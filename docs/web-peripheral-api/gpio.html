GPIO — Digital I/O ระดับ Register

ระดับ Peripheral (SPL ดิบ)
ไฟล์: #include "ch32v00x_gpio.h" — นี่คือไลบรารี Standard Peripheral Library (SPL) ระดับต่ำ เขียนตั้งค่าผ่าน struct GPIO_InitTypeDef และเรียกฟังก์ชันที่แก้ register ตรง ๆ (CFGLR, INDR, OUTDR, BSHR, BCR) ต้องเปิด clock เองด้วย RCC_APB2PeriphClockCmd ก่อนใช้งานทุกครั้ง
ถ้าต้องการ API ที่ใช้ง่ายกว่านี้ (เช่น pinMode(), digitalWrite()) ดูที่ SimpleGPIO — Digital I/O (SimpleHAL wrapper) ซึ่งสร้างต่อยอดจากไลบรารีนี้

ผังการทำงาน GPIO (SPL)

flowchart LR
    RCC["RCC_APB2PeriphClockCmd(GPIOx)"] --> INIT["GPIO_Init()"]
    INIT --> MODE{"GPIO_Mode"}
    MODE --> OUT["Out_PP / Out_OD"]
    MODE --> IN["IN_FLOATING / IPD / IPU"]
    MODE --> AIN["AIN (ADC)"]
    MODE --> AF["AF_PP / AF_OD"]
    OUT --> WR["GPIO_SetBits / GPIO_ResetBits / GPIO_WriteBit"]
    IN --> RD["GPIO_ReadInputDataBit"]
    AF --> REMAP["GPIO_PinRemapConfig()"]
    IN --> EXTICFG["GPIO_EXTILineConfig() → EXTI_Init()"]

GPIO_InitTypeDef — โครงสร้างตั้งค่า

สมาชิกชนิดคำอธิบาย
GPIO_Pinuint16_tขาที่ต้องการตั้งค่า ใช้ GPIO_Pin_0 .. GPIO_Pin_7 หรือ OR รวมกันได้หลายขา หรือ GPIO_Pin_All
GPIO_SpeedGPIOSpeed_TypeDefความเร็วเอาต์พุต ใช้เฉพาะเมื่อเป็นโหมด output/AF เท่านั้น
GPIO_ModeGPIOMode_TypeDefโหมดการทำงานของขา (input/output/AF/analog)

GPIOMode_TypeDef (ค่า GPIO_Mode)

ค่าคงที่ค่า hexความหมาย
GPIO_Mode_AIN0x00Analog Input — ใช้กับ ADC
GPIO_Mode_IN_FLOATING0x04Input Floating — ไม่มี pull ภายใน ต้องมี pull ภายนอกถ้าจำเป็น
GPIO_Mode_IPD0x28Input Pull-Down ภายใน
GPIO_Mode_IPU0x48Input Pull-Up ภายใน
GPIO_Mode_Out_OD0x14Output Open-Drain
GPIO_Mode_Out_PP0x10Output Push-Pull
GPIO_Mode_AF_OD0x1CAlternate Function Open-Drain (เช่น I2C)
GPIO_Mode_AF_PP0x18Alternate Function Push-Pull (เช่น USART TX, PWM)

GPIOSpeed_TypeDef (ค่า GPIO_Speed)

ค่าคงที่ความเร็ว
GPIO_Speed_10MHz10 MHz
GPIO_Speed_2MHz2 MHz
GPIO_Speed_30MHz30 MHz (มี alias GPIO_Speed_50MHz ชี้มาที่ค่าเดียวกัน)

GPIO_pins_define

Macroค่า
GPIO_Pin_0 .. GPIO_Pin_7bitmask ขาที่ 0-7 (0x0001 .. 0x0080)
GPIO_Pin_All0xFFFF — ทุกขาในพอร์ต
CH32V003 ใช้จริงแค่ขา 0-7 ต่อพอร์ต (register กว้าง 32 บิตแต่ตัว MCU มีแค่ 8 pin ต่อพอร์ต) ค่า GPIO_Pin_8 ขึ้นไปไม่มีนิยามในไฟล์นี้

BitAction (สำหรับ GPIO_WriteBit)

ค่าคงที่ความหมาย
Bit_RESET= 0, เขียนขาเป็น LOW
Bit_SET= 1, เขียนขาเป็น HIGH

GPIO_Remap_define (สำหรับ GPIO_PinRemapConfig)

Macroใช้ทำอะไร
GPIO_Remap_SPI1ย้ายขา SPI1 ไปตำแหน่งสำรอง
GPIO_PartialRemap_I2C1 / GPIO_FullRemap_I2C1ย้ายขา I2C1 บางส่วน/ทั้งหมด
GPIO_PartialRemap1_USART1 / GPIO_PartialRemap2_USART1 / GPIO_FullRemap_USART1ย้ายขา USART1 แบบต่าง ๆ
GPIO_PartialRemap1_TIM1 / GPIO_PartialRemap2_TIM1 / GPIO_FullRemap_TIM1ย้ายขา TIM1 (PWM)
GPIO_PartialRemap1_TIM2 / GPIO_PartialRemap2_TIM2 / GPIO_FullRemap_TIM2ย้ายขา TIM2 (PWM)
GPIO_Remap_PA1PA2 (alias GPIO_Remap_PA1_2)เปิดใช้งาน PA1/PA2 เป็น GPIO ทั่วไป
GPIO_Remap_ADC1_ETRGINJ / GPIO_Remap_ADC1_ETRGREGย้าย external trigger ของ ADC1
GPIO_Remap_LSI_CALเปิดฟังก์ชันสอบเทียบ LSI ออกทางขา
GPIO_Remap_SDI_Disableปิดใช้งาน SDI (Single-wire Debug Interface)

GPIO_Port_Sources / GPIO_Pin_sources (สำหรับ GPIO_EXTILineConfig)

Macroความหมาย
GPIO_PortSourceGPIOAเลือกพอร์ต A เป็นแหล่งของ EXTI line
GPIO_PortSourceGPIOCเลือกพอร์ต C เป็นแหล่งของ EXTI line
GPIO_PortSourceGPIODเลือกพอร์ต D เป็นแหล่งของ EXTI line
GPIO_PinSource0 .. GPIO_PinSource7เลขขา (0-7) ที่จะผูกกับ EXTI line ตามเลขเดียวกัน

ตัวอย่างการใช้งาน

1. Output Push-Pull: กระพริบ LED ด้วย SPL ดิบ

#include "ch32v00x_gpio.h"
#include "ch32v00x_rcc.h"

void LED_Init(void)
{
    RCC_APB2PeriphClockCmd(RCC_APB2Periph_GPIOC, ENABLE);   // 1. เปิด clock พอร์ตก่อนเสมอ

    GPIO_InitTypeDef GPIO_InitStructure = {0};
    GPIO_InitStructure.GPIO_Pin = GPIO_Pin_0;
    GPIO_InitStructure.GPIO_Speed = GPIO_Speed_10MHz;
    GPIO_InitStructure.GPIO_Mode = GPIO_Mode_Out_PP;         // push-pull output
    GPIO_Init(GPIOC, &GPIO_InitStructure);
}

int main(void)
{
    LED_Init();
    while (1) {
        GPIO_SetBits(GPIOC, GPIO_Pin_0);     // HIGH
        Delay_Ms(500);
        GPIO_ResetBits(GPIOC, GPIO_Pin_0);   // LOW
        Delay_Ms(500);
    }
}

2. Input Pull-Up: อ่านปุ่มกด

RCC_APB2PeriphClockCmd(RCC_APB2Periph_GPIOC, ENABLE);

GPIO_InitTypeDef GPIO_InitStructure = {0};
GPIO_InitStructure.GPIO_Pin = GPIO_Pin_1;
GPIO_InitStructure.GPIO_Mode = GPIO_Mode_IPU;   // input pull-up ภายใน
GPIO_Init(GPIOC, &GPIO_InitStructure);

if (GPIO_ReadInputDataBit(GPIOC, GPIO_Pin_1) == Bit_RESET) {
    // ปุ่มกด (ต่อ PC1 กับ GND) = อ่านได้ LOW
}

3. Alternate Function + Remap: ย้าย USART1 ไปตำแหน่งสำรอง

RCC_APB2PeriphClockCmd(RCC_APB2Periph_GPIOD | RCC_APB2Periph_AFIO, ENABLE);

GPIO_PinRemapConfig(GPIO_PartialRemap1_USART1, ENABLE);  // ย้ายขา USART1 TX/RX

GPIO_InitTypeDef GPIO_InitStructure = {0};
GPIO_InitStructure.GPIO_Pin = GPIO_Pin_5;         // ขา TX (ตำแหน่งใหม่หลัง remap)
GPIO_InitStructure.GPIO_Speed = GPIO_Speed_10MHz;
GPIO_InitStructure.GPIO_Mode = GPIO_Mode_AF_PP;   // alternate function push-pull
GPIO_Init(GPIOD, &GPIO_InitStructure);

4. EXTI Interrupt บนขา GPIO (ดูรายละเอียดเต็มที่หน้า EXTI)

RCC_APB2PeriphClockCmd(RCC_APB2Periph_GPIOC | RCC_APB2Periph_AFIO, ENABLE);

GPIO_InitTypeDef GPIO_InitStructure = {0};
GPIO_InitStructure.GPIO_Pin = GPIO_Pin_1;
GPIO_InitStructure.GPIO_Mode = GPIO_Mode_IPU;     // ต้องเป็น input ก่อนตั้ง interrupt
GPIO_Init(GPIOC, &GPIO_InitStructure);

GPIO_EXTILineConfig(GPIO_PortSourceGPIOC, GPIO_PinSource1);  // ผูก PC1 -> EXTI Line1
// ต่อด้วย EXTI_Init() + NVIC_Init() — ดูตัวอย่างเต็มในหน้า exti.html

ปัญหาที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
เรียก GPIO_Init() แล้วขาไม่ตอบสนองเลย ลืมเปิด clock ให้พอร์ตด้วย RCC_APB2PeriphClockCmd(RCC_APB2Periph_GPIOx, ENABLE) ก่อน — เขียน register ของพอร์ตที่ยังไม่มี clock จะไม่มีผลใด ๆ เรียก RCC_APB2PeriphClockCmd เปิด clock พอร์ตนั้นเสมอ ก่อน GPIO_Init()
GPIO_Mode_AF_PP/AF_OD ตั้งแล้วแต่อุปกรณ์ต่อพ่วง (USART/SPI/I2C/PWM) ไม่ทำงาน โหมด AF บอกแค่ว่าขาให้วงจร alternate function ควบคุม — ต้องเปิด peripheral clock และ config peripheral นั้นเอง (เช่น USART_Init) ด้วย ไม่ใช่แค่ตั้ง GPIO ตรวจว่าเปิด clock ของ peripheral ที่เกี่ยวข้อง และเรียก _Init() ของ peripheral นั้นครบแล้ว
ใช้ GPIO_PinRemapConfig() แล้ว remap ไม่ทำงาน ลืมเปิด clock ของ AFIO (RCC_APB2Periph_AFIO) — การเขียน AFIO->PCFR1 ต้องมี clock ของ AFIO ก่อน เพิ่ม RCC_APB2PeriphClockCmd(RCC_APB2Periph_AFIO, ENABLE) คู่กับ clock ของพอร์ต GPIO
GPIO_ReadOutputDataBit() คืนค่าไม่ตรงกับสถานะจริงบนขา ฟังก์ชันนี้อ่านจากรีจิสเตอร์ OUTDR (ค่าที่ MCU สั่งล่าสุด) ไม่ใช่ค่าที่วัดจากขาจริง ถ้าขาโหลดเกิน/ช็อตกับภายนอกค่าจริงบนขาอาจต่างไป ถ้าต้องการอ่านสถานะจริงของขา (รวมถึงขา output ที่ถูกภายนอกบังคับ) ให้ใช้ GPIO_ReadInputDataBit() แทน — register INDR อ่านสถานะไฟฟ้าจริงบนขาเสมอ ไม่ว่าโหมดจะเป็น input หรือ output
ใช้ GPIO_EXTILineConfig() กับหลายพอร์ตแล้ว interrupt ตัวหนึ่งหายไปเงียบ ๆ EXTI Line ผูกกับเลขขา ไม่ใช่พอร์ต — GPIO_EXTILineConfig(GPIO_PortSourceGPIOC, GPIO_PinSource1) กับ GPIO_EXTILineConfig(GPIO_PortSourceGPIOD, GPIO_PinSource1) ใช้ EXTI Line 1 เดียวกัน — เรียกครั้งหลังจะทับครั้งแรก เลือกใช้ pin source ที่เลขต่างกันสำหรับแต่ละ interrupt เช่น PC1 (Line1) กับ PD2 (Line2)
GPIO_Write() ทำให้ขาอื่นในพอร์ตเปลี่ยนสถานะโดยไม่ตั้งใจ GPIO_Write() เขียนทับ OUTDR ทั้ง 16 บิตในครั้งเดียว ไม่ใช่แก้เฉพาะขาที่ต้องการ ใช้ GPIO_SetBits()/GPIO_ResetBits()/GPIO_WriteBit() แทนเมื่อต้องการแก้แค่บางขา เพราะเป็น atomic bit operation ผ่าน BSHR/BCR
