EXTI — External Interrupt/Event Controller

ระดับ Peripheral (SPL ดิบ)
ไฟล์: #include "ch32v00x_exti.h" — เป็นไลบรารี SPL ระดับต่ำที่คุมวงจร EXTI (External Interrupt/Event) ตั้งค่าผ่าน struct EXTI_InitTypeDef แล้วแก้ register INTENR/EVENR/RTENR/FTENR/INTFR ตรง ๆ EXTI ใช้คู่กับ GPIO_EXTILineConfig() (จาก ch32v00x_gpio.h) เพื่อเลือกว่าขาไหนจะเป็นแหล่งของ interrupt line
ถ้าต้องการ API ที่ใช้ง่ายกว่านี้ ดูที่ SimpleGPIO — attachInterrupt() (SimpleHAL wrapper) ซึ่งซ่อนรายละเอียด EXTI/NVIC ไว้ให้หมดแล้ว

ผังการทำงาน EXTI

flowchart LR
    A["GPIO_Init() โหมด input"] --> B["GPIO_EXTILineConfig(port, pin)"]
    B --> C["EXTI_Init() ตั้ง line/mode/trigger"]
    C --> D["NVIC_Init() เปิด EXTI7_0_IRQn"]
    D --> E["เกิดขอบสัญญาณจริง"]
    E --> F["EXTI7_0_IRQHandler()"]
    F --> G["EXTI_GetITStatus() ตรวจ line"]
    G --> H["ทำงานตาม logic"]
    H --> I["EXTI_ClearITPendingBit() ล้าง flag"]

EXTI_InitTypeDef — โครงสร้างตั้งค่า

สมาชิกชนิดคำอธิบาย
EXTI_Lineuint32_tเลือก line ที่จะตั้งค่า ใช้ EXTI_Line0 .. EXTI_Line9 หรือ OR รวมกันได้หลาย line
EXTI_ModeEXTIMode_TypeDefโหมด Interrupt หรือ Event
EXTI_TriggerEXTITrigger_TypeDefขอบสัญญาณที่จะกระตุ้น (ขึ้น/ลง/ทั้งสองขอบ)
EXTI_LineCmdFunctionalStateENABLE เพื่อเปิดใช้งาน line นี้ หรือ DISABLE เพื่อปิด

EXTI_Lines

Macroค่า bitmaskหมายเหตุ
EXTI_Line0 .. EXTI_Line70x0001 .. 0x0080ผูกกับขา GPIO หมายเลข 0-7 ผ่าน GPIO_EXTILineConfig()
EXTI_Line80x0100เชื่อมต่อภายในกับสัญญาณ PVD Output (ไม่ใช่ขา GPIO)
EXTI_Line90x0200เชื่อมต่อภายในกับเหตุการณ์ PWR Auto Wake-up (ไม่ใช่ขา GPIO)

EXTIMode_TypeDef

ค่าคงที่ความหมาย
EXTI_Mode_Interruptสร้าง interrupt request เข้า NVIC เมื่อเกิดขอบสัญญาณ
EXTI_Mode_Eventสร้าง event ภายใน (เช่น ปลุก MCU จาก low-power) โดยไม่เข้า interrupt handler

EXTITrigger_TypeDef

ค่าคงที่ความหมาย
EXTI_Trigger_Risingกระตุ้นที่ขอบขาขึ้น (LOW→HIGH)
EXTI_Trigger_Fallingกระตุ้นที่ขอบขาลง (HIGH→LOW)
EXTI_Trigger_Rising_Fallingกระตุ้นทั้งสองขอบ

ตารางอ้างอิงฟังก์ชัน

ฟังก์ชันพารามิเตอร์คืนค่าคำอธิบายหมายเหตุ
void EXTI_DeInit(void) ไม่มี void ล้างค่า INTENR, EVENR, RTENR, FTENR เป็น 0 และล้าง INTFR ทั้งหมด (ปิดทุก line กลับสถานะเริ่มต้น) เรียกตอนต้นโปรแกรมเพื่อล้างค่า EXTI ก่อนตั้งค่าใหม่
void EXTI_Init(EXTI_InitTypeDef *EXTI_InitStruct) EXTI_InitStruct: pointer struct ตั้งค่า void ตั้งค่า mode/trigger/enable ของ line ตาม struct — ฟังก์ชันหลักที่ใช้เปิด EXTI ต้องเรียก GPIO_EXTILineConfig() ก่อนเสมอ (สำหรับ line 0-7) ไม่งั้น line จะยังผูกกับพอร์ต default (GPIOA)
void EXTI_StructInit(EXTI_InitTypeDef *EXTI_InitStruct) EXTI_InitStruct: pointer struct void ใส่ค่า default ให้ struct: EXTI_Line=ไม่มี line, EXTI_Mode_Interrupt, EXTI_Trigger_Falling, DISABLE เรียกก่อนแก้บาง field เพื่อกัน garbage value ใน struct
void EXTI_GenerateSWInterrupt(uint32_t EXTI_Line) EXTI_Line: EXTI_Line_x ที่จะบังคับกระตุ้น void สั่งกระตุ้น interrupt/event ของ line นั้นด้วยซอฟต์แวร์ (ผ่าน SWIEVR) โดยไม่ต้องรอสัญญาณจริง มีประโยชน์ตอน debug/ทดสอบ ISR โดยไม่ต้องมีฮาร์ดแวร์จริงมากระตุ้น
FlagStatus EXTI_GetFlagStatus(uint32_t EXTI_Line) EXTI_Line: line ที่ต้องการเช็ค FlagStatus (SET/RESET) อ่านค่า flag ค้าง (pending) ของ line จาก INTFR โดยตรง ต่างจาก EXTI_GetITStatus ตรงที่ไม่เช็คว่า line เปิดใช้งาน (INTENR) อยู่หรือไม่
void EXTI_ClearFlag(uint32_t EXTI_Line) EXTI_Line: line ที่ต้องการล้าง void ล้าง flag ค้างของ line นั้นใน INTFR เขียนการทำงานเหมือนกับ EXTI_ClearITPendingBit ทุกประการ
ITStatus EXTI_GetITStatus(uint32_t EXTI_Line) EXTI_Line: line ที่ต้องการเช็ค ITStatus (SET/RESET) เช็คว่า line นี้ทั้งค้าง (INTFR) และเปิดใช้งานอยู่ (INTENR) พร้อมกันหรือไม่ ใช้ใน ISR เพื่อกรองว่า interrupt ที่เข้ามาเป็นของ line ไหน — แม่นกว่า EXTI_GetFlagStatus
void EXTI_ClearITPendingBit(uint32_t EXTI_Line) EXTI_Line: line ที่ต้องการล้าง void ล้างบิตค้างของ line ใน INTFR (เขียน 1 เพื่อล้าง) ต้องเรียกใน ISR ทุกครั้ง ก่อนออกจาก handler ไม่งั้น interrupt จะเข้าซ้ำทันที

ตัวอย่างการใช้งาน

1. ตั้ง Interrupt ขอบขาลงบนขา PC1 (ปุ่มกด)

#include "ch32v00x_gpio.h"
#include "ch32v00x_exti.h"
#include "ch32v00x_misc.h"

void Button_EXTI_Init(void)
{
    // 1. เปิด clock พอร์ต GPIO และ AFIO
    RCC_APB2PeriphClockCmd(RCC_APB2Periph_GPIOC | RCC_APB2Periph_AFIO, ENABLE);

    // 2. ตั้งขาเป็น input pull-up
    GPIO_InitTypeDef GPIO_InitStructure = {0};
    GPIO_InitStructure.GPIO_Pin = GPIO_Pin_1;
    GPIO_InitStructure.GPIO_Mode = GPIO_Mode_IPU;
    GPIO_Init(GPIOC, &GPIO_InitStructure);

    // 3. ผูกขา PC1 เข้ากับ EXTI Line1
    GPIO_EXTILineConfig(GPIO_PortSourceGPIOC, GPIO_PinSource1);

    // 4. ตั้งค่า EXTI line
    EXTI_InitTypeDef EXTI_InitStructure = {0};
    EXTI_InitStructure.EXTI_Line = EXTI_Line1;
    EXTI_InitStructure.EXTI_Mode = EXTI_Mode_Interrupt;
    EXTI_InitStructure.EXTI_Trigger = EXTI_Trigger_Falling;  // ปุ่มกด = ขอบลง
    EXTI_InitStructure.EXTI_LineCmd = ENABLE;
    EXTI_Init(&EXTI_InitStructure);

    // 5. เปิด NVIC ให้ interrupt line 0-7 (CH32V003 รวม EXTI0-7 ไว้ที่ IRQ เดียว)
    NVIC_InitTypeDef NVIC_InitStructure = {0};
    NVIC_InitStructure.NVIC_IRQChannel = EXTI7_0_IRQn;
    NVIC_InitStructure.NVIC_IRQChannelPreemptionPriority = 1;
    NVIC_InitStructure.NVIC_IRQChannelSubPriority = 2;
    NVIC_InitStructure.NVIC_IRQChannelCmd = ENABLE;
    NVIC_Init(&NVIC_InitStructure);
}

2. IRQ Handler อ่านและล้าง flag ให้ถูก line

volatile uint16_t buttonPressCount = 0;

void EXTI7_0_IRQHandler(void) __attribute__((interrupt("WCH-Interrupt-fast")));
void EXTI7_0_IRQHandler(void)
{
    if (EXTI_GetITStatus(EXTI_Line1) != RESET) {
        buttonPressCount++;              // งานให้เร็วที่สุด ห้ามหน่วงเวลาใน ISR
        EXTI_ClearITPendingBit(EXTI_Line1);  // ต้องล้างเสมอ ไม่งั้นเข้า ISR ซ้ำไม่จบ
    }
}

3. หลาย EXTI Line พร้อมกัน (ต้องเลขขาไม่ซ้ำกัน)

// PC1 -> Line1, PD2 -> Line2 (คนละเลขขา ใช้พร้อมกันได้)
GPIO_EXTILineConfig(GPIO_PortSourceGPIOC, GPIO_PinSource1);
GPIO_EXTILineConfig(GPIO_PortSourceGPIOD, GPIO_PinSource2);

EXTI_InitTypeDef EXTI_InitStructure = {0};
EXTI_InitStructure.EXTI_Line = EXTI_Line1 | EXTI_Line2;   // ตั้งสอง line พร้อมกันได้ใน struct เดียว
EXTI_InitStructure.EXTI_Mode = EXTI_Mode_Interrupt;
EXTI_InitStructure.EXTI_Trigger = EXTI_Trigger_Rising_Falling;
EXTI_InitStructure.EXTI_LineCmd = ENABLE;
EXTI_Init(&EXTI_InitStructure);

// ใน handler ต้องแยกตรวจทีละ line
void EXTI7_0_IRQHandler(void)
{
    if (EXTI_GetITStatus(EXTI_Line1) != RESET) {
        EXTI_ClearITPendingBit(EXTI_Line1);
        // ... งานของ PC1
    }
    if (EXTI_GetITStatus(EXTI_Line2) != RESET) {
        EXTI_ClearITPendingBit(EXTI_Line2);
        // ... งานของ PD2
    }
}

4. ปิดการใช้งาน EXTI Line เมื่อเลิกใช้

EXTI_InitTypeDef EXTI_InitStructure = {0};
EXTI_InitStructure.EXTI_Line = EXTI_Line1;
EXTI_InitStructure.EXTI_LineCmd = DISABLE;   // field อื่นไม่มีผลตอน DISABLE
EXTI_Init(&EXTI_InitStructure);

ปัญหาที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
ตั้ง EXTI_Init() แล้ว interrupt ไม่เกิดขึ้นเลย ลืมเรียก GPIO_EXTILineConfig() ก่อน — ถ้าไม่เรียก line จะยังผูกกับพอร์ต/ขา default (มักเป็น GPIOA) ไม่ใช่ขาที่ตั้งใจใช้จริง และลืมเปิด clock RCC_APB2Periph_AFIO ด้วย (เขียน AFIO->EXTICR ต้องมี clock ของ AFIO) เรียกตามลำดับเสมอ: เปิด clock GPIO+AFIO → GPIO_Init() โหมด input → GPIO_EXTILineConfig() → EXTI_Init() → NVIC_Init()
ใช้ interrupt สองขาที่เลขท้ายเหมือนกัน (เช่น PC1 กับ PD1) แล้วตัวหนึ่งหายไปเงียบ ๆ EXTI Line ผูกกับเลขขา ไม่ใช่ port — GPIO_PinSource1 มีได้แค่ตัวเดียวต่อครั้ง เรียก GPIO_EXTILineConfig() ครั้งหลังจะทับการตั้งค่าของครั้งแรกใน AFIO->EXTICR โดยไม่มี error ใด ๆ เลือกขาที่เลขท้ายไม่ซ้ำกันสำหรับแต่ละ interrupt เช่น PC1 (Line1) คู่กับ PD2 (Line2) แทน PC1 คู่กับ PD1
ลืมเรียก EXTI_ClearITPendingBit() ใน ISR ทำให้ MCU เหมือนค้าง/เข้า ISR รัวถี่ยิบ ถ้าไม่ล้างบิตค้างใน INTFR ก่อนออกจาก handler ตัว interrupt controller จะเห็นว่ายังมี flag ค้างอยู่ แล้วเรียก ISR ซ้ำทันทีที่ออกจาก handler เดิม กลายเป็นวนไม่จบ เรียก EXTI_ClearITPendingBit(EXTI_Line_x) เป็นบรรทัดสุดท้ายของการจัดการ line นั้นใน ISR เสมอ
ตั้งหลาย line ใน struct เดียว (EXTI_Line1 | EXTI_Line2) แล้วแยกแยะไม่ได้ว่า interrupt มาจากขาไหน CH32V003 รวม EXTI Line 0-7 เข้า IRQ เดียวคือ EXTI7_0_IRQn / EXTI7_0_IRQHandler — handler ตัวเดียวต้องรับผิดชอบทุก line ที่เปิดไว้ในช่วง 0-7 ใน handler ต้องเช็คทีละ line ด้วย EXTI_GetITStatus(EXTI_Line_x) แล้วล้างเฉพาะ line ที่ตรวจพบ ก่อนจะทำ logic ของ line นั้น
ขาที่ตั้ง EXTI ไม่เคยเข้า ISR เลยแม้สัญญาณเปลี่ยนจริง ลืมตั้งขาเป็นโหมด input (GPIO_Mode_IN_FLOATING/IPD/IPU) ก่อน — ถ้าขายังเป็นโหมด output หรือ analog input (GPIO_Mode_AIN) วงจรตรวจจับขอบสัญญาณของ EXTI จะไม่เห็นการเปลี่ยนแปลงเดิน ต้อง GPIO_Init() ตั้งขาเป็น input ก่อนเสมอ แล้วค่อยเรียก GPIO_EXTILineConfig() และ EXTI_Init()
