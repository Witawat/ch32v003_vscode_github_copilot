OPA — Operational Amplifier

ระดับสูง
Header: #include "ch32v00x_opa.h" — นี่คือไลบรารีระดับล่าง (Standard Peripheral Library) เขียนตรงเข้าเรจิสเตอร์ EXTEN->EXTEN_CTR โดยตรง ถ้าต้องการ API ที่มี "โหมด" สำเร็จรูป (Voltage Follower / Non-Inverting / Inverting / Comparator พร้อมสูตรคำนวณ gain) ดูที่ SimpleOPAMP (web-simplehal-api/opamp.html) แทน

ภาพรวม

CH32V003 มี Operational Amplifier ภายใน 1 ตัว ควบคุมผ่านเรจิสเตอร์รวม EXTEN_CTR (Extended Configuration Register) เพียง 3 บิต:

  • EN (bit 16) — เปิด/ปิดวงจร OPA
  • NSEL (bit 17) — เลือกขาอินพุตลบ (CHN0/CHN1)
  • PSEL (bit 18) — เลือกขาอินพุตบวก (CHP0/CHP1)
ไม่มี "โหมด" ในไลบรารีนี้: ไฟล์ ch32v00x_opa.h/.c ทำหน้าที่แค่ เลือกขา input (PSEL/NSEL) และเปิด/ปิดวงจร เท่านั้น — ไม่มีบิตหรือฟังก์ชันสำหรับเลือก "Voltage Follower" / "Non-Inverting" / "Inverting" / "Comparator" ในฮาร์ดแวร์เลย พฤติกรรมของ op-amp (buffer, ขยายสัญญาณ, เปรียบเทียบ) เกิดจากการต่อวงจรภายนอก (feedback resistor ระหว่างขา output กับขา input ลบ) ทั้งหมด ไม่ใช่ค่าที่ตั้งผ่านซอฟต์แวร์ — "โหมด" ที่เห็นใน SimpleOPAMP wrapper เป็นแค่ชื่อเรียกตามวิธีต่อวงจรที่แนะนำ ไม่มีอยู่จริงในเรจิสเตอร์
ไม่ต้องเปิด peripheral clock: EXTEN_CTR เป็นเรจิสเตอร์ configuration พิเศษที่ถูก clock ตลอดเวลา (ไม่ผูกกับ RCC APB/AHB gate เหมือน peripheral อื่น) เรียก OPA_Init()/OPA_Cmd() ได้ทันทีโดยไม่ต้องเรียก RCC_xxxPeriphClockCmd() ก่อน

Typedef และค่าคงที่

ชื่อชนิดค่า/สมาชิกคำอธิบาย
OPA_PSEL_TypeDef enum CHP0 = 0, CHP1 = 1 เลือกขาอินพุตบวก (non-inverting input)
OPA_NSEL_TypeDef enum CHN0 = 0, CHN1 = 1 เลือกขาอินพุตลบ (inverting input)
OPA_InitTypeDef struct PSEL (OPA_PSEL_TypeDef), NSEL (OPA_NSEL_TypeDef) โครงสร้างพารามิเตอร์สำหรับ OPA_Init()
ชื่อ enum ไม่มี prefix: ค่า CHP0, CHP1, CHN0, CHN1 ประกาศแบบ global ไม่มี prefix OPA_ นำหน้า (ต่างจาก wrapper ที่ใช้ OPAMP_CHP0 เป็นต้น) — ระวังชนกับชื่ออื่นถ้ามี macro/enum ชื่อซ้ำในโปรเจกต์

ตารางอ้างอิงฟังก์ชัน

ฟังก์ชันพารามิเตอร์คืนค่าคำอธิบาย
OPA_DeInit(void) — void รีเซ็ต bit 16-18 (EN, NSEL, PSEL) ของ EXTEN_CTR กลับเป็น 0 ทั้งหมด (ปิด OPA และล้างการเลือก channel)
OPA_Init(OPA_InitTypeDef *OPA_InitStruct) OPA_InitStruct: pointer ไปยังโครงสร้างที่ตั้งค่า PSEL/NSEL void เลือกขาอินพุตบวก/ลบ (เขียนบิต 17-18 ของ EXTEN_CTR แบบ read-modify-write) — ไม่แตะบิต EN
OPA_StructInit(OPA_InitTypeDef *OPA_InitStruct) OPA_InitStruct: pointer โครงสร้างที่จะถูกเติมค่า void เติมค่า default ลงโครงสร้าง: PSEL = CHP0, NSEL = CHN0
OPA_Cmd(FunctionalState NewState) NewState: ENABLE/DISABLE void เปิด/ปิดวงจร OPA (ตั้ง/เคลียร์บิต 16 ของ EXTEN_CTR) เท่านั้น — ไม่ยุ่งกับ channel
Doxygen comment ของ OPA_Cmd คลาดเคลื่อน: comment ในซอร์ส (ch32v00x_opa.c) ระบุพารามิเตอร์ OPA_NUM - Select OPA แต่ลายเซ็นฟังก์ชันจริงมีแค่ NewState พารามิเตอร์เดียว — CH32V003 มี OPA เพียงตัวเดียว ไม่มีให้เลือกเบอร์ อย่าคาดหวัง overload หรือพารามิเตอร์เพิ่มตามที่ comment บอก

ลำดับการเรียกใช้ฟังก์ชัน

flowchart TD

    B --> C["s.PSEL = CHP0; s.NSEL = CHN0;

ตัวอย่างโค้ด

1. Init + Enable พื้นฐาน (channel 0 ทั้งคู่)

#include "ch32v00x.h"

OPA_InitTypeDef OPA_InitStructure;

int main(void)
{
    SystemCoreClockUpdate();

    OPA_StructInit(&OPA_InitStructure);   // PSEL = CHP0, NSEL = CHN0 (default)
    OPA_Init(&OPA_InitStructure);         // เขียนค่าลง EXTEN_CTR
    OPA_Cmd(ENABLE);                      // ★ ต้องเรียกแยก — เปิดวงจรจริง

    while (1)
    {
        // output ของ OPA ออกที่ขา OPO บนบอร์ด — วัดผลผ่าน ADC ภายนอกฟังก์ชันนี้เอง
    }
}

2. เลือก channel เอง (PSEL=CHP1, NSEL=CHN1)

#include "ch32v00x.h"

int main(void)
{
    SystemCoreClockUpdate();

    OPA_InitTypeDef OPA_InitStructure;
    OPA_InitStructure.PSEL = CHP1;
    OPA_InitStructure.NSEL = CHN1;

    OPA_Init(&OPA_InitStructure);
    OPA_Cmd(ENABLE);

    while (1)
    {
        // ...
    }
}

3. Voltage Follower (buffer) — ต่อ output กลับเข้า input ลบเอง

#include "ch32v00x.h"

// ต่อวงจรจริงบนบอร์ด: ขา OPO (output) ต่อสายตรงกลับเข้าขา OPN0 (CHN0)
// -> เกิดผล voltage follower (gain = 1) ล้วนๆ จากการต่อวงจร ไม่ใช่จาก register ใดๆ

int main(void)
{
    SystemCoreClockUpdate();

    OPA_InitTypeDef OPA_InitStructure;
    OPA_InitStructure.PSEL = CHP0;   // สัญญาณ input เข้าที่นี่
    OPA_InitStructure.NSEL = CHN0;   // ต้องต่อสายจาก OPO กลับมาที่ขานี้จริงบนบอร์ด

    OPA_Init(&OPA_InitStructure);
    OPA_Cmd(ENABLE);

    while (1)
    {
        // ...
    }
}

4. ปิดใช้งาน / รีเซ็ตกลับค่าเริ่มต้น

OPA_Cmd(DISABLE);   // ปิดวงจร แต่ channel ที่เลือกไว้ยังอยู่
// หรือ
OPA_DeInit();        // ล้างทั้ง EN + PSEL + NSEL กลับเป็น 0 (ต้อง Init+Cmd ใหม่ทั้งหมดถ้าจะใช้ต่อ)

ปัญหาที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
เรียก OPA_Init() แล้ว output ยังไม่ออก OPA_Init() ตั้งค่าแค่ PSEL/NSEL เท่านั้น — ไม่แตะบิต EN เลย ต้องเรียก OPA_Cmd(ENABLE) แยกต่างหากเสมอหลัง OPA_Init()
อยากได้ Non-Inverting/Inverting gain แต่ตั้งค่าอย่างไรก็ไม่มีฟังก์ชันให้เลือก mode ไลบรารีนี้ไม่มีแนวคิด "mode" ในซอฟต์แวร์ — เป็นแค่ raw op-amp ที่เลือกได้เฉพาะ input pin ต้องต่อตัวต้านทาน feedback ภายนอกจริงบนบอร์ดระหว่างขา output (OPO) กับขา input ลบ (OPN0/OPN1) เพื่อกำหนดพฤติกรรม gain เอง — ซอฟต์แวร์ทำได้แค่เลือกว่า input บวก/ลบมาจากขาไหน
เขียนโค้ดปิด/เปิด OPA ซ้ำๆ แล้วบิตอื่นในเรจิสเตอร์เดียวกันเปลี่ยนไปด้วยโดยไม่ตั้งใจ EXTEN_CTR เป็นเรจิสเตอร์รวมที่ใช้ร่วมกับฟีเจอร์อื่น (เช่น EXTEN_LOCKUP_EN bit 6, EXTEN_LDO_TRIM bit 10) — ฟังก์ชัน OPA ในไลบรารีนี้ทำ read-modify-write เฉพาะบิตของตัวเองอยู่แล้ว (ปลอดภัย) แต่ถ้าไปเขียน EXTEN->EXTEN_CTR ตรงๆ เองที่อื่นในโปรแกรม อาจไปทับบิต OPA โดยไม่ตั้งใจ อย่าเขียน EXTEN->EXTEN_CTR = value; แบบเขียนทับทั้งเรจิสเตอร์เอง ให้ใช้ฟังก์ชัน OPA_* หรือทำ read-modify-write เองด้วย mask (7<<16) เท่านั้น
ต้องการอ่านผลเปรียบเทียบ (comparator) แบบ digital HIGH/LOW แต่หาไม่เจอฟังก์ชันอ่านสถานะ ไลบรารี OPA นี้ไม่มีเรจิสเตอร์/ฟังก์ชันอ่านผลลัพธ์ใดๆ เลย (ไม่มี status flag เหมือน comparator ของ STM32 บางรุ่น) ต้องอ่านค่าที่ขา output (OPO) ผ่านโมดูล ADC หรือ GPIO digital input เอง — ไลบรารีนี้ทำหน้าที่แค่ config input/enable เท่านั้น
เรียก OPA_DeInit() แล้วเรียก OPA_Cmd(ENABLE) ต่อทันที แต่ output ผิดเพี้ยน OPA_DeInit() ล้างทั้ง PSEL/NSEL กลับเป็น 0 (= CHP0/CHN0) พร้อมกับ EN — ถ้าเรียก OPA_Cmd(ENABLE) ทันทีโดยไม่เรียก OPA_Init() ใหม่ จะได้ channel 0/0 เสมอ ไม่ใช่ channel เดิมที่เคยตั้งไว้ หลัง OPA_DeInit() ต้องเรียก OPA_Init() เพื่อตั้ง channel ใหม่ก่อน แล้วค่อย OPA_Cmd(ENABLE)
