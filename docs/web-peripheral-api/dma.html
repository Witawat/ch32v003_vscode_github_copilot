DMA — Direct Memory Access

ระดับขั้นสูง (Raw SPL)
Header: #include "ch32v00x_dma.h" — SPL ระดับล่างสำหรับตั้งค่าช่อง DMA โดยตรงผ่าน struct DMA_InitTypeDef CH32V003 มี DMA controller ตัวเดียว (DMA1) พร้อม 7 ช่อง (DMA1_Channel1 … DMA1_Channel7) แต่ละช่องผูกตายตัวกับอุปกรณ์ต่อพ่วงเฉพาะ (fixed hardware mapping) ไม่สามารถสลับช่องให้อุปกรณ์อื่นเองแบบ DMAMUX ได้
อยากได้อะไรที่ง่ายกว่านี้? ถ้าต้องการแค่ถ่ายโอนข้อมูลแบบไม่ต้องยุ่งกับ struct/เรจิสเตอร์เอง ดูที่ SimpleDMA แทน — หน้านี้เหมาะสำหรับกรณีที่ต้องคุมรายละเอียดเต็มรูปแบบ เช่น DMA คู่กับ ADC scan ต่อเนื่อง, memory-to-memory, หรือจัดการ interrupt เอง

Interrupt / Flag (ตัวอย่างช่อง 1 — DMA1_IT_xx1 / DMA1_FLAG_xx1 มีครบทุกช่อง 1-7)

ค่าคงที่ (ใช้กับ DMA_ITConfig)ค่าคงที่ Flag เฉพาะช่อง (ใช้กับ DMA_GetFlagStatus/DMA_ClearFlag)คำอธิบาย
DMA_IT_TCDMA1_FLAG_TC1 … DMA1_FLAG_TC7Transfer Complete — ถ่ายโอนครบตามที่ตั้งไว้
DMA_IT_HTDMA1_FLAG_HT1 … DMA1_FLAG_HT7Half Transfer — ถ่ายโอนได้ครึ่งหนึ่ง (มีประโยชน์กับ circular double-buffer)
DMA_IT_TEDMA1_FLAG_TE1 … DMA1_FLAG_TE7Transfer Error — เกิดข้อผิดพลาดระหว่างถ่ายโอน (เช่น bus error)
—DMA1_FLAG_GL1 … DMA1_FLAG_GL7Global flag — รวมสถานะ TC/HT/TE ของช่องนั้นไว้บิตเดียว

สำหรับ DMA_GetITStatus/DMA_ClearITPendingBit ใช้ชุดค่าคงที่รูปแบบเดียวกันแต่ขึ้นต้นด้วย DMA1_IT_ แทน DMA1_FLAG_ (เช่น DMA1_IT_TC1) ความหมายเดียวกัน

การจับคู่ช่อง DMA กับอุปกรณ์ (Fixed Hardware Mapping)

สำคัญ: บน CH32V003 การจับคู่ระหว่างช่อง DMA กับอุปกรณ์ต่อพ่วงถูกกำหนดตายตัวโดยฮาร์ดแวร์ ไม่ใช่เลือกได้อิสระเหมือน DMAMUX ใน MCU บางตระกูล — ADC1 ผูกกับ DMA1_Channel1 เท่านั้น ถ้าใช้ช่องอื่นกับ ADC จะไม่มีข้อผิดพลาดตอนคอมไพล์แต่ DMA จะไม่ทำงานเลย (เงียบๆ ไม่มีข้อมูลไหล) ให้ตรวจสอบตาราง DMA request mapping ในคู่มืออ้างอิง CH32V003 Reference Manual ก่อนกำหนดช่องให้อุปกรณ์อื่น เช่น SPI1 หรือ USART1

ตัวอย่างโค้ด

1. Memory-to-Memory Copy (คัดลอกหน่วยความจำโดยไม่ใช้ CPU)

#include "ch32v00x_dma.h"
#include "ch32v00x_rcc.h"

uint8_t src_buf[64] = { /* ... ข้อมูลต้นทาง ... */ };
uint8_t dst_buf[64];

void DMA_M2M_Copy(void)
{
    DMA_InitTypeDef DMA_InitStructure = {0};

    RCC_AHBPeriphClockCmd(RCC_AHBPeriph_DMA1, ENABLE);

    DMA_DeInit(DMA1_Channel3);
    DMA_InitStructure.DMA_PeripheralBaseAddr = (uint32_t)src_buf;
    DMA_InitStructure.DMA_MemoryBaseAddr     = (uint32_t)dst_buf;
    DMA_InitStructure.DMA_DIR                = DMA_DIR_PeripheralSRC;
    DMA_InitStructure.DMA_BufferSize         = sizeof(src_buf);
    DMA_InitStructure.DMA_PeripheralInc      = DMA_PeripheralInc_Enable;  // M2M ต้อง Enable ทั้งสองฝั่ง
    DMA_InitStructure.DMA_MemoryInc          = DMA_MemoryInc_Enable;
    DMA_InitStructure.DMA_PeripheralDataSize = DMA_PeripheralDataSize_Byte;
    DMA_InitStructure.DMA_MemoryDataSize     = DMA_MemoryDataSize_Byte;
    DMA_InitStructure.DMA_Mode               = DMA_Mode_Normal;          // ห้ามใช้ Circular กับ M2M
    DMA_InitStructure.DMA_Priority           = DMA_Priority_Medium;
    DMA_InitStructure.DMA_M2M                = DMA_M2M_Enable;
    DMA_Init(DMA1_Channel3, &DMA_InitStructure);

    DMA_Cmd(DMA1_Channel3, ENABLE);   // M2M เริ่มทำงานทันที ไม่ต้องรอคำขอจากอุปกรณ์

    while (DMA_GetFlagStatus(DMA1_FLAG_TC3) == RESET);
    DMA_ClearFlag(DMA1_FLAG_TC3);
}

2. Peripheral-to-Memory พร้อม Transfer-Complete Interrupt

#include "ch32v00x_dma.h"
#include "ch32v00x_rcc.h"
#include "ch32v00x_misc.h"

#define BUF_LEN 32
volatile uint8_t rx_buf[BUF_LEN];
volatile uint8_t rx_done = 0;

void DMA_USART_RxInit(void)
{
    DMA_InitTypeDef DMA_InitStructure = {0};
    NVIC_InitTypeDef NVIC_InitStructure = {0};

    RCC_AHBPeriphClockCmd(RCC_AHBPeriph_DMA1, ENABLE);

    // ตัวอย่าง: USART1_RX ผูกกับ DMA1_Channel5 บน CH32V003 (ตรวจสอบ mapping จริงจาก Reference Manual)
    DMA_DeInit(DMA1_Channel5);
    DMA_InitStructure.DMA_PeripheralBaseAddr = (uint32_t)&USART1->DATAR;
    DMA_InitStructure.DMA_MemoryBaseAddr     = (uint32_t)rx_buf;
    DMA_InitStructure.DMA_DIR                = DMA_DIR_PeripheralSRC;
    DMA_InitStructure.DMA_BufferSize         = BUF_LEN;
    DMA_InitStructure.DMA_PeripheralInc      = DMA_PeripheralInc_Disable;
    DMA_InitStructure.DMA_MemoryInc          = DMA_MemoryInc_Enable;
    DMA_InitStructure.DMA_PeripheralDataSize = DMA_PeripheralDataSize_Byte;
    DMA_InitStructure.DMA_MemoryDataSize     = DMA_MemoryDataSize_Byte;
    DMA_InitStructure.DMA_Mode               = DMA_Mode_Normal;
    DMA_InitStructure.DMA_Priority           = DMA_Priority_High;
    DMA_InitStructure.DMA_M2M                = DMA_M2M_Disable;
    DMA_Init(DMA1_Channel5, &DMA_InitStructure);

    DMA_ITConfig(DMA1_Channel5, DMA_IT_TC, ENABLE);

    NVIC_InitStructure.NVIC_IRQChannel = DMA1_Channel5_IRQn;
    NVIC_InitStructure.NVIC_IRQChannelPreemptionPriority = 1;
    NVIC_InitStructure.NVIC_IRQChannelSubPriority = 0;
    NVIC_InitStructure.NVIC_IRQChannelCmd = ENABLE;
    NVIC_Init(&NVIC_InitStructure);

    DMA_Cmd(DMA1_Channel5, ENABLE);
    // ... เปิด USART_DMACmd(USART1, USART_DMAReq_Rx, ENABLE) ด้วย ...
}

void DMA1_Channel5_IRQHandler(void) __attribute__((interrupt("WCH-Interrupt-fast")));
void DMA1_Channel5_IRQHandler(void)
{
    if (DMA_GetITStatus(DMA1_IT_TC5) != RESET) {
        rx_done = 1;
        DMA_ClearITPendingBit(DMA1_IT_TC5);   // ต้องล้างเสมอ ไม่งั้น ISR ยิงซ้ำไม่หยุด
    }
}

3. รีสตาร์ทรอบถ่ายโอนใน Normal Mode โดยไม่เรียก DMA_Init ใหม่

void DMA_Restart(DMA_Channel_TypeDef *ch, uint16_t len)
{
    DMA_Cmd(ch, DISABLE);              // ต้องปิดก่อนแก้ counter
    DMA_ClearFlag(DMA1_FLAG_TC1);      // ล้าง flag รอบก่อน (ปรับเลขช่องตามที่ใช้จริง)
    DMA_SetCurrDataCounter(ch, len);   // ตั้งจำนวนใหม่
    DMA_Cmd(ch, ENABLE);               // เริ่มรอบใหม่
}

ใน DMA_Mode_Normal เมื่อถ่ายโอนครบแล้ว DMA จะหยุดเอง ต้อง disable/set counter/enable ใหม่ทุกครั้งถ้าต้องการถ่ายโอนรอบถัดไป (ต่างจาก DMA_Mode_Circular ที่วนอัตโนมัติ)

ปัญหาที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
ตั้งค่า DMA ครบแล้วแต่ไม่มีข้อมูลไหลเลย ใช้ช่อง DMA ที่ไม่ตรงกับ mapping ตายตัวของอุปกรณ์นั้น (เช่น ผูก ADC1 กับ DMA1_Channel3 ทั้งที่ ADC1 ใช้ได้แค่ Channel1) — ไม่มี error ตอนคอมไพล์หรือรัน แค่เงียบๆ ไม่ทำงาน ตรวจตาราง DMA request mapping ใน CH32V003 Reference Manual ให้ตรงกับอุปกรณ์ที่ใช้ก่อนตั้งค่า
DMA ทำงานรอบแรกแล้วหยุด ไม่ยอมถ่ายรอบถัดไป ใช้ DMA_Mode_Normal ซึ่งหยุดอัตโนมัติเมื่อ DMA_BufferSize ถ่ายครบ ใช้ DMA_Mode_Circular ถ้าต้องการวนซ้ำอัตโนมัติ หรือเรียก DMA_SetCurrDataCounter + DMA_Cmd ใหม่ทุกรอบถ้าจงใจใช้ Normal
Interrupt Transfer-Complete ยิงครั้งเดียวแล้วไม่ยิงอีกเลย (ทั้งที่ DMA ยังทำงานต่อ) ลืมเรียก DMA_ClearITPendingBit/DMA_ClearFlag ใน ISR ทำให้ flag ค้าง บาง MCU/สถาปัตยกรรมจะไม่ trigger ซ้ำจนกว่าจะล้างก่อน ล้าง flag ที่ต้นๆ หรือท้ายๆ ของ ISR ทุกครั้งก่อนออกจากฟังก์ชัน
ข้อมูลใน buffer เพี้ยน/ไม่ตรงตำแหน่งที่คาดไว้ ตั้ง DMA_PeripheralDataSize กับ DMA_MemoryDataSize ไม่ตรงกัน (เช่น อุปกรณ์เป็น HalfWord แต่ buffer เป็น uint8_t) ทำให้ address คำนวณผิด ให้ DataSize ทั้งสองฝั่งตรงกับชนิดข้อมูลจริง เช่น ADC 10-bit ควรใช้ HalfWord ทั้งคู่ และ buffer เป็น uint16_t
Compile ผ่านแต่ใช้ DMA_M2M_Enable ร่วมกับ DMA_Mode_Circular แล้วพฤติกรรมแปลกๆ Errata/ข้อจำกัดฮาร์ดแวร์: circular buffer mode ใช้ไม่ได้กับ memory-to-memory ตามที่ระบุใน comment ของ header เอง ใช้ DMA_Mode_Normal เสมอเมื่อ DMA_M2M_Enable
เขียนทับ buffer เดิมทุกครั้ง ไม่เลื่อนตำแหน่ง ลืมตั้ง DMA_MemoryInc = DMA_MemoryInc_Enable — ค่า default จาก DMA_StructInit คือ Disable ตั้ง DMA_MemoryInc_Enable เมื่อถ่ายลง array/buffer ที่ต้องการเก็บหลายค่า
