ตัวอย่างการใช้งาน

1. เข้าถึง register โดยตรง (Bare-metal)

// อ่านค่า GPIOC input register โดยตรง
uint8_t pin_state = GPIOC->INDR & GPIO_Pin_1;

// เขียนค่า GPIOA output register โดยตรง
GPIOA->OUTDR |= GPIO_Pin_0;   // set PA0 high
GPIOA->OUTDR &= ~GPIO_Pin_0;  // set PA0 low

// ใช้ bit set/reset register (เขียน atomic)
GPIOA->BSHR = GPIO_Pin_0;     // set PA0 high
GPIOA->BCR = GPIO_Pin_0;      // set PA0 low

2. ตรวจสอบ clock ของ peripheral

// ตรวจสอบว่า USART1 clock เปิดอยู่หรือไม่
if (RCC->APB2PCENR & RCC_APB2Periph_USART1) {
    // USART1 clock เปิดอยู่
}

3. Debug register (ดูค่าใน register ขณะทำงาน)

// ใน debug mode สามารถดูค่า register ได้โดยตรง
volatile uint32_t cr = RCC->CTLR;      // RCC Control Register
volatile uint32_t cfg = RCC->CFGR0;    // RCC Configuration Register
volatile uint32_t en = RCC->APB2PCENR; // APB2 Peripheral Clock Enable

ปัญหาที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
อ่าน register ได้ค่า 0 ทั้งที่ตั้งค่าไว้แล้ว ยังไม่ได้เปิด clock ของ peripheral นั้น — register จะไม่ตอบสนองถ้า clock ยังปิดอยู่ เรียก RCC_APB2PeriphClockCmd() หรือ RCC_APB1PeriphClockCmd() ตาม bus ที่ peripheral สังกัด
เขียน register แล้วค่าไม่เปลี่ยน register นั้นเป็น read-only หรือต้องเขียน sequence เฉพาะ (เช่น Flash key unlock) ตรวจสอบ datasheet ว่า register นั้นเขียนได้หรือไม่ และต้องปลดล็อกก่อนหรือไม่
