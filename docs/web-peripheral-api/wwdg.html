WWDG — Window Watchdog (วอทช์ด็อกแบบหน้าต่าง)

ระดับสูง
Header: #include <ch32v00x_wwdg.h> — นี่คือ Standard Peripheral Library (SPL) ระดับต่ำ เขียนเรจิสเตอร์ตรง ไม่มีการตรวจสอบพารามิเตอร์หรือ abstraction ใดๆ ผู้ใช้ต้องคำนวณค่า window/counter เองและเข้าใจพฤติกรรมฮาร์ดแวร์
ทางเลือกที่ง่ายกว่า: ถ้าต้องการ API ที่ตั้งค่าให้อัตโนมัติ ดูที่ SimpleWWDG (SimpleHAL API Reference) ซึ่งห่อ (wrap) ฟังก์ชันชุดนี้ไว้อีกชั้น

ภาพรวม

WWDG (Window Watchdog) เป็นวอทช์ด็อกที่ใช้นาฬิกา PCLK1 (นาฬิการะบบหลัก ไม่ใช่ LSI แบบ IWDG) มีจุดต่างสำคัญจาก IWDG คือ ต้อง reload counter ภายในช่วงเวลาที่กำหนด (window) เท่านั้น — ถ้า reload เร็วเกินไป (ก่อนเข้าหน้าต่าง) ก็รีเซ็ตทันที ไม่ใช่แค่ reload ช้าเกินไปเหมือน IWDG

  • Counter (T[6:0]) — นับถอยหลัง 7 บิต ช่วง 0x40-0x7F เมื่อ T6 หลุดจาก 1 เป็น 0 (ค่าลดผ่าน 0x3F) จะเกิดรีเซ็ตทันที
  • Window (W[6:0]) — ค่าเปรียบเทียบ ต้องน้อยกว่า 0x80 ห้าม refresh counter ตราบใดที่ T ยังมากกว่า W (ช่วง "ปิด")
  • Prescaler (WDGTB) — หาร (PCLK1/4096) ด้วย 1/2/4/8
  • Early Wakeup Interrupt (EWI) — แจ้งเตือนก่อนรีเซ็ตจริง ให้โอกาส reload ใน ISR
ต่างจาก IWDG: IWDG แค่ "ห้ามช้าเกินไป" แต่ WWDG "ห้ามทั้งเร็วเกินไปและช้าเกินไป" — ต้อง reload ในหน้าต่างเวลาที่กำหนดพอดี

ผังการทำงาน WWDG

flowchart LR

ค่าคงที่ (Constants)

WWDG_Prescaler — ตัวหารนาฬิกา (WDGTB)

ค่าคงที่ค่าCounter clock
WWDG_Prescaler_10x00000000(PCLK1/4096)/1
WWDG_Prescaler_20x00000080(PCLK1/4096)/2
WWDG_Prescaler_40x00000100(PCLK1/4096)/4
WWDG_Prescaler_80x00000180(PCLK1/4096)/8

ช่วงค่า Counter / Window

พารามิเตอร์ช่วงค่าที่ถูกต้องคำอธิบาย
Counter (WWDG_SetCounter/WWDG_Enable)0x40 - 0x7Fบิต T6 ต้องเป็น 1 เสมอตอนตั้งค่า มิฉะนั้นรีเซ็ตทันที (T6=0 คือเงื่อนไขรีเซ็ต)
WindowValue (WWDG_SetWindowValue)0x00 - 0x7Fต้องน้อยกว่า 0x80 (มาสก์ 7 บิต); เปรียบเทียบกับ downcounter T[6:0]

สูตรคำนวณเวลา

// คาบนับ 1 tick (ms) = 4096 * prescaler_divide / PCLK1_Hz * 1000
// เวลาจนรีเซ็ต (จาก Counter=C ถึง 0x3F) = tick_ms * (C - 0x3F)
// เวลาจนหน้าต่างเปิด (จาก Counter=C ถึง Window=W) = tick_ms * (C - W)
//
// ตัวอย่าง: PCLK1 = 48MHz, prescaler = 8
// tick = 4096 * 8 / 48,000,000 = 0.6827 ms ต่อ 1 นับ
// Counter = 0x7F (127), Window = 0x50 (80)
//   หน้าต่างปิดจนถึง 0x50: (127-80)*0.6827 ms ≈ 32.1 ms  (reload ก่อนหน้านี้ = รีเซ็ตทันที)
//   ตั้งแต่ 0x50 ถึงขอบรีเซ็ต 0x3F: (80-63)*0.6827 ms ≈ 11.6 ms (ต้อง reload ในช่วงนี้)

ตารางอ้างอิงฟังก์ชัน

ฟังก์ชันพารามิเตอร์คืนค่าคำอธิบายหมายเหตุ
WWDG_DeInit() - void รีเซ็ตเรจิสเตอร์ WWDG ทั้งหมดกลับค่า default ผ่าน RCC APB1 peripheral reset ใช้ตอนเริ่มต้นก่อนตั้งค่าใหม่ทั้งหมด
WWDG_SetPrescaler(WWDG_Prescaler) WWDG_Prescaler: WWDG_Prescaler_1/2/4/8 void ตั้งบิต WDGTB ใน CFGR (แก้เฉพาะ field prescaler ค่าอื่นใน CFGR คงเดิม) เรียกก่อน WWDG_Enable
WWDG_SetWindowValue(WindowValue) WindowValue: uint8_t < 0x80 void ตั้งบิต W[6:0] ใน CFGR ต้องน้อยกว่า Counter เริ่มต้น เพื่อให้มีช่วง "หน้าต่างปิด" จริง
WWDG_EnableIT() - void ตั้งบิต 9 (EWI) ใน CFGR เปิด Early Wakeup Interrupt ต้องเปิด NVIC สำหรับ WWDG_IRQn เองแยกต่างหาก; ISR ต้องเคลียร์ flag ด้วย WWDG_ClearFlag()
WWDG_SetCounter(Counter) Counter: uint8_t 0x40-0x7F void เขียนค่า Counter ลง CTLR โดยไม่แตะบิต WDGA (เปิด/ปิด) ใช้สำหรับ "ป้อนอาหาร"/reload ระหว่างทำงาน ต้องเรียกเฉพาะตอนอยู่ในหน้าต่างเปิดเท่านั้น
WWDG_Enable(Counter) Counter: uint8_t 0x40-0x7F void ตั้งบิต WDGA (เปิดใช้งาน) พร้อมโหลดค่า Counter เริ่มต้นในการเขียนครั้งเดียว เมื่อเปิดแล้วปิดด้วยซอฟต์แวร์ไม่ได้ — ต้อง reset MCU เท่านั้น
WWDG_GetFlagStatus() - FlagStatus (SET/RESET) อ่านค่า STATR (บิต EWIF) — แจ้งว่าเกิด Early Wakeup Interrupt แล้วหรือยัง เรียกใน WWDG_IRQHandler เพื่อยืนยันสาเหตุ interrupt
WWDG_ClearFlag() - void เขียน STATR = 0 ล้าง flag EWIF ต้องเรียกใน ISR มิฉะนั้น interrupt จะค้าง (pending ซ้ำ)

ตัวอย่างโค้ด

1. Init WWDG พื้นฐาน + reload ใน main loop (ไม่ใช้ interrupt)

#include <ch32v00x.h>
#include <ch32v00x_wwdg.h>

void WWDG_Init_Basic(void)
{
    RCC_APB1PeriphClockCmd(RCC_APB1Periph_WWDG, ENABLE);

    WWDG_DeInit();
    WWDG_SetPrescaler(WWDG_Prescaler_8);
    WWDG_SetWindowValue(0x50);   // หน้าต่างเปิดเมื่อ Counter ลดต่ำกว่า 0x50
    WWDG_Enable(0x7F);           // เริ่ม Counter ที่ค่าสูงสุด แล้วเปิดใช้งานทันที
}

int main(void)
{
    SystemCoreClockUpdate();
    WWDG_Init_Basic();

    while (1) {
        // ... งานหลัก วนรอบให้คาบเวลาตรงกับหน้าต่างที่คำนวณไว้ ...
        Delay_Ms(15);            // ต้องอยู่ในช่วงหน้าต่างเปิดพอดี (ดูสูตรคำนวณด้านบน)
        WWDG_SetCounter(0x7F);   // reload กลับค่าสูงสุด
    }
}

2. ใช้ Early Wakeup Interrupt (EWI) เพื่อ reload อัตโนมัติก่อนหน้าต่างปิด

#include <ch32v00x.h>
#include <ch32v00x_wwdg.h>

void WWDG_Init_WithEWI(void)
{
    RCC_APB1PeriphClockCmd(RCC_APB1Periph_WWDG, ENABLE);

    WWDG_DeInit();
    WWDG_SetPrescaler(WWDG_Prescaler_8);
    WWDG_SetWindowValue(0x50);

    NVIC_InitTypeDef nvic = {0};
    nvic.NVIC_IRQChannel = WWDG_IRQn;
    nvic.NVIC_IRQChannelPreemptionPriority = 2;
    nvic.NVIC_IRQChannelSubPriority = 0;
    nvic.NVIC_IRQChannelCmd = ENABLE;
    NVIC_Init(&nvic);

    WWDG_EnableIT();      // เปิด EWI - จะเกิด interrupt ตอน Counter ลดถึง 0x40
    WWDG_Enable(0x7F);
}

// ต้อง reload ทันทีใน ISR ก่อนที่ Counter จะลดผ่าน 0x3F (จุดรีเซ็ต)
void WWDG_IRQHandler(void) __attribute__((interrupt("WCH-Interrupt-fast")));
void WWDG_IRQHandler(void)
{
    if (WWDG_GetFlagStatus() != RESET) {
        WWDG_SetCounter(0x7F);   // reload ก่อนขอบเขต — ต้องอยู่ในหน้าต่างเปิดอยู่แล้วตอนนี้
        WWDG_ClearFlag();
    }
}

3. คำนวณ Window/Prescaler จากเวลาที่ต้องการ (PCLK1 = 48MHz)

// ต้องการคาบ reload window กว้างประมาณ 10-20ms
// tick (prescaler=8) = 4096*8/48,000,000 = 0.6827 ms
// เลือก Window = 0x60 (96), Counter เริ่ม = 0x7F (127)
//   หน้าต่างปิดนาน (127-96)*0.6827 ≈ 21.2 ms
//   หน้าต่างเปิดนาน (96-63)*0.6827 ≈ 22.5 ms  <- ต้อง reload ในช่วงนี้
WWDG_SetPrescaler(WWDG_Prescaler_8);
WWDG_SetWindowValue(0x60);
WWDG_Enable(0x7F);

ปัญหาที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
Reload แล้ว MCU รีเซ็ตทันที (ตรงข้ามกับที่ตั้งใจ) เรียก WWDG_SetCounter() ก่อน Counter ลดต่ำกว่า Window (ยังอยู่ในหน้าต่าง "ปิด") — WWDG ถือว่าเป็นการ refresh ที่ผิดเวลาและรีเซ็ตทันที ไม่ใช่แค่ไม่มีผล reload เฉพาะเมื่อ Counter (T) ต่ำกว่า Window (W) แล้วเท่านั้น คำนวณช่วงเวลาให้แม่นด้วยสูตร tick ก่อน หรือใช้ EWI เพื่อให้ฮาร์ดแวร์บอกจังหวะที่ถูกต้อง
เปิด WWDG แล้วปิดไม่ได้ ดีบักลำบาก WWDG_Enable() ตั้งบิต WDGA แล้วไม่มีฟังก์ชัน Disable — ต้องรีเซ็ตชิปเท่านั้นจึงจะปิดได้ อย่าเปิด WWDG ตอนช่วง debug จนกว่าจะพร้อม reload ต่อเนื่องในหน้าต่างที่ถูกต้อง
ตั้ง Counter หรือ Window ผิดช่วงแล้วพฤติกรรมประหลาด Counter ต้องอยู่ 0x40-0x7F (บิต T6=1) และ WindowValue ต้อง < 0x80 — ค่านอกช่วงนี้ทำให้ตรรกะ "หน้าต่าง" ผิดเพี้ยนหรือรีเซ็ตทันทีตอนเปิดใช้งาน ตรวจสอบค่าที่ส่งให้อยู่ในช่วงตามสเปกเสมอ โดยเฉพาะ Counter เริ่มต้นต้องมีบิต 6 เป็น 1
ISR ของ EWI เกิดซ้ำถี่/ค้าง ลืมเรียก WWDG_ClearFlag() ใน WWDG_IRQHandler ทำให้ flag EWIF ค้างเป็น pending เคลียร์ flag ทุกครั้งใน ISR หลังจัดการ reload เสร็จ
คำนวณเวลาหน้าต่างผิดเมื่อเปลี่ยนความถี่ระบบ สูตรเวลาอิงกับ PCLK1 ของระบบ ไม่ใช่ค่าคงที่แบบ LSI ของ IWDG — ถ้าปรับ system clock (เช่นเปลี่ยนจาก 48MHz เป็นความถี่อื่น) ค่า tick ทั้งหมดเปลี่ยนตาม คำนวณ Window/Counter ใหม่ทุกครั้งที่เปลี่ยนความถี่ system clock อย่าใช้ค่าคงที่ตายตัวข้ามโปรเจกต์ที่ clock ต่างกัน
