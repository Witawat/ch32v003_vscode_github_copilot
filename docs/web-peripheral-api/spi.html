SPI — Serial Peripheral Interface

ระดับสูง
ไลบรารี: #include <ch32v00x_spi.h> — นี่คือ Standard Peripheral Library (SPL) ระดับล่าง เขียนคอนฟิกผ่าน struct SPI_InitTypeDef โดยตรงกับเรจิสเตอร์ฮาร์ดแวร์ และตรวจสอบสถานะการรับ-ส่งด้วยการ poll flag เอง (เช่น SPI_I2S_FLAG_TXE/_RXNE) ไม่มี buffer อัตโนมัติแบบ wrapper
อยากได้อะไรที่ง่ายกว่านี้? ถ้าต้องการ SPI แบบเรียกใช้งานง่าย ดูที่ SimpleSPI (SimpleHAL wrapper) แทน หน้านี้เหมาะสำหรับคนที่ต้องการควบคุมรายละเอียดระดับ register/flag เอง

ผังการทำงานของ SPI Master (Polling Model)

flowchart LR
    INIT["SPI_Init()"] --> EN["SPI_Cmd(ENABLE)"]
    EN --> NSS["ลด NSS/CS ลง (software GPIO)"]
    NSS --> TXE["รอ SPI_I2S_FLAG_TXE"]
    TXE --> SEND["SPI_I2S_SendData()"]
    SEND --> RXNE["รอ SPI_I2S_FLAG_RXNE"]
    RXNE --> RECV["SPI_I2S_ReceiveData()"]
    RECV --> NSSUP["ยก NSS/CS ขึ้น"]

SPI_InitTypeDef — โครงสร้างการตั้งค่า

สมาชิกชนิดค่าที่เป็นไปได้คำอธิบาย
SPI_Directionuint16_tSPI_Direction_2Lines_FullDuplex, _2Lines_RxOnly, _1Line_Rx, _1Line_Txโหมดทิศทางข้อมูล (2 สายเต็มดูเพล็กซ์ หรือสายเดียว)
SPI_Modeuint16_tSPI_Mode_Master, SPI_Mode_Slaveโหมด Master หรือ Slave
SPI_DataSizeuint16_tSPI_DataSize_8b, SPI_DataSize_16bขนาดข้อมูลต่อครั้ง 8 หรือ 16 บิต
SPI_CPOLuint16_tSPI_CPOL_Low, SPI_CPOL_Highระดับ clock ตอน idle (Mode 0/2 = Low, Mode 1/3 = High)
SPI_CPHAuint16_tSPI_CPHA_1Edge, SPI_CPHA_2Edgeขอบ clock ที่ใช้จับข้อมูล (ขอบแรก/ขอบที่สอง)
SPI_NSSuint16_tSPI_NSS_Hard, SPI_NSS_SoftNSS จัดการโดยฮาร์ดแวร์ (ขา NSS) หรือซอฟต์แวร์ (บิต SSI)
SPI_BaudRatePrescaleruint16_tSPI_BaudRatePrescaler_2 ถึง _256ตัวหารความถี่ clock ระบบ → SCK (2,4,8,...,256)
SPI_FirstBituint16_tSPI_FirstBit_MSB, SPI_FirstBit_LSBลำดับบิตที่ส่งก่อน (LSB ไม่รองรับโหมด Slave)
SPI_CRCPolynomialuint16_tค่าพหุนาม เช่น 7พหุนามสำหรับคำนวณ CRC (ถ้าใช้ฟีเจอร์ CRC)

SPI_StructInit() ตั้งค่าเริ่มต้น: Direction=2Lines_FullDuplex, Mode=Slave, DataSize=8b, CPOL=Low, CPHA=1Edge, BaudRatePrescaler=2, FirstBit=MSB, CRCPolynomial=7 — ต้องเปลี่ยน Mode/NSS เองถ้าจะใช้เป็น Master

Flag และ Interrupt Macros

กลุ่มค่าคงที่ความหมาย
Flag (ใช้กับ SPI_I2S_GetFlagStatus)SPI_I2S_FLAG_RXNEReceive buffer ไม่ว่าง (มีข้อมูลรอให้อ่าน)
SPI_I2S_FLAG_TXETransmit buffer ว่าง (ส่งข้อมูลใหม่ได้)
SPI_I2S_FLAG_BSYบัส SPI กำลังส่ง/รับข้อมูลอยู่
SPI_FLAG_MODFMode Fault — เกิดข้อขัดแย้งของ Master/Slave (NSS ผิดพลาด)
SPI_FLAG_CRCERRค่า CRC ที่รับไม่ตรงกับที่คำนวณ
SPI_I2S_FLAG_OVROverrun — ข้อมูลใหม่มาทับที่ยังไม่ได้อ่าน
Interrupt (ใช้กับ SPI_I2S_ITConfig/GetITStatus)SPI_I2S_IT_TXEinterrupt เมื่อ transmit buffer ว่าง
SPI_I2S_IT_RXNEinterrupt เมื่อรับข้อมูลมาใหม่
SPI_I2S_IT_ERRเปิดกลุ่ม error interrupt (mode fault/overrun/CRC)
SPI_I2S_IT_OVR, SPI_IT_MODF, SPI_IT_CRCERRสถานะ error แต่ละประเภท (ใช้ตรวจใน ISR)
อื่นๆSPI_I2S_DMAReq_Tx, SPI_I2S_DMAReq_Rxคำขอ DMA (ใช้กับ SPI_I2S_DMACmd)

ตัวอย่างโค้ด

1. เริ่มต้นใช้งาน SPI1 (Master, Full-Duplex, NSS Software)

#include <ch32v00x_spi.h>
#include <ch32v00x_gpio.h>
#include <ch32v00x_rcc.h>

#define CS_PIN   GPIO_Pin_4   /* PC4 = Chip Select (software) */

void SPI1_Init_Master(void)
{
    GPIO_InitTypeDef GPIO_InitStructure = {0};
    SPI_InitTypeDef SPI_InitStructure = {0};

    RCC_APB2PeriphClockCmd(RCC_APB2Periph_GPIOC | RCC_APB2Periph_SPI1, ENABLE);

    /* PC5=SCK, PC6=MOSI -> AF Push-Pull */
    GPIO_InitStructure.GPIO_Pin = GPIO_Pin_5 | GPIO_Pin_6;
    GPIO_InitStructure.GPIO_Mode = GPIO_Mode_AF_PP;
    GPIO_InitStructure.GPIO_Speed = GPIO_Speed_10MHz;
    GPIO_Init(GPIOC, &GPIO_InitStructure);

    /* PC7=MISO -> Input floating */
    GPIO_InitStructure.GPIO_Pin = GPIO_Pin_7;
    GPIO_InitStructure.GPIO_Mode = GPIO_Mode_IN_FLOATING;
    GPIO_Init(GPIOC, &GPIO_InitStructure);

    /* PC4=CS -> Output push-pull ควบคุมด้วยซอฟต์แวร์ */
    GPIO_InitStructure.GPIO_Pin = CS_PIN;
    GPIO_InitStructure.GPIO_Mode = GPIO_Mode_Out_PP;
    GPIO_Init(GPIOC, &GPIO_InitStructure);
    GPIO_SetBits(GPIOC, CS_PIN);   /* idle = HIGH (ไม่เลือกอุปกรณ์) */

    SPI_InitStructure.SPI_Direction = SPI_Direction_2Lines_FullDuplex;
    SPI_InitStructure.SPI_Mode = SPI_Mode_Master;
    SPI_InitStructure.SPI_DataSize = SPI_DataSize_8b;
    SPI_InitStructure.SPI_CPOL = SPI_CPOL_Low;
    SPI_InitStructure.SPI_CPHA = SPI_CPHA_1Edge;
    SPI_InitStructure.SPI_NSS = SPI_NSS_Soft;
    SPI_InitStructure.SPI_BaudRatePrescaler = SPI_BaudRatePrescaler_16;
    SPI_InitStructure.SPI_FirstBit = SPI_FirstBit_MSB;
    SPI_InitStructure.SPI_CRCPolynomial = 7;
    SPI_Init(SPI1, &SPI_InitStructure);

    /* จำเป็น! ไม่ตั้งค่านี้ SPI จะค้างในโหมด Slave แม้ตั้ง SPI_Mode_Master ไว้แล้ว */
    SPI_NSSInternalSoftwareConfig(SPI1, SPI_NSSInternalSoft_Set);

    SPI_Cmd(SPI1, ENABLE);
}

2. ส่ง-รับ 1 byte แบบ Full-Duplex (poll TXE/RXNE)

uint8_t SPI1_TransferByte(uint8_t data)
{
    while (SPI_I2S_GetFlagStatus(SPI1, SPI_I2S_FLAG_TXE) == RESET);
    SPI_I2S_SendData(SPI1, data);

    while (SPI_I2S_GetFlagStatus(SPI1, SPI_I2S_FLAG_RXNE) == RESET);
    return (uint8_t)SPI_I2S_ReceiveData(SPI1);
}

3. อ่าน/เขียน register ของอุปกรณ์ SPI ภายนอก ด้วย CS ควบคุมเอง

uint8_t SPI1_ReadReg(uint8_t regAddr)
{
    uint8_t value;

    GPIO_ResetBits(GPIOC, CS_PIN);           /* CS ลง = เลือกอุปกรณ์ */
    SPI1_TransferByte(regAddr | 0x80);        /* บิต 0x80 = read command (แล้วแต่ IC) */
    value = SPI1_TransferByte(0xFF);          /* ส่ง dummy byte เพื่อ clock ข้อมูลออกมา */
    GPIO_SetBits(GPIOC, CS_PIN);             /* CS ขึ้น = ปล่อยอุปกรณ์ */

    return value;
}
NSS Software ต้องคู่กับ SSI: เมื่อตั้ง SPI_NSS = SPI_NSS_Soft ต้องเรียก SPI_NSSInternalSoftwareConfig(SPI1, SPI_NSSInternalSoft_Set) เพิ่มเสมอ ไม่งั้นฮาร์ดแวร์จะมองว่า NSS เป็น LOW ตลอดเวลา ทำให้ peripheral ตกกลับไปอยู่ในสถานะ Slave โดยอัตโนมัติแม้ config เป็น Master

ปัญหาที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
ส่งข้อมูลไม่ออก / SPI_I2S_SendData() ไม่มีผล ลืมเรียก SPI_Cmd(SPI1, ENABLE) หลัง SPI_Init() ต้องเรียก SPI_Cmd(ENABLE) เสมอก่อนใช้งานจริง — SPI_Init() ตั้งค่าเรจิสเตอร์แต่ไม่เปิด peripheral
ตั้ง SPI_Mode_Master แล้วแต่พฤติกรรมเหมือน Slave / ไม่ generate clock ใช้ SPI_NSS_Soft แต่ไม่ได้เรียก SPI_NSSInternalSoftwareConfig(SPI_NSSInternalSoft_Set) เรียก SPI_NSSInternalSoftwareConfig() เป็น Set ทันทีหลัง SPI_Init() เมื่อใช้ NSS แบบซอฟต์แวร์
อ่านค่าจาก SPI_I2S_ReceiveData() ได้ข้อมูลเก่า/ค้าง อ่านโดยไม่เช็ค SPI_I2S_FLAG_RXNE ก่อน ต้อง poll SPI_I2S_GetFlagStatus(SPI_I2S_FLAG_RXNE) จนเป็น SET ก่อนอ่านทุกครั้ง
ข้อมูลสลับบิต/อ่านผิด (bit-reversed) หรือสื่อสารไม่ตรงกับอุปกรณ์อื่น ตั้ง CPOL/CPHA ไม่ตรงกับ SPI Mode ที่อุปกรณ์ปลายทางต้องการ (Mode 0-3) ตรวจ datasheet อุปกรณ์ว่าใช้ Mode ใด แล้วจับคู่ CPOL/CPHA ให้ตรง (Mode0=CPOL_Low+CPHA_1Edge, Mode3=CPOL_High+CPHA_2Edge)
MOSI/SCK ทำงานแต่ไม่มีข้อมูลรับกลับมา ตั้ง MISO pin เป็นโหมดอื่นที่ไม่ใช่ Input (เช่น Output ทับกัน) หรือลืมตั้ง GPIO เป็น AF MOSI/SCK = GPIO_Mode_AF_PP, MISO (ฝั่ง Master) = GPIO_Mode_IN_FLOATING
อุปกรณ์ SPI หลายตัวบนบัสเดียวกันรับคำสั่งพร้อมกันผิดตัว ใช้ NSS ฮาร์ดแวร์ร่วมกันหลายอุปกรณ์ หรือไม่ได้ยก CS ของตัวที่ไม่ได้ใช้งานขึ้น ใช้ CS แยกขา GPIO ธรรมดา ควบคุมด้วยซอฟต์แวร์ทีละตัว (ตัวอย่างที่ 3) แทนการแชร์ NSS ฮาร์ดแวร์เดียว
