I2C — Inter-Integrated Circuit

ระดับสูง
ไลบรารี: #include <ch32v00x_i2c.h> — นี่คือ Standard Peripheral Library (SPL) ระดับล่าง เขียนคอนฟิกผ่าน struct I2C_InitTypeDef โดยตรงกับเรจิสเตอร์ฮาร์ดแวร์ และตรวจสอบสถานะการสื่อสารด้วยการ poll event/flag เอง (ไม่มี buffer, ไม่มี interrupt callback อัตโนมัติแบบ wrapper)
อยากได้อะไรที่ง่ายกว่านี้? ถ้าต้องการ I2C แบบเรียกใช้งานง่าย ไม่ต้อง poll event เอง ดูที่ SimpleI2C (SimpleHAL wrapper) แทน หน้านี้เหมาะสำหรับคนที่ต้องการควบคุมรายละเอียดระดับ register/event เอง

ผังการทำงานของ I2C Master (Event Polling Model)

flowchart LR
    INIT["I2C_Init()"] --> EN["I2C_Cmd(ENABLE)"]
    EN --> START["I2C_GenerateSTART()"]
    START --> EVT5["รอ EVT5:
I2C_EVENT_MASTER_MODE_SELECT"]
    EVT5 --> ADDR["I2C_Send7bitAddress()"]

I2C_InitTypeDef — โครงสร้างการตั้งค่า

สมาชิกชนิดค่าที่เป็นไปได้คำอธิบาย
I2C_ClockSpeeduint32_tค่าความถี่ (Hz) เช่น 100000, 400000ความถี่สัญญาณนาฬิกา SCL ต้อง < 400kHz
I2C_Modeuint16_tI2C_Mode_I2Cโหมดการทำงาน (CH32V003 มีเพียงโหมดเดียว)
I2C_DutyCycleuint16_tI2C_DutyCycle_2, I2C_DutyCycle_16_9อัตราส่วน Tlow/Thigh เมื่อความเร็ว > 100kHz (Fast mode)
I2C_OwnAddress1uint16_tที่อยู่ 7 หรือ 10 บิตที่อยู่ของตัวเองเมื่อทำงานเป็น Slave
I2C_Ackuint16_tI2C_Ack_Enable, I2C_Ack_Disableเปิด/ปิดการตอบรับ ACK อัตโนมัติ
I2C_AcknowledgedAddressuint16_tI2C_AcknowledgedAddress_7bit, _10bitขนาดที่อยู่ที่ยอมรับเมื่อเป็น Slave

I2C_StructInit() ตั้งค่าเริ่มต้น: ClockSpeed=5000, Mode=I2C_Mode_I2C, DutyCycle=I2C_DutyCycle_2, OwnAddress1=0, Ack=Disable, AcknowledgedAddress=7bit

I2C Event Macros (ใช้กับ I2C_CheckEvent)

ค่าคงที่เหล่านี้แทนรูปแบบ flag รวมกันในเรจิสเตอร์ STAR1/STAR2 ที่บอกจุดต่างๆ ของการสื่อสาร — ใช้คู่กับ I2C_CheckEvent(I2Cx, EVENT)

Master events

Macroเมื่อไหร่ (EVTx)ความหมาย
I2C_EVENT_MASTER_MODE_SELECTEVT5ส่ง START สำเร็จแล้ว บัสพร้อมส่งที่อยู่
I2C_EVENT_MASTER_TRANSMITTER_MODE_SELECTEDEVT6Slave ตอบรับที่อยู่แล้ว โหมด Transmitter (ส่งข้อมูล)
I2C_EVENT_MASTER_RECEIVER_MODE_SELECTEDEVT6Slave ตอบรับที่อยู่แล้ว โหมด Receiver (รับข้อมูล)
I2C_EVENT_MASTER_MODE_ADDRESS10EVT9ส่ง header ที่อยู่ 10-bit ท่อนแรกสำเร็จ
I2C_EVENT_MASTER_BYTE_RECEIVEDEVT7รับข้อมูล 1 byte แล้ว อ่านได้ด้วย I2C_ReceiveData()
I2C_EVENT_MASTER_BYTE_TRANSMITTINGEVT8ข้อมูลอยู่ใน shift register กำลังส่งออก (TXE=1)
I2C_EVENT_MASTER_BYTE_TRANSMITTEDEVT8_2ส่งข้อมูลออกบัสสำเร็จจริง (TXE=1, BTF=1) — เชื่อถือได้กว่า EVT8

Slave events

Macroเมื่อไหร่ (EVTx)ความหมาย
I2C_EVENT_SLAVE_RECEIVER_ADDRESS_MATCHEDEVT1Master เรียกที่อยู่ตรงกับ OwnAddress1 โหมด Receiver
I2C_EVENT_SLAVE_TRANSMITTER_ADDRESS_MATCHEDEVT1Master เรียกที่อยู่ตรงกับ OwnAddress1 โหมด Transmitter
I2C_EVENT_SLAVE_RECEIVER_SECONDADDRESS_MATCHEDEVT1ตรงกับที่อยู่ที่สอง (Dual Address) โหมด Receiver
I2C_EVENT_SLAVE_TRANSMITTER_SECONDADDRESS_MATCHEDEVT1ตรงกับที่อยู่ที่สอง โหมด Transmitter
I2C_EVENT_SLAVE_GENERALCALLADDRESS_MATCHEDEVT1Master ส่ง General Call address (0x00)
I2C_EVENT_SLAVE_BYTE_RECEIVEDEVT2รับข้อมูล 1 byte จาก Master แล้ว
I2C_EVENT_SLAVE_STOP_DETECTEDEVT4ตรวจพบ STOP condition จาก Master
I2C_EVENT_SLAVE_BYTE_TRANSMITTEDEVT3ส่งข้อมูล 1 byte ให้ Master สำเร็จ (BTF=1)
I2C_EVENT_SLAVE_BYTE_TRANSMITTINGEVT3กำลังส่งข้อมูลให้ Master (TXE=1)
I2C_EVENT_SLAVE_ACK_FAILUREEVT3_2Master ส่ง NACK บอกให้หยุดส่งข้อมูล (AF flag)

ตัวอย่างโค้ด

1. เริ่มต้นใช้งาน I2C1 (Master, 100kHz)

#include <ch32v00x_i2c.h>
#include <ch32v00x_gpio.h>
#include <ch32v00x_rcc.h>

void I2C1_Init_Master(void)
{
    GPIO_InitTypeDef GPIO_InitStructure = {0};
    I2C_InitTypeDef I2C_InitStructure = {0};

    RCC_APB2PeriphClockCmd(RCC_APB2Periph_GPIOC, ENABLE);
    RCC_APB1PeriphClockCmd(RCC_APB1Periph_I2C1, ENABLE);

    /* PC1=SDA, PC2=SCL ต้องเป็น AF Open-Drain เสมอสำหรับ I2C */
    GPIO_InitStructure.GPIO_Pin = GPIO_Pin_1 | GPIO_Pin_2;
    GPIO_InitStructure.GPIO_Mode = GPIO_Mode_AF_OD;
    GPIO_InitStructure.GPIO_Speed = GPIO_Speed_10MHz;
    GPIO_Init(GPIOC, &GPIO_InitStructure);

    I2C_InitStructure.I2C_ClockSpeed = 100000;
    I2C_InitStructure.I2C_Mode = I2C_Mode_I2C;
    I2C_InitStructure.I2C_DutyCycle = I2C_DutyCycle_2;
    I2C_InitStructure.I2C_OwnAddress1 = 0x00;
    I2C_InitStructure.I2C_Ack = I2C_Ack_Enable;
    I2C_InitStructure.I2C_AcknowledgedAddress = I2C_AcknowledgedAddress_7bit;
    I2C_Init(I2C1, &I2C_InitStructure);

    I2C_Cmd(I2C1, ENABLE);
}

2. Master เขียนข้อมูล (Write) — START → Address → Data → STOP

uint8_t I2C1_WriteByte(uint8_t slaveAddr, uint8_t regAddr, uint8_t data)
{
    /* EVT5: ส่ง START */
    I2C_GenerateSTART(I2C1, ENABLE);
    while (!I2C_CheckEvent(I2C1, I2C_EVENT_MASTER_MODE_SELECT));

    /* EVT6: ส่งที่อยู่ slave โหมด Transmitter */
    I2C_Send7bitAddress(I2C1, slaveAddr, I2C_Direction_Transmitter);
    while (!I2C_CheckEvent(I2C1, I2C_EVENT_MASTER_TRANSMITTER_MODE_SELECTED));

    /* EVT8: ส่ง register address */
    I2C_SendData(I2C1, regAddr);
    while (!I2C_CheckEvent(I2C1, I2C_EVENT_MASTER_BYTE_TRANSMITTED));

    /* EVT8_2: ส่งข้อมูลจริง */
    I2C_SendData(I2C1, data);
    while (!I2C_CheckEvent(I2C1, I2C_EVENT_MASTER_BYTE_TRANSMITTED));

    I2C_GenerateSTOP(I2C1, ENABLE);
    return 0;
}

3. Master อ่านข้อมูล (Read) พร้อม ACK/NACK — อ่านหลาย byte

void I2C1_ReadBytes(uint8_t slaveAddr, uint8_t regAddr, uint8_t *buf, uint8_t len)
{
    /* เขียน register address ก่อน (repeated-start pattern) */
    I2C_GenerateSTART(I2C1, ENABLE);
    while (!I2C_CheckEvent(I2C1, I2C_EVENT_MASTER_MODE_SELECT));
    I2C_Send7bitAddress(I2C1, slaveAddr, I2C_Direction_Transmitter);
    while (!I2C_CheckEvent(I2C1, I2C_EVENT_MASTER_TRANSMITTER_MODE_SELECTED));
    I2C_SendData(I2C1, regAddr);
    while (!I2C_CheckEvent(I2C1, I2C_EVENT_MASTER_BYTE_TRANSMITTED));

    /* Repeated START แล้วเปลี่ยนเป็น Receiver */
    I2C_GenerateSTART(I2C1, ENABLE);
    while (!I2C_CheckEvent(I2C1, I2C_EVENT_MASTER_MODE_SELECT));
    I2C_AcknowledgeConfig(I2C1, ENABLE);   /* เปิด ACK ไว้ก่อนสำหรับ byte ที่ไม่ใช่ตัวสุดท้าย */
    I2C_Send7bitAddress(I2C1, slaveAddr, I2C_Direction_Receiver);
    while (!I2C_CheckEvent(I2C1, I2C_EVENT_MASTER_RECEIVER_MODE_SELECTED));

    for (uint8_t i = 0; i < len; i++) {
        if (i == len - 1) {
            /* byte สุดท้าย: ปิด ACK ก่อนรับ เพื่อให้ตอบ NACK แล้วตามด้วย STOP */
            I2C_AcknowledgeConfig(I2C1, DISABLE);
            I2C_GenerateSTOP(I2C1, ENABLE);
        }
        while (!I2C_CheckEvent(I2C1, I2C_EVENT_MASTER_BYTE_RECEIVED));
        buf[i] = I2C_ReceiveData(I2C1);
    }
}
ลำดับ NACK สำคัญมาก: ต้องเรียก I2C_AcknowledgeConfig(I2C1, DISABLE) และ I2C_GenerateSTOP() ก่อนอ่าน byte สุดท้ายด้วย I2C_ReceiveData() เสมอ ถ้าเรียกหลังอ่านจะไปมีผลกับ byte ถัดไปแทน (ผิดจังหวะ ทำให้ค้างหรืออ่านเกิน)

ปัญหาที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
บัสค้าง (bus lockup) ไม่ตอบสนอง ลืม poll I2C_CheckEvent() ให้ครบทุกขั้นตอน หรือ slave ไม่ตอบ ACK เช็ค event ทุกขั้น (START → ADDR → DATA) ตามลำดับ, เพิ่ม timeout กันค้างตลอดกาล, ถ้าค้างจริงใช้ I2C_SoftwareResetCmd() กู้บัส
อ่าน byte สุดท้ายผิด/บัสไม่ปล่อย เรียก I2C_AcknowledgeConfig(DISABLE) + I2C_GenerateSTOP() หลังอ่าน I2C_ReceiveData() แทนที่จะเรียกก่อน ต้องตั้ง NACK และ STOP ก่อนอ่าน byte สุดท้ายเสมอ (ดูตัวอย่างที่ 3)
ใช้ event ผิด ทำให้ I2C_CheckEvent ไม่คืน SUCCESS ตลอด (ค้างที่ while loop) สลับใช้ event ของ Master กับ Slave ผิด หรือใช้ Transmitter event ทั้งที่เป็นโหมด Receiver ตรวจสอบทิศทางที่ตั้งใน I2C_Send7bitAddress() ให้ตรงกับ event ที่รอ (Transmitter → _TRANSMITTER_MODE_SELECTED, Receiver → _RECEIVER_MODE_SELECTED)
GPIO ตั้งผิดโหมด สื่อสารไม่ได้เลย ตั้ง SDA/SCL เป็น GPIO_Mode_Out_PP แทนที่จะเป็น Open-Drain I2C ต้องใช้ GPIO_Mode_AF_OD (Alternate Function Open-Drain) เสมอ เพราะบัสต้องรองรับหลายอุปกรณ์ดึงลงพร้อมกันได้
ความเร็วบัสไม่ตรงที่ตั้งไว้ I2C_Init() คำนวณค่าจาก PCLK1 จริง ไม่ใช่ค่าคงที่ ถ้า clock system ตั้งผิดจะได้ความเร็วบัสผิด เรียก RCC_GetClocksFreq()/ตรวจ system clock ให้ถูกต้องก่อน I2C_Init()
ลืมเปิด peripheral clock (RCC) ของ I2C1 ไม่ได้เรียก RCC_APB1PeriphClockCmd(RCC_APB1Periph_I2C1, ENABLE) เปิด RCC clock ทั้ง GPIO port และ I2C1 ก่อนเรียก I2C_Init() เสมอ
