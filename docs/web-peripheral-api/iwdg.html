IWDG — Independent Watchdog (วอทช์ด็อกอิสระ)

ระดับสูง
Header: #include <ch32v00x_iwdg.h> — นี่คือ Standard Peripheral Library (SPL) ระดับต่ำ เขียนเรจิสเตอร์ตรง ไม่มีการตรวจสอบพารามิเตอร์หรือ abstraction ใดๆ ผู้ใช้ต้องคำนวณค่า prescaler/reload เองและเข้าใจพฤติกรรมฮาร์ดแวร์
ทางเลือกที่ง่ายกว่า: ถ้าต้องการ API ที่คำนวณ timeout เป็น ms ให้อัตโนมัติ ดูที่ SimpleIWDG (SimpleHAL API Reference) ซึ่งห่อ (wrap) ฟังก์ชันชุดนี้ไว้อีกชั้น

ภาพรวม

IWDG (Independent Watchdog) เป็นวอทช์ด็อกฮาร์ดแวร์ที่ใช้นาฬิกา LSI ~40kHz ซึ่งเป็นอิสระจากนาฬิการะบบหลัก (HSE/HSI/PLL) — แม้ระบบนาฬิกาหลักหยุดทำงาน IWDG ก็ยังนับถอยหลังต่อไปและรีเซ็ต MCU ได้ตามปกติ

  • Reload register (RLDR) — 12 บิต ค่า 0x000-0xFFF (0-4095)
  • Prescaler (PSCR) — 7 ระดับ หาร 4 ถึง 256
  • Key register (CTLR) — เขียนผ่าน "key" พิเศษ 3 ค่า (0x5555 / 0xAAAA / 0xCCCC) เท่านั้น ไม่มีฟิลด์อื่นให้เขียนตรง

ผังการทำงาน IWDG

flowchart LR
    WA["IWDG_WriteAccessCmd(Enable)"] --> SP["IWDG_SetPrescaler()"]
    SP --> SR["IWDG_SetReload()"]
    SR --> RC["IWDG_ReloadCounter()"]
    RC --> EN["IWDG_Enable()"]
    EN --> LOOP["นับถอยหลังจาก Reload → 0"]
    LOOP -->|"เรียก IWDG_ReloadCounter() ทัน"| RC
    LOOP -->|"ไม่เรียกทัน (โปรแกรมค้าง)"| RST["MCU รีเซ็ต"]

ค่าคงที่ (Constants)

IWDG_WriteAccess — คีย์เปิด/ปิดการเขียน PSCR และ RLDR

ค่าคงที่ค่าคำอธิบาย
IWDG_WriteAccess_Enable0x5555เปิดการเขียนเรจิสเตอร์ PSCR/RLDR (ต้องเรียกก่อน SetPrescaler/SetReload ทุกครั้ง)
IWDG_WriteAccess_Disable0x0000ปิดการเขียน (ป้องกันการเขียนโดยไม่ตั้งใจ)

IWDG_Prescaler — ตัวหารนาฬิกา LSI (~40kHz)

ค่าคงที่ค่าหารด้วยTimeout สูงสุด* (Reload=0xFFF)
IWDG_Prescaler_40x004≈ 0.41 s
IWDG_Prescaler_80x018≈ 0.82 s
IWDG_Prescaler_160x0216≈ 1.64 s
IWDG_Prescaler_320x0332≈ 3.28 s
IWDG_Prescaler_640x0464≈ 6.55 s
IWDG_Prescaler_1280x05128≈ 13.1 s
IWDG_Prescaler_2560x06256≈ 26.2 s

*คำนวณจาก LSI = 40kHz พอดี ในความเป็นจริง LSI มี tolerance กว้าง ค่าจริงจึงคลาดเคลื่อนได้มาก — ดูหัวข้อ "ปัญหาที่พบบ่อย"

IWDG_Flag — บิตสถานะใน STATR

ค่าคงที่ค่าความหมาย
IWDG_FLAG_PVU0x0001Prescaler Value Update — ค่า PSCR กำลังถูกอัปเดตเข้าโดเมนนาฬิกา LSI ยังไม่เสร็จ
IWDG_FLAG_RVU0x0002Reload Value Update — ค่า RLDR กำลังถูกอัปเดตเข้าโดเมนนาฬิกา LSI ยังไม่เสร็จ

สูตรคำนวณ Timeout

// timeout (ms) = prescaler_divide * (Reload + 1) / 40   (LSI ~40kHz)
// ตัวอย่าง: ต้องการ timeout ~500ms
// เลือก prescaler = 32 (หาร 32) -> Reload = 500*40/32 - 1 = 624 (0x270)
IWDG_WriteAccessCmd(IWDG_WriteAccess_Enable);
IWDG_SetPrescaler(IWDG_Prescaler_32);
IWDG_SetReload(624);
IWDG_ReloadCounter();
IWDG_Enable();

ตารางอ้างอิงฟังก์ชัน

ฟังก์ชันพารามิเตอร์คืนค่าคำอธิบายหมายเหตุ
IWDG_WriteAccessCmd(IWDG_WriteAccess) IWDG_WriteAccess: IWDG_WriteAccess_Enable / _Disable void เขียนคีย์ 0x5555 ลง CTLR เพื่อปลดล็อกการเขียน PSCR/RLDR ต้องเรียกก่อน IWDG_SetPrescaler/IWDG_SetReload ทุกครั้ง มิฉะนั้นค่าจะไม่ถูกเขียน
IWDG_SetPrescaler(IWDG_Prescaler) IWDG_Prescaler: IWDG_Prescaler_4 ... _256 void เขียนค่าตัวหารลง PSCR ต้อง unlock ด้วย IWDG_WriteAccessCmd(Enable) ก่อน
IWDG_SetReload(Reload) Reload: uint16_t ช่วง 0-0x0FFF (0-4095) void เขียนค่าโหลดใหม่ลง RLDR ต้อง unlock ด้วย IWDG_WriteAccessCmd(Enable) ก่อน
IWDG_ReloadCounter() - void เขียนคีย์ 0xAAAA ลง CTLR สั่งให้ counter โหลดค่าจาก RLDR ใหม่ ("ป้อนอาหาร" watchdog) ต้องเรียกซ้ำๆ ในลูปหลักก่อน timeout หมด
IWDG_Enable() - void เขียนคีย์ 0xCCCC ลง CTLR เปิดใช้งาน IWDG แล้วรอจนกว่า RCC_RSTSCKR bit1 (IWDGRSTF) จะไม่ว่าง เมื่อเปิดแล้วปิดด้วยซอฟต์แวร์ไม่ได้ — ต้อง reset MCU เท่านั้น
IWDG_GetFlagStatus(IWDG_FLAG) IWDG_FLAG: IWDG_FLAG_PVU / IWDG_FLAG_RVU FlagStatus (SET/RESET) อ่านบิตจาก STATR ตรวจว่าเรจิสเตอร์กำลังอัปเดตอยู่หรือไม่ ควร poll จนเป็น RESET ก่อนเขียนค่าใหม่ซ้ำ (การอัปเดตข้ามโดเมนนาฬิกาใช้เวลาหลายรอบ LSI)

ตัวอย่างโค้ด

1. Init IWDG timeout ~1 วินาที + ป้อนใน main loop

#include <ch32v00x.h>
#include <ch32v00x_iwdg.h>

void IWDG_Init_1s(void)
{
    // prescaler=64 -> LSI/64 = 625 Hz, ต้องการ 1000ms
    // Reload = 1000 * 40 / 64 - 1 = 624 (0x270)
    IWDG_WriteAccessCmd(IWDG_WriteAccess_Enable);
    IWDG_SetPrescaler(IWDG_Prescaler_64);
    IWDG_SetReload(624);
    IWDG_ReloadCounter();
    IWDG_Enable();
}

int main(void)
{
    SystemCoreClockUpdate();
    IWDG_Init_1s();

    while (1) {
        // ... งานหลัก ...
        IWDG_ReloadCounter();   // ต้องเรียกให้ทันภายใน ~1s เสมอ
        Delay_Ms(200);
    }
}

2. ตรวจสอบว่ารีเซ็ตล่าสุดเกิดจาก IWDG หรือไม่

#include <ch32v00x.h>
#include <ch32v00x_iwdg.h>

int main(void)
{
    SystemCoreClockUpdate();

    if (RCC_GetFlagStatus(RCC_FLAG_IWDGRST) != RESET) {
        // รีเซ็ตครั้งล่าสุดเกิดจาก IWDG timeout -> โปรแกรมเคยค้าง
        RCC_ClearFlag();
    }

    IWDG_Init_1s();
    while (1) {
        IWDG_ReloadCounter();
        Delay_Ms(200);
    }
}

3. รอให้ PVU/RVU เคลียร์ก่อนเขียนค่าซ้ำ (เปลี่ยน timeout ระหว่างรันไทม์)

void IWDG_ChangeTimeout(uint8_t prescaler, uint16_t reload)
{
    IWDG_WriteAccessCmd(IWDG_WriteAccess_Enable);

    while (IWDG_GetFlagStatus(IWDG_FLAG_PVU) == SET);  // รอ prescaler เก่าเขียนเสร็จ
    IWDG_SetPrescaler(prescaler);

    while (IWDG_GetFlagStatus(IWDG_FLAG_RVU) == SET);  // รอ reload เก่าเขียนเสร็จ
    IWDG_SetReload(reload);

    IWDG_ReloadCounter();
}

ปัญหาที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
Timeout จริงคลาดเคลื่อนจากที่คำนวณมาก LSI เป็น RC oscillator ภายใน ไม่ผ่านการ trim ละเอียดเหมือน HSE — ค่าความถี่เปลี่ยนตามอุณหภูมิ/แรงดันไฟ ±tolerance กว้าง (ไม่ใช่ 40kHz แม่นยำ) อย่าตั้ง timeout ให้ตึงเกินไป เผื่อ margin อย่างน้อย 2-3 เท่าของคาบ loop จริง และเลือก reload ให้ห่างขอบเขต
เปิด IWDG แล้วปิดไม่ได้ ดีบักลำบาก เมื่อเรียก IWDG_Enable() แล้ว ไม่มีฟังก์ชัน Disable ในฮาร์ดแวร์ — ต้องรีเซ็ตชิปเท่านั้นจึงจะปิดได้ อย่าเปิด IWDG ตอนช่วง debug ถ้ายังไม่พร้อม feed ต่อเนื่อง หรือหยุด debugger ค้างไว้ที่ breakpoint จะโดน IWDG รีเซ็ตกลางคัน
ตั้งค่า Prescaler/Reload ใหม่แล้วไม่มีผล ลืมเรียก IWDG_WriteAccessCmd(IWDG_WriteAccess_Enable) ก่อน — ถ้าไม่ unlock การเขียน PSCR/RLDR จะถูกฮาร์ดแวร์เพิกเฉย เรียก IWDG_WriteAccessCmd(Enable) ทุกครั้งก่อน SetPrescaler/SetReload
เขียนค่าซ้ำเร็วเกินไปแล้วค่าไม่อัปเดต การเขียน PSCR/RLDR ต้อง sync ข้ามจากโดเมนนาฬิกา APB ไปยังโดเมน LSI ใช้เวลาหลายรอบ LSI (ไม่ใช่ทันที) — ถ้าเขียนซ้ำถี่เกินไปจะชนกับการอัปเดตที่ยังค้างอยู่ ตรวจ IWDG_GetFlagStatus(IWDG_FLAG_PVU/RVU) ให้เป็น RESET ก่อนเขียนค่าครั้งถัดไป
Reload() ไม่ได้เรียกทัน MCU รีเซ็ตเอง โค้ดใน main loop มีจุดที่ block นาน (เช่น รอ I2C/SPI timeout, delay ยาว) เกินกว่า timeout ที่ตั้งไว้ ตั้ง timeout ให้ครอบคลุม worst-case ของทุก path ในลูป หรือเรียก IWDG_ReloadCounter() เพิ่มในจุดที่ block นาน
