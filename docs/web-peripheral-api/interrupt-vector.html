ข้อควรระวัง: CH32V003 รวม EXTI Line 0-7 เข้า IRQ เดียว (EXTI7_0_IRQn) — handler ตัวเดียวต้องรับผิดชอบทุก line ที่เปิดไว้ ต้องเช็คทีละ line ด้วย EXTI_GetITStatus()

Handler Attribute

CH32V003 ใช้ RISC-V ต้องประกาศ handler ด้วย attribute เฉพาะ:

// ประกาศ ISR สำหรับ CH32V003 (WCH RISC-V)
void EXTI7_0_IRQHandler(void) __attribute__((interrupt("WCH-Interrupt-fast")));
void EXTI7_0_IRQHandler(void)
{
    if (EXTI_GetITStatus(EXTI_Line1) != RESET) {
        // ทำงานตาม logic
        EXTI_ClearITPendingBit(EXTI_Line1);
    }
}

หมายเหตุเรื่อง Attribute

Attributeคำอธิบาย
__attribute__((interrupt))บังคับให้ compiler สร้าง prologue/epilogue สำหรับ interrupt (save/restore context)
"WCH-Interrupt-fast"โหมด fast interrupt ของ WCH — ใช้ register bank ช่วยลด latency

ตัวอย่างการใช้งาน

1. เปิด Interrupt หลายตัว

#include "ch32v00x_misc.h"

// เปิด TIM1 Update interrupt
NVIC_InitTypeDef NVIC_InitStructure = {0};
NVIC_InitStructure.NVIC_IRQChannel = TIM1_UP_IRQn;
NVIC_InitStructure.NVIC_IRQChannelPreemptionPriority = 1;
NVIC_InitStructure.NVIC_IRQChannelSubPriority = 0;
NVIC_InitStructure.NVIC_IRQChannelCmd = ENABLE;
NVIC_Init(&NVIC_InitStructure);

// เปิด USART1 interrupt
NVIC_InitStructure.NVIC_IRQChannel = USART1_IRQn;
NVIC_InitStructure.NVIC_IRQChannelPreemptionPriority = 2;
NVIC_InitStructure.NVIC_IRQChannelSubPriority = 0;
NVIC_Init(&NVIC_InitStructure);

2. หลาย ISR แยกตาม peripheral

// TIM1 Update ISR
void TIM1_UP_IRQHandler(void) __attribute__((interrupt("WCH-Interrupt-fast")));
void TIM1_UP_IRQHandler(void)
{
    if (TIM_GetITStatus(TIM1, TIM_IT_Update) != RESET) {
        TIM_ClearITPendingBit(TIM1, TIM_IT_Update);
        // ทำงานทุกครั้งที่ TIM1 overflow
    }
}

// USART1 ISR
void USART1_IRQHandler(void) __attribute__((interrupt("WCH-Interrupt-fast")));
void USART1_IRQHandler(void)
{
    if (USART_GetITStatus(USART1, USART_IT_RXNE) != RESET) {
        uint8_t data = USART_ReceiveData(USART1);
        // ประมวลผลข้อมูลที่รับมา
    }
}

3. Nested Interrupt (Interrupt ซ้อน)

// ตั้งค่า priority group
NVIC_PriorityGroupConfig(NVIC_PriorityGroup_2);  // 2 bit preemption + 2 bit sub

// TIM1 priority สูง (preemption=0) — ขัดจังหวะตัวอื่นได้
NVIC_InitStructure.NVIC_IRQChannel = TIM1_UP_IRQn;
NVIC_InitStructure.NVIC_IRQChannelPreemptionPriority = 0;
NVIC_InitStructure.NVIC_IRQChannelCmd = ENABLE;
NVIC_Init(&NVIC_InitStructure);

// USART1 priority ต่ำ (preemption=3) — ถูกขัดจังหวะได้
NVIC_InitStructure.NVIC_IRQChannel = USART1_IRQn;
NVIC_InitStructure.NVIC_IRQChannelPreemptionPriority = 3;
NVIC_Init(&NVIC_InitStructure);

ปัญหาที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
ประกาศ ISR แล้ว linker ไม่เจอ ลืม __attribute__((interrupt("WCH-Interrupt-fast"))) — compiler จะสร้าง function ธรรมดาที่ไม่ save context เพิ่ม attribute ทุกครั้งที่ประกาศ ISR สำหรับ CH32V003
ISR ทำงานแล้วค้าง ไม่ return ลืมเรียก ClearITPendingBit() — interrupt flag ยังค้างอยู่ ทำให้เข้า ISR ซ้ำทันที เรียก ClearITPendingBit() เป็นบรรทัดสุดท้ายของ ISR เสมอ
ตั้ง priority แล้วไม่สามารถ nested ได้ ลืมเรียก NVIC_PriorityGroupConfig() — ค่า default อาจไม่รองรับ preemption เรียก NVIC_PriorityGroupConfig() ตอนเริ่มต้นโปรแกรม
EXTI หลาย line ทำงานไม่ครบ CH32V003 รวม EXTI0-7 เป็น IRQ เดียว — ต้องเช็คทีละ line ใน handler ใช้ if (EXTI_GetITStatus(EXTI_Line_x) != RESET) เช็คทีละ line
