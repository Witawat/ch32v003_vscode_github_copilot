USART — Universal Synchronous/Asynchronous Receiver Transmitter

ระดับขั้นสูง (Raw SPL)
Header: #include <ch32v00x_usart.h> — นี่คือ Standard Peripheral Library (SPL) ระดับล่าง ต้องตั้งค่า struct เอง เปิด clock เอง (RCC_APB2PeriphClockCmd) และตั้งค่า GPIO เอง (AFIO alternate function) ก่อนเรียกใช้งาน — ไม่มีการ auto-config ให้เหมือน wrapper
อยากง่ายกว่านี้? ถ้าไม่จำเป็นต้องคุมทุกบิตเอง ให้ใช้ SimpleUSART แทน — มี ring buffer รับข้อมูลผ่าน interrupt ให้พร้อมใช้ ไม่ต้องตั้งค่า NVIC/GPIO เอง

ผังการทำงาน USART (SPL)

flowchart LR
    RCC["RCC_APB2PeriphClockCmd\n(USART1 + GPIO)"] --> GPIOCFG["GPIO_Init\n(TX=AF_PP, RX=IN_FLOATING)"]
    GPIOCFG --> INIT["USART_Init\n(baud/word/stop/parity)"]
    INIT --> CMD["USART_Cmd(ENABLE)"]
    CMD --> TXP["Polling TX:\nUSART_SendData + รอ USART_FLAG_TXE"]
    CMD --> RXP["Polling RX:\nรอ USART_FLAG_RXNE + USART_ReceiveData"]
    CMD --> ITC["USART_ITConfig(USART_IT_RXNE, ENABLE)\n+ NVIC_Init"]
    ITC --> ISR["USART1_IRQHandler\nUSART_GetITStatus → USART_ReceiveData"]

USART_InitTypeDef — โครงสร้างตั้งค่าหลัก

ส่ง pointer ของ struct นี้ให้ USART_Init(USART1, &init) เพื่อกำหนดค่าทั้งหมดในครั้งเดียว

ฟิลด์ชนิดค่าที่ใช้ได้คำอธิบาย
USART_BaudRateuint32_tเช่น 9600, 115200อัตราบิต — คำนวณ BRR จาก PCLK2 (USART1 อยู่บน APB2)
USART_WordLengthuint16_tUSART_WordLength_8b, USART_WordLength_9bจำนวนบิตข้อมูลต่อเฟรม
USART_StopBitsuint16_tUSART_StopBits_1, _0_5, _2, _1_5จำนวนบิตหยุด
USART_Parityuint16_tUSART_Parity_No, _Even, _Oddโหมดพาริตี — เมื่อเปิดใช้ พาริตีจะแทรกที่บิต MSB ของเฟรม
USART_Modeuint16_tUSART_Mode_Rx, USART_Mode_Tx (OR รวมกันได้)เปิดโหมดรับ/ส่ง — ปกติใช้ USART_Mode_Rx | USART_Mode_Tx
USART_HardwareFlowControluint16_t_None, _RTS, _CTS, _RTS_CTSควบคุมการไหลด้วยฮาร์ดแวร์ (CH32V003 ปกติใช้ _None)
USART_StructInit: เรียก USART_StructInit(&init) ก่อนได้ เพื่อเซ็ตค่าเริ่มต้น (9600, 8b, 1 stop, no parity, Rx+Tx, no flow control) แล้วค่อยแก้เฉพาะฟิลด์ที่ต้องการ

USART_ClockInitTypeDef — สำหรับโหมด Synchronous เท่านั้น

ฟิลด์ค่าที่ใช้ได้คำอธิบาย
USART_ClockUSART_Clock_Disable, USART_Clock_Enableเปิด/ปิดสัญญาณนาฬิกา SCLK ออกขา
USART_CPOLUSART_CPOL_Low, USART_CPOL_Highขั้วสัญญาณนาฬิกาขณะ idle
USART_CPHAUSART_CPHA_1Edge, USART_CPHA_2Edgeขอบที่ใช้จับบิตข้อมูล
USART_LastBitUSART_LastBit_Disable, USART_LastBit_Enableส่งพัลส์นาฬิกาของบิตสุดท้ายออกขา SCLK หรือไม่

ใช้กับ USART_ClockInit(USART1, &clockInit) — เกี่ยวข้องเฉพาะเมื่อใช้ USART แบบซิงโครนัส (มีขา SCLK) ซึ่งไม่ค่อยได้ใช้งานทั่วไปในโปรเจกต์ CH32V003

ค่าคงที่ Flag และ Interrupt

กลุ่มค่าคงที่ความหมาย
USART_FLAG (ใช้กับ USART_GetFlagStatus/USART_ClearFlag)USART_FLAG_CTSสถานะขา CTS เปลี่ยน
USART_FLAG_LBDตรวจพบ LIN Break
USART_FLAG_TXETransmit Data Register ว่าง — ใช้บ่อยที่สุดสำหรับ polling TX
USART_FLAG_TCส่งข้อมูลเสร็จสมบูรณ์ (shift register ว่างด้วย)
USART_FLAG_RXNEมีข้อมูลใหม่ให้อ่าน — ใช้บ่อยที่สุดสำหรับ polling RX
USART_FLAG_IDLEตรวจพบเส้น idle
USART_FLAG_OREOverrun error — รับข้อมูลใหม่ทับข้อมูลเก่าที่ยังไม่ได้อ่าน
USART_FLAG_NENoise error
USART_FLAG_FEFraming error
USART_IT (ใช้กับ USART_ITConfig/USART_GetITStatus/USART_ClearITPendingBit)USART_IT_TXEขัดจังหวะเมื่อ TX ว่าง
USART_IT_TCขัดจังหวะเมื่อส่งเสร็จ
USART_IT_RXNEขัดจังหวะเมื่อมีข้อมูลรับเข้ามา — ตัวที่ใช้บ่อยสุดสำหรับ RX แบบ interrupt
USART_IT_IDLEขัดจังหวะเมื่อพบเส้น idle
USART_IT_PEขัดจังหวะ parity error
USART_IT_LBDขัดจังหวะ LIN break detect
USART_IT_ERR / _ORE / _NE / _FE / _CTSขัดจังหวะกลุ่ม error (ต้องเปิด USART_IT_ERR ด้วย)

ตัวอย่างโค้ด

1. USART_Init + Polling TX (ส่งข้อมูล)

#include <ch32v00x.h>

void USART1_Init_115200(void)
{
    GPIO_InitTypeDef  GPIO_InitStructure = {0};
    USART_InitTypeDef USART_InitStructure = {0};

    RCC_APB2PeriphClockCmd(RCC_APB2Periph_GPIOD | RCC_APB2Periph_USART1, ENABLE);

    /* PD5 = USART1_TX (alternate function push-pull) */
    GPIO_InitStructure.GPIO_Pin = GPIO_Pin_5;
    GPIO_InitStructure.GPIO_Speed = GPIO_Speed_50MHz;
    GPIO_InitStructure.GPIO_Mode = GPIO_Mode_AF_PP;
    GPIO_Init(GPIOD, &GPIO_InitStructure);

    /* PD6 = USART1_RX (floating input) */
    GPIO_InitStructure.GPIO_Pin = GPIO_Pin_6;
    GPIO_InitStructure.GPIO_Mode = GPIO_Mode_IN_FLOATING;
    GPIO_Init(GPIOD, &GPIO_InitStructure);

    USART_InitStructure.USART_BaudRate = 115200;
    USART_InitStructure.USART_WordLength = USART_WordLength_8b;
    USART_InitStructure.USART_StopBits = USART_StopBits_1;
    USART_InitStructure.USART_Parity = USART_Parity_No;
    USART_InitStructure.USART_Mode = USART_Mode_Rx | USART_Mode_Tx;
    USART_InitStructure.USART_HardwareFlowControl = USART_HardwareFlowControl_None;
    USART_Init(USART1, &USART_InitStructure);

    USART_Cmd(USART1, ENABLE);   /* อย่าลืมบรรทัดนี้! */
}

void USART1_SendByte(uint8_t b)
{
    while (USART_GetFlagStatus(USART1, USART_FLAG_TXE) == RESET);
    USART_SendData(USART1, b);
}

void USART1_SendString(const char *s)
{
    while (*s) USART1_SendByte(*s++);
}

2. Polling RX (รับข้อมูล)

uint8_t USART1_ReceiveByte(void)
{
    while (USART_GetFlagStatus(USART1, USART_FLAG_RXNE) == RESET);
    return (uint8_t)USART_ReceiveData(USART1);   /* การอ่านนี้เคลียร์ RXNE ให้อัตโนมัติ */
}

int main(void)
{
    SystemCoreClockUpdate();
    USART1_Init_115200();

    while (1) {
        uint8_t c = USART1_ReceiveByte();
        USART1_SendByte(c);   /* echo กลับ */
    }
}

3. รับข้อมูลผ่าน Interrupt (RXNE ISR)

#include <ch32v00x.h>

volatile uint8_t rxByte;
volatile uint8_t rxReady = 0;

void USART1_Init_IT(void)
{
    /* ... ตั้งค่า GPIO + USART_Init + USART_Cmd เหมือนตัวอย่างที่ 1 ... */

    USART_ITConfig(USART1, USART_IT_RXNE, ENABLE);

    NVIC_InitTypeDef NVIC_InitStructure = {0};
    NVIC_InitStructure.NVIC_IRQChannel = USART1_IRQn;
    NVIC_InitStructure.NVIC_IRQChannelPreemptionPriority = 1;
    NVIC_InitStructure.NVIC_IRQChannelSubPriority = 1;
    NVIC_InitStructure.NVIC_IRQChannelCmd = ENABLE;
    NVIC_Init(&NVIC_InitStructure);
}

void USART1_IRQHandler(void) __attribute__((interrupt("WCH-Interrupt-fast")));
void USART1_IRQHandler(void)
{
    if (USART_GetITStatus(USART1, USART_IT_RXNE) != RESET) {
        rxByte = (uint8_t)USART_ReceiveData(USART1);  /* อ่านค่าเคลียร์ RXNE ให้อัตโนมัติ */
        rxReady = 1;
    }
}

CH32V003 ใช้ attribute __attribute__((interrupt("WCH-Interrupt-fast"))) สำหรับ ISR ตามธรรมเนียมของ RISC-V core นี้ — ดูตัวอย่างอื่นในโปรเจกต์ที่ User/

ปัญหาที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
ตั้งค่าครบแล้วแต่ไม่ส่ง/ไม่รับข้อมูลเลย ลืมเรียก USART_Cmd(USART1, ENABLE) หลัง USART_Init ตรวจสอบว่ามีบรรทัด USART_Cmd อยู่เสมอหลังตั้งค่า struct
Baud rate ผิด (ตัวหนังสือมั่ว/ไม่ตรงกัน) USART_Init คำนวณ BRR จาก PCLK2 (สำหรับ USART1) ผ่าน RCC_GetClocksFreq — ถ้า clock ระบบตั้งไว้ผิด (เช่นลืม SystemCoreClockUpdate()) ค่าที่คำนวณจะผิดตาม เรียก SystemCoreClockUpdate() ก่อนตั้งค่า USART เสมอ และตรวจสอบว่า RCC clock config ตรงกับที่ต้องการ
ข้อมูลรับเพี้ยน/หายบางไบต์ (Overrun) ไม่ได้อ่าน USART_ReceiveData ให้ทันก่อนไบต์ถัดไปมาถึง ทำให้เกิด USART_FLAG_ORE อ่าน DATAR ให้เร็ว (ใช้ interrupt แทน polling ถ้า loop หลักช้า) หรือใช้ ring buffer อย่าง SimpleUSART
ใช้ Interrupt แต่ ISR ไม่ถูกเรียก ลืมตั้งค่า NVIC (NVIC_Init กับ USART1_IRQn) หรือลืม USART_ITConfig(..., ENABLE) ตรวจสอบทั้งสองขั้นตอน — USART_ITConfig เปิดที่ตัว peripheral, NVIC_Init เปิดที่ตัว core
ลืมตั้งค่า GPIO ให้ตรงโหมด ขา TX ต้องเป็น GPIO_Mode_AF_PP (alternate function push-pull) ไม่ใช่ output ธรรมดา ขา RX ต้องเป็น GPIO_Mode_IN_FLOATING ตรวจสอบโหมด GPIO ให้ตรงตามตารางในเอกสาร datasheet ของแต่ละขา
