ADC — Analog-to-Digital Converter

ระดับขั้นสูง (Raw SPL)
Header: #include "ch32v00x_adc.h" — นี่คือ Standard Peripheral Library (SPL) ระดับล่าง ทำงานผ่าน struct ADC_InitTypeDef และเขียนตรงไปยังเรจิสเตอร์ฮาร์ดแวร์ ไม่มี การช่วยเหลือแบบ lazy-init, ไม่มี compile-time validation ของ pin เหมือน SimpleHAL — ผู้ใช้ต้องตั้งค่า RCC clock, GPIO analog mode, calibration และลำดับการเปิดใช้งานเองทั้งหมด
อยากได้อะไรที่ง่ายกว่านี้? ถ้าต้องการแค่ analogRead(pin) แบบไม่ต้องยุ่งกับเรจิสเตอร์ ดูที่ SimpleADC แทน — หน้านี้เหมาะสำหรับกรณีที่ต้องการควบคุมละเอียด เช่น DMA scan หลายช่อง, injected channel, analog watchdog, หรือปรับแต่ง sample time เอง

    J -- ใช่ --> L["ADC_DMACmd(ADC1, ENABLE) + ตั้งค่า DMA channel"]

ค่าคงที่คำอธิบาย

ตัวอย่างโค้ด

1. แปลงค่าเดียวแบบ Polling (single channel, software trigger)

#include "ch32v00x_adc.h"
#include "ch32v00x_gpio.h"
#include "ch32v00x_rcc.h"

void ADC_SingleInit(void)
{
    GPIO_InitTypeDef  GPIO_InitStructure = {0};
    ADC_InitTypeDef   ADC_InitStructure = {0};

    RCC_APB2PeriphClockCmd(RCC_APB2Periph_GPIOC | RCC_APB2Periph_ADC1, ENABLE);
    RCC_ADCCLKConfig(RCC_PCLK2_Div8);   // ADCCLK = PCLK2/8 (ห้ามเกิน max ตาม datasheet)

    // ตั้งขา PC4 เป็น analog input
    GPIO_InitStructure.GPIO_Pin  = GPIO_Pin_4;
    GPIO_InitStructure.GPIO_Mode = GPIO_Mode_AIN;
    GPIO_Init(GPIOC, &GPIO_InitStructure);

    ADC_DeInit(ADC1);
    ADC_InitStructure.ADC_Mode = ADC_Mode_Independent;
    ADC_InitStructure.ADC_ScanConvMode = DISABLE;
    ADC_InitStructure.ADC_ContinuousConvMode = DISABLE;
    ADC_InitStructure.ADC_ExternalTrigConv = ADC_ExternalTrigConv_None;
    ADC_InitStructure.ADC_DataAlign = ADC_DataAlign_Right;
    ADC_InitStructure.ADC_NbrOfChannel = 1;
    ADC_Init(ADC1, &ADC_InitStructure);

    ADC_RegularChannelConfig(ADC1, ADC_Channel_2, 1, ADC_SampleTime_241Cycles);

    ADC_Cmd(ADC1, ENABLE);

    // ลำดับ calibration ต้องทำก่อนแปลงค่าจริงเสมอ
    ADC_ResetCalibration(ADC1);
    while (ADC_GetResetCalibrationStatus(ADC1));
    ADC_StartCalibration(ADC1);
    while (ADC_GetCalibrationStatus(ADC1));
}

uint16_t ADC_ReadOnce(void)
{
    ADC_SoftwareStartConvCmd(ADC1, ENABLE);
    while (ADC_GetFlagStatus(ADC1, ADC_FLAG_EOC) == RESET);
    return ADC_GetConversionValue(ADC1);
}

2. Scan หลายช่อง + DMA แบบต่อเนื่อง (circular)

#include "ch32v00x_adc.h"
#include "ch32v00x_dma.h"
#include "ch32v00x_gpio.h"
#include "ch32v00x_rcc.h"

#define NUM_CH 3
volatile uint16_t adc_buf[NUM_CH];   // ต้อง volatile เพราะ DMA เขียนโดยไม่ผ่าน CPU

void ADC_DMA_ScanInit(void)
{
    GPIO_InitTypeDef GPIO_InitStructure = {0};
    ADC_InitTypeDef  ADC_InitStructure  = {0};
    DMA_InitTypeDef  DMA_InitStructure  = {0};

    RCC_APB2PeriphClockCmd(RCC_APB2Periph_GPIOD | RCC_APB2Periph_ADC1, ENABLE);
    RCC_AHBPeriphClockCmd(RCC_AHBPeriph_DMA1, ENABLE);
    RCC_ADCCLKConfig(RCC_PCLK2_Div8);

    GPIO_InitStructure.GPIO_Pin  = GPIO_Pin_2 | GPIO_Pin_3 | GPIO_Pin_4;
    GPIO_InitStructure.GPIO_Mode = GPIO_Mode_AIN;
    GPIO_Init(GPIOD, &GPIO_InitStructure);

    // DMA1 Channel1 คือช่องที่ผูกกับ ADC1 ตายตัวบน CH32V003
    DMA_DeInit(DMA1_Channel1);
    DMA_InitStructure.DMA_PeripheralBaseAddr = (uint32_t)&ADC1->RDATAR;
    DMA_InitStructure.DMA_MemoryBaseAddr     = (uint32_t)adc_buf;
    DMA_InitStructure.DMA_DIR                = DMA_DIR_PeripheralSRC;
    DMA_InitStructure.DMA_BufferSize         = NUM_CH;
    DMA_InitStructure.DMA_PeripheralInc      = DMA_PeripheralInc_Disable;
    DMA_InitStructure.DMA_MemoryInc          = DMA_MemoryInc_Enable;
    DMA_InitStructure.DMA_PeripheralDataSize = DMA_PeripheralDataSize_HalfWord;
    DMA_InitStructure.DMA_MemoryDataSize     = DMA_MemoryDataSize_HalfWord;
    DMA_InitStructure.DMA_Mode               = DMA_Mode_Circular;
    DMA_InitStructure.DMA_Priority           = DMA_Priority_High;
    DMA_InitStructure.DMA_M2M                = DMA_M2M_Disable;
    DMA_Init(DMA1_Channel1, &DMA_InitStructure);
    DMA_Cmd(DMA1_Channel1, ENABLE);

    ADC_DeInit(ADC1);
    ADC_InitStructure.ADC_Mode = ADC_Mode_Independent;
    ADC_InitStructure.ADC_ScanConvMode = ENABLE;
    ADC_InitStructure.ADC_ContinuousConvMode = ENABLE;
    ADC_InitStructure.ADC_ExternalTrigConv = ADC_ExternalTrigConv_None;
    ADC_InitStructure.ADC_DataAlign = ADC_DataAlign_Right;
    ADC_InitStructure.ADC_NbrOfChannel = NUM_CH;
    ADC_Init(ADC1, &ADC_InitStructure);

    ADC_RegularChannelConfig(ADC1, ADC_Channel_2, 1, ADC_SampleTime_241Cycles);
    ADC_RegularChannelConfig(ADC1, ADC_Channel_3, 2, ADC_SampleTime_241Cycles);
    ADC_RegularChannelConfig(ADC1, ADC_Channel_4, 3, ADC_SampleTime_241Cycles);

    ADC_DMACmd(ADC1, ENABLE);
    ADC_Cmd(ADC1, ENABLE);

    ADC_ResetCalibration(ADC1);
    while (ADC_GetResetCalibrationStatus(ADC1));
    ADC_StartCalibration(ADC1);
    while (ADC_GetCalibrationStatus(ADC1));

    ADC_SoftwareStartConvCmd(ADC1, ENABLE);   // circular DMA + continuous mode = สแกนวนไม่หยุด
}
// อ่านค่าล่าสุดได้ตลอดเวลาโดยตรงจาก adc_buf[0..2] โดยไม่ต้อง poll ADC เลย

3. Injected Channel แทรกระหว่างลำดับ Regular

#include "ch32v00x_adc.h"

void ADC_InjectedInit(void)
{
    // ... เปิด clock, ตั้ง GPIO analog, ADC_Init ปกติเหมือนตัวอย่างที่ 1 ...

    ADC_InjectedSequencerLengthConfig(ADC1, 1);                      // มี injected 1 ช่อง
    ADC_InjectedChannelConfig(ADC1, ADC_Channel_7, 1, ADC_SampleTime_73Cycles);
    ADC_ExternalTrigInjectedConvConfig(ADC1, ADC_ExternalTrigInjecConv_None);
    ADC_ExternalTrigInjectedConvCmd(ADC1, ENABLE);
    ADC_SetInjectedOffset(ADC1, ADC_InjectedChannel_1, 0);           // ไม่ชดเชย offset

    ADC_Cmd(ADC1, ENABLE);
    ADC_ResetCalibration(ADC1);
    while (ADC_GetResetCalibrationStatus(ADC1));
    ADC_StartCalibration(ADC1);
    while (ADC_GetCalibrationStatus(ADC1));
}

uint16_t ADC_ReadInjected(void)
{
    ADC_SoftwareStartInjectedConvCmd(ADC1, ENABLE);
    while (ADC_GetFlagStatus(ADC1, ADC_FLAG_JEOC) == RESET);
    ADC_ClearFlag(ADC1, ADC_FLAG_JEOC);
    return ADC_GetInjectedConversionValue(ADC1, ADC_InjectedChannel_1);
}

Injected group เหมาะกับกรณีที่ต้องแอบแปลงช่องพิเศษ (เช่น อ่านกระแสฉุกเฉิน) แทรกกลางลำดับ regular โดยไม่รบกวนลำดับ scan หลัก

4. Analog Watchdog แจ้งเตือนเมื่อแรงดันเกินช่วง

#include "ch32v00x_adc.h"
#include "ch32v00x_misc.h"

void ADC_WatchdogInit(void)
{
    // ... ADC_Init + ADC_RegularChannelConfig ปกติ ...

    ADC_AnalogWatchdogThresholdsConfig(ADC1, 900, 100);  // เตือนถ้าค่านอกช่วง 100-900 (10-bit)
    ADC_AnalogWatchdogSingleChannelConfig(ADC1, ADC_Channel_2);
    ADC_AnalogWatchdogCmd(ADC1, ADC_AnalogWatchdog_SingleRegEnable);
    ADC_ITConfig(ADC1, ADC_IT_AWD, ENABLE);

    NVIC_InitTypeDef NVIC_InitStructure = {0};
    NVIC_InitStructure.NVIC_IRQChannel = ADC1_IRQn;
    NVIC_InitStructure.NVIC_IRQChannelPreemptionPriority = 1;
    NVIC_InitStructure.NVIC_IRQChannelSubPriority = 0;
    NVIC_InitStructure.NVIC_IRQChannelCmd = ENABLE;
    NVIC_Init(&NVIC_InitStructure);
}

void ADC1_IRQHandler(void) __attribute__((interrupt("WCH-Interrupt-fast")));
void ADC1_IRQHandler(void)
{
    if (ADC_GetITStatus(ADC1, ADC_IT_AWD) != RESET) {
        // แรงดันหลุดช่วงที่กำหนด — จัดการ event ที่นี่
        ADC_ClearITPendingBit(ADC1, ADC_IT_AWD);
    }
}

ปัญหาที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
ค่าที่อ่านได้เพี้ยน/ไม่แม่นยำตั้งแต่ครั้งแรก ลืมทำลำดับ calibration (ADC_ResetCalibration → รอ → ADC_StartCalibration → รอ) ก่อนเริ่มแปลงค่าจริง ทำ calibration ทุกครั้งหลัง ADC_Cmd(ADC1, ENABLE) และก่อนเรียก ADC_SoftwareStartConvCmd ครั้งแรก
ADC ไม่ทำงานเลย หรือค่าที่ได้คงที่ 0/1023 ตลอด ลืมตั้งขา GPIO เป็น GPIO_Mode_AIN (ยังเป็น digital I/O อยู่) หรือลืมเปิด clock ของ ADC1/GPIO ตรวจว่าตั้ง GPIO_Mode_AIN และเปิด RCC_APB2PeriphClockCmd ทั้งของ GPIO และ ADC1 แล้ว
DMA ไม่ยอมส่งข้อมูลต่อรอบถัดไป (ทำงานแค่ครั้งเดียว) ใช้ DMA_Mode_Normal แทน DMA_Mode_Circular ทำให้ DMA หยุดหลัง buffer เต็มรอบเดียว ใช้ DMA_Mode_Circular คู่กับ ADC_ContinuousConvMode = ENABLE ถ้าต้องการสแกนวนไม่หยุด
ตั้งค่า ADC clock เร็วเกินไปจนอ่านค่าไม่นิ่ง/ผิดสเปก ลืมเรียก RCC_ADCCLKConfig() หรือตั้ง prescaler น้อยไป ทำให้ ADCCLK เกินค่าสูงสุดตาม datasheet เรียก RCC_ADCCLKConfig(RCC_PCLK2_Div8) หรือค่าที่หารแล้วไม่เกิน max ADCCLK ตามสเปกจริงของ MCU
Interrupt (EOC/AWD/JEOC) ยิงรัวไม่หยุด หรือ ISR ค้าง ลืมล้าง flag ด้วย ADC_ClearITPendingBit / ADC_ClearFlag ใน ISR ทุก ISR ต้องเรียก ADC_ClearITPendingBit(ADC1, ADC_IT_xxx) ก่อนออกจากฟังก์ชันเสมอ
Scan หลายช่องแต่ได้ผลลัพธ์แค่ช่องเดียว ลืมตั้ง ADC_ScanConvMode = ENABLE หรือ ADC_NbrOfChannel ไม่ตรงกับจำนวนที่ตั้งด้วย ADC_RegularChannelConfig ตรวจว่า ADC_NbrOfChannel เท่ากับจำนวนครั้งที่เรียก ADC_RegularChannelConfig และเปิด scan mode แล้ว
