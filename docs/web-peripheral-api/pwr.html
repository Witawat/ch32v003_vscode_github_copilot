PWR — Power Control

ระดับสูง
Header: #include "ch32v00x_pwr.h" — นี่คือไลบรารีระดับล่าง (Standard Peripheral Library) เขียนตรงเข้าเรจิสเตอร์ PWR โดยตรง ไม่มีการห่อหุ้ม/ค่า default ให้เหมือน wrapper — ถ้าต้องการ API ที่ใช้ง่ายกว่า (มีฟังก์ชันคำนวณ AWU อัตโนมัติ, ตรวจ wakeup flag ให้) ดูที่ SimplePWR (web-simplehal-api/pwr.html) แทน

ภาพรวม

โมดูล PWR ของ CH32V003 มีเรจิสเตอร์หลัก 3 กลุ่ม: CTLR (ควบคุม PVD และโหมด Standby), CSR (สถานะ PVD), และ AWUCSR/AWUPSC/AWUWR (Auto Wake-Up timer ที่ใช้ปลุกจาก Standby)

ต้องเปิด clock ของ PWR ก่อนใช้งานเสมอ: ฟังก์ชันในไฟล์ ch32v00x_pwr.c เขียนเรจิสเตอร์ PWR->CTLR/PWR->AWUCSR โดยตรง ไม่มีการเปิด peripheral clock ให้เอง ต้องเรียก RCC_APB1PeriphClockCmd(RCC_APB1Periph_PWR, ENABLE); ก่อนเรียกฟังก์ชัน PWR ใดๆ มิฉะนั้นการเขียนเรจิสเตอร์จะไม่มีผล
ไม่มีฟังก์ชัน "Sleep Mode" ในไลบรารีนี้: ต่างจาก wrapper (PWR_EnterSleepMode()ใน SimplePWR) ไฟล์ ch32v00x_pwr.h มีฟังก์ชันเข้าโหมดประหยัดไฟเพียงตัวเดียวคือ PWR_EnterSTANDBYMode() โหมด Sleep (CPU หยุด peripherals ทำงานต่อ) ทำได้ด้วยการเรียก core intrinsic __WFI() หรือ __WFE() ตรงๆ โดยไม่ต้องยุ่งกับ PWR->CTLR เลย (ไม่ตั้งบิต PDDS = CPU จะกลับมาทำงานต่อแบบ sleep ธรรมดา ไม่ใช่ standby)

ตัวอย่างโค้ด

1. Sleep Mode ด้วย WFI (CPU หยุด peripherals ทำงานต่อ)

#include "ch32v00x.h"

int main(void)
{
    SystemCoreClockUpdate();

    // ไม่มีฟังก์ชัน PWR_EnterSleepMode() ในไลบรารีนี้ —
    // sleep ธรรมดาทำได้ด้วยการเรียก __WFI() ตรงๆ โดยไม่แตะ PWR->CTLR
    while (1)
    {
        // ... งานปกติ ...
        __WFI();   // CPU หยุดจนกว่าจะมี interrupt ใดๆ มาปลุก
        // กลับมาทำงานต่อจากบรรทัดนี้ (ไม่ reset)
    }
}

2. ตั้งค่า PVD ตรวจจับแรงดันต่ำพร้อม interrupt

#include "ch32v00x.h"

void PVD_Config(void)
{
    RCC_APB1PeriphClockCmd(RCC_APB1Periph_PWR, ENABLE);

    PWR_PVDLevelConfig(PWR_PVDLevel_3V3);   // threshold 3.3V
    PWR_PVDCmd(ENABLE);
}

int main(void)
{
    SystemCoreClockUpdate();
    PVD_Config();

    while (1)
    {
        if (PWR_GetFlagStatus(PWR_FLAG_PVDO) == SET)
        {
            // VDD ต่ำกว่า 3.3V — เตือนผู้ใช้ / บันทึกข้อมูลก่อนไฟดับ
        }
    }
}

3. เข้า Standby Mode พร้อม Auto Wake-Up (AWU)

#include "ch32v00x.h"

int main(void)
{
    SystemCoreClockUpdate();
    RCC_APB1PeriphClockCmd(RCC_APB1Periph_PWR, ENABLE);

    // ตรวจก่อนว่ารอบนี้ตื่นจาก standby หรือเพิ่งเปิดไฟใหม่ (ต้องเช็คผ่าน RCC ไม่ใช่ PWR)
    if (RCC_GetFlagStatus(RCC_FLAG_LPWRRST) != RESET)
    {
        RCC_ClearFlag();   // ★ ต้องเคลียร์ก่อนเข้า standby รอบถัดไป มิฉะนั้น flag จะค้าง
        // ... โค้ดกรณีตื่นจาก standby ...
    }

    // ตั้ง AWU ให้ปลุกทุกๆ ~1 วินาที (prescaler=4096 -> ~32ms/count, window=0x3F -> ~2s)
    PWR_AWU_SetPrescaler(PWR_AWU_Prescaler_4096);
    PWR_AWU_SetWindowValue(0x1F);
    PWR_AutoWakeUpCmd(ENABLE);

    // ... งานก่อน standby ...

    PWR_EnterSTANDBYMode(PWR_STANDBYEntry_WFI);
    // ไม่มีโค้ดใดทำงานต่อจากบรรทัดนี้ — Standby = รีเซ็ตทั้งชิป, main() เริ่มใหม่ทั้งหมด
}
ต่างจาก Sleep: PWR_EnterSTANDBYMode() ไม่ได้ "คืนค่า" กลับมาที่บรรทัดถัดไปเหมือน __WFI() ธรรมดา — เพราะตั้งบิต PDDS ทำให้ MCU ตัดไฟโดเมน 1.8V เมื่อตื่นขึ้นมาจะเป็นการ รีเซ็ตทั้งชิป (เทียบเท่า power-on ใหม่) โค้ดหลังบรรทัด PWR_EnterSTANDBYMode() จะไม่ถูกรันต่อ

ปัญหาที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
เรียกฟังก์ชัน PWR แล้วไม่มีผลอะไรเลย (เช่น PVD ไม่ทำงาน, AWU ไม่ปลุก) ลืมเปิด peripheral clock — ch32v00x_pwr.c เขียนเรจิสเตอร์ตรงๆ โดยไม่เปิด clock ให้เอง เรียก RCC_APB1PeriphClockCmd(RCC_APB1Periph_PWR, ENABLE); ก่อนใช้ฟังก์ชัน PWR ใดๆ เป็นอย่างแรก
ตื่นจาก Standby แล้วตัวแปร/สถานะเดิมหายหมด Standby mode ตัดไฟโดเมน 1.8V ทั้งหมด — SRAM ไม่ถูก retain เหมือนโหมด Sleep ออกแบบโปรแกรมให้ตรวจ RCC_GetFlagStatus(RCC_FLAG_LPWRRST) ที่ต้น main() แล้วกู้สถานะจาก flash/EEPROM แทนการพึ่ง RAM
ตื่นจาก Standby ครั้งเดียวแล้วกลับเข้า Standby ใหม่ทันที (loop วนไม่หยุด) ไม่ได้เคลียร์ flag รีเซ็ตของ RCC (RCC_FLAG_LPWRRST) — ค้างเป็น SET ข้ามรอบ ทำให้ logic เข้าใจผิดว่ายังอยู่ในสถานะ "เพิ่งตื่น" ตลอด เรียก RCC_ClearFlag() ทันทีหลังตรวจสอบสาเหตุการรีเซ็ตแล้ว
ใช้ PWR_GetFlagStatus() เพื่อเช็คว่าตื่นจาก Standby แล้วได้ผลผิด/ไม่คาดคิด โมดูล PWR ของ CH32V003 มี flag เดียวคือ PWR_FLAG_PVDO เท่านั้น — ไม่มี WUF/SBF แบบ STM32 ใช้ RCC_GetFlagStatus(RCC_FLAG_LPWRRST) จากโมดูล RCC แทนสำหรับตรวจ wakeup จาก standby
ตั้งใจเข้า Sleep mode แต่ MCU กลับรีเซ็ตทั้งชิป (พฤติกรรมเหมือน Standby) เผลอเรียก PWR_EnterSTANDBYMode() แทนที่จะเรียก __WFI() ตรงๆ — ฟังก์ชันนี้ตั้งบิต PDDS เสมอ ไม่มีโหมด "sleep เฉยๆ" ให้เรียกแยก ถ้าต้องการแค่ sleep (CPU หยุดชั่วคราว, กลับมาทำงานต่อ) ให้เรียก __WFI()/__WFE() ตรงๆ โดยไม่ผ่านฟังก์ชัน PWR เลย
PWR_AWU_SetWindowValue() ตั้งค่าแล้วดูเหมือนปลุกเร็ว/ช้ากว่าที่คำนวณ ค่า WindowValue ต้องน้อยกว่า 0x3F (63) ตามที่ระบุใน header — ค่าที่เกินจะถูกมาสก์ทิ้งบางส่วนโดย AWUWR_MASK ตรวจสอบว่า WindowValue < 0x3F เสมอ และคำนวณ timeout จาก prescaler และ LSI 128kHz ให้ตรงก่อนตั้งค่า
