TIM — Timer, PWM, Input Capture, Encoder

ระดับขั้นสูง (Raw SPL)
Header: #include <ch32v00x_tim.h> — Standard Peripheral Library ระดับล่างของ Timer ตั้งค่าผ่าน struct โดยตรง ไม่มี auto-calculate prescaler/period ให้เหมือน wrapper ต้องคำนวณเองจาก system clock
CH32V003 มี Timer แค่ 2 ตัว: TIM1 เป็น Advanced-control timer (มี dead-time, break input, complementary output N — เหมาะกับควบคุมมอเตอร์) อยู่บน APB2 และ TIM2 เป็น General-purpose timer (ไม่มี BDTR/ complementary output) อยู่บน APB1 — ฟีเจอร์ที่ระบุว่า "TIM1 only" ในเอกสารนี้จะไม่มีผลกับ TIM2 (ฮาร์ดแวร์ไม่รองรับ ถึงเรียกฟังก์ชันก็ไม่เกิดอะไรขึ้น)
อยากง่ายกว่านี้? ถ้าแค่ต้องการ interrupt เป็นจังหวะหรือ PWM ธรรมดา ให้ใช้ SimpleTIM หรือ SimplePWM แทน — คำนวณ prescaler/period ให้อัตโนมัติจากความถี่ที่ต้องการ

ผังการทำงาน TIM (SPL)

flowchart TB
    RCC["RCC_APB2/APB1PeriphClockCmd\n(TIM1/TIM2)"] --> TB["TIM_TimeBaseInit\n(prescaler, period, counter mode)"]
    TB --> BASIC["โหมด Update Interrupt\nTIM_ITConfig(TIM_IT_Update)"]
    TB --> PWM["โหมด PWM Output\nTIM_OCxInit(OCMode_PWM1) + TIM_Cmd"]
    TB --> IC["โหมด Input Capture\nTIM_ICInit + TIM_GetCaptureX"]
    PWM --> ADV["TIM1 เท่านั้น:\nTIM_BDTRConfig + TIM_CtrlPWMOutputs(ENABLE)"]
    BASIC --> ISR["TIMx_UP_IRQHandler\nTIM_GetITStatus → TIM_ClearITPendingBit"]

ตัวอย่างโค้ด

1. Timebase Interrupt — สร้าง Update Event ทุก 1ms (TIM2)

#include <ch32v00x.h>

volatile uint32_t msTick = 0;

void TIM2_Init_1ms(void)
{
    RCC_APB1PeriphClockCmd(RCC_APB1Periph_TIM2, ENABLE);

    /* สมมติ SystemCoreClock = 48MHz, ต้องการ tick 1ms
       48MHz / (Prescaler+1) / (Period+1) = 1000Hz */
    TIM_TimeBaseInitTypeDef TIM_TimeBaseStructure = {0};
    TIM_TimeBaseStructure.TIM_Period = 999;          /* นับ 0-999 = 1000 tick */
    TIM_TimeBaseStructure.TIM_Prescaler = 47;         /* 48MHz/48 = 1MHz counter clock */
    TIM_TimeBaseStructure.TIM_ClockDivision = TIM_CKD_DIV1;
    TIM_TimeBaseStructure.TIM_CounterMode = TIM_CounterMode_Up;
    TIM_TimeBaseInit(TIM2, &TIM_TimeBaseStructure);

    TIM_ITConfig(TIM2, TIM_IT_Update, ENABLE);

    NVIC_InitTypeDef NVIC_InitStructure = {0};
    NVIC_InitStructure.NVIC_IRQChannel = TIM2_IRQn;
    NVIC_InitStructure.NVIC_IRQChannelPreemptionPriority = 0;
    NVIC_InitStructure.NVIC_IRQChannelSubPriority = 0;
    NVIC_InitStructure.NVIC_IRQChannelCmd = ENABLE;
    NVIC_Init(&NVIC_InitStructure);

    TIM_Cmd(TIM2, ENABLE);
}

void TIM2_IRQHandler(void) __attribute__((interrupt("WCH-Interrupt-fast")));
void TIM2_IRQHandler(void)
{
    if (TIM_GetITStatus(TIM2, TIM_IT_Update) != RESET) {
        msTick++;
        TIM_ClearITPendingBit(TIM2, TIM_IT_Update);  /* ต้องเคลียร์ ไม่งั้น ISR วนซ้ำ */
    }
}

2. PWM Output บน TIM1 Channel 1 พร้อมปรับ Duty Cycle

#include <ch32v00x.h>

void TIM1_PWM_Init(uint16_t period)
{
    RCC_APB2PeriphClockCmd(RCC_APB2Periph_TIM1 | RCC_APB2Periph_GPIOD, ENABLE);

    GPIO_InitTypeDef GPIO_InitStructure = {0};
    GPIO_InitStructure.GPIO_Pin = GPIO_Pin_2;          /* PD2 = TIM1_CH1 (ตัวอย่าง) */
    GPIO_InitStructure.GPIO_Speed = GPIO_Speed_50MHz;
    GPIO_InitStructure.GPIO_Mode = GPIO_Mode_AF_PP;
    GPIO_Init(GPIOD, &GPIO_InitStructure);

    TIM_TimeBaseInitTypeDef TIM_TimeBaseStructure = {0};
    TIM_TimeBaseStructure.TIM_Period = period;         /* กำหนดความถี่ PWM */
    TIM_TimeBaseStructure.TIM_Prescaler = 0;
    TIM_TimeBaseStructure.TIM_ClockDivision = TIM_CKD_DIV1;
    TIM_TimeBaseStructure.TIM_CounterMode = TIM_CounterMode_Up;
    TIM_TimeBaseInit(TIM1, &TIM_TimeBaseStructure);

    TIM_OCInitTypeDef TIM_OCInitStructure = {0};
    TIM_OCInitStructure.TIM_OCMode = TIM_OCMode_PWM1;
    TIM_OCInitStructure.TIM_OutputState = TIM_OutputState_Enable;
    TIM_OCInitStructure.TIM_Pulse = 0;                 /* duty เริ่มต้น 0% */
    TIM_OCInitStructure.TIM_OCPolarity = TIM_OCPolarity_High;
    TIM_OC1Init(TIM1, &TIM_OCInitStructure);

    TIM_OC1PreloadConfig(TIM1, TIM_OCPreload_Enable);  /* duty เปลี่ยนแบบไม่กระตุก */
    TIM_ARRPreloadConfig(TIM1, ENABLE);

    TIM_CtrlPWMOutputs(TIM1, ENABLE);   /* TIM1 (advanced timer) ต้องเปิดบรรทัดนี้ ไม่งั้นไม่มีสัญญาณออก */
    TIM_Cmd(TIM1, ENABLE);
}

void TIM1_SetDutyPercent(uint16_t period, uint8_t percent)
{
    TIM_SetCompare1(TIM1, (uint32_t)period * percent / 100);
}

3. Input Capture — วัดความกว้างพัลส์ / ความถี่บน Channel 1

#include <ch32v00x.h>

volatile uint16_t capturedWidth = 0;
volatile uint8_t  edgeState = 0;    /* 0 = รอขอบขึ้น, 1 = รอขอบลง */
volatile uint16_t risingValue = 0;

void TIM2_IC_Init(void)
{
    RCC_APB1PeriphClockCmd(RCC_APB1Periph_TIM2, ENABLE);
    /* ... ตั้งขา TIM2_CH1 เป็น GPIO_Mode_IN_FLOATING ก่อนหน้านี้ ... */

    TIM_TimeBaseInitTypeDef TIM_TimeBaseStructure = {0};
    TIM_TimeBaseStructure.TIM_Period = 0xFFFF;
    TIM_TimeBaseStructure.TIM_Prescaler = 47;    /* 48MHz/48 = 1MHz => นับ 1 tick = 1us */
    TIM_TimeBaseStructure.TIM_ClockDivision = TIM_CKD_DIV1;
    TIM_TimeBaseStructure.TIM_CounterMode = TIM_CounterMode_Up;
    TIM_TimeBaseInit(TIM2, &TIM_TimeBaseStructure);

    TIM_ICInitTypeDef TIM_ICInitStructure = {0};
    TIM_ICInitStructure.TIM_Channel = TIM_Channel_1;
    TIM_ICInitStructure.TIM_ICPolarity = TIM_ICPolarity_Rising;
    TIM_ICInitStructure.TIM_ICSelection = TIM_ICSelection_DirectTI;
    TIM_ICInitStructure.TIM_ICPrescaler = TIM_ICPSC_DIV1;
    TIM_ICInitStructure.TIM_ICFilter = 0;
    TIM_ICInit(TIM2, &TIM_ICInitStructure);

    TIM_ITConfig(TIM2, TIM_IT_CC1, ENABLE);
    /* ... NVIC_Init กับ TIM2_IRQn เหมือนตัวอย่างที่ 1 ... */
    TIM_Cmd(TIM2, ENABLE);
}

void TIM2_IRQHandler(void) __attribute__((interrupt("WCH-Interrupt-fast")));
void TIM2_IRQHandler(void)
{
    if (TIM_GetITStatus(TIM2, TIM_IT_CC1) != RESET) {
        uint16_t val = TIM_GetCapture1(TIM2);
        if (edgeState == 0) {
            risingValue = val;
            edgeState = 1;
            TIM_OC1PolarityConfig(TIM2, TIM_OCPolarity_Low);  /* สลับไปจับขอบลงครั้งถัดไป */
        } else {
            capturedWidth = val - risingValue;   /* หน่วย us — ความกว้างพัลส์ HIGH */
            edgeState = 0;
            TIM_OC1PolarityConfig(TIM2, TIM_OCPolarity_High);
        }
        TIM_ClearITPendingBit(TIM2, TIM_IT_CC1);
    }
}

สำหรับวัดทั้งคาบและ duty cycle พร้อมกันในครั้งเดียว ใช้ TIM_PWMIConfig ซึ่งตั้งช่องคู่ (เช่น CH1+CH2) ให้จับขอบตรงข้ามกันอัตโนมัติ แทนการสลับ polarity เองแบบตัวอย่างข้างต้น

ปัญหาที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
PWM บน TIM1 ไม่มีสัญญาณออกเลย TIM1 เป็น advanced-control timer ต้องเปิด Main Output Enable เพิ่มเติมจาก TIM_Cmd เรียก TIM_CtrlPWMOutputs(TIM1, ENABLE) คู่กับ TIM_Cmd(TIM1, ENABLE) เสมอ — ขั้นตอนนี้ ไม่จำเป็นสำหรับ TIM2
เปลี่ยน duty cycle แล้วสัญญาณ PWM กระตุก/มีขอบแหว่ง ลืมเปิด preload buffer ทำให้ค่า CCR/ARR ใหม่มีผลทันทีกลางคาบ แทนที่จะรอ update event รอบถัดไป เปิด TIM_OCxPreloadConfig(TIMx, TIM_OCPreload_Enable) และ TIM_ARRPreloadConfig(TIMx, ENABLE) ก่อนเริ่ม PWM
ใช้ Interrupt แล้ว ISR ถูกเรียกซ้ำไม่หยุด / ค้าง ลืมเคลียร์ pending bit ในตัว ISR ทำให้ NVIC เข้า handler ซ้ำทันทีที่ออกจาก ISR เรียก TIM_ClearITPendingBit(TIMx, TIM_IT_*) เป็นบรรทัดสุดท้ายของ ISR เสมอ
ตั้งค่า TIM_RepetitionCounter หรือ complementary output (OCxN) กับ TIM2 แล้วไม่มีผล ฟีเจอร์เหล่านี้มีเฉพาะใน advanced-control timer (TIM1) เท่านั้น — TIM2 เป็น general-purpose timer ไม่มีเรจิสเตอร์รองรับ ใช้ TIM1 แทนถ้าต้องการ dead-time / complementary output / repetition counter
Input Capture อ่านค่าคาบ/ความกว้างพัลส์ผิดเมื่อ counter overflow ค่า TIM_GetCaptureX เป็น 16-bit วนกลับที่ 0xFFFF ถ้าลบกันตรง ๆ โดยไม่ตรวจ overflow จะได้ค่าติดลบ (wrap) ที่ผิด ลบแบบ unsigned 16-bit ตามธรรมชาติ (uint16_t diff = new - old; จะ wrap ถูกต้องเองถ้าไม่เกิน 1 รอบ) หรือใช้ prescaler ให้ 1 tick มีหน่วยเวลาที่เหมาะกับความถี่สัญญาณ
