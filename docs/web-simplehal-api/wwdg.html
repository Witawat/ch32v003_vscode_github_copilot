SimpleWWDG — Window Watchdog

ระดับสูง
ไลบรารี: SimpleHAL.h — ใช้ ไม่ต้อง include ไฟล์ย่อยเอง ทุกฟังก์ชัน WWDG รวมอยู่ใน SimpleHAL.h แล้ว

ภาพรวม

Window Watchdog (WWDG) ใช้ PCLK1 เป็นแหล่งนาฬิกา ต่างจาก IWDG ที่ใช้ LSI 40kHz — WWDG มี window protection ที่ป้องกันการ refresh เร็วเกินไป นั่นคือต้อง refresh ในช่วงเวลาที่กำหนดเท่านั้น

  • 7-bit counter — 0x40 ถึง 0x7F (64-127)
  • Window — 0x40 ถึง 0x7F — ต้อง refresh เมื่อ counter อยู่ในช่วง window < counter < 0x40
  • 4 prescalers — 1, 2, 4, 8
  • Early wakeup interrupt — เมื่อ counter = 0x40 ก่อนรีเซ็ตที่ 0x3F
Auto-clamp: ถ้า window > counter เริ่มต้น library จะ clamp window ให้เท่ากับ counter โดยอัตโนมัติ — ป้องกันการรีเซ็ตวนซ้ำถาวร

แนวคิด Window Protection

WWDG แตกต่างจาก IWDG ตรงที่มี "หน้าต่าง" (window) — ต้อง refresh ภายในหน้าต่างเท่านั้น:

  • Refresh เร็วเกินไป (counter > window) → รีเซ็ต — จับการทำงานผิดปกติที่ loop เร็วเกิน
  • Refresh ช้าเกินไป (counter < 0x40 → 0x3F) → รีเซ็ต — จับกรณีโค้ดค้าง
  • Refresh ถูกเวลา (0x40 <= counter <= window) → ปกติ
    counter: 0x7F ──────► window ──────► 0x40 ───► 0x3F
                  ^^^^^^^^^^^^^^^^^^^^^^^^        ^^^^^^
                  REFRESH เร็วเกิน → RESET        RESET (T6 bit)

ตารางอ้างอิงฟังก์ชัน

ฟังก์ชันพารามิเตอร์คืนค่าคำอธิบายใช้ร่วมกับ
WWDG_SimpleInit(counter, window) counter: uint8_t (0x40-0x7F)
window: uint8_t (0x40-0x7F)		 void เริ่ม WWDG ด้วย prescaler=8 (ค่าเริ่มต้น) 1 ใน 3 ทาง init ที่ใช้แทนกัน (SimpleInit / Init / InitWithInterrupt) — เลือกทางเดียว ห้ามเรียกซ้อนกัน; ตามด้วย WWDG_Refresh(counter) ในลูปเสมอ
WWDG_Refresh(counter) counter: uint8_t (0x40-0x7F) void ต้อง refresh counter ในช่วง window — เร็วหรือช้าเกิน = รีเซ็ต ต้องเรียกหลังหนึ่งใน WWDG_SimpleInit()/WWDG_Init()/WWDG_InitWithInterrupt() — ค่า counter ที่ส่งควรตรงกับค่าที่ใช้ตอน init เสมอ
WWDG_Init(counter, window, prescaler) counter: uint8_t
window: uint8_t
prescaler: WWDG_PRESCALER_*		 void ตั้งค่า WWDG เต็มรูปแบบ ใช้แทน WWDG_SimpleInit() เมื่อต้องกำหนด prescaler เอง — ค่า counter/window ควรมาจาก WWDG_CalcTimeout() (คำนวณย้อนกลับ) เพื่อให้ได้ timeout ที่ต้องการ
WWDG_InitWithInterrupt(counter, window, prescaler) เหมือน WWDG_Init void เริ่ม WWDG + เปิด early wakeup interrupt (EWI) เมื่อ counter=0x40 ใช้แทน WWDG_Init() (ไม่ใช่เรียกคู่กัน) เมื่อต้องการ early-warning callback — ต้องเรียก WWDG_SetCallback() ก่อน ฟังก์ชันนี้เสมอ มิฉะนั้น interrupt แรกจะไม่มี callback ให้เรียก
WWDG_SetCallback(callback) callback: void (*)(void) void ตั้ง callback เมื่อ counter=0x40 (เตือนก่อนรีเซ็ต) ต้องเรียกก่อน WWDG_InitWithInterrupt() — ใช้ไม่ได้ผลถ้าใช้คู่กับ WWDG_SimpleInit()/WWDG_Init() เพราะสองทางนั้นไม่เปิด interrupt
WWDG_CalcTimeout(prescaler, counter) prescaler: WWDG_PRESCALER_*
counter: uint8_t		 uint32_t (µs) คำนวณ timeout จริง หน่วย µs ใช้ก่อนเรียก WWDG_Init()/InitWithInterrupt() เพื่อตรวจว่าค่า counter/prescaler ที่เลือกให้ timeout ตรงตามต้องการหรือไม่ (ดูตาราง Timeout range ด้านบน)
WWDG_GetInterruptFlag() — uint8_t (1/0) ตรวจสอบ early wakeup interrupt flag ใช้ได้เฉพาะเมื่อ init ด้วย WWDG_InitWithInterrupt() — เรียกคู่กับ WWDG_ClearInterruptFlag() เพื่อเคลียร์หลังตรวจ
WWDG_ClearInterruptFlag() — void เคลียร์ early wakeup interrupt flag เรียกหลัง WWDG_GetInterruptFlag() คืน 1 หรือในตัว callback ที่ตั้งด้วย WWDG_SetCallback()
WWDG_Disable() — void ⚠️ รีเซ็ต configuration registers เท่านั้น — WDGA bit หยุดไม่ได้! เรียกได้ทุกเมื่อหลัง init แต่ ไม่ได้ทำให้ WWDG_Refresh() เลิกจำเป็น — ยังต้อง refresh ต่อไปเหมือนเดิม (ดู CRITICAL ด้านล่าง)
CRITICAL: เมื่อ WDGA bit ถูกเซ็ตแล้ว ไม่สามารถหยุด WWDG ได้ ด้วยซอฟต์แวร์ — WWDG_Disable() แค่เคลียร์ config register ไม่ได้หยุด watchdog จริง

ตัวอย่างโค้ด

Basic WWDG — counter 100, window 80

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include <SimpleHAL.h>

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();

    pinMode(PC0, PIN_MODE_OUTPUT);

    // counter=100 (0x64), window=80 (0x50)
    // ต้อง refresh เมื่อ counter อยู่ในช่วง 0x40-0x50
    WWDG_SimpleInit(100, 80);

    while (1) {
        digitalToggle(PC0);
        Delay_Ms(5);       // ทำงาน
        WWDG_Refresh(100); // refresh — ต้องไม่เร็วหรือช้าเกิน
    }
}

WWDG พร้อม Early Warning Interrupt

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include <SimpleHAL.h>

volatile uint8_t wwdg_warning = 0;

void OnWWDG_Warning(void) {
    wwdg_warning = 1;  // counter = 0x40 → ทำ save ข้อมูลก่อนรีเซ็ต
}

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();

    pinMode(PC0, PIN_MODE_OUTPUT);

    WWDG_SetCallback(OnWWDG_Warning);
    WWDG_InitWithInterrupt(127, 100, WWDG_PRESCALER_8);

    while (1) {
        digitalToggle(PC0);
        Delay_Ms(5);

        if (wwdg_warning) {
            wwdg_warning = 0;
            // บันทึกข้อมูลสำคัญก่อนระบบรีเซ็ต
        }

        WWDG_Refresh(127);
    }
}
