SimpleDMA — ถ่ายโอนข้อมูลความเร็วสูง สูง

ภาพรวม

SimpleDMA ควบคุม DMA Controller ของ CH32V003 จำนวน 7 channels ถ่ายโอนข้อมูลโดยไม่ใช้ CPU รองรับโหมด M2M (Memory-to-Memory), P2M (Peripheral-to-Memory), M2P (Memory-to-Peripheral), circular mode สำหรับ ring buffer, callback เมื่อโอนเสร็จ/ครึ่งทาง/เกิด error และผสานการทำงานกับอุปกรณ์ต่อพ่วงโดยตรง

คุณสมบัติรายละเอียด
Channels7 channels (CH1–CH7)
DirectionM2M, P2M, M2P
ModeNormal + Circular
Data size8-bit / 16-bit / 32-bit
Peripheral integrationADC, USART, SPI, I2C, TIM1, TIM2
Header#include "SimpleDMA.h"

DMA Channels

DMA มี 7 channels — channel ที่ มีเลขต่ำกว่า จะมี priority สูงกว่า (CH1 สูงสุด → CH7 ต่ำสุด)

ChannelPriorityPeripheral Mapping (Fixed)
CH1สูงสุดADC1, TIM1_CH1, TIM1_CH4, TIM2_CH3, USART1_TX
CH2ADC1, TIM1_CH2, TIM1_TRG, USART1_RX
CH3ADC1, TIM1_CH3, TIM2_UP, USART1_TX
CH4ADC1, TIM1_CH4, TIM2_CH1, USART1_RX, I2C1_TX
CH5ADC1, TIM1_UP, TIM2_CH2, TIM2_CH4, USART1_TX, I2C1_RX
CH6ADC1, TIM1_CH3, TIM1_UP, TIM1_COM, SPI1_RX
CH7ต่ำสุดADC1, TIM2_CH1, TIM2_CH2, TIM2_CH4, SPI1_TX
ข้อควรระวัง: peripheral-to-DMA-channel mapping เป็น fixed mapping — ไม่สามารถเลือก channel เองได้อิสระ ต้องใช้ channel ที่ตรงกับ peripheral นั้น ๆ เช่น I2C1_TX ต้องใช้ CH4 เท่านั้น, I2C1_RX ต้องใช้ CH5 เท่านั้น

ตารางฟังก์ชัน — Basic

ฟังก์ชันคืนค่าคำอธิบายใช้ร่วมกับ
DMA_SimpleInit(config)voidเริ่มต้น DMA ตาม config structต้องเรียกหลัง peripheral (ADC/USART/SPI/I2C/TIM) ถูก init/enable แล้วเท่านั้น — ถ้าจะใช้ callback ให้เรียก DMA_SetXxxCallback() ก่อน DMA_Start() เพื่อไม่พลาดรอบแรก
DMA_Start(channel)voidเริ่มการถ่ายโอนเรียกหลัง DMA_SimpleInit() และหลังตั้ง callback (ถ้าใช้) แล้ว
DMA_Stop(channel)voidหยุดการถ่ายโอนจำเป็นสำหรับ circular mode เพราะ DMA จะวนต่อไปเรื่อยๆ ไม่หยุดเอง — เรียก DMA_Start() ซ้ำเพื่อเริ่มใหม่
DMA_GetStatus(channel)DMA_Statusตรวจสอบสถานะ (IDLE / BUSY / COMPLETE / ERROR)ใช้ poll แทน callback เมื่อไม่ต้องการ interrupt — เรียกหลัง DMA_Start()
DMA_WaitComplete(channel, timeout_ms)uint8_tรอจนถ่ายโอนเสร็จ — 1=สำเร็จ 0=timeoutใช้แทนการ poll DMA_GetStatus() เอง — เป็นฟังก์ชันที่ DMA_MemCopy()/DMA_MemSet() เรียกใช้ภายในเพื่อทำ blocking
DMA_Reset(channel)voidรีเซ็ต channel (soft reset)เรียกก่อน DMA_SimpleInit() ใหม่ เมื่อต้องการเปลี่ยน config ของ channel เดิมทั้งหมด
DMA_GetRemainingCount(channel)uint16_tจำนวน data unit ที่ยังเหลืออยู่ใช้ตรวจความคืบหน้าระหว่าง DMA_Start() กับ callback/DMA_GetStatus() คืน COMPLETE
DMA_EnableInterrupt(channel, enable)voidเปิด/ปิด interrupt สำหรับ channel นั้นต้องเปิด (enable=1) ก่อน callback (DMA_SetXxxCallback()) จะถูกเรียกได้จริง — ฟังก์ชัน DMA_SetXxxCallback ส่วนใหญ่เปิดให้อัตโนมัติแล้ว เรียกเองเฉพาะกรณีปิดไปก่อนหน้า
DMA_GetChannelBase(channel)DMA_Channel_TypeDef*แปลง DMA_Channel เป็น pointer register จริงใช้ในระดับ low-level เมื่อต้องเข้าถึง DMA register โดยตรง (เช่น MADDR, CNTR)
DMA_GetChannelIRQn(channel)IRQn_Typeแปลง DMA_Channel เป็นหมายเลข IRQใช้เมื่อต้องเรียก NVIC_EnableIRQ()/NVIC_SetPriority() เองสำหรับ channel นั้น

Callbacks

ฟังก์ชัน callback ถูกเรียกจากภายใน ISR — ควรทำให้สั้นที่สุด:

ฟังก์ชันคำอธิบายใช้ร่วมกับ
DMA_SetTransferCompleteCallback(ch, callback)เรียกเมื่อถ่ายโอนครบทุก unitต้องเรียกก่อน DMA_Start()/DMA_MemCopyAsync() เสมอ — ตั้งหลังเริ่มโอนแล้วอาจพลาด callback ของรอบแรกถ้าโอนเสร็จเร็ว
DMA_SetErrorCallback(ch, callback)เรียกเมื่อเกิด DMA errorตั้งก่อน DMA_Start() เช่นเดียวกัน — ใช้คู่กับ DMA_GetStatus() เพื่อ debug เมื่อ error เกิดขึ้น
DMA_SetHalfTransferCallback(ch, callback)เรียกเมื่อถ่ายโอนได้ครึ่งทาง — เหมาะกับ ping-pong bufferตั้งก่อน DMA_Start() — ใช้คู่กับ circular mode (mode = DMA_MODE_CIRCULAR) เพื่อประมวลผลครึ่ง buffer ที่โอนเสร็จแล้วขณะอีกครึ่งกำลังโอนอยู่
Ping-Pong Buffer: ใช้ DMA_SetHalfTransferCallback เพื่อประมวลผลครึ่งแรกของ buffer ขณะที่ DMA ยังคงเติมครึ่งหลังต่อ — เพิ่ม throughput ได้โดยที่ไม่ต้องหยุด DMA

ข้อผิดพลาดที่พบบ่อย

ลืม init peripheral ก่อน DMA — ต้อง config และ enable peripheral (ADC, USART, SPI, I2C, TIM) ก่อนเรียก DMA_SimpleInit เสมอ

DMA ต้องการให้ peripheral พร้อมก่อนเริ่มถ่ายโอน — ถ้า peripheral ยังไม่ enable การถ่ายโอนจะไม่เกิดขึ้น

Buffer size ไม่ตรงกับ data size — ระวัง mismatch ระหว่าง data_size (8/16/32-bit) กับจำนวน element ใน buffer

เช่น ตั้งค่า data_size = 16-bit แต่ buffer_size = 100 → จะถ่ายโอน 100 half-words = 200 bytes — ต้องประกาศ buffer เป็น uint16_t buf[100]

Channel conflict กับ fixed peripheral mappings — ไม่สามารถเลือก channel ได้ตามใจ

เช่น I2C1_TX ต้องใช้ CH4, I2C1_RX ต้องใช้ CH5 — ใช้ channel อื่นจะไม่ทำงาน

Circular mode ไม่หยุดเอง — ถ้าใช้ circular mode ต้องเรียก DMA_Stop() เพื่อหยุดเอง เพราะ DMA จะวนต่อไปเรื่อย ๆ

ตัวอย่างโค้ด

Memory Copy — Blocking

#include "SimpleHAL.h"

uint8_t src[64] = {0xAA, 0xBB, 0xCC, /* ... */};
uint8_t dst[64];

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();

    DMA_MemCopy(dst, src, 64);
    // dst contains copy of src — complete when this line returns

    while (1) { }
}

Memory Copy — Async with Callback

#include "SimpleHAL.h"

uint8_t src[256], dst[256];
volatile uint8_t done = 0;

void transfer_done(DMA_Channel channel) {
    done = 1;
}

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();

    DMA_MemCopyAsync(DMA_CH1, dst, src, 256);
    DMA_SetTransferCompleteCallback(DMA_CH1, transfer_done);

    while (!done);  // รอ callback
    // dst เสร็จแล้ว

    while (1) { }
}

ADC Circular DMA — อ่านต่อเนื่อง

#include "SimpleHAL.h"

#define ADC_BUF_SIZE  64
uint16_t adc_buffer[ADC_BUF_SIZE];

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();

    ADC_Init(ADC_CH_PA1);  // init peripheral ก่อน
    DMA_ADC_Init(DMA_CH1, adc_buffer, ADC_BUF_SIZE, 1);  // circular=1

    while (1) {
        uint16_t avg = DMA_analogReadAverage(adc_buffer, ADC_BUF_SIZE);
        USART_Print("Avg: "); USART_PrintNum(avg); USART_Print("\r\n");
        Delay_Ms(100);
    }
}

USART DMA TX — ส่งข้อมูลโดยไม่ block CPU

#include "SimpleHAL.h"

uint8_t tx_data[] = "Hello via DMA!\r\n";

void tx_complete(DMA_Channel channel) {
    // ส่งเสร็จแล้ว — ทำงานอย่างอื่นต่อได้
}

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();

    USART_SimpleInit(USART_BAUD_115200, USART_PIN_DEFAULT);
    DMA_USART_InitTx(DMA_CH1, tx_data, sizeof(tx_data) - 1);
    DMA_SetTransferCompleteCallback(DMA_CH1, tx_complete);
    DMA_USART_Transmit(DMA_CH1, tx_data, sizeof(tx_data) - 1);

    // CPU ว่างทำงานอื่น — DMA ส่งข้อมูลเอง
    while (1) { }
}

PWM Waveform Generator — DMA Update Duty

#include "SimpleHAL.h"

// Look-Up Table สำหรับ sine wave (32 steps)
uint16_t sine_lut[32] = {
    32767, 39148, 44768, 49105, 51725, 52365, 50967, 47687,
    42877, 37006, 30595, 24189, 18358, 13589, 10278, 8741,
    9231, 11963, 16990, 24189, 33252, 43806, 55337, 67252,
    78907, 89690, 99002, 106336, 111326, 113754, 113563, 110858
};

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();

    PWM_Init(TIM1, PWM_CH1, PA1, 1000, 50);  // init TIM ก่อน DMA
    uint32_t ccr_addr = DMA_TIM_GetCCRAddress(TIM1, 1);

    DMA_Config_t dma_cfg = {
        .channel = DMA_CH1,
        .periph_addr = ccr_addr,
        .mem_addr = (uint32_t)sine_lut,
        .buffer_size = 32,
        .mode = DMA_MODE_CIRCULAR,
        .direction = DMA_DIR_MEM_TO_PERIPH,
        // ...
    };
    DMA_SimpleInit(&dma_cfg);
    DMA_Start(DMA_CH1);

    while (1) { }  // PWM output follows sine LUT automatically
}
