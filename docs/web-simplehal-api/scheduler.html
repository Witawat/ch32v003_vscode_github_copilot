SimpleScheduler — Cooperative Task Scheduler

ระดับผู้เชี่ยวชาญ
ไม่ใช่ส่วนหนึ่งของ SimpleHAL.h: ต่างจากโมดูลอื่นในเอกสารนี้ — SimpleScheduler อยู่ที่ User/Lib/SimpleScheduler/ (ไม่ใช่ User/SimpleHAL/) ต้อง #include "SimpleScheduler.h" เอง และเพิ่มไฟล์ SimpleScheduler.c เข้าโปรเจกต์ (เพิ่มใน Makefile/IDE) ก่อนคอมไพล์

ภาพรวม

SimpleScheduler เป็นตัวจัดตารางงานแบบ cooperative (non-preemptive) round-robin — ใช้ทำงานหลายๆ งานบน single-core โดยไม่ต้องใช้ RTOS แต่ละ task มี interval (ms) เป็นอิสระของตัวเอง Scheduler จะเรียก task เมื่อถึงเวลาที่กำหนดโดยใช้ Get_CurrentMs() เป็น time base

  • สูงสุด 8 tasks — ปรับได้ผ่าน #define SCHEDULER_MAX_TASKS ก่อน include
  • แต่ละ task มี interval เป็นอิสระ — ตั้งเวลาแยกกันได้
  • Enable/Disable ได้ — หยุด task ชั่วคราวโดยไม่ต้อง remove
  • เปลี่ยน interval ขณะรันได้ — ไม่ต้อง remove แล้ว add ใหม่
Cooperative ไม่ใช่ preemptive: Task ต้องไม่บล็อกนานเกิน interval ของตัวเอง (ห้ามใช้ Delay_Ms ยาวๆ หรือ loop รอนานในฟังก์ชัน task) เพราะจะทำให้ task อื่นทั้งหมดช้าตามไปด้วย — task ต้อง return โดยเร็วเสมอ

ตารางอ้างอิงฟังก์ชัน

ฟังก์ชันพารามิเตอร์คืนค่าคำอธิบายใช้ร่วมกับ
Scheduler_Init() ไม่มี void เคลียร์ตาราง task ทั้งหมด ต้องเรียกก่อน Scheduler_AddTask() ทุกครั้ง — ต้องมี Timer_Init() เรียกไปก่อนแล้วด้วย เพราะ scheduler ใช้ Get_CurrentMs() เป็น time base
Scheduler_AddTask(func, interval_ms) func: void (*)(void)
interval_ms: uint32_t (ต้อง > 0)		 Scheduler_Status เพิ่ม task ใหม่เข้าตาราง คืน SCHEDULER_ERROR_PARAM ถ้า func=NULL หรือ interval_ms=0; คืน SCHEDULER_ERROR_FULL ถ้าเต็ม 8 tasks — เช็ค return value ก่อนใช้งานจริง
Scheduler_Run() ไม่มี void ตรวจสอบและเรียก task ที่ถึงเวลา เรียกซ้ำใน while(1) ของ main loop — ห้ามใส่ Delay_Ms ใน loop เดียวกัน เพราะจะทำให้ทุก task ถูก delay ไปด้วย
Scheduler_RemoveTask(id) id: uint8_t (index ตอน add) Scheduler_Status ลบ task ออกจากตาราง — index ของ task ที่เหลือจะเลื่อนขึ้นมาแทนที่ คืน SCHEDULER_ERROR_PARAM ถ้า id เกินจำนวน task ปัจจุบัน — ระวัง: id ของ task ตัวถัดไปจะเปลี่ยนหลัง remove เพราะ array เลื่อนขึ้นมาเติมช่องว่าง
Scheduler_SetInterval(id, interval_ms) id: uint8_t
interval_ms: uint32_t		 Scheduler_Status เปลี่ยน interval ของ task ที่รันอยู่แล้วโดยไม่ต้อง remove/add ใหม่ คืน SCHEDULER_ERROR_PARAM ถ้า id ไม่มีอยู่จริง
Scheduler_EnableTask(id) id: uint8_t Scheduler_Status เปิดใช้งาน task ที่เคยถูก disable — รีเซ็ต last_run_ms เป็นเวลาปัจจุบันด้วย (นับ interval ใหม่) ใช้คู่กับ Scheduler_DisableTask() เพื่อหยุด/เริ่ม task ชั่วคราวโดยไม่เสีย slot
Scheduler_DisableTask(id) id: uint8_t Scheduler_Status ปิดใช้งาน task ชั่วคราว — Scheduler_Run() จะข้าม task นี้ไป task ยังอยู่ในตาราง (ไม่เสีย slot จาก 8 ตัว) ต่างจาก RemoveTask() ที่ลบทิ้งจริง
Scheduler_GetTaskCount() ไม่มี uint8_t จำนวน task ทั้งหมดที่ add ไว้ (รวมทั้งที่ disable) ใช้เช็คว่าเหลือ slot ว่างกี่ตัวก่อนเรียก AddTask() เพิ่ม (สูงสุด SCHEDULER_MAX_TASKS = 8)

Scheduler_Status (ค่าคืนกลับ)

ค่าคงที่ค่าความหมาย
SCHEDULER_OK0สำเร็จ
SCHEDULER_ERROR_FULL1ตารางเต็ม (ครบ SCHEDULER_MAX_TASKS แล้ว)
SCHEDULER_ERROR_PARAM2พารามิเตอร์ผิด (func=NULL, interval=0, หรือ id ไม่มีอยู่จริง)
SCHEDULER_ERROR_NOTASK3ไม่มี task ในตาราง

ปรับจำนวน Task สูงสุด

ค่า default คือ 8 tasks สามารถเปลี่ยนได้โดย #define ก่อน #include:

#define SCHEDULER_MAX_TASKS  4
#include "SimpleScheduler.h"
ข้อควรระวัง: CH32V003 มี RAM เพียง 2KB — แต่ละ task ใช้ sizeof(Scheduler_Task) ≈ 13-16 byte เพิ่มจำนวน task มากเกินไปอาจกิน RAM โดยไม่จำเป็นถ้าไม่ได้ใช้ครบ

ข้อผิดพลาดที่พบบ่อย

Task บล็อกนานเกิน interval — เช่นใช้ Delay_Ms(2000) ในฟังก์ชัน task ที่ตั้ง interval 500ms

เพราะเป็น cooperative scheduler ไม่ preempt — task ที่บล็อกนานจะทำให้ ทุก task อื่นค้างตามไปด้วย ต้องออกแบบ task ให้ return เร็วเสมอ ใช้ non-blocking state machine แทนถ้าต้องรอ

ลืมเรียก Timer_Init() ก่อน Scheduler_Init() — Scheduler ใช้ Get_CurrentMs() เป็น time base ซึ่งต้องพึ่ง Timer ที่ init แล้ว

ถ้าลืม interval ทั้งหมดจะคำนวณผิดตั้งแต่ต้น (เวลาไม่เดินหรือเดินผิด)

ไม่เช็ค return value ของ Scheduler_AddTask() — ถ้าตารางเต็ม (เกิน 8 tasks) ฟังก์ชันจะคืน SCHEDULER_ERROR_FULL เงียบๆ โดย task นั้นจะไม่ถูกเพิ่มเข้าไปเลย

ควรเช็คค่าที่คืนกลับเสมอ โดยเฉพาะเวลา add หลาย task ติดกัน

ใช้ id ผิดหลัง RemoveTask() — Scheduler_RemoveTask() เลื่อน array ทำให้ id ของ task ตัวถัดไปเปลี่ยน

ถ้าเก็บ id ของ task อื่นไว้ใช้ภายหลัง ต้องเรียก RemoveTask() ด้วยความระมัดระวัง หรือหลีกเลี่ยงการ remove ถ้าใช้ DisableTask() แทนได้

ตัวอย่างโค้ด

Multi-Task พื้นฐาน (LED + Sensor)

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include <SimpleHAL.h>
#include "SimpleScheduler.h"

void led_task(void) {
    digitalToggle(PC0);  // กระพริบทุก 500ms
}

void sensor_task(void) {
    // อ่านเซนเซอร์ทุก 1000ms — ต้องไม่บล็อกนาน
    uint16_t val = analogRead(PA1);
    (void)val;
}

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();               // ต้อง init ก่อน Scheduler
    pinMode(PC0, PIN_MODE_OUTPUT);

    Scheduler_Init();
    Scheduler_AddTask(led_task, 500);
    Scheduler_AddTask(sensor_task, 1000);

    while (1) {
        Scheduler_Run();        // ห้ามใส่ Delay_Ms ใน loop นี้
    }
}

Enable/Disable Task ขณะรัน

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include <SimpleHAL.h>
#include "SimpleScheduler.h"

void blink_task(void) { digitalToggle(PC0); }

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    pinMode(PC0, PIN_MODE_OUTPUT);
    pinMode(PC1, PIN_MODE_INPUT_PULLUP);  // ปุ่มเปิด/ปิดกระพริบ

    Scheduler_Init();
    Scheduler_Status st = Scheduler_AddTask(blink_task, 300);
    if (st != SCHEDULER_OK) while (1);  // เช็ค error เสมอ

    uint8_t enabled = 1;
    while (1) {
        Scheduler_Run();

        if (digitalRead(PC1) == LOW) {
            if (enabled) { Scheduler_DisableTask(0); enabled = 0; }
            else         { Scheduler_EnableTask(0);  enabled = 1; }
            Delay_Ms(200);  // กันปุ่มเด้ง — ใช้นอก scheduler loop เป็นครั้งคราวได้
        }
    }
}
