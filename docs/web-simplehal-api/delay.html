SimpleDelay — การหน่วงเวลา

ระดับเริ่มต้น
ไลบรารี: SimpleHAL.h — SimpleDelay ใช้ SysTick timer เป็นฐานเวลา 1ms เริ่มต้นอัตโนมัติผ่าน constructor แบบ singleton

ภาพรวม

SimpleDelay เป็นระบบหน่วงเวลาที่ทำงานบน SysTick interrupt ทุก 1ms — แม่นยำและไม่เปลือง hardware timer ตัวอื่น รองรับทั้งแบบ blocking (หยุดรอ) และ non-blocking (ทำงานต่อไปได้)

  • Blocking: Delay_Ms(), Delay_Us() — หยุด CPU จนครบเวลา
  • Non-blocking: Start_Timer() + Is_Timer_Expired() — ทำงานอื่นต่อได้ระหว่างรอ
  • Timestamp: Get_CurrentMs(), Get_CurrentUs() — จับเวลาผ่านไป
  • Timer_t: struct สำหรับสร้าง software timer หลายตัวพร้อมกัน
สำคัญ: ต้องเรียก Timer_Init() หลัง SystemCoreClockUpdate() ใน main() — ถ้าลืม Delay_Ms() และฟังก์ชันทั้งหมดจะคืนค่า 0 ทันที ไม่มีการหน่วงเวลา

ตารางอ้างอิงฟังก์ชัน

ฟังก์ชันพารามิเตอร์คืนค่าคำอธิบายใช้ร่วมกับ
Timer_Init() — void เริ่มต้น SysTick interrupt 1ms ต้องเรียกใน main() ก่อนใช้ทุกฟังก์ชัน
Delay_Ms(n) n: มิลลิวินาที (uint32_t) void หน่วงเวลาแบบ blocking หน่วย ms ใช้ใน main loop เท่านั้น, ห้ามใช้ใน ISR
Delay_Us(n) n: ไมโครวินาที (uint32_t) void หน่วงเวลาแบบ blocking หน่วย µs ใช้สำหรับ pulse สั้นๆ (เช่น ultrasonic trigger 10µs)
Get_CurrentMs() — uint32_t คืนค่าเวลาที่ผ่านไปตั้งแต่เริ่ม หน่วย ms ใช้กับ IS_TIMEOUT หรือ ELAPSED_TIME
Get_CurrentUs() — uint32_t คืนค่าเวลาที่ผ่านไปตั้งแต่เริ่ม หน่วย µs ใช้สำหรับวัดช่วงเวลาสั้น
Get_TickMicros() — uint32_t คืนค่า µs ภายใน tick ปัจจุบัน (0-999) รวมกับ Get_CurrentMs() เพื่อความละเอียดสูง
Start_Timer(timer, ms, repeat) timer: Timer_t*
ms: มิลลิวินาที
repeat: 0=ครั้งเดียว, 1=วนซ้ำ		 void เริ่ม non-blocking timer ใช้กับ Is_Timer_Expired() ใน main loop
Is_Timer_Expired(timer) timer: Timer_t* uint8_t ตรวจสอบ timer หมดเวลา — 1=หมด, 0=ยัง ใช้ร่วมกับ Start_Timer, เรียกใน main loop บ่อยๆ
Reset_Timer(timer, repeat) timer: Timer_t*
repeat: 0/1		 void รีเซ็ต timer เริ่มนับใหม่ ใช้กับ timer ที่ start แล้ว
Stop_Timer(timer) timer: Timer_t* void หยุด timer Is_Timer_Expired จะคืน 0 หลังจากเรียก
ELAPSED_TIME(start, current) start: เวลาเริ่มต้น (ms)
current: เวลาปัจจุบัน (ms)		 uint32_t (macro) คำนวณเวลาที่ผ่านไป ปลอดภัย overflow ใช้กับ Get_CurrentMs() แทนการลบตรง
IS_TIMEOUT(start, timeout) start: เวลาเริ่มต้น (ms)
timeout: ระยะหมดเวลา (ms)		 uint8_t (macro) ตรวจสอบ timeout ปลอดภัย overflow — 1=หมด, 0=ยัง ทางเลือกแทน Timer_t สำหรับเคสง่ายๆ

โครงสร้าง Timer_t

Timer_t เป็น struct สำหรับสร้าง software timer ที่ทำงานแบบ non-blocking — ใช้ timer หลายตัวพร้อมกันได้ไม่มีข้อจำกัด (ใช้ RAM ประมาณ 16 ไบต์ต่อ timer)

typedef struct {
    uint32_t start_time;  // เวลาเริ่มนับ (ms)
    uint32_t duration;    // ระยะเวลาที่ตั้ง (ms)
    uint8_t  active;      // 1=กำลังนับ, 0=หยุด
    uint8_t  repeat;      // 1=วนซ้ำอัตโนมัติ, 0=ครั้งเดียว
} Timer_t;
ฟิลด์ชนิดคำอธิบาย
start_timeuint32_ttimestamp เริ่มนับ — ตั้งโดย Start_Timer()
durationuint32_tระยะเวลาเป็น ms — กำหนดตอน Start_Timer(timer, ms, repeat)
activeuint8_tสถานะ — Start_Timer ตั้งเป็น 1, Stop_Timer ตั้งเป็น 0
repeatuint8_t1 = รีเซ็ตอัตโนมัติเมื่อหมดเวลา, 0 = หยุดหลังหมด
 
Timer_t ledTimer;
Timer_t btnTimer;

Start_Timer(&ledTimer, 500, 1);   // repeat ทุก 500ms
Start_Timer(&btnTimer, 50, 0);    // ครั้งเดียว 50ms (debounce)

Blocking vs Non-Blocking

ลักษณะBlocking (Delay_Ms)Non-blocking (Timer_t)
CPU ระหว่างรอหยุดนิ่ง — ทำอะไรไม่ได้ทำงานอย่างอื่นต่อได้
ใช้หลายงานพร้อมกันไม่ได้ — ต้องทำทีละงานได้ — ใช้หลาย Timer_t
ความซับซ้อนของโค้ดง่าย — 2 บรรทัดปานกลาง — ต้องเช็ค Is_Timer_Expired
ใช้ใน ISRห้ามเด็ดขาดใช้ได้ (แต่เช็ค status เท่านั้น)
ความแม่นยำ±1msขึ้นกับความถี่เรียกเช็ค
แนะนำ: สำหรับ LED กระพริบตัวเดียว Delay_Ms ก็พอ — สำหรับโปรแกรมที่ต้องทำหลายอย่างพร้อมกัน (LED + ปุ่ม + UART) ใช้ Timer_t non-blocking ดีกว่า

ตัวอย่างโค้ด

Blocking Blink (LED กระพริบ 1Hz)

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include <SimpleHAL.h>

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();                        // ★ ต้องเรียกก่อน Delay
    pinMode(PC0, PIN_MODE_OUTPUT);

    while (1) {
        digitalWrite(PC0, HIGH);
        Delay_Ms(500);                   // CPU หยุด 500ms
        digitalWrite(PC0, LOW);
        Delay_Ms(500);
    }
}

Non-Blocking Blink (LED กระพริบแบบไม่ง่วง)

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include <SimpleHAL.h>

Timer_t ledTimer;

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    pinMode(PC0, PIN_MODE_OUTPUT);
    Start_Timer(&ledTimer, 500, 1);      // repeat ทุก 500ms

    while (1) {
        if (Is_Timer_Expired(&ledTimer)) {
            digitalToggle(PC0);           // เปลี่ยนสถานะเมื่อ timer หมด
        }
        // ทำงานอย่างอื่นที่นี่ได้ — CPU ไม่ได้หยุด!
    }
}

Multi-Timer (LED 2 ความถี่ + Debounce ปุ่ม)

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include <SimpleHAL.h>

Timer_t ledFast, ledSlow, btnDebounce;
uint8_t btnPressed = 0;

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    pinMode(PC0, PIN_MODE_OUTPUT);          // LED เร็ว
    pinMode(PC1, PIN_MODE_OUTPUT);          // LED ช้า
    pinMode(PC2, PIN_MODE_INPUT_PULLUP);    // ปุ่ม

    Start_Timer(&ledFast, 100, 1);
    Start_Timer(&ledSlow, 750, 1);

    while (1) {
        if (Is_Timer_Expired(&ledFast))  digitalToggle(PC0);
        if (Is_Timer_Expired(&ledSlow))  digitalToggle(PC1);

        // อ่านปุ่มพร้อม debounce 50ms
        if (digitalRead(PC2) == LOW && !btnPressed) {
            Start_Timer(&btnDebounce, 50, 0);
            btnPressed = 1;
        }
        if (btnPressed && Is_Timer_Expired(&btnDebounce)) {
            if (digitalRead(PC2) == LOW) {
                // ปุ่มกดจริง — ทำบางอย่าง
            }
            btnPressed = 0;
        }
    }
}

วัดเวลาที่ผ่านไป (Elapsed Time)

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include <SimpleHAL.h>

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    pinMode(PC0, PIN_MODE_OUTPUT);

    uint32_t startTime = Get_CurrentMs();

    while (1) {
        uint32_t now = Get_CurrentMs();
        uint32_t elapsed = ELAPSED_TIME(startTime, now);  // ปลอดภัย overflow

        if (elapsed >= 3000) {
            digitalWrite(PC0, HIGH);   // ทำบางอย่างหลังผ่านไป 3s
            startTime = now;
        } else {
            digitalWrite(PC0, LOW);
        }
    }
}

ปัญหาที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
Delay_Ms ไม่หน่วง (คืน 0 ทันที) ลืมเรียก Timer_Init() หลัง SystemCoreClockUpdate() เพิ่ม Timer_Init() ใน main() ก่อนใช้ delay ทุกตัว
MCU ค้าง/ไม่ตอบสนอง เรียก Delay_Ms หรือ Delay_Us ใน callback interrupt (ISR) ห้ามใช้ delay แบบ blocking ใน ISR — ใช้ flag + main loop แทน
Timer นับผิดหลังจาก 49.7 วัน Get_CurrentMs() overflow ที่ 2^32 ms (~49.7 วัน) ใช้ macro ELAPSED_TIME หรือ IS_TIMEOUT — จัดการ overflow ให้อัตโนมัติ
Get_CurrentUs() overflow เร็ว overflow ทุก 71.6 นาที (2^32 µs) ใช้ Get_CurrentMs() แทนสำหรับงานนาน — ใช้ Get_CurrentUs() เฉพาะงานสั้นๆ (< 1 ชม.)
Is_Timer_Expired ไม่เคยเป็น 1 ลืม Start_Timer(&timer, ms, repeat) หรือ timer ถูก Stop_Timer แล้ว ตรวจสอบว่ามี Start_Timer และไม่ได้เรียก Stop_Timer ก่อนหน้า

ข้อควรรู้เพิ่มเติม

SysTick กับ Hardware Timer

  • SysTick เป็น hardware timer ภายใน Cortex-M core — ไม่ใช่ timer ทั่วไปของ CH32V003
  • ใช้งาน SysTick ไม่เปลือง TIM1/TIM2 — เหลือไว้ให้ PWM, input capture, encoder
  • ความละเอียด 1ms (1kHz) — เข้ากันกับ HAL ทั่วไป

ลำดับการเริ่มต้นที่ถูกต้อง

int main(void) {
    SystemCoreClockUpdate();   // 1. อัปเดต SystemCoreClock
    Timer_Init();              // 2. เริ่ม SysTick — ต้องหลัง SystemCoreClockUpdate
    // ... โค้ดอื่น  ...
    pinMode(PC0, PIN_MODE_OUTPUT);
    // Delay_Ms, Get_CurrentMs ใช้งานได้แล้ว
}
ผิด: สลับลำดับ — Timer_Init() ก่อน SystemCoreClockUpdate() จะได้ความถี่ผิด หน่วงเวลาไม่ตรง

Overflow และวิธีรับมือ

การลบ timestamp ตรงๆ now - start จะผิด เมื่อ overflow — ใช้ macro ด้านล่างแทน:

// ผิด — พังเมื่อ overflow
uint32_t elapsed = Get_CurrentMs() - startTime;

// ถูก — ปลอดภัย overflow
uint32_t elapsed = ELAPSED_TIME(startTime, Get_CurrentMs());

// ตรวจ timeout
if (IS_TIMEOUT(startTime, 1000)) {
    // ผ่านไป 1 วินาที
}
