SimpleSPI ระดับกลาง

สื่อสาร SPI ความเร็วสูง — Hardware SPI แบบ Full-Duplex

ภาพรวม

SimpleSPI ใช้ Hardware SPI ของ CH32V003 รองรับ Full-Duplex (ส่งและรับพร้อมกัน), โหมด 0-3, ความเร็ว 8 ระดับ, และ Pin Config 2 แบบ

SPI เป็นโปรโตคอลสื่อสาร 4 สาย: SCK (นาฬิกา), MOSI (Master Out Slave In), MISO (Master In Slave Out), NSS/CS (Chip Select) — เหมาะกับเซ็นเซอร์, จอแสดงผล, การ์ด SD และอุปกรณ์ความเร็วสูง

คำเตือนสำคัญ — ข้อจำกัดแพ็กเกจ
SOP-8 และ SOP-16 ไม่มี SPI Hardware — โค้ดส่วน SPI จะถูกตัดออกตอนคอมไพล์ด้วย #if defined(CH32V003) / #if defined(CH32V003A8P6)
ถ้าอยู่บน SOP-8/SOP-16 ให้ใช้ shiftOut() / shiftIn() แทน (ดู โมดูล GPIO)

โหมด SPI

เลือกโหมด SPI ผ่านพารามิเตอร์แรกของ SPI_SimpleInit() — โหมด 0 เป็นค่าเริ่มต้นและใช้แพร่หลายที่สุด

โหมดCPOLCPHAขอบนาฬิกาใช้กับ
Mode 000ขาขึ้นอ่าน ขาลงเขียนทั่วไป, LCD, SD Card
Mode 101ขาลงอ่าน ขาขึ้นเขียนเซ็นเซอร์บางตัว
Mode 210ขาลงอ่าน ขาขึ้นเขียนไม่ค่อยใช้
Mode 311ขาขึ้นอ่าน ขาลงเขียนไม่ค่อยใช้
  • CPOL (Clock Polarity): ระดับสัญญาณนาฬิกาตอนว่าง — 0 = LOW, 1 = HIGH
  • CPHA (Clock Phase): ตำแหน่งการสุ่มข้อมูล — 0 = ขอบแรก, 1 = ขอบที่สอง

Pin ConfigSCKMISOMOSINSSรองรับแพ็กเกจ

REMAP เปิดใช้ผ่าน AFIO (Alternate Function I/O) — จะใช้ Pin Config นี้ก็ต่อเมื่อ DEFAULT ชนกับโมดูลอื่น

ตารางฟังก์ชัน

ฟังก์ชันรีเทิร์นคำอธิบาย
SPI_SimpleInit(mode, speed, pin_config) void เริ่มต้น SPI — กำหนดโหมด, ความเร็ว, และขา
SPI_Transfer(data) uint8_t ส่ง 1 byte พร้อมรับกลับ 1 byte
SPI_TransferBuffer(tx, rx, len) void ส่ง buffer พร้อมรับกลับ buffer
SPI_Write(data, len) void ส่งอย่างเดียว (ไม่สนใจข้อมูลรับ)
SPI_Read(data, len, dummy_byte) void รับอย่างเดียว — ส่ง dummy_byte เพื่อสร้างนาฬิกา
SPI_SetCS(state) void ควบคุม CS pin — 0 = LOW (active/เลือก), 1 = HIGH (ปล่อย)
SPI_SetCSPin(port, pin) void เปลี่ยน CS pin ไปขาอื่น — สำหรับหลายอุปกรณ์
SPI_SetBitOrder(order) void เปลี่ยนลำดับบิต — MSB_FIRST หรือ LSB_FIRST
SPI_SetSpeed(speed) void เปลี่ยนความเร็วตอนรันไทม์

การควบคุม CS (Chip Select)

ต้องควบคุม CS เองก่อนและหลัง Transfer ทุกครั้ง — SPI Hardware ไม่ จัดการ CS ให้อัตโนมัติ

// รูปแบบมาตรฐาน
SPI_SetCS(0);              // CS LOW → เลือกอุปกรณ์
SPI_Transfer(0x42);        // ส่ง/รับข้อมูล
SPI_SetCS(1);              // CS HIGH → ปล่อยอุปกรณ์

ถ้าใช้หลายอุปกรณ์ต่อ SPI bus เดียวกัน:

// อุปกรณ์ A — CS ที่ PA1
SPI_SetCSPin('A', 1);
SPI_SetCS(0);
SPI_Transfer(0x10);
SPI_SetCS(1);

// อุปกรณ์ B — CS ที่ PA2
SPI_SetCSPin('A', 2);
SPI_SetCS(0);
SPI_Transfer(0x20);
SPI_SetCS(1);

ลำดับบิต (Bit Order)

Enumคำอธิบาย
MSB_FIRSTส่งบิตสูงสุด (Bit 7) ก่อน — ค่าเริ่มต้น
LSB_FIRSTส่งบิตต่ำสุด (Bit 0) ก่อน — บางอุปกรณ์ต้องการ
SPI_SetBitOrder(LSB_FIRST);  // เปลี่ยนเป็น LSB first
SPI_Transfer(data);
SPI_SetBitOrder(MSB_FIRST);  // กลับเป็น MSB first

ตัวอย่างโค้ด

Transfer พื้นฐาน

#include "simplehal.h"

void setup() {
    SPI_SimpleInit(0, SPI_1MHZ, SPI_PINS_DEFAULT);
}

void loop() {
    SPI_SetCS(0);
    uint8_t response = SPI_Transfer(0x42);  // ส่ง 0x42 รับ response
    SPI_SetCS(1);

    if (response == 0xFF) {
        // อุปกรณ์ตอบสนอง
    }

    Delay_Ms(100);
}

Transfer แบบ Buffer

uint8_t tx_data[] = {0x01, 0x02, 0x03, 0x04};
uint8_t rx_data[4];

SPI_SetCS(0);
SPI_TransferBuffer(tx_data, rx_data, 4);
SPI_SetCS(1);

// rx_data[0..3] มีข้อมูลที่รับกลับมา

หลายอุปกรณ์ — เปลี่ยน CS Pin

void setup() {
    SPI_SimpleInit(0, SPI_2MHZ, SPI_PINS_DEFAULT);
}

uint8_t read_sensor_a() {
    SPI_SetCSPin('A', 1);        // CS → PA1
    SPI_SetCS(0);
    uint8_t val = SPI_Transfer(0x00);
    SPI_SetCS(1);
    return val;
}

uint8_t read_sensor_b() {
    SPI_SetCSPin('A', 2);        // CS → PA2
    SPI_SetCS(0);
    uint8_t val = SPI_Transfer(0x00);
    SPI_SetCS(1);
    return val;
}

ส่งข้อมูลอย่างเดียว (Write Only)

uint8_t data[] = {0x10, 0x20, 0x30, 0x40};

SPI_SetCS(0);
SPI_Write(data, 4);   // ส่ง 4 bytes — ไม่สนใจข้อมูลที่รับกลับ
SPI_SetCS(1);

รับข้อมูลอย่างเดียว (Read Only)

uint8_t buffer[8];

SPI_SetCS(0);
SPI_Read(buffer, 8, 0xFF);  // ส่ง 0xFF (dummy) เพื่อรับ 8 bytes
SPI_SetCS(1);
