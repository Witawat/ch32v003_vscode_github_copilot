โมดูล 9: SimpleTIM + SimpleTIM_Ext — Timer, Stopwatch, Countdown

ระดับกลาง

Part A: SimpleTIM

ภาพรวม

SimpleTIM ใช้ TIM1 และ TIM2 เป็น hardware timer ตั้งเวลา — คำนวณ prescaler และ period อัตโนมัติตามความถี่ที่ต้องการ รองรับ interrupt callback สำหรับงาน periodic

  • คำนวณ prescaler + period อัตโนมัติจากความถี่ที่ต้องการ
  • รองรับ interrupt callback — เหมาะสำหรับงานที่ต้องทำเป็นจังหวะ
  • Timer Ownership Guard ป้องกันการแย่ง timer กับ SimplePWM
  • ฟังก์ชันอ่าน/เขียน counter และ period โดยตรง

Timer Ownership Guard

SimpleTIM ใช้ระบบเดียวกับ SimplePWM เพื่อป้องกันการแย่ง TIM1/TIM2

Owner Stateความหมาย
TIM_OWNER_NONEยังไม่ถูกใช้ — ใครก็ได้ใช้ก่อน
TIM_OWNER_PWMถูกใช้โดย SimplePWM
TIM_OWNER_TIMERถูกใช้โดย SimpleTIM
TIM_OWNER_TIMEXTถูกใช้โดย SimpleTIM_Ext
สำคัญ: ถ้า timer ถูกครอบครองแล้ว — ฟังก์ชันจะ return ทันทีโดยไม่ทำงาน Guard ป้องกันการเขียน register ทับกัน

Timer Resource Map

Timerใช้โดยข้อจำกัด
TIM1PWM หรือ SimpleTIMใช้กับ PWM หรือ TIM — เลือกอย่างใดอย่างหนึ่ง
TIM2PWM หรือ SimpleTIM หรือ SimpleTIM_Extใช้กับ PWM หรือ TIM หรือ TIM_Ext — เลือกอย่างใดอย่างหนึ่ง
TIM2 ขัดแย้งกับ PWM2 และ TIM_Ext: ถ้าใช้ TIM2 เป็น SimpleTIM — จะใช้ PWM2_CH1~CH4 หรือ Stopwatch/Countdown ร่วมกันไม่ได้

ตัวอย่างโค้ด — งาน Periodic

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"

void onTimerTick(void)
{
    // callback — ห้ามใช้ Delay ภายในนี้
    digitalToggle(PC0);  // LED toggle ทุก 1ms
}

int main(void)
{
    SystemCoreClockUpdate();
    Timer_Init();

    pinMode(PC0, PIN_MODE_OUTPUT);

    TIM_SimpleInit(TIM1, 1000);           // 1kHz = 1ms
    TIM_AttachInterrupt(TIM1, onTimerTick);
    TIM_Start(TIM1);

    while(1)
    {
        // main loop อิสระ — ไม่ถูกบล็อก
    }
}
Callback ห้ามใช้ Delay: ภายใน interrupt callback — ห้ามเรียก Delay_Ms(), Delay_Us(), printf() — จะทำให้โปรแกรมค้าง

Part B: SimpleTIM_Ext — Stopwatch + Countdown

ภาพรวม

SimpleTIM_Ext ใช้ TIM2 ที่ 1kHz เป็นฐานนับเวลา — รองรับทั้ง Stopwatch (นับขึ้น) และ Countdown (นับถอยหลัง) พร้อม format เวลาแบบ HH:MM:SS

  • ใช้ TIM2 ที่ 1kHz — ขัดแย้งกับ PWM2 และ PWM2_CH1~CH4
  • รองรับเวลา format HHMMSS, MMSS, SS
  • โหมดแสดงผล: NORMALIZED (60 วิ = 1 นาที) และ RAW (ค่าตรง)
  • Countdown มี alarm callback เมื่อหมดเวลา
TIM2 ขัดแย้ง: SimpleTIM_Ext ใช้ TIM2 — ไม่สามารถใช้ร่วมกับ PWM2 (PWM2_CH1~CH4) หรือ MOTIM2 ได้

รูปแบบเวลาและโหมดแสดงผล

ตัวอย่างโค้ด

Stopwatch

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"

int main(void)
{
    SystemCoreClockUpdate();
    Timer_Init();

    USART_Printf_Init(USART1, BAUD_115200);

    Stopwatch_Init();
    Stopwatch_Start();

    Delay_Ms(5000);  // หน่วง 5 วิ

    uint32_t elapsed = Stopwatch_GetTotalSeconds();
    printf("Elapsed: %lu seconds\r\n", elapsed);

    char buf[16];
    Stopwatch_GetTimeString(buf, MMSS, DISPLAY_NORMALIZED);
    printf("Time: %s\r\n", buf);

    while(1);
}

Countdown with Alarm

void onAlarm(void)
{
    // callback — ห้ามใช้ delay
    printf("Time's up!\r\n");
}

int main(void)
{
    SystemCoreClockUpdate();
    Timer_Init();

    USART_Printf_Init(USART1, BAUD_115200);

    Countdown_Init(0, 0, 10);  // 10 วินาที
    Countdown_SetAlarmCallback(onAlarm);
    Countdown_Start();

    while(!Countdown_IsFinished())
    {
        char buf[12];
        Countdown_GetTimeString(buf, SS, DISPLAY_NORMALIZED);
        printf("Remaining: %s\r\n", buf);
        Delay_Ms(1000);
    }

    printf("Done!\r\n");
    while(1);
}

อ่านเวลาแบบ RAW

Stopwatch_Init();
Stopwatch_Start();

Delay_Ms(125000);  // 125 วิ = 2 นาที 5 วิ

// NORMALIZED: "02:05"
// RAW:         "02:05" (ค่าเท่ากันในกรณีนี้)
char buf[16];
Stopwatch_GetTimeString(buf, MMSS, DISPLAY_NORMALIZED);
printf("Normalized: %s\r\n", buf);

Stopwatch_GetTimeString(buf, MMSS, DISPLAY_RAW);
printf("Raw: %s\r\n", buf);
