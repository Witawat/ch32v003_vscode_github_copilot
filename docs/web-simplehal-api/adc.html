SimpleADC ระดับกลาง

อ่านค่า Analog — ADC 10-bit พร้อมฟีเจอร์ลด Noise และชดเชยแรงดัน

ภาพรวม

SimpleADC จัดการ ADC (Analog-to-Digital Converter) 10-bit ของ CH32V003 — ให้ค่า 0-1023, รองรับ 8 ช่องภายนอก + 2 ช่องภายใน, ทำงานแบบ Lazy Init (เริ่มต้นอัตโนมัติเมื่อเรียกใช้งานครั้งแรก)

ช่องภายในประกอบด้วย VREFINT (แรงดันอ้างอิงภายใน ~1.2V) และ VCALINT — ใช้ประโยชน์ในการคำนวณ VDD จริงและชดเชยแรงดันเลี้ยง

ตารางฟังก์ชัน

ฟังก์ชันรีเทิร์นคำอธิบาย
ADC_SimpleInit() void เปิดทุก channel — เรียกครั้งแรกพอ (Lazy Init)
ADC_SimpleInitChannels(channels, count) void เปิดเฉพาะ channel ที่ระบุ — ประหยัดพลังงานกว่า
ADC_EnableChannel(channel) void เปิด channel เพิ่มหลังจาก init แล้ว
ADC_Read(channel) uint16_t อ่านค่า ADC 0-1023
ADC_ReadMultiple(channels, values, count) void อ่านหลาย channel ในครั้งเดียว
ADC_ReadAverage(channel, samples) uint16_t อ่านค่าเฉลี่ยจาก N ตัวอย่าง — ลด noise
ADC_ToVoltage(adc_value, vref) float แปลงค่า ADC → แรงดัน (V)
ADC_ReadVoltage(channel, vref) float อ่าน + แปลงเป็นแรงดันทันที
ADC_ToPercent(adc_value) float แปลงค่า ADC → เปอร์เซ็นต์ (0.0 - 100.0)
ADC_ReadVrefInt() uint16_t อ่านค่า VREFINT (~1.2V) จาก channel ภายใน
ADC_GetVDD() float คำนวณ VDD จริงจาก VREFINT
ADC_ReadVoltageCompensated(channel) float อ่าน ADC แล้วชดเชย VDD ให้
ADC_GetBatteryPercent(vdd, v_min, v_max) float คำนวณเปอร์เซ็นต์แบตเตอรี่จาก VDD

การชดเชย VDD (VDD Compensation)

ADC ของ CH32V003 ใช้ VDD เป็นแรงดันอ้างอิง — ถ้า VDD ไม่ใช่ 3.3V พอดี (เช่น ใช้แบตเตอรี่), ค่าที่อ่านได้จะคลาดเคลื่อน

VREFINT (~1.2V) เป็นแรงดันอ้างอิงภายในที่คงที่ ไม่ขึ้นกับ VDD — ใช้เป็น "ไม้บรรทัด" เพื่อคำนวณว่า VDD จริงเป็นเท่าไหร่ แล้วปรับค่าอ่าน ADC ให้ถูกต้อง

// วิธีที่ 1: อ่านแบบชดเชยอัตโนมัติ
float voltage = ADC_ReadVoltageCompensated(ADC_CH_PA2);
// ค่านี้ถูกปรับตาม VDD จริงแล้ว — แม่นยำทั้งที่ 3.3V, 2.8V, 4.2V

// วิธีที่ 2: ทำเองทีละขั้น
float vdd = ADC_GetVDD();                         // VDD จริง = ?
float voltage = ADC_ReadVoltage(ADC_CH_PA2, vdd); // แปลงด้วย VDD จริง
หลักการ: VDD_จริง = 1.2V × 4096 / ADC_ReadVrefInt() — CH32V003 มี ADC 10-bit จึงใช้ 1024 แทน 4095 (สูตรภายในปรับค่าไว้แล้ว)

รูปแบบอ่านค่าแบตเตอรี่ (Battery Monitor)

ใช้ VREFINT คำนวณ VDD → แปลงเป็นเปอร์เซ็นต์แบตเตอรี่:

void setup() {
    ADC_SimpleInit();  // เปิดทุก channel (รวม VREFINT)
}

void loop() {
    float vdd = ADC_GetVDD();                           // VDD จริง
    float bat_pct = ADC_GetBatteryPercent(vdd, 3.0, 4.2); // % แบต (3.0V=0%, 4.2V=100%)

    // แสดงผล — SimpleHAL ไม่มีฟังก์ชันพิมพ์ float จึงแปลงเป็นจำนวนเต็มทศนิยม 2/1 ตำแหน่ง
    int32_t vdd_x100 = (int32_t)(vdd * 100.0f);
    int32_t pct_x10 = (int32_t)(bat_pct * 10.0f);
    USART_Print("VDD: ");
    USART_PrintNum(vdd_x100 / 100);
    USART_Print(".");
    USART_PrintNum(vdd_x100 % 100);
    USART_Print("V, Battery: ");
    USART_PrintNum(pct_x10 / 10);
    USART_Print(".");
    USART_PrintNum(pct_x10 % 10);
    USART_Print("%\n");

    Delay_Ms(1000);
}

การลด Noise

สัญญาณ ADC มักมี noise โดยเฉพาะในสภาพแวดล้อมที่มีสัญญาณรบกวน — ใช้ ADC_ReadAverage() เพื่ออ่านหลายครั้งแล้วหาค่าเฉลี่ย:

// อ่านครั้งเดียว — มีโอกาสได้ค่ากระโดดจาก noise
uint16_t raw = ADC_Read(ADC_CH_PA2);

// อ่านเฉลี่ย 10 ครั้ง — ค่าเสถียรกว่า เหมาะกับสัญญาณที่มี noise
uint16_t smooth = ADC_ReadAverage(ADC_CH_PA2, 10);

// อ่านเฉลี่ย 50 ครั้ง — เรียบมาก แต่ตอบสนองช้า
uint16_t very_smooth = ADC_ReadAverage(ADC_CH_PA2, 50);
แนะนำ: สำหรับสัญญาณทั่วไป ใช้ 8-16 samples ก็เพียงพอ ยิ่งใช้มาก ค่าจะยิ่งเรียบแต่ตอบสนองช้าลง — หาจุดสมดุลตามความต้องการของโปรเจกต์

ข้อผิดพลาดที่พบบ่อย

  1. อ่าน channel ที่ยังไม่ init
    ต้องเรียก ADC_SimpleInit() หรือ ADC_SimpleInitChannels() ก่อน
    ถ้าใช้ ADC_SimpleInitChannels() อย่าลืมใส่ channel ที่ต้องการใน array
  2. PD7 ไม่รองรับ ADC
    แม้จะอยู่ใน Port D เหมือน PD2-PD6 แต่ PD7 ไม่มีฟังก์ชัน ADC — ต้องใช้ขาอื่น
  3. ลืมเรียก ADC_SimpleInit
    แม้จะเป็น Lazy Init แต่ ต้อง มีการเรียกอย่างน้อยครั้งหนึ่ง — ADC จะไม่ถูกเปิดเองลอย ๆ
  4. VDD ไม่เสถียรแล้วไม่อ่าน VREFINT
    ถ้าใช้แบตเตอรี่หรือแหล่งจ่ายที่ไม่ใช่ 3.3V คงที่ — ต้องใช้ ADC_ReadVoltageCompensated() หรือ ADC_GetVDD() เสมอ
  5. ใช้ alias เก่า
    ADC_CH_A0 ถึง ADC_CH_A7 — ใช้ alias ใหม่ ADC_CH_PA2 ฯลฯ แทน

ตัวอย่างโค้ด

อ่านค่า Analog พื้นฐาน

#include "simplehal.h"

void setup() {
    ADC_SimpleInit();         // เปิด ADC — ทำครั้งเดียว
}

void loop() {
    uint16_t val = ADC_Read(ADC_CH_PA2);
    float voltage = ADC_ToVoltage(val, 3.3);
    float percent = ADC_ToPercent(val);

    // SimpleHAL ไม่มีฟังก์ชันพิมพ์ float — แปลงเป็นจำนวนเต็มทศนิยม 2/1 ตำแหน่ง
    int32_t v_x100 = (int32_t)(voltage * 100.0f);
    int32_t pct_x10 = (int32_t)(percent * 10.0f);
    USART_Print("ADC: ");
    USART_PrintNum(val);
    USART_Print(", V: ");
    USART_PrintNum(v_x100 / 100);
    USART_Print(".");
    USART_PrintNum(v_x100 % 100);
    USART_Print(", %: ");
    USART_PrintNum(pct_x10 / 10);
    USART_Print(".");
    USART_PrintNum(pct_x10 % 10);
    USART_Print("\n");
    Delay_Ms(500);
}

อ่านหลาย Channel

void setup() {
    // เปิดเฉพาะ channel ที่ใช้ — ประหยัดพลังงาน
    uint8_t channels[] = {ADC_CH_PA2, ADC_CH_PA1, ADC_CH_PC4};
    ADC_SimpleInitChannels(channels, 3);
}

void loop() {
    uint8_t ch_list[] = {ADC_CH_PA2, ADC_CH_PA1, ADC_CH_PC4};
    uint16_t values[3];

    ADC_ReadMultiple(ch_list, values, 3);

    USART_Print("PA2: ");
    USART_PrintNum(values[0]);
    USART_Print(", PA1: ");
    USART_PrintNum(values[1]);
    USART_Print(", PC4: ");
    USART_PrintNum(values[2]);
    USART_Print("\n");
    Delay_Ms(500);
}

void loop() {
    float vdd = ADC_GetVDD();
    float bat_pct = ADC_GetBatteryPercent(vdd, 3.0, 4.2);

    if (bat_pct < 10.0) {
        // แบตเตอรี่ใกล้หมด — เตือน
    }

    // SimpleHAL ไม่มีฟังก์ชันพิมพ์ float — แปลงเป็นจำนวนเต็มทศนิยม 2/1 ตำแหน่ง

อ่านค่าแรงดันแบบชดเชย VDD

void setup() {
    ADC_SimpleInit();
}

void loop() {
    // ชดเชย VDD อัตโนมัติ — แม่นยำแม้ VDD เปลี่ยน
    float voltage = ADC_ReadVoltageCompensated(ADC_CH_PA2);

    // SimpleHAL ไม่มีฟังก์ชันพิมพ์ float — แปลงเป็นทศนิยม 3 ตำแหน่งด้วยจำนวนเต็ม
    int32_t v_x1000 = (int32_t)(voltage * 1000.0f);
    USART_Print("Voltage: ");
    USART_PrintNum(v_x1000 / 1000);
    USART_Print(".");
    USART_PrintNum(v_x1000 % 1000);
    USART_Print("V (VDD-compensated)\n");
    Delay_Ms(500);
}

ลด Noise ด้วยค่าเฉลี่ย

void setup() {
    ADC_SimpleInit();
}

void loop() {
    // อ่านค่าเฉลี่ย 16 ตัวอย่าง — เสถียรสำหรับเซ็นเซอร์
    uint16_t avg = ADC_ReadAverage(ADC_CH_PA2, 16);
    float voltage = ADC_ToVoltage(avg, 3.3);

    // SimpleHAL ไม่มีฟังก์ชันพิมพ์ float — แปลงเป็นทศนิยม 3 ตำแหน่งด้วยจำนวนเต็ม
    int32_t v_x1000 = (int32_t)(voltage * 1000.0f);
    USART_Print("Average: ");
    USART_PrintNum(avg);
    USART_Print(" -> ");
    USART_PrintNum(v_x1000 / 1000);
    USART_Print(".");
    USART_PrintNum(v_x1000 % 1000);
    USART_Print("V\n");
    Delay_Ms(100);
}

เปิด Channel เพิ่มภายหลัง

void setup() {
    // เริ่มต้นด้วย PA2 เท่านั้น
    uint8_t ch[] = {ADC_CH_PA2};
    ADC_SimpleInitChannels(ch, 1);
}

void loop() {
    uint16_t val1 = ADC_Read(ADC_CH_PA2);

    // อยากอ่าน PC4 เพิ่มภายหลัง
    ADC_EnableChannel(ADC_CH_PC4);
    uint16_t val2 = ADC_Read(ADC_CH_PC4);

    USART_Print("PA2: ");
    USART_PrintNum(val1);
    USART_Print(", PC4: ");
    USART_PrintNum(val2);
    USART_Print("\n");
    Delay_Ms(500);
}
