Simple1Wire — 1-Wire Protocol

ระดับผู้เชี่ยวชาญ
ไลบรารี: SimpleHAL.h — ใช้ ไม่ต้อง include ไฟล์ย่อยเอง ทุกฟังก์ชัน 1-Wire รวมอยู่ใน SimpleHAL.h แล้ว

ภาพรวม

1-Wire เป็นโปรโตคอลสื่อสารของ Dallas/Maxim — ใช้สายสัญญาณเส้นเดียว (1 data + GND) สื่อสารกับอุปกรณ์หลายตัวบนบัสเดียวกัน เช่น DS18B20, DS2431 EEPROM, iButton

  • สายเดียว — 1 data line + GND
  • Multi-device — ต่อได้หลายตัวบนบัสเดียวกัน
  • ROM Search — ค้นหา device address ทั้งหมดบนบัส
  • CRC8 — ตรวจสอบความถูกต้องของข้อมูล
Hardware: ต้องต่อ ตัวต้านทาน pull-up 4.7kΩ ภายนอกระหว่าง data line กับ VCC (3.3V) — ใช้ GPIO pin ใดก็ได้

ROM Commands

CommandCodeคำอธิบาย
SKIP_ROM0xCCคุยกับทุกอุปกรณ์บนบัส (ใช้ได้กรณีมีตัวเดียว)
READ_ROM0x33อ่าน ROM — ใช้ได้เมื่อมีอุปกรณ์ตัวเดียวเท่านั้น
MATCH_ROM0x55เลือกอุปกรณ์ตาม ROM address
SEARCH_ROM0xF0ค้นหาอุปกรณ์ทั้งหมด — Maxim AN187 algorithm
ALARM_SEARCH0xECค้นหาเฉพาะอุปกรณ์ที่ alarm active

ROM Address Format

    [ Family Code 1B ] [ Serial Number 6B ] [ CRC8 1B ]
   เช่น DS18B20 → 28 xx xx xx xx xx xx CR

8 byte = Family (1B) + Serial (6B) + CRC8 (1B) — CRC ครอบคลุมทั้ง 8 byte

Search Algorithm — Maxim AN187

ROM Search ใช้ binary tree traversal — ค้นหา ROM address ทีละ 1 bit (LSB first) โดยอ่านพร้อมกันจากทุกอุปกรณ์:

  1. Read 2 bits: bit จริง + complement — ถ้าทุกตัวตอบ 0 → bit = 0
  2. Conflict detection: ถ้า complement ต่างจาก bit → มีอุปกรณ์หลายตัว → เลือก branch
  3. Write direction: เขียน 0 หรือ 1 เพื่อเลือก branch — อุปกรณ์ที่ ROM ไม่ตรงจะหยุดตอบ
  4. ทำซ้ำ 64 รอบ จนได้ ROM address ครบ 64 bit
Interrupt-safe: Search algorithm ต้องการ timing ที่แม่นยำ — library จะ ปิด IRQ ชั่วคราว ในช่วง critical section

ตัวอย่างโค้ด

อ่านอุณหภูมิจาก DS18B20 ตัวเดียว

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include <SimpleHAL.h>

#define ONE_WIRE_PIN PC0

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();

    OneWire_Bus* bus = OneWire_Init(ONE_WIRE_PIN);
    if (!bus) while(1);  // bus เต็ม (ไม่ควรเกิด)

    while (1) {
        uint8_t data[9];

        // DS18B20 — เริ่มแปลงอุณหภูมิ
        OneWire_Reset(bus);
        OneWire_SkipROM(bus);
        OneWire_WriteByte(bus, 0x44);  // Convert T

        Delay_Ms(750);  // รอแปลงเสร็จ (max 750ms)

        // อ่านผล
        OneWire_Reset(bus);
        OneWire_SkipROM(bus);
        OneWire_WriteByte(bus, 0xBE);  // Read Scratchpad
        OneWire_ReadBytes(bus, data, 9);

        // ตรวจ CRC
        if (!OneWire_VerifyCRC(data, 9)) continue;

        // แปลงอุณหภูมิ — int16 ใน data[0:1]
        int16_t temp_raw = (data[1] << 8) | data[0];
        float temp_c = temp_raw / 16.0f;

        (void)temp_c;
        Delay_Ms(1000);
    }
}

ค้นหาอุปกรณ์ทั้งหมดบนบัส

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include <SimpleHAL.h>

#define ONE_WIRE_PIN PC0
uint8_t roms[10][8];  // เก็บ ROM ได้สูงสุด 10 ตัว
uint8_t device_count = 0;

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();

    OneWire_Bus* bus = OneWire_Init(ONE_WIRE_PIN);

    OneWire_ResetSearch(bus);
    while (OneWire_Search(bus) && device_count < 10) {
        OneWire_GetAddress(bus, roms[device_count]);
        device_count++;
    }

    // device_count = จำนวนอุปกรณ์ที่พบบนบัส
    while (1) {
        Delay_Ms(1000);
    }
}

ตรวจสอบ CRC

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include <SimpleHAL.h>

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();

    uint8_t data[] = {0x28, 0xAA, 0xBB, 0xCC, 0xDD, 0xEE, 0xFF, 0x00};

    uint8_t crc = OneWire_CRC8(data, 8);

    bool valid = OneWire_VerifyCRC(data, 8);

    (void)crc;
    (void)valid;

    while (1) { Delay_Ms(1000); }
}
