SimpleOPAMP — Operational Amplifier

ระดับผู้เชี่ยวชาญ
ไลบรารี: SimpleHAL.h — ใช้ ไม่ต้อง include ไฟล์ย่อยเอง ทุกฟังก์ชัน OPAMP รวมอยู่ใน SimpleHAL.h แล้ว

ภาพรวม

CH32V003 มี Operational Amplifier (OPAMP) ภายใน 1 ตัว — ใช้ขยายสัญญาณอนาล็อก เปรียบเทียบแรงดัน โดยไม่ต้องใช้ IC ภายนอก

  • 4 โหมดการทำงาน: Voltage Follower, Non-Inverting, Inverting, Comparator
  • 2 input channels: CHP0/CHP1 (ขาบวก), CHN0/CHN1 (ขาลบ)
  • Output: อ่านผ่าน ADC หรือใช้เป็น digital output (comparator mode)
ต้องใช้ตัวต้านทานภายนอก: โหมด Non-Inverting และ Inverting ต้องต่อ R ภายนอก เพื่อกำหนด gain — OPAMP ภายในไม่มี R feedback ในตัว

โหมดการทำงาน

โหมดGainสูตรใช้เมื่อ
OPAMP_VOLTAGE_FOLLOWER 1 Vout = Vin Buffer สัญญาณ, impedance matching
OPAMP_NON_INVERTING >1 Vout = Vin * (1 + R1/R2) ขยายสัญญาณเฟสไม่กลับ
OPAMP_INVERTING <0 Vout = -Vin * (R2/R1) ขยายสัญญาณกลับเฟส
OPAMP_COMPARATOR ∞ (digital) Vout = HIGH if Vin+ > Vin- เปรียบเทียบแรงดันสองสัญญาณ

วงจรแต่ละโหมด

Voltage Follower (gain = 1)

    Vin ──────┬──► CHPx (OPAMP +)
              │
              └────────┐
                   ┌───┴───┐
                   │ OPAMP │────► Vout → ADC
                   └───┬───┘
                       │
                       └──► CHNx (OPAMP -) [ต่อตรงกับ output]

ใช้ buffer สัญญาณ — input impedance สูง output impedance ต่ำ

Non-Inverting (gain > 1)

    Vin ────────────► CHPx (OPAMP +)

                    R1
              ┌───[====]───┐
              │            │
              │   ┌───┴───┐│
              └───┤ OPAMP ├┴───► Vout
                  └───┬───┘
                      │
              ┌───[====]───┤ GND
              │    R2      │
              └──► CHNx ◄──┘

Gain = 1 + R1/R2 — output ไม่กลับเฟส

Inverting (gain < 0)

    Vin ──[ R1 ]──┬──► CHNx (OPAMP -)
                  │
              ┌───┴───┐
              │       │
    GND ──────┤ OPAMP ├───► Vout
              │       │
              └───┬───┘
                  │
                  └──[ R2 ]──┘ (feedback)

Gain = -(R2/R1) — output กลับเฟส

Comparator

    Vin+ ─────► CHPx (OPAMP +)
                          ┌───┐
    Vin- ─────► CHNx ─────┤   ├──► Vout (digital HIGH/LOW)
                          └───┘

Vin+ > Vin- → HIGH, Vin+ < Vin- → LOW — ไม่ต้องใช้ R ภายนอก

ตารางอ้างอิงฟังก์ชัน

ฟังก์ชันพารามิเตอร์คืนค่าคำอธิบาย
OPAMP_SimpleInit(mode) mode: OPAMP_VOLTAGE_FOLLOWER / OPAMP_COMPARATOR void เริ่ม OPAMP แบบง่าย — ใช้ channel 0 ทั้งคู่
OPAMP_Enable() — void เปิด OPAMP
OPAMP_Disable() — void ปิด OPAMP (ประหยัดพลังงาน)
OPAMP_SetMode(mode) mode: OPAMP_* void เปลี่ยนโหมดขณะทำงาน
OPAMP_Init(pos_ch, neg_ch) pos_ch: OPAMP_CHP0/OPAMP_CHP1
neg_ch: OPAMP_CHN0/OPAMP_CHN1		 void เริ่ม OPAMP + เลือก channel เอง
OPAMP_ConfigVoltageFollower(pos) pos: OPAMP_CHP0/OPAMP_CHP1 void ตั้งโหมด voltage follower
OPAMP_ConfigNonInverting(pos, neg) pos: OPAMP_CHP*
neg: OPAMP_CHN*		 void ตั้งโหมด non-inverting
OPAMP_ConfigInverting(pos, neg) pos: OPAMP_CHP*
neg: OPAMP_CHN*		 void ตั้งโหมด inverting
OPAMP_ConfigComparator(pos, neg) pos: OPAMP_CHP*
neg: OPAMP_CHN*		 void ตั้งโหมด comparator
OPAMP_SetChannels(pos, neg) pos: OPAMP_CHP*
neg: OPAMP_CHN*		 void เปลี่ยน channel โดยไม่เปลี่ยนโหมด
OPAMP_CalculateGainNonInv(r1, r2) r1, r2: R ภายนอก (ohm) float คำนวณ gain = 1 + R1/R2
OPAMP_CalculateGainInv(r1, r2) r1, r2: R ภายนอก (ohm) float คำนวณ gain = -(R2/R1)
OPAMP_CalculateR2NonInv(r1, gain) r1: ohm, gain: >1 uint32_t คำนวณ R2 จาก R1 และ gain ที่ต้องการ
OPAMP_CalculateR2Inv(r1, gain) r1: ohm, gain: magnitude uint32_t คำนวณ R2 จาก R1 และ gain magnitude
OPAMP_IsEnabled() — uint8_t (1/0) ตรวจสอบ OPAMP เปิดอยู่หรือไม่
OPAMP_GetConfig() — uint32_t อ่านค่า configuration register ปัจจุบัน

การอ่าน Output ผ่าน ADC

OPAMP output ต่อเข้ากับ ADC channel ภายใน — อ่านค่าด้วย analogRead() หรือ ADC_Read()

OPAMP_ConfigVoltageFollower(OPAMP_CHP0);
OPAMP_Enable();

uint16_t output = analogRead(ADC_OPAMP);  // อ่าน output OPAMP

ตัวอย่างโค้ด

Voltage Follower — Buffer สัญญาณ

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include <SimpleHAL.h>

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();

    // Buffer สัญญาณจาก CHP0 — output = input
    OPAMP_ConfigVoltageFollower(OPAMP_CHP0);
    OPAMP_Enable();

    while (1) {
        uint16_t val = analogRead(ADC_OPAMP);
        Delay_Ms(100);
    }
}

Non-Inverting — ขยาย x2

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include <SimpleHAL.h>

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();

    // R1 = 10kΩ, R2 = 10kΩ → gain = 1 + 10k/10k = 2
    OPAMP_ConfigNonInverting(OPAMP_CHP0, OPAMP_CHN0);
    OPAMP_Enable();
    // ต่อ R1=10k ระหว่าง output กับ CHN0
    // ต่อ R2=10k ระหว่าง CHN0 กับ GND

    float gain = OPAMP_CalculateGainNonInv(10000, 10000); // 2.0
    (void)gain;

    while (1) {
        uint16_t val = analogRead(ADC_OPAMP);
        Delay_Ms(100);
    }
}

Comparator — ตรวจจับแรงดันเกิน threshold

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include <SimpleHAL.h>

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();

    pinMode(PC0, PIN_MODE_OUTPUT);

    // CHP0 = สัญญาณที่ต้องการวัด, CHN0 = แรงดันอ้างอิง
    OPAMP_ConfigComparator(OPAMP_CHP0, OPAMP_CHN0);
    OPAMP_Enable();

    while (1) {
        // อ่าน output digital (HIGH = CHP0 > CHN0)
        uint16_t result = analogRead(ADC_OPAMP);
        if (result > 512) {
            digitalWrite(PC0, HIGH);  // เกิน threshold
        } else {
            digitalWrite(PC0, LOW);
        }
        Delay_Ms(10);
    }
}
