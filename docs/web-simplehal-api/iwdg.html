SimpleIWDG — Independent Watchdog

ระดับสูง
ไลบรารี: SimpleHAL.h — ใช้ ไม่ต้อง include ไฟล์ย่อยเอง ทุกฟังก์ชัน IWDG รวมอยู่ใน SimpleHAL.h แล้ว

ภาพรวม

Independent Watchdog (IWDG) เป็น hardware watchdog ที่ใช้แหล่งจ่ายสัญญาณนาฬิกา LSI 40kHz ซึ่งแยกอิสระจากระบบนาฬิกาหลัก (HSE/HSI/PLL) — แม้ระบบนาฬิกาหลักจะล่ม IWDG ก็ยังทำงานต่อไป

  • LSI 40kHz — แหล่งนาฬิกาอิสระ ไม่พึ่ง system clock
  • 12-bit reload counter — ค่า reload 0x000-0xFFF (0-4095)
  • 7 prescalers — 4, 8, 16, 32, 64, 128, 256
  • Timeout — ตั้งแต่ 0.1ms ถึง 26.2s (LSI 40kHz; จริงอาจถึง 32.7s ถ้า LSI 30kHz)
LSI tolerance ±25%: เวลา timeout จริงอาจคลาดเคลื่อน ±25% จากค่าที่คำนวณ — ตรวจสอบ Datasheet และเผื่อ margin ไว้เสมอ

ตารางอ้างอิงฟังก์ชัน

ฟังก์ชันพารามิเตอร์คืนค่าคำอธิบาย
IWDG_SimpleInit(timeout_ms) timeout_ms: uint32_t (1-32767) void ตั้ง timeout (ms) — เลือก prescaler ที่เหมาะสมให้อัตโนมัติ
IWDG_Feed() — void รีเซ็ต counter — หมาต้องให้อาหารก่อนหมดเวลา ไม่งั้นระบบรีเซ็ต
IWDG_Init(prescaler, reload) prescaler: IWDG_PRESCALER_*
reload: uint16_t (0-4095)		 void ตั้งค่า prescaler และ reload ด้วยตนเอง (อิสระเต็มที่)
IWDG_IsBusy() — uint8_t กำลังอัปเดต register? (1=กำลังอัปเดต, 0=พร้อม)
IWDG_GetTimeout(prescaler, reload) prescaler: IWDG_PRESCALER_*
reload: uint16_t		 uint32_t (ms) คำนวณ timeout จริงจาก prescaler + reload
IWDG_WasResetCause() — uint8_t (1/0) รีเซ็ตล่าสุดเกิดจาก IWDG หรือไม่?
IWDG_ClearResetFlag() — void ⚠️ เคลียร์ทุก reset flag (ไม่ใช่แค่ IWDG)
คำแนะนำ: ใช้ IWDG_SimpleInit() สำหรับกรณีทั่วไป — library จะเลือก prescaler ที่ใกล้เคียง timeout ที่ต้องการที่สุดให้อัตโนมัติ

การทำงานร่วมกับ yield()

SimpleHAL มีระบบ yield() สำหรับให้อาหาร watchdog อัตโนมัติ — เรียก arduino_SetIWDGActive() แล้ว yield() จะ feed IWDG ให้ทุกครั้ง

IWDG_SimpleInit(1000);   // timeout 1 วินาที
arduino_SetIWDGActive(); // เปิด auto-feed ผ่าน yield()

while (1) {
    // ทำงานอื่น...
    Delay_Ms(100);
    // yield() ถูกเรียกใน Delay_Ms → feed IWDG อัตโนมัติ
}
ข้อควรระวัง: yield() จะ feed เฉพาะตอนที่ถูกเรียก — ถ้าโค้ดติด loop ที่ไม่มี yield() หรือ Delay นานกว่า timeout → IWDG จะรีเซ็ตอยู่ดี

คำเตือน

เมื่อเริ่มแล้วหยุดไม่ได้: IWDG เมื่อเริ่มทำงานแล้ว ไม่สามารถหยุดได้ ด้วยซอฟต์แวร์ — เป็นข้อจำกัดระดับ hardware การรีเซ็ตหรือ power cycle เท่านั้นที่หยุด IWDG ได้
IWDG_ClearResetFlag() เคลียร์ทุก reset flag (power-on, external reset, IWDG, WWDG, software, low-power) — ไม่ใช่แค่ IWDG

ตัวอย่างโค้ด

Basic Watchdog — timeout 2 วินาที

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include <SimpleHAL.h>

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();

    IWDG_SimpleInit(2000);  // ถ้าไม่ feed ภายใน 2 วินาที → รีเซ็ต

    pinMode(PC0, PIN_MODE_OUTPUT);

    while (1) {
        digitalToggle(PC0);   // กระพริบ LED
        Delay_Ms(500);        // yield() → feed IWDG อัตโนมัติ
    }
}

ตรวจจับสาเหตุการรีเซ็ต

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include <SimpleHAL.h>

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();

    if (IWDG_WasResetCause()) {
        // รีเซ็ตล่าสุดเกิดจาก watchdog → ระบบค้างก่อนหน้า
        IWDG_ClearResetFlag();
        // แจ้งเตือน, บันทึก log, หรือกู้คืนสถานะ
    }

    IWDG_SimpleInit(1000);

    while (1) {
        // ทำงานปกติ
        IWDG_Feed();  // ให้อาหารเอง ไม่พึ่ง yield()
        Delay_Ms(100);
    }
}

int main(void) {

    while (1) {
        // ทำงาน...
        IWDG_Feed();
    }
}
