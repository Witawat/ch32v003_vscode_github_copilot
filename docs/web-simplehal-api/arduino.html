SimpleArduino — Arduino API Compatibility

ระดับเริ่มต้น
ไลบรารี: SimpleHAL.h — ใช้ ไม่ต้อง include ไฟล์ย่อยเอง ทุกฟังก์ชัน Arduino compatibility รวมอยู่ใน SimpleHAL.h แล้ว

ภาพรวม

SimpleArduino เป็นเลเยอร์เลียนแบบ Arduino API — ทำให้ผู้ที่คุ้นเคยกับ Arduino สามารถย้ายมาใช้ CH32V003 ได้ทันที โดยใช้ชื่อฟังก์ชันและรูปแบบที่คุ้นเคย

ฟังก์ชันทั้งหมดเป็น alias macro หรือ wrapper function — ไม่มี overhead เพิ่ม เพราะถูกแปลงเป็นฟังก์ชัน SimpleHAL โดยตรงตอนคอมไพล์

ตารางอ้างอิงฟังก์ชัน

ฟังก์ชันพารามิเตอร์คืนค่าคำอธิบายใช้ร่วมกับ
digitalPinToInterrupt(pin) pin: GPIO pin uint8_t แปลง pin เป็น EXTI line number — ใช้กับ attachInterrupt ผลลัพธ์ส่งต่อให้ฟังก์ชัน interrupt แบบ Arduino ที่ต้องการ EXTI line โดยตรง — ถ้าใช้ attachInterrupt(pin, cb, mode) ของ SimpleGPIO ปกติไม่จำเป็นต้องเรียกฟังก์ชันนี้เอง
randomSeed(seed) seed: uint32_t void ตั้ง seed สำหรับ PRNG ควรเรียกก่อน _randomMax()/_randomRange() ครั้งแรกเสมอ — ถ้าไม่เรียก ค่าสุ่มจะซ้ำรูปแบบเดิมทุกครั้งที่เปิดเครื่อง (มักใช้ analogRead() จาก pin ที่ไม่ต่ออะไรเป็น seed เพื่อสุ่มจาก noise)
_randomMax(max) max: long long สุ่ม [0, max-1] — ⚠️ ไม่ใช่ random() เพราะชื่อชนกับ stdlib เรียกได้หลัง randomSeed() — หรือใช้ #define ENABLE_ARDUINO_RANDOM_MACRO ก่อน include เพื่อเรียกเป็น random(max) แทน
_randomRange(min, max) min: long
max: long		 long สุ่ม [min, max-1] เช่นเดียวกับ _randomMax() — ควรเรียก randomSeed() ก่อนใช้งานจริงครั้งแรก
yield() — void Cooperative multitasking — feed IWDG ถ้า active ถูกเรียกอัตโนมัติภายใน delay()/Delay_Ms() — ถ้าต้องการทำงานพิเศษระหว่างรอ ให้ประกาศฟังก์ชัน void yield(void) ของตัวเอง (override) ไม่ต้องเรียกตรงๆ
arduino_SetIWDGActive() — void เปิด auto-feed IWDG ผ่าน yield() — ถูกเรียกโดย SimpleIWDG ถูกเรียกอัตโนมัติจาก IWDG_SimpleInit() อยู่แล้ว — ไม่ต้องเรียกเองในโค้ดผู้ใช้ทั่วไป
dtostrf(val, width, precision, buf) val: double
width: int
precision: int
buf: char*		 char* แปลง float → string (ไม่ต้องใช้ FPU) buf ต้องจองไว้ล่วงหน้าอย่างน้อย width+2 byte ก่อนเรียก — ผลลัพธ์มักส่งต่อให้ USART_Println(buf)/USART_Print(buf) พิมพ์ออก

Optional USART Extensions

ฟังก์ชันกลุ่มนี้เปิดใช้งานโดยการ #define ก่อน include:

USART_Println Series — ต้องการ #define ENABLE_USART_PRINTLN

ฟังก์ชันพารามิเตอร์คำอธิบาย
USART_Println(str)str: const char*พิมพ์ string + ขึ้นบรรทัดใหม่
USART_PrintlnNum(num)num: int32_tพิมพ์ตัวเลข + ขึ้นบรรทัดใหม่
USART_PrintlnHex(num, upper)num: uint32_t
upper: uppercase? (1/0)		พิมพ์เลขฐาน 16 + ขึ้นบรรทัดใหม่

USART_PrintFloat — ต้องการ #define ENABLE_USART_PRINTFLOAT

ฟังก์ชันพารามิเตอร์คำอธิบาย
USART_PrintFloat(val, dec)val: float
dec: ตำแหน่งทศนิยม		พิมพ์ float ผ่าน USART
USART_PrintlnFloat(val, dec)val: float
dec: ตำแหน่งทศนิยม		พิมพ์ float + ขึ้นบรรทัดใหม่

ทำไม _randomMax ไม่ใช่ random()

stdlib.h (ไลบรารีมาตรฐาน C) มีฟังก์ชัน long random(void) อยู่แล้ว — ชื่อ random() จึงซ้ำกัน SimpleHAL เลือกใช้ _randomMax() และ _randomRange() เพื่อหลีกเลี่ยงการชน

ต้องการชื่อ random() แบบ Arduino: #define ENABLE_ARDUINO_RANDOM_MACRO ก่อน #include <SimpleHAL.h> — จะเปิด macro overload ชื่อ random

dtostrf — Float to String (ไม่ต้องใช้ FPU)

dtostrf() แปลงเลขทศนิยมเป็น string โดยไม่ต้องใช้ FPU — เหมาะกับ CH32V003 ที่ไม่มี FPU hardware

char buf[12];
dtostrf(3.14159, 8, 3, buf);  // "   3.142"
// width=8, precision=3 → มีช่องว่างด้านหน้า
Buffer size: เผื่อ buffer อย่างน้อย width + 2 byte — สำหรับ null terminator และอาจมีเครื่องหมายลบ

yield() และ IWDG Integration

yield() เป็นจุดแทรกสำหรับ cooperative multitasking — ถูกเรียกในฟังก์ชันที่รอนาน เช่น Delay_Ms(), Delay_Us(), ลูปต่างๆ

เมื่อเปิด IWDG ด้วย IWDG_SimpleInit() — library จะเรียก arduino_SetIWDGActive() อัตโนมัติ ทำให้ yield() feed watchdog ให้ทุกครั้ง

IWDG_SimpleInit(2000);   // timeout 2s
// arduino_SetIWDGActive() ถูกเรียกอัตโนมัติ

while (1) {
    // ทำงานหนัก...
    Delay_Ms(500);       // yield() → feed IWDG → ป้องกันรีเซ็ต
}
Custom yield: สามารถ override yield() ได้ — ประกาศฟังก์ชัน void yield(void) เองเพื่อเพิ่มงานที่ต้องการทำระหว่างรอ

ตัวอย่างโค้ด

Arduino-Style Blink

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include <SimpleHAL.h>

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();

    // ใช้ชื่อฟังก์ชันแบบ Arduino ได้ทันที
    pinMode(PC0, OUTPUT);

    while (1) {
        digitalWrite(PC0, HIGH);
        delay(500);
        digitalWrite(PC0, LOW);
        delay(500);
    }
}

Random Number

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include <SimpleHAL.h>

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();

    randomSeed(analogRead(PC1));  // ใช้ ADC noise เป็น seed

    pinMode(PC0, PIN_MODE_OUTPUT);

    while (1) {
        long r = _randomRange(100, 500);  // สุ่ม [100, 499]
        digitalWrite(PC0, HIGH);
        delay(r);
        digitalWrite(PC0, LOW);
        delay(200);
    }
}

dtostrf — พิมพ์ Float ผ่าน USART

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#define ENABLE_USART_PRINTLN
#define ENABLE_USART_PRINTFLOAT
#include <SimpleHAL.h>

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();

    USART_Printf_Init(115200);

    float temperature = 25.75f;
    char buf[12];

    dtostrf(temperature, 6, 2, buf);  // " 25.75"
    USART_Println(buf);

    // หรือใช้ฟังก์ชันเฉพาะ
    USART_PrintlnFloat(temperature, 2);  // "25.75"

    while (1) { Delay_Ms(1000); }
}

yield() — Custom Cooperative Task

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include <SimpleHAL.h>

volatile uint8_t task_flag = 0;

void OnTimer(void) {
    task_flag = 1;
}

// Override yield — ทำงานเบื้องหลังระหว่าง delay
void yield(void) {
    if (task_flag) {
        task_flag = 0;
        digitalToggle(PC1);  // กระพริบ LED อีกตัวระหว่างรอ
    }
}

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();

    pinMode(PC0, PIN_MODE_OUTPUT);
    pinMode(PC1, PIN_MODE_OUTPUT);

    // ตั้ง Timer interrupt สำหรับ task_flag
    // ... (ดูโมดูล TIM)

    while (1) {
        digitalWrite(PC0, HIGH);
        delay(1000);
        digitalWrite(PC0, LOW);
        delay(1000);
        // ระหว่าง 2 วินาทีนี้ yield() ทำงาน → PC1 กระพริบ
    }
}
