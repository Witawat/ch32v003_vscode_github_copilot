SimpleUSART — สื่อสาร Serial กลาง

ภาพรวม

SimpleUSART ใช้ hardware USART ของ CH32V003 รับส่งข้อมูลผ่าน interrupt (ISR) พร้อม RX ring buffer 64 bytes รองรับ 7 baud rate และ 4 pin configuration สามารถเลือกสลับขาใช้งานได้หลายโหมด

คุณสมบัติรายละเอียด
HardwareUSART (built-in CH32V003)
RX BufferRing buffer 64 bytes (ISR-driven)
Baud rates7 ค่า (9600 — 460800)
Pin remap4 configs (DEFAULT, REMAP1, REMAP2, FULL_REMAP)

RX Byte Hook

USART_RxByteHook(uint8_t data) เป็นฟังก์ชัน weak ที่ถูกเรียกจากภายใน ISR ทุกครั้งที่มี byte เข้ามา ไลบรารีอื่น (เช่น TJC Serial Display) สามารถ override ฟังก์ชันนี้เพื่อตรวจสอบ/ดักข้อมูล โดยไม่มีผลกระทบต่อ ring buffer หลัก

วิธีใช้ — ประกาศในโค้ดของตัวเอง:

void USART_RxByteHook(uint8_t data) {
    // ส่งต่อไปให้ไลบรารี TJC จัดการ
    tjc_feed(data);
}
จุดสำคัญ: hook นี้ทำงานภายใน ISR ควรประมวลผลให้สั้นที่สุด อย่าใช้ Delay_Ms หรือ loop นานภายใน hook

Ring Buffer

RX ใช้ ring buffer ขนาด 64 bytes — RX interrupt (USART1_IRQHandler) จะอ่าน data register และเขียนเข้า buffer โดยอัตโนมัติ ฝั่งผู้ใช้เรียก USART_Read() / USART_ReadBytes() เพื่อดึงข้อมูลออก

  • Buffer เต็ม → byte ใหม่จะถูกทิ้ง (silent drop)
  • เรียก USART_Available() เพื่อตรวจสอบก่อนอ่าน
  • เรียก USART_Flush() เพื่อล้าง buffer ทั้งหมด

ปรับขนาด Buffer

ค่า default คือ 64 bytes สามารถเปลี่ยนได้โดย #define ก่อน #include:

#define USART_RX_BUFFER_SIZE 128
#include "simple_usart.h"
ข้อควรระวัง: CH32V003 มี RAM เพียง 2KB — การเพิ่ม buffer มากเกินไปอาจทำให้ RAM ไม่พอ

ข้อผิดพลาดที่พบบ่อย

ลืมเรียก SystemCoreClockUpdate() — ทำให้ baud rate ผิดเพี้ยน

SystemCoreClockUpdate() ต้องถูกเรียกก่อน USART_SimpleInit() เสมอ (เป็น ส่วนหนึ่งของ 3 ขั้นตอนบังคับ)

ใช้ REMAP1 บน SOP8 — PD0/PD1 ไม่ถูก bond ออกมาบนแพ็กเกจ SOP8

คอมไพล์จะขึ้น #error — เปลี่ยนไปใช้ DEFAULT หรือเปลี่ยนชิปเป็น TSSOP20/QFN20

USART ISR ชนกับไลบรารีอื่น — หากมีไลบรารีอื่นที่ใช้ USART1_IRQHandler (เช่น DMA, LIN) จะเกิด conflict ต้องเลือกใช้อย่างใดอย่างหนึ่ง

ตัวอย่างโค้ด

Hello World

#include "simplehal.h"

int main(void) {
    SystemCoreClockUpdate();
    USART_SimpleInit(USART_BAUD_115200, USART_PIN_DEFAULT);

    USART_Print("Hello from CH32V003!\r\n");

    uint32_t counter = 0;
    while (1) {
        USART_Print("Count: ");
        USART_PrintNum(counter++);
        USART_Print("\r\n");
        Delay_Ms(1000);
    }
}

Echo

#include "simplehal.h"

int main(void) {
    SystemCoreClockUpdate();
    USART_SimpleInit(USART_BAUD_115200, USART_PIN_DEFAULT);

    USART_Print("Echo ready\r\n");
    while (1) {
        if (USART_Available()) {
            uint8_t c = USART_Read();
            USART_WriteByte(c);  // ส่งกลับทันที
        }
    }
}

Number Printing

int16_t value = -12345;
USART_Print("Signed: ");   USART_PrintNum(value);  USART_Print("\r\n");

uint32_t hex = 0xDEADBEEF;
USART_Print("Hex: 0x");    USART_PrintHex(hex, 1);  USART_Print("\r\n");
USART_Print("hex: 0x");    USART_PrintHex(hex, 0);  USART_Print("\r\n");
// Output:
// Signed: -12345
// Hex: 0xDEADBEEF
// hex: 0xdeadbeef

Hex Dump

void hex_dump(const uint8_t *data, uint16_t len) {
    for (uint16_t i = 0; i < len; i++) {
        if (data[i] < 0x10) USART_WriteByte('0');
        USART_PrintHex(data[i], 1);
        USART_WriteByte(' ');
    }
    USART_Print("\r\n");
}

// ใช้งาน
uint8_t buf[] = {0x48, 0x65, 0x6C, 0x6C, 0x6F};
hex_dump(buf, 5);  // Output: 48 65 6C 6C 6F
