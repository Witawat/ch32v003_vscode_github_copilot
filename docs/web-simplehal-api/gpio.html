SimpleGPIO — Digital I/O

ระดับเริ่มต้น
ไลบรารี: SimpleHAL.h — ใช้ ไม่ต้อง include ไฟล์ย่อยเอง ทุกฟังก์ชัน GPIO รวมอยู่ใน SimpleHAL.h แล้ว

ผังการทำงาน SimpleGPIO

flowchart LR
    PM["pinMode()"] --> OUT["Output"]
    PM --> IN["Input"]
    PM --> INT["Interrupt"]
    OUT --> DW["digitalWrite()"]
    OUT --> TG["digitalToggle()"]
    OUT --> AW["analogWrite()"]
    IN --> DR["digitalRead()"]
    IN --> AR["analogRead()"]
    INT --> AI["attachInterrupt()"]
    INT --> DI["detachInterrupt()"]

รายชื่อ Pin ที่ใช้งานได้

พอร์ทPinฟังก์ชันพิเศษหมายเหตุ
PAPA1, PA2ADC, PWMTSSOP20/SOP16/QFN20 เท่านั้น
PCPC0, PC1, PC2, PC3, PC4, PC5, PC6, PC7ADC (PC1-PC7), PWMครบทุกแพ็กเกจ
PDPD0, PD1, PD2, PD3, PD4, PD5, PD6, PD7ADC (PD1-PD7), PWMPD0/PD1 = SWIO ใช้ร่วมกับ debugger; SOP8 ไม่มี PD ทั้งหมด
ข้อจำกัด SOP8: PD0-PD7 ไม่มี ในแพ็กเกจ SOP8 (8 ขา) — ใช้ได้เฉพาะ PC0-PC7 เท่านั้น SOP8 มีเพียง 6 pin I/O (PC0-PC5), PC6/PC7 ซ้อนกับ NRST/MCLR

โหมด Pin (Pin Modes)

ค่าคงที่คำอธิบายใช้เมื่อ
PIN_MODE_INPUTInput floatingอ่านสัญญาณภายนอก (ต้องมี pull ภายนอก)
PIN_MODE_OUTPUTOutput push-pullขับ LED, relay, transistor โดยตรง
PIN_MODE_INPUT_PULLUPInput + pull-up ภายในปุ่มกดต่อ GND, อ่านสวิตช์
PIN_MODE_INPUT_PULLDOWNInput + pull-down ภายในปุ่มกดต่อ VCC
PIN_MODE_OUTPUT_ODOutput open-drainI2C, ต่อบัส shared, level-shifting

การตรวจสอบ Pin (Compile-Time Validation)

SimpleHAL มี macro สำหรับตรวจสอบ pin ตั้งแต่ตอนคอมไพล์ — ไม่ต้องรอรันแล้วเจอบั๊ก

Macroคืนค่าคำอธิบาย
IS_VALID_PIN(pin)1 = ใช้ได้, 0 = ใช้ไม่ได้ตรวจ pin มีจริงตามแพ็กเกจ (SOP8 จำกัดแค่ PC0-PC5)
IS_ADC_PIN(pin)1 = เป็น ADC, 0 = ไม่ใช่ตรวจ pin รองรับ ADC หรือไม่
IS_PWM_PIN(pin)1 = เป็น PWM, 0 = ไม่ใช่ตรวจ pin รองรับ PWM หรือไม่
ใช้ PACKAGE_SOP8 แต่เรียก pinMode(PD0, ...) = compile error ทันที — macro IS_VALID_PIN จะจับตั้งแต่ตอนคอมไพล์

ตารางอ้างอิงฟังก์ชัน

ฟังก์ชันพารามิเตอร์คืนค่าคำอธิบายใช้ร่วมกับ
pinMode(pin, mode) pin: หมายเลข pin
mode: PIN_MODE_*		 void ตั้งโหมด pin (input/output/pullup/ฯลฯ) ต้องเรียกก่อนใช้งาน pin ทุกครั้ง
pinModeMultiple(pins_array, mode) pins_array: const uint8_t[]
mode: PIN_MODE_*		 void (macro) ตั้งโหมดหลาย pin พร้อมกัน ใช้กับ array const uint8_t pins[] = {PC0, PC1, PC2};
digitalWrite(pin, value) pin: หมายเลข pin
value: HIGH/LOW		 void เขียน HIGH (3.3V) หรือ LOW (0V) ใช้หลัง pinMode(pin, PIN_MODE_OUTPUT)
digitalRead(pin) pin: หมายเลข pin uint8_t (HIGH/LOW) อ่านสถานะลอจิกจาก pin ใช้หลัง pinMode(pin, PIN_MODE_INPUT*)
digitalToggle(pin) pin: หมายเลข pin void สลับสถานะ HIGH↔LOW ใช้หลัง pinMode(pin, PIN_MODE_OUTPUT)
digitalWriteMultiple(pins, values) pins: array pin
values: array HIGH/LOW		 void (macro) เขียนหลาย pin พร้อมกัน uint8_t vals[] = {HIGH, LOW, HIGH};
attachInterrupt(pin, callback, mode) pin: หมายเลข pin
callback: ฟังก์ชัน
mode: CHANGE/RISING/FALLING		 void ติด interrupt hardware ที่ pin ต้อง pinMode(pin, PIN_MODE_INPUT*) ก่อน; callback ห้ามใช้ Delay_Ms
detachInterrupt(pin) pin: หมายเลข pin void ยกเลิก interrupt ที่ pin ใช้หลัง attachInterrupt
pulseIn(pin, state, timeout) pin: หมายเลข pin
state: HIGH/LOW
timeout: µs (0=ไม่จำกัด)		 uint32_t (µs) วัดความกว้าง pulse หน่วย µs ใช้กับ ultrasonic HC-SR04 วัดระยะทาง
shiftOut(dataPin, clockPin, bitOrder, value) dataPin, clockPin: pin
bitOrder: MSBFIRST/LSBFIRST
value: uint8_t		 void ส่งข้อมูลแบบ software SPI 74HC595 shift register + latch pin
shiftIn(dataPin, clockPin, bitOrder) dataPin, clockPin: pin
bitOrder: MSBFIRST/LSBFIRST		 uint8_t รับข้อมูลแบบ software SPI 74HC165 parallel-in serial-out
portWrite(port, value) port: 'C' หรือ 'D'
value: 0-255		 void เขียนทั้งพอร์ท 8 bit พร้อมกัน เร็วกว่า digitalWrite ทีละ pin
portRead(port) port: 'C' หรือ 'D' uint8_t อ่านทั้งพอร์ท 8 bit อ่าน 8 input พร้อมกัน
analogRead(pin) pin: หมายเลข pin ADC uint16_t (0-1023) อ่านค่า ADC 10-bit (macro) compile-time validate ว่า pin เป็น ADC
analogReadMilliVolts(pin) pin: หมายเลข pin ADC uint16_t (mV) อ่าน ADC แปลงเป็น mV โดยตรง (macro) ใช้ Vref=3.3V; compile-time validate
analogWrite(pin, value) pin: หมายเลข pin PWM
value: 0-255		 void (macro) เขียน PWM 8-bit, auto-init 1kHz เรียกครั้งแรก auto-config Timer PWM

กลุ่มฟังก์ชันที่ใช้ร่วมกัน

1. Basic LED: pinMode → digitalWrite → Delay_Ms

pinMode(PC0, PIN_MODE_OUTPUT);
while(1) {
    digitalWrite(PC0, HIGH);
    Delay_Ms(500);
    digitalWrite(PC0, LOW);
    Delay_Ms(500);
}

กระพริบ LED 1Hz — ใช้ฟังก์ชันพื้นฐานที่สุด 3 ตัว

2. Button: pinMode(PULLUP) → digitalRead

pinMode(PC1, PIN_MODE_INPUT_PULLUP);
if (digitalRead(PC1) == LOW) {
    // ปุ่มถูกกด (ต่อปุ่มระหว่าง pin กับ GND)
}

pull-up ภายใน = ไม่ต้องต่อ R ภายนอก; LOW = ปุ่มกด (วงจรครบ GND)

3. Interrupt: pinMode → attachInterrupt

volatile uint8_t flag = 0;

void onButton() {
    flag = 1;  // callback ต้องเร็ว ห้ามใช้ Delay
}

pinMode(PC2, PIN_MODE_INPUT_PULLUP);
attachInterrupt(PC2, onButton, FALLING);  // จับขอบลง
while(1) {
    if (flag) {
        flag = 0;
        digitalToggle(PC0);  // ทำงานใน main loop
    }
}
Callback interrupt: ตัวแปรที่ใช้ใน callback ต้องประกาศเป็น volatile และห้ามเรียก Delay_Ms ใน callback เพราะทำให้ MCU ค้างใน ISR
EXTI Line ชนกันข้าม port: ฮาร์ดแวร์ CH32V003 ผูก interrupt line ตามเลขขา ไม่ใช่ port — attachInterrupt(PC1, ...) กับ attachInterrupt(PD1, ...) ใช้ EXTI Line 1 เดียวกัน เรียกพร้อมกันไม่ได้ (ตัวหลังจะทับตัวแรกเงียบๆ) ถ้าต้องใช้ interrupt หลายขา ให้เลือกขาที่ เลขท้ายต่างกัน เช่น PC1 กับ PD2 แทน PC1 กับ PD1

4. Pulse Measure: pinMode → pulseIn

pinMode(PC3, PIN_MODE_INPUT);
uint32_t duration = pulseIn(PC3, HIGH, 25000);  // timeout 25ms
float distance = duration / 58.0;  // cm (HC-SR04)

วัดความกว้าง pulse จาก ultrasonic — state บอกขอบที่รอ, timeout=0 คือไม่จำกัดเวลา

5. Shift Register: shiftOut + latch pin

pinMode(dataPin, PIN_MODE_OUTPUT);
pinMode(clockPin, PIN_MODE_OUTPUT);
pinMode(latchPin, PIN_MODE_OUTPUT);

digitalWrite(latchPin, LOW);
shiftOut(dataPin, clockPin, MSBFIRST, 0b10101010);
digitalWrite(latchPin, HIGH);  // latch — ออก 8 ขาพร้อมกัน

74HC595: shiftOut ส่งข้อมูลทีละบิต → latch HIGH = ข้อมูลออก 8 ขาพร้อมกัน

ปัญหาที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
LED ไม่ติด ต่อกลับขั้ว LED หรือลืม R จำกัดกระแส LED ขายาว (+) ต่อ R 220Ω → MCU pin; ขาสั้น (-) → GND
ปุ่มอ่านค่าผิด/เด้ง ลืม pull-up หรือใช้ INPUT ธรรมดาไม่มี pull ใช้ PIN_MODE_INPUT_PULLUP + debounce ในโค้ด
MCU รีเซ็ต/ค้าง ดึงกระแสเกิน 8mA ต่อ pin (ต่อ LED โดยไม่มี R) ต่อ R 220Ω-470Ω อนุกรมกับ LED เสมอ
Interrupt ไม่ทำงาน ลืม pinMode(pin, PIN_MODE_INPUT*) ก่อน attachInterrupt ต้องตั้ง pin เป็น INPUT mode ก่อน attachInterrupt
Pin compile error ใช้ pin ที่ไม่มีในแพ็กเกจ (เช่น PD0 บน SOP8) ตรวจสอบ IS_VALID_PIN(pin) หรือดูตาราง pin ด้านบน
analogWrite ไม่มี PWM pin นั้นไม่รองรับ PWM ใช้ IS_PWM_PIN(pin) เช็คก่อน หรือดู pin ที่ระบุ PWM

ตัวอย่างโค้ด

LED Blink 1Hz

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include <SimpleHAL.h>

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    pinMode(PC0, PIN_MODE_OUTPUT);

    while (1) {
        digitalWrite(PC0, HIGH);
        Delay_Ms(500);
        digitalWrite(PC0, LOW);
        Delay_Ms(500);
    }
}

Button + LED (กดติด ปล่อยดับ)

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include <SimpleHAL.h>

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    pinMode(PC0, PIN_MODE_OUTPUT);
    pinMode(PC1, PIN_MODE_INPUT_PULLUP);  // ปุ่มต่อระหว่าง PC1 กับ GND

    while (1) {
        if (digitalRead(PC1) == LOW) {
            digitalWrite(PC0, HIGH);  // กด = ติด
        } else {
            digitalWrite(PC0, LOW);   // ปล่อย = ดับ
        }
    }
}

Interrupt Counter (นับจำนวนครั้งกดปุ่ม)

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include <SimpleHAL.h>

volatile uint16_t count = 0;

void buttonISR() {
    count++;  // นับทุกครั้งที่กด (ขอบลง)
}

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    pinMode(PC0, PIN_MODE_OUTPUT);        // LED แสดงผล
    pinMode(PC1, PIN_MODE_INPUT_PULLUP);  // ปุ่ม
    attachInterrupt(PC1, buttonISR, FALLING);

    while (1) {
        // กระพริบตามจำนวน count
        for (uint16_t i = 0; i < count; i++) {
            digitalWrite(PC0, HIGH);
            Delay_Ms(200);
            digitalWrite(PC0, LOW);
            Delay_Ms(200);
        }
        Delay_Ms(1000);
    }
}

Ultrasonic HC-SR04 วัดระยะ

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include <SimpleHAL.h>

#define TRIG PC0
#define ECHO PC1

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    pinMode(TRIG, PIN_MODE_OUTPUT);
    pinMode(ECHO, PIN_MODE_INPUT);

    while (1) {
        // ส่ง pulse 10µs
        digitalWrite(TRIG, LOW);
        Delay_Us(2);
        digitalWrite(TRIG, HIGH);
        Delay_Us(10);
        digitalWrite(TRIG, LOW);

        // วัด echo
        uint32_t duration = pulseIn(ECHO, HIGH, 25000);
        uint16_t distance = duration / 58;  // cm

        Delay_Ms(100);
    }
}

ความเข้ากันได้กับแพ็กเกจ

แพ็กเกจขา I/OPin ที่ใช้ได้ข้อจำกัด
SOP8 6 PC0, PC1, PC2, PC3, PC4, PC5 PD ทั้งหมดไม่มี; PC6/PC7 = NRST/MCLR
SOP16 14 PA1-PA2, PC0-PC7, PD0, PD1, PD6, PD7 PD2-PD5 ไม่มี
TSSOP20 18 PA1-PA2, PC0-PC7, PD0-PD7 ครบทุก pin
QFN20 18 PA1-PA2, PC0-PC7, PD0-PD7 ครบทุก pin
คำแนะนำ: ใช้ #define CH32V003_PACKAGE PACKAGE_SOP8 (หรือแพ็กเกจที่ใช้จริง) ก่อน #include <SimpleHAL.h> — macro IS_VALID_PIN จะป้องกันการคอมไพล์ผ่านถ้าใช้ pin ที่ไม่มีในแพ็กเกจนั้น
