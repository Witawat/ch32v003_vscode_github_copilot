🤔 ทำไมต้องใช้?

DMA (Direct Memory Access) ถ่ายโอนข้อมูลระหว่าง memory และอุปกรณ์ต่อพ่วงโดย ไม่ใช้ CPU — เหมาะสำหรับงานที่ต้องการ throughput สูง:

  • ADC ต่อเนื่อง — อ่านค่าด้วยความถี่สูงแบบ circular buffer โดย CPU ไม่ต้องคอย polling
  • USART ส่งข้อมูลยาว — ส่ง buffer ขนาดใหญ่โดยไม่ต้องรอทีละไบต์
  • SPI display buffer — ส่ง framebuffer ไปจอโดย CPU ว่างทำงานอื่น
  • Waveform Generator — ใช้ DMA เขียน sine table ไปยัง TIM CCR สร้าง sine wave บน PWM

📚 DMA Channel Map

CH32V003 มี 7 DMA channels — แต่ละ channel ผูกกับอุปกรณ์ต่อพ่วงเฉพาะ:

Channelอุปกรณ์ต่อพ่วงใช้สำหรับ
CH1ADC1ADC continuous conversion
CH2SPI1_RX / USART1_TXSPI รับข้อมูล / USART ส่งข้อมูล
CH3SPI1_TX / USART1_RXSPI ส่งข้อมูล / USART รับข้อมูล
CH4I2C1_TX / TIM1_CH4I2C ส่ง / TIM1 channel 4
CH5I2C1_RX / TIM1_UpdateI2C รับ / TIM1 update event
CH6USART1_RX / TIM1_CH3USART รับ / TIM1 channel 3
CH7USART1_TX / TIM2_CH2USART ส่ง / TIM2 channel 2
Channel Conflict: SPI1_RX กับ USART1_TX ใช้ CH2 ร่วมกัน — ถ้าใช้ SPI และ USART พร้อมกัน ต้องวางแผนให้ดี

ขั้นที่ 2: DMA MemCopy (M2M)

    uint8_t src[256], dst[256];

    // เติม src ด้วยข้อมูลทดสอบ
    for (int i = 0; i < 256; i++)
        src[i] = i;

    // ก็อปปี้ผ่าน DMA — CPU ไม่ต้องทำอะไร
    DMA_MemCopy(dst, src, 256);

🔍 DMA_MemCopy เป็นแบบ blocking — จะ return เมื่อ transfer เสร็จ เร็วกว่าการใช้ CPU copy ทีละไบต์สำหรับ buffer ใหญ่

ขั้นที่ 3: ADC DMA Circular Buffer

    #define BUF_SIZE 100
    uint16_t adc_buf[BUF_SIZE];

    // เริ่ม ADC DMA แบบ circular
    DMA_ADC_Init(adc_buf, BUF_SIZE);
    DMA_analogReadStart(10000);  // sample @ 10kHz

    while (1) {
        // อ่านค่าเฉลี่ยจาก buffer — ไม่ต้อง polling ADC
        uint16_t avg = DMA_analogReadAverage(0);
        USART_Print("ADC Avg: ");
        USART_PrintNum(avg);
        USART_Print("\r\n");
        Delay_Ms(500);
    }

🔍 Circular mode — buffer จะถูกเติมวนไปเรื่อย ๆ โดยอัตโนมัติ DMA_analogReadAverage คำนวณค่าเฉลี่ยจากข้อมูลใน buffer ล่าสุด

🟢 ตัวอย่างขั้นต้น

Fast MemCopy — ใช้ DMA ก็อปปี้ 128 bytes วัดประสิทธิภาพ

#include "simplehal.h"
#include <string.h>

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);

    uint8_t src[128], dst[128];
    memset(dst, 0, 128);

    // เติมข้อมูลทดสอบ
    for (int i = 0; i < 128; i++)
        src[i] = i;

    uint32_t t0 = Get_CurrentMs();
    DMA_MemCopy(dst, src, 128);
    uint32_t t1 = Get_CurrentMs();

    USART_Print("DMA copy 128 bytes: ");
    USART_PrintNum(t1 - t0);
    USART_Print(" ms\r\n");

    // ตรวจสอบผลลัพธ์
    uint8_t ok = 1;
    for (int i = 0; i < 128; i++) {
        if (dst[i] != src[i]) { ok = 0; break; }
    }
    USART_Print("Verify: ");
    USART_Print(ok ? "PASS" : "FAIL");
    USART_Print("\r\n");

    while (1);
}

🟡 ตัวอย่างขั้นกลาง

แนวคิด: ADC Continuous Circular Buffer — อ่าน 100 samples ต่อเนื่อง พร้อมคำนวณค่าเฉลี่ย 
#include "simplehal.h"

#define ADC_BUF_SIZE 100

uint16_t adc_buffer[ADC_BUF_SIZE];

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
    ADC_SimpleInit();

    DMA_ADC_Init(adc_buffer, ADC_BUF_SIZE);
    DMA_analogReadStart(5000);  // 5kHz sampling

    while (1) {
        // Circular DMA จะเติม buffer วนไปเรื่อย ๆ
        uint16_t avg = DMA_analogReadAverage(0);

        // แสดงค่า raw ทั้งหมด
        uint16_t min = 4095, max = 0;
        for (int i = 0; i < ADC_BUF_SIZE; i++) {
            if (adc_buffer[i] < min) min = adc_buffer[i];
            if (adc_buffer[i] > max) max = adc_buffer[i];
        }

        USART_Print("ADC: avg=");
        USART_PrintNum(avg);
        USART_Print(" min=");
        USART_PrintNum(min);
        USART_Print(" max=");
        USART_PrintNum(max);
        USART_Print("\r\n");
        Delay_Ms(200);
    }
}

🔴 ตัวอย่างขั้นสูง

แนวคิด: DMA PWM Sine Wave Generator — เขียน sine table ไปยัง TIM1 CCR ด้วย DMA ที่ update rate สร้าง sine wave บน PWM output 
#include "simplehal.h"
#include <math.h>

#define SINE_SAMPLES 64

// คำนวณ sine table ตอน compile-time (ใช้ math.h, เรียกครั้งเดียว)
uint16_t sine_table[SINE_SAMPLES];

void generateSineTable(void) {
    for (int i = 0; i < SINE_SAMPLES; i++) {
        float rad = 2.0f * 3.14159f * i / SINE_SAMPLES;
        // sine value 0..1 → map to 0..999 (PWM duty 0-100%)
        sine_table[i] = (uint16_t)((sin(rad) + 1.0f) / 2.0f * 999);
    }
}

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();

    generateSineTable();

    // ตั้งค่า PWM บน TIM1 CH1 (PA1) @ 1kHz
    PWM_SimpleInit(TIM_1, TIM_CH1, 1000, 500); // 50% duty เริ่มต้น

    // ใช้ DMA อัปเดต CCR ตาม sine table
    // TIM update event → DMA เขียน sine_table → CCR → sine wave บน PWM output
    DMA_TIM_InitCapture(sine_table, SINE_SAMPLES, TIM_CH1);
    DMA_Start(DMA_CH5);  // TIM1_Update → CH5

    while (1) {
        // CPU ว่างทำอย่างอื่น — DMA จัดการ waveform เอง
    }
}

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
Channel conflict 2 peripherals ต้องการใช้ DMA channel เดียวกัน (เช่น SPI RX + USART TX = CH2) วางแผน channel ล่วงหน้า — หรือใช้ interrupt แทน DMA สำหรับ peripheral ใด peripheral หนึ่ง
ข้อมูลเสียหาย buffer ถูกเขียนทับก่อนที่ CPU จะอ่าน — circular mode วนกลับมาแล้ว ใช้ half-transfer callback — อ่านครึ่งแรกขณะ DMA เขียนครึ่งหลัง (double buffering)
Circular mode ไม่หยุด ตั้ง mode เป็น circular ต้องการ one-shot ใช้ DMA_MODE_NORMAL สำหรับ one-shot transfer
Callback context ผิด เรียก Delay_Ms() หรือ printf() ใน DMA callback → ทำงานใน ISR context DMA callback ทำงานใน ISR — ต้องสั้น รวดเร็ว ห้ามเรียก blocking functions — ใช้ flag ให้ main loop จัดการ
Mem-to-Mem Circular ไม่ทำงาน CH32V003 DMA hardware ไม่รองรับ circular mode สำหรับ memory-to-memory ใช้ peripheral-to-memory หรือ memory-to-peripheral แทน — M2M ใช้เฉพาะ normal mode

💡 FAQ

Q: DMA vs CPU copy — ตัวไหนเร็วกว่า?
A: DMA เหนือกว่า เมื่อ buffer ใหญ่ (มากกว่า 64 bytes) หรือเมื่อต้องการให้ CPU ว่างทำงานอื่น — สำหรับ buffer เล็ก CPU อาจเร็วกว่าเพราะไม่มี overhead ในการตั้งค่า DMA

Q: Circular mode คืออะไร?
A: Circular mode ทำให้ DMA วนกลับไปเขียนที่ต้น buffer เมื่อถึงจุดสิ้นสุด — เหมาะกับ ADC ต่อเนื่องที่ต้องการ ring buffer ของค่าล่าสุด

Q: ใช้ DMA พร้อมกันหลาย channels ได้ไหม?
A: ได้ — แต่ละ channel เป็นอิสระ แต่ priority ถูกกำหนดโดยหมายเลข channel (CH1 สูงสุด) — ระวัง channel conflict

Q: DMA transfer size สูงสุดเท่าไร?
A: Counter register เป็น 16-bit — สูงสุด 65,535 transfers ต่อครั้ง
