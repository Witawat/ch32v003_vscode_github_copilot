🤔 ทำไมต้องใช้?

USART (Universal Synchronous/Asynchronous Receiver-Transmitter) คือช่องทางสื่อสารที่สำคัญที่สุดสำหรับการ debug — แทนที่จะเดาว่า MCU กำลังทำอะไร คุณส่งข้อความออกมาดูบน Serial Monitor ได้เลย นอกจากนี้ยังใช้รับคำสั่งจากคอมพิวเตอร์ ส่งข้อมูลเซนเซอร์ขึ้นกราฟแบบ real-time หรือคุยกับอุปกรณ์อื่น ๆ เช่น Bluetooth, WiFi module, GPS

📚 ภาพรวมและหลักการทำงาน

CH32V003 มี USART 1 ช่อง (USART1) รองรับ baud rate ตั้งแต่ 9600 ถึง 460800 การสื่อสารเป็นแบบ asynchronous serial — ใช้ 2 สาย: TX (ส่งข้อมูลออก) และ RX (รับข้อมูลเข้า) ไม่ต้องมี clock

SimpleUSART จัดการให้:

  • TX — ใช้ blocking send (วนรอจนกว่า transmit register ว่าง) ง่ายและไม่เปลือง RAM
  • RX — ใช้ interrupt-driven ring buffer (ข้อมูลไม่หายแม้ CPU ยุ่งอยู่) ขนาดปริยาย 64 bytes

📋 API Reference

ฟังก์ชันคำอธิบายตัวอย่าง
USART_SimpleInit(baud, pins)เริ่มต้น USART ด้วย baud rate และ pin configUSART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
USART_Print(str)ส่งสตริง (null-terminated) ออกทาง TXUSART_Print("Hello\r\n");
USART_WriteByte(byte)ส่ง 1 ไบต์ออกทาง TX (binary data)USART_WriteByte(0xAA);
USART_PrintNum(num)ส่งตัวเลขแบบ decimal (int32_t) ออกทาง TXUSART_PrintNum(12345);
USART_PrintHex(num, uppercase)ส่งตัวเลขแบบ hex (0x...) ออกทาง TX — uppercase: 1 = ตัวพิมพ์ใหญ่ (A-F), 0 = ตัวพิมพ์เล็ก (a-f)USART_PrintHex(0xAB, 1); // "0xAB"
USART_Available()ตรวจสอบว่ามีข้อมูลรออ่านใน RX buffer หรือไม่ — คืน 1 (มีข้อมูล) หรือ 0 (ไม่มี) เท่านั้น ไม่ใช่จำนวน byteif (USART_Available()) { ... }
USART_Read()อ่าน 1 ไบต์จาก RX buffer — ต้องมั่นใจว่ามีข้อมูลก่อน (เช็ค Available ก่อน)uint8_t c = USART_Read();
USART_ReadBytes(buf, len)อ่านหลายไบต์จาก RX buffer เข้า arrayUSART_ReadBytes(buf, 10);
USART_Flush()ล้าง RX buffer ทิ้งทั้งหมด (เริ่มรับใหม่)USART_Flush();
USART_RxByteHook(byte)ฟังก์ชัน callback — ถูกเรียกทุกครั้งที่มี byte เข้ามาจาก interrupt (ให้ผู้ใช้ override)void USART_RxByteHook(uint8_t b) { ... }

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: ตั้งค่าระบบ + Init USART

#include "SimpleHAL.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();

    // เริ่มต้น USART: baud 115200, ใช้ pin PD5(TX) + PD6(RX)
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);

    USART_Print("CH32V003 USART Ready!\r\n");

    while (1) { }
}

🔍 USART_SimpleInit() ทำทุกอย่าง: เปิด clock ให้ USART1, ตั้งค่า baud rate, เปิด RX interrupt, ตั้งค่า pin TX/RX — เรียกครั้งเดียวจบ

ขั้นที่ 2: ส่งข้อมูล — Print, PrintNum, PrintHex

    USART_Print("=== Demo Send ===\r\n");

    USART_Print("Counter: ");
    uint32_t counter = 0;
    USART_PrintNum(counter);
    USART_Print("\r\n");

    USART_Print("Reg value: ");
    USART_PrintHex(0x5A, 2);
    USART_Print("\r\n");

🔍 USART_Print() รับเฉพาะ string เท่านั้น — ต้องการส่งตัวเลขต้องใช้ USART_PrintNum() หรือ USART_PrintHex() ต่อท้ายด้วย "\r\n" เพื่อขึ้นบรรทัดใหม่ (CR+LF)

ขั้นที่ 3: รับข้อมูล — Available + Read

    while (1) {
        if (USART_Available()) {          // มีข้อมูลรอ?
            uint8_t ch = USART_Read();    // อ่าน 1 ตัวอักษร
            USART_WriteByte(ch);           // echo กลับไป
        }
    }

🔍 ต้องเช็ค USART_Available() ก่อนเสมอ — ถ้าอ่านโดยไม่มีข้อมูล USART_Read() อาจคืนค่า 0 หรือค้าง (ขึ้นกับการ implement)

ขั้นที่ 4: Command Parser — แยกคำสั่งจาก Serial

char cmd_buffer[32];
uint8_t cmd_idx = 0;

while (1) {
    while (USART_Available()) {
        char ch = USART_Read();

        if (ch == '\r' || ch == '\n') {     // กด Enter = ส่งคำสั่ง
            cmd_buffer[cmd_idx] = '\0';     // null-terminate
            ProcessCommand(cmd_buffer);      // แยกวิเคราะห์คำสั่ง
            cmd_idx = 0;
        } else if (cmd_idx < 31) {
            cmd_buffer[cmd_idx++] = ch;     // เก็บตัวอักษร
        }
    }

    // งานอื่น ๆ ใน main loop...
}

void ProcessCommand(char* cmd) {
    if (strcmp(cmd, "LED ON") == 0) {
        digitalWrite(PA1, HIGH);
        USART_Print("LED ON\r\n");
    } else if (strcmp(cmd, "LED OFF") == 0) {
        digitalWrite(PA1, LOW);
        USART_Print("LED OFF\r\n");
    } else {
        USART_Print("Unknown command: ");
        USART_Print(cmd);
        USART_Print("\r\n");
    }
}

🔍 สร้าง buffer เก็บตัวอักษรที่พิมพ์เข้ามา — เมื่อเจอ Enter (CR/LF) ให้นำ buffer ไป parse หาคำสั่งที่รู้จัก ใช้วิธีนี้สร้าง command-line interface บน Serial Monitor ได้

🟡 ตัวอย่างขั้นกลาง

Command Line Receiver — รับคำสั่ง "LON" / "LOFF" จาก Serial Monitor มาควบคุม LED

#include "SimpleHAL.h"
#include <string.h>

char rx_buffer[32];
uint8_t rx_idx = 0;

void ProcessCmd(char* cmd) {
    if (strcmp(cmd, "LON") == 0) {
        digitalWrite(PA1, HIGH);
        USART_Print("LED is ON\r\n");
    } else if (strcmp(cmd, "LOFF") == 0) {
        digitalWrite(PA1, LOW);
        USART_Print("LED is OFF\r\n");
    } else if (strcmp(cmd, "HELP") == 0) {
        USART_Print("Commands: LON, LOFF, HELP\r\n");
    } else {
        USART_Print("Unknown: ");
        USART_Print(cmd);
        USART_Print("\r\n");
    }
}

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();

    pinMode(PA1, PIN_MODE_OUTPUT);
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);

    USART_Print("CH32V003 Command Line Ready\r\n");
    USART_Print("Type HELP for commands\r\n");
    USART_Print("> ");

    while (1) {
        while (USART_Available()) {
            char ch = USART_Read();
            if (ch == '\r' || ch == '\n') {
                if (rx_idx > 0) {
                    rx_buffer[rx_idx] = '\0';
                    ProcessCmd(rx_buffer);
                    rx_idx = 0;
                    USART_Print("> ");
                }
            } else if (rx_idx < 31) {
                rx_buffer[rx_idx++] = ch;
                USART_WriteByte(ch);  // echo
            }
        }
    }
}

🔴 ตัวอย่างขั้นสูง

Binary Protocol with Checksum — ส่ง/รับข้อมูลเป็นเฟรมที่มี header 0xAA 0x55 + length + payload + XOR checksum เหมาะกับระบบที่ต้องการความน่าเชื่อถือสูง (หุ่นยนต์, โดรน)

#include "SimpleHAL.h"

#define FRAME_HEADER1  0xAA
#define FRAME_HEADER2  0x55

typedef struct {
    uint8_t header1;
    uint8_t header2;
    uint8_t length;
    uint8_t command;
    uint8_t data[16];
    uint8_t checksum;
} Frame_t;

uint8_t CalcXOR(uint8_t* buf, uint8_t len) {
    uint8_t xor_result = 0;
    for (uint8_t i = 0; i < len; i++) {
        xor_result ^= buf[i];
    }
    return xor_result;
}

void SendFrame(Frame_t* frame) {
    frame->header1 = FRAME_HEADER1;
    frame->header2 = FRAME_HEADER2;

    uint8_t* p = (uint8_t*)frame;
    frame->checksum = CalcXOR(p, frame->length + 3);

    for (uint8_t i = 0; i < frame->length + 4; i++) {
        USART_WriteByte(p[i]);
    }
}

uint8_t ReceiveFrame(Frame_t* frame) {
    static uint8_t sync = 0;

    while (USART_Available()) {
        uint8_t b = USART_Read();

        if (!sync) {
            if (b == FRAME_HEADER1) sync = 1;
            continue;
        }
        if (sync == 1) {
            if (b == FRAME_HEADER2) sync = 2;
            else sync = 0;
            continue;
        }

        // Got header — read rest of frame
        frame->header1 = FRAME_HEADER1;
        frame->header2 = FRAME_HEADER2;
        frame->length = b;

        uint8_t remain = frame->length;
        uint8_t* p = &frame->command;
        for (uint8_t i = 0; i < remain; i++) {
            while (!USART_Available());       // wait for next byte
            p[i] = USART_Read();
        }

        while (!USART_Available());
        frame->checksum = USART_Read();

        // Verify checksum
        uint8_t* raw = (uint8_t*)frame;
        uint8_t calc = CalcXOR(raw, frame->length + 3);
        sync = 0;

        if (calc == frame->checksum) return 1;  // OK
        return 0;                                // checksum error
    }
    return 0xFF;  // no complete frame yet
}

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();

    pinMode(PA1, PIN_MODE_OUTPUT);
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);

    USART_Print("Binary Protocol Ready\r\n");

    Frame_t txFrame;
    txFrame.length = 1;
    txFrame.command = 0x01;       // LED command

    while (1) {
        txFrame.data[0] = 1;     // LED ON
        SendFrame(&txFrame);
        Delay_Ms(1000);

        txFrame.data[0] = 0;     // LED OFF
        SendFrame(&txFrame);
        Delay_Ms(1000);
    }
}

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
Serial Monitor แสดงตัวอักษรแปลก / baud rate ผิด ลืมเรียก SystemCoreClockUpdate() — USART baud rate generator ใช้ค่า clock ผิด วาง SystemCoreClockUpdate() เป็นบรรทัดแรกสุดใน main() เสมอ
USART_Read() คืนค่า 0 ตลอด เรียก USART_Read() โดยไม่เช็ค USART_Available() ก่อน — ไม่มีข้อมูลให้อ่าน เช็คด้วย if (USART_Available()) ก่อนเรียก USART_Read()
ข้อมูลหายเมื่อรับเร็วเกิน RX ring buffer เต็ม — ขนาดปริยาย 64 bytes อาจไม่พอ เพิ่ม USART_RX_BUFFER_SIZE ใน simplehal_config.h (เช่น 256) หรืออ่านข้อมูลบ่อยขึ้น
CH32V003 มี USART ช่องเดียว — ต้องการใช้หลายอุปกรณ์ USART1 ใช้ได้ครั้งละ 1 ชุดของ pin — แต่ remap ช่วยเลือกชุด pin ได้ ใช้ remap หรือใช้ SoftwareSerial (bit-banging) สำหรับอุปกรณ์ที่ไม่ต้องการความเร็ว

💡 เกร็ดความรู้ / FAQ

Q: ใช้โปรแกรมอะไรเป็น Serial Monitor?
A: MounRiver Studio มี Serial Monitor ในตัว (Window → Show View → Terminal) หรือใช้ PuTTY (ฟรี), Arduino IDE Serial Monitor, SerialPlot (พล็อตกราฟ real-time) — ตั้ง baud rate ให้ตรงกับในโค้ด 115200 8N1

Q: RX ring buffer คืออะไร?
A: เมื่อมีข้อมูลเข้ามาทาง RX interrupt จะถูกเก็บลง buffer วงกลม (ring/circular buffer) โดยอัตโนมัติ — โปรแกรมหลักสามารถอ่านเมื่อไรก็ได้ ข้อมูลไม่หายระหว่างที่ CPU ยุ่งอยู่ ขนาดปริยาย 64 bytes ถ้าอ่านออกไม่ทัน buffer ล้น ข้อมูลเก่าจะถูกเขียนทับ

Q: ทำไมต้องลงท้าย string ด้วย \r\n?
A: \r (carriage return) เลื่อน cursor ไปต้นบรรทัด \n (line feed) เลื่อน cursor ลงบรรทัดใหม่ — Terminal ส่วนใหญ่ต้องการทั้งคู่เพื่อแสดงผลถูกต้อง ถ้าใช้แค่ \n ข้อความจะไม่ขึ้นบรรทัดใหม่

Q: USART_RxByteHook ใช้ยังไง?
A: ฟังก์ชัน USART_RxByteHook(uint8_t byte) เป็น weak function — คุณ override ได้โดยเขียนฟังก์ชันชื่อเดียวกันในโค้ดของคุณ มันจะถูกเรียกทุกครั้งที่มี byte ใหม่เข้ามาจาก interrupt ใช้สำหรับ: dump hex แบบ real-time, ตรวจจับ byte พิเศษ, หรือสร้าง protocol parser แบบ state machine

✏️ ลองทำดู [เฉลย]

  1. Echo Server — ส่งกลับทุกตัวอักษรที่รับเข้ามา (ตัวใหญ่ ตัวเล็ก ตัวเลข) — ห้ามใช้ USART_Print ทั้งสตริง ต้องใช้ USART_WriteByte ทีละตัว
  2. LED PWM ควบคุมด้วย Serial — รับคำสั่ง LED=50 ผ่าน Serial Monitor เพื่อตั้งความสว่าง LED (0–100%) ด้วย software PWM โดยใช้ atoi() แปลงข้อความเป็นตัวเลข
  3. อ่าน ADC ส่งทาง Serial ทุก 100ms — ใช้ ADC อ่านค่าแรงดัน (เช่นจาก potentiometer) แล้วส่งค่า raw + voltage ผ่าน USART ทุก 100ms ด้วย non-blocking timer แสดงผลเป็นตารางตัวเลข
