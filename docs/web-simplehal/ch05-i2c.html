บทที่ 5: I2C — สื่อสารกับ Sensor และ EEPROM

ระดับกลาง

สิ่งที่ต้องรู้ก่อน

  • บทที่ 4: USART (debug print)
  • บทที่ 2: GPIO

อุปกรณ์ที่ต้องใช้

  • CH32V003 Dev Board
  • ตัวต้านทาน 4.7kΩ × 2 (pull-up)
  • เซนเซอร์ I2C ใด ๆ (MPU6050, BH1750, OLED)

ทำไมต้องใช้?

เซนเซอร์ส่วนใหญ่ใช้ I2C — อ่านอุณหภูมิ ความเร่ง แสง จอ OLED — ใช้แค่ 2 สายเท่านั้น

API Reference

ฟังก์ชันคำอธิบาย
I2C_SimpleInit()เริ่มต้น I2C hardware
I2C_Write()เขียนข้อมูลไปยัง device
I2C_Read()อ่านข้อมูลจาก device
I2C_WriteReg()เขียนค่า register
I2C_ReadReg()อ่านค่า register
I2C_TryReadReg()ลองอ่าน register (มี error handling)
I2C_WriteRegMulti()เขียนหลาย register ต่อเนื่อง
I2C_ReadRegMulti()อ่านหลาย register ต่อเนื่อง
I2C_Scan()สแกนหา device บน bus
I2C_IsDeviceReady()ตรวจสอบว่า device พร้อมหรือไม่

Status Codes

รหัสความหมาย
I2C_OKสำเร็จ
I2C_ERROR_TIMEOUTการสื่อสารใช้เวลานานเกิน
I2C_ERROR_NACKDevice ไม่ตอบรับ (NACK)
I2C_ERROR_BUS_BUSYBus ติดค้าง ถูกจองอยู่
I2C_ERROR_PARAMพารามิเตอร์ไม่ถูกต้อง

ขั้นตอนการใช้งาน (Step by Step)

1. เริ่มต้น I2C

I2C_SimpleInit(I2C_100KHZ, I2C_PINS_DEFAULT) — อย่าลืมต่อ pull-up resistors

2. สแกน Bus

I2C_Scan() — หา address ของอุปกรณ์ที่ต่ออยู่

3. อ่าน Register

I2C_ReadReg(addr, reg) — อ่านค่าจาก register ที่ต้องการ

4. เขียน Register

I2C_WriteReg(addr, reg, data) — เขียนค่าลง register

5. Multi-byte

I2C_ReadRegMulti() / I2C_WriteRegMulti() — อ่าน/เขียนหลายไบต์ต่อเนื่อง

ระดับขั้นต้น: Read/Write AT24C32 EEPROM

#include "SimpleHAL.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
    I2C_SimpleInit(I2C_100KHZ, I2C_PINS_DEFAULT);

    uint8_t write_data = 0x42;
    I2C_WriteReg(0x50, 0x00, write_data);

    Delay_Ms(5);

    uint8_t read_data = I2C_ReadReg(0x50, 0x00);
    USART_Print("EEPROM[0x00] = 0x");
    USART_PrintHex(read_data, 0);
    USART_Print("\r\n");

    while (1);
}

ระดับขั้นกลาง: อ่าน MPU6050 Accelerometer

แนวคิด: ปลุก MPU6050 จาก sleep mode แล้วอ่านค่า accelerometer X/Y/Z (6 ไบต์) 
#include "SimpleHAL.h"

#define MPU6050_ADDR 0x68

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
    I2C_SimpleInit(I2C_100KHZ, I2C_PINS_DEFAULT);

    // ปลุก MPU6050 จาก sleep
    I2C_WriteReg(MPU6050_ADDR, 0x6B, 0x00);
    Delay_Ms(10);

    int16_t accel[3];
    while (1) {
        uint8_t raw[6];
        I2C_ReadRegMulti(MPU6050_ADDR, 0x3B, raw, 6);

        accel[0] = (raw[0] << 8) | raw[1]; // X
        accel[1] = (raw[2] << 8) | raw[3]; // Y
        accel[2] = (raw[4] << 8) | raw[5]; // Z

        USART_Print("X=");
        USART_PrintNum(accel[0]);
        USART_Print(" Y=");
        USART_PrintNum(accel[1]);
        USART_Print(" Z=");
        USART_PrintNum(accel[2]);
        USART_Print("\r\n");
        Delay_Ms(500);
    }
}

ระดับขั้นสูง: Multi-device I2C พร้อม Error Recovery

แนวคิด: สแกน bus ก่อนใช้งาน, ใช้ I2C_TryReadReg() ป้องกัน NACK, retry เมื่อ fail 
#include "SimpleHAL.h"

void safe_read(uint8_t addr, uint8_t reg, uint8_t retries) {
    uint8_t data;
    for (uint8_t i = 0; i < retries; i++) {
        if (I2C_TryReadReg(addr, reg, &data) == I2C_OK) {
            USART_Print("Addr 0x");
            USART_PrintHex(addr, 0);
            USART_Print(" Reg 0x");
            USART_PrintHex(reg, 0);
            USART_Print(" = 0x");
            USART_PrintHex(data, 0);
            USART_Print("\r\n");
            return;
        }
        USART_Print("Retry ");
        USART_PrintNum(i + 1);
        USART_Print("...\r\n");
        Delay_Ms(10);
    }
    USART_Print("Failed after ");
    USART_PrintNum(retries);
    USART_Print(" retries\r\n");
}

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
    I2C_SimpleInit(I2C_100KHZ, I2C_PINS_DEFAULT);

    USART_Print("--- I2C Bus Scan ---\r\n");
    for (uint8_t addr = 1; addr < 127; addr++) {
        if (I2C_IsDeviceReady(addr)) {
            USART_Print("Found device at 0x");
            USART_PrintHex(addr, 0);
            USART_Print("\r\n");
        }
    }

    safe_read(0x68, 0x75, 3); // MPU6050 WHO_AM_I
    while (1);
}

ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
NACK ตลอดเวลาAddress ผิด หรือไม่มี pull-upเช็ค address ด้วย I2C_Scan, ต่อ 4.7kΩ pull-up
Bus stuckSCL ค้างต่ำClock 9 ครั้งเพื่อ release bus
400kHz เพี้ยนสายยาวเกินไปลดความเร็วลง หรือใช้สายสั้น
I2C_ReadReg คืน 0xFFDevice ไม่ตอบสนองใช้ I2C_TryReadReg แทน

คำถามที่พบบ่อย (FAQ)

7-bit vs 8-bit address ต่างกันอย่างไร?

SimpleHAL ใช้ 7-bit address — ระบบจะเลื่อนบิตให้เอง ไม่ต้องเลื่อนซ้ายเอง

Pull-up จำเป็นไหม?

จำเป็น! ถ้าไม่มี pull-up resistor bus จะไม่ทำงานเลย เพราะ CH32V003 ไม่มี internal pull-up ที่แรงพอ

Hardware I2C vs Software I2C ใช้อะไรดี?

Hardware I2C (บทนี้) เร็วกว่า ไม่กิน CPU — Software I2C ดูที่ บทที่ 6
