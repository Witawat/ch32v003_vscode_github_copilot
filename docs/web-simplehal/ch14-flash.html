Flash Layout

Pageขนาดการใช้งาน
Page 25464 bytesConfig (ตั้งค่าผู้ใช้)
Page 25564 bytesData (ข้อมูลทั่วไป)
หมายเหตุ: ใช้งานได้จริง ~60 bytes (หัก CRC overhead) — Endurance ~10,000 รอบ

ขั้นตอนการใช้งาน (Step by Step)

1. เริ่มต้น Flash

Flash_Init() — เรียกครั้งแรกใน main

2. โหลด Config

Flash_LoadConfig(&my_config, sizeof(my_config)) — ถ้า CRC OK จะคืน 1

3. ตรวจสอบผล

ถ้าโหลดไม่สำเร็จ → ใช้ค่า default แล้วบันทึกทับด้วย Flash_SaveConfig

4. บันทึกเมื่อมีการเปลี่ยนแปลง

Flash_SaveConfig(&my_config, sizeof(my_config)) — เรียกเฉพาะเมื่อข้อมูลเปลี่ยน

ระดับขั้นต้น: Save/Load Config Struct

#include "SimpleHAL.h"

typedef struct {
    uint8_t brightness;
    uint16_t interval_ms;
    uint8_t uart_enabled;
} AppConfig;

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(USART_115200);
    Flash_Init();

    AppConfig cfg;
    if (!Flash_LoadConfig(&cfg, sizeof(cfg))) {
        // ยังไม่มี config → ใช้ default
        cfg.brightness = 50;
        cfg.interval_ms = 1000;
        cfg.uart_enabled = 1;
        Flash_SaveConfig(&cfg, sizeof(cfg));
        printf("ใช้ค่า default\r\n");
    } else {
        printf("โหลด config สำเร็จ\r\n");
    }

    printf("Brightness: %d\r\n", cfg.brightness);
    printf("Interval: %d ms\r\n", cfg.interval_ms);
    printf("UART: %s\r\n", cfg.uart_enabled ? "ON" : "OFF");

    while (1);
}

ระดับขั้นกลาง: Config Manager พร้อม Version Check

แนวคิด: เพิ่ม version field ใน struct — ถ้า version ไม่ตรง → รีเซ็ตเป็นค่า default อัตโนมัติ 
#include "SimpleHAL.h"

#define CONFIG_VERSION 2

typedef struct {
    uint8_t version;
    uint8_t brightness;
    uint16_t interval_ms;
    uint8_t uart_enabled;
} AppConfig;

void config_load(AppConfig *cfg) {
    if (!Flash_LoadConfig(cfg, sizeof(*cfg))) {
        goto defaults;
    }
    if (cfg->version != CONFIG_VERSION) {
        printf("Version mismatch: v%d → v%d\r\n", cfg->version, CONFIG_VERSION);
        goto defaults;
    }
    return;

defaults:
    cfg->version = CONFIG_VERSION;
    cfg->brightness = 50;
    cfg->interval_ms = 1000;
    cfg->uart_enabled = 1;
    Flash_SaveConfig(cfg, sizeof(*cfg));
    printf("ใช้ค่า default\r\n");
}

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(USART_115200);
    Flash_Init();

    AppConfig cfg;
    config_load(&cfg);
    printf("Brightness=%d Interval=%d\r\n", cfg.brightness, cfg.interval_ms);

    // เปลี่ยน config เมื่อกดปุ่ม
    cfg.brightness += 10;
    Flash_SaveConfig(&cfg, sizeof(cfg));

    while (1);
}

ระดับขั้นสูง: Wear Leveling — บันทึก Counter ทุก 60 วินาที

แนวคิด: ใช้ counter นับจำนวนครั้งที่เขียน — หลีกเลี่ยงการเขียนซ้ำที่อยู่เดิม ๆ เพื่อยืดอายุ Flash (สูงสุด ~10,000 รอบ/หน้า) 
#include "SimpleHAL.h"

typedef struct {
    uint32_t write_count;
    uint32_t total_runtime_s;
    uint8_t led_brightness;
} PersistentData;

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(USART_115200);
    Flash_Init();

    PersistentData data;
    if (!Flash_LoadConfig(&data, sizeof(data))) {
        data.write_count = 0;
        data.total_runtime_s = 0;
        data.led_brightness = 50;
        Flash_SaveConfig(&data, sizeof(data));
    }

    printf("Write count: %lu, Runtime: %lus\r\n",
           data.write_count, data.total_runtime_s);

    uint32_t last_save = Get_CurrentMs();
    while (1) {
        if (Get_CurrentMs() - last_save >= 60000) {
            last_save = Get_CurrentMs();
            data.total_runtime_s += 60;
            data.write_count++;
            Flash_SaveConfig(&data, sizeof(data));
            printf("Saved: count=%lu runtime=%lus\r\n",
                   data.write_count, data.total_runtime_s);
        }
        // งานอื่น ๆ...
    }
}

ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
CRC error → ใช้ defaultFlash ว่างเปล่าหรือข้อมูลเสียบันทึกค่า default ใหม่
ข้อมูลผิดพลาดInvalid → ไม่เคยถูกบันทึกเรียก Flash_SaveConfig ครั้งแรก
Flash เสื่อมเร็วเขียนบ่อยเกิน ~10k cyclesบันทึกเฉพาะเมื่อข้อมูลเปลี่ยน
ข้อมูลเกิน 60 bytesStruct ใหญ่เกินหน้าเดียวแยก struct หรือบีบอัดข้อมูล
ข้อมูลหายหลังเขียนInterrupt ขัดจังหวะระหว่างเขียนอัปเดตเป็น SimpleHAL v2.1+ (IRQ-safe)

คำถามที่พบบ่อย (FAQ)

Flash vs EEPROM ต่างกันอย่างไร?

Flash ในตัวชิป — ฟรี ไม่ต้องซื้อเพิ่ม แต่ endurance ต่ำกว่า (~10k รอบ) ส่วน EEPROM ภายนอก endurance สูง (~1M รอบ) แต่ต้องซื้อเพิ่ม

ข้อมูลเก็บได้นานแค่ไหน?

~10+ ปี ที่อุณหภูมิใช้งานปกติ

เขียนบ่อยได้ไหม?

ไม่ควรเกิน 10,000 ครั้งต่อหน้า — สำหรับ counter ที่เขียนทุกวินาที ให้ใช้วิธีทอนรอบ (บันทึกทุก 60 วินาที) หรือใช้ wear leveling

ลองทำดู (แบบฝึกหัด)

  1. บันทึกและโหลด boolean config (เปิด/ปิด LED) ข้าม reset
  2. Counter ที่คงค่าแม้ปิด-เปิดเครื่อง — นับจำนวนครั้งที่เปิดเครื่อง
  3. Factory reset button — ล้าง Flash กลับเป็นค่า default เมื่อกดปุ่มค้าง 5 วินาที
