ขั้นตอนการใช้งาน (Step by Step)

1. เลือกโหมดที่เหมาะสม

Sleep: idle สั้น (<30s), RAM ยังอยู่ — Standby: idle นาน (>30s), ต้อง save state ลง Flash

2. ปิด peripheral ที่ไม่ใช้

ปิด ADC, TIM, USART ที่ไม่จำเป็นก่อนเข้า sleep — ลดกระแสได้มาก

3. เรียก Sleep/Standby

PWR_Sleep() หรือ PWR_Standby(timeout)

4. จัดการ wakeup

Sleep — ต่อจากบรรทัดที่เรียกต่อได้เลย / Standby — เริ่ม main ใหม่เหมือนเปิดเครื่อง

ระดับขั้นต้น: Sleep เมื่อ Idle

#include "SimpleHAL.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(USART_115200);

    // ตั้ง TIM1 interrupt ทุก 1 วินาทีเป็น wakeup source
    TIM_SimpleInit(TIM1, 1000); // 1Hz

    while (1) {
        printf("Wake up!\r\n");

        // Flush USART buffer ก่อน sleep
        while (USART_GetFlagStatus(USART1, USART_FLAG_TC) == RESET);

        PWR_Sleep(); // หลับรอ TIM1 interrupt
        // ตื่นต่อจากตรงนี้
    }
}

ระดับขั้นกลาง: Standby + AWU — Deep Sleep 10 วินาที

แนวคิด: ใช้ Flash บันทึก ADC counter ก่อนเข้า Standby — หลังตื่นอ่านค่ากลับมา — อ่าน ADC ใหม่ — สะสมต่อ 
#include "SimpleHAL.h"

typedef struct {
    uint32_t adc_samples;
    uint32_t boot_count;
} StateData;

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    Flash_Init();
    ADC_SimpleInit();

    StateData state = {0};
    if (Flash_LoadConfig(&state, sizeof(state))) {
        printf("Standby wakeup #%lu\r\n", state.boot_count);
    }

    state.boot_count++;
    state.adc_samples++;

    uint16_t adc_val = ADC_Read();
    printf("ADC: %d, Samples: %lu\r\n", adc_val, state.adc_samples);

    // บันทึกก่อนเข้า Standby
    Flash_SaveConfig(&state, sizeof(state));

    // เข้า Standby 10 วินาที
    printf("Entering standby for 10s...\r\n");
    Delay_Ms(50); // รอให้ USART ส่งให้หมด
    PWR_Standby(10000); // ตื่นด้วย AWU 10 วิ
    // ไม่มาถึงบรรทัดนี้ — MCU reset แล้วเริ่ม main ใหม่
}

ระดับขั้นสูง: Smart Sleep — เลือกโหมดตามระยะเวลา

แนวคิด: Idle สั้น (< 30s) → Sleep (ตื่นเร็ว) — Idle นาน (> 30s) → Standby (ประหยัดสุด) — Button EXTI wakeup 
#include "SimpleHAL.h"

#define IDLE_THRESHOLD_MS 30000

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    Flash_Init();
    ADC_SimpleInit();
    GPIO_Init(GPIOB, GPIO_Pin_1, GPIO_Mode_IPU);

    uint32_t idle_start = Get_CurrentMs();

    while (1) {
        // ทำงานปกติ — สั้นมาก
        uint16_t adc = ADC_Read();
        printf("ADC: %d\r\n", adc);

        idle_start = Get_CurrentMs();

        // Sleep 1 วินาที
        TIM_SimpleInit(TIM1, 1000);
        while (USART_GetFlagStatus(USART1, USART_FLAG_TC) == RESET);
        PWR_Sleep();

        // ตรวจสอบว่า idle นานเท่าไหร่
        if (Get_CurrentMs() - idle_start > IDLE_THRESHOLD_MS) {
            printf("Long idle → Standby\r\n");
            Delay_Ms(50);
            // Standby รอปุ่ม EXTI
            PWR_StandbyUntilInterrupt();
            // Reset → เริ่มใหม่
        }
    }
}

Battery Life Estimation

#include "SimpleHAL.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(USART_115200);

    // 1000mAh, 1% active, 5mA active, 5uA standby
    float days = PWR_CalculateBatteryLife(1000, 1, 5, 5);
    printf("Battery life: ~%.0f days\r\n", days);
    // ~200 days

    while (1);
}

ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
ตื่นจาก Standby แล้วค้างไม่ได้ clear wakeup flagเรียก PWR_ClearStandbyFlag()
RAM หายหลัง StandbyStandby ปิดหมด — RAM ไม่คงอยู่Save state to Flash ก่อนเข้า Standby
Sleep ไม่ประหยัดADC/TIM ยังทำงานอยู่ปิด peripheral ก่อนเรียก PWR_Sleep
AWU timeout ไม่แม่นยำLSI ±25% toleranceเผื่อเวลา margin 30%

คำถามที่พบบ่อย (FAQ)

Sleep vs Standby ต่างกันอย่างไร?

Sleep — ตื่นเร็ว RAM ยังอยู่ (1-2mA) / Standby — ประหยัดสุดแต่ RAM หาย ต้อง boot ใหม่ (2-5µA)

แบตเตอรี่ 1000mAh ใช้ได้นานแค่ไหน?

Active 1% ของเวลา, Sleep ตลอด → ~200 วัน (คำนวณด้วย PWR_CalculateBatteryLife)

WKUP pin (PA0) ใช้ใน CH32V003 ได้หรือไม่?

CH32V003 ไม่มี WKUP pin โดยเฉพาะ — ใช้ EXTI แทนได้ สำหรับ wake from Standby

ลองทำดู (แบบฝึกหัด)

  1. Sleep 1s แล้วตื่นด้วย TIM1 interrupt → กระพริบ LED
  2. Standby 10s ด้วย AWU — ตื่นอ่าน ADC ใหม่ทุกครั้ง
  3. คำนวณอายุแบตเตอรี่ด้วย PWR_CalculateBatteryLife สำหรับอุปกรณ์ IoT
