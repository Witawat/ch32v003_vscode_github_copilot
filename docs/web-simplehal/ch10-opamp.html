🤔 ทำไมต้องใช้?

OPAMP ภายใน CH32V003 ช่วยให้คุณ ขยายสัญญาณอ่อน จากเซนเซอร์ (เช่น pH, thermocouple), buffer impedance ให้สัญญาณไม่เพี้ยนเมื่อขับโหลด, หรือใช้เป็น comparator เปรียบเทียบแรงดัน — ทั้งหมดนี้ ไม่ต้องใช้ IC ภายนอก ลดต้นทุน BOM และพื้นที่บอร์ด

📋 API Reference

ฟังก์ชันคำอธิบายตัวอย่าง
OPAMP_SimpleInit(mode)เริ่มต้น OPAMP ด้วยโหมดที่ระบุOPAMP_SimpleInit(OPAMP_MODE_VOLTAGE_FOLLOWER);
OPAMP_Enable()เปิดการทำงาน OPAMPOPAMP_Enable();
OPAMP_Disable()ปิด OPAMP (ประหยัดพลังงาน)OPAMP_Disable();
OPAMP_IsEnabled()ตรวจสอบว่า OPAMP เปิดอยู่หรือไม่if (OPAMP_IsEnabled()) { ... }
OPAMP_SetMode(mode)เปลี่ยนโหมด OPAMP ระหว่างทำงานOPAMP_SetMode(OPAMP_MODE_COMPARATOR);
OPAMP_GetConfig(pos, neg)อ่านค่า channel configuration ปัจจุบันผ่าน out-pointerOPAMP_Channel_Positive pos; OPAMP_Channel_Negative neg; OPAMP_GetConfig(&pos, &neg);
OPAMP_Init(chp, chn)เริ่มต้นด้วยการเลือก channel เอง (ไม่รวมโหมด — ใช้ OPAMP_SetMode() แยกถ้าต้องการ)OPAMP_Init(OPAMP_CHP0, OPAMP_CHN0);
OPAMP_SetChannels(chp, chn)เปลี่ยน channel inputOPAMP_SetChannels(CHP0, CHN0);
OPAMP_ConfigVoltageFollower()ตั้งค่าเป็น Voltage Follower สำเร็จรูปOPAMP_ConfigVoltageFollower();
OPAMP_ConfigNonInverting()ตั้งค่าโหมด Non-InvertingOPAMP_ConfigNonInverting();
OPAMP_ConfigInverting()ตั้งค่าโหมด InvertingOPAMP_ConfigInverting();
OPAMP_ConfigComparator()ตั้งค่าโหมด ComparatorOPAMP_ConfigComparator();
OPAMP_CalculateGainNonInv(Rf, Rin)คำนวณ gain ของวงจร Non-Invertingfloat g = OPAMP_CalculateGainNonInv(10000, 10000); // 2.0
OPAMP_CalculateGainInv(Rf, Rin)คำนวณ gain ของวงจร Invertingfloat g = OPAMP_CalculateGainInv(10000, 10000); // 1.0
OPAMP_CalculateR2NonInv(R1, gain)คำนวณค่า Rf เมื่อทราบ gain และ Rinuint32_t rf = OPAMP_CalculateR2NonInv(10000, 2.0); // 10000
OPAMP_CalculateR2Inv(R1, gain)คำนวณค่า Rf สำหรับ Invertinguint32_t rf = OPAMP_CalculateR2Inv(10000, 2.0); // 20000
สำคัญ: ฟังก์ชัน OPAMP_CalculateGain* และ OPAMP_CalculateR2* ใช้ float — ให้เรียกเฉพาะตอน init (ก่อน while loop) เท่านั้น ห้ามเรียกใน main loop เพราะ CH32V003 ไม่มี FPU การคำนวณ float ใน loop จะกิน CPU มาก

ขั้นที่ 3: อ่านสัญญาณผ่าน ADC

    // Non-Inverting gain = 2 ด้วย R1=R2=10kΩ

🟢 ตัวอย่างขั้นต้น

Voltage Follower — Buffer สัญญาณเพื่อ read ผ่าน ADC ได้อย่างแม่นยำโดยไม่ดึงกระแสจากแหล่งสัญญาณ

#include "simplehal.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);

    OPAMP_SimpleInit(OPAMP_MODE_VOLTAGE_FOLLOWER);
    OPAMP_Enable();
    ADC_SimpleInit();

    while (1) {
        uint16_t raw = ADC_Read(PA2);
        float voltage = raw * 3.3f / 4095.0f;

        // SimpleHAL ไม่มีฟังก์ชันพิมพ์ float — แปลง voltage เป็นทศนิยม 3 ตำแหน่งด้วยจำนวนเต็ม
        int32_t v_x1000 = (int32_t)(voltage * 1000.0f);
        USART_Print("Vout = ");
        USART_PrintNum(v_x1000 / 1000);
        USART_Print(".");
        USART_PrintNum(v_x1000 % 1000);
        USART_Print(" V\r\n");
        Delay_Ms(500);
    }
}

🟡 ตัวอย่างขั้นกลาง

แนวคิด: ใช้ OPAMP เป็น Comparator — เปรียบเทียบแรงดัน PA2 (V+) กับ PA1 (V−) ถ้า V+ > V− ให้ output HIGH 
#include "simplehal.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();

    pinMode(PC0, PIN_MODE_OUTPUT);  // LED แสดงผล

    OPAMP_SimpleInit(OPAMP_MODE_COMPARATOR);
    OPAMP_Enable();

    while (1) {
        if (OPAMP_IsEnabled()) {
            // V+ (PA2) > V− (PA1) = output goes high
            uint16_t vp = ADC_Read(PA2);
            uint16_t vn = ADC_Read(PA1);

            if (vp > vn) {
                digitalWrite(PC0, HIGH);  // LED ติด
            } else {
                digitalWrite(PC0, LOW);   // LED ดับ
            }
        }
        Delay_Ms(100);
    }
}

🔴 ตัวอย่างขั้นสูง

แนวคิด: Non-Inverting Amplifier Gain = 2 — ขยายสัญญาณ 2 เท่า ด้วย R1 = R2 = 10kΩ ภายนอก 
#include "simplehal.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
    ADC_SimpleInit();

    // คำนวณ gain ตอน init (float OK — เรียกครั้งเดียว)
    float gain = OPAMP_CalculateGainNonInv(10000, 10000);

    // SimpleHAL ไม่มีฟังก์ชันพิมพ์ float — แปลง gain เป็นทศนิยม 2 ตำแหน่งด้วยจำนวนเต็ม
    int32_t gain_x100 = (int32_t)(gain * 100.0f);
    USART_Print("Configured Gain: ");
    USART_PrintNum(gain_x100 / 100);
    USART_Print(".");
    USART_PrintNum(gain_x100 % 100);
    USART_Print("\r\n");

    OPAMP_SimpleInit(OPAMP_MODE_NON_INVERTING);
    OPAMP_Enable();

    while (1) {
        uint16_t raw = ADC_Read(PA2);
        float vout = raw * 3.3f / 4095.0f;

        // Vin = Vout / Gain (คำนวณย้อนกลับ)
        float vin = vout / gain;
        float actual_gain = vout / vin;

        // SimpleHAL ไม่มีฟังก์ชันพิมพ์ float — แปลง vin/vout (3 ตำแหน่ง) และ gain (2 ตำแหน่ง) เป็นจำนวนเต็ม
        int32_t vin_x1000 = (int32_t)(vin * 1000.0f);
        int32_t vout_x1000 = (int32_t)(vout * 1000.0f);
        int32_t gain_x100 = (int32_t)(actual_gain * 100.0f);
        USART_Print("Vin=");
        USART_PrintNum(vin_x1000 / 1000);
        USART_Print(".");
        USART_PrintNum(vin_x1000 % 1000);
        USART_Print("V  Vout=");
        USART_PrintNum(vout_x1000 / 1000);
        USART_Print(".");
        USART_PrintNum(vout_x1000 % 1000);
        USART_Print("V  Gain=");
        USART_PrintNum(gain_x100 / 100);
        USART_Print(".");
        USART_PrintNum(gain_x100 % 100);
        USART_Print("\r\n");
        Delay_Ms(500);
    }
}

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
Input เกิน VDD สัญญาณ input สูงเกิน 3.3V — ทำให้ OPAMP เสียหาย ใช้ voltage divider ลดแรงดันก่อนเข้า OPAMP
Gain ไม่ตรงค่าที่คำนวณ ค่าตัวต้านทานภายนอกผิด หรือ tolerance สูง วัดค่า R ด้วยมัลติมิเตอร์จริง — ใช้ R 1% tolerance
สัญญาณ oscillate ไม่มี capacitor feedback — OPAMP มี phase margin ไม่พอ เพิ่ม capacitor ขนาดเล็ก (~10–100pF) คร่อม Rf
ADC อ่านค่า OPAMP ไม่ถึง VDD Rail-to-rail headroom — OPAMP ภายในไปไม่ถึง 0V / 3.3V เต็ม เผื่อ headroom ~100–300mV จาก rail — output อยู่ในช่วง ~0.1V ถึง ~3.0V
โปรแกรมช้าผิดปกติ เรียก OPAMP_CalculateGain* ใน main loop — float บน MCU ที่ไม่มี FPU คำนวณ gain ครั้งเดียวตอน init — เก็บเป็น const

💡 FAQ

Q: OPAMP ในตัว vs ใช้ IC ภายนอกต่างกันอย่างไร?
A: OPAMP ในตัว สะดวก ลด BOM — ไม่ต้องซื้อ IC เพิ่ม ไม่ต้องเดินวงจรภายนอก แต่มีข้อจำกัดด้าน ความแม่นยำและ bandwidth ที่ต่ำกว่า IC เฉพาะทาง สำหรับงานเซนเซอร์ทั่วไปเพียงพอแล้ว

Q: ใช้ pin ไหนได้บ้าง?
A: CHP0 = PA2, CHP1 = PC4, CHN0 = PA1, CHN1 = PC4 — แต่ละโหมดใช้คู่ channel ต่างกัน OPAMP_SimpleInit เลือกให้อัตโนมัติ

Q: ใช้ OPAMP พร้อมกับฟังก์ชันอื่นได้ไหม?
A: ระวัง pin conflict — PA1 และ PA2 ถูกใช้โดย OPAMP แล้วจะใช้เป็น GPIO, USART, หรือ ADC channel ปกติไม่ได้ ต้องตรวจสอบ resource guard

✏️ ลองทำดู [เฉลย]

  1. Buffer สัญญาณเซนเซอร์ pH — ต่อเซนเซอร์ pH เข้า OPAMP ในโหมด Voltage Follower อ่าน ADC แสดงค่าเป็น voltage
  2. Comparator ตรวจจับ threshold — ตั้ง V+ ด้วย potentiometer เป็น threshold, V− รับสัญญาณจากเซนเซอร์ — เมื่อเกิน threshold ให้ LED ติด
  3. Amplifier gain tests — ทดสอบ Non-Inverting ด้วย R หลายค่า (10k/10k, 10k/22k, 10k/47k) วัด gain จริงเทียบกับค่าคำานวณ
