🤔 ทำไมต้องใช้?

ทุกโปรเจกต์ต้องใช้เวลา ไม่ว่าจะเป็นการกระพริบไฟทุก 500ms, การหน่วงเวลาก่อนอ่านเซนเซอร์, การจับ timeout รอการตอบกลับจากอุปกรณ์ภายนอก, หรือการวัดความเร็วว่าโค้ดส่วนไหนใช้เวลานานแค่ไหน (profiling)

SimpleHAL แบ่งการทำงานด้านเวลาออกเป็น 2 กลุ่มใหญ่: Blocking Delay (โปรแกรมหยุดรอ) และ Non-Blocking Timer (โปรแกรมทำงานต่อได้ระหว่างรอ)

📚 ภาพรวมและหลักการทำงาน

SimpleHAL ใช้ SysTick Timer ซึ่งเป็น hardware timer 24-bit ภายใน Cortex-M0 core มาเป็นฐานเวลาสำหรับระบบ delay ทั้งหมด:

  • Timer_Init() ตั้งค่า SysTick ให้สร้าง interrupt ทุก 1 มิลลิวินาที
  • ภายใน interrupt handler จะเพิ่มตัวนับ global current_ms ขึ้นทีละ 1
  • Delay_Ms(n) อ่านค่า current_ms แล้ววนลูปจนผ่านไป n มิลลิวินาที
  • Get_CurrentMs() อ่านค่า current_ms ณ ขณะนั้น — ใช้สร้าง non-blocking timer ได้

สำหรับระดับ microseconds: Get_CurrentUs() อ่านค่า SysTick counter register โดยตรงแล้วคำนวณกลับเป็นไมโครวินาที ให้ความละเอียด ~20μs ที่ 48MHz

📋 API Reference

ฟังก์ชันคำอธิบายตัวอย่าง
Timer_Init()เริ่มต้น SysTick Timer (ต้องเรียกก่อนฟังก์ชันอื่น)Timer_Init();
Delay_Ms(n)หน่วงเวลา n มิลลิวินาที (blocking)Delay_Ms(500);
Delay_Us(n)หน่วงเวลา n ไมโครวินาที (blocking, ใช้ busy-wait)Delay_Us(100);
Get_CurrentMs()อ่านค่า millisecond counter ตั้งแต่ระบบเริ่ม — ใช้สร้าง non-blocking timeruint32_t t = Get_CurrentMs();
Get_CurrentUs()อ่านค่า microsecond counter (รวมจาก SysTick counter + count register)uint32_t t = Get_CurrentUs();
Get_TickMicros()คืนค่าจำนวน microseconds ต่อ SysTick — ปกติ 1000uint32_t t = Get_TickMicros();
Start_Timer(timer, ms, repeat)เริ่มจับเวลา Timer_t เป็นเวลา ms มิลลิวินาที (repeat: 1 = ทำงานซ้ำอัตโนมัติ, 0 = ครั้งเดียว)Start_Timer(&myTimer, 1000, 0);
Is_Timer_Expired(timer)ตรวจสอบว่า Timer_t ครบเวลาหรือยัง (คืน 1 = ครบ, 0 = ยัง)if (Is_Timer_Expired(&myTimer)) { ... }
Reset_Timer(timer, repeat)รีเซ็ต Timer_t ให้เริ่มนับใหม่ (ใช้กับ Start_Timer)Reset_Timer(&myTimer, 0);
Stop_Timer(timer)หยุด Timer_t (ทำให้ Is_Timer_Expired คืน 1 ทันที)Stop_Timer(&myTimer);

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: SystemCoreClockUpdate + Timer_Init (ต้องเป็น 2 บรรทัดแรกเสมอ)

#include "SimpleHAL.h"

int main(void) {
    SystemCoreClockUpdate();  // ← ต้องเรียกก่อน!
    Timer_Init();              // ← ต้องเรียกเป็นที่สอง!

    // ... โค้ดอื่น ๆ ต่อจากนี้ ...
    while (1) { }
}

🔍 SystemCoreClockUpdate() อัปเดตตัวแปร SystemCoreClock ให้มีค่าความถี่ที่แท้จริง (48MHz) — ถ้าลืมเรียก SysTick จะนับผิด Timer_Init() ใช้ค่า SystemCoreClock ตั้งค่า SysTick reload ให้เกิด interrupt ทุก 1ms พอดี

ขั้นที่ 2: Blocking Delay — ง่าย สั้น แต่โปรแกรมหยุดระหว่างรอ

#include "SimpleHAL.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();

    pinMode(PA1, PIN_MODE_OUTPUT);

    while (1) {
        digitalToggle(PA1);   // toggle LED
        Delay_Ms(500);         // รอ 500 ms — โปรแกรมหยุดตรงนี้
        // อะไรที่อยู่หลัง Delay_Ms จะไม่ทำงานจนกว่าครบเวลา
    }
}

🔍 Delay_Ms() และ Delay_Us() เหมาะกับโปรแกรมง่าย ๆ ที่มีงานแค่ 1–2 อย่าง แต่ถ้าต้องทำงานหลายอย่างพร้อมกัน (multitasking) ควรใช้ non-blocking แทน

ขั้นที่ 3: Non-Blocking Timer — โปรแกรมทำงานต่อได้ระหว่างรอ

#include "SimpleHAL.h"

Timer_t ledTimer;

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();

    pinMode(PA1, PIN_MODE_OUTPUT);
    Start_Timer(&ledTimer, 500, 0);  // เริ่มจับเวลา 500 ms (ครั้งเดียว)

    while (1) {
        if (Is_Timer_Expired(&ledTimer)) {
            digitalToggle(PA1);       // toggle LED
            Start_Timer(&ledTimer, 500, 0);  // รีเซ็ต timer ใหม่
        }

        // โค้ดอื่น ๆ ทำงานได้ตลอดโดยไม่ต้องรอ!
        // เช่น อ่านเซนเซอร์, รับ Serial, ประมวลผล...
    }
}

🔍 Is_Timer_Expired() ตรวจสอบแค่ชั่วขณะแล้วคืนค่าทันที — loop ยังคงวิ่งต่อไป ไม่มีจุดหยุดรอ หลังทำงานเสร็จต้องเรียก Start_Timer() อีกครั้งเพื่อเริ่มรอบใหม่

ขั้นที่ 4: อ่านเวลาปัจจุบัน — ใช้กับ timeout, profiling, debounce

#include "SimpleHAL.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();

    uint32_t start_ms = Get_CurrentMs();

    // ทำงานบางอย่าง...
    for (volatile int i = 0; i < 100000; i++);

    uint32_t elapsed_ms = Get_CurrentMs() - start_ms;

    // elapsed_ms คือเวลาที่ใช้ไป (ms)
    // ถ้าใช้ USART จะส่งค่าไปดูได้:
    // USART_PrintNum(elapsed_ms);

    while (1) { }
}

🔍 Get_CurrentMs() ช่วยให้เขียน timeout pattern ได้ — เช่น "ถ้าผ่านไป 5 วินาทีแล้วยังไม่ได้คำตอบ ให้เลิกรอ" หรือใช้วัดว่าโค้ดส่วนไหนช้า (profiling)

🟢 ตัวอย่างขั้นต้น

LED กระพริบด้วย blocking delay — เข้าใจง่ายที่สุด

#include "SimpleHAL.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    pinMode(PA1, PIN_MODE_OUTPUT);

    while (1) {
        digitalWrite(PA1, HIGH);
        Delay_Ms(100);
        digitalWrite(PA1, LOW);
        Delay_Ms(900);
    }
}

🟡 ตัวอย่างขั้นกลาง

Non-Blocking Multi-Task — LED กระพริบ 3 ความเร็ว + heartbeat + อ่านเซนเซอร์ ทั้งหมดทำงานพร้อมกันโดยไม่หยุดรอ

#include "SimpleHAL.h"

Timer_t led1Timer;
Timer_t led2Timer;
Timer_t heartbeatTimer;

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();

    pinMode(PA1, PIN_MODE_OUTPUT);  // LED1: กระพริบเร็ว 200ms
    pinMode(PA2, PIN_MODE_OUTPUT);  // LED2: กระพริบช้า 1s
    pinMode(PC0, PIN_MODE_OUTPUT);  // heartbeat: 50ms on, 2s off

    Start_Timer(&led1Timer, 200, 0);
    Start_Timer(&led2Timer, 1000, 0);
    Start_Timer(&heartbeatTimer, 50, 0);

    while (1) {
        if (Is_Timer_Expired(&led1Timer)) {
            digitalToggle(PA1);
            Start_Timer(&led1Timer, 200, 0);
        }

        if (Is_Timer_Expired(&led2Timer)) {
            digitalToggle(PA2);
            Start_Timer(&led2Timer, 1000, 0);
        }

        if (Is_Timer_Expired(&heartbeatTimer)) {
            digitalToggle(PC0);
            if (digitalRead(PC0) == HIGH) {
                Start_Timer(&heartbeatTimer, 50, 0);    // ON 50ms
            } else {
                Start_Timer(&heartbeatTimer, 2000, 0);  // OFF 2s
            }
        }

        // ตรงนี้ว่าง — ทำงานเพิ่มได้อีก!
    }
}

🔴 ตัวอย่างขั้นสูง

Timeout pattern + การวัดความกว้างพัลส์ + profiling — ใช้ Get_CurrentMs() และ Get_CurrentUs() ในสถานการณ์จริง

#include "SimpleHAL.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();

    pinMode(PC1, PIN_MODE_INPUT_PULLUP);
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);

    USART_Print("=== Pulse Width Measurement ===\r\n");

    while (1) {
        // === Timeout Pattern ===
        // รอให้ pin เป็น LOW (กดปุ่ม) แต่ไม่รอเกิน 5 วินาที
        uint32_t timeout_start = Get_CurrentMs();
        while (digitalRead(PC1) == HIGH) {
            if (Get_CurrentMs() - timeout_start > 5000) {
                USART_Print("Timeout: no button press in 5s\r\n");
                break;
            }
        }

        // === Pulse Width Measurement ===
        // รอให้ pin กลับ HIGH แล้ววัดว่าอยู่ LOW นานแค่ไหน
        if (digitalRead(PC1) == LOW) {
            uint32_t pulse_start = Get_CurrentUs();
            while (digitalRead(PC1) == LOW);  // รอปล่อย
            uint32_t pulse_width = Get_CurrentUs() - pulse_start;

            USART_Print("Pulse width: ");
            USART_PrintNum(pulse_width);
            USART_Print(" us\r\n");
        }

        // === Profiling ===
        uint32_t t1 = Get_CurrentUs();
        // โค้ดที่ต้องการวัดความเร็ว...
        for (volatile int i = 0; i < 1000; i++) { __NOP(); }
        uint32_t t2 = Get_CurrentUs();

        USART_Print("Elapsed: ");
        USART_PrintNum(t2 - t1);
        USART_Print(" us\r\n");

        Delay_Ms(500);
    }
}

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
Delay_Ms() ไม่ทำงานเลย ลืมเรียก Timer_Init() — SysTick ยังไม่ได้เริ่มทำงาน เพิ่ม Timer_Init(); บรรทัดที่ 2 ใน main()
Get_CurrentUs() คืนค่า 0 เสมอ ยังไม่ได้ init SysTick หรือเรียก Get_CurrentUs ก่อน Timer_Init ย้าย Timer_Init() ไว้บนสุด ตามด้วยฟังก์ชันอื่น
Timer_t ทำงานผิด — หมดเวลาตลอดหรือไม่หมดเลย ประกาศ Timer_t เป็น local variable ในฟังก์ชัน — ค่าไม่ persist ประกาศเป็น global หรือ static
ห้ามใช้ delay ใน ISR ISR ต้องทำงานให้เสร็จเร็วที่สุด — delay จะ block ทั้งระบบ ใช้ flag แทน — ให้ main loop เช็ค flag แล้วค่อย delay

💡 เกร็ดความรู้ / FAQ

Q: Blocking vs Non-Blocking — ควรใช้อะไรเมื่อไร?
A: Blocking (Delay_Ms/Delay_Us) — ใช้กับโปรแกรมสั้นที่มีงานเดียว เช่น กระพริบไฟ, สร้างสัญญาณ clock Non-Blocking (Timer_t) — ใช้เมื่อต้องการทำหลายอย่างพร้อมกัน เช่น อ่านปุ่ม + กระพริบ LED + ส่ง Serial พร้อมกัน

Q: Timer_t ใช้ได้สูงสุดกี่ตัว?
A: ไม่จำกัด — Timer_t เป็นแค่ struct (ขนาด ~12 bytes) ประกาศเท่าไรก็ได้ ใช้แค่ RAM ไม่ใช้ hardware timer เพิ่ม

Q: Get_CurrentMs() ใช้ไปนาน ๆ จะ overflow ไหม?
A: ตัวแปรเป็น uint32_t — จะ overflow ทุก ~49.7 วัน แล้วกลับไปเริ่มที่ 0 ใหม่ การคำนวณผลต่างของเวลา (now - last) ยังถูกต้องอยู่เพราะใช้ unsigned arithmetic แบบ wrap-around

Q: Get_CurrentUs() กับ Get_CurrentMs() ต่างกันยังไง?
A: Get_CurrentMs() ละเอียด 1ms อ่านจากตัวแปร global — เร็วและเหมาะกับงานทั่วไป Get_CurrentUs() ละเอียด ~20μs อ่านจาก hardware counter โดยตรง — ช้ากว่าเล็กน้อย แต่ละเอียดกว่า ใช้กับงานที่ต้องการความแม่นยำสูง

✏️ ลองทำดู [เฉลย]

  1. LED กระพริบ 3 ความเร็วด้วย Timer_t — ใช้ Timer_t 3 ตัว แยกกัน ทำให้ LED 3 ดวงกระพริบที่ 100ms, 500ms, และ 1 วินาที โดยไม่มี blocking
  2. นับเวลากดปุ่มค้าง — ใช้ Get_CurrentMs() จับเวลาว่าผู้ใช้กดปุ่มค้างนานกี่มิลลิวินาที แล้วแสดงผลทาง USART (ถ้ามี) หรือเปลี่ยนความถี่กระพริบ LED ตามระยะเวลากด
  3. สร้าง Fade LED ด้วย non-blocking timer — ใช้ Timer_t + software PWM (เปลี่ยน duty cycle ทีละนิดทุก 2ms) ทำให้ LED ค่อย ๆ สว่างขึ้นและค่อย ๆ ดับลง โดยที่ main loop ยังว่างทำงานอื่นได้
