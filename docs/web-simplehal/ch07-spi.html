ระดับขั้นต้น: MAX7219 7-Segment Driver

แนวคิด: เขียน register + data พร้อม toggle CS ทุกครั้ง 
#include "SimpleHAL.h"

#define MAX7219_CS PC0

void max7219_write(uint8_t reg, uint8_t data) {
    digitalWrite(MAX7219_CS, LOW);
    SPI_Transfer(reg);
    SPI_Transfer(data);
    digitalWrite(MAX7219_CS, HIGH);
}

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    SPI_SimpleInit(SPI_CLOCK_DIV_16, SPI_PINS_DEFAULT);
    SPI_SetCSPin(MAX7219_CS);

    // Initialize MAX7219
    max7219_write(0x09, 0x00); // No decode
    max7219_write(0x0A, 0x01); // Intensity
    max7219_write(0x0B, 0x07); // Scan limit (8 digits)
    max7219_write(0x0C, 0x01); // Shutdown off

    // Display "8" on digit 0
    max7219_write(0x01, 0x08);

    while (1);
}

ระดับขั้นกลาง: W25Q Flash Memory

แนวคิด: อ่าน JEDEC ID, เขียน enable, อ่านข้อมูลที่ address 
#include "SimpleHAL.h"

#define W25Q_CS PC0

void w25q_cs_low(void)  { digitalWrite(W25Q_CS, LOW); }
void w25q_cs_high(void) { digitalWrite(W25Q_CS, HIGH); }

uint32_t w25q_read_jedec(void) {
    w25q_cs_low();
    SPI_Transfer(0x9F); // JEDEC ID command
    uint32_t id = (SPI_Transfer(0xFF) << 16) | (SPI_Transfer(0xFF) << 8) | SPI_Transfer(0xFF);
    w25q_cs_high();
    return id;
}

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(USART_115200);
    SPI_SimpleInit(SPI_CLOCK_DIV_4, SPI_PINS_DEFAULT);
    SPI_SetCSPin(W25Q_CS);
    digitalWrite(W25Q_CS, HIGH);

    uint32_t jedec = w25q_read_jedec();
    printf("JEDEC ID: 0x%06lX\r\n", jedec);

    while (1);
}

ระดับขั้นสูง: SPI + DMA สำหรับ Bulk Display Buffer

แนวคิด: ใช้ DMA ส่ง buffer ขนาดใหญ่ไปยัง display — ลด CPU overhead ได้อย่างมาก (~60fps) 
#include "SimpleHAL.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    SPI_SimpleInit(SPI_CLOCK_DIV_2, SPI_PINS_DEFAULT);

    // ดูบทที่ 13 DMA สำหรับรายละเอียดการใช้งาน SPI + DMA
    // SPI_DMA_Init() + SPI_DMA_Send(buffer, size)
    // ทำให้ส่งข้อมูล display buffer ได้โดย CPU แทบไม่ได้ใช้

    while (1);
}

ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
ข้อมูลผิดเพี้ยนMode ไม่ตรงกับ datasheetเช็ค CPOL/CPHA ใน datasheet อุปกรณ์
Device ไม่ตอบสนองลืมดึง CS LOWต้อง digitalWrite(CS, LOW) ก่อนทุกครั้ง
สัญญาณเสียสายยาวเกินไป หรือ breadboardใช้สายสั้น, ลดความเร็ว SPI
MISO อ่านค่าสุ่มไม่มี pull-upถ้า MISO เป็น open-drain ต้อง pull-up

คำถามที่พบบ่อย (FAQ)

SPI vs I2C ต่างกันอย่างไร?

SPI เร็วกว่า ใช้ 4 สาย (ไม่มี address) — I2C 2 สาย ใช้ address เลือก device

SOP-8 ใช้ SPI ได้ไหม?

ไม่มี hardware SPI — ใช้ shiftOut() / shiftIn() แบบ bit-bang แทน
