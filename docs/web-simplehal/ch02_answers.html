📝 เฉลย — บทที่ 2: GPIO — Digital Input/Output

ข้อ 1: LED 3 ดวงวิ่ง Knight Rider (PC0→PC1→PC2→PC1→PC0)

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include "SimpleHAL.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();

    uint8_t leds[] = {PC0, PC1, PC2};
    pinModeMultiple(leds, PIN_MODE_OUTPUT);

    int idx = 0, dir = 1;

    while (1) {
        digitalWrite(PC0, LOW);
        digitalWrite(PC1, LOW);
        digitalWrite(PC2, LOW);
        digitalWrite(leds[idx], HIGH);

        Delay_Ms(150);

        idx += dir;
        if (idx >= 3) { idx = 1; dir = -1; }
        if (idx < 0)  { idx = 1; dir =  1; }
    }
}
🧠 แนวคิด: pinModeMultiple ตั้ง LED 3 ดวงเป็น output พร้อมกัน — array leds[] เก็บ pin — วน idx ไปข้างหน้า 0→1→2 แล้วกลับทิศ 2→1→0 ด้วย dir flag — ก่อนเปลี่ยนดวงต้องดับทุกดวงด้วย digitalWrite(LOW) ก่อน แล้วติดเฉพาะดวงปัจจุบัน — Delay_Ms(150) ควบคุมความเร็ว

ข้อ 2: ปุ่มกด 2 ปุ่มเปลี่ยนความเร็วกระพริบ

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include "SimpleHAL.h"

volatile int speed_ms = 500;
volatile uint8_t btn_flag = 0;

void btn_up_isr(void) {
    if (speed_ms > 50) speed_ms -= 50;
}
void btn_down_isr(void) {
    if (speed_ms < 2000) speed_ms += 50;
}

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();

    pinMode(PC0, PIN_MODE_OUTPUT);
    pinMode(PC1, PIN_MODE_INPUT_PULLUP);
    pinMode(PC2, PIN_MODE_INPUT_PULLUP);
    attachInterrupt(PC1, btn_up_isr, FALLING);
    attachInterrupt(PC2, btn_down_isr, FALLING);

    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);

    while (1) {
        digitalToggle(PC0);
        USART_Print("Speed=");
        USART_PrintNum(speed_ms);
        USART_Print("ms\r\n");
        Delay_Ms(speed_ms);
    }
}
🧠 แนวคิด: ใช้ volatile int speed_ms แชร์ระหว่าง ISR และ main — btn_up_isr ลดค่า → เร็วขึ้น (delay สั้นลง), btn_down_isr เพิ่มค่า → ช้าลง — clamp ไม่ให้ต่ำกว่า 50ms หรือเกิน 2000ms — ใช้ attachInterrupt 2 ตัวแยกกันสำหรับ 2 ปุ่ม

ข้อ 3: ใช้ pulseIn วัด double-click

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include "SimpleHAL.h"

volatile uint32_t last_press_ms = 0;
volatile uint8_t  click_count = 0;

void btn_isr(void) {
    uint32_t now = Get_CurrentMs();
    if (now - last_press_ms < 400) {
        click_count++;  // double-click!
    } else {
        click_count = 1;  // single click ใหม่
    }
    last_press_ms = now;
}

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();

    pinMode(PC0, PIN_MODE_OUTPUT);
    pinMode(PC1, PIN_MODE_INPUT_PULLUP);
    attachInterrupt(PC1, btn_isr, FALLING);

    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);

    while (1) {
        if (click_count == 2) {
            click_count = 0;
            USART_Print("DOUBLE CLICK!\r\n");
            for (int i = 0; i < 5; i++) {
                digitalToggle(PC0);
                Delay_Ms(80);
            }
        }
        Delay_Ms(10);
    }
}
🧠 แนวคิด: ISR จับเวลาจาก Get_CurrentMs — ถ้าห่าง < 400ms = double-click — ถ้าห่าง > 400ms = single click ใหม่ — main loop เช็ค click_count == 2 → กระพริบ LED เร็ว 5 ครั้งยืนยัน — ใช้ volatile กับตัวแปรที่แชร์ระหว่าง ISR/main

🔌 วงจรประกอบ — Knight Rider 3 LED + 2 ปุ่ม

        CH32V003
        ──────────
        ─          ─
   PC0 ───── R220Ω ─────▶ LED1 ────── GND
   PC1 ───── R220Ω ─────▶ LED2 ────── GND
   PC2 ───── R220Ω ─────▶ LED3 ────── GND
        ─                       ─
   PC3 ───── ปุ่มเร่ง ───── GND
   PC4 ───── ปุ่มลด  ───── GND
        ──────────
💡 ทำไมต้อง R 220Ω: LED ใช้กระแส ~10mA — ถ้าไม่มี R จะดึงกระแสเกินชิปพัง = (3.3V - 2V) / 220Ω ≈ 6mA — ใช้ internal pull-up (~40kΩ) สำหรับปุ่มกด ไม่ต้องต่อ R pull-up ภายนอก
