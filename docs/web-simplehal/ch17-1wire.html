บทที่ 17: 1-Wire — DS18B20 และอุปกรณ์ One-Wire

ระดับสูง

สิ่งที่ต้องรู้ก่อน

  • บทที่ 3: Delay (timing-critical! ต้องแม่นยำ microsecond)
  • บทที่ 1: USART (แสดงผลอุณหภูมิ)

อุปกรณ์ที่ต้องใช้

  • CH32V003 Dev Board
  • DS18B20 เซนเซอร์อุณหภูมิ
  • ตัวต้านทาน 4.7kΩ pull-up
  • Breadboard + สายไฟ

ทำไมต้องใช้?

DS18B20 เป็นเซนเซอร์อุณหภูมิยอดนิยม — ใช้สายเดียวต่อได้หลายตัว, กันน้ำ, แม่นยำ ±0.5°C

API Reference

ฟังก์ชันคำอธิบาย
OneWire_Init(pin)เริ่มต้น 1-Wire bus บน pin ที่กำหนด — คืนค่า OneWire_Bus* handle ที่ต้องเก็บไว้ส่งให้ทุกฟังก์ชันด้านล่าง
OneWire_Reset(bus)ส่ง reset pulse ตรวจ presence
OneWire_WriteByte(bus, data)เขียน 1 ไบต์ไปยัง bus
OneWire_ReadByte(bus)อ่าน 1 ไบต์จาก bus
OneWire_SkipROM(bus)SkipROM — ส่งคำสั่งถึงทุก device
OneWire_ReadROM(bus, rom)อ่าน 64-bit ROM address (เฉพาะ single device)
OneWire_MatchROM(bus, rom)เลือก device ตาม ROM address
OneWire_Select(bus, rom)เลือก device (Reset + MatchROM รวมกัน)
OneWire_ResetSearch(bus)เริ่มค้นหา device ทั้งหมดบน bus
OneWire_Search(bus)ค้นหา device ถัดไป (คืน true เมื่อพบ, ผลอยู่ใน bus->rom — ดึงด้วย OneWire_GetAddress)
OneWire_AlarmSearch(bus)ค้นหา device ที่มี alarm ตั้งไว้ (เหมือน Search แต่ใช้ alarm command)
OneWire_GetAddress(bus, rom)ดึง ROM address ที่พบล่าสุดจาก OneWire_Search/AlarmSearch — ไม่ได้รับ index
OneWire_WriteBit(bus, bit)เขียน 1 bit
OneWire_ReadBit(bus)อ่าน 1 bit
OneWire_WriteBytes(bus, data, len)เขียนหลายไบต์
OneWire_ReadBytes(bus, buf, len)อ่านหลายไบต์
OneWire_CRC8(data, len)คำนวณ CRC8 (ไม่ต้องใช้ bus — คำนวณจาก buffer ล้วนๆ)
OneWire_VerifyCRC(data, len)ตรวจสอบ CRC8 ของข้อมูล (ไม่ต้องใช้ bus)
OneWire_Depower(bus)ปล่อย bus (parasite power mode)
OneWire_GetBusByPin(pin)ดึง bus handle ที่เคย Init ไว้แล้ว จาก pin (ไม่ต้อง Init ซ้ำ)
สำคัญ: ทุกฟังก์ชันข้างบน (ยกเว้น OneWire_CRC8/OneWire_VerifyCRC) ต้องรับ bus (ค่าที่ได้จาก OneWire_Init()) เป็น argument แรกเสมอ — เก็บค่านี้ไว้เป็นตัวแปรตั้งแต่ต้น main

ขั้นตอนอ่าน DS18B20 (Step by Step)

0. เริ่มต้น bus (ทำครั้งเดียวตอนต้น)

OneWire_Bus* bus = OneWire_Init(PA1); — เก็บ bus ไว้ ต้องส่งให้ทุกฟังก์ชันด้านล่าง

1. Reset + Presence Check

OneWire_Reset(bus) — ตรวจสอบว่ามี device ต่ออยู่หรือไม่

2. SkipROM / MatchROM

OneWire_SkipROM(bus) (อุปกรณ์ตัวเดียว) หรือ OneWire_MatchROM(bus, rom) (หลายตัว)

3. Convert T (0x44)

OneWire_WriteByte(bus, 0x44) — สั่งให้เซนเซอร์วัดอุณหภูมิ

4. รอ 750ms

Delay_Ms(750) — เวลาที่เซนเซอร์ใช้ในการแปลงค่า

5. Read Scratchpad (0xBE)

อ่าน 9 ไบต์ — ไบต์ 0-1 = raw temperature

6. CRC Verify

OneWire_VerifyCRC(buf, 9) — ตรวจสอบความถูกต้อง (ฟังก์ชันนี้ไม่ต้องใช้ bus)

7. แปลงค่า

temp = (int16_t)(raw[1] << 8 | raw[0]) / 16.0f

ระดับขั้นต้น: Single DS18B20 อ่านอุณหภูมิ

#include "SimpleHAL.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);

    OneWire_Bus* bus = OneWire_Init(PA1);  // OneWire_Init รับ SimpleGPIO pin เดียว ไม่ใช่ (port, pin)

    while (1) {
        if (!OneWire_Reset(bus)) {
            printf("No DS18B20 found!\r\n");
            Delay_Ms(1000);
            continue;
        }

        OneWire_SkipROM(bus);
        OneWire_WriteByte(bus, 0x44); // Convert T
        Delay_Ms(750);

        if (!OneWire_Reset(bus)) {
            printf("Device lost!\r\n");
            continue;
        }

        OneWire_SkipROM(bus);
        OneWire_WriteByte(bus, 0xBE); // Read Scratchpad

        uint8_t raw[9];
        for (uint8_t i = 0; i < 9; i++) {
            raw[i] = OneWire_ReadByte(bus);
        }

        if (!OneWire_VerifyCRC(raw, 9)) {
            printf("CRC error, retrying...\r\n");
            continue;
        }

        int16_t temp_raw = (raw[1] << 8) | raw[0];
        float temp_c = temp_raw / 16.0f;
        printf("Temperature: %.2f °C\r\n", temp_c);
        Delay_Ms(1000);
    }
}

ระดับขั้นกลาง: Multi-device Scan — ค้นหา ROM ทุกตัว

แนวคิด: ใช้ OneWire_Search ค้นหา ROM address ของทุก device บน bus แล้วอ่านแต่ละตัวด้วย MatchROM 
#include "SimpleHAL.h"

#define MAX_DEVICES 4

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
    OneWire_Bus* bus = OneWire_Init(PA1);

    uint8_t roms[MAX_DEVICES][8];
    uint8_t count = 0;

    OneWire_ResetSearch(bus);
    while (count < MAX_DEVICES && OneWire_Search(bus)) {
        OneWire_GetAddress(bus, roms[count]);  // OneWire_Search ไม่คืน ROM ตรงๆ ต้องดึงด้วย GetAddress
        printf("Device %d: ", count + 1);
        for (uint8_t i = 0; i < 8; i++) {
            printf("%02X", roms[count][i]);
        }
        printf("\r\n");
        count++;
    }
    printf("Found %d device(s)\r\n", count);

    while (1) {
        for (uint8_t i = 0; i < count; i++) {
            OneWire_Select(bus, roms[i]);
            OneWire_WriteByte(bus, 0x44); // Convert T

            Delay_Ms(750);

            OneWire_Select(bus, roms[i]);
            OneWire_WriteByte(bus, 0xBE);

            uint8_t raw[9];
            for (uint8_t j = 0; j < 9; j++) {
                raw[j] = OneWire_ReadByte(bus);
            }

            if (OneWire_VerifyCRC(raw, 9)) {
                int16_t t = (raw[1] << 8) | raw[0];
                printf("Sensor %d: %.2f °C\r\n", i + 1, t / 16.0f);
            }
        }
        Delay_Ms(2000);
    }
}

ระดับขั้นสูง: Simultaneous Conversion — ประหยัดเวลารอ

แนวคิด: ใช้ SkipROM + Convert T broadcast สั่งให้ทุกเซนเซอร์วัดพร้อมกัน — รอแค่ครั้งเดียว — แล้วอ่านแต่ละตัว — ประหยัดเวลา N×750ms 
#include "SimpleHAL.h"

#define MAX_DEVICES 4

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
    OneWire_Bus* bus = OneWire_Init(PA1);

    uint8_t roms[MAX_DEVICES][8];
    uint8_t count = 0;

    OneWire_ResetSearch(bus);
    while (count < MAX_DEVICES && OneWire_Search(bus)) {
        OneWire_GetAddress(bus, roms[count]);
        count++;
    }

    printf("Found %d devices\r\n", count);

    while (1) {
        // Broadcast Convert T to ALL devices
        OneWire_Reset(bus);
        OneWire_SkipROM(bus);
        OneWire_WriteByte(bus, 0x44);

        // รอครั้งเดียว! (แทน N×750ms)
        Delay_Ms(750);

        // อ่านแต่ละตัว
        for (uint8_t i = 0; i < count; i++) {
            OneWire_Select(bus, roms[i]);
            OneWire_WriteByte(bus, 0xBE);

            uint8_t raw[9];
            for (uint8_t j = 0; j < 9; j++) {
                raw[j] = OneWire_ReadByte(bus);
            }

            if (OneWire_VerifyCRC(raw, 9)) {
                int16_t t = (raw[1] << 8) | raw[0];
                printf("S%d: %.2f°C  ", i + 1, t / 16.0f);
            }
        }
        printf("\r\n");
        Delay_Ms(2000);
    }
}

ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
Reset ไม่พบ deviceไม่มี pull-up resistorต่อ 4.7kΩ ระหว่าง DQ กับ VDD
CRC error บ่อยสายยาวเกินไป / noiseใช้สายสั้น, ลด noise, ใช้ shielded cable
อ่านได้ตัวเดียวใช้ SkipROM (ส่งถึงทุกตัว)ใช้ MatchROM สำหรับ multi-device
อุณหภูมิผิดResolution default 9-bitเปลี่ยน resolution เป็น 12-bit

คำถามที่พบบ่อย (FAQ)

ต่อ DS18B20 แบบ Parasite Power ได้ไหม?

ได้ — ใช้แค่ 2 สาย (DQ + GND, VDD ต่อ GND) — แต่ต้องใช้ strong pull-up ระหว่าง conversion

ต่อ DS18B20 ได้กี่ตัวบน bus เดียว?

หลายสิบตัวบน bus เดียว (มี unique 64-bit ROM address) — ทุกตัวใช้ data line ร่วมกัน

สายยาวสุดเท่าไหร่?

~100 เมตร ด้วย 2.2kΩ pull-up — สายยาวมากกว่าใช้ shielded cable

ลองทำดู (แบบฝึกหัด)

  1. อ่านอุณหภูมิจาก DS18B20 ตัวเดียว — แสดงผลผ่าน USART
  2. ต่อ DS18B20 2 ตัวบน bus เดียว — อ่านอุณหภูมิแยกแต่ละตัว
  3. ใช้ Simultaneous Conversion — วัดอุณหภูมิ 3 ตัวพร้อมกัน
