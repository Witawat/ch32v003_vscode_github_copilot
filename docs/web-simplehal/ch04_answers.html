📝 เฉลย — บทที่ 4: USART — สื่อสารผ่าน Serial

ข้อ 1: Echo Server — ส่งกลับทุกตัวอักษร

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include "SimpleHAL.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();

    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
    USART_Print("Echo Server Ready\r\n");

    while (1) {
        if (USART_Available()) {
            uint8_t ch = USART_Read();
            USART_WriteByte(ch);  // ส่งกลับอักษรเดิม
            if (ch == '\r') {
                USART_WriteByte('\n');  // เพิ่ม newline หลัง CR
            }
        }
    }
}
🧠 แนวคิด: USART_Available เช็คว่ามีข้อมูล → USART_Read รับ 1 byte → USART_WriteByte ส่งกลับ byte เดิมทันที — ตรวจเฉพาะ \r (\n จะตามมาแล้วในบาง terminal) — ง่ายสุด: echo ทุกตัวอักษรโดยไม่ต้อง buffer

ข้อ 2: LED PWM ควบคุมด้วยคำสั่ง Serial "LED=50"

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include <stdlib.h>

#define CMD_BUF_SIZE 16

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();

    PWM_Init(PWM2_CH3, 1000);  // PC0, 1kHz
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
    USART_Print("Commands: LED=0 ถึง LED=100\r\n");

    char cmd[CMD_BUF_SIZE];
    uint8_t idx = 0;

    while (1) {
        if (USART_Available()) {
            char c = (char)USART_Read();
            if (c == '\r' || c == '\n') {
                if (idx > 0) {
                    cmd[idx] = '\0';
                    if (cmd[0]=='L' && cmd[1]=='E' && cmd[2]=='D' && cmd[3]=='=') {
                        int val = atoi(&cmd[4]);
                        if (val < 0) val = 0;
                        if (val > 100) val = 100;
                        PWM_SetDutyCycle(PWM2_CH3, (uint8_t)val);
                        USART_Print("LED=");
                        USART_PrintNum(val);
                        USART_Print("%\r\n");
                    } else {
                        USART_Print("Unknown\r\n");
                    }
                    idx = 0;
                }
            } else if (idx < CMD_BUF_SIZE - 1) {
                cmd[idx++] = c;
            }
        }
    }
}
🧠 แนวคิด: สะสม char ใน buffer จนเจอ \r หรือ \n → เซ็ค 4 ตัวแรกเป็น "LED=" → atoi(&cmd[4]) แปลงตัวเลขหลัง '=' → clamp 0-100 → PWM_SetDutyCycle — รวม <stdlib.h> เพื่อใช้ atoi — cmd_buf + idx = simple string builder

ข้อ 3: อ่าน ADC ส่งทาง Serial ทุก 100ms

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include "SimpleHAL.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();

    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
    ADC_SimpleInit();

    Timer_t adc_timer;
    Start_Timer(&adc_timer, 100, 1);  // 100ms

    USART_Print(" ADC  |  RAW  |  mV\r\n");
    USART_Print("------|-------|------\r\n");

    uint16_t sample = 0;
    while (1) {
        if (Is_Timer_Expired(&adc_timer)) {
            uint16_t raw = ADC_Read(ADC_CH_PD2);
            uint16_t mv  = (uint16_t)(((uint32_t)raw * 3300) / 1023);

            USART_Print("  PD2  | ");
            USART_PrintNum(raw);
            USART_Print(" | ");
            USART_PrintNum(mv);
            USART_Print("\r\n");

            sample++;
        }
    }
}
🧠 แนวคิด: Timer_t 100ms non-blocking → แต่ละ tick อ่าน ADC_Read(PD2) + คำนวณ mV = raw × 3300 / 1023 → USART_PrintNum แสดงผล — พิมพ์ header ตารางก่อนเข้า loop — ใช้ uint32_t คูณก่อนหารป้องกัน overflow — sample counter (ไว้สำหรับ debug) — main loop ไม่อั้น CPU ว่างทำงานอื่น
