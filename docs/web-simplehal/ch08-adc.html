API Reference

ฟังก์ชันคำอธิบาย
ADC_SimpleInit()เริ่มต้น ADC
ADC_SimpleInitChannels()เริ่มต้น ADC แบบหลาย channel
ADC_EnableChannel()เปิดใช้งาน channel เพิ่ม
ADC_Read()อ่านค่า raw 0-1023
ADC_ReadMultiple()อ่านหลาย channel
ADC_ReadVoltage()อ่านเป็นแรงดัน (V)
ADC_ToVoltage()แปลง raw → แรงดัน
ADC_ReadAverage()อ่านเฉลี่ย (ลด noise)
ADC_ToPercent()แปลง raw → เปอร์เซ็นต์
ADC_ReadVrefInt()อ่าน internal Vref (~1.2V)
ADC_GetVDD()คำนวณแรงดัน VDD จริง
ADC_ReadVoltageCompensated()อ่านแรงดันแบบชดเชย VDD
ADC_GetBatteryPercent()คำนวณเปอร์เซ็นต์แบตเตอรี่

ระดับขั้นต้น: อ่าน Potentiometer

#include "SimpleHAL.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
    ADC_SimpleInit(); // เปิดทุก channel

    while (1) {
        uint16_t raw = ADC_Read(ADC_CH_0);            // 0-1023, PA2
        float voltage = ADC_ToVoltage(raw, 3.3);      // 0-3.3V
        printf("Raw=%d Voltage=%.2fV\r\n", raw, voltage);
        Delay_Ms(500);
    }
}

ระดับขั้นกลาง: ตรวจสอบแบตเตอรี่

#include "SimpleHAL.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);

    while (1) {
        float vdd = ADC_GetVDD();
        float pct = ADC_GetBatteryPercent(vdd, 3.0, 4.2);  // Li-ion: 3.0V (หมด) - 4.2V (เต็ม)

        printf("VDD=%.3fV Battery=%.0f%%\r\n", vdd, pct);
        Delay_Ms(2000);
    }
}

ระดับขั้นสูง: Multi-Channel Scan พร้อม VDD Compensation

#include "SimpleHAL.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);

    uint8_t channels[] = {ADC_CH_0, ADC_CH_3, ADC_CH_7};
    ADC_SimpleInitChannels(channels, 3);

    while (1) {
        uint16_t results[3];
        ADC_ReadMultiple(channels, results, 3);

        for (int i = 0; i < 3; i++) {
            float v = ADC_ReadVoltageCompensated(channels[i]);
            printf("CH%d=%.3fV ", channels[i], v);
        }
        printf("\r\n");
        Delay_Ms(500);
    }
}

ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
ค่าแกว่ง / มี noiseสัญญาณรบกวนใช้ ADC_ReadAverage() แทน
ค่าเพี้ยนตอน VDD ตกVDD เป็น referenceใช้ ADC_ReadVoltageCompensated()
ไม่มี FPUCH32V003 ไม่มี hardware floatหลีกเลี่ยง float ถ้าทำได้ ใช้ int แทน
อ่าน internal channel ไม่ได้ยังไม่ได้ initv2.1+ มี lazy-init — เรียกแล้วใช้งานได้เลย

คำถามที่พบบ่อย (FAQ)

ADC 10-bit ได้ความละเอียดเท่าไร?

0-1023 = 1024 steps — ที่ 3.3V ประมาณ 3.2mV ต่อขั้น

ทำไมอ่านค่าผิด?

อาจยังไม่ได้เรียก ADC_SimpleInit() — ต้องเรียกก่อนอ่านทุกครั้ง

analogReadMilliVolts?

เป็น macro ที่คำนวณตรงจาก analogRead(pin): (analogRead(pin) * 3300) / 1023 — อ่านค่าเป็น mV ทันทีโดยอิง Vref 3.3V ตายตัว (ไม่ได้เรียก ADC_ReadVoltage())

ลองทำดู (แบบฝึกหัด)

  1. อ่าน potentiometer แสดงผลเป็นเปอร์เซ็นต์ 0-100%
  2. วัดแรงดันแบตเตอรี่และแสดง % คงเหลือ
  3. วัดอุณหภูมิด้วย LM35 (10mV/°C)
