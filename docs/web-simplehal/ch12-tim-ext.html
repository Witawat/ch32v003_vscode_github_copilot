🤔 ทำไมต้องใช้?

Stopwatch และ Countdown เป็นโมดูลสำเร็จรูปที่ต่อยอดจาก Hardware Timer — ไม่ต้องเขียน logic จับเวลาเอง ใช้ทำ นาฬิกาจับเวลาแข่งขัน, นับถอยหลังทำอาหาร, interval training, หรือ lap timer ได้ทันที

เปลี่ยน Timer เริ่มต้น: ดีฟอลต์ใช้ TIM_2 — เปลี่ยนเป็น TIM_1 ด้วย #define TIMEXT_DEFAULT_TIMER TIM_1 ก่อน include SimpleHAL

🟢 ตัวอย่างขั้นต้น

Stopwatch แสดงผลผ่าน USART — ใช้ปุ่ม Start/Stop แสดงเวลา HH:MM:SS

#include "simplehal.h"
#include <stdio.h>

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(USART_115200);

    pinMode(PC1, PIN_MODE_INPUT_PULLUP);

    Stopwatch_Init();
    char buf[16];

    printf("--- Stopwatch Ready ---\r\n");

    while (1) {
        if (digitalRead(PC1) == LOW) {
            Delay_Ms(30);
            if (Stopwatch_IsRunning())
                Stopwatch_Stop();
            else
                Stopwatch_Start();
            while (digitalRead(PC1) == LOW);
            Delay_Ms(30);
        }

        Stopwatch_GetTimeString(buf, TIME_FORMAT_HHMMSS);
        printf("\rTime: %s", buf);
        Delay_Ms(200);
    }
}

🟡 ตัวอย่างขั้นกลาง

แนวคิด: Countdown Timer 60 วินาที — เมื่อหมดเวลาให้ Buzzer ดัง 
#include "simplehal.h"
#include <stdio.h>

void onAlarm(void) {
    // callback เมื่อนับถอยหลังถึง 0
    digitalWrite(PA2, HIGH);  // เปิด Buzzer
}

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(USART_115200);

    pinMode(PC1, PIN_MODE_INPUT_PULLUP); // Start
    pinMode(PC3, PIN_MODE_INPUT_PULLUP); // Stop
    pinMode(PA2, PIN_MODE_OUTPUT);       // Buzzer

    Countdown_InitFromSeconds(60);  // 60 วินาที
    Countdown_SetAlarmCallback(onAlarm);

    char buf[16];
    printf("--- Countdown 60s Ready ---\r\n");

    while (1) {
        if (digitalRead(PC1) == LOW) {
            Delay_Ms(30);
            Countdown_Start();
            digitalWrite(PA2, LOW);  // ปิด buzzer
            while (digitalRead(PC1) == LOW);
            Delay_Ms(30);
        }

        if (digitalRead(PC3) == LOW) {
            Delay_Ms(30);
            Countdown_Stop();
            while (digitalRead(PC3) == LOW);
            Delay_Ms(30);
        }

        Countdown_GetTimeString(buf, TIME_FORMAT_MMSS);
        printf("\rRemaining: %s", buf);

        if (Countdown_IsFinished()) {
            printf("\r\n*** TIME'S UP! ***\r\n");
            Delay_Ms(3000);
            Countdown_Reset();
        }

        Delay_Ms(200);
    }
}

🔴 ตัวอย่างขั้นสูง

แนวคิด: Lap Timer — จับเวลา 5 laps, แสดง split time แต่ละ lap 
#include "simplehal.h"
#include <stdio.h>

#define MAX_LAPS 5

uint32_t lap_times[MAX_LAPS];
uint8_t lap_count = 0;

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(USART_115200);

    pinMode(PC1, PIN_MODE_INPUT_PULLUP); // Start
    pinMode(PC2, PIN_MODE_INPUT_PULLUP); // Lap
    pinMode(PC3, PIN_MODE_INPUT_PULLUP); // Reset

    Stopwatch_Init();
    char buf[16];

    printf("--- Lap Timer (5 Laps) ---\r\n");

    while (1) {
        // Start / Stop
        if (digitalRead(PC1) == LOW) {
            Delay_Ms(30);
            if (Stopwatch_IsRunning())
                Stopwatch_Stop();
            else
                Stopwatch_Start();
            while (digitalRead(PC1) == LOW);
            Delay_Ms(30);
        }

        // Record Lap
        if (digitalRead(PC2) == LOW && lap_count < MAX_LAPS) {
            Delay_Ms(30);
            lap_times[lap_count] = Stopwatch_GetTotalMilliseconds();
            printf("Lap %d: %lums\r\n", lap_count + 1, lap_times[lap_count]);
            lap_count++;
            while (digitalRead(PC2) == LOW);
            Delay_Ms(30);
        }

        // Reset
        if (digitalRead(PC3) == LOW) {
            Delay_Ms(30);
            Stopwatch_Reset();
            lap_count = 0;
            printf("--- Reset ---\r\n");
            while (digitalRead(PC3) == LOW);
            Delay_Ms(30);
        }

        if (lap_count >= MAX_LAPS) {
            printf("=== ALL %d LAPS ===\r\n", MAX_LAPS);
            for (uint8_t i = 0; i < MAX_LAPS; i++)
                printf("  Lap %d: %lums\r\n", i + 1, lap_times[i]);
            lap_count = 0;
        }

        Stopwatch_GetTimeString(buf, TIME_FORMAT_MMSS);
        printf("\rTime: %s", buf);
        Delay_Ms(100);
    }
}

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
ใช้ SimpleTIM TIM_2 พร้อมกันไม่ได้ Stopwatch จอง TIM_2 อยู่ — SimpleTIM ก็อยากใช้ TIM_2 ใช้คนละ timer: #define TIMEXT_DEFAULT_TIMER TIM_1 หรือใช้ SIMTIM กับ TIM_1
Stopwatch หยุดระหว่าง sleep MCU เข้า standby/sleep → TIM2 ปิด อย่าใช้ sleep mode เมื่อต้องการจับเวลา — ใช้ WFI หรือ WFE แทน
buffer เล็กเกินสำหรับ string ใช้ char[8] เก็บ HH:MM:SS → เขียนเกิน buffer ใช้ TIME_BUFFER_SIZE_HHMMSS (ค่าประมาณ 16) แทนการกำหนดเลขเอง
Countdown ไม่แจ้งเตือน ลืม poll Countdown_IsFinished() ใน main loop เรียก Countdown_IsFinished() ทุก loop iteration — หรือใช้ SetAlarmCallback ที่จะถูกเรียกใน ISR

💡 FAQ

Q: Stopwatch กับ SimpleTIM ต่างกันอย่างไร?
A: Stopwatch สร้างมาเพื่อจับเวลาที่เพิ่มขึ้น (elapsed time) — มี Start/Stop/Reset สำเร็จรูป SimpleTIM เป็น timer interrupt พื้นฐาน — คุณต้องเขียน logic เองทั้งหมด

Q: จับเวลาได้นานสุดเท่าไร?
A: ใช้ uint32_t เก็บมิลลิวินาที — ค่าสูงสุด ≈ 4,294,967,295ms ≈ 49.7 วัน

Q: ความละเอียดในการจับเวลาเป็นเท่าไร?
A: Stopwatch_GetTotalMilliseconds() ให้ความละเอียด 1ms — เกินพอกับการจับเวลาแบบคน (ปฏิกิริยามนุษย์ ~150–300ms)

✏️ ลองทำดู [เฉลย]
