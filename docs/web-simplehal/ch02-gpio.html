🤔 ทำไมต้องใช้?

GPIO (General Purpose Input/Output) คือประตูให้ MCU คุยกับโลกภายนอก ทุกอย่างที่ MCU ทำกับอุปกรณ์จริงล้วนต้องใช้ GPIO: เปิดปิดไฟ, อ่านสถานะปุ่มกด, วัดชีพจรจากเซนเซอร์, สั่งมอเตอร์หมุน, ส่งข้อมูลแบบ Parallel ไปจอ LCD ฯลฯ

SimpleGPIO ห่อหุ้มการตั้งค่า GPIO ของ CH32V003 ให้ใช้งานง่ายด้วย 5 ฟังก์ชันหลัก ภายในไม่กี่บรรทัด แทนที่จะต้องเขียน register เองทีละ bit

📚 ภาพรวมและหลักการทำงาน

GPIO แต่ละ pin บน CH32V003 มีโหมดการทำงานหลักดังนี้:

โหมดชื่อใน SimpleHALใช้เมื่อ
Push-Pull OutputPIN_MODE_OUTPUTขับ LED, สั่งอุปกรณ์อื่น, ให้สัญญาณ HIGH/LOW
Open-Drain OutputPIN_MODE_OUTPUT_ODใช้กับบัสที่ใช้ pull-up ภายนอก เช่น I2C
Input (Floating)PIN_MODE_INPUTต่อกับ R pull-down ภายนอก
Input + Pull-UpPIN_MODE_INPUT_PULLUPปุ่มกดต่อ GND (นิยมที่สุด)
Input + Pull-DownPIN_MODE_INPUT_PULLDOWNปุ่มกดต่อ VCC

ภายใน CH32V003 แต่ละ Port (PA, PC, PD) มี register ควบคุม 3 ตัว: GPIOx_CFGLR (ตั้งโหมด), GPIOx_OUTDR (เขียน output), GPIOx_INDR (อ่าน input) — SimpleHAL จัดการ register เหล่านี้ให้คุณโดยอัตโนมัติ

📋 API Reference

ฟังก์ชันคำอธิบายตัวอย่าง
pinMode(pin, mode)ตั้งโหมดให้ pin (Output/Input/PullUp/...)pinMode(PA1, PIN_MODE_OUTPUT);
digitalWrite(pin, value)เขียนค่า HIGH หรือ LOW ไปที่ pindigitalWrite(PA1, HIGH);
digitalRead(pin)อ่านค่า digital จาก pin (คืน HIGH/LOW)uint8_t v = digitalRead(PC1);
digitalToggle(pin)สลับสถานะ pin HIGH↔LOWdigitalToggle(PA1);

detachInterrupt(pin)ยกเลิก interrupt ที่ผูกไว้detachInterrupt(PC1);
pinModeMultiple(pins_array, mode)ตั้งโหมดหลาย pin พร้อมกัน (รับ array ของ pin)uint8_t leds[] = {PC0, PC1, PC2, PC3}; pinModeMultiple(leds, PIN_MODE_OUTPUT);

portWrite(port, value)เขียนครบ 8-bit ไปที่พอร์ต (รับ GPIO_TypeDef* เช่น GPIOC)portWrite(GPIOC, 0xAA);
portRead(port)อ่านครบ 8-bit จากพอร์ต (รับ GPIO_TypeDef* เช่น GPIOC)uint8_t v = portRead(GPIOC);
pulseIn(pin, state, timeout)วัดความกว้างพัลส์ (μs)uint32_t w = pulseIn(PC1, HIGH, 10000);
shiftOut(dataPin, clockPin, order, data)ส่งข้อมูลแบบ Serial Shift (8-bit)shiftOut(PA1, PA2, LSBFIRST, 0x55);
shiftIn(dataPin, clockPin, order)รับข้อมูลแบบ Serial Shift (8-bit)uint8_t v = shiftIn(PA2, PA1, MSBFIRST);

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Include Header + ตั้งค่าระบบ

#include "simplehal.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();

    // ... โค้ด GPIO จะใส่ตรงนี้ ...

    while (1) {
        // main loop
    }
}

🔍 SystemCoreClockUpdate() ต้องเรียกก่อนเสมอเพื่อให้ MCU รู้ความถี่ clock ที่แท้จริง Timer_Init() ตั้งค่า SysTick สำหรับระบบหน่วงเวลา

ขั้นที่ 2: ตั้งค่า pinMode สำหรับ LED และปุ่มกด

    // LED เป็น Output ดันกระแสออก
    pinMode(PA1, PIN_MODE_OUTPUT);
    pinMode(PA2, PIN_MODE_OUTPUT);
    pinMode(PC0, PIN_MODE_OUTPUT);

    // ปุ่มกดใช้ internal pull-up (กด = LOW)
    pinMode(PC1, PIN_MODE_INPUT_PULLUP);
    pinMode(PC2, PIN_MODE_INPUT_PULLUP);

🔍 ปุ่มกดต่อกับ GND — เมื่อไม่กด pull-up จะดึงให้เป็น HIGH (3.3V) เมื่อกดจะต่อลง GND ทำให้อ่านได้ LOW การใช้ internal pull-up ทำให้ไม่ต้องต่อ R ภายนอก

ขั้นที่ 3: อ่านปุ่มกด + ควบคุม LED (Polling)

    while (1) {
        // อ่านปุ่มแบบ polling
        if (digitalRead(PC1) == LOW) {   // กด = LOW (pull-up)
            digitalWrite(PA1, HIGH);      // LED1 ติด
            Delay_Ms(50);                 // debounce
            while (digitalRead(PC1) == LOW); // รอปล่อย
            Delay_Ms(50);
        }

        if (digitalRead(PC2) == LOW) {
            digitalWrite(PA1, LOW);       // LED1 ดับ
            Delay_Ms(50);
            while (digitalRead(PC2) == LOW);
            Delay_Ms(50);
        }
    }

🔍 การอ่านปุ่มกดแบบ polling ง่ายที่สุด แต่โปรแกรมจะติดรอตอนกดค้าง Delay_Ms(50) ช่วยกันสัญญาณ bounce ของหน้าสัมผัส

ขั้นที่ 4: Interrupt (ทางเลือก — ขั้นสูง)

volatile uint8_t btn_flag = 0;
volatile uint32_t last_isr_ms = 0;

void onBtnPressed(void) {
    uint32_t now = Get_CurrentMs();
    if (now - last_isr_ms < 30) return;  // software debounce
    last_isr_ms = now;
    btn_flag = 1;
}

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();

    pinMode(PA1, PIN_MODE_OUTPUT);
    pinMode(PC1, PIN_MODE_INPUT_PULLUP);

    attachInterrupt(PC1, onBtnPressed, FALLING);

    while (1) {
        if (btn_flag) {
            btn_flag = 0;
            digitalToggle(PA1);  // สลับ LED ทุกครั้งที่กด
        }
        // โปรแกรมทำงานอย่างอื่นได้ตลอดเวลา
    }
}

🔍 Interrupt ช่วยให้โปรแกรมไม่ต้องคอย polling — เมื่อ pin มีการเปลี่ยนแปลง (FALLING = HIGH→LOW) ISR จะถูกเรียกทันที volatile จำเป็นเพราะ flag ถูกแก้ไขจาก ISR

🟢 ตัวอย่างขั้นต้น

LED กระพริบ 1 ดวง — ใช้ digitalToggle() + Delay_Ms()

#include "simplehal.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();

    pinMode(PA1, PIN_MODE_OUTPUT);

    while (1) {
        digitalToggle(PA1);
        Delay_Ms(500);
    }
}

🟡 ตัวอย่างขั้นกลาง

กดปุ่มเปิด-ปิด LED — ใช้ digitalRead() + digitalWrite() แบบ polling

#include "simplehal.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();

    pinMode(PA1, PIN_MODE_OUTPUT);               // LED
    pinMode(PC1, PIN_MODE_INPUT_PULLUP);          // ปุ่มกด

    uint8_t led_state = 0;

    while (1) {
        if (digitalRead(PC1) == LOW) {           // กดปุ่ม
            Delay_Ms(30);                         // debounce
            if (digitalRead(PC1) == LOW) {       // ยืนยันว่ากดจริง
                led_state = !led_state;           // toggle state
                digitalWrite(PA1, led_state);
                while (digitalRead(PC1) == LOW); // รอปล่อยปุ่ม
                Delay_Ms(30);
            }
        }
    }
}

🔴 ตัวอย่างขั้นสูง

Interrupt-driven push button พร้อม software debounce ด้วย Get_CurrentMs() + ISR flag — โปรแกรมไม่ติดขัดตอนกดปุ่ม

#include "simplehal.h"

volatile uint8_t btn1_flag = 0;
volatile uint8_t btn2_flag = 0;
volatile uint32_t btn1_last = 0;
volatile uint32_t btn2_last = 0;

void btn1_isr(void) {
    uint32_t now = Get_CurrentMs();
    if (now - btn1_last < 40) return;
    btn1_last = now;
    btn1_flag = 1;
}

void btn2_isr(void) {
    uint32_t now = Get_CurrentMs();
    if (now - btn2_last < 40) return;
    btn2_last = now;
    btn2_flag = 1;
}

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();

    pinMode(PA1, PIN_MODE_OUTPUT);   // LED1
    pinMode(PA2, PIN_MODE_OUTPUT);   // LED2
    pinMode(PC1, PIN_MODE_INPUT_PULLUP);
    pinMode(PC2, PIN_MODE_INPUT_PULLUP);

    attachInterrupt(PC1, btn1_isr, FALLING);
    attachInterrupt(PC2, btn2_isr, FALLING);

    while (1) {
        if (btn1_flag) {
            btn1_flag = 0;
            digitalToggle(PA1);
        }
        if (btn2_flag) {
            btn2_flag = 0;
            digitalToggle(PA2);
        }
        // CPU ว่างทำอย่างอื่น
    }
}

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
ปุ่มกดอ่านค่าผิด pinMode ไม่ได้ตั้ง pull-up — pin ลอย (floating) ทำให้อ่านค่าไม่แน่นอน ใช้ PIN_MODE_INPUT_PULLUP หรือต่อ R pull-up ภายนอก 10kΩ
LED ไม่ติด กลับขั้ว LED / ไม่มีตัวต้านทาน / กระแสเกิน pin พัง ตรวจสอบขั้ว LED (แอโนดต่อ pin, แคโทดต่อ GND) ต้องมี R 220Ω เสมอ
MCU รีเซ็ตผิดปกติ กระแสรวมของทุก pin เกิน 8mA ต่อ pin หรือ 25mA รวม ใช้ R คำนวณกระแส: I = (3.3V − Vf_LED) / R; ใช้ transistor ขับโหลดหนัก
Interrupt ไม่ทำงาน ลืมตั้ง pinMode เป็น input ก่อนเรียก attachInterrupt เรียก pinMode(pin, PIN_MODE_INPUT_PULLUP) ก่อนเสมอ
"Pin not valid" ตอน compile ใช้ pin ที่ไม่มีในแพ็คเกจ (เช่น PD0 บน SOP-8) ตรวจสอบ CH32V003_PACKAGE ใน simplehal_config.h

💡 เกร็ดความรู้ / FAQ

Q: HIGH และ LOW คืออะไร?
A: HIGH = 3.3V (แรงดัน logic สูง), LOW = 0V (แรงดัน logic ต่ำ) — CH32V003 เป็น MCU 3.3V ห้ามต่อ 5V เข้า pin โดยตรง

Q: ทำไมต่อ LED ต้องมีตัวต้านทาน?
A: LED ไม่มีตัวจำกัดกระแสในตัวเอง — ถ้าต่อ pin โดยตรง LED จะกินกระแสไม่จำกัด อาจทำให้ pin MCU หรือ LED เสียหาย R 220Ω จำกัดกระแสที่ ~8mA (ปลอดภัย)

Q: Active-HIGH กับ Active-LOW ต่างกันยังไง?
A: Active-HIGH = OUTPUT HIGH → LED ติด (ต่อ LED แคโทดลง GND) Active-LOW = OUTPUT LOW → LED ติด (ต่อ LED แอโนดเข้า VCC) — SimpleHAL ใช้ Active-HIGH เป็นหลัก

Q: internal pull-up มีค่าเท่าไร?
A: Internal pull-up ของ CH32V003 มีค่าประมาณ 40kΩ — แรงพอสำหรับปุ่มกด แต่ถ้าสายยาวหรือมีสัญญาณรบกวนมาก ควรใช้ R ภายนอก 4.7–10kΩ

✏️ ลองทำดู [เฉลย]

  1. LED 3 ดวงวิ่ง Knight Rider — ใช้ digitalWrite() ไล่เปิดปิด LED ทีละดวงวนไปเรื่อย ๆ (ซ้าย→ขวา→ซ้าย) ด้วย Delay_Ms(100)
  2. ปุ่มกดเปลี่ยนความเร็วกระพริบ — ใช้ 2 ปุ่ม: ปุ่มหนึ่งเร่งความเร็ว อีกปุ่มลดความเร็ว LED โดยอ่านปุ่มแบบ interrupt + flag
  3. ใช้ pulseIn() วัด double-click — ต่อปุ่มกดเข้ากับ PC1 ใช้ pulseIn วัดระยะเวลาระหว่างการกดครั้งแรกกับครั้งที่สอง แยก single-click กับ double-click
