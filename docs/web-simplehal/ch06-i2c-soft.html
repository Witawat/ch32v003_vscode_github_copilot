ระดับขั้นต้น: I2C Bus ที่ 2

แนวคิด: HW I2C คุยกับเซนเซอร์หลัก + SW I2C คุยกับอุปกรณ์เสริม 
#include "SimpleHAL.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);

    // Bus 1: Hardware I2C (sensor หลัก)
    I2C_SimpleInit(I2C_100KHZ, I2C_PINS_DEFAULT);

    // Bus 2: Software I2C (อุปกรณ์เสริม) — ใช้ PA1=SCL, PA2=SDA
    I2C_Soft_Init(PA1, PA2, I2C_SOFT_100KHZ);

    printf("Dual I2C bus ready\r\n");
    while (1) {
        // อ่านจาก bus 1 (HW)
        uint8_t val1 = I2C_ReadReg(0x50, 0x00);

        // อ่านจาก bus 2 (SW) — เขียน register address ก่อน แล้วค่อยอ่านค่า
        uint8_t val2;
        uint8_t reg = 0x00;
        I2C_Soft_Write(0x50, ®, 1, 0);
        I2C_Soft_Read(0x50, &val2, 1);

        Delay_Ms(500);
    }
}

ระดับขั้นกลาง: ดีบักด้วย Logic Analyzer

แนวคิด: ใช้ ignore_ack=1 เพื่อส่งข้อมูลแม้ไม่มี device — เหมาะกับการตรวจสอบสัญญาณ 
#include "SimpleHAL.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);

    // Software I2C บนขาใดก็ได้ — debug mode ignore_ack
    I2C_Soft_Init(PD2, PD3, I2C_SOFT_100KHZ);

    printf("Sending test data (ignore ack)...\r\n");
    while (1) {
        // ส่งข้อมูลโดยไม่สนใจว่า device ตอบ ACK หรือไม่
        uint8_t test_data[] = {0x00, 0x42};
        I2C_Soft_Write(0x50, test_data, 2, 1); // ignore_ack=1
        Delay_Ms(1000);
    }
}

ระดับขั้นสูง: Custom SHT31 Read ด้วย Low-level API

แนวคิด: ใช้ Start, Stop, WriteByte, ReadByte ควบคุมทุกขั้นตอนเอง — รองรับ protocol ที่ซับซ้อน 
#include "SimpleHAL.h"

#define SHT31_ADDR 0x44

typedef struct {
    uint8_t SCL;
    uint8_t SDA;
} I2C_Soft_Handle;

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);

    I2C_Soft_Init(PC6, PC7, I2C_SOFT_100KHZ);

    while (1) {
        // Start → Write address → Stop → Delay → Start → Read data → Stop
        I2C_Soft_Start();
        I2C_Soft_WriteByte(SHT31_ADDR << 1);    // Write mode
        I2C_Soft_WriteByte(0x2C);               // Command MSB
        I2C_Soft_WriteByte(0x06);               // Command LSB
        I2C_Soft_Stop();

        Delay_Ms(15); // รอการวัด

        I2C_Soft_Start();
        I2C_Soft_WriteByte((SHT31_ADDR << 1) | 1); // Read mode
        uint8_t data[6];
        for (int i = 0; i < 5; i++) {
            data[i] = I2C_Soft_ReadByte(1); // ack=1 (ยังมีข้อมูลเหลือ)
        }
        data[5] = I2C_Soft_ReadByte(0); // ack=0 → NAK (ไบต์สุดท้าย)
        I2C_Soft_Stop();

        uint16_t temp_raw = (data[0] << 8) | data[1];
        float temp = -45.0f + 175.0f * (temp_raw / 65535.0f);
        printf("Temp = %.1f C\r\n", temp);
        Delay_Ms(2000);
    }
}

คำถามที่พบบ่อย (FAQ)

เมื่อไหร่ควรใช้ Software I2C?

เมื่อต้องการ I2C bus ที่ 2, ใช้กับ Logic Analyzer เพื่อดีบัก, หรือต้องการอิสระในการเลือก pin

ลองทำดู (แบบฝึกหัด)

  1. สร้าง I2C bus ที่ 2 สำหรับต่อจอ OLED
  2. ใช้ Logic Analyzer ดูสัญญาณ I2C ที่สร้างจาก software
  3. Implement custom read protocol ด้วย low-level API
