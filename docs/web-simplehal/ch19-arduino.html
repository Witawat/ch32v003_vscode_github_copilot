ขั้นตอนการใช้งาน (Step by Step)

1. Include SimpleHAL.h

#include "SimpleHAL.h" — ฟังก์ชัน Arduino-style มีให้พร้อมใช้เลย

2. เรียก millis() / delay() ได้ทันที

หลังจาก Timer_Init() — millis() และ delay() ใช้ได้ทันที

3. randomSeed ก่อนใช้ random

randomSeed(ADC_Read()) — ใช้ค่า noise จาก ADC เป็น seed

4. dtostrf — บัฟเฟอร์ต้องใหญ่พอ

บัฟเฟอร์ต้องมีขนาด width + 2 ไบต์ (เผื่อ null terminator และเครื่องหมาย)

ระดับขั้นต้น: Blink Non-blocking ด้วย millis()

#include "SimpleHAL.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    GPIO_Init(GPIOC, GPIO_Pin_0, GPIO_Mode_Out_PP);

    uint32_t last_toggle = millis();
    uint8_t led_state = 0;

    while (1) {
        if (millis() - last_toggle >= 500) {
            last_toggle = millis();
            led_state = !led_state;
            if (led_state) {
                GPIO_WriteBit(GPIOC, GPIO_Pin_0, 1);
            } else {
                GPIO_WriteBit(GPIOC, GPIO_Pin_0, 0);
            }
        }

        // ยังทำงานอื่นต่อได้ ไม่ block เหมือน delay()
    }
}

ระดับขั้นกลาง: Random LED Dice

แนวคิด: ใช้ randomSeed จาก ADC noise → สุ่มเลข 1-6 (เหมือนลูกเต๋า) → กระพริบ LED ตามจำนวนที่สุ่มได้ 
#include "SimpleHAL.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(USART_115200);
    GPIO_Init(GPIOC, GPIO_Pin_0, GPIO_Mode_Out_PP);

    randomSeed(ADC_Read()); // seed จาก ADC noise

    while (1) {
        uint8_t dice = _randomRange(1, 7); // 1-6
        printf("Dice: %d\r\n", dice);

        // กระพริบ LED ตามแต้ม
        for (uint8_t i = 0; i < dice; i++) {
            GPIO_WriteBit(GPIOC, GPIO_Pin_0, 1);
            delay(200);
            GPIO_WriteBit(GPIOC, GPIO_Pin_0, 0);
            delay(200);
        }

        delay(2000); // รอ 2 วิ ก่อนทอยใหม่
    }
}

ระดับขั้นสูง: yield() + IWDG — Cooperative Multitasking

แนวคิด: ใช้ yield() ใน main loop เพื่อ feed IWDG อัตโนมัติ — ไม่ต้องเรียก IWDG_Feed เอง — เหมาะกับ multi-tasking 
#include "SimpleHAL.h"

void task_blink_led(void) {
    static uint32_t last = 0;
    if (millis() - last >= 500) {
        last = millis();
        GPIO_WriteBit(GPIOC, GPIO_Pin_0,
            (GPIO_ReadOutputDataBit(GPIOC, GPIO_Pin_0)) ? 0 : 1);
    }
}

void task_print_ms(void) {
    static uint32_t last = 0;
    if (millis() - last >= 2000) {
        last = millis();
        printf("Uptime: %lu ms\r\n", millis());
    }
}

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(USART_115200);
    GPIO_Init(GPIOC, GPIO_Pin_0, GPIO_Mode_Out_PP);

    IWDG_SimpleInit(2000); // เปิด watchdog

    while (1) {
        task_blink_led();
        task_print_ms();

        yield(); // เรียกทุก loop = auto-feed IWDG
    }
}

ข้อควรระวังสำคัญ

  • _randomMax / _randomRange — ใช้ชื่อนี้เพราะ random ชนกับ stdlib.h
  • ต้องเรียก randomSeed() ก่อนใช้ random
  • dtostrf — บัฟเฟอร์ต้องใหญ่กว่า width อย่างน้อย 2 bytes
  • delay() ใน ISR = ห้ามเด็ดขาด

ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
random() ชนกับ stdlib.hMacro conflictใช้ _randomMax() หรือ #define ENABLE_ARDUINO_RANDOM_MACRO
yield() ไม่ feed IWDGยังไม่ได้เริ่ม IWDGเรียก IWDG_SimpleInit() ก่อน loop
dtostrf output เพี้ยนBuffer เล็กเกินไปเพิ่ม buffer เป็น width + 2
delay() ใน ISR → ค้างISR block main loopห้ามใช้ delay ใน ISR — ใช้ flag แทน

คำถามที่พบบ่อย (FAQ)

ทำไมต้องมี SimpleArduino?

เพื่อให้คนที่ย้ายจาก Arduino มา CH32V003 เริ่มต้นได้ง่าย — ใช้ชื่อฟังก์ชันเดิม ไม่ต้องเรียนรู้ใหม่ทั้งหมด

ต่างจาก Arduino จริงอย่างไร?

API ชื่อเหมือนกัน — แต่ hardware ข้างล่างเป็น CH32V003 เต็ม ๆ — ประสิทธิภาพดีกว่า เร็วกว่า ถูกกว่า

ฟังก์ชันไหนใช้แทนกันได้บ้าง?

millis, delay, random, dtostrf — ครอบคลุมฟังก์ชันพื้นฐานที่ใช้บ่อยที่สุด

  3. พิมพ์แรงดัน float ด้วย dtostrf — อ่าน ADC แปลงเป็นแรงดัน แล้วแสดงผล
