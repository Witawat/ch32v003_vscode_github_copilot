บทที่ 9: PWM — ควบคุมความสว่างและมอเตอร์

ทำไมต้องใช้?

API Reference

ฟังก์ชันคำอธิบาย
PWM_Init()เริ่มต้น PWM channel
PWM_InitRemap()เริ่มต้น PWM แบบ remap
PWM_SetDutyCycle()ตั้งค่า duty cycle เป็น % (0-100)
PWM_SetDutyCycleRaw()ตั้งค่า duty cycle แบบ raw value
PWM_Write()analogWrite — ตั้งค่า 0-255
PWM_SetFrequency()ตั้งค่าความถี่ PWM
PWM_GetPeriod()อ่านค่า period ปัจจุบัน
PWM_Start()เริ่ม PWM output
PWM_Stop()หยุด PWM output
PWM_GetDutyCycle()อ่านค่า duty cycle ปัจจุบัน
PWM_AdvancedInit()เริ่มต้น PWM แบบปรับละเอียด
PWM_SetPolarity()ตั้งค่าขั้วสัญญาณ (active high/low)

Timer Resource Guard

ถ้า TIM1 หรือ TIM2 ถูกใช้โดย SimpleTIM อยู่แล้ว — PWM_Init() จะ return ทันที โดยไม่เขียนทับ

ระดับขั้นต้น: Fade LED

แนวคิด: วนลูป 0% → 100% duty cycle 
#include "SimpleHAL.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    PWM_Init(PWM1_CH1, 1000); // 1kHz

    while (1) {
        // Fade in
        for (uint8_t d = 0; d <= 100; d++) {
            PWM_SetDutyCycle(PWM1_CH1, d);
            Delay_Ms(10);
        }
        // Fade out
        for (uint8_t d = 100; d > 0; d--) {
            PWM_SetDutyCycle(PWM1_CH1, d);
            Delay_Ms(10);
        }
    }
}

ระดับขั้นกลาง: Servo Control

แนวคิด: ตั้งค่า PWM 50Hz, คำนวณ pulse width สำหรับมุม 0-180° (ปกติ 500-2500μs) 
#include "SimpleHAL.h"

#define SERVO_PWM PWM1_CH1

void servo_set_angle(uint8_t angle) {
    // 0° = 500us, 180° = 2500us, period = 20000us (50Hz)
    uint16_t period = PWM_GetPeriod(SERVO_PWM);
    uint16_t pulse = 500 + (2000 * angle / 180);
    uint16_t raw = (uint32_t)pulse * period / 20000;
    PWM_SetDutyCycleRaw(SERVO_PWM, raw);
}

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();

    // Servo ต้องการ 50Hz
    PWM_Init(SERVO_PWM, 50);

    while (1) {
        servo_set_angle(0);   Delay_Ms(1000);
        servo_set_angle(90);  Delay_Ms(1000);
        servo_set_angle(180); Delay_Ms(1000);
    }
}

ระดับขั้นสูง: DC Motor Speed + Direction (L298N)

แนวคิด: ใช้ 2 PWM channels + direction control — IN1=PWM, IN2=PWM ควบคุมทั้งความเร็วและทิศทาง 
#include "SimpleHAL.h"

#define MOTOR_IN1 PWM1_CH1
#define MOTOR_IN2 PWM1_CH2

void motor_set(int8_t speed) {
    // speed: -100..0..100 (ลบ=ถอยหลัง, บวก=เดินหน้า)
    if (speed >= 0) {
        PWM_SetDutyCycle(MOTOR_IN1, speed);
        PWM_SetDutyCycle(MOTOR_IN2, 0);
    } else {
        PWM_SetDutyCycle(MOTOR_IN1, 0);
        PWM_SetDutyCycle(MOTOR_IN2, -speed);
    }
}

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();

    // มอเตอร์ต้องการความถี่สูง >10kHz กันเสียงหอน
    PWM_Init(MOTOR_IN1, 20000);
    PWM_Init(MOTOR_IN2, 20000);

    while (1) {
        motor_set(50);   Delay_Ms(2000); // เดินหน้า 50%
        motor_set(0);    Delay_Ms(1000); // หยุด
        motor_set(-50);  Delay_Ms(2000); // ถอยหลัง 50%
        motor_set(0);    Delay_Ms(1000);
    }
}

ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
Servo สั่นคำนวณ pulse width ผิดตรวจสอบ servo datasheet (ปกติ 500-2500μs)
PWM_Write ไม่ตรง %255 ≠ 100%PWM_Write(pin, 255) = 100% เทียบกับ analogWrite
PC4 ADC+PWM ชนกันขาเดี่ยวใช้ทั้ง ADC และ PWMเลือกใช้อย่างใดอย่างหนึ่ง
Frequency sharingตั้งความถี่ผิด channelตั้งความถี่ก่อน init channel — ทุก channel timer เดียวกันใช้ freq เดียวกัน

คำถามที่พบบ่อย (FAQ)

analogWrite กับ PWM_Write ต่างกันอย่างไร?

analogWrite(pin, 0-255) เป็น macro ที่เรียก PWM_Write() — ให้ความรู้สึกแบบ Arduino

ความถี่ PWM ไหนเหมาะกับอะไร?

LED >100Hz (ไม่กระพริบ), Servo = 50Hz (มาตรฐาน), Motor >10kHz (ไม่หอน), Audio >20kHz

SOP-8 มีกี่ PWM channels?

แค่ 2 channels: PWM1_CH1 + PWM2_CH1

ลองทำดู (แบบฝึกหัด)

  1. Fade LED เข้า-ออก พร้อมปรับความเร็ว fade
  2. หมุน Servo 0° → 90° → 180° → 0° วนซ้ำ
  3. RGB LED 3 สี PWM — ผสมสีได้ทุกสี
