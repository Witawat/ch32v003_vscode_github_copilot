📝 เฉลย — บทที่ 3: Delay & Timer — ควบคุมเวลา

ข้อ 1: LED 3 ดวงกระพริบ 3 ความเร็วด้วย Timer_t

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include "SimpleHAL.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();

    pinMode(PC0, PIN_MODE_OUTPUT);
    pinMode(PC1, PIN_MODE_OUTPUT);
    pinMode(PC2, PIN_MODE_OUTPUT);

    Timer_t t1, t2, t3;
    Start_Timer(&t1, 100, 1);   // 100ms = 5Hz
    Start_Timer(&t2, 500, 1);   // 500ms = 1Hz
    Start_Timer(&t3, 1000, 1);  // 1000ms = 0.5Hz

    while (1) {
        if (Is_Timer_Expired(&t1)) digitalToggle(PC0);
        if (Is_Timer_Expired(&t2)) digitalToggle(PC1);
        if (Is_Timer_Expired(&t3)) digitalToggle(PC2);
    }
}
🧠 แนวคิด: Timer_t 3 ตัว independent กัน — repeat=1 ให้วนซ้ำอัตโนมัติ — Is_Timer_Expired เช็คทีละตัว — Non-blocking: main loop ไม่หยุดรอ LED ทั้ง 3 ดวงทำงานพร้อมกันด้วยความถี่ต่างกัน

ข้อ 2: นับเวลากดปุ่มค้าง

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include "SimpleHAL.h"

volatile uint32_t press_start = 0;
volatile uint32_t press_duration = 0;
volatile uint8_t  press_done = 0;

void btn_isr(void) {
    if (digitalRead(PC1) == LOW) {
        press_start = Get_CurrentMs();  // กด → เริ่มจับ
    } else {
        press_duration = Get_CurrentMs() - press_start;  // ปล่อย
        press_done = 1;
    }
}

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();

    pinMode(PC1, PIN_MODE_INPUT_PULLUP);
    attachInterrupt(PC1, btn_isr, CHANGE);
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);

    while (1) {
        if (press_done) {
            press_done = 0;
            USART_Print("Hold: ");
            USART_PrintNum((int32_t)press_duration);
            USART_Print(" ms\r\n");
        }
        Delay_Ms(10);
    }
}
🧠 แนวคิด: attachInterrupt ใช้ CHANGE mode → จับทั้ง FALLING (กด) และ RISING (ปล่อย) — ISR อ่านสถานะ pin ทันที: LOW=เริ่มจับเวลา, HIGH=คำนวณ elapsed โดยใช้ Get_CurrentMs — กดค้างนาน = ตัวเลขมาก — volatile flag ให้ main loop แสดงผล

ข้อ 3: Fade LED ด้วย non-blocking timer

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include "SimpleHAL.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();

    pinMode(PC0, PIN_MODE_OUTPUT);
    PWM_Init(PWM2_CH3, 1000);  // ใช้ PWM จริงง่ายกว่า software

    Timer_t fade_timer;
    Start_Timer(&fade_timer, 2, 1);  // 2ms tick
    int8_t brightness = 0, dir = 1;

    while (1) {
        if (Is_Timer_Expired(&fade_timer)) {
            brightness += dir;
            if (brightness >= 100) dir = -1;
            if (brightness <= 0)   dir = 1;
            PWM_SetDutyCycle(PWM2_CH3, brightness);
        }
        // main loop ว่างทำงานอื่นได้
    }
}
🧠 แนวคิด: ใช้ Timer_t 2ms เป็น tick — counter brightness เพิ่ม/ลดทีละ 1 ด้วย dir flag — เมื่อถึง 100 หรือ 0 กลับทิศ — PWM_SetDutyCycle อัปเดตสว่าง/มืดในแต่ละ tick — Non-blocking: ใช้ PWM จริงบน PC0 (PWM2_CH3) ไม่ต้อง software PWM — main loop ว่างทำงานอื่น
