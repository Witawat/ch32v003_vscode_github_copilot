🤔 ทำไมต้องใช้?

Hardware Timer ให้คุณทำงานตามจังหวะที่แม่นยำระดับ hardware — ไม่ต้องใช้ Delay_Ms() แบบ blocking ที่หยุดโปรแกรมทั้งหมด ใช้สร้าง scheduler, multi-rate tasks, PWM alternative timing, หรือ input capture สำหรับวัดความถี่สัญญาณ

📚 Timer Instances

TimerประเภทChannelหมายเหตุ
TIM_1Advanced4 channelsรองรับ PWM, complementary output
TIM_2General4 channelsใช้เป็น Timer หรือ PWM ได้

🔒 Timer Resource Guard

SimpleHAL มีระบบป้องกันการแย่งใช้ Timer ตัวเดียวกัน — ตรวจสอบผ่าน global variable:

ค่าความหมาย
TIM_OWNER_NONETimer ยังไม่มีใครใช้
TIM_OWNER_PWMTimer ถูกใช้โดย PWM module
TIM_OWNER_TIMERTimer ถูกใช้โดย SimpleTIM (บทนี้)
TIM_OWNER_TIMEXTTimer ถูกใช้โดย Stopwatch/Countdown (บทที่ 12)
สำคัญ: ถ้า PWM ใช้ TIM_1 อยู่แล้ว จะเรียก TIM_SimpleInit(TIM_1) ไม่ได้ — ต้องใช้ TIM_2 หรือแก้ไขการตั้งค่า PWM

📋 API Reference

ฟังก์ชันคำอธิบายตัวอย่าง
TIM_SimpleInit(timer)เริ่มต้น timer สำหรับ periodic interruptTIM_SimpleInit(TIM_2);
TIM_Start(timer)เริ่มนับ timerTIM_Start(TIM_2);
TIM_Stop(timer)หยุดนับ timerTIM_Stop(TIM_2);
TIM_SetFrequency(timer, freq)ตั้งความถี่ interrupt (Hz)TIM_SetFrequency(TIM_2, 1000); // 1kHz
TIM_AttachInterrupt(timer, callback)ผูก callback function กับ timer interruptTIM_AttachInterrupt(TIM_2, onTick);
TIM_DetachInterrupt(timer)ยกเลิก callbackTIM_DetachInterrupt(TIM_2);
Simple_TIM_GetCounter(timer)อ่านค่า counter ปัจจุบันuint32_t cnt = Simple_TIM_GetCounter(TIM_2);
Simple_TIM_SetCounter(timer, val)ตั้งค่า counter ใหม่Simple_TIM_SetCounter(TIM_2, 0);
TIM_GetPeriod(timer)อ่านค่า period (auto-reload)uint32_t per = TIM_GetPeriod(TIM_2);
TIM_SetMode(timer, mode)เปลี่ยนโหมด timerTIM_SetMode(TIM_2, TIM_MODE_UP);
TIM_AdvancedInit(timer, prescaler, period, mode)ตั้งค่า timer แบบละเอียดTIM_AdvancedInit(TIM_1, 47, 999, TIM_MODE_UP);
TIM_SetPrescaler(timer, psc)ตั้งค่า prescalerTIM_SetPrescaler(TIM_2, 4800);
TIM_GenerateUpdate(timer)สร้าง update event ด้วย softwareTIM_GenerateUpdate(TIM_2);
สูตรความถี่:
Freq = 48MHz / (prescaler + 1) / (period + 1)
ตัวอย่าง: 1kHz = 48,000,000 / (4800) / (10) ⇒ prescaler=4799, period=9

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Include + ประกาศ Callback

#include "simplehal.h"

volatile uint32_t tick_count = 0;

void onTimerTick(void) {
    tick_count++;  // ทำงานใน ISR — ต้องสั้นและเร็ว
}

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();

🔍 volatile สำคัญ — ตัวแปรที่แก้ไขใน ISR ต้องประกาศเป็น volatile เสมอ เพื่อป้องกัน compiler optimize ทิ้ง

ขั้นที่ 2: ตั้งค่า Timer + ผูก Callback

    TIM_SimpleInit(TIM_2);
    TIM_SetFrequency(TIM_2, 1000);        // 1kHz = 1000 interrupts/วินาที
    TIM_AttachInterrupt(TIM_2, onTimerTick);
    TIM_Start(TIM_2);

🔍 TIM_SetFrequency จะคำนวณ prescaler และ period ให้อัตโนมัติจาก clock 48MHz

ขั้นที่ 3: Main Loop — ทำงานพร้อมจังหวะ Timer

    pinMode(PA1, PIN_MODE_OUTPUT);

    while (1) {
        // LED toggle ทุก 500ms (tick_count ถูกเพิ่มใน ISR)
        if (tick_count >= 500) {
            tick_count = 0;
            digitalToggle(PA1);
        }
        // CPU ว่างทำงานอย่างอื่นได้ตลอดเวลา
    }

🔍 main loop ไม่เคยถูกบล็อก — ตรวจสอบ flag/tick แทนการใช้ Delay แตกต่างจาก Delay_Ms(500) แบบ blocking อย่างสิ้นเชิง

🟢 ตัวอย่างขั้นต้น

LED กระพริบด้วย Timer Interrupt — ตั้ง Timer 2Hz → toggle ทุก interrupt = LED กระพริบ 1Hz

#include "simplehal.h"

void onTimerTick(void) {
    digitalToggle(PA1);  // toggle โดยตรงใน ISR — ทำงานเร็วมาก
}

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();

    pinMode(PA1, PIN_MODE_OUTPUT);

    TIM_SimpleInit(TIM_2);
    TIM_SetFrequency(TIM_2, 2);  // 2Hz = toggle LED 1Hz
    TIM_AttachInterrupt(TIM_2, onTimerTick);
    TIM_Start(TIM_2);

    while (1) {
        // CPU idle — ทำอะไรก็ได้
    }
}

🟡 ตัวอย่างขั้นกลาง

แนวคิด: Simple Scheduler — 1kHz tick → ทุก 100ms อ่าน ADC, ทุก 1s heartbeat LED 
#include "simplehal.h"

volatile uint32_t ticks = 0;

void tick_isr(void) {
    ticks++;
}

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
    ADC_SimpleInit();

    pinMode(PA1, PIN_MODE_OUTPUT);

    TIM_SimpleInit(TIM_2);
    TIM_SetFrequency(TIM_2, 1000);  // 1kHz base tick
    TIM_AttachInterrupt(TIM_2, tick_isr);
    TIM_Start(TIM_2);

    uint32_t last_adc = 0, last_hb = 0;

    while (1) {
        uint32_t now = ticks;

        // Task 1: อ่าน ADC ทุก 100ms
        if (now - last_adc >= 100) {
            last_adc = now;
            uint16_t val = ADC_Read(PA2);
            USART_Print("ADC: ");
            USART_PrintNum(val);
            USART_Print("\r\n");
        }

        // Task 2: Heartbeat LED ทุก 1000ms
        if (now - last_hb >= 1000) {
            last_hb = now;
            digitalToggle(PA1);
        }
    }
}

🔴 ตัวอย่างขั้นสูง

แนวคิด: Input Capture — วัด pulse width ด้วย TIM2 free-running @1MHz + GPIO interrupt 
#include "simplehal.h"

volatile uint32_t pulse_start = 0;
volatile uint32_t pulse_width = 0;

void onRisingEdge(void) {
    pulse_start = Simple_TIM_GetCounter(TIM_2);  // จับเวลา start
}

void onFallingEdge(void) {
    uint32_t now = Simple_TIM_GetCounter(TIM_2);
    pulse_width = now - pulse_start;  // คำนวณความกว้างพัลส์
}

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);

    // TIM2 free-running @ 1MHz (1 tick = 1μs)
    TIM_SimpleInit(TIM_2);
    TIM_SetFrequency(TIM_2, 1000000);
    TIM_Start(TIM_2);

    // ตั้ง PC1 เป็น input + attach interrupt
    pinMode(PC1, PIN_MODE_INPUT_PULLUP);
    attachInterrupt(PC1, onRisingEdge, RISING);
    // เปลี่ยน edge ใน callback — หรือใช้ทั้งสองขา

    while (1) {
        if (pulse_width > 0) {
            USART_Print("Pulse width: ");
            USART_PrintNum(pulse_width);
            USART_Print(" us\r\n");
            pulse_width = 0;
        }
        Delay_Ms(500);
    }
}

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
PWM กับ Timer ชนกัน ใช้ TIM_1 ทั้ง PWM และ Timer interrupt พร้อมกัน ใช้คนละ timer — PWM ใช้ TIM_1, Timer ใช้ TIM_2
Callback ทำงานนานเกินไป วนลูป หรือ print ใน ISR — ทำให้ ISR ใช้เวลานาน ระบบสะดุด ISR ควรสั้นที่สุด — ใช้ flag ให้ main loop จัดการ
ตั้งความถี่ต่ำมาก ๆ ไม่ได้ Prescaler มีค่า max 65535 — ความถี่ต่ำสุด = 48M/65536/65536 ≈ 0.011Hz ใช้ software counter ใน ISR แทน — นับ tick จนครบแล้วค่อยทำงาน
tick overflow หลังทำงานนาน uint32_t ms จะ overflow ที่ ~49.7 วัน ใช้ subtraction ในการเปรียบเทียบเวลา: if (now - last >= interval) — overflow-safe

💡 FAQ

Q: SimpleTIM ต่างจาก SimpleDelay อย่างไร?
A: SimpleTIM ใช้ hardware timer interrupt — โปรแกรมไม่หยุด, SimpleDelay ใช้ SysTick busy-wait — โปรแกรมหยุดรอ การเลือกใช้ขึ้นกับงาน: ถ้าต้องการ non-blocking ใช้ SimpleTIM

Q: TIM_1 กับ TIM_2 ต่างกันอย่างไร?
A: ทั้งคู่ใช้ API เดียวกัน — ต่างกันที่ hardware: TIM_1 เป็น Advanced timer (4 channels, complementary output), TIM_2 เป็น General purpose (4 channels) — สำหรับ periodic interrupt ใช้ตัวไหนก็ได้

Q: เปลี่ยนความถี่ Timer ระหว่างรันได้ไหม?
A: ได้ — เรียก TIM_Stop() → TIM_SetFrequency() → TIM_Start() ใหม่
