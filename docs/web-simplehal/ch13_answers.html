📝 เฉลย — บทที่ 13: DMA — Direct Memory Access

ข้อ 1: DMA copy 256 bytes วัดเวลา เทียบ CPU

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include <string.h>

#define BUF_SIZE 256

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);

    uint8_t src[BUF_SIZE], dst1[BUF_SIZE], dst2[BUF_SIZE];
    for (int i = 0; i < BUF_SIZE; i++) src[i] = (uint8_t)i;

    // CPU copy
    uint32_t t0 = Get_CurrentUs();
    for (int i = 0; i < BUF_SIZE; i++) dst1[i] = src[i];
    uint32_t cpu_time = Get_CurrentUs() - t0;

    // DMA copy
    uint32_t t1 = Get_CurrentUs();
    DMA_MemCopy(dst2, src, BUF_SIZE);
    uint32_t dma_time = Get_CurrentUs() - t1;

    USART_Print("CPU copy: "); USART_PrintNum((int32_t)cpu_time);
    USART_Print(" us\r\n");
    USART_Print("DMA copy: "); USART_PrintNum((int32_t)dma_time);
    USART_Print(" us\r\n");

    // Verify
    uint8_t ok = (memcmp(dst1, dst2, BUF_SIZE) == 0);
    USART_Print(ok ? "Data OK!\r\n" : "Data MISMATCH!\r\n");

    while (1);
}
🧠 แนวคิด: จับเวลา Get_CurrentUs ก่อนและหลังแต่ละวิธี — CPU copy: for loop ธรรมดา — DMA copy: DMA_MemCopy แบบ blocking — เปรียบเทียบเวลา µs — DMA เร็วกว่าที่ buffer ใหญ่ (>64 bytes) และ CPU ว่างทำงานอื่นได้ (แต่แบบ blocking CPU รอ) — memcmp ตรวจสอบข้อมูลถูกต้อง

ข้อ 2: ADC DMA circular buffer 10kHz

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include "SimpleHAL.h"

#define BUF_SIZE 100

uint16_t adc_buf[BUF_SIZE];

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    ADC_SimpleInit();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);

    // เริ่ม ADC DMA circular (circular=1)
    DMA_analogReadStart(PD2, adc_buf, BUF_SIZE, 1);

    Timer_t report_timer;
    Start_Timer(&report_timer, 1000, 1);  // ทุก 1 วินาที

    USART_Print("ADC DMA started\r\n");

    while (1) {
        if (Is_Timer_Expired(&report_timer)) {
            // คำนวณค่าเฉลี่ย
            uint32_t sum = 0;
            for (int i = 0; i < BUF_SIZE; i++) {
                sum += adc_buf[i];
            }
            uint16_t avg = (uint16_t)(sum / BUF_SIZE);
            uint16_t latest = adc_buf[BUF_SIZE - 1];

            USART_Print("Avg="); USART_PrintNum(avg);
            USART_Print(" Latest="); USART_PrintNum(latest);
            USART_Print(" mV=");
            USART_PrintNum((int32_t)(((uint32_t)avg * 3300) / 1023));
            USART_Print("\r\n");
        }
    }
}
🧠 แนวคิด: DMA_analogReadStart circular mode → DMA เขียนค่า ADC ลง buffer เรื่อย ๆ แบบ ring buffer — buffer 100 samples อัปเดตตลอด — main loop: Timer_t 1 วินาที เช็ค → คำนวณค่าเฉลี่ยจาก 100 samples → แสดง avg + latest + mV — CPU ว่าง ไม่ต้อง poll ADC ทีละครั้ง — ideal for data logger

ข้อ 3: USART DMA ส่ง 500 bytes + LED กระพริบพร้อมกัน

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include "SimpleHAL.h"

#define MSG_SIZE 500

volatile uint8_t dma_busy = 0;

void dma_done_callback(DMA_Channel ch) {
    dma_busy = 0;
}

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
    pinMode(PC0, PIN_MODE_OUTPUT);

    // เตรียมข้อความ 500 bytes
    uint8_t msg[MSG_SIZE];
    for (int i = 0; i < MSG_SIZE; i++) msg[i] = 'A' + (i % 26);

    // ต้องเรียก InitTx ก่อนเสมอ — DMA_USART_Transmit แค่ตั้งค่า address/length แล้ว start
    // ไม่ได้ผูก DMA channel เข้ากับ USART_DATAR หรือเปิด USART_DMACmd ให้
    DMA_USART_InitTx(DMA_CH2, msg, MSG_SIZE);

    // ตั้ง DMA callback
    DMA_SetTransferCompleteCallback(DMA_CH2, dma_done_callback);

    while (1) {
        dma_busy = 1;
        DMA_USART_Transmit(DMA_CH2, msg, MSG_SIZE);  // non-blocking send

        // CPU แสดง LED กระพริบ — พิสูจน์ว่าไม่ติดขัด!
        while (dma_busy) {
            digitalWrite(PC0, HIGH); Delay_Ms(50);
            digitalWrite(PC0, LOW);  Delay_Ms(50);
        }

        USART_Print("\r\n--- Done ---\r\n");
        Delay_Ms(2000);
    }
}
🧠 แนวคิด: DMA_USART_Transmit non-blocking → ส่ง 500 bytes โดย DMA ทำงานเบื้องหลัง — dma_busy flag เซ็ตก่อนส่ง, เคลียร์ตอน callback — while(dma_busy): LED toggle ทุก 50ms — พิสูจน์ว่า CPU ไม่หยุดรอ USART — DMA transfer complete → callback → main loop แสดง "Done" — นี่คือข้อได้เปรียบของ DMA: CPU อิสระระหว่างถ่ายโอนข้อมูล
