ex01_LED_Blink — กระพริบ LED 1Hz

📖 แนวคิด

โปรแกรมแรกของ MCU ทุกตัว — ทำให้ LED กระพริบ 1Hz (ติด 500ms / ดับ 500ms) สอนการตั้งค่า pin output และใช้ delay แบบ blocking

🔌 วงจร

    CH32V003                  LED
    PC0 ----/\/\/\---->|---- GND
           220 Ohm
    - ขาแอโนด (+ ยาว) ต่อ R 220Ω → PC0
    - ขาแคโทด (- สั้น) ต่อ GND

🧭 ผังการทำงาน

flowchart TD
    A["SystemCoreClockUpdate()"] --> B["Timer_Init()"]
    B --> C["pinMode(PC0, OUTPUT)"]
    C --> D["while(1)"]
    D --> E["digitalWrite(PC0, HIGH)"]
    E --> F["Delay_Ms(500)"]
    F --> G["digitalWrite(PC0, LOW)"]
    G --> H["Delay_Ms(500)"]
    H --> D

💻 โค้ด + อธิบายทีละบรรทัด

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include <SimpleHAL.h>

int main(void)
{
    SystemCoreClockUpdate();   // ★ ต้องเป็นบรรทัดแรก — อัปเดต SystemCoreClock
    Timer_Init();              // ★ ต้องเรียกก่อน Delay_Ms — เริ่ม SysTick 1ms
    pinMode(PC0, PIN_MODE_OUTPUT);  // ตั้ง PC0 เป็น output push-pull

    while(1)
    {
        digitalWrite(PC0, HIGH);   // LED ติด (3.3V ที่ PC0)
        Delay_Ms(500);             // ค้าง 500ms
        digitalWrite(PC0, LOW);    // LED ดับ (0V ที่ PC0)
        Delay_Ms(500);             // ค้าง 500ms
    }
}
  • SystemCoreClockUpdate(): ดึงค่าความถี่จริงจาก register — USART, PWM, Timer ทุกตัวใช้ค่านี้ ไม่เรียก = ทุกอย่างเพี้ยน
  • Timer_Init(): เปิด SysTick interrupt ทุก 1ms — Delay_Ms, Get_CurrentMs, Timer_t ทุกตัวทำงานไม่ได้ถ้าไม่เรียก
  • pinMode: ทุกครั้งที่ใช้ pin ต้องตั้งค่าก่อน — PIN_MODE_OUTPUT = push-pull output, PIN_MODE_INPUT_PULLUP = input มี pull-up ในตัว

⚠️ จุดควรระวัง

  • ลืม SystemCoreClockUpdate() → Delay_Ms ทำงานผิด (ค่าจริงอาจสั้นกว่าที่ตั้ง)
  • ลืม Timer_Init() → Delay_Ms คืนทันที (delay = 0ms)
  • LED ต้องมี R 220Ω เสมอ — ต่อตรง = LED + MCU pin พัง, pin output กระแสสูงสุด 8mA

✅ ตรวจสอบก่อนรัน

  • LED ต่อถูกขั้ว? แอโนด→R→PC0, แคโทด→GND
  • R 220Ω อยู่ระหว่าง PC0 กับ LED? (ไม่ใช่ระหว่าง LED กับ GND)
  • GND ต่อครบวงจร? MCU GND ต้องต่อร่วมกับ LED GND

ex02_Button_Input — อ่านปุ่มกด + Debounce + USART

📖 แนวคิด

อ่านปุ่มกดด้วย digitalRead — ใช้ internal pull-up (ACTIVE_LOW) — debounce แบบ non-blocking ด้วย ELAPSED_TIME — แสดงผลผ่าน USART

🔌 วงจร

    PC1 -- Button -- GND    (ใช้ PIN_MODE_INPUT_PULLUP ภายใน ~40kΩ)
    PD5(TX) -- USB-to-Serial -- PC  (115200 baud)
    PD6(RX) -- USB-to-Serial -- PC

🧭 ผังการทำงาน

flowchart TD
    A["SystemCoreClockUpdate()"] --> B["Timer_Init()"]
    B --> C["USART_SimpleInit()"]
    C --> D["pinMode(BUTTON, INPUT_PULLUP)"]
    D --> E["while(1)"]
    E --> F["digitalRead(BUTTON)"]
    F --> G{"State changed?"}
    G -->|"Yes"| H["บันทึกเวลา"]
    G -->|"No"| E
    H --> I{"Debounce 50ms?"}
    I -->|"Yes"| J["USART_Print(Press)"]
    I -->|"No"| E
    J --> E

💻 โค้ด

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include <SimpleHAL.h>

#define BUTTON_PIN   PC1
#define DEBOUNCE_MS  50

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
    pinMode(BUTTON_PIN, PIN_MODE_INPUT_PULLUP);

    uint8_t last_state = HIGH;     // ปกติ HIGH (pull-up)
    uint32_t last_debounce = 0;

    USART_Print("Button Test Ready!\r\n");

    while(1) {
        uint8_t reading = digitalRead(BUTTON_PIN);
        if(reading != last_state) {
            last_debounce = Get_CurrentMs();
            last_state = reading;
        }
        if(ELAPSED_TIME(last_debounce, Get_CurrentMs()) >= DEBOUNCE_MS) {
            if(reading == LOW) {
                USART_Print("Button pressed!\r\n");
            }
        }
        IWDG_Feed();
    }
}
  • PIN_MODE_INPUT_PULLUP: เปิด pull-up ~40kΩ ภายใน — ไม่ต้องต่อ R ภายนอก — ปกติอ่านได้ HIGH, กด = LOW
  • ELAPSED_TIME: คำนวณเวลาที่ผ่านไปแบบ overflow-safe ใช้กับ Get_CurrentMs()
  • Debounce non-blocking: บันทึกเวลาที่ state เปลี่ยน → รอ 50ms → อ่านค่าซ้ำ → ถ้ายังเปลี่ยนอยู่ = กดจริง

⚠️ จุดควรระวัง

  • ACTIVE_LOW = กดแล้วอ่าน LOW — ถ้าต้องการ HIGH เมื่อกด ใช้ PIN_MODE_INPUT_PULLDOWN (หรือต่อ R ภายนอก)
  • Debounce จำเป็น! — ปุ่มกลไกมี "bounce" ~10-50ms — อ่านตรง ๆ โดยไม่ debounce ได้ค่า bounce หลายร้อยครั้ง
  • USART baud 115200 — ต้องตรงกับ serial monitor (Arduino IDE / PuTTY / screen)

ex03_External_Interrupt — Interrupt ขอบตก

📖 แนวคิด

ใช้ attachInterrupt ให้ MCU หยุด main loop ชั่วครู่เมื่อมีสัญญาณขอบตก (FALLING) ที่ปุ่ม — นับจำนวน interrupt — toggle LED ใน ISR

🧭 ผังการทำงาน

flowchart TD
    A["SystemCoreClockUpdate()"] --> B["Timer_Init()"]
    B --> C["pinMode(PC0, OUTPUT)"]
    C --> D["pinMode(PC1, INPUT)"]
    D --> E["attachInterrupt(FALLING)"]
    E --> F["ISR: interrupt_count++; digitalToggle(PC0)"]
    F --> G["while(1)"]
    G --> H{"1s elapsed?"}
    H -->|"Yes"| I["Print interrupt_count"]
    H -->|"No"| G
    I --> G

💻 โค้ด

volatile uint32_t interrupt_count = 0;  // ★ volatile — เปลี่ยนใน ISR

void btn_isr(void) {
    interrupt_count++;                   // นับ interrupt
    digitalToggle(PC0);                  // สลับ LED
}

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
    pinMode(PC0, PIN_MODE_OUTPUT);
    pinMode(PC1, PIN_MODE_INPUT_PULLUP);
    attachInterrupt(PC1, btn_isr, FALLING);
    USART_Print("Interrupt Test Ready!\r\n");

    while(1) {
        // แสดงจำนวน interrupt ทุก 1 วิ (non-blocking)
        static uint32_t last_print = 0;
        if(ELAPSED_TIME(last_print, Get_CurrentMs()) >= 1000) {
            last_print = Get_CurrentMs();
            USART_Print("Count: ");
            USART_PrintNum(interrupt_count);
            USART_Print("\r\n");
        }
        IWDG_Feed();
    }
}
  • volatile: บอกคอมไพเลอร์ว่าตัวแปรนี้เปลี่ยนค่าได้โดยไม่รู้ตัว (ISR) — ห้าม optimization ตัดทิ้ง
  • attachInterrupt(pin, cb, mode): RISING/FALLING/CHANGE — callback ถูกเรียกจาก ISR context — จำกัดเวลา

⚠️ จุดควรระวัง

  • ห้ามใช้ Delay_Ms() หรือ USART_Print() ใน ISR! — Delay_Ms รอ SysTick แต่ ISR ขัดจังหวะ priority สูงกว่า → deadlock
  • ตัวแปรที่ใช้ร่วมกัน ISR↔main loop ต้อง volatile เสมอ
  • ISR ควรสั้นที่สุด — ตั้ง flag แล้วให้ main loop จัดการ

ex04_GPIO_Multiple — Running Light 4 LED

📖 แนวคิด

ใช้ pinModeMultiple ตั้งค่า 4 pins พร้อมกัน + digitalWriteMultiple เขียนพร้อมกัน — ทำ running light (ดวงละ 250ms วนไป)

🔌 วงจร

    PC0 ---R---|LED1|---GND
    PC1 ---R---|LED2|---GND
    PC2 ---R---|LED3|---GND
    PC3 ---R---|LED4|---GND

🧭 ผังการทำงาน

flowchart TD
    A["pinModeMultiple(4 pins, OUTPUT)"] --> B["while(1)"]
    B --> C["for i = 0 to 3"]
    C --> D["digitalWriteMultiple(1 << i)"]
    D --> E["Delay_Ms(250)"]
    E --> C

💻 โค้ด

uint8_t pins[] = {PC0, PC1, PC2, PC3};
pinModeMultiple(pins, 4, PIN_MODE_OUTPUT);

while(1) {
    for(uint8_t i = 0; i < 4; i++) {
        digitalWriteMultiple(pins, 4, 1 << i);  // เปิดทีละดวง
        Delay_Ms(250);
    }
}

ex05_Port_Operations — อ่าน/เขียนทั้งพอร์ต

📖 แนวคิด

ใช้ portWrite อ่าน/เขียน 8-bit ทั้งพอร์ตพร้อมกัน — เร็วกว่าเขียนทีละ pin — ใช้อ่าน GPIOD (PD2-PD7 ที่เป็น input)

🧭 ผังการทำงาน

flowchart TD
    A["SystemCoreClockUpdate()"] --> B["Timer_Init()"]
    B --> C["USART_SimpleInit()"]
    C --> D["portRead(GPIOD)"]
    D --> E["USART_PrintHex(portD)"]
    E --> F["portWrite(GPIOC, 0xAA)"]
    F --> G["while(1)"]

💻 โค้ด

// อ่านทั้งพอร์ต D
uint8_t port_d = portRead(GPIOD);
USART_Print("GPIOD: 0x");
USART_PrintHex(port_d, 2);

// เขียนทั้งพอร์ต C
portWrite(GPIOC, 0b10101010);  // PC7=HIGH, PC6=LOW, PC5=HIGH, ...
  • portRead: อ่าน 8-bit จากพอร์ต — รวมทุก pin — เหมาะกับ sensor 8-bit แบบขนาน
  • portWrite: เขียน 8-bit ทั้งพอร์ต — ส่งทีละ byte — เร็วกว่า digitalWrite 8 ครั้ง x3

ex06_Shift_Register — 74HC595 + Knight Rider

📖 แนวคิด

ใช้ shiftOut ส่งข้อมูล 8-bit แบบ serial ไปยัง 74HC595 — ขยาย output จาก 3 pins → 8 outputs — ทำเอฟเฟกต์ LED วิ่งไปวิ่งมา (Knight Rider)

🔌 วงจร

    PC0 -- DATA (DS) -- 74HC595
    PC1 -- CLK  (SH_CP)
    PC2 -- LATCH (ST_CP)
    74HC595 outputs Q0-Q7 -- R 220Ω -- LED 8 ดวง -- GND

🧭 ผังการทำงาน

flowchart TD
    A["pinMode(DATA,CLK,LATCH, OUT)"] --> B["while(1)"]
    B --> C["for i = 0 to 7"]
    C --> D["LATCH LOW"]
    D --> E["shiftOut(, 1 << i)"]
    E --> F["LATCH HIGH"]
    F --> G["Delay_Ms(100)"]
    G --> C

💻 โค้ด

#define DATA_PIN    PC0
#define CLOCK_PIN   PC1
#define LATCH_PIN   PC2

pinMode(DATA_PIN, PIN_MODE_OUTPUT);
pinMode(CLOCK_PIN, PIN_MODE_OUTPUT);
pinMode(LATCH_PIN, PIN_MODE_OUTPUT);

while(1) {
    // Knight Rider effect — LED วิ่งไปมา
    for(uint8_t i = 0; i < 8; i++) {
        digitalWrite(LATCH_PIN, LOW);
        shiftOut(DATA_PIN, CLOCK_PIN, LSBFIRST, 1 << i);
        digitalWrite(LATCH_PIN, HIGH);
        Delay_Ms(100);
    }
    for(uint8_t i = 6; i > 0; i--) {
        digitalWrite(LATCH_PIN, LOW);
        shiftOut(DATA_PIN, CLOCK_PIN, LSBFIRST, 1 << i);
        digitalWrite(LATCH_PIN, HIGH);
        Delay_Ms(100);
    }
}
  • shiftOut: ส่ง 1 byte แบบ MSBFIRST หรือ LSBFIRST — bit ละ 1 CLK pulse — 74HC595 รับ serial → แปลงเป็น parallel
  • LATCH: เมื่อ LATCH HIGH → output เปลี่ยนพร้อมกัน — ตอน LATCH LOW ข้อมูลกำลัง shift ยังไม่แสดงผล

ex07_PulseIn_Measurement — วัดระยะ HC-SR04

📖 แนวคิด

ใช้ pulseIn วัดความกว้าง pulse จาก HC-SR04 ultrasonic sensor — TRIG 10µs → sensor ส่ง 8 pulse 40kHz → ECHO pulse กว้างตามระยะ

🧭 ผังการทำงาน

flowchart TD
    A["pinMode(TRIG, OUT; ECHO, IN)"] --> B["while(1)"]
    B --> C["digitalWrite(TRIG, HIGH)"]
    C --> D["Delay_Us(10)"]
    D --> E["digitalWrite(TRIG, LOW)"]
    E --> F["pulseIn(ECHO, HIGH, 30000)"]
    F --> G{"duration > 0?"}
    G -->|"Yes"| H["distance = duration × 0.0343 / 2"]
    G -->|"No"| I["Print Error"]
    H --> J["USART_Print(distance)"]
    I --> J
    J --> K["Delay_Ms(60)"]
    K --> B

💻 โค้ด

pinMode(TRIG, PIN_MODE_OUTPUT);
pinMode(ECHO, PIN_MODE_INPUT);

while(1) {
    digitalWrite(TRIG, HIGH);     // ส่ง trigger pulse 10µs
    Delay_Us(10);
    digitalWrite(TRIG, LOW);

    uint32_t duration = pulseIn(ECHO, HIGH, 30000);  // จับเวลา pulse (max 30ms)
    if(duration) {
        float distance = (float)duration * 0.0343f / 2.0f;  // sound speed / 2
        USART_Print("Distance: ");
        USART_PrintFloat(distance, 1);
        USART_Print(" cm\r\n");
    }
    Delay_Ms(60);  // ต้องรอ > 60ms ระหว่างวัด
}
  • pulseIn: รอ HIGH → จับเวลาจนถึง LOW — timeout 30ms (ถ้าไม่มี echo) — blocking!
  • Distance = duration × 0.0343 / 2: duration = ไป-กลับ → หาร 2, เสียงเดินทาง 343m/s = 0.0343cm/µs

⚠️ จุดควรระวัง

  • ECHO pin ให้ 5V — CH32V003 I/O ทน 5V ได้ แต่ถ้าไม่แน่ใจ ใช้ voltage divider 2:1
  • pulseIn เป็น blocking — ถ้าต้อง non-blocking ใช้ interrupt + Timer_t
  • วัดถี่เกิน 60ms → echo สะท้อนกัน → ค่าเพี้ยน

ex09_Error_Handling — ตรวจสอบ Pin ก่อนใช้

📖 แนวคิด

ใช้ IS_VALID_PIN / IS_ADC_PIN / IS_PWM_PIN เช็คก่อนใช้ pin — ป้องกัน SOP-8 ใช้ pin ที่ไม่มีอยู่จริง — ใช้ SOP-8 หรือเปลี่ยน PACKAGE เพื่อดูผลต่าง

🧭 ผังการทำงาน

flowchart TD
    A["USART_SimpleInit()"] --> B["for each test pin"]
    B --> C{"IS_VALID_PIN?"}
    C -->|"Yes"| D["pinMode OK"]
    C -->|"No"| E["Print SKIP"]
    D --> F{"IS_ADC_PIN?"}
    F -->|"Yes"| G["analogRead OK"]
    F -->|"No"| H["Print SKIP"]
    G --> I["next pin"]
    E --> I
    H --> I
    I --> B

💻 โค้ด

#define CH32V003_PACKAGE  PACKAGE_SOP8     // เลือกแพ็กเกจเพื่อทดสอบ
#include "SimpleHAL.h"

static uint8_t safe_pinMode(uint8_t pin, GPIO_PinMode mode) {
    if(!IS_VALID_PIN(pin)) {
        USART_Print("  [SKIP] pinMode(");
        USART_PrintNum(pin);
        USART_Print(") — pin not available\r\n");
        return 0;
    }
    pinMode(pin, mode);
    return 1;
}

static uint8_t safe_analogRead(uint8_t pin) {
    if(!IS_ADC_PIN(pin)) {
        USART_Print("  [SKIP] analogRead(…)\r\n");
        return 0;
    }
    uint16_t val = _analogRead_impl(pin);
    USART_Print("  [OK] analogRead = ");
    USART_PrintNum(val);
    USART_Print("\r\n");
    return val;
}

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);

    USART_Print("SOP-8 Pin Validation:\r\n");
    uint8_t test_pins[] = {PA1, PC0, PC1, PC2, PC4, PD6, PD1};
    for(uint8_t i = 0; i < sizeof(test_pins); i++) {
        safe_pinMode(test_pins[i], PIN_MODE_OUTPUT);
    }
    while(1) { IWDG_Feed(); }
}
  • IS_VALID_PIN(pin): compile-time macro — ถ้า pin ไม่มีในแพ็กเกจ → false (แต่ compile ผ่าน — runtime เงียบ)
  • IS_ADC_PIN(pin): เช็ค pin นั้นเป็น ADC input ได้ไหม — SOP-8 มี ADC: PA1, PC4, PD1, PD4, PD6
  • IS_PWM_PIN(pin): เช็ค pin นั้นมี PWM output ไหม — SOP-8 มี PWM: PA1, PA2, PD1, PD4, PD6

⚠️ จุดควรระวัง

  • SOP-8 มีแค่ 6 pins: PA1, PA2, PC1, PC2, PC4, PD6 — PC0, PC3, PC5-PC7, PD0, PD2-PD5, PD7 ใช้ไม่ได้!
  • pinMode/digitalWrite ผิด pin → เงียบไม่ทำงาน ไม่แจ้ง error — ต้องตรวจสอบเอง
  • _analogRead_impl / _analogWrite_impl = runtime validation (ไม่ใช่ compile error เหมือน API ปกติ)
