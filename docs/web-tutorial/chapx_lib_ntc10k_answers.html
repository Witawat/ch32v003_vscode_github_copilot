📝 เฉลย — NTC10K เทอร์มิสเตอร์วัดอุณหภูมิ

ข้อ 1: Temp Calibration ด้วย Reference Thermometer

#define NTC_ADC_CH  ADC_CH_PC4
float cal_offset = 0.0f;

void ntc_calibrate(float ref_temp_c) {
    // วัดอุณหภูมิจาก NTC 10 ครั้งแล้วเฉลี่ย
    float ntc_sum = 0;
    for(int i = 0; i < 10; i++) {
        float temp = NTC_ReadTemperature();
        ntc_sum += temp;
        Delay_Ms(50);
    }
    float ntc_avg = ntc_sum / 10.0f;

    cal_offset = ref_temp_c - ntc_avg;

    USART_Print("Calibration: NTC=");
    USART_PrintFloat(ntc_avg, 2);
    USART_Print(" Ref=");
    USART_PrintFloat(ref_temp_c, 2);
    USART_Print(" Offset=");
    USART_PrintFloat(cal_offset, 2);
    USART_Println(" °C");
}

float ntc_read_calibrated(void) {
    return NTC_ReadTemperature() + cal_offset;
}
🧠 แนวคิด: เปรียบเทียบค่าที่วัดได้กับ reference thermometer ที่รู้ค่าจริง — หาค่า offset เชิงเส้น — เก็บ cal_offset เป็น global แล้วบวกทุกครั้งที่อ่าน — Steinhart-Hart equation ให้ค่าอุณหภูมิจากความต้านทาน NTC — B-constant 3950 สำหรับ NTC 10K ทั่วไป

ข้อ 2: Voltage Divider Calculation

// วงจร: VCC(3.3V) → R_fixed(10k) → ADC_PIN → NTC → GND
// Vout = VCC * R_ntc / (R_fixed + R_ntc)
// => R_ntc = R_fixed * Vout / (VCC - Vout)

void ntc_voltage_divider_demo(void) {
    USART_Println("ADC | Vout   | R_ntc   | Temp C");
    USART_Println("----|--------|---------|-------");

    for(uint16_t adc = 0; adc <= 1023; adc += 50) {
        float vout = adc * 3.3f / 1023.0f;
        float r_ntc = 10000.0f * vout / (3.3f - vout);
        float temp = NTC_ReadTemperature();

        USART_PrintNum(adc); USART_Print(" | ");
        USART_PrintFloat(vout, 3); USART_Print(" | ");
        USART_PrintFloat(r_ntc, 1); USART_Print(" | ");
        USART_PrintFloat(temp, 1);
        USART_Println("");
    }
}
🧠 แนวคิด: Voltage divider: ADC อ่านแรงดันที่จุดกึ่งกลางระหว่าง R_fixed กับ NTC — สูตร R_ntc = R_fixed × Vout / (VCC − Vout) — กรอง out-of-range: ADC ~0 = NTC short, ADC ~1023 = NTC ขาด — ตาราง ADC→Temp ช่วยดีบักและเลือก R_fixed ที่เหมาะสม

ข้อ 3: เทียบความแม่นยำ NTC10K vs DS18B20

void ntc_vs_ds18b20(void) {
    OLED_Handle oled;
    OLED_Init(&oled, OLED_128x64, 0x3C);

    DS18B20_Device* ds = DS18B20_Init(PC3);

    char buf[64];
    while(1) {
        float ntc_temp = ntc_read_calibrated();
        float ds_temp = DS18B20_ReadTemperatureBlocking(ds);

        float diff = fabsf(ntc_temp - ds_temp);

        snprintf(buf, sizeof(buf), "NTC:  %.1f C ", ntc_temp);
        OLED_DrawStringAlign(&oled, 0, 0, buf, ALIGN_LEFT);

        snprintf(buf, sizeof(buf), "DS18: %.1f C ", ds_temp);
        OLED_DrawStringAlign(&oled, 0, 10, buf, ALIGN_LEFT);

        snprintf(buf, sizeof(buf), "Diff: %.2f C ", diff);
        OLED_DrawStringAlign(&oled, 0, 20, buf, ALIGN_LEFT);

        if(diff < 0.5f)       snprintf(buf, sizeof(buf), "Excellent    ");
        else if(diff < 1.0f)  snprintf(buf, sizeof(buf), "Good         ");
        else if(diff < 2.0f)  snprintf(buf, sizeof(buf), "Acceptable   ");
        else                  snprintf(buf, sizeof(buf), "Recalibrate!  ");
        OLED_DrawStringAlign(&oled, 0, 30, buf, ALIGN_LEFT);

        USART_PrintFloat(ntc_temp, 2);
        USART_Print(",");
        USART_PrintFloat(ds_temp, 2);
        USART_Print(",");
        USART_PrintFloat(diff, 3);
        USART_Println("");  // CSV output

        Delay_Ms(2000);
    }
}
🧠 แนวคิด: DS18B20 ความแม่นยำ ±0.5°C ใช้เป็น reference — NTC ทั่วไป ±1–2°C — แสดงผลต่างบน OLED และ serial CSV ให้ import Excel วิเคราะห์ — NTC ข้อดี: ราคาถูก ตอบสนองเร็ว — DS18B20 ข้อดี: digital ไม่ต้อง calibrate, one-wire bus ต่อหลายตัวได้
