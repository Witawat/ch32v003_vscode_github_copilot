AHT10/AHT20 — Temp+Hum I2C

ระดับกลาง

🧭 แผนภาพการทำงาน

flowchart TD
    A["AHT10_Init(&sensor)"] --> B["AHT10_Read(&sensor, &t, &h)"]
    B --> C["float t (°C), float h (%)"]
    C --> B

🔌 การต่อวงจร

        AHT10 / AHT20 (I2C)
┌────────────────────────────────────┐
│ PC1 ──── SDA                       │
│ PC2 ──── SCL                       │
│ 3.3V ─── VCC                       │
│ GND  ─── GND                       │
│                                    │
│ I2C addr = 0x38 (fixed)            │
│ AHT20 = drop-in replacement        │
└────────────────────────────────────┘
MCU Pin→อุปกรณ์ขาหมายเหตุ
PC1↔SDA-สายข้อมูล I2C — ต้องมี R 4.7kΩ pull-up ไป 3.3V
PC2→SCL-สายนาฬิกา I2C — ต้องมี R 4.7kΩ pull-up ไป 3.3V
3.3V→VCC-ไฟเลี้ยง 3.3V
GND→GND-กราวด์ร่วม

📚 หลักการทำงาน

AHT10 เซนเซอร์วัดอุณหภูมิ+ความชื้นแบบ I2C — address 0x38 คงที่ — AHT20 ใช้แทนได้ทันที — Accuracy ±0.3°C / ±2%RH

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Init I2C + AHT10

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "AHT10.h"

AHT10_Instance sensor;

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    I2C_SimpleInit(I2C_SPEED_100K, I2C_PINS_DEFAULT);

    if(!AHT10_Init(&sensor)) {
        USART_Println("AHT10 Init Failed!");
        while(1);
    }
    USART_Println("AHT10 Ready!");

🔍 AHT10_Init(&sensor) — instance-based, ส่ง calibration command + ตรวจสอบ — address 0x38 คงที่

ขั้นที่ 2: อ่านค่า

    float t, h;
    if(AHT10_Read(&sensor, &t, &h)) {
        USART_Print("T: "); USART_PrintFloat(t, 1);
        USART_Print("C  H: "); USART_PrintFloat(h, 1);
        USART_Println("%");
    } else {
        USART_Println("Read error");
    }

🔍 AHT10_Read(&sensor, &t, &h) — trigger + wait + read ในฟังก์ชันเดียว — คืนค่า TRUE ถ้าสำเร็จ, FALSE ถ้า CRC fail

ขั้นที่ 3: Loop + OLED Display

    OLED_Handle oled;
    OLED_Init(&oled, OLED_128x64, 0x3C);
    OLED_SetFont(&oled, FONT_8x16);

    Timer_t tmr;
    Start_Timer(&tmr, 2000, 1);

    float t, h;

    while(1) {
        if(Is_Timer_Expired(&tmr)) {
            if(AHT10_Read(&sensor, &t, &h)) {
                OLED_Clear(&oled);
                snprintf(buf, sizeof(buf), "Temp: %.1fC", t);
                OLED_DrawStringAlign(&oled, 0, 0, buf, ALIGN_LEFT);
                snprintf(buf, sizeof(buf), "Hum : %.1f %%", h);
                OLED_DrawStringAlign(&oled, 0, 20, buf, ALIGN_LEFT);
                OLED_Update(&oled);
            }
        }

    }
}

🔍 AHT10_Read รวม trigger+delay+read — ใช้เวลา ~80ms — ระบบ Non-Blocking ด้วย Timer

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
Init fail — AHT10 ไม่ตอบสาย I2C หลวม หรือ addr ผิดเช็คสาย SDA/SCL — pull-up 4.7kΩ
ค่าแกว่ง ±2°CSelf-heating จากการวัดต่อเนื่องลด sampling rate เหลือทุก 2-5 วินาที

🛠️ สร้าง Product: Incubator Monitor

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "AHT10.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    I2C_SimpleInit(I2C_SPEED_100K, I2C_PINS_DEFAULT);
    USART_Init(USART1, BAUD_115200);

    AHT10_Instance sensor;
    AHT10_Init(&sensor);
    pinMode(PC0, PIN_MODE_OUTPUT);  // Relay = Heater
    digitalWrite(PC0, LOW);

    Timer_t tmr;
    Start_Timer(&tmr, 2000, 1);

    float t, h;
    while(1) {
        if(Is_Timer_Expired(&tmr)) {
            if(AHT10_Read(&sensor, &t, &h)) {
                if(t < 37.0f) digitalWrite(PC0, HIGH);
                if(t > 37.8f) digitalWrite(PC0, LOW);
                USART_Print("T:"); USART_PrintFloat(t, 1);
                USART_Print(" H:"); USART_PrintFloat(h, 1);
                USART_Println("");
            }
        }
        IWDG_Feed();
    }
}
