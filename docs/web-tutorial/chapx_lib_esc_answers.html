📝 เฉลย — ESC (Electronic Speed Controller)

ข้อ 1: Thrust Test Ramp — ค่อย ๆ เพิ่มกำลังวัดแรงขับ

#include "simple_esc.h"

ESC_Instance esc;

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    ESC_Init(&esc, PWM1_CH1, 1000, 2000);

    // Arm ESC
    ESC_Arm(&esc);

    // Ramp จาก 0% → 100% ทีละ 1%
    for (uint8_t throttle = 0; throttle <= 100; throttle++) {
        ESC_SetThrottle(&esc, throttle);
        Delay_Ms(150);  // ค้างแต่ละระดับ 150ms เพื่อให้แรงขับคงที่

        // ถ้าต้องการ log ค่า ADC จาก load cell ที่นี่
    }

    // ค้างที่ 100% 2 วินาที
    Delay_Ms(2000);

    // Ramp ลงกลับ 0%
    for (int8_t t = 100; t >= 0; t--) {
        ESC_SetThrottle(&esc, (uint8_t)t);
        Delay_Ms(100);
    }

    ESC_SetThrottle(&esc, 0);

    while (1) {
        // idle
    }
}
🧠 แนวคิด: ESC ต้อง arm ก่อนใช้งาน — ส่ง throttle 0% ค้างไว้ 3 วิ — ramp ขึ้นทีละ 1% หน่วง 150ms ต่อ step เพื่อให้ motor มีเวลาตอบสนองต่อการเปลี่ยน RPM — การเพิ่มช้า ๆ ทำให้แรงขับค่อยเป็นค่อยไป ป้องกัน prop strike และ over-current — loop วัดแรงขับได้โดยอ่าน ADC จาก load cell หรือ strain gauge ในช่วง delay — หลังจบ ramp ลงช้ากว่ารอบขึ้นเล็กน้อยเพื่อให้ prop เบรกอย่างปลอดภัย — ระวัง: ต้องยึดมอเตอร์กับแท่นทดสอบให้แน่นก่อนรัน

ข้อ 2: Battery Voltage Monitor ระหว่างรันมอเตอร์

#include "simple_esc.h"
#include "simple_adc.h"

ESC_Instance esc;

#define BAT_LOW_MV   3300   // 3.3V per cell (3S = 9.9V)
#define BAT_CRIT_MV  3000   // 3.0V per cell — cut off
#define VOLT_DIVIDER 4.0f   // 1:4 voltage divider ratio

uint16_t read_battery_mv(void) {
    uint16_t adc = ADC_Read(ADC_CH0);
    float v_adc = (float)adc * 3.3f / 4095.0f;
    float v_bat = v_adc * VOLT_DIVIDER;
    return (uint16_t)(v_bat * 1000.0f);
}

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    ADC_SimpleInit(ADC_CH0);
    ESC_Init(&esc, PWM1_CH1, 1000, 2000);

    ESC_SetThrottle(&esc, 0);
    Delay_Ms(3000);

    for (uint8_t t = 10; t <= 100; t += 10) {
        ESC_SetThrottle(&esc, t);
        Delay_Ms(200);

        uint16_t bat_mv = read_battery_mv();
        if (bat_mv < BAT_CRIT_MV) {
            ESC_SetThrottle(&esc, 0);
            // Blink LED for warning
            while (1) {}
        }
    }

    ESC_SetThrottle(&esc, 0);
    while (1) {}
}
🧠 แนวคิด: ใช้ voltage divider (เช่น 10k+3.3k) ลดแรงดันแบต LiPo ให้อยู่ในช่วง 0–3.3V ที่ ADC อ่านได้ — read_battery_mv อ่าน ADC แล้วคูณกลับด้วยอัตราส่วน divider — เมื่อแรงดันต่ำกว่า critical (3.0V/cell) ให้ตัด throttle ทันทีเพื่อป้องกันแบตเสียหาย — ทดสอบทีละ 10% throttle เพื่อดูกำลังโหลดเทียบกับแรงดันตก — แบตคุณภาพดีแรงดันจะตกน้อยเมื่อโหลดสูง — ควรเพิ่ม hysteresis ป้องกันการตัดต่อซ้ำไปมา (cut at 3.0V, restore at 3.3V) — ใช้ averaging หลาย sampling เพื่อลด noise

ข้อ 3: Automated Thrust Curve Logger — ส่งข้อมูลความเร็วผ่าน UART

#include "simple_esc.h"
#include "simple_adc.h"
#include "simple_uart.h"

ESC_Instance esc;

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    ADC_SimpleInit(ADC_CH0);
    UART_Init(USART1, 115200);
    ESC_Init(&esc, PWM1_CH1, 1000, 2000);

    ESC_SetThrottle(&esc, 0);
    Delay_Ms(3000);
    UART_Printf(USART1, "Throttle(%%),ADC_Raw\r\n");

    for (uint8_t t = 5; t <= 100; t += 5) {
        ESC_SetThrottle(&esc, t);
        Delay_Ms(500);

        uint32_t sum = 0;
        for (uint8_t s = 0; s < 32; s++) {
            sum += ADC_Read(ADC_CH0);
            Delay_Ms(2);
        }
        uint16_t avg = sum / 32;

        UART_Printf(USART1, "%d,%d\r\n", t, avg);
    }

    ESC_SetThrottle(&esc, 0);
    UART_Printf(USART1, "DONE\r\n");
    while (1) {}
}
🧠 แนวคิด: ส่งข้อมูลผ่าน UART แบบ CSV ให้เปิด Serial Plotter ใน IDE หรือ import เข้า Excel — ทดสอบทีละ 5% throttle เพื่อความละเอียดของกราฟ — หน่วง 500ms ต่อระดับให้แรงขับ stabilise — อ่าน ADC 32 ครั้งแล้วหาค่าเฉลี่ยเพื่อลด noise จาก vibration — ข้อมูลที่ได้: แกน X = throttle%, แกน Y = ADC (หรือแรงขับ) — ถ้ามี RPM sensor เพิ่มคอลัมน์ RPM ด้วย — ต่อกับแบตที่ชาร์จเต็มก่อนเริ่มทดสอบเพื่อผลลัพธ์ที่แม่นยำ — ห้ามใช้ ESC แบบไม่มี load นานเกิน 30 วินาที มิฉะนั้นมอเตอร์จะร้อนเกิน
