📝 เฉลย — SoftUART

ข้อ 1: SoftUART เชื่อม ESP01 สำหรับดีบัก — ใช้ 2 UART พร้อมกัน

// HW UART1 (PD5=TX, PD6=RX) → PC terminal
// SoftUART (PD3=TX, PD4=RX) → ESP01

SoftUART_Instance suart;
char esp_line[128];
uint8_t esp_idx = 0;

void softuart_esp01_debug(void) {
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
    Timer_Init();

    SoftUART_Status st = SoftUART_Init(&suart, PD3, PD4, 9600);
    //  ESP01 default firmware @ 9600 — TX ของ ESP ต่อ RX (PD4) ของ MCU
    if (st != SOFTUART_OK) { USART_Println("SoftUART init fail"); return; }

    USART_Println("=== Dual UART Debug Mode ===");
    USART_Println("HW UART1 → PC Terminal | SoftUART → ESP01 @9600");

    while (1) {
        // ส่ง AT command ไป ESP01 ผ่าน SoftUART
        SoftUART_WriteString(&suart, "AT\r\n");
        Delay_Ms(500);

        // อ่าน response แบบ blocking-with-timeout (เพราะ Available() ใช้ไม่ได้)
        esp_idx = 0;
        uint32_t deadline = Timer_GetMs() + 2000;  // รวมทั้ง response 2s
        while (Timer_GetMs() < deadline && esp_idx < sizeof(esp_line) - 1) {
            uint8_t c;
            if (SoftUART_ReadByte(&suart, &c, 50) == SOFTUART_OK) {
                if (c == '\n') {                  // จบบรรทัด → parse ทันที
                    esp_line[esp_idx] = '\0';
                    USART_Print("[ESP] ");
                    USART_Println(esp_line);
                    if (strstr(esp_line, "OK"))      USART_Println("  ← ESP01 OK");
                    if (strstr(esp_line, "ERROR"))   USART_Println("  ← ESP01 ERROR");
                    esp_idx = 0;
                } else if (c != '\r') {
                    esp_line[esp_idx++] = (char)c;
                }
            }
        }

        Delay_Ms(3000);   // รอ 3s ก่อนส่ง AT ใหม่
    }
}
🧠 แนวคิด: CH32V003 มี HW UART 1 ช่อง — SoftUART สร้างพอร์ตอนุกรมเพิ่มบน GPIO ธรรมดาด้วย bit-bang — HW UART คุยกับ PC ที่ 115200 bps, SoftUART คุยกับ ESP01 ที่ 9600 bps (default firmware) — ใช้ SoftUART_ReadByte(&suart, &c, 50) แบบ blocking-with-timeout เพราะ Available() เป็น known issue — สะสม byte จนเจอ '\n' แล้ว parse ทันที — มี deadline รวม 2 วินาทีกันค้าง — เหมาะสำหรับดีบัก WiFi module โดยไม่ต้องใช้ USB-to-Serial แยก

ข้อ 2: Bridge HW UART ↔ SoftUART

SoftUART_Instance suart;
// เชื่อมต่อ PC ↔ ESP01 ผ่าน CH32V003
// HW UART (PD5,PD6) ↔ PC @ 115200
// SoftUART (PD3,PD4) ↔ ESP01 @ 9600 (ต้องตั้ง ESP01 firmware ให้เท่ากันถ้าจะ bridge จริง)

void uart_bridge(void) {
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
    Timer_Init();
    SoftUART_Init(&suart, PD3, PD4, 9600);

    USART_Println("[Bridge] HW↔Soft UART Bridge active");

    while (1) {
        // === HW → Soft: PC ส่งไป ESP01 ===
        uint8_t c;
        while (USART_ReadByte(&c) == 0) {     // HW UART มี Available จริง
            SoftUART_WriteByte(&suart, c);    // ส่งต่อไป ESP01
            USART_WriteByte(c);               // echo กลับ PC ให้เห็นว่าพิมพ์อะไร
            if (c == '\r') SoftUART_WriteByte(&suart, '\n');  // terminal CR → CRLF
        }

        // === Soft → HW: ESP01 ตอบกลับ PC (ใช้ ReadByte timeout สั้นๆ) ===
        uint8_t s;
        if (SoftUART_ReadByte(&suart, &s, 5) == SOFTUART_OK) {
            USART_WriteByte(s);               // ส่งอักษรออก PC
        }

        Delay_Ms(1);   // ยึด CPU
    }
}
🧠 แนวคิด: Bridge ทำหน้าที่ส่งต่อ byte แบบ transparent — HW UART รับจาก PC → ส่งออก SoftUART — ฝั่ง SoftUART ใช้ ReadByte(&suart, &s, 5) timeout 5ms ต่อ byte แทน Available() เพราะ ring buffer เป็น known issue — แปลง CR เป็น CRLF ให้ terminal — โหมดนี้ต้องการ baudrate เท่ากันทั้งสองฝั่ง (จริง ๆ bridge สำหรับ ESP01 ที่ 9600 → PC terminal ที่ 115200 ต้องมี buffer ถ้า traffic หนาแน่น — ที่นี่ bridge พอร์ตเดียวกันเพื่อเรียบง่าย)

ข้อ 3: Packet Protocol บน SoftUART พร้อม CRC-8

SoftUART_Instance suart;
#define PKT_START   0xAA
#define PKT_END     0x55
#define PKT_MAX_PAYLOAD 56

typedef struct {
    uint8_t start;
    uint8_t len;
    uint8_t cmd;
    uint8_t data[PKT_MAX_PAYLOAD];
    uint8_t crc;
    uint8_t end;
} soft_pkt_t;

uint8_t crc8(const uint8_t *data, uint8_t len) {
    uint8_t crc = 0xFF;
    while (len--) {
        crc ^= *data++;
        for (uint8_t i = 0; i < 8; i++)
            crc = (crc & 0x80) ? (crc << 1) ^ 0x07 : crc << 1;
    }
    return crc;
}

void pkt_send(uint8_t cmd, const uint8_t *data, uint8_t len) {
    if (len > PKT_MAX_PAYLOAD) return;

    uint8_t buf[2 + PKT_MAX_PAYLOAD];
    buf[0] = len;
    buf[1] = cmd;
    if (len) memcpy(&buf[2], data, len);

    SoftUART_WriteByte(&suart, PKT_START);
    SoftUART_WriteByte(&suart, len);
    SoftUART_WriteByte(&suart, cmd);
    SoftUART_Write(&suart, data, len);          // ส่ง payload ทีเดียว
    SoftUART_WriteByte(&suart, crc8(buf, 2 + len));
    SoftUART_WriteByte(&suart, PKT_END);
}

// รับ packet แบบ state machine — ใช้ ReadByte blocking-with-timeout
int8_t pkt_recv(soft_pkt_t *pkt, uint32_t timeout_ms) {
    uint8_t state = 0, idx = 0, crc_in = 0;
    uint8_t crc_buf[2 + PKT_MAX_PAYLOAD];
    uint32_t deadline = Timer_GetMs() + timeout_ms;

    while (Timer_GetMs() < deadline) {
        uint8_t b;
        if (SoftUART_ReadByte(&suart, &b, 50) != SOFTUART_OK) continue;

        switch (state) {
            case 0:  // start
                if (b == PKT_START) { state = 1; idx = 0; }
                break;
            case 1:  // len
                pkt->len = b; crc_buf[idx++] = b; state = 2;
                if (b == 0) state = 4;
                break;
            case 2:  // cmd
                pkt->cmd = b; crc_buf[idx++] = b; state = 3;
                break;
            case 3:  // data
                pkt->data[idx - 2] = b; crc_buf[idx++] = b;
                if (idx - 2 >= pkt->len) state = 4;
                break;
            case 4:  // crc
                crc_in = b; state = 5;
                break;
            case 5:  // end
                if (b == PKT_END) {
                    uint8_t calc = crc8(crc_buf, idx);
                    return (calc == crc_in) ? 1 : -1;   // OK / CRC mismatch
                }
                return -1;                               // end marker ผิด
        }
    }
    return 0;  // timeout — ยังรับไม่สมบูรณ์
}
🧠 แนวคิด: โปรโตคอล packet แบบ start byte (0xAA) + len + cmd + payload + CRC-8-CCITT + end byte (0x55) — ใช้ state machine เก็บทีละ byte ด้วย SoftUART_ReadByte(&suart, &b, 50) แบบ blocking-with-timeout 50ms/byte (เพราะ Available() เป็น known issue) — มี deadline รวมทั้ง packet เพื่อกันค้าง — คืนค่า 1 = OK, -1 = CRC/end ผิด, 0 = timeout — เหมาะกับ SoftUART ที่อาจมี noise มากกว่า HW UART — ส่ง payload ทีเดียวด้วย SoftUART_Write(&suart, data, len) เพื่อลด overhead
