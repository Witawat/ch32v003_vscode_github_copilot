📝 เฉลย — Buzzer

ข้อ 1: Melody Player — เล่นทำนอง Happy Birthday

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "Buzzer.h"

#define BUZZER_PIN PC4

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    Buzzer_Init(BUZZER_PIN, BUZZER_ACTIVE_HIGH);

    uint16_t notes[] = {
        262,262,294,262,349,330,    // Happy birthday to you
        262,262,294,262,392,349,    // Happy birthday to you
        262,262,523,440,349,330,294, // Happy birthday dear X
        466,466,440,349,392,349      // Happy birthday to you
    };

    uint16_t dur[] = {
        300,150,400,400,400,800,
        300,150,400,400,400,800,
        300,150,400,400,400,400,400,
        300,150,400,400,400,800
    };

    uint8_t count = sizeof(notes) / sizeof(notes[0]);

    while(1) {
        for(uint8_t i = 0; i < count; i++) {
            Buzzer_Tone(notes[i], dur[i]);
            Delay_Ms(50);  // pause ระหว่างโน้ต
        }
        Delay_Ms(2000);  // pause ก่อนเล่นซ้ำ
    }
}
🧠 แนวคิด: ใช้ Passive Buzzer — array notes[] เก็บความถี่ Hz (C4=262, D4=294, E4=330, F4=349, G4=392, A4=440, C5=523) — array dur[] เก็บความยาวโน้ตใน ms — Buzzer_Tone(freq, ms) สร้าง PWM ที่ความถี่นั้นตามเวลาที่กำหนด — Delay_Ms(50) เว้นจังหวะระหว่างโน้ต — ต้องต่อ Passive Buzzer (ไม่มี oscillator ในตัว)

ข้อ 2: PWM Frequency Sweep — กวาดความถี่ 100-4000Hz

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "Buzzer.h"

#define BUZZER_PIN PC4

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_Init(115200);
    Buzzer_Init(BUZZER_PIN, BUZZER_ACTIVE_HIGH);

    USART_Println("Frequency Sweep: 100Hz → 4000Hz");

    while(1) {
        // กวาดความถี่ต่ำ → สูง
        for(uint16_t freq = 100; freq <= 4000; freq += 50) {
            Buzzer_Tone(freq, 30);
            Delay_Ms(10);
            USART_Print("Freq: "); USART_PrintNum(freq);
            USART_Println(" Hz");
        }

        Delay_Ms(500);

        // กวาดความถี่สูง → ต่ำ
        for(uint16_t freq = 4000; freq >= 100; freq -= 50) {
            Buzzer_Tone(freq, 30);
            Delay_Ms(10);
        }

        Delay_Ms(1000);

        // เฉพาะช่วงเสียงแหลม (siren effect)
        for(uint16_t freq = 2000; freq <= 4000; freq += 100) {
            Buzzer_Tone(freq, 50);
        }
        for(uint16_t freq = 4000; freq >= 2000; freq -= 100) {
            Buzzer_Tone(freq, 50);
        }
    }
}
🧠 แนวคิด: for loop เปลี่ยนความถี่ทีละ 50Hz จาก 100→4000 — Buzzer_Tone(freq, 30) เล่นแต่ละความถี่ 30ms — ใช้ Passive Buzzer เพราะ Active มีความถี่คงที่ — sweep loop แรก: เสียงต่ำไปสูง (เหมือน siren ตำรวจ) — sweep loop ที่สอง: กลับทิศ — sweep loop ที่สาม: เฉพาะช่วง 2-4kHz (เสียงแหลม) ให้ผลเหมือน siren ฉุกเฉิน — ดู response frequency ของ Passive buzzer ว่าดังสุดที่ความถี่ไหน

ข้อ 3: Dual Buzzer Alarm — Active + Passive สลับแพทเทิร์น

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "Buzzer.h"

#define ACTIVE_PIN  PC0
#define PASSIVE_PIN PC4

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();

    Buzzer_Init(ACTIVE_PIN, BUZZER_ACTIVE_HIGH);
    Buzzer_Init(PASSIVE_PIN, BUZZER_ACTIVE_HIGH);

    while(1) {
        // ===== ALARM LEVEL 1: Active สั้น =====
        for(uint8_t i = 0; i < 3; i++) {
            Buzzer_Beep(150);
            Delay_Ms(150);
        }
        Delay_Ms(500);

        // ===== ALARM LEVEL 2: Active+Passive สลับ =====
        for(uint8_t i = 0; i < 4; i++) {
            Buzzer_Beep(100);
            Delay_Ms(50);
            Buzzer_Tone(800, 100);
            Delay_Ms(50);
        }
        Delay_Ms(800);

        // ===== ALARM LEVEL 3: Passive ต่อเนื่อง 2kHz =====
        Buzzer_Tone(2000, 1000);
        Delay_Ms(200);

        // ===== ALARM LEVEL 4: ทั้งคู่ดังพร้อม + ความถี่เปลี่ยน =====
        for(uint8_t i = 0; i < 5; i++) {
            digitalWrite(ACTIVE_PIN, HIGH);
            Buzzer_Tone(1000 + i*300, 150);
            digitalWrite(ACTIVE_PIN, LOW);
            Delay_Ms(150);
        }

        Delay_Ms(3000);  // pause ระหว่างรอบ
    }
}
🧠 แนวคิด: ต่อ Buzzer 2 ตัว: Active (PC0) + Passive (PC4) — Level 1: Active beep สั้นๆ 3 ครั้ง (แจ้งเตือนเบา) — Level 2: Active+Passive สลับกัน (เสียง 2 โทน — เตือนกลาง) — Level 3: Passive ต่อเนื่อง 1 วิ (เตือนหนัก) — Level 4: ทั้งคู่พร้อม + ความถี่ไล่ขึ้น (ฉุกเฉินสูงสุด) — ใช้ทั้ง Buzzer_Beep (Active) และ Buzzer_Tone (Passive) — เลือกแพทเทิร์นตาม severity
