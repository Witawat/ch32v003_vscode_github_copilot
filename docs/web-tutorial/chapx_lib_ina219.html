INA219 — วัดแรงดัน/กระแส/กำลังไฟฟ้า ผ่าน I2C

ระดับสูง

🧭 แผนภาพการทำงาน

flowchart TD
    A["INA219_Init(&ina, addr, shunt, max_amps)"] --> B["ตั้งค่า Calibration Register"]
    B --> C["เริ่มวัดอัตโนมัติ (Continuous Mode)"]
    C --> D["อ่าน Bus Voltage (Shunt+Bus)"]
    D --> E["อ่าน Shunt Voltage (mV)"]
    E --> F["คำนวณกระแส: I = Vshunt / Rshunt"]
    F --> G["คำนวณกำลัง: P = Vbus × I"]
    G --> H["แสดงผล / บันทึกข้อมูล"]
    H --> C

🔌 การต่อวงจร

        INA219 High-Side Current Sense
   ┌──────────────────────────────────────┐
   │  LOAD (+) ──Rshunt(0.1Ω)── VCC(3.3V) │
   │  Vin+ ──────┤ ├── Vin-               │
   │                  │                   │
   │  MCU                INA219           │
   │  PC1─SDA────SDA    VCC─3.3V          │
   │  PC2─SCL────SCL    GND─GND           │
   │                  A0─GND (addr 0x40)  │
   │                  A1─GND              │
   └──────────────────────────────────────┘

ต่อ Shunt Resistor 0.1Ω ระหว่าง Load(+) กับ VCC — วัด High-Side (ด้านบวกของโหลด) รองรับแรงดัน 0-32V, กระแสสูงสุด ±3.2A (ที่ shunt 0.1Ω)

MCU Pin→อุปกรณ์ขาหมายเหตุ
PC1↔SDA-สายข้อมูล I2C — ต้องมี R 4.7kΩ pull-up ไป 3.3V
PC2→SCL-สายนาฬิกา I2C — ต้องมี R 4.7kΩ pull-up ไป 3.3V
Vin+←Shunt (Load side)-ด้านโหลดของ shunt resistor 0.1Ω — วัด high-side current
Vin-←Shunt (VCC side)-ด้าน VCC ของ shunt resistor 0.1Ω
3.3V→VCC-ไฟเลี้ยง 3.3V
GND→GND-กราวด์ร่วม
GND→A0-ต่อ GND = ตั้งค่า I2C address เป็น 0x40
GND→A1-ต่อ GND = ตั้งค่า I2C address เป็น 0x40

📚 หลักการทำงาน

INA219 เป็น Bi-directional Current/Power Monitor แบบ I2C วัดแรงดันตกคร่อม Shunt Resistor แล้วคำนวณกระแสด้วยกฎของโอห์ม I = Vshunt / Rshunt — มี ADC 12-bit ภายใน วัดได้ทั้ง Bus Voltage (0-32V) และ Shunt Voltage (±320mV full-scale)

Calibration Register ต้องตั้งค่าให้ถูกต้องตาม Rshunt และ Max Current ที่ใช้ — Lib จะคำนวณ calibration value ให้อัตโนมัติจากค่าที่กำหนด (shunt Ω และ max_amps)

4 Addresses เลือกผ่านขา A0/A1 — ต่อ GND หรือ VCC ได้ 4 ค่า: 0x40, 0x41, 0x44, 0x45 — ต่อหลายตัวบน I2C bus เดียวกันได้

การวัดกำลัง ใช้ Bus Voltage × Current — INA219 คำนวณ power register ให้ในตัว แต่ Lib อ่าน Bus Voltage จริง (ไม่รวม shunt drop) แล้วคูณกระแสให้ถูกต้อง

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Init INA219 + ตั้งค่า

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "INA219.h"

INA219_Instance ina;

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    I2C_Init(PC1, PC2, I2C_SPEED_STANDARD);
    USART_Init(115200);

    INA219_Init(&ina, 0x40, 0.1f, 3.2f);
    //              inst addr shunt max_amps
    USART_Print("INA219 Ready\r\n");

🔍 INA219_Init(&ina, addr, shunt, max_amps) — addr 0x40 (A0=GND,A1=GND), shunt 0.1Ω, max_amps 3.2A — Lib คำนวณ Calibration ให้โดยอัตโนมัติ

ขั้นที่ 2: อ่านค่าแรงดัน กระแส กำลัง

    float busV = INA219_GetBusVoltage(&ina);
    float shuntV = INA219_GetShuntVoltage(&ina);
    float current = INA219_GetCurrent(&ina);
    float power = INA219_GetPower(&ina);

    USART_Print("Bus:");  USART_PrintFloat(busV, 3);
    USART_Print("V Shunt:"); USART_PrintFloat(shuntV*1000, 2);
    USART_Print("mV I:"); USART_PrintFloat(current, 3);
    USART_Print("A P:"); USART_PrintFloat(power, 3);
    USART_Print("W\r\n");

🔍 GetBusVoltage() คืนค่าแรงดันที่ขั้วโหลด — GetShuntVoltage() คืนค่าแรงดันตกคร่อม shunt (V) — GetCurrent() คืนค่ากระแสที่คำนวณจาก Vshunt/Rshunt — GetPower() คำนวณจาก BusV × Current

ขั้นที่ 3: Loop วัดแบตเตอรี่ + คำนวณ mAh

    float mah_used = 0;
    Timer_t tmr_sample;
    Start_Timer(&tmr_sample, 1000, 1); // sample ทุก 1 วิ

    while(1) {
        if(Is_Timer_Expired(&tmr_sample)) {
            float v = INA219_GetBusVoltage(&ina);
            float a = INA219_GetCurrent(&ina);
            mah_used += (a * 1000.0f) / 3600.0f; // mAh

            USART_Print("V:"); USART_PrintFloat(v, 3);
            USART_Print(" A:"); USART_PrintFloat(a*1000, 1);
            USART_Print("mA mAh:"); USART_PrintFloat(mah_used, 1);
            USART_Print("\r\n");
            if(v < 2.8f) USART_Print("BATTERY LOW!\r\n");
        }
    }
}

🔍 สุ่มตัวอย่างทุก 1 วินาที — สะสม mAh จากกระแส × เวลา — mAh = (I(A) × 1000) / 3600 ต่อวินาที — ตรวจจับแบตเตอรี่ต่ำเมื่อแรงดันต่ำกว่า 2.8V

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
อ่านค่าได้ 0 ตลอดShunt ต่อผิดขั้ว หรือ Vin+ Vin- สลับตรวจสอบ Vin+ ต่อด้าน VCC, Vin- ต่อด้าน Load
ค่ากระแสผิดเพี้ยนRshunt ไม่ตรงกับค่าที่กำหนดใน Initวัดค่า shunt จริงด้วยมัลติมิเตอร์ — ป้อนค่าที่ถูกต้องใน Init
Bus Voltage อ่านได้ 0ใช้ High-Side แต่ต่อผิดด้าน GNDHigh-Side ต้องต่อ shunt ด้านบวก ไม่ใช่ด้าน GND
I2C ไม่ตอบสนองAddress ผิด (A0/A1 ต่อผิด)ตรวจสอบ A0/A1 — 0x40(A0=GND,A1=GND), 0x41(VCC,GND), 0x44(GND,VCC), 0x45(VCC,VCC)
กระแสอ่านได้ติดลบโหลดต่อกลับขั้ว หรือ current directionกลับขั้ว Vin+ Vin- หรือใช้ abs() กับค่ากระแส

🛠️ สร้าง Product: Battery Fuel Gauge

INA219 + OLED แสดงแรงดัน กระแส mAh ที่ใช้ — หน้าจออัพเดททุก 1 วิ

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "INA219.h"
#include "oled_i2c.h"

int main(void)
{
    SystemCoreClockUpdate();
    Timer_Init();
    I2C_Init(PC1, PC2, I2C_SPEED_STANDARD);
    USART_Init(115200);

    INA219_Instance ina;
    INA219_Init(&ina, 0x40, 0.1f, 3.2f);

    OLED_Handle oled;
    OLED_Init(&oled, OLED_128x64, 0x3C);
    OLED_SetFont(&oled, FONT_8x16);

    float mah_used = 0;
    Timer_t tmr;
    Start_Timer(&tmr, 1000, 1);

    char buf[64];
    while(1) {
        if(Is_Timer_Expired(&tmr)) {
            float v = INA219_GetBusVoltage(&ina);
            float a = INA219_GetCurrent(&ina);
            float w = INA219_GetPower(&ina);
            mah_used += (a * 1000.0f) / 3600.0f;

            OLED_Clear(&oled);
            snprintf(buf, sizeof(buf), "V: %.2fV", v);
            OLED_DrawStringAlign(&oled, 0, 0, buf, ALIGN_LEFT);

            snprintf(buf, sizeof(buf), "I: %.1fmA", a * 1000);
            OLED_DrawStringAlign(&oled, 0, 20, buf, ALIGN_LEFT);

            snprintf(buf, sizeof(buf), "P: %.1fmW", w * 1000);
            OLED_DrawStringAlign(&oled, 0, 40, buf, ALIGN_LEFT);

            snprintf(buf, sizeof(buf), "Used: %.1fmAh", mah_used);
            OLED_DrawStringAlign(&oled, 0, 60, buf, ALIGN_LEFT);

            if(v < 2.8f) {
                snprintf(buf, sizeof(buf), "LOW!");
                OLED_DrawStringAlign(&oled, 70, 20, buf, ALIGN_LEFT);
            }

            OLED_Update(&oled);
        }
    }
}

✏️ ลองทำดู

  1. ใช้ INA219 หลายตัว (2 address) วัดกระแส 2 ช่องพร้อมกัน — ใช้ USART พิมพ์เปรียบเทียบ
  2. เพิ่ม EEPROM บันทึกค่า mAh สะสม — เมื่อ reboot อ่านค่ากลับมาใช้ต่อ
  3. เพิ่ม Relay — ตัดโหลดอัตโนมัติเมื่อแรงดันต่ำกว่า threshold ที่กำหนด (Low Voltage Cutoff)
  4. วัด Solar Panel — คำนวณ Wh และ efficiency เทียบกับเวลาที่แสงตกกระทบ
