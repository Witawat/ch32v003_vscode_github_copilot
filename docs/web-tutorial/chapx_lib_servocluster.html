รู้จักขา

  • SDA (PC1): สายข้อมูล I2C ควบคุม PCA9685
  • SCL (PC2): สัญญาณนาฬิกา I2C
  • VCC (3.3V): จ่ายไฟเลี้ยงวงจรลอจิกของ PCA9685
  • GND: กราวด์ร่วมระหว่าง MCU, PCA9685 และแหล่งจ่าย 5V
  • V+ (5V External): จ่ายไฟให้ servo ผ่าน PCA9685 ต้องใช้แหล่งจ่าย 5V แยก
  • CH0–CH15: ขาสัญญาณ PWM ไปยัง servo แต่ละตัว

วิธีต่อทีละขั้น

  1. ต่อ PC1 ไป SDA ของ PCA9685 — สายข้อมูล I2C
  2. ต่อ PC2 ไป SCL ของ PCA9685 — สายนาฬิกา I2C
  3. ต่อ 3.3V ไป VCC ของ PCA9685 — จ่ายไฟลอจิก
  4. ต่อ GND ไป GND ของ PCA9685 — กราวด์ร่วม
  5. ต่อ แหล่งจ่าย 5V 3A ไป V+ ของ PCA9685 — จ่ายไฟให้ servo
  6. ต่อ CH0 ไป Signal ของ Servo 1, CH1 ไป Servo 2 ไปเรื่อยๆ

⚠️ PCA9685 ขับ servo ได้สูงสุด 16 ตัวผ่าน I²C — ต่อ V+ เข้ากับ External 5V (ต้องจ่ายกระแสได้ ≥3A สำหรับ 16 servo) — ใช้ Address 0x40 (default)

📚 หลักการทำงาน

PCA9685 คือ I²C 16-channel 12-bit PWM driver — สร้าง PWM 50Hz ได้ 16 ช่องพร้อมกันจาก hardware ตัวเดียว ทำให้ CH32V003 ควบคุม servo จำนวนมากได้โดยไม่เปลือง GPIO

ServoCluster ห่อหุ้ม PCA9685 + เพิ่มระบบ Easing Curves 10 แบบ: Linear, QuadEaseIn, QuadEaseOut, QuadEaseInOut, Cubic, Elastic, Bounce, Back, Sine, Expo — ทำให้การเคลื่อนไหวดูเป็นธรรมชาติ

Non-Blocking Update(): ServoCluster_MoveTo(ch, target_angle, duration_ms, curve) ตั้งเป้าหมายแล้ว ServoCluster_Update() ใน main loop จะคำนวณตำแหน่งระหว่างทางอัตโนมัติ — ไม่ต้องใช้ Delay

Callback: เมื่อ servo ถึงเป้าหมาย จะเรียกฟังก์ชัน callback — ใช้ทำ Animation Sequence หรือ Pick-and-Place

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Init PCA9685 + เพิ่ม Servo

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "ServoCluster.h"

ServoCluster_Instance sc;

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    I2C_Init(PC2, PC1);  // SCL, SDA

    ServoCluster_Init(&sc, SERVOCLUSTER_PCA9685, 16);
    ServoCluster_AddServo(&sc, 0, 500, 2500);   // CH0: 0.5-2.5ms
    ServoCluster_AddServo(&sc, 1, 600, 2400);   // CH1: custom range
    ServoCluster_AddServo(&sc, 2, 500, 2500);   // CH2
    ServoCluster_AddServo(&sc, 3, 500, 2500);   // CH3

🔍 ServoCluster_Init(cluster, i2c_addr) ตั้งค่า PCA9685 50Hz — ServoCluster_AddServo(cluster, channel, us_min, us_max) กำหนด pulse width range ของ servo แต่ละตัว (calibrate ได้)

ขั้นที่ 2: MoveTo พร้อม Easing

    // ย้าย servo 4 ตัวไปคนละมุมด้วย easing ต่างกัน
    ServoCluster_MoveTo(&sc, 0, 180, 1000, EASE_IN_OUT_QUAD);
    ServoCluster_MoveTo(&sc, 1, 90,  1500, EASE_ELASTIC_OUT);
    ServoCluster_MoveTo(&sc, 2, 45,  800,  EASE_BOUNCE_OUT);
    ServoCluster_MoveTo(&sc, 3, 135, 1200, EASE_LINEAR);

    // ใช้ callback เมื่อ CH0 ถึงเป้าหมาย
    sc.servos[0].onComplete = on_servo0_done;

🔍 ServoCluster_MoveTo(cluster, ch, angle, duration_ms, curve) — กำหนดมุมเป้าหมาย, เวลาที่ใช้เดินทาง, และ easing curve — แต่ละ servo เคลื่อนไหวอิสระพร้อมกัน

ขั้นที่ 3: Non-Blocking Update Loop

    while(1) {
        ServoCluster_Update(&sc);  // Non-Blocking — คำนวณทุก servo

        // ตรวจสอบว่า servo ทั้งหมดถึงเป้าหมายหรือยัง
        if(ServoCluster_IsAllDone(&sc)) {
            Delay_Ms(500);
            ServoCluster_MoveTo(&sc, 0, 0, 1000, EASE_IN_OUT_CUBIC);
            ServoCluster_MoveTo(&sc, 1, 0, 1000, EASE_IN_OUT_CUBIC);
            ServoCluster_MoveTo(&sc, 2, 180, 1000, EASE_IN_OUT_CUBIC);
            ServoCluster_MoveTo(&sc, 3, 180, 1000, EASE_IN_OUT_CUBIC);
        }
    }
}

void on_servo0_done(void) {
    // Trigger sequence ถัดไปเมื่อ servo 0 ถึงเป้าหมาย
}

🔍 ServoCluster_Update() ต้องเรียกใน main loop ตลอด — มันคำนวณตำแหน่ง interpolated จาก elapsed time และ easing curve — ServoCluster_IsAllDone() เช็คว่าทุก servo ถึง target หมด

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
Servo ไม่ขยับเลยลืมเรียก ServoCluster_Update() ใน loopต้องเรียก ServoCluster_Update(&sc) ใน main while(1) ทุกรอบ
PCA9685 ไม่ตอบสนองI²C address ไม่ตรง หรือไม่ได้ต่อ pull-upตรวจสอบ address (0x40 default) — ถ้าใช้หลายบอร์ดต้องตั้ง address ให้ไม่ชนกัน
Servo สั่นระหว่างเคลื่อนที่External 5V จ่ายกระแสไม่พอสำหรับหลาย servoใช้ 5V supply ≥ 3A สำหรับ 16 servo — ต่อ capacitor 1000μF ขนาน V+
Easing ไม่ smoothเรียก ServoCluster_Update() ช้าเกินไปloop ต้องวน ≥ 50Hz — ห้ามมี Delay_Ms นาน ๆ ใน loop
Timer overflow ในการคำนวณduration นานเกิน 65 วินาที (uint16_ms)แบ่งการเคลื่อนที่เป็นช่วง ๆ — หรือใช้ uint32_t สำหรับ long moves

🛠️ สร้าง Product: Robot Arm 3-DOF Pick & Place

แขนกล 3 แกน — Base (CH0), Shoulder (CH1), Elbow (CH2) + Gripper (CH3) — ทำ Pick and Place sequence อัตโนมัติ

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "ServoCluster.h"

ServoCluster_Instance arm;
uint8_t step = 0;

void next_step(void) {
    step = (step + 1) % 6;
    switch(step) {
        case 0: // Home
            ServoCluster_MoveTo(&arm,0,90,1000,EASE_IN_OUT_QUAD);
            ServoCluster_MoveTo(&arm,1,45,1000,EASE_IN_OUT_QUAD);
            ServoCluster_MoveTo(&arm,2,90,1000,EASE_IN_OUT_QUAD);
            ServoCluster_MoveTo(&arm,3,90,500,EASE_LINEAR); break;
        case 1: // Move to pick
            ServoCluster_MoveTo(&arm,0,30,1200,EASE_IN_OUT_QUAD);
            ServoCluster_MoveTo(&arm,1,120,1200,EASE_IN_OUT_QUAD);
            ServoCluster_MoveTo(&arm,2,150,1200,EASE_IN_OUT_QUAD); break;
        case 2: // Grab
            ServoCluster_MoveTo(&arm,3,0,500,EASE_IN_BACK); break;
        case 3: // Lift
            ServoCluster_MoveTo(&arm,1,60,800,EASE_OUT_QUAD);
            ServoCluster_MoveTo(&arm,2,90,800,EASE_OUT_QUAD); break;
        case 4: // Move to place
            ServoCluster_MoveTo(&arm,0,150,1500,EASE_IN_OUT_QUAD);
            ServoCluster_MoveTo(&arm,1,130,1500,EASE_IN_OUT_QUAD);
            ServoCluster_MoveTo(&arm,2,160,1500,EASE_IN_OUT_QUAD); break;
        case 5: // Release
            ServoCluster_MoveTo(&arm,3,90,300,EASE_LINEAR); break;
    }
}

int main(void) {
    SystemCoreClockUpdate(); Timer_Init();
    I2C_Init(PC2, PC1);
    ServoCluster_Init(&arm, SERVOCLUSTER_PCA9685, 16);
    ServoCluster_AddServo(&arm, 0, 500, 2500); // Base
    ServoCluster_AddServo(&arm, 1, 600, 2400); // Shoulder
    ServoCluster_AddServo(&arm, 2, 500, 2500); // Elbow
    ServoCluster_AddServo(&arm, 3, 700, 2300); // Gripper
    arm.servos[3].onComplete = next_step; // Auto-advance
    next_step(); // Start sequence

    while(1) {
        ServoCluster_Update(&arm);
    }
}

✏️ ลองทำดู

  1. เพิ่ม easing curves ใหม่ — เช่น Overshoot, Anticipate, Custom bezier — แล้วนำไปใช้กับแขนกล
  2. ทำระบบบันทึกและเล่นซ้ำ — ใช้ Flash เก็บบันทึกตำแหน่ง servo แล้วเล่น animation sequence
  3. ควบคุมแขนกลด้วย Joystick + Easing — ขยับ joystick แล้ว servo เคลื่อนที่ตามด้วย easing แบบ real-time
