YL-69 — Soil Moisture Sensor วัดความชื้นดินผ่าน ADC

ระดับกลาง

🧭 แผนภาพการทำงาน

flowchart TD
    A["GPIO VCC Control → HIGH (เปิดไฟเลี้ยง)"] --> B["ADC_Read → 0-1023"]
    B --> C["Map ADC → % ความชื้น"]
    C --> D{"ความชื้น < threshold?"}
    D -->|"Yes (ดินแห้ง)"| E["เปิดปั๊มน้ำ / แจ้งเตือน"]

📚 หลักการทำงาน

YL-69 (HL-69) เป็น Soil Moisture Sensor ใช้หลักการวัดความต้านทานของดิน — Probe 2 ขั้วเสียบลงดิน — เมื่อดินเปียก ความต้านทานต่ำ → ADC สูง — เมื่อดินแห้ง ความต้านทานสูง → ADC ต่ำ — Module มักมาพร้อม LM393 Comparator สำหรับ Digital Output (threshold ปรับด้วย pot)

Corrosion Warning: Probe เป็นโลหะ — ถ้าเปิดไฟเลี้ยงตลอดเวลา จะเกิด electrolysis ทำให้ probe กัดกร่อนเร็ว — ต้องใช้ GPIO เปิดไฟเลี้ยงเฉพาะตอนวัด (~100ms) แล้วปิดทันที — เพิ่มอายุการใช้งาน probe ได้หลายเท่า

ทำ Mapping % ความชื้น: calibrate โดยการจุ่ม probe ในน้ำ (100%) และแห้งสนิท (0%) — สมการ: Moisture% = 100 - ((ADC - ADC_min) / (ADC_max - ADC_min)) × 100 — ค่า ADC_min กับ ADC_max ขึ้นกับดินแต่ละชนิด

การใช้งาน: รดน้ำต้นไม้อัตโนมัติ, ตรวจจับน้ำรั่ว, วัดระดับน้ำ (เปลี่ยน probe เป็นแบบกันน้ำ), Smart farming

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Init + VCC Control ด้วย GPIO

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"

#define SOIL_VCC PC0
#define SOIL_ADC PD2
#define SOIL_CH ADC_CH0

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    ADC_Init(SOIL_ADC, SOIL_CH);
    pinMode(SOIL_VCC, PIN_MODE_OUTPUT);
    digitalWrite(SOIL_VCC, LOW); // ปิดตอนเริ่ม
    USART_Init(115200);

    USART_Print("Soil Moisture Ready\r\n");

🔍 SOIL_VCC เป็น GPIO ควบคุมไฟเลี้ยงเซนเซอร์ — เริ่มต้นปิดไว้ (LOW) — เปิดเฉพาะตอนวัด — ลดการกัดกร่อนของ probe

ขั้นที่ 2: อ่านค่า + คำนวณ % ความชื้น

    // ค่า calibrate (วัดเองตามดินที่ใช้)
    uint16_t adc_dry = 200;   // ADC ตอนดินแห้งสนิท
    uint16_t adc_wet = 950;   // ADC ตอนดินเปียกมาก

    digitalWrite(SOIL_VCC, HIGH); // เปิดไฟเลี้ยง
    Delay_Ms(50);                  // รอ stabilize

    uint16_t adc = ADC_Read(SOIL_ADC);

    digitalWrite(SOIL_VCC, LOW);  // ปิดทันที!

    if(adc < adc_dry) adc = adc_dry;
    if(adc > adc_wet) adc = adc_wet;

    float moisture = 100.0f - ((float)(adc - adc_dry)
                     / (float)(adc_wet - adc_dry)) * 100.0f;

    USART_Print("Moisture: ");
    USART_PrintFloat(moisture, 1);
    USART_Print("%\r\n");

🔍 เปิด VCC → รอ 50ms stabilize → อ่าน ADC → ปิด VCC ทันที! — Map ADC เป็น % — adc_dry และ adc_wet ต้อง calibrate ตามดินที่ใช้ — ค่าจะกลับกับความชื้น (ดินแห้ง=ADC ต่ำ, ดินเปียก=ADC สูง) — บางแผงค่าอาจกลับกัน ต้องทดสอบ

ขั้นที่ 3: รดน้ำอัตโนมัติ + วัดเป็นระยะ

    pinMode(PC1, PIN_MODE_OUTPUT); // ปั๊มน้ำ Relay
    Timer_t tmr;
    Start_Timer(&tmr, 10000, 1); // วัดทุก 10 วิ

    while(1) {
        if(Is_Timer_Expired(&tmr)) {
            digitalWrite(SOIL_VCC, HIGH);
            Delay_Ms(50);
            uint16_t adc = ADC_Read(SOIL_ADC);
            digitalWrite(SOIL_VCC, LOW);

            float moisture = 100.0f - ((float)(adc - 200)
                             / (float)(950 - 200)) * 100.0f;

            USART_Print("Moisture: ");
            USART_PrintFloat(moisture, 1);
            USART_Print("% ");

            if(moisture < 30.0f) {
                USART_Print("→ WATERING!");
                digitalWrite(PC1, HIGH); // เปิดปั๊ม
                Delay_Ms(3000);          // รด 3 วิ
                digitalWrite(PC1, LOW);  // ปิดปั๊ม
            }
            USART_Print("\r\n");
        }
    }
}

🔍 ใช้ Non-Blocking Timer วัดทุก 10 วินาที — ลดการเปิดไฟเลี้ยงถี่เกิน — เมื่อความชื้น < 30% → เปิดปั๊ม 3 วิ — ปิด VCC ทันทีหลังอ่านเสร็จ

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
Probe กัดกร่อนเร็วมากเปิดไฟเลี้ยงค้างตลอดเวลาใช้ GPIO เปิดเฉพาะตอนวัด — ปิดทันทีหลังอ่าน — ใช้ transistor ถ้ากระแสสูง
ค่า ADC กระโดดไปมาNoise จาก switching / หน้าสัมผัสไม่ดีใช้ averaging 5-10 ครั้ง — ต่อ 10µF cap ที่ VCC-GND
ค่า % ไม่ตรงไม่ได้ calibrate ตามดินที่ใช้วัด adc_dry (กลางอากาศ) และ adc_wet (จุ่มน้ำ) — บันทึกไว้
วัดแล้ว ADC เป็น 0VCC เปิดไม่ทัน หรือ GPIO จ่ายกระแสไม่พอเพิ่ม Delay 50-100ms หลังเปิด VCC — ถ้ากระแสเกิน 8mA ใช้ transistor
ดินเปียกแต่อ่านค่าแห้งต่อสายสลับ หรือ module มี buffer กลับค่าบาง module ใช้ LM393 กลับ logic — ทดสอบแล้วกลับค่า ADC

🛠️ สร้าง Product: ระบบรดน้ำอัตโนมัติ

YL-69 + Relay + OLED — วัดความชื้น 3 ต้น — รดน้ำเมื่อต่ำกว่า threshold — แสดงสถานะบน OLED

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "OLED.h"

#define PUMP_PIN PC3
#define SOIL_VCC PC0
#define SOIL_ADC PD2
#define SOIL_CH ADC_CH0
#define DRY_THRESHOLD 30.0f

int main(void)
{
    SystemCoreClockUpdate();
    Timer_Init();
    I2C_Init(PC1, PC2, I2C_SPEED_STANDARD);
    ADC_Init(SOIL_ADC, SOIL_CH);
    USART_Init(115200);

    pinMode(SOIL_VCC, PIN_MODE_OUTPUT);
    pinMode(PUMP_PIN, PIN_MODE_OUTPUT);
    digitalWrite(SOIL_VCC, LOW);
    digitalWrite(PUMP_PIN, LOW);

    OLED_t oled;
    OLED_Init(&oled, 0x3C, OLED_128x64);
    OLED_SetFont(&oled, FONT_8x16);

    Timer_t tmr;
    Start_Timer(&tmr, 30000, 1); // วัดทุก 30 วิ

    while(1)
    {
        if(Is_Timer_Expired(&tmr))
        {
            digitalWrite(SOIL_VCC, HIGH);
            Delay_Ms(50);
            uint16_t adc = ADC_Read(SOIL_ADC);
            digitalWrite(SOIL_VCC, LOW);

            float moisture = 100.0f - ((float)(adc - 200)
                             / (float)(950 - 200)) * 100.0f;
            if(moisture < 0) moisture = 0;
            if(moisture > 100) moisture = 100;

            OLED_Clear(&oled);
            OLED_SetCursor(&oled, 0, 0);
            OLED_Print(&oled, "Soil Moisture");

            OLED_SetCursor(&oled, 0, 2);
            OLED_PrintFloat(&oled, "", moisture, 1);
            OLED_Print(&oled, " %");

            if(moisture < DRY_THRESHOLD) {
                digitalWrite(PUMP_PIN, HIGH);
                OLED_SetCursor(&oled, 0, 4);
                OLED_Print(&oled, "PUMP ON!");
                Delay_Ms(3000);
                digitalWrite(PUMP_PIN, LOW);
            } else {
                OLED_SetCursor(&oled, 0, 4);
                OLED_Print(&oled, "Soil OK");
            }

            OLED_SetCursor(&oled, 0, 6);
            OLED_Print(&oled, "ADC:");
            OLED_PrintDec(&oled, adc);

            OLED_Display(&oled);
        }
    }
}

✏️ ลองทำดู

  1. วัดความชื้นหลายจุด — ใช้ analog mux (CD4051) หรือ ADC หลายช่อง — รดน้ำเฉพาะจุดที่แห้ง
  2. เพิ่ม RTC — รดน้ำตามเวลาที่กำหนด — เช้า-เย็น — ตรวจสอบความชื้นก่อนรด
  3. Calibrate อัตโนมัติ — กดปุ่มเพื่อบันทึกค่าแห้ง/เปียก — เก็บใน EEPROM
  4. เพิ่ม Water Level Sensor — ตรวจจับน้ำในถัง — หยุดปั๊มถ้าน้ำหมด
