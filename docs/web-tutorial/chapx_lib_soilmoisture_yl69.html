📚 หลักการทำงาน

YL-69 (HL-69) เป็น Soil Moisture Sensor ใช้หลักการวัดความต้านทานของดิน — Probe 2 ขั้วเสียบลงดิน — เมื่อดินเปียก ความต้านทานต่ำ → ADC สูง — เมื่อดินแห้ง ความต้านทานสูง → ADC ต่ำ

Corrosion Warning: Probe เป็นโลหะ — ถ้าเปิดไฟเลี้ยงตลอดเวลา จะเกิด electrolysis ทำให้ probe กัดกร่อนเร็ว — Library จัดการ VCC เปิดเฉพาะตอนวัด (~100ms) แล้วปิดทันที — เพิ่มอายุ probe ได้หลายเท่า

การใช้งาน: รดน้ำต้นไม้อัตโนมัติ, ตรวจจับน้ำรั่ว, วัดระดับน้ำ, Smart farming

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Init SoilMoisture

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "SoilMoisture.h"

SoilMoisture_Instance s;

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_Init(115200);

    SoilMoisture_Init(&s, ADC_CH_PC4);
    USART_Println("Soil Moisture Ready!");

🔍 SoilMoisture_Init(&s, ADC_CH_PC4) — ตั้งค่า ADC + VCC control — library จัดการเปิด/ปิด VCC อัตโนมัติเพื่อป้องกัน corrosion

ขั้นที่ 2: อ่าน % ความชื้น

    float pct = SoilMoisture_Read(&s);

    USART_Print("Moisture: ");
    USART_PrintFloat(pct, 1);
    USART_Println("%");

    if(pct < 30.0f) {
        USART_Println("DRY — Need water!");
    } else if(pct > 80.0f) {
        USART_Println("WET — Soil is saturated");
    } else {
        USART_Println("OK — Moisture is good");
    }

🔍 SoilMoisture_Read(&s) — อ่านความชื้นเป็น % (0-100) — จัดการ VCC เปิด/ปิด อัตโนมัติ — < 30% = แห้งต้องรดน้ำ, > 80% = แฉะเกิน

ขั้นที่ 3: รดน้ำอัตโนมัติ + วัดเป็นระยะ

    pinMode(PC1, PIN_MODE_OUTPUT); // ปั๊มน้ำ Relay
    Timer_t tmr;
    Start_Timer(&tmr, 10000, 1); // วัดทุก 10 วิ

    while(1) {
        if(Is_Timer_Expired(&tmr)) {
            float pct = SoilMoisture_Read(&s);

            USART_Print("Moisture: ");
            USART_PrintFloat(pct, 1);
            USART_Print("% ");

            if(pct < 30.0f) {
                USART_Print("→ WATERING!");
                digitalWrite(PC1, HIGH); // เปิดปั๊ม
                Delay_Ms(3000);          // รด 3 วิ
                digitalWrite(PC1, LOW);  // ปิดปั๊ม
            }
            USART_Println("");
        }
    }
}

🔍 ใช้ Non-Blocking Timer วัดทุก 10 วินาที — เมื่อความชื้น < 30% → เปิดปั๊ม 3 วิ — VCC จัดการอัตโนมัติด้วย library

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
Probe กัดกร่อนเร็วมากเปิดไฟเลี้ยงค้างตลอดเวลาใช้ library จัดการ VCC อัตโนมัติ — เปิดเฉพาะตอนวัด
ค่า ADC กระโดดไปมาNoise จาก switching / หน้าสัมผัสไม่ดีใช้ averaging 5-10 ครั้ง — ต่อ 10µF cap ที่ VCC-GND
ค่า % ไม่ตรงไม่ได้ calibrate ตามดินที่ใช้SoilMoisture_Read คืนค่าปกติ 0-100 — ปรับ calibration ใน library config
ดินเปียกแต่อ่านค่าแห้งต่อสายสลับ หรือ module มี buffer กลับค่าบาง module ใช้ LM393 กลับ logic — ทดสอบแล้วกลับค่าใน code

🛠️ สร้าง Product: ระบบรดน้ำอัตโนมัติ

YL-69 + Relay + OLED — วัดความชื้น — รดน้ำเมื่อต่ำกว่า threshold

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "SoilMoisture.h"
#include "oled_i2c.h"

#define PUMP_PIN PC3
#define DRY_THRESHOLD 30.0f

int main(void)
{
    SystemCoreClockUpdate();
    Timer_Init();
    I2C_Init(PC1, PC2, I2C_SPEED_STANDARD);
    USART_Init(115200);

    SoilMoisture_Instance s;
    SoilMoisture_Init(&s, ADC_CH_PC4);

    pinMode(PUMP_PIN, PIN_MODE_OUTPUT);
    digitalWrite(PUMP_PIN, LOW);

    OLED_Handle oled;
    OLED_Init(&oled, OLED_128x64, 0x3C);
    OLED_SetFont(&oled, FONT_8x16);

    Timer_t tmr;
    Start_Timer(&tmr, 30000, 1); // วัดทุก 30 วิ

    while(1)
    {
        if(Is_Timer_Expired(&tmr))
        {
            float pct = SoilMoisture_Read(&s);
            if(pct < 0) pct = 0;
            if(pct > 100) pct = 100;

            OLED_Clear(&oled);
            OLED_SetCursor(&oled, 0, 0);
            OLED_Print(&oled, "Soil Moisture");

            OLED_SetCursor(&oled, 0, 2);
            OLED_PrintFloat(&oled, "", pct, 1);
            OLED_Print(&oled, " %");

            if(pct < DRY_THRESHOLD) {
                digitalWrite(PUMP_PIN, HIGH);
                OLED_SetCursor(&oled, 0, 4);
                OLED_Print(&oled, "PUMP ON!");
                Delay_Ms(3000);
                digitalWrite(PUMP_PIN, LOW);
            } else {
                OLED_SetCursor(&oled, 0, 4);
                OLED_Print(&oled, "Soil OK");
            }

            OLED_Update(&oled);
        }
    }
}

✏️ ลองทำดู

  1. วัดความชื้นหลายจุด — ใช้ SoilMoisture หลาย instance — รดน้ำเฉพาะจุดที่แห้ง
  2. เพิ่ม RTC — รดน้ำตามเวลาที่กำหนด — เช้า-เย็น — ตรวจสอบความชื้นก่อนรด
  3. Calibrate อัตโนมัติ — กดปุ่มเพื่อบันทึกค่าแห้ง/เปียก — เก็บใน EEPROM
  4. เพิ่ม Water Level Sensor — ตรวจจับน้ำในถัง — หยุดปั๊มถ้าน้ำหมด
