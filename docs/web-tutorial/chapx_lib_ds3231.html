🔌 การต่อวงจร

                DS3231 RTC Module I2C
    ┌──────────────────────────────────────────────┐
    │ CH32V003                         DS3231      │
    │ PC1(SDA)────────SDA   ┌─────────────────────┤
    │ PC2(SCL)────────SCL   │                     │
    │ 3.3V───────────VCC    │ Address = 0x68      │
    │ GND────────────GND    │ Accuracy: ±2ppm     │
    │                        │ +--BAT--CR2032---┐ │
    └──────────────────────────────────────────────┘
MCU Pin→อุปกรณ์ขาหมายเหตุ
PC1↔SDA-สายข้อมูล I2C — ต้องมี R 4.7kΩ pull-up ไป 3.3V
PC2→SCL-สายนาฬิกา I2C — ต้องมี R 4.7kΩ pull-up ไป 3.3V
3.3V→VCC-ไฟเลี้ยง 3.3V
GND→GND-กราวด์ร่วม
CR2032→BAT-แบตเตอรี่สำรอง 3V — รักษาเวลาเมื่อไม่มีไฟหลัก

รู้จักขา

  • VCC: ไฟเลี้ยง 3.3V
  • GND: กราวด์
  • SDA (PC1): สายข้อมูล I2C
  • SCL (PC2): สายนาฬิกา I2C
  • BAT: ขาใส่แบต CR2032 สำรองเวลา 3V
  • 32K/SQW: ขาเอาต์พุตคล็อก 32kHz หรือสัญญาณแจ้งเตือน (optional)
  • pull-up 4.7kΩ: ต่อระหว่าง SDA→3.3V และ SCL→3.3V (โมดูลส่วนมากมีในตัว)

วิธีต่อทีละขั้น

  1. ต่อ VCC ของ DS3231 ไป 3.3V
  2. ต่อ GND ของ DS3231 ไป GND
  3. ต่อ SDA ของ DS3231 ไป PC1 (ขา 13) ของ CH32V003
  4. ต่อ SCL ของ DS3231 ไป PC2 (ขา 14) ของ CH32V003
  5. ใส่ CR2032 ที่ขา BAT เพื่อสำรองเวลาเมื่อไม่มีไฟหลัก
  6. (optional) ต่อขา 32K หรือ SQW ไปขา MCU ถ้าต้องการสัญญาณนาฬิกาหรือ interrupt
  7. ตรวจสอบว่าโมดูลมี R pull-up 4.7kΩ ในตัวหรือไม่ — ถ้าไม่มี ต่อ R เพิ่มจาก SDA→3.3V และ SCL→3.3V

📚 หลักการทำงาน

DS3231 เป็น RTC (Real-Time Clock) แบบ I2C — แม่นยำสูง TCXO ในตัว, เก็บวินาที-นาที-ชั่วโมง-วัน-เดือน-ปี, จัดการ leap year อัตโนมัติ, มี temperature sensor ในตัว

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Init + ตั้งเวลาเริ่มต้น

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "DS3231.h"

DS3231_Instance rtc;

int main(void) {
    SystemCoreClockUpdate();
    I2C_Init(I2C1, I2C_SPEED_STANDARD);
    USART_Init(USART1, BAUD_115200);

    DS3231_Init(&rtc);  // addr 0x68 fixed
    USART_Println("DS3231 RTC Ready!");

🔍 DS3231_Init(&rtc) — instance-based, I2C address 0x68 fixed

ขั้นที่ 2: อ่านเวลา + ตั้ง Alarm

    DS3231_DateTime dt;
    DS3231_GetDateTime(&rtc, &dt);

    USART_Print("Time: ");
    USART_PrintInt(dt.hour); USART_Print(":");
    USART_PrintInt(dt.minute); USART_Print(":");
    USART_PrintInt(dt.second); USART_Println("");

    USART_Print("Date: ");
    USART_PrintInt(dt.day); USART_Print("/");
    USART_PrintInt(dt.month); USART_Print("/");
    USART_PrintInt(dt.year); USART_Println("");

    // ตั้ง Alarm 1: ทุกวัน 08:00:00
    DS3231_SetAlarm1(&rtc, DS3231_A1_MATCH_HR_MIN_SEC, 0, 8, 0, 0);

🔍 DS3231_GetDateTime(&rtc, &dt) — อ่านวัน/เวลาเป็น struct, DS3231_SetAlarm1() — ตั้ง alarm

ขั้นที่ 3: Loop — แสดงผล + ตรวจสอบ Alarm

    while(1) {
        DS3231_GetDateTime(&rtc, &dt);
        USART_PrintInt(dt.hour); USART_Print(":");
        USART_PrintInt(dt.minute); USART_Print(":");
        USART_PrintInt(dt.second); USART_Print(" | ");
        USART_PrintInt(dt.day); USART_Print("/");
        USART_PrintInt(dt.month); USART_Print("/");
        USART_PrintInt(dt.year); USART_Println("");

        if(DS3231_IsAlarm1Fired(&rtc)) {
            USART_Println("ALARM 1 FIRED!");
            digitalToggle(PC0);
            DS3231_ClearAlarmFlag(&rtc, 1);
        }

        float temp;
        DS3231_GetTemperature(&rtc, &temp);
        USART_Print("Temp: "); USART_PrintFloat(temp, 1);
        USART_Println(" C");

        Delay_Ms(1000);
    }
}

🔍 DS3231_GetTemperature(&rtc, &temp) — อ่านอุณหภูมิภายในชิป, DS3231_IsAlarm1Fired(&rtc) — เช็ค flag alarm

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
เวลารีเซ็ตทุกครั้งที่เปิดไฟไม่มี battery backup หรือ battery หมดต่อ CR2032 ที่ขา BAT

🛠️ สร้าง Product: Digital Alarm Clock

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "DS3231.h"

int main(void)
{
    SystemCoreClockUpdate();
    I2C_Init(I2C1, I2C_SPEED_STANDARD);
    USART_Init(USART1, BAUD_115200);
    pinMode(PC0, PIN_MODE_OUTPUT);

    DS3231_Instance rtc;
    DS3231_Init(&rtc);

    DS3231_DateTime dt = {2025, 1, 1, 12, 0, 0, DS3231_WEDNESDAY, 0};
    DS3231_SetDateTime(&rtc, dt.year, dt.month, dt.day, dt.hour, dt.minute, dt.second, dt.day_of_week);
    DS3231_SetAlarm1(&rtc, DS3231_A1_MATCH_HR_MIN_SEC, 0, 8, 0, 0);

    Timer_t tmr;
    Start_Timer(&tmr, 1000, 1);

    while(1)
    {
        if(Is_Timer_Expired(&tmr)) {
            DS3231_GetDateTime(&rtc, &dt);
            USART_PrintInt(dt.hour); USART_Print(":");
            USART_PrintInt(dt.minute); USART_Print(":");
            USART_PrintInt(dt.second); USART_Print(" | ");
            float temp;
            DS3231_GetTemperature(&rtc, &temp);
            USART_PrintFloat(temp, 1); USART_Println(" C");
        }

        if(DS3231_IsAlarm1Fired(&rtc)) {
            digitalWrite(PC0, HIGH);
            USART_Println("*** ALARM ***");
            Delay_Ms(10000);
            digitalWrite(PC0, LOW);
            DS3231_ClearAlarmFlag(&rtc, 1);
        }
    }
}
