LCD 16×2 ตัวอักษรผ่าน I2C (ลด pin usage) — 4 ปุ่มสำหรับนำทางเมนู — ทุกปุ่มใช้ internal pull-up (ACTIVE_LOW)

📚 หลักการทำงาน

LCDMenu: ไลบรารีเมนูแบบลำดับชั้น (hierarchical) สำหรับ LCD1602 16×2 — ใช้ร่วมกับ LCD1602_I2C และ Button — ออกแบบมาสำหรับ MCU หน่วยความจำจำกัดด้วย static memory pool

Menu Tree: สร้างเมนูเป็นลำดับชั้น — เพิ่ม item ด้วย LCDMenu_AddItem() — แต่ละ item มีชื่อ, ประเภท, และ callback — เปลี่ยนหน้าโดย UP/DOWN, เข้าด้วย ENTER, กลับด้วย BACK

4 Callback Types:

  • SUBMENU — เข้าไปยังเมนูย่อย — เหมือน folder — ไม่มี callback แทนที่ด้วยการเปลี่ยนหน้า
  • EDIT — แก้ไขค่าตัวเลข (int/float) — UP/DOWN เปลี่ยนค่า, ENTER ยืนยัน — callback จะได้รับค่าใหม่
  • TOGGLE — เปิด/ปิด — ENTER สลับสถานะ — callback ได้รับ 0 หรือ 1
  • ACTION — เรียก callback ทันทีที่กด ENTER — เหมือนปุ่ม execute
  • BACK — กลับไปเมนูก่อนหน้า — ไม่ต้องมี callback

Static Memory Pool: ทุกอย่างจัดสรรตอน compile — ไม่มี malloc — ปลอดภัยสำหรับ MCU RAM 2KB — กำหนดจำนวน item สูงสุดใน header

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Init LCD + Menu + ปุ่ม

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "LCDMenu.h"
#include "LCD1602_I2C.h"

LCDMenu_Handle menu;
uint8_t setpoint = 40;
float pid_kp = 2.0f;
uint8_t wifi_enabled = 0;

void on_setpoint(void) { USART_Print("Setpoint: "); USART_PrintNum(setpoint); }
void on_toggle_wifi(LCDMenu_Handle *m, uint8_t val) {
    wifi_enabled = val; USART_Println(val ? "WiFi ON" : "WiFi OFF");
}

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_Init(115200);

    LCD1602_I2C_Init(0x27);
    LCDMenu_Init(&menu, 0x27, 16, 2);  // LCD addr, 16 chars, 2 rows
    pinMode(PC3, PIN_MODE_INPUT_PULLUP);  // UP
    pinMode(PC4, PIN_MODE_INPUT_PULLUP);  // DOWN
    pinMode(PC5, PIN_MODE_INPUT_PULLUP);  // ENTER
    pinMode(PC6, PIN_MODE_INPUT_PULLUP);  // BACK

🔍 LCDMenu_Init(&menu, addr, cols, rows) — ต้อง Init LCD1602_I2C ก่อน — ใช้ addr เดียวกัน — ปุ่มทั้ง 4 ตัวต้องตั้งเป็น INPUT_PULLUP เพื่อใช้ internal pull-up

ขั้นที่ 2: สร้าง Menu Tree + Callbacks

    LCDMenu_AddItem(&menu, "Temperature",   LCDMENU_SUBMENU, NULL);
    LCDMenu_AddItem(&menu, "  Set Max",     LCDMENU_EDIT,    on_setpoint);
    LCDMenu_AddItem(&menu, "  Set Min",     LCDMENU_EDIT,    NULL);
    LCDMenu_AddItem(&menu, "  Back",        LCDMENU_BACK,    NULL);
    LCDMenu_AddItem(&menu, "WiFi Control",  LCDMENU_TOGGLE,  on_toggle_wifi);
    LCDMenu_AddItem(&menu, "PID Tuning",    LCDMENU_SUBMENU, NULL);
    LCDMenu_AddItem(&menu, "  Kp=",         LCDMENU_EDIT,    NULL);
    LCDMenu_AddItem(&menu, "  Ki=",         LCDMENU_EDIT,    NULL);
    LCDMenu_AddItem(&menu, "  Back",        LCDMENU_BACK,    NULL);
    LCDMenu_AddItem(&menu, "Save & Exit",   LCDMENU_ACTION,  on_save);
    USART_Println("Menu Ready!");

🔍 LCDMenu_AddItem(&menu, name, type, callback) — ชื่อที่มีเว้นวรรคนำหน้า (" ") คือ item ใน submenu — ต้องมี BACK ทุก submenu — callback เป็น optional สำหรับ SUBMENU/BACK

ขั้นที่ 3: Loop + ปุ่ม Navigate

    while(1) {
        uint8_t up    = !digitalRead(PC3);  // Active LOW → invert
        uint8_t down  = !digitalRead(PC4);
        uint8_t enter = !digitalRead(PC5);
        uint8_t back  = !digitalRead(PC6);

        LCDMenu_Update(&menu, up, down, enter, back);
        Delay_Ms(50);  // 20Hz refresh — ลด CPU load
        IWDG_Feed();
    }
}

🔍 LCDMenu_Update(&menu, up, down, enter, back) — ส่งสถานะปุ่ม 4 ตัวเป็น 0/1 — Active LOW ต้อง invert ก่อนส่ง — refresh 50ms พอดีสำหรับการตอบสนองของมนุษย์ — อย่าลืม IWDG_Feed เพื่อกัน MCU รีเซ็ต

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
จอไม่แสดงอะไรเลยลืม LCD1602_I2C_Init() ก่อน LCDMenu_Init()ต้อง Init LCD ก่อน — LCD1602_I2C_Init(0x27) แล้วตามด้วย LCDMenu_Init()
กด UP/DOWN แล้วกระโดดหลายช่องRefresh rate เร็วเกินไป + ไม่มี debounceเพิ่ม Delay_Ms เป็น 80-100ms — หรือใช้ Button library สำหรับ debounce
SUBMENU ไม่ยอมเข้าลืม BACK item — หรือลูกศรไม่ตรงกับ ENTERทุก SUBMENU ต้องมี BACK เป็น item สุดท้าย — เช็ค pin mapping ปุ่ม
TOGGLE ค่าไม่เปลี่ยนCallback signature ไม่ตรง — ต้องรับ (LCDMenu_Handle*, uint8_t)Callback TOGGLE ต้องเป็น void cb(LCDMenu_Handle *m, uint8_t val)
EDIT item ไม่เซฟค่าไม่ได้ใช้ LCDMenu_GetValue() ใน callbackใน callback ของ EDIT ให้เรียก LCDMenu_GetValue(&menu) เพื่อดึงค่าที่แก้ไข
RAM หมด — menu item หายเกิน MAX_ITEMS ที่ตั้งใน headerเพิ่ม #define LCDMENU_MAX_ITEMS 32 ก่อน include — หรือลดจำนวน item

🛠️ สร้าง Product: Settings Menu

เมนูปรับแต่งระบบ: WiFi SSID, Temperature Setpoint, Fan Toggle — บันทึกค่าลง Flash เมื่อกด Save

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "LCDMenu.h"
#include "LCD1602_I2C.h"
#include "Flash.h"

LCDMenu_Handle menu;
uint8_t setpoint = 40;
uint8_t fan_on = 0;

void on_setpoint(void) { setpoint = LCDMenu_GetValue(&menu); }
void on_fan(LCDMenu_Handle *m, uint8_t val) {
    fan_on = val;
    pinMode(PC0, PIN_MODE_OUTPUT); digitalWrite(PC0, val);
}
void on_save(void) {
    Flash_Unlock();
    Flash_WriteByte(0x0800FC00, setpoint);
    Flash_WriteByte(0x0800FC01, fan_on);
    Flash_Lock();
    USART_Println("Settings Saved!");
}

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_Init(115200);
    LCD1602_I2C_Init(0x27);
    LCDMenu_Init(&menu, 0x27, 16, 2);
    pinMode(PC3, PIN_MODE_INPUT_PULLUP);
    pinMode(PC4, PIN_MODE_INPUT_PULLUP);
    pinMode(PC5, PIN_MODE_INPUT_PULLUP);
    pinMode(PC6, PIN_MODE_INPUT_PULLUP);

    LCDMenu_AddItem(&menu, "Temperature",  LCDMENU_SUBMENU, NULL);
    LCDMenu_AddItem(&menu, "  Setpoint",   LCDMENU_EDIT,    on_setpoint);
    LCDMenu_AddItem(&menu, "  Back",       LCDMENU_BACK,    NULL);
    LCDMenu_AddItem(&menu, "Fan Control",  LCDMENU_TOGGLE,  on_fan);
    LCDMenu_AddItem(&menu, "Save & Exit",  LCDMENU_ACTION,  on_save);

    while(1) {
        LCDMenu_Update(&menu, !digitalRead(PC3), !digitalRead(PC4),
                       !digitalRead(PC5), !digitalRead(PC6));
        Delay_Ms(50);
        IWDG_Feed();
    }
}

✏️ ลองทำดู

  1. เพิ่ม Password Protect — ก่อนเข้า Settings Menu ต้องพิมพ์รหัส 4 หลักด้วย KeyMatrix
  2. Rotary Encoder Navigation — ใช้ Rotary Encoder แทนปุ่ม UP/DOWN — หมุน=เลื่อน, กด=Enter — ใช้ RotaryEncoder_OnRotate() + LCDMenu_MoveCursor()
  3. Dynamic Menu — เพิ่ม Wi-Fi network list ที่ scan แล้วแสดงแบบ realtime — ใช้ LCDMenu_AddItem() ใน runtime
