LCD 16×2 ตัวอักษรผ่าน I2C (ลด pin usage) — 4 ปุ่มสำหรับนำทางเมนู — ทุกปุ่มใช้ internal pull-up (ACTIVE_LOW) — ต้อง init Button_Instance ก่อน LCDMenu_Init

📚 หลักการทำงาน

LCDMenu: ไลบรารีเมนูแบบลำดับชั้น (hierarchical) สำหรับ LCD1602 16×2 — ใช้ร่วมกับ LCD1602_I2C และ Button — ออกแบบมาสำหรับ MCU หน่วยความจำจำกัดด้วย static memory pool

Menu Tree: สร้างเมนูเป็นลำดับชั้น — เพิ่ม item ด้วย LCDMenu_AddItem() — แต่ละ item มีชื่อ, ประเภท, และ callback — เปลี่ยนหน้าโดย UP/DOWN, เข้าด้วย ENTER, กลับด้วย BACK

4 Callback Types:

  • SUBMENU — เข้าไปยังเมนูย่อย — เหมือน folder — ไม่มี callback แทนที่ด้วยการเปลี่ยนหน้า
  • EDIT — แก้ไขค่าตัวเลข (int/float) — UP/DOWN เปลี่ยนค่า, ENTER ยืนยัน — callback จะได้รับค่าใหม่
  • TOGGLE — เปิด/ปิด — ENTER สลับสถานะ — callback ได้รับ 0 หรือ 1
  • ACTION — เรียก callback ทันทีที่กด ENTER — เหมือนปุ่ม execute
  • BACK — กลับไปเมนูก่อนหน้า — ไม่ต้องมี callback

Static Memory Pool: ทุกอย่างจัดสรรตอน compile — ไม่มี malloc — ปลอดภัยสำหรับ MCU RAM 2KB — กำหนดจำนวน item สูงสุดใน header

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Init Buttons + LCD + Menu

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "LCDMenu.h"
#include "lcd1602_i2c.h"

LCD1602_Handle lcd;
Button_Instance btn_up, btn_down, btn_enter, btn_back;
LCDMenu_Handle menu;

uint8_t setpoint = 40;

void on_setpoint(void) { USART_Print("Setpoint: "); USART_PrintNum(setpoint); }

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_Init(115200);

    // Init LCD1602 I2C
    LCD_Init(&lcd, 0x27, LCD_16x2);

    // Init 4 buttons
    Button_Init(&btn_up,   PC3, BUTTON_ACTIVE_LOW);
    Button_Init(&btn_down, PC4, BUTTON_ACTIVE_LOW);
    Button_Init(&btn_enter,PC5, BUTTON_ACTIVE_LOW);
    Button_Init(&btn_back, PC6, BUTTON_ACTIVE_LOW);

    // Init menu with LCD + buttons
    LCDMenu_Init(&menu, &lcd, &btn_up, &btn_down, &btn_enter, &btn_back);
    LCDMenu_CreateRoot(&menu, "Main Menu");

🔍 LCDMenu_Init(&menu, &lcd, &btn_up, &btn_down, &btn_enter, &btn_back) — ต้อง Init LCD และ Button_Instance ก่อน — LCDMenu_CreateRoot(&menu, "Main Menu") สร้าง root menu

ขั้นที่ 2: สร้าง Menu Tree + Callbacks

    LCDMenu_AddItem(&menu, "Temperature",   LCDMENU_SUBMENU, NULL);
    LCDMenu_AddItem(&menu, "  Set Max",     LCDMENU_EDIT,    on_setpoint);
    LCDMenu_AddItem(&menu, "  Set Min",     LCDMENU_EDIT,    NULL);
    LCDMenu_AddItem(&menu, "  Back",        LCDMENU_BACK,    NULL);
    LCDMenu_AddItem(&menu, "WiFi Control",  LCDMENU_TOGGLE,  NULL);
    LCDMenu_AddItem(&menu, "PID Tuning",    LCDMENU_SUBMENU, NULL);
    LCDMenu_AddItem(&menu, "  Kp=",         LCDMENU_EDIT,    NULL);
    LCDMenu_AddItem(&menu, "  Ki=",         LCDMENU_EDIT,    NULL);
    LCDMenu_AddItem(&menu, "  Back",        LCDMENU_BACK,    NULL);
    LCDMenu_AddItem(&menu, "Save & Exit",   LCDMENU_ACTION,  NULL);
    USART_Println("Menu Ready!");

🔍 LCDMenu_AddItem(&menu, name, type, callback) — ชื่อที่มีเว้นวรรคนำหน้า (" ") คือ item ใน submenu — ต้องมี BACK ทุก submenu — callback เป็น optional สำหรับ SUBMENU/BACK

ขั้นที่ 3: Loop — Update Menu

    while(1) {
        LCDMenu_Update(&menu);  // จัดการ input + render อัตโนมัติ
        Delay_Ms(50);
        IWDG_Feed();
    }
}

🔍 LCDMenu_Update(&menu) — อ่านปุ่มทั้งหมด + จัดการ navigation + render LCD — เรียกครั้งเดียว — refresh 50ms พอดีสำหรับการตอบสนองของมนุษย์ — อย่าลืม IWDG_Feed เพื่อกัน MCU รีเซ็ต

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
จอไม่แสดงอะไรเลยลืม LCD_Init() ก่อน LCDMenu_Init()ต้อง Init LCD ก่อน — LCD_Init(&lcd, 0x27, 16, 2) แล้วตามด้วย LCDMenu_Init()
กด UP/DOWN แล้วกระโดดหลายช่องRefresh rate เร็วเกินไป + ไม่มี debounceเพิ่ม Delay_Ms เป็น 80-100ms — Button_Instance มี debounce ในตัว
SUBMENU ไม่ยอมเข้าลืม BACK item — หรือลูกศรไม่ตรงกับ ENTERทุก SUBMENU ต้องมี BACK เป็น item สุดท้าย — เช็ค pin mapping ปุ่ม
TOGGLE ค่าไม่เปลี่ยนCallback signature ไม่ตรงCallback TOGGLE ต้องเป็น void cb(uint8_t val)
EDIT item ไม่เซฟค่าไม่ได้ใช้ค่าที่แก้ไขใน callbackใน callback ของ EDIT ให้ดึงค่าที่แก้ไขมาใช้
RAM หมด — menu item หายเกิน MAX_ITEMS ที่ตั้งใน headerเพิ่ม #define LCDMENU_MAX_ITEMS 32 ก่อน include — หรือลดจำนวน item

🛠️ สร้าง Product: Settings Menu

เมนูปรับแต่งระบบ: WiFi SSID, Temperature Setpoint, Fan Toggle — บันทึกค่าลง Flash เมื่อกด Save

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "LCDMenu.h"
#include "lcd1602_i2c.h"

LCD1602_Handle lcd;
Button_Instance btn_up, btn_down, btn_enter, btn_back;
LCDMenu_Handle menu;
uint8_t setpoint = 40;
uint8_t fan_on = 0;

void on_setpoint(void) { /* get value from menu */ }
void on_fan(uint8_t val) {
    fan_on = val;
    pinMode(PC0, PIN_MODE_OUTPUT); digitalWrite(PC0, val);
}
void on_save(void) {
    Flash_Unlock();
    Flash_WriteByte(0x0800FC00, setpoint);
    Flash_WriteByte(0x0800FC01, fan_on);
    Flash_Lock();
    USART_Println("Settings Saved!");
}

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_Init(115200);

    LCD_Init(&lcd, 0x27, LCD_16x2);
    Button_Init(&btn_up,   PC3, BUTTON_ACTIVE_LOW);
    Button_Init(&btn_down, PC4, BUTTON_ACTIVE_LOW);
    Button_Init(&btn_enter,PC5, BUTTON_ACTIVE_LOW);
    Button_Init(&btn_back, PC6, BUTTON_ACTIVE_LOW);

    LCDMenu_Init(&menu, &lcd, &btn_up, &btn_down, &btn_enter, &btn_back);
    LCDMenu_CreateRoot(&menu, "Settings");

    LCDMenu_AddItem(&menu, "Temperature",  LCDMENU_SUBMENU, NULL);
    LCDMenu_AddItem(&menu, "  Setpoint",   LCDMENU_EDIT,    on_setpoint);
    LCDMenu_AddItem(&menu, "  Back",       LCDMENU_BACK,    NULL);
    LCDMenu_AddItem(&menu, "Fan Control",  LCDMENU_TOGGLE,  on_fan);
    LCDMenu_AddItem(&menu, "Save & Exit",  LCDMENU_ACTION,  on_save);

    while(1) {
        LCDMenu_Update(&menu);
        Delay_Ms(50);
        IWDG_Feed();
    }
}
