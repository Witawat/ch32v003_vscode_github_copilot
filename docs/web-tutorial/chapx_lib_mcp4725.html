🔌 การต่อวงจร

            MCP4725 12-bit DAC
   ┌──────────────────────────────────┐
   │  MCU              MCP4725        │
   │  PC1─SDA──────────SDA   VDD─3.3V │
   │  PC2─SCL──────────SCL   GND─GND  │
   │                   A0─GND (0x60)  │
   │  Vout ──────────── Output        │
   └──────────────────────────────────┘
MCU Pin→อุปกรณ์ขาหมายเหตุ
PC1↔SDA-สายข้อมูล I2C — ต้องมี R 4.7kΩ pull-up ไป 3.3V
PC2↔SCL-สายนาฬิกา I2C — ต้องมี R 4.7kΩ pull-up ไป 3.3V
3.3V→VDD-ไฟเลี้ยง 3.3V
GND→GND-กราวด์ร่วม; A0→GND=0x60, A0→VDD=0x61

รู้จักขา

  • VDD: ไฟเลี้ยง 3.3V
  • GND: กราวด์
  • SDA (PC1): สายข้อมูล I2C
  • SCL (PC2): สายนาฬิกา I2C
  • A0: เลือก I2C address — GND=0x60, VDD=0x61
  • Vout: ขาเอาต์พุต 0–3.3V
  • pull-up 4.7kΩ: ต่อระหว่าง SDA→3.3V และ SCL→3.3V (โมดูลส่วนมากมีในตัว)

วิธีต่อทีละขั้น

  1. ต่อ VDD ของ MCP4725 ไป 3.3V
  2. ต่อ GND ของ MCP4725 ไป GND
  3. ต่อ SDA ของ MCP4725 ไป PC1 (ขา 13) ของ CH32V003
  4. ต่อ SCL ของ MCP4725 ไป PC2 (ขา 14) ของ CH32V003
  5. ต่อ A0 ไป GND เพื่อใช้ address 0x60 (หรือ VDD=0x61)
  6. ต่อ Vout ไปยังอุปกรณ์ที่ต้องการรับสัญญาณ analog (เช่น oscilloscope, op-amp)
  7. ตรวจสอบว่าโมดูลมี R pull-up 4.7kΩ ในตัวหรือไม่ — ถ้าไม่มี ต่อ R เพิ่มจาก SDA→3.3V และ SCL→3.3V

ต่อ I2C ปกติ — A0 pin เลือก address (GND=0x60, VDD=0x61) — Output ต่อตรงเข้า Oscilloscope

📚 หลักการทำงาน

MCP4725 เป็น DAC 12-bit สื่อสารผ่าน I2C — รับค่า 0-4095 แปลงเป็นแรงดัน 0V ถึง VDD (3.3V) — ความละเอียด 3.3V/4096 ≈ 0.8mV ต่อ step

MCP4725_SetVoltage(&dac, volt, vref) ตั้งแรงดันโดยตรง — vref เป็น MCP4725_VREF_VDD หรือค่าแรงดันอ้างอิงภายนอก

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Init MCP4725

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "MCP4725.h"

MCP4725_Instance dac;

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    I2C_Init(PC1, PC2, I2C_SPEED_FAST);
    USART_Init(115200);

    MCP4725_Init(&dac, 0x60);
    USART_Print("MCP4725 Ready\r\n");

🔍 MCP4725_Init(&dac, addr) — ตั้งค่า I2C address — ใช้ I2C Fast Mode (400kHz)

ขั้นที่ 2: เขียนค่า DAC

    // ตั้งแรงดัน 1.65V (ครึ่ง VDD)
    MCP4725_SetVoltage(&dac, 1.65f, MCP4725_VREF_VDD);

    // ตั้งแรงดัน 0.825V
    MCP4725_SetVoltage(&dac, 0.825f, MCP4725_VREF_VDD);

    // Write raw value (0-4095)
    MCP4725_WriteRaw(&dac, 2048);

🔍 MCP4725_SetVoltage(&dac, volt, vref) — ตั้งแรงดันโดยตรง ไม่ต้องคำนวณ raw value เอง — MCP4725_VREF_VDD ใช้ VDD เป็นแรงดันอ้างอิง

ขั้นที่ 3: สร้าง Waveform Generator

    const uint16_t sine_table[64] = {
        2048,2248,2447,2642,2831,3013,3185,3346,
        3495,3630,3750,3853,3939,4007,4056,4085,
        4095,4085,4056,4007,3939,3853,3750,3630,
        3495,3346,3185,3013,2831,2642,2447,2248,
        2048,1847,1649,1453,1264,1082,910,749,
        600,465,345,242,156,88,39,10,
        0,10,39,88,156,242,345,465,
        600,749,910,1082,1264,1453,1649,1847
    };

    uint8_t idx = 0;
    while(1) {
        MCP4725_WriteRaw(&dac, sine_table[idx]);
        idx = (idx + 1) % 64;
        Delay_Us(100); // ~156Hz
    }
}

🔍 ใช้ตาราง sine 64 จุด pre-computed — วนลูปด้วย WriteRaw — 100µs delay = ~156Hz sine wave

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้

🛠️ สร้าง Product: Analog Waveform Generator

MCP4725 สร้าง Sine, Triangle, Sawtooth, Square wave

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "MCP4725.h"

int main(void)
{
    SystemCoreClockUpdate();
    Timer_Init();
    I2C_Init(PC1, PC2, I2C_SPEED_FAST);
    USART_Init(115200);

    MCP4725_Instance dac;
    MCP4725_Init(&dac, 0x60);

    uint16_t sine64[64] = {2048,2248,2447,2642,2831,3013,3185,3346,
        3495,3630,3750,3853,3939,4007,4056,4085,
        4095,4085,4056,4007,3939,3853,3750,3630,
        3495,3346,3185,3013,2831,2642,2447,2248,
        2048,1847,1649,1453,1264,1082,910,749,
        600,465,345,242,156,88,39,10,
        0,10,39,88,156,242,345,465,
        600,749,910,1082,1264,1453,1649,1847};

    uint8_t idx = 0;
    while(1)
    {
        MCP4725_WriteRaw(&dac, sine64[idx]);
        idx = (idx + 1) % 64;
        Delay_Us(100);
    }
}
