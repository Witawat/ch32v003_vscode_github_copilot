MCP4725 — DAC 12-Bit I2C พร้อม EEPROM

📚 หลักการทำงาน

MCP4725 เป็น DAC 12-bit สื่อสารผ่าน I2C — รับค่า 0-4095 แปลงเป็นแรงดัน 0V ถึง VDD (3.3V) — ความละเอียด 3.3V/4096 ≈ 0.8mV ต่อ step — มี Buffer Amplifier ในตัว ขับโหลดได้สูง

EEPROM ภายใน เก็บค่า DAC default — เมื่อเปิดเครื่อง MCP4725 จะอ่านค่า EEPROM มาใช้เป็นค่าเริ่มต้น — ใช้สำหรับงานที่ต้องการค่าเริ่มต้นคงที่แม้ไฟดับ

Fast Write Mode เขียนค่า DAC โดยไม่เขียน EEPROM — เร็วกว่า ใช้สร้าง Waveform (sine, triangle, sawtooth) ได้ — I2C Standard 100kHz ส่ง 3 bytes ≈ 240µs ต่อการอัพเดท 1 ครั้ง

Write + EEPROM เขียนทั้ง DAC และ EEPROM — ช้ากว่า (~25ms เพราะ EEPROM write cycle) — ใช้เมื่อต้องการให้ค่าคงอยู่หลังปิดเครื่อง

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Init MCP4725

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "MCP4725.h"

MCP4725_t dac;

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    I2C_Init(PC1, PC2, I2C_SPEED_FAST); // Fast mode 400kHz
    USART_Init(115200);

    MCP4725_Init(&dac, 0x60); // addr=0x60 (A0=GND)
    USART_Print("MCP4725 Ready\r\n");

🔍 MCP4725_Init(dev, addr) — ตั้งค่า I2C address — ใช้ I2C Fast Mode (400kHz) เพื่อความเร็วสูงสุดในการอัพเดท waveform

ขั้นที่ 2: เขียนค่า DAC + อ่าน EEPROM

    // Fast Write — แค่เปลี่ยน output
    MCP4725_FastWrite(&dac, 2048); // ~1.65V (ครึ่ง VDD)

    // Write + Save EEPROM
    MCP4725_WriteEEPROM(&dac, 1024); // ~0.825V — เก็บใน EEPROM
    USART_Print("Saved 1024 to EEPROM\r\n");

    // อ่านค่าใน EEPROM
    uint16_t saved = MCP4725_ReadEEPROM(&dac);
    USART_Print("EEPROM value: ");
    USART_PrintDec(saved);
    USART_Print("\r\n");

🔍 FastWrite(value) อัพเดท DAC ทันที ~240µs — WriteEEPROM(value) เขียนทั้ง DAC + EEPROM ~25ms — ReadEEPROM() อ่านค่าที่บันทึกไว้ใน EEPROM

ขั้นที่ 3: สร้าง Waveform Generator

    // Sine wave table (pre-computed)
    const uint16_t sine_table[64] = {
        2048,2248,2447,2642,2831,3013,3185,3346,
        3495,3630,3750,3853,3939,4007,4056,4085,
        4095,4085,4056,4007,3939,3853,3750,3630,
        3495,3346,3185,3013,2831,2642,2447,2248,
        2048,1847,1649,1453,1264,1082,910,749,
        600,465,345,242,156,88,39,10,
        0,10,39,88,156,242,345,465,
        600,749,910,1082,1264,1453,1649,1847
    };

    uint8_t idx = 0;
    while(1) {
        MCP4725_FastWrite(&dac, sine_table[idx]);
        idx = (idx + 1) % 64;
        Delay_Us(100); // ~156Hz
    }
}

🔍 ใช้ตาราง sine 64 จุด pre-computed — วนลูปด้วย Fast Write — 100µs delay = ~156Hz sine wave — เปลี่ยน delay เพื่อเปลี่ยนความถี่ — เชื่อมต่อกับ Oscilloscope ดู waveform ได้

🛠️ สร้าง Product: Analog Waveform Generator

MCP4725 สร้าง Sine, Triangle, Sawtooth, Square wave — เลือก waveform ผ่านปุ่ม — ปรับความถี่ด้วย Potentiometer (ADC)

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "MCP4725.h"

const uint16_t sine64[64] = {
    2048,2248,2447,2642,2831,3013,3185,3346,
    3495,3630,3750,3853,3939,4007,4056,4085,
    4095,4085,4056,4007,3939,3853,3750,3630,
    3495,3346,3185,3013,2831,2642,2447,2248,
    2048,1847,1649,1453,1264,1082,910,749,
    600,465,345,242,156,88,39,10,
    0,10,39,88,156,242,345,465,
    600,749,910,1082,1264,1453,1649,1847
};

int main(void)
{
    SystemCoreClockUpdate();
    Timer_Init();
    I2C_Init(PC1, PC2, I2C_SPEED_FAST);
    ADC_Init(PD2, ADC_CH0);
    USART_Init(115200);

    MCP4725_t dac;
    MCP4725_Init(&dac, 0x60);

    pinMode(PC0, PIN_MODE_INPUT_PULLUP); // ปุ่มเปลี่ยน waveform
    uint8_t wave_type = 0;
    uint8_t idx = 0;

    while(1)
    {
        uint16_t freq = ADC_Read(PD2); // 0-1023 → frequency
        uint16_t delay_us = 5000 / (freq + 1); // 5kHz max ÷ (freq+1)

        if(GPIO_Read(PC0) == 0) {
            Delay_Ms(50); // debounce
            wave_type = (wave_type + 1) % 4;
            while(GPIO_Read(PC0) == 0);
        }

        uint16_t val;
        switch(wave_type) {
            case 0: val = sine64[idx]; break;
            case 1: val = (idx * 64) % 4096; break;         // Sawtooth
            case 2: val = (idx < 32) ? (idx*128) : (4095 - (idx-32)*128); break; // Triangle
            case 3: val = (idx < 32) ? 4095 : 0; break;     // Square
        }

        MCP4725_FastWrite(&dac, val);
        idx = (idx + 1) % 64;
        Delay_Us(delay_us);
    }
}

✏️ ลองทำดู

  1. ใช้ตาราง 256 จุดสร้าง sine wave ความละเอียดสูง — ลองหาค่าสูงสุดของ sampling rate ที่ CH32V003 ทำได้
  2. สร้าง Arbitrary Waveform — รับค่าผ่าน USART แล้วเล่น waveform ตามที่ผู้ใช้กำหนด
  3. ต่อ RC Low-Pass Filter ที่ Output — ดูผลการกรองสัญญาณที่ความถี่ cutoff ต่าง ๆ
  4. สร้าง Audio Tone Generator — เปลี่ยน MCP4725 เป็น sine wave 440Hz (Note A4) — ต่อลำโพงผ่าน amplifier
