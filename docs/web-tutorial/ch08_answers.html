📝 เฉลย — บทที่ 8: SPI

ข้อ 1: อ่าน JEDEC ID จาก W25Q Flash

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include "SimpleHAL.h"

#define CS_FLASH  PC4

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
    SPI_SimpleInit(SPI_MODE0, SPI_1MHZ, SPI_PINS_DEFAULT);

    pinMode(CS_FLASH, PIN_MODE_OUTPUT);
    digitalWrite(CS_FLASH, HIGH);  // CS inactive
    SPI_SetCSPin(GPIOC, GPIO_Pin_4);

    while (1) {
        // Send JEDEC ID command (0x9F)
        digitalWrite(CS_FLASH, LOW);
        SPI_Transfer(0x9F);
        uint8_t mfg = SPI_Transfer(0x00);  // Manufacturer ID
        uint8_t mem = SPI_Transfer(0x00);  // Memory Type
        uint8_t cap = SPI_Transfer(0x00);  // Capacity
        digitalWrite(CS_FLASH, HIGH);

        USART_Print("JEDEC ID: ");
        USART_PrintHex(mfg, 1);
        USART_Print(" ");
        USART_PrintHex(mem, 1);
        USART_Print(" ");
        USART_PrintHex(cap, 1);

        // W25Q32: EF 40 16 (Winbond, QSPI, 32Mbit)
        USART_Print("\r\n");

        Delay_Ms(2000);
    }
}
🧠 แนวคิด: W25Qxx ใช้ SPI Mode 0 — เริ่มด้วย CS LOW เพื่อเลือก Flash — ส่ง command 0x9F (Read JEDEC ID) — ต่อด้วย dummy byte 0x00 3 ครั้งเพื่อสร้าง clock ให้รับข้อมูลกลับ — CS HIGH เพื่อปล่อย bus — JEDEC ID 3 bytes: [Manufacturer] [Memory Type] [Capacity] — W25Q32 = EF 40 16, W25Q64 = EF 40 17

ข้อ 2: เปลี่ยน SPI speed แบบ runtime

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include "SimpleHAL.h"

#define CS_FLASH  PC4

void read_jedec(void) {
    digitalWrite(CS_FLASH, LOW);
    SPI_Transfer(0x9F);
    uint8_t id[3];
    id[0] = SPI_Transfer(0x00);
    id[1] = SPI_Transfer(0x00);
    id[2] = SPI_Transfer(0x00);
    digitalWrite(CS_FLASH, HIGH);

    USART_Print("ID: ");
    USART_PrintHex(id[0], 1); USART_Print(" ");
    USART_PrintHex(id[1], 1); USART_Print(" ");
    USART_PrintHex(id[2], 1); USART_Print("\r\n");
}

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
    SPI_SimpleInit(SPI_MODE0, SPI_1MHZ, SPI_PINS_DEFAULT);

    pinMode(CS_FLASH, PIN_MODE_OUTPUT);
    digitalWrite(CS_FLASH, HIGH);
    SPI_SetCSPin(GPIOC, GPIO_Pin_4);

    // ทดสอบที่ 1MHz
    USART_Print("Speed: 1MHz\r\n");
    read_jedec();
    Delay_Ms(500);

    // เปลี่ยนเป็น 500kHz
    SPI_SetSpeed(SPI_500KHZ);
    USART_Print("Speed: 500kHz\r\n");
    read_jedec();
    Delay_Ms(500);

    // เปลี่ยนเป็น 8MHz
    SPI_SetSpeed(SPI_8MHZ);
    USART_Print("Speed: 8MHz\r\n");
    read_jedec();
    Delay_Ms(500);

    USART_Print("Done — all speeds OK!\r\n");
    while (1);
}
🧠 แนวคิด: SPI_SetSpeed(speed) เปลี่ยนความเร็ว SPI ได้ทันทีที่ runtime — ไม่ต้อง re-init — speed constants: SPI_500KHZ, SPI_1MHZ, SPI_2MHZ, SPI_4MHZ, SPI_8MHZ — W25Qxx รองรับสูงสุด ~50MHz — ข้อควรระวัง: ความเร็วสูงมากอาจมีปัญหา noise หากสายยาวหรือ breadboard — 8MHz ควรใช้ PCB ที่มี layout ดี

ข้อ 3: ใช้ SOP-8 — เปลี่ยนไปใช้ shiftOut แทน SPI HW

#define CH32V003_PACKAGE  PACKAGE_SOP8
#include "SimpleHAL.h"

#define DATA_PIN   PC1    // DS (SER)
#define CLOCK_PIN  PC2    // SHCP (SRCLK)
#define LATCH_PIN  PC4    // STCP (RCLK)

void hc595_write(uint8_t data) {
    digitalWrite(LATCH_PIN, LOW);
    shiftOut(DATA_PIN, CLOCK_PIN, MSBFIRST, data);
    digitalWrite(LATCH_PIN, HIGH);
}

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();

    pinMode(DATA_PIN, PIN_MODE_OUTPUT);
    pinMode(CLOCK_PIN, PIN_MODE_OUTPUT);
    pinMode(LATCH_PIN, PIN_MODE_OUTPUT);

    uint8_t pattern = 0x01;

    while (1) {
        hc595_write(pattern);   // แสดง LED
        hc595_write(~pattern);  // ตรงข้าม

        // Knight Rider 8-bit
        pattern = (pattern << 1);
        if (pattern == 0) pattern = 0x01;

        Delay_Ms(100);
    }
}
🧠 แนวคิด: SOP-8/SOP-16 ไม่มี Hardware SPI (ไม่มี PC5/PC6/PC7) — ใช้ shiftOut() แบบ software bit-bang แทน — 74HC595 เป็น shift register 8-bit — 3 pins control: DATA (serial in), CLOCK (shift), LATCH (output) — shiftOut(DATA_PIN, CLOCK_PIN, MSBFIRST, data) ส่ง 8 bits ทีละ bit — LATCH ใช้ pulse เพื่ออัปเดต output — ช้ากว่า HW SPI แต่ยืดหยุ่น ใช้ pin ไหนก็ได้ — สำหรับ W25Q Flash: SOP-8 ใช้ shiftOut() + shiftIn() ได้เช่นกัน

🔌 วงจรประกอบ — SPI (W25Q Flash)

        CH32V003              W25Qxx Flash
        ┌──────────┐       ┌──────────────┐
        │    PC5 ──┼──────▶│ SCK (Clock)  │
        │    PC6 ──┼──────▶│ MOSI (Data)  │
        │    PC7 ──┼───────│ MISO (Data)  │
        │    PC4 ──┼──────▶│ CS  (Select) │
        │    VCC ──┼───────│ VCC          │
        │    GND ──┼───────│ GND          │
        └──────────┘       └──────────────┘
        
        SPI_PINS_DEFAULT: SCK=PC5, MOSI=PC6, MISO=PC7, CS=PC4
        ⚠️ SOP-8 ไม่มี PC5(SCK) → SPI HW ใช้ไม่ได้ → ใช้ shiftOut() แทน
💡 หลักการ SPI: Full-duplex — ส่ง 1 byte พร้อมรับ 1 byte — CS ต้อง LOW ก่อนส่ง, HIGH หลังส่งเสร็จ

🔌 วงจรประกอบ — SOP-8: 74HC595 (shiftOut)

        ┌──────────┐       ┌──────────────┐
        │    PD5 ──┼──────▶│ DS  (Data)   │
        │    PD6 ──┼──────▶│ SHCP(Clock)  │
        │    PD4 ──┼──────▶│ STCP(Latch)  │
        │    VCC ──┼───────│ VCC, MR      │
        │    GND ──┼───────│ GND, OE      │
        └──────────┘       └──┬──┬──┬──┬──┘
                            Q0 Q1 Q2 ... Q7
                             │  │  │      │
                            LED LED LED  LED
💡 หลักการ shiftOut: ส่งข้อมูลทีละ bit ผ่าน DS — SHCP clock แต่ละ pulse เลื่อนข้อมูลเข้า 1 bit — STCP latch copy ข้อมูลออก Q0-Q7 ทั้ง 8 ขาพร้อมกัน
