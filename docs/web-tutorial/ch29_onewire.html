ex01_DS18B20_Temperature — อ่านอุณหภูมิจาก DS18B20

📖 แนวคิด

DS18B20 เป็นเซ็นเซอร์อุณหภูมิแบบ 1-Wire — สื่อสารด้วยโปรโตคอลเฉพาะ: Reset → SkipROM → Convert → Read Scratchpad ใช้ OneWire_Init, OneWire_WriteByte, OneWire_ReadByte

🔌 วงจร

    CH32V003              DS18B20
    PC0 ----[4.7kΩ]----+-- DQ (ขา 2)
                         |
                        VDD (3.3V)
                       GND
    - ต้องมี R pull-up 4.7kΩ เสมอ — 1-Wire ใช้ open-drain

💻 โค้ด + อธิบายทีละบรรทัด

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include <SimpleHAL.h>

#define ONE_WIRE_PIN  PC0

int main(void)
{
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);

    OneWire_Init(ONE_WIRE_PIN);                    // ตั้งค่า pin เป็น 1-Wire

    while(1)
    {
        if(OneWire_Reset() == 0) {                  // ส่ง reset pulse — รอ presence
            OneWire_WriteByte(0xCC);                // SkipROM — ข้ามการระบุ ROM
            OneWire_WriteByte(0x44);                // เริ่ม Convert T
            Delay_Ms(750);                           // รอ conversion เสร็จ (~750ms)

            OneWire_Reset();
            OneWire_WriteByte(0xCC);                // SkipROM อีกครั้ง
            OneWire_WriteByte(0xBE);                // Read Scratchpad

            uint8_t lsb = OneWire_ReadByte();       // อ่าน byte ต่ำ
            uint8_t msb = OneWire_ReadByte();       // อ่าน byte สูง
            int16_t raw = (msb << 8) | lsb;
            float temp = raw * 0.0625f;              // DS18B20: 1 LSB = 0.0625°C

            USART_Print("Temperature: ");
            USART_PrintFloat(temp, 2);
            USART_Print(" °C\r\n");
        }
        Delay_Ms(1000);
    }
}
  • OneWire_Init(pin): ตั้ง pin เป็น open-drain output พร้อม pull-up 4.7kΩ ภายนอก
  • OneWire_Reset(): ดึง LOW 480µs → ปล่อย → รอ presence pulse จาก slave (60-240µs) — คืน 0 ถ้าเจออุปกรณ์
  • OneWire_WriteByte(byte): ส่ง 1 byte แบบ serial 1-Wire — bit ละ time slot ~60µs
  • OneWire_ReadByte(): อ่าน 1 byte — MCU ดึง LOW 1µs → ปล่อย → อ่าน bit ที่ slave ส่ง

⚠️ จุดควรระวัง

  • ต้องมี R pull-up 4.7kΩ — 1-Wire ใช้ open-drain ขาด R = สื่อสารไม่ได้
  • Delay_Ms(750) ระหว่าง Convert — DS18B20 ใช้เวลาแปลงสูงสุด 750ms (12-bit resolution)
  • ถ้าไม่เจอ DS18B20 (OneWire_Reset != 0) — เช็คสาย + pull-up + pin

✅ ตรวจสอบก่อนรัน

  • DS18B20 ต่อ VDD 3.3V? DQ มี R 4.7kΩ ขึ้น VDD?
  • ลอง scan ด้วย ex02 เพื่อดูว่าเจอ ROM address ไหม

ex02_ROM_Search — ค้นหาอุปกรณ์ 1-Wire ทั้งหมด

📖 แนวคิด

ใช้ OneWire_Search เพื่อค้นหา ROM address ของอุปกรณ์ 1-Wire ทุกตัวบนบัส — ใช้ GetAddress ดึง address ที่พบ — ตรวจสอบ CRC8 เพื่อความถูกต้อง

💻 โค้ด

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include <SimpleHAL.h>

#define ONE_WIRE_PIN  PC0

int main(void)
{
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);

    OneWire_Init(ONE_WIRE_PIN);

    OneWire_Reset();                                 // รีเซ็ตบัส + รอ presence

    // ค้นหาครั้งแรก — ต้องส่ง ROM command ก่อน
    OneWire_WriteByte(0xF0);                         // SEARCH ROM
    uint8_t addr[8];
    if(OneWire_Search(addr)) {                       // วนค้นหา address
        USART_Print("Found device:\r\n");
        for(uint8_t i = 0; i < 8; i++) {
            USART_PrintHex(addr[i], 2);
            USART_Print(" ");
        }
        USART_Print("\r\n");

        if(OneWire_CRC8(addr, 7) == addr[7]) {      // ตรวจ CRC
            USART_Print("CRC8 OK\r\n");
        } else {
            USART_Print("CRC8 ERROR!\r\n");
        }
    } else {
        USART_Print("No device found\r\n");
    }

    while(1) { IWDG_Feed(); }
}
  • OneWire_Search(buffer): อัลกอริทึม ROM search — ส่ง 0xF0 (SEARCH ROM) → อ่าน bit พร้อม inversion → หา ROM ตัวแรก
  • OneWire_CRC8(data, len): คำนวณ CRC-8/Maxim (x⁸ + x⁵ + x⁴ + 1) — ตรวจสอบ byte สุดท้ายของ ROM
  • 0xF0 SEARCH ROM: สั่งให้อุปกรณ์ทุกตัวส่ง ROM address — ถ้ามีหลายตัวจะชนกัน ต้องใช้ search algorithm

⚠️ จุดควรระวัง

  • มีหลายอุปกรณ์บนบัส — 1-Wire ใช้ line arbitration — ต้อง SEARCH ROM เพื่อแยก address
  • OneWire_Search ค้นหาเจอทีละ device — ถ้าหลายตัว เรียกซ้ำจนกว่าคืน 0
  • CRC8 จำเป็น — ROM address ผิด 1 bit = เลือกผิดอุปกรณ์

ex03_Multi_Device_Select — เลือกอุปกรณ์หลายตัว

📖 แนวคิด

ใช้ OneWire_Select + matchROM เพื่อเลือกอุปกรณ์ 1-Wire ตัวใดตัวหนึ่งจากหลายตัวบนบัส — ส่ง ROM address 8 byte ตามด้วยคำสั่ง

💻 โค้ด

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include <SimpleHAL.h>

#define ONE_WIRE_PIN  PC0

// สมมติ ROM address ที่ค้นพบจาก ex02
uint8_t sensor1[8] = {0x28, 0xAA, 0xBB, 0xCC, 0xDD, 0xEE, 0xFF, 0x12};
uint8_t sensor2[8] = {0x28, 0x11, 0x22, 0x33, 0x44, 0x55, 0x66, 0x98};

static float read_temperature(uint8_t *rom)
{
    OneWire_Reset();
    OneWire_Select(rom);                             // MATCH ROM — เลือก device
    OneWire_WriteByte(0x44);                         // Convert T

    Delay_Ms(750);

    OneWire_Reset();
    OneWire_Select(rom);
    OneWire_WriteByte(0xBE);                         // Read Scratchpad

    uint8_t lsb = OneWire_ReadByte();
    uint8_t msb = OneWire_ReadByte();
    int16_t raw = (msb << 8) | lsb;
    return raw * 0.0625f;
}

int main(void)
{
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
    OneWire_Init(ONE_WIRE_PIN);

    while(1)
    {
        float t1 = read_temperature(sensor1);
        float t2 = read_temperature(sensor2);
        USART_Print("Sensor1: "); USART_PrintFloat(t1, 2);
        USART_Print(" | Sensor2: "); USART_PrintFloat(t2, 2);
        USART_Print(" °C\r\n");
        Delay_Ms(1000);
    }
}
  • OneWire_Select(rom): ส่ง 0x55 (MATCH ROM) ตามด้วย ROM address 8 byte — เฉพาะ device ที่ match เท่านั้นตอบสนอง
  • matchROM: ส่ง ROM address ทีละ byte — อุปกรณ์ที่ตรงกันจะยอมรับคำสั่งถัดไป

ex04_MatchROM — ค้นหา + MatchROM ทีละตัว

📖 แนวคิด

รวม OneWire_Search กับ OneWire_Select — ค้นหาอุปกรณ์ก่อน แล้ว MatchROM ทีละตัวเพื่ออ่านค่า — รองรับหลายอุปกรณ์อัตโนมัติ

💻 โค้ด

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include <SimpleHAL.h>

#define ONE_WIRE_PIN  PC0
#define MAX_DEVICES   10

int main(void)
{
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
    OneWire_Init(ONE_WIRE_PIN);

    // ค้นหาอุปกรณ์ทั้งหมด
    uint8_t devices[MAX_DEVICES][8];
    uint8_t count = 0;

    OneWire_Reset();
    while(OneWire_Search(devices[count]) && count < MAX_DEVICES) {
        if(OneWire_CRC8(devices[count], 7) != devices[count][7]) {
            USART_Print("CRC fail on device ");
            USART_PrintNum(count);
            USART_Print("\r\n");
            continue;
        }
        count++;
    }
    USART_Print("Found ");
    USART_PrintNum(count);
    USART_Print(" device(s)\r\n");

    while(1)
    {
        for(uint8_t i = 0; i < count; i++) {
            OneWire_Reset();
            OneWire_Select(devices[i]);              // MatchROM
            OneWire_WriteByte(0x44);
            Delay_Ms(750);

            OneWire_Reset();
            OneWire_Select(devices[i]);
            OneWire_WriteByte(0xBE);

            uint8_t lsb = OneWire_ReadByte();
            uint8_t msb = OneWire_ReadByte();
            int16_t raw = (msb << 8) | lsb;

            USART_Print("Dev");
            USART_PrintNum(i);
            USART_Print(": ");
            USART_PrintFloat(raw * 0.0625f, 2);
            USART_Print(" °C\r\n");
        }
        Delay_Ms(1000);
    }
}
  • OneWire_Search ใน loop: เรียกซ้ำ — แต่ละครั้งเจอ 1 address — คืน 0 เมื่อหาครบทุกตัว
  • OneWire_Select + ROM: ใช้ 0x55 MATCH ROM — ระบุ device ที่จะคุยด้วย
  • ค้นหา + MatchROM = โซลูชันสมบูรณ์: ไม่ต้อง hardcode ROM address — รองรับการเปลี่ยนอุปกรณ์

⚠️ จุดควรระวัง

  • 1-Wire timing เป็น µs — ถ้า interrupt มาขัด timing จะเพี้ยน — ปิด global interrupt ขณะสื่อสารหรือใช้ critical section
  • OneWire_Search ใช้เวลานานถ้ามีหลายอุปกรณ์ — แต่ละ bit ใช้ 2 time slots ~120µs × 64 bit × จำนวนรอบ
  • ถ้าไม่มีอุปกรณ์ — OneWire_Search คืน 0 ทันที
