MCU Pin→อุปกรณ์ขาหมายเหตุ
PC0←HC-SR501 PIROUTHIGH = ตรวจจับ motion ได้
3.3V / 5V→HC-SR501 PIRVCCเลือกไฟตาม jumper
GND→HC-SR501 PIRGND

PC0─OUT (INPUT_PULLUP หรือ INPUT) — ใช้ไฟ 3.3V หรือ 5V — มีปุ่มปรับ Sensitivity (ระยะ) และ Delay Time (นานในการค้าง HIGH หลังตรวจจับ)

รู้จักขา

  • OUT: ขาสัญญาณ ต่อกับ PC0 — HIGH เมื่อเจอ motion, LOW เมื่อไม่มี
  • VCC: ไฟเลี้ยง 3.3V หรือ 5V (เลือกตาม jumper)
  • GND: ต่อกราวด์ร่วมกับ MCU

วิธีต่อทีละขั้น

  1. ต่อ PC0 ไป OUT ของ PIR — อ่านสถานะ motion
  2. ต่อ 3.3V หรือ 5V ไป VCC — ดู jumper เลือกไฟให้ตรง
  3. ต่อ GND ไป GND — กราวด์ร่วม
  4. ปรับ pot: Sensitivity (ระยะตรวจจับ 3–7m) และ Delay Time (เวลาค้างสัญญาณ 5–300s)

📚 หลักการทำงาน

PIR (Passive Infrared) เซนเซอร์ตรวจจับการเปลี่ยนแปลงของรังสีอินฟราเรด — เมื่อมีมนุษย์หรือสัตว์ (แหล่งความร้อน) เคลื่อนที่ผ่านหน้าการตรวจจับ → pyroelectric sensor ตรวจจับความแตกต่างของ IR → output HIGH

Fresnel Lens: เลนส์พลาสติกทรงโดมรวมแสง IR — แบ่งพื้นที่ตรวจจับเป็นโซน — การเคลื่อนผ่านโซนทำให้เกิดการเปลี่ยนแปลง IR → trigger

2 โหมดการทำงาน:

  • Repeatable (L): ทุกการเคลื่อนไหว → reset delay timer → output ค้าง HIGH ถ้ายังมีคนเคลื่อนที่
  • Non-Repeatable (H): Trigger ครั้งเดียว → รอ delay → LOW → รอถึงจะ trigger ใหม่

Interrupt vs Polling: PIR_SetCallback ใช้ interrupt จับทุก edge แม่นกว่า — PIR_SetDebounce ตั้ง lockout กัน trigger ซ้ำ — PIR_Update() ต้องเรียกใน loop

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Init + Setup Callback

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "PIR.h"

volatile uint8_t motionFlag = 0;

// Interrupt callback
void my_callback(void) {
    motionFlag = 1;
}

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_Init(USART1, BAUD_115200);

    pinMode(PC1, PIN_MODE_OUTPUT);  // LED indicator

    // PIR: pin=PC0, callback=my_callback
    PIR_Init(PC0);
    PIR_SetCallback(my_callback);
    PIR_SetDebounce(100, 5000);  // 100ms lockout, 5000ms duration
    USART_Println("PIR Ready...");

🔍 PIR_Init(pin) เริ่มขา PIN — PIR_SetCallback(cb) ตั้งฟังก์ชันที่จะเรียกเมื่อตรวจจับ motion — PIR_SetDebounce(lockout, duration) ตั้งเวลากันซ้ำ (100ms) และ duration ค้าง (5000ms)

ขั้นที่ 2: Update + อ่านค่า

    // PIR_Update() ต้องเรียกใน loop เพื่อตรวจสอบ state
    while(1) {
        PIR_Update();

        if(motionFlag) {
            motionFlag = 0;
            USART_Println("Motion detected!");

            // อ่านระยะเวลาที่ motion ค้างอยู่
            uint32_t dur = PIR_GetMotionDuration();
            USART_Print("Duration: ");
            USART_PrintInt(dur);
            USART_Println(" ms");

            digitalWrite(PC1, HIGH);  // LED ON
        } else {
            digitalWrite(PC1, LOW);   // LED OFF
        }

        Delay_Ms(100);  // check every 100ms
    }
}

🔍 PIR_Update() ต้องเรียกใน loop — ตรวจสอบ state และเรียก callback — PIR_GetMotionDuration() อ่านเวลาที่ motion ค้าง — callback ถูกเรียกใน main loop ผ่าน Update()

ขั้นที่ 3: Loop + แสดงผล

    uint32_t lastMotion = 0;
    uint16_t motionCount = 0;

    while(1) {
        PIR_Update();

        if(motionFlag) {
            motionFlag = 0;
            motionCount++;
            lastMotion = Get_CurrentMs();

            USART_Print("Motion #");
            USART_PrintInt(motionCount);
            USART_Println(" detected!");

            digitalWrite(PC1, HIGH);
        }

        // Auto off LED after 5 seconds of no motion
        if(Get_CurrentMs() - lastMotion > 5000) {
            digitalWrite(PC1, LOW);
        }

        Delay_Ms(100);
    }
}

🔍 ใช้ callback flag + main loop pattern — motionFlag ถูก set ใน callback, clear ใน main loop — LED ติด 5 วิหลัง motion ล่าสุด — นับจำนวนครั้งที่ตรวจจับได้ — ปรับ delay pot ที่ PIR เพิ่มความ long-hold

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
Trigger ตลอดเวลา (false trigger)Sensitivity pot สูงเกิน, แหล่งความร้อนใกล้เคียงลด sensitivity, หัน PIR ออกจากหน้าต่าง/แอร์/ฮีตเตอร์
ไม่ trigger แม้มีคนเดินผ่านSensitivity pot ต่ำเกิน, lens สกปรกเพิ่ม sensitivity, ทำความสะอาด Fresnel lens
Callback ไม่ถูกเรียกลืมเรียก PIR_Update() ใน loopต้องเรียก PIR_Update() ทุกรอบ loop — ไลบรารีใช้ polling ภายใน
Output ค้าง HIGH ไม่ยอมกลับ LOWDelay pot ตั้งนานเกิน (300s)ลด delay time, หรือใช้ Repeatable mode (L) แทน H
PIR ใช้ไม่ได้กับสัตว์ขนาดเล็กสัตว์เล็ก → IR เปลี่ยนแปลงน้อย → ไม่ถึง thresholdใช้ PIR ความไวสูง หรือใช้ microwave sensor (RCWL-0516)

🛠️ สร้าง Product: ไฟเปิดอัตโนมัติเมื่อมีคนเดินผ่าน

PIR ตรวจจับ motion → Relay เปิดไฟ → ไฟดับหลังไม่มีคน 30 วิ

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "PIR.h"

volatile uint8_t motionFlag = 0;
void my_callback(void) { motionFlag = 1; }

int main(void)
{
    SystemCoreClockUpdate();
    Timer_Init();

    pinMode(PC1, PIN_MODE_OUTPUT);  // Relay (ไฟ)
    PIR_Init(PC0);
    PIR_SetCallback(my_callback);
    PIR_SetDebounce(100, 5000);

    Timer_t tmr_off;
    uint8_t lightOn = 0;

    while(1)
    {
        PIR_Update();

        if(motionFlag) {
            motionFlag = 0;

            if(!lightOn) {
                digitalWrite(PC1, HIGH);  // เปิดไฟ
                lightOn = 1;
            }
            Start_Timer(&tmr_off, 30000, 0);  // 30 วิ ไม่มีคน → ปิด
        }

        if(lightOn && Is_Timer_Expired(&tmr_off)) {
            digitalWrite(PC1, LOW);  // ปิดไฟ
            lightOn = 0;
        }

        Delay_Ms(100);
    }
}

✏️ ลองทำดู

  1. เปรียบเทียบ Repeatable (L) vs Non-Repeatable (H) mode — สังเกตพฤติกรรมตอนมีคนยืนอยู่กับที่ vs เดินผ่าน
  2. ทำคนนับ — PIR ตรวจจับเข้า/ออก — แสดงจำนวนคนในห้องบน OLED
  3. ใช้ PIR + BH1750 — เปิดไฟเฉพาะตอนมืด AND มีคน — ประหยัดไฟกว่าตรวจจับอย่างเดียว
