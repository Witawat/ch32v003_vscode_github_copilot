📝 เฉลย — บทที่ 10: Power Management (Sleep / Standby / AWU / PVD)

หมายเหตุ: SimpleHAL และ standard peripheral (ch32v00x_pwr.h/ch32v00x_rcc.h) ไม่มี BKP register wrapper — เก็บข้อมูลข้าม reset ใน Flash เท่านั้น (ดูบทที่ 11)

ข้อ 1: Standby + AWU 10 วินาที + reset counter (เก็บใน Flash)

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include "SimpleHAL.h"

// state ที่จะเก็บข้าม reset — ใช้ Flash เพราะไม่มี BKP register
typedef struct {
    uint32_t magic;
    uint32_t wake_count;
    uint16_t crc;
} PersistState;

int main(void) {
    SystemCoreClockUpdate();

    // โหลด state จาก flash ที่เก็บไว้
    PersistState st;
    uint8_t loaded = Flash_LoadConfig(&st, sizeof(st));

    if (PWR_WasStandbyWakeup()) {
        // ——— ตื่นจาก Standby (AWU ที่ตั้งไว้ 10ms → 10s) ———
        Timer_Init();
        USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
        pinMode(PC0, PIN_MODE_OUTPUT);
        PWR_ClearStandbyFlag();

        if (!loaded || st.magic != 0xBEEF1234) {
            st.magic = 0xBEEF1234;
            st.wake_count = 0;
        }
        st.wake_count++;

        USART_Print("=== Wake #");
        USART_PrintNum((int32_t)st.wake_count);
        USART_Print(" from STANDBY (AWU 10s) ===\r\n");

        // LED blink ตาม wake_count mod 10
        for (uint32_t i = 0; i < (st.wake_count % 10); i++) {
            digitalWrite(PC0, HIGH);
            Delay_Ms(100);
            digitalWrite(PC0, LOW);
            Delay_Ms(100);
        }

        // บันทึก counter กลับลง flash (ระวัง wear — ทน ~10,000-80,000 รอบ/page)
        Flash_SaveConfig(&st, sizeof(st));

        USART_Print("Wake source: AWU (auto wake-up 10s)\r\n");
        USART_Print("Sleeping 10s...\r\n");
        Delay_Ms(50);  // รอ USART ส่งเสร็จก่อนเข้า Standby
    } else {
        // ——— Power-on ครั้งแรก ———
        Timer_Init();
        USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
        USART_Print("=== COLD BOOT ===\r\n");
        USART_Print("Wake source: POWER-ON\r\n");
        st.magic = 0xBEEF1234;
        st.wake_count = 0;
        Flash_SaveConfig(&st, sizeof(st));
        Delay_Ms(50);
    }

    // ตั้ง AWU 10 วินาที → เข้า Standby (auto prescaler ให้)
    // 10 วินาทีน้อยกว่า max 30s → PWR_Standby(timeout_ms) จัดการให้เอง
    USART_Print("Entering Standby for 10s...\r\n");
    Delay_Ms(30);
    PWR_Standby(10000);

    // ไม่มีทางมาถึง — Standby = reset
    while (1);
}
🧠 แนวคิด: PWR_Standby(uint32_t timeout_ms) รับ timeout 1-30000ms (สูงสุด 30s) — 10 วินาทีใช้สั้นกว่า max → SimpleHAL auto-select prescaler+window ที่เหมาะสมให้ ไม่ต้องเลือกเอง — RAM สูญหาย! ทุกครั้งที่ตื่น = reset → main() เริ่มใหม่ — PWR_WasStandbyWakeup() แยก reset จาก standby vs power-on — ไม่มี BKP register ใน SimpleHAL → เก็บ counter ใน Flash ผ่าน Flash_LoadConfig()/Flash_SaveConfig() — ระวัง Flash wear: ทนได้ ~10,000-80,000 รอบ/page — ถ้า wake ทุกนาทีต้องใช้ wear leveling หรือ external EEPROM — ถ้าต้องการ AWU แม่น ใช้ PWR_AWU_CALC_WINDOW(PWR_AWU_PRESCALER_10240, 10000) + PWR_ConfigureAWU() + PWR_EnterStandbyMode() แทน

ข้อ 2: คำนวณอายุแบต CR2032 ด้วย PWR_CalculateBatteryLife (4 args) + LDO quiescent

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include "SimpleHAL.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
    ADC_SimpleInit();
    pinMode(PC0, PIN_MODE_OUTPUT);

    // ——— Scenario: CR2032 + LDO 3.3V (quiescent ~1µA) ———
    // CR2032 capacity ≈ 220mAh (typical)
    // MCU active 50ms @ 4mA ทุก 10s → active_time_percent ≈ 0.5%
    // MCU standby rest → ~5µA (AWU on) + LDO quiescent ~1µA = ~6µA

    const uint16_t battery_mah       = 220;   // CR2032 220mAh
    const uint8_t  active_time_pct   = 1;     // ประมาณ 1% (over-estimate เผื่อ)
    const uint16_t active_current_ma = 4;     // active 4mA @ 24MHz
    const uint16_t standby_current_ua = 6;   // 5µA AWU + 1µA LDO quiescent

    // คำนวณอายุแบตเป็นชั่วโมง — signature จริง 4 args
    uint32_t hours = PWR_CalculateBatteryLife(
        battery_mah, active_time_pct,
        active_current_ma, standby_current_ua);

    USART_Print("Battery (CR2032 ");
    USART_PrintNum(battery_mah);
    USART_Print("mAh) life = ");
    USART_PrintNum((int32_t)hours);
    USART_Print(" hours = ");
    USART_PrintNum((int32_t)(hours / 24));
    USART_Print(" days = ");
    USART_PrintNum((int32_t)(hours / 24 / 30));
    USART_Print(" months\r\n");

    // เทียบแบบ manual (ตรวจสูตรเดียวกัน):
    //   weighted_current_uA = active_pct * active_mA * 1000
    //                       + (100 - active_pct) * standby_uA
    //   = 1 * 4000 + 99 * 6 = 4000 + 594 = 4594 uA
    //   hours = (battery_mah * 1000) / weighted_current_uA
    //        = (220 * 1000) / 4594 ≈ 47.9 ชม.

    // เปรียบเทียบ 3 กรณี: ไม่ sleep / Sleep / Standby
    uint32_t h_nosleep = PWR_CalculateBatteryLife(220, 100, 4, 0);   // active ตลอด → ~55 ชม.
    uint32_t h_sleep    = PWR_CalculateBatteryLife(220, 10, 4, 2000); // sleep 90%, 2mA
    uint32_t h_standby  = PWR_CalculateBatteryLife(220, 1,  4, 6);     // standby 99%, 6µA

    USART_Print("No-sleep:  ");  USART_PrintNum((int32_t)h_nosleep);  USART_Print(" h\r\n");
    USART_Print("Sleep:     ");   USART_PrintNum((int32_t)h_sleep);    USART_Print(" h\r\n");
    USART_Print("Standby:   ");  USART_PrintNum((int32_t)h_standby);  USART_Print(" h\r\n");

    // ประมาณ standby current ด้วย API (PVD on/off × AWU on/off)
    uint32_t sb_awu  = PWR_EstimateStandbyCurrent(0, 1);  // AWU only ≈ 5µA
    uint32_t sb_pvd  = PWR_EstimateStandbyCurrent(1, 1);  // AWU+PVD  ≈ 10µA
    USART_Print("Est standby (AWU only): ");
    USART_PrintNum((int32_t)sb_awu); USART_Print(" uA\r\n");
    USART_Print("Est standby (AWU+PVD): ");
    USART_PrintNum((int32_t)sb_pvd); USART_Print(" uA\r\n");

    // วัดแบตจริงผ่าน ADC (VREFINT) → % แบต
    float vdd = ADC_GetVDD();
    float batt_pct = ADC_GetBatteryPercent(vdd, 2.0f, 3.0f);  // CR2032 range
    USART_Print("VDD = ");
    USART_PrintNum((int32_t)(vdd * 1000));
    USART_Print(" mV  Batt = ");
    USART_PrintNum((int32_t)batt_pct);
    USART_Print("%\r\n");

    while (1) {
        digitalWrite(PC0, !digitalRead(PC0));
        Delay_Ms(500);
    }
}
🧠 แนวคิด: PWR_CalculateBatteryLife(battery_mah, active_time_percent, active_current_ma, standby_current_ua) รับ 4 arguments และคืน ชั่วโมง — สูตรภายในคือค่าเฉลี่ยถ่วงน้ำหนัก: hours = (battery_mah × 1000) / (active_pct × active_mA × 1000 + (100 − active_pct) × standby_uA) — ไม่ใช่ signature 3 args (vdd, vmin, vmax) และไม่ใช่ % แบต — % แบตใช้ ADC_GetBatteryPercent() ของ SimpleADC แทน — อย่าลืม LDO quiescent current (~1-2µA) บวกเข้า standby_current_ua ไม่งั้นประเมินอายุเกินจริง — ในทางปฏิบัติ CR2032 self-discharge ~1-3% ต่อปี ทำให้อายุจริง ~1-2 ปี แม้ทฤษฎีจะคำนวณได้หลายปี — PWR_EstimateStandbyCurrent(pvd, awu) ประมาณกระแส standby ใช้เป็นค่าประมาณก่อนเขียน code จริง

ข้อ 3: PVD low-voltage detect + safe shutdown + Standby

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include "SimpleHAL.h"

typedef struct {
    uint32_t magic;
    uint32_t unsafe_count;   // นับครั้งที่ VDD ต่ำกว่า threshold
    uint32_t last_vdd_mv;
    uint16_t crc;
} PersistState;

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
    ADC_SimpleInit();
    pinMode(PC0, PIN_MODE_OUTPUT);

    // เปิด PVD ที่ 3.3V — ถ้า VDD ต่ำกว่า 3.3V flag จะติด
    PWR_EnablePVD(PWR_PVD_3V3);

    PersistState st;
    if (!Flash_LoadConfig(&st, sizeof(st)) || st.magic != 0xCAFE1234) {
        st.magic = 0xCAFE1234;
        st.unsafe_count = 0;
        st.last_vdd_mv = 0;
    }

    USART_Print("PVD Safe-Shutdown demo — monitoring VDD...\r\n");

    while (1) {
        float vdd = ADC_GetVDD();

        if (PWR_GetPVDStatus()) {
            // ——— LOW VOLTAGE DETECTED — safe shutdown ———
            USART_Print("!!! LOW VOLTAGE — VDD=");
            USART_PrintNum((int32_t)(vdd * 1000));
            USART_Print("mV < 3.3V threshold\r\n");

            // บันทึก state ลง flash ก่อน shutdown
            st.unsafe_count++;
            st.last_vdd_mv = (uint32_t)(vdd * 1000);
            Flash_SaveConfig(&st, sizeof(st));
            USART_Print("State saved to flash. Safe shutdown...\r\n");

            // กระพริบ LED เตือน 5 รอบ
            for (int i = 0; i < 5; i++) {
                digitalWrite(PC0, HIGH); Delay_Ms(80);
                digitalWrite(PC0, LOW);  Delay_Ms(80);
            }

            // เข้า Standby จนกว่าจะมีคนจ่ายไฟใหม่/กด RESET
            // ใช้ StandbyUntilInterrupt เพราะไม่ต้องการ wake อัตโนมัติ
            PWR_StandbyUntilInterrupt();
            // ตื่นแล้ว = reset → main() เริ่มใหม่ → เช็คว่ายัง low อยู่ไหม
        }

        // สถานะปกติ — กระพริบ LED ช้าๆ และวัด VDD ทุก 1s
        digitalWrite(PC0, !digitalRead(PC0));
        USART_Print("VDD = ");
        USART_PrintNum((int32_t)(vdd * 1000));
        USART_Print(" mV  OK\r\n");
        Delay_Ms(1000);
    }
}
🧠 แนวคิด: PWR_EnablePVD(level) เปิด voltage detector ที่ threshold 8 ระดับ (2.9V-4.4V) — PWR_GetPVDStatus() คืน 1 ถ้า VDD ต่ำกว่า threshold — ในตัวอย่างนี้ใช้ทำ safe shutdown: บันทึก state ลง Flash → กระพริบ LED เตือน → เข้า PWR_StandbyUntilInterrupt() (ปลุกด้วย EXTI หรือกด RESET เท่านั้น ไม่ใช่ AWU) เพื่อป้องกันการ wake อัตโนมัติซ้ำจนแบตหมด — เมื่อจ่ายไฟใหม่/กด RESET จะ reset → main() เริ่มใหม่ → เช็ค PWR_GetPVDStatus() อีกครั้ง ถ้า VDD ยังต่ำอยู่จะกลับเข้า Standby ทันที — สำคัญ: เช็ค PVD ใน loop เปิด PVD ที่ระดับที่เหมาะกับแบตของคุณ — สำหรับ Li-Ion 1-cell ใช้ PWR_PVD_3V3 หรือ PWR_PVD_3V5 — สำหรับ CR2032 ใช้ PWR_PVD_2V9 — PWR_EstimateStandbyCurrent(1, 0) ประมาณกระแส standby เมื่อเปิด PVD (อาจ ~10µA)
