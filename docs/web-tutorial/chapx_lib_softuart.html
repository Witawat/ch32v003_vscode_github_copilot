Software UART — Bit-Bang บน GPIO, สร้าง UART Port เพิ่ม

ระดับสูง

🧭 แผนภาพการทำงาน

flowchart TD
    A["SoftUART_Init(&uart, tx, rx, baud)"] --> B["SoftUART_WriteString(&uart, str)"]
    B --> C["Bit-bang TX: start→8bit→stop"]
    C --> D["SoftUART_Available(&uart)?"]
    D -->|"Yes"| E["SoftUART_Read(&uart)"]
    D -->|"No"| D

🔌 การต่อวงจร

              Software UART — GPIO ธรรมดา

    │ CH32V003                          Device     │
    │ PC0(TX)──────────RX               ┌─────────┤
    │ PC1(RX)──────────TX               │ e.g.    │

PC0 (TX)→DeviceRXMCU→อุปกรณ์

GND↔DeviceGNDกราวด์ร่วม

📚 หลักการทำงาน

Bit-Bang UART จำลองการส่ง-รับข้อมูลแบบ asynchronous serial โดยใช้ GPIO + delay แม่นยำ — TX: สร้าง start bit → 8 data bits → stop bit, RX: รอ falling edge → sample ที่กลางบิต

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Init + Setup

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "SoftUART.h"

SoftUART_Instance uart;

int main(void) {
    SystemCoreClockUpdate();
    USART_Init(USART1, BAUD_115200);  // debug

    SoftUART_Init(&uart, PC0, PC1, BAUD_9600);
    //             inst   TX   RX   baud
    USART_Println("SoftUART Ready @ 9600bps");

🔍 SoftUART_Init(&uart, tx, rx, baud) — instance-based, ใช้ GPIO 2 ขา, บอด ≤ 19200

ขั้นที่ 2: ส่งข้อมูล (TX)

    SoftUART_WriteString(&uart, "Hello from SoftUART!\r\n");
    SoftUART_WriteChar(&uart, 'A');
    SoftUART_WriteInt(&uart, 12345);

    // ส่ง buffer
    uint8_t buf[] = {0x01, 0x02, 0x03, 0x04};
    SoftUART_WriteBuf(&uart, buf, 4);

🔍 SoftUART_WriteString(&uart, str) — ส่ง string, WriteChar — ส่ง 1 ตัว, WriteBuf — ส่ง byte array

ขั้นที่ 3: รับข้อมูล (RX) + Loop

    char rx_buf[32]; uint8_t idx = 0;
    while(1) {
        if(SoftUART_Available(&uart)) {
            char c = SoftUART_Read(&uart);
            USART_Write(USART1, c);      // echo to debug
            rx_buf[idx++] = c;
            if(c == '\n' || idx >= 31) {
                rx_buf[idx] = '\0';
                USART_Print("SoftRX: ");
                USART_Println(rx_buf);
                idx = 0;
            }
        }
    }
}

🔍 SoftUART_Available(&uart) — เช็ค RX buffer แบบ non-blocking, SoftUART_Read(&uart) — อ่าน 1 ไบต์

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
ข้อมูลออกมาขยะบอดเรตไม่ตรงกันทั้งสองฝั่งใช้ค่ามาตรฐาน: 1200-19200
ข้อมูลเพี้ยนบางไบต์CPU ถูก interrupt ขัดจังหวะปิด interrupt ช่วง RX critical

🛠️ สร้าง Product: 3-Port Serial Hub

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "SoftUART.h"

int main(void)
{
    SystemCoreClockUpdate();
    USART_Init(USART1, BAUD_115200);   // Port 0: Debug
    USART_Init(USART2, BAUD_9600);     // Port 1: GPS

    SoftUART_Instance uart;
    SoftUART_Init(&uart, PC0, PC1, BAUD_9600); // Port 2: BT

    while(1)
    {
        if(USART_Available(USART2)) {
            USART_Write(USART1, USART_Read(USART2));
        }

        if(SoftUART_Available(&uart)) {
            USART_Write(USART1, SoftUART_Read(&uart));
        }

        if(USART_Available(USART1)) {
            char c = USART_Read(USART1);
            SoftUART_WriteChar(&uart, c);
        }
    }
}
