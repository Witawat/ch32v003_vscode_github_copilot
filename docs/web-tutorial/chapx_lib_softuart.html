Software UART — Bit-Bang บน GPIO, สร้าง UART Port เพิ่ม

PC0 (TX)→DeviceRXMCU ส่ง → อุปกรณ์ — TTL level 3.3V
PC1 (RX)←DeviceTXอุปกรณ์ ส่ง → MCU — TTL level 3.3V

รู้จักขา

  • TX: ส่งข้อมูลจาก MCU ไปอุปกรณ์ — ใช้ขา PC0 (GPIO ธรรมดา)
  • RX: รับข้อมูลจากอุปกรณ์เข้า MCU — ใช้ขา PC1 (GPIO ธรรมดา)
  • GND: กราวด์ร่วม — ต้องต่อทั้งสองฝั่ง

หมายเหตุ: SoftUART ไม่มีขา VCC — จ่ายไฟแยกให้อุปกรณ์ตาม spec ของมัน — ระดับแรงดัน TTL 3.3V (เช่น GPS NEO-6M, HC-05, เซนเซอร์ที่มี TX/RX TTL)

📚 หลักการทำงาน

Bit-Bang UART จำลองการส่ง-รับข้อมูลแบบ asynchronous serial โดยใช้ GPIO + delay แม่นยำ — TX: สร้าง start bit → 8 data bits → stop bit, RX: รอ falling edge แล้ว sample ที่กลางบิต

เนื่องจาก CH32V003 มี USART ฮาร์ดแวร์แค่ 1 ช่อง SoftUART จึงใช้สร้างพอร์ตอนุกรมเพิ่มบน GPIO ธรรมดา ใช้ได้กับอุปกรณ์ TTL ทุกชนิดที่มีขา TX/RX

⚠️ ข้อจำกัดที่ต้องรู้ (อ่านก่อนเขียน code)

⚠️ RX interrupt buffering เป็น known issue — ring buffer ใน instance ไม่ถูกเติมข้อมูลจริง (LIB_AUDIT #10) ทำให้ SoftUART_Available(&uart) คืน 0 เสมอ

→ ใช้ SoftUART_ReadByte(&uart, &data, timeout_ms) แบบ blocking-with-timeout เป็น pattern หลัก — เรียกแล้วรอจนกว่าจะได้ byte หรือหมดเวลา timeout

ข้อจำกัดเพิ่มเติมจาก LIMITATIONS.md:

  • Baud > 38400 → timing error สะสม ~8% ต่อ 10 bits → ใช้ baud ≤ 38400 (แนะนำ 9600–38400)
  • Baud = 0 → SoftUART_Init คืน SOFTUART_ERROR (ป้องกัน DIV/0)
  • TX/RX เป็น blocking — ปิด interrupt ชั่วคราวระหว่าง bit-bang

ห้ามใช้ SoftUART_Read() (ไม่มีใน header) หรือคาดหวังว่า Available() จะทำงานเหมือน ring buffer ปกติ

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Init + Setup

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "SoftUART.h"

SoftUART_Instance uart;

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();                                   // จำเป็น — SoftUART ใช้ Delay_Us
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);  // debug ทาง HW UART

    SoftUART_Status st = SoftUART_Init(&uart, PC0, PC1, 9600);
    //                                inst   TX   RX   baud
    if (st != SOFTUART_OK) {
        USART_Println("SoftUART Init FAILED!");
        while (1);
    }
    USART_Println("SoftUART Ready @ 9600bps");

🔍 SoftUART_Init(&uart, tx, rx, baud) — instance-based, ใช้ GPIO 2 ขา — คืน SoftUART_Status (SOFTUART_OK/ERROR) — ต้องเรียก Timer_Init() ก่อนเสมอเพราะใช้ Delay_Us

ขั้นที่ 2: ส่งข้อมูล (TX)

    // ส่ง string
    SoftUART_WriteString(&uart, "Hello from SoftUART!\r\n");

    // ส่ง 1 byte (เช่นส่ง byte คำสั่งไปเซนเซอร์)
    SoftUART_WriteByte(&uart, 0xA5);

    // ส่ง byte array
    uint8_t buf[] = {0x01, 0x02, 0x03, 0x04};
    SoftUART_Write(&uart, buf, 4);

    // ส่งแบบ formatted (เหมือน printf)
    for (uint16_t i = 0; i < 5; i++) {
        SoftUART_Printf(&uart, "count=%u\r\n", i);
    }

🔍 WriteString, WriteByte, Write(buf, len), Printf(fmt, ...) — ทุกฟังก์ชันส่ง handle &uart เป็น argument แรกเสมอ — TX blocking สั้นๆ ปิด interrupt ระหว่าง bit-bang

ขั้นที่ 3: รับข้อมูล (RX) — ใช้ ReadByte + timeout เป็น pattern หลัก

    USART_Println("Waiting for byte (timeout 5s)...");
    uint8_t data;
    SoftUART_Status rs = SoftUART_ReadByte(&uart, &data, 5000);
    //                                       inst  out   timeout_ms
    if (rs == SOFTUART_OK) {
        USART_Print("Got byte: 0x");
        USART_PrintHex(data);
        USART_Println("");
    } else if (rs == SOFTUART_TIMEOUT) {
        USART_Println("ReadByte timeout — ไม่มีข้อมูลเข้าใน 5 วินาที");
    } else {
        USART_Println("ReadByte error — parity/timing");
    }

    while (1) {
        // echo แบบ blocking-with-timeout 10ms ต่อ byte
        uint8_t c;
        if (SoftUART_ReadByte(&uart, &c, 10) == SOFTUART_OK) {
            SoftUART_WriteByte(&uart, c);   // echo กลับ
        }
    }
}

🔍 ไม่ใช้ SoftUART_Available() เพราะเป็น known issue (คืน 0 ตลอด) — ใช้ SoftUART_ReadByte(&uart, &data, timeout_ms) แทน: รอจนกว่าจะได้ byte หรือหมด timeout — ฟังก์ชันคืน SOFTUART_OK / SOFTUART_TIMEOUT / SOFTUART_ERROR / SOFTUART_PARITY

🛠️ Product: GPS NMEA Reader (อ่านบรรทัดทีละ byte)

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "SoftUART.h"

SoftUART_Instance gps_uart;

// อ่านบรรทัด NMEA หนึ่งบรรทัดด้วย ReadByte loop แบบ blocking-with-timeout
// คืนความยาวบรรทัด (รวม '\0') หรือ 0 ถ้า timeout
uint16_t gps_readline(char *buf, uint16_t max_len, uint32_t timeout_ms) {
    uint16_t idx = 0;
    uint32_t deadline = Timer_GetMs() + timeout_ms;

    while (idx < max_len - 1) {
        if (SoftUART_ReadByte(&gps_uart, (uint8_t *)&buf[idx], 50) == SOFTUART_OK) {
            if (buf[idx] == '\n') {           // จบบรรทัด
                buf[idx] = '\0';
                return idx + 1;
            }
            if (buf[idx] != '\r') idx++;
        }
        if (Timer_GetMs() >= deadline) {       // รวม timeout ทั้งบรรทัด
            buf[idx] = '\0';
            return 0;
        }
    }
    buf[max_len - 1] = '\0';
    return idx;
}

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);

    // GPS NEO-6M default 9600  baud — TX ของ GPS ต่อ RX ของ MCU (PC1)
    SoftUART_Init(&gps_uart, PC0, PC1, 9600);
    USART_Println("GPS NMEA reader ready @ 9600bps");

    char line[96];
    while (1) {
        if (gps_readline(line, sizeof(line), 2000)) {
            // สนใจเฉพาะ $GPGGA หรือ $GPRMC
            if (strncmp(line, "$GPGGA", 6) == 0 || strncmp(line, "$GPRMC", 6) == 0) {
                USART_Println(line);
            }
        }
    }
}

🔍 อ่านทีละ byte ด้วย SoftUART_ReadByte(&gps_uart, &c, 50) (timeout 50ms ต่อ byte) — สะสมใน buffer จนเจอ '\n' ถือว่าจบบรรทัด — มี deadline รวมทั้งบรรทัด 2 วินาที (เก็บเวลาเริ่มต้น — เมื่อไม่ได้ใช้ Available()) — เป็น pattern มาตรฐานเพราะ RX ring buffer เป็น known issue

✏️ ลองทำดู

  1. ทำ dual-port debug — HW UART คุยกับ PC, SoftUART ส่ง "AT\r\n" ไป ESP01 แล้วใช้ ReadByte loop อ่าน response ส่งต่อขึ้น PC terminal
  2. ทำ GPS data logger — อ่านบรรทัด NMEA $GPRMC ด้วย ReadByte loop, แยก field ด้วย strtok แล้วส่ง timestamp ผ่าน HW UART
  3. ทำ packet protocol กับ CRC-8 — ใช้ SoftUART_Write(buf, len) ส่งทั้ง packet, ฝั่งรับใช้ ReadByte เก็บใน buffer จนครบ len แล้วตรวจ CRC
