ต่อ I2C ปกติ — ADDR pin เลือก 4 address (GND=0x48, VDD=0x49, SDA=0x4A, SCL=0x4B) — ALRT pin (optional) ใช้ขัดจังหวะเมื่อค่าผ่าน threshold

📚 หลักการทำงาน

ADS1115 เป็น ADC แบบ Delta-Sigma ความละเอียด 16-bit สื่อสารผ่าน I2C มี 4 ช่องสัญญาณ Analog Input — วัดได้ทั้งแบบ Single-Ended (เทียบ GND) และ Differential (AIN0-AIN1, AIN0-AIN3, AIN1-AIN3, AIN2-AIN3)

PGA (Programmable Gain Amplifier) ขยายสัญญาณก่อนเข้า ADC — มี 6 ระดับ: ±6.144V (gain 1), ±4.096V (2), ±2.048V (4), ±1.024V (8), ±0.512V (16), ±0.256V (16) — เลือกให้ครอบคลุม range สัญญาณ จะได้ความละเอียดสูงสุด

Data Rate 8, 16, 32, 64, 128, 250, 475, 860 SPS — ยิ่งเร็ว noise ยิ่งมาก — ใช้ 128 SPS สำหรับงานทั่วไป

Comparator + Alert ตั้งค่า Threshold High/Low — เมื่อค่าผ่าน threshold ALRT pin จะเปลี่ยน state — ใช้ตรวจจับ over/under voltage ได้โดยไม่ต้อง polling

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Init ADS1115 + Config

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "ADS1115.h"

ADS1115_t adc;

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    I2C_Init(PC1, PC2, I2C_SPEED_STANDARD);
    USART_Init(115200);

    ADS1115_Init(&adc, 0x48);                   // addr=0x48 (ADDR=GND)
    ADS1115_SetGain(&adc, ADS1115_PGA_4096);    // ±4.096V
    ADS1115_SetDataRate(&adc, ADS1115_DR_128);  // 128 SPS
    USART_Print("ADS1115 Ready\r\n");

🔍 ADS1115_Init(dev, addr) — ตั้งค่าเริ่มต้นด้วย I2C address — SetGain() เลือก PGA gain ให้เหมาะกับช่วงแรงดัน — SetDataRate() กำหนด sampling speed

ขั้นที่ 2: อ่านค่า ADC ทั้ง 4 ช่อง

    for(int ch = 0; ch < 4; ch++) {
        int16_t raw = ADS1115_Read(&adc, ch);
        float voltage = raw * (4.096f / 32768.0f);
        USART_Print("CH"); USART_PrintDec(ch);
        USART_Print(": "); USART_PrintDec(raw);
        USART_Print(" → "); USART_PrintFloat(voltage, 4);
        USART_Print("V\r\n");
        Delay_Ms(10);
    }

🔍 ADS1115_Read(dev, channel) อ่าน raw value 16-bit signed — แปลงเป็นแรงดัน: V = raw × (FSR / 2^15) — FSR คือ PGA full-scale range — ค่า raw เป็น signed (+32767 ถึง -32768)

ขั้นที่ 3: Differential Mode + Comparator

    ADS1115_SetChannel(&adc, ADS1115_CH_DIFF_0_1);
    ADS1115_SetComparator(&adc, 1000, 2000, ADS1115_COMP_TRAD);
    // ALRT pin LOW เมื่อค่า > 2000 หรือ < 1000

    while(1) {
        int16_t diff = ADS1115_ReadDiff(&adc);
        float v = diff * (4.096f / 32768.0f);
        USART_Print("Diff A0-A1: ");
        USART_PrintFloat(v, 4);
        USART_Print("V\r\n");

        if(GPIO_Read(PD4) == 0) {
            USART_Print("ALERT!\r\n");
        }
        Delay_Ms(500);
    }
}

🔍 Differential Mode วัดผลต่างระหว่าง 2 ช่อง — SetComparator() ตั้ง threshold hi/lo — ALRT pin active LOW เมื่อผ่าน threshold — ไม่ต้อง polling เองตลอดเวลา

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
ค่าที่อ่านได้กระโดดไปมาData Rate สูงเกินไป / Noise จาก supplyใช้ 64-128 SPS, เพิ่ม 0.1µF cap ที่ input, ใช้ averaging
แรงดันที่คำนวณผิดเพี้ยนใช้ FSR ผิดจาก PGA gain ที่ตั้งไว้ตรวจสอบ PGA gain — FSR = 6.144/(gain) เช่น gain=4 → FSR=1.536V? ไม่ใช่! FSR=±4.096V
I2C address ไม่เจอADDR pin ต่อผิด0x48(GND), 0x49(VDD), 0x4A(SDA), 0x4B(SCL) — ใช้ I2C Scanner ตรวจสอบ
อ่าน channel ไม่ได้ค่าInput ลอย (floating)ทุก channel ที่ไม่ได้ใช้ควรต่อ GND — อย่าปล่อยลอย
Differential อ่านค่าแปลกInput เกิน common-mode rangeDifferential ต้องอยู่ในช่วง GND ถึง VDD — PGA=±6.144V แต่ common-mode ต้อง ≤ VDD
ALRT pin ไม่ทำงานComparator ไม่ได้ตั้งค่าหรือ mode ผิดเรียก SetComparator ก่อน — เลือก mode TRAD (traditional) หรือ WINDOW

🛠️ สร้าง Product: 4-Channel Data Logger

ADS1115 วัด 4 ช่อง + บันทึกลง W25Qxx Flash ทุก 1 วิ — อ่านข้อมูลภายหลังผ่าน USART

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "ADS1115.h"
#include "W25Qxx.h"

#define LOG_INTERVAL_MS 1000
#define MAX_LOGS 512

typedef struct { uint32_t timestamp; int16_t ch[4]; } LogEntry_t;

int main(void)
{
    SystemCoreClockUpdate();
    Timer_Init();
    I2C_Init(PC1, PC2, I2C_SPEED_STANDARD);
    SPI_Init(PC4, PC5, PC6, SPI_MODE0);
    USART_Init(115200);

    ADS1115_t adc;
    ADS1115_Init(&adc, 0x48);
    ADS1115_SetGain(&adc, ADS1115_PGA_4096);
    ADS1115_SetDataRate(&adc, ADS1115_DR_128);

    W25Qxx_Init();
    W25Qxx_EraseSector(0);
    USART_Print("Data Logger Ready\r\n");

    Timer_t tmr;
    Start_Timer(&tmr, LOG_INTERVAL_MS, 1);
    uint32_t addr = 0;
    uint16_t count = 0;

    while(1) {
        if(Is_Timer_Expired(&tmr) && count < MAX_LOGS) {
            LogEntry_t log;
            log.timestamp = Get_CurrentMs();
            for(int i = 0; i < 4; i++) {
                log.ch[i] = ADS1115_Read(&adc, i);
            }

            W25Qxx_WritePage(addr, (uint8_t*)&log, sizeof(LogEntry_t));
            addr += sizeof(LogEntry_t);
            count++;

            USART_Print("Logged #"); USART_PrintDec(count);
            USART_Print(" t="); USART_PrintDec(log.timestamp);
            for(int i = 0; i < 4; i++) {
                USART_Print(" CH"); USART_PrintDec(i);
                USART_Print("="); USART_PrintDec(log.ch[i]);
            }
            USART_Print("\r\n");
        }
    }
}

✏️ ลองทำดู

  1. วัด Load Cell แบบ Differential — ต่อ Wheatstone Bridge เข้า AIN0-AIN1 ใช้ PGA=±0.512V เพื่อความละเอียดสูงสุด
  2. Alert เมื่อแรงดันเกิน 3.0V — ใช้ Comparator + ALRT pin ขัดจังหวะ MCU ด้วย EXTI
  3. วัด Thermocouple Type-K — ใช้ Differential + Cold Junction Compensation ผ่าน CH2
  4. ทำ Oscilloscope อย่างง่าย — sample 860 SPS ต่อเนื่อง เก็บ buffer ใน RAM แล้วส่งผ่าน USART ทีเดียว
