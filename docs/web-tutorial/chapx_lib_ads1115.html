ต่อ I2C ปกติ — ADDR pin เลือก 4 address (GND=0x48, VDD=0x49, SDA=0x4A, SCL=0x4B)

MCU Pin→อุปกรณ์ขาหมายเหตุ
PC1↔SDA-สายข้อมูล I2C — ต้องมี R 4.7kΩ pull-up ไป 3.3V
PC2→SCL-สายนาฬิกา I2C — ต้องมี R 4.7kΩ pull-up ไป 3.3V
3.3V→VDD-ไฟเลี้ยง 3.3V
GND→GND-กราวด์ร่วม
GND→ADDR-ตั้งค่า I2C address เป็น 0x48
PD4→ALRT-ขาแจ้งเตือน (optional)

รู้จักขา

  • VCC: ไฟเลี้ยง 3.3V
  • GND: กราวด์
  • SDA (PC1): สายข้อมูล I2C
  • SCL (PC2): สายนาฬิกา I2C
  • ADDR: เลือก I2C address (GND=0x48, VDD=0x49, SDA=0x4A, SCL=0x4B)
  • ALRT (PD4): ขาแจ้งเตือน (optional)
  • pull-up 4.7kΩ: ต่อระหว่าง SDA→3.3V และ SCL→3.3V (โมดูลส่วนมากมีในตัว)

วิธีต่อทีละขั้น

  1. ต่อ VCC ของ ADS1115 ไป 3.3V
  2. ต่อ GND ของ ADS1115 ไป GND
  3. ต่อ SDA ของ ADS1115 ไป PC1 (ขา 13) ของ CH32V003
  4. ต่อ SCL ของ ADS1115 ไป PC2 (ขา 14) ของ CH32V003
  5. ต่อ ADDR ไป GND เพื่อใช้ address 0x48 (หรือ VDD=0x49, SDA=0x4A, SCL=0x4B)
  6. (optional) ต่อ ALRT ไป PD4 เพื่อรับสัญญาณแจ้งเตือน
  7. ตรวจสอบว่าโมดูลมี R pull-up 4.7kΩ ในตัวหรือไม่ — ถ้าไม่มี ต่อ R เพิ่มจาก SDA→3.3V และ SCL→3.3V

📚 หลักการทำงาน

ADS1115 เป็น ADC แบบ Delta-Sigma ความละเอียด 16-bit สื่อสารผ่าน I2C มี 4 ช่องสัญญาณ Analog Input — ADS1115_ReadVoltage() คืนค่าแรงดันเป็น float โดยตรง — PGA gain ตั้งด้วย ADS1115_SetGain()

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Init ADS1115

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "ADS1115.h"

ADS1115_Instance ads;

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    I2C_Init(PC1, PC2, I2C_SPEED_STANDARD);
    USART_Init(115200);

    ADS1115_Init(&ads, 0x48);
    ADS1115_SetGain(&ads, ADS1115_GAIN_1);  // ±4.096V
    USART_Print("ADS1115 Ready\r\n");

🔍 ADS1115_Init(&ads, addr) — ตั้งค่าเริ่มต้นด้วย I2C address — SetGain() เลือก PGA gain — ADS1115_GAIN_1 = ±4.096V

ขั้นที่ 2: อ่านค่า ADC ทั้ง 4 ช่อง

    for(int ch = 0; ch < 4; ch++) {
        float v = ADS1115_ReadVoltage(&ads, ch);
        USART_Print("CH"); USART_PrintDec(ch);
        USART_Print(": "); USART_PrintFloat(v, 4);
        USART_Print("V\r\n");
        Delay_Ms(10);
    }

🔍 ADS1115_ReadVoltage(&ads, ch) อ่านแรงดันเป็น float โดยตรง (V) — ไม่ต้องคำนวณ raw → voltage เอง

ขั้นที่ 3: Loop + Data Logger

    Timer_t tmr;
    Start_Timer(&tmr, 1000, 1);

    while(1) {
        if(Is_Timer_Expired(&tmr)) {
            for(int i = 0; i < 4; i++) {
                float v = ADS1115_ReadVoltage(&ads, i);
                USART_Print("CH"); USART_PrintDec(i);
                USART_Print("="); USART_PrintFloat(v, 4);
                USART_Print("V  ");
            }
            USART_Print("\r\n");
        }
    }
}

🔍 อ่าน 4 ช่องทุก 1 วินาที — ADS1115_ReadVoltage จัดการแปลง raw เป็นแรงดันให้ — สะดวกกว่าใช้ raw value

🛠️ สร้าง Product: 4-Channel Data Logger

ADS1115 วัด 4 ช่อง + บันทึกลง W25Qxx Flash ทุก 1 วิ

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "ADS1115.h"
#include "W25Qxx.h"

int main(void)
{
    SystemCoreClockUpdate();
    Timer_Init();
    I2C_Init(PC1, PC2, I2C_SPEED_STANDARD);
    SPI_Init(PC4, PC5, PC6, SPI_MODE0);
    USART_Init(115200);

    ADS1115_Instance ads;
    ADS1115_Init(&ads, 0x48);
    ADS1115_SetGain(&ads, ADS1115_GAIN_1);

    W25Qxx_Instance flash;
    W25Qxx_Init(&flash, PC4);
    W25Qxx_EraseSector(&flash, 0);

    Timer_t tmr;
    Start_Timer(&tmr, 1000, 1);

    while(1) {
        if(Is_Timer_Expired(&tmr)) {
            float values[4];
            for(int i = 0; i < 4; i++) {
                values[i] = ADS1115_ReadVoltage(&ads, i);
            }
            W25Qxx_Write(&flash, 0, (uint8_t*)values, sizeof(values));

            USART_Print("Logged: ");
            for(int i = 0; i < 4; i++) {
                USART_PrintFloat(values[i], 3);
                USART_Print(" ");
            }
            USART_Print("\r\n");
        }
    }
}

✏️ ลองทำดู

  1. วัด Load Cell — ใช้ ADS1115_ReadVoltage อ่านแรงดันจาก Wheatstone Bridge
  2. วัด Thermocouple — ใช้ ADS1115 อ่านแรงดันเล็กๆ จาก thermocouple
  3. ทำ Oscilloscope อย่างง่าย — sample ต่อเนื่อง เก็บ buffer ใน RAM
