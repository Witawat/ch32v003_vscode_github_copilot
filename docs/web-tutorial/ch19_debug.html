#define DEBUG_MODE  1

⚠️ ข้อควรระวัง: อย่า print ใน loop เร็ว (เช่นทุก 1ms) — USART baud 115200 = ~11.5KB/s — print เร็วเกิน buffer overflow → ข้อมูลหาย

📚 ระดับ 2: GPIO/LED Debug — ดีบักโดยไม่ใช้ USART

LED Blink Code — บอกสถานะ

void blink_code(uint8_t count) {
    for(int i = 0; i < count; i++) {
        digitalWrite(PC0, HIGH); Delay_Ms(200);
        digitalWrite(PC0, LOW);  Delay_Ms(200);
    }
    Delay_Ms(1000);  // pause between codes
}

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    pinMode(PC0, PIN_MODE_OUTPUT);

    // Startup blink: 1 = init เริ่ม, 2 = init สำเร็จ
    blink_code(1);  // กระพริบ 1 ครั้ง = กำลัง init

    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
    blink_code(2);  // กระพริบ 2 ครั้ง = USART ready

    // Error blink: 3 = sensor fail, 4 = flash fail
    if(!sensor_init()) { blink_code(3); while(1); }
    blink_code(5);  // 5 ครั้ง = ทุกอย่าง OK

    while(1) { /* main loop */ }
}

Toggle Pin — จับด้วย Logic Analyzer

#define DEBUG_PIN  PC2
pinMode(DEBUG_PIN, PIN_MODE_OUTPUT);

📚 ระดับ 3: ดีบักด้วยมัลติมิเตอร์

วัดอะไรวิธีวัดค่าที่ควรได้ถ้าผิด → ปัญหาคือ
VDDโหมด DCV — จิ้ม VDD กับ GND3.2-3.4Vแหล่งจ่ายไฟ, LDO, สายไฟ
GND ครบวงจรโหมด Continuity (🔊)มีเสียงสาย GND หลวม/ขาด
Pin output HIGHDCV — pin กับ GND~3.3VpinMode, MCU พัง
Pin output LOWDCV — pin กับ GND~0Vshort circuit
กระแส MCUDCA — อนุกรมกับ VDD3-6mA (active)short, pin overcurrent
กระแส SleepDCA — อนุกรมกับ VDD~0.5mAperipheral ไม่ปิด
กระแส StandbyDCA — อนุกรมกับ VDD~5µApin floating, IO ไม่ได้ตั้งค่า
R ต่อเนื่องΩ — คร่อมตัวต้านทาน~220Ω / ~10kΩR เสีย, ค่าผิด
สายไฟขาดContinuity — ปลายถึงปลายมีเสียงสายไฟหักใน
💡 มัลติมิเตอร์มือสอง — ลงทุน ~100-300 บาท — เป็นอุปกรณ์ดีบักที่คุ้มค่าที่สุด — ใช้ตรวจ hardware ได้แทบทุกปัญหา

📚 ระดับ 4: Logic Analyzer — มองเห็นสัญญาณ

อุปกรณ์ + ซอฟต์แวร์

    USB Logic Analyzer 8ch 24MHz (~$10)
    ┌────────┐
    │ USB ───┼── PC (PulseView / sigrok)
    │ CH0────┼── PC0 (GPIO)
    │ CH1────┼── PD5 (USART TX)
    │ CH2────┼── PC1 (I2C SDA)
    │ CH3────┼── PC4 (PWM)
    │ GND────┼── GND (MCU)
    └────────┘

    ใช้จับสัญญาณ:
    ├─ USART: Baud rate, data bits, parity
    ├─ I2C:   Start/Stop, Address, Data, ACK/NACK
    ├─ SPI:   CS, SCK, MOSI, MISO timing
    ├─ PWM:   Frequency, Duty Cycle
    └─ GPIO:  Pulse width, timing, interrupt latency

ตัวอย่าง: ตรวจสอบ I2C ว่า sensor ตอบไหม

// ต่อ Logic Analyzer CH1=SDA, CH2=SCL
I2C_SimpleInit(I2C_SPEED_100K, I2C_PINS_DEFAULT);

uint8_t data;
uint8_t result = I2C_Read(0x76, 0xD0, &data, 1);
// result=1 → sensor ตอบ → 🟢
// result=0 → sensor ไม่ตอบ → ดู Logic Analyzer:
//   - SDA stuck LOW? → bus hang → reset bus
//   - No Start condition? → I2C init fail
//   - Address wrong? → เช็ค datasheet (0x76 vs 0xEC)

// ส่วนที่ 1: Memory Configuration
Memory Configuration
  FLASH  0x00000000  0x00004000    // 16KB
  RAM    0x20000000  0x00000800    // 2KB

// ส่วนที่ 2: ตัวแปร Global — ใช้ RAM เท่าไหร่
.bss           0x20000000       0x10 main.o
.bss           0x20000010       0x20 SimpleUSART.o
//                 ↑ 0x10 = 16 bytes, 0x20 = 32 bytes

// ส่วนที่ 3: Stack (สำคัญมาก)
// Stack = RAM_END - _heap_start
// ถ้า Stack เหลือน้อย → array ใหญ่ในฟังก์ชัน → stack overflow!

วิธีเช็ค Stack เหลือ

// เติม RAM ด้วย pattern ตอนเริ่ม
extern uint32_t _heap_start;
void fill_stack_pattern(void) {
    uint32_t *p = (uint32_t*)0x20000000;
    uint32_t *end = &_heap_start;
    while(p < end) { *p++ = 0xDEADBEEF; }
}

// เช็คหลังรัน — pattern ถูกเขียนทับ = stack ใช้ถึงตรงนั้น
uint32_t stack_used(void) {
    uint32_t *p = (uint32_t*)0x20000000;
    while(*p == 0xDEADBEEF && (uint32_t)p < 0x20000800) p++;
    return 0x20000800 - (uint32_t)p;
}

🛠️ สร้าง Tool: System Health Monitor

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"

void print_system_info(void) {
    USART_Println("=== System Health ===");

    USART_Print("  SysClk: ");
    USART_PrintNum(SystemCoreClock);
    USART_Println(" Hz");

    USART_Print("  ChipID: ");
    USART_PrintHex(DBGMCU_GetCHIPID(), 8);

    USART_Print("  VDD: ");
    USART_PrintNum(ADC_GetVDD());
    USART_Println(" mV");

    USART_Print("  Reset: ");
    if(RCC_GetFlagStatus(RCC_FLAG_PINRST))  USART_Print("PIN ");
    if(RCC_GetFlagStatus(RCC_FLAG_PORRST))  USART_Print("POR ");
    if(RCC_GetFlagStatus(RCC_FLAG_SFTRST))  USART_Print("SFT ");
    if(RCC_GetFlagStatus(RCC_FLAG_IWDGRST)) USART_Print("IWDG ");
    if(RCC_GetFlagStatus(RCC_FLAG_WWDGRST)) USART_Print("WWDG ");
    if(RCC_GetFlagStatus(RCC_FLAG_LPWRRST)) USART_Print("LPWR ");
    USART_Println("");

    RCC_ClearFlag();  // เคลียร์ทุก flag หลังอ่าน
}

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
    ADC_SimpleInit();
    Delay_Ms(500);

    print_system_info();

    pinMode(PC0, PIN_MODE_OUTPUT);
    while(1) {
        digitalToggle(PC0);
        Delay_Ms(1000);
    }
}

✏️ ลองทำดู

  1. จงใจใส่ bug — ลบ SystemCoreClockUpdate() — สังเกตอาการ USART เพี้ยน → ใช้ Print Debug หาสาเหตุ
  2. ต่อ Logic Analyzer — จับสัญญาณ I2C ระหว่างอ่าน BMP280 — ดู addressing + ACK
  3. ใช้ blink_code() แทน USART — ดัดแปลงให้กระพริบบอก error code ต่าง ๆ ได้ 10 แบบ
  4. เช็ค Stack Usage — เติม pattern 0xDEADBEEF → รันโปรแกรม 10 นาที → ดูว่า pattern ถูกเขียนทับไปแค่ไหน
