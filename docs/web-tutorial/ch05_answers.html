📝 เฉลย — บทที่ 5: PWM

ข้อ 1: LED Fade 2 ดวงสลับทิศทาง

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include "SimpleHAL.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();

    // LED1 = PD2 (PWM1_CH1), LED2 = PA1 (PWM1_CH2)
    PWM_Init(PWM1_CH1, 1000);  // 1kHz
    PWM_Init(PWM1_CH2, 1000);

    int duty1 = 0, duty2 = 100;

    while (1) {
        PWM_SetDutyCycle(PWM1_CH1, duty1);   // ดวง1: 0→100
        PWM_SetDutyCycle(PWM1_CH2, duty2);   // ดวง2: 100→0 (สลับทิศ)

        Delay_Ms(15);

        duty1++;
        duty2--;

        if (duty1 > 100) { duty1 = 0; duty2 = 100; }
    }
}
🧠 แนวคิด: LED 2 ดวงใช้ PWM คนละ channel — PWM1_CH1 (PD2) เพิ่ม duty จาก 0→100 ส่วน PWM1_CH2 (PA1) ลดจาก 100→0 — พร้อมกันในทิศตรงข้าม — เมื่อ duty1 ถึง 100 ทั้งคู่รีเซ็ตกลับจุดเริ่ม — delay 15ms × 100 steps = 1.5 วิต่อรอบ

ข้อ 2: Servo กวาด 0-180° วนลูป

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include "SimpleHAL.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();

    // Servo = PD2 (PWM1_CH1) @50Hz
    PWM_Init(PWM1_CH1, 50);

    int angle = 0, dir = 1;

    while (1) {
        // แปลง 0-180° → duty 5%-10% @50Hz
        uint8_t duty = (uint8_t)(5 + (angle * 5 / 180));
        PWM_SetDutyCycle(PWM1_CH1, duty);

        Delay_Ms(20);

        angle += dir;
        if (angle >= 180) dir = -1;
        if (angle <= 0)   dir = 1;
    }
}
🧠 แนวคิด: Servo ใช้ PWM 50Hz — duty 5% (1ms) = 0°, 10% (2ms) = 180° — สูตร 5 + angle*5/180 map มุม 0-180° → duty 5-10% — delay 20ms × 180 steps = 3.6 วิต่อเที่ยว รวมขากลับ ~7.2 วิ — สำคัญ: ต้องใช้ไฟ 5V ภายนอกเลี้ยง Servo — GND ต่อรวมกับ MCU

ข้อ 3: ใช้ PWM_SetRemap เปลี่ยน pin PWM ไปขาอื่น

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include "SimpleHAL.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();

    // PWM1_CH2 ปกติใช้ PA1 — remap ไป PC3 แทน
    PWM_SetRemap(PWM1_CH2, PWM_PIN_PC3);
    PWM_Init(PWM1_CH2, 1000);

    int duty = 0, dir = 1;

    while (1) {
        PWM_SetDutyCycle(PWM1_CH2, duty);
        Delay_Ms(10);
        duty += dir;
        if (duty >= 100) dir = -1;
        if (duty <= 0)   dir = 1;
    }
}
🧠 แนวคิด: PWM_SetRemap(channel, alternate_pin) เปลี่ยน mapping ก่อน PWM_Init() — PWM1_CH2 ปกติ = PA1 แต่ remap ไป PC3 ได้ — ตัวเลือก remap: PWM_PIN_PD2, PWM_PIN_PA1, PWM_PIN_PC3, PWM_PIN_PC4 — การ remap มีประโยชน์เมื่อ pin เดิมถูกใช้โดยอุปกรณ์อื่นแล้ว
