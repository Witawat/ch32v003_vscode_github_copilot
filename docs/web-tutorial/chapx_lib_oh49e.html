OH49E เป็น Linear Hall Sensor แบบ Ratiometric — Output เป็น analog voltage แปรผันตามสนามแม่เหล็ก — ที่ 0 Gauss output ≈ ½VCC (2.5V@5V หรือ 1.65V@3.3V)

📚 หลักการทำงาน

OH49E เป็น Linear Hall Effect Sensor — ตรวจจับสนามแม่เหล็กและคืนค่าแรงดัน analog แบบ Ratiometric (แปรผันตาม VCC) — Sensitivity ประมาณ 1.4mV/Gauss — วัดได้ทั้งขั้ว North และ South — Output อยู่ระหว่าง 0V ถึง VCC

Ratiometric Output หมายความว่าแรงดัน output เป็นสัดส่วนกับ VCC — ที่ 0 Gauss output = VCC/2 — ถ้า VCC=3.3V → zero=1.65V — ถ้า VCC=5V → zero=2.5V — การใช้ ADC ต้อง calibrate zero ทุกครั้งที่เปลี่ยน VCC

การคำนวณ Gauss: อ่าน ADC → แปลงเป็นแรงดัน → ลบด้วย zero voltage → หารด้วย sensitivity — Gauss = (Vout - Vzero) / Sensitivity — ค่าบวก = South, ค่าลบ = North

การใช้งาน: วัดความแรงสนามแม่เหล็ก, ตรวจจับตำแหน่งแม่เหล็ก, Current Sensor (วัดสนามแม่เหล็กรอบสายไฟ), ตรวจจับการเปิด-ปิด (Proximity Switch)

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Init ADC + Calibrate Zero

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"

#define HALL_PIN PD2
#define HALL_CH ADC_CH0
#define SENSITIVITY 1.4f // mV/Gauss

float zero_voltage;

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    ADC_Init(HALL_PIN, HALL_CH);
    USART_Init(115200);

    // Calibrate zero — อ่าน 100 ครั้งเฉลี่ยตอนไม่มีแม่เหล็ก
    uint32_t sum = 0;
    for(int i = 0; i < 100; i++) {
        sum += ADC_Read(HALL_PIN);
        Delay_Ms(10);
    }
    zero_voltage = (sum / 100.0f) * (3.3f / 1024.0f);
    USART_Print("Zero: ");
    USART_PrintFloat(zero_voltage, 3);
    USART_Print("V\r\n");

🔍 Calibrate ด้วยการหาค่าเฉลี่ยตอนไม่มีแม่เหล็ก — ใช้ 100 samples — zero voltage = ค่ากลางที่ใช้เป็น reference — ต้อง calibrate ใหม่ถ้าเปลี่ยน VCC

ขั้นที่ 2: อ่านค่า ADC + คำนวณ Gauss

    while(1) {
        uint16_t adc = ADC_Read(HALL_PIN);
        float voltage = adc * (3.3f / 1024.0f);
        float diff = voltage - zero_voltage;
        float gauss = (diff * 1000.0f) / SENSITIVITY; // mV → Gauss

        USART_Print("ADC:"); USART_PrintDec(adc);
        USART_Print(" V:"); USART_PrintFloat(voltage, 3);
        USART_Print(" Gauss:"); USART_PrintFloat(gauss, 1);

        if(gauss > 10) {
            USART_Print(" SOUTH+");
        } else if(gauss < -10) {
            USART_Print(" NORTH-");
        } else {
            USART_Print(" ~ZERO");
        }
        USART_Print("\r\n");
        Delay_Ms(200);
    }
}

🔍 แปลง ADC → Voltage → Gauss — diff เป็น Volt, คูณ 1000 เป็น mV ก่อนหาร sensitivity — threshold ±10 Gauss กัน noise — South pole = +Gauss, North pole = -Gauss

ขั้นที่ 3: ตรวจจับแม่เหล็ก + LED แสดงขั้ว

    pinMode(PC0, PIN_MODE_OUTPUT); // LED South
    pinMode(PC1, PIN_MODE_OUTPUT); // LED North

    while(1) {
        uint16_t adc = ADC_Read(HALL_PIN);
        float voltage = adc * (3.3f / 1024.0f);
        float gauss = ((voltage - zero_voltage) * 1000.0f) / SENSITIVITY;

        if(gauss > 15) {
            digitalWrite(PC0, HIGH); // South detected
            digitalWrite(PC1, LOW);
        } else if(gauss < -15) {
            digitalWrite(PC0, LOW);
            digitalWrite(PC1, HIGH); // North detected
        } else {
            digitalWrite(PC0, LOW);
            digitalWrite(PC1, LOW);
        }
        Delay_Ms(50);
    }
}

🔍 ใช้ hysteresis ±15 Gauss — LED 2 ดวงแสดงขั้วแม่เหล็ก — South = PC0 ติด, North = PC1 ติด, ไม่มีแม่เหล็ก = ดับทั้งคู่

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
ค่า ADC ไม่เปลี่ยนเมื่อเอาแม่เหล็กเข้าใกล้แม่เหล็กอ่อนเกิน/อยู่ไกลเกินใช้แม่เหล็ก Neodymium — ให้ใกล้ผิวเซนเซอร์ที่สุด — sensitivity ~1.4mV/G
ค่า Gauss เพี้ยนหลังจากเปลี่ยน VCCRatiometric — zero shift ตาม VCCCalibrate ใหม่ทุกครั้งที่เปลี่ยน VCC — หรือใช้แรงดันอ้างอิงคงที่
ค่ากระพือแม้ไม่มีแม่เหล็กNoise จาก ADC + switching power supplyใช้ averaging 10-20 samples — ต่อ 0.1µF cap ที่ output
Output ติดอยู่ที่ 0V หรือ VCCแม่เหล็กแรงมากจนเซนเซอร์อิ่มตัวOH49E มี linear range ±1000G — ถ้าเกินจะ clip — ถอยแม่เหล็กออก
ทิศทางขั้วกลับกันติดตั้งเซนเซอร์กลับด้านOH49E ด้านที่มีตัวอักษรหันเข้าแม่เหล็ก South → output สูงขึ้น

🛠️ สร้าง Product: Magnetic Field Strength Meter

OH49E + OLED — แสดงค่า Gauss แบบ Real-Time + กราฟแท่ง — Calibrate อัตโนมัติตอนเริ่ม

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "OLED.h"

#define HALL_PIN PD2
#define HALL_CH ADC_CH0
#define SENSITIVITY 1.4f

int main(void)
{
    SystemCoreClockUpdate();
    Timer_Init();
    I2C_Init(PC1, PC2, I2C_SPEED_STANDARD);
    ADC_Init(HALL_PIN, HALL_CH);
    USART_Init(115200);

    OLED_t oled;
    OLED_Init(&oled, 0x3C, OLED_128x64);
    OLED_SetFont(&oled, FONT_8x16);

    float zero_v = HALL_Calibrate(HALL_PIN, 100);

    while(1)
    {
        uint16_t adc = ADC_Read(HALL_PIN);
        float voltage = adc * (3.3f / 1024.0f);
        float gauss = ((voltage - zero_v) * 1000.0f) / SENSITIVITY;

        OLED_Clear(&oled);
        OLED_SetCursor(&oled, 0, 0);
        OLED_PrintFloat(&oled, "Field:", gauss, 1);
        OLED_Print(&oled, " G");

        const char* pole = (gauss > 5) ? " SOUTH" :
                           (gauss < -5) ? " NORTH" : " --";
        OLED_SetCursor(&oled, 80, 0);
        OLED_Print(&oled, pole);

        // Bar graph
        int bar_w = (int)(fabs(gauss) * 0.3f);
        if(bar_w > 100) bar_w = 100;
        OLED_SetCursor(&oled, 0, 3);
        for(int i = 0; i < bar_w; i++) {
            OLED_Print(&oled, "|");
        }

        OLED_SetCursor(&oled, 0, 5);
        OLED_PrintFloat(&oled, "ADC:", (float)adc, 0);
        OLED_Print(&oled, " ");
        OLED_PrintFloat(&oled, "V:", voltage, 2);

        OLED_Display(&oled);
        Delay_Ms(100);
    }
}

✏️ ลองทำดู

  1. ทำ Current Sensor แบบไม่สัมผัส — พันสายไฟรอบ Toroid Core แล้ววาง OH49E ใน gap — calibrate เทียบกับ DMM
  2. ทำ RPM Sensor — ติดแม่เหล็กบนมอเตอร์ — นับจำนวนครั้งที่สนามแม่เหล็กเปลี่ยน — คำนวณ RPM
  3. ทำ Linear Position Sensor — ติดแม่เหล็กกับ slider — วัดระยะทางจากค่า Gauss ในแนว linear
  4. ทำ Magnetic Door Sensor — ตรวจจับการเปิด-ปิดประตูด้วยแม่เหล็ก — ส่งสถานะผ่าน USART
