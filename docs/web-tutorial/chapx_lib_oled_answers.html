📝 เฉลย — OLED Display

ข้อ 1: แสดงข้อความ 2 ภาษา (ไทย+อังกฤษ) สลับบรรทัด

#include "oled_i2c.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    OLED_Handle oled;
    OLED_Init(&oled, OLED_128x64, 0x3C);

    OLED_SetFont(FONT_16X16_THAI);
    OLED_Print(0, 0, "สวัสดี");
    OLED_SetFont(FONT_6X8);
    OLED_Print(60, 0, "Hello");
    OLED_SetFont(FONT_8X16);
    OLED_Print(0, 20, "อุณหภูมิ:");
    OLED_SetFont(FONT_6X8);
    OLED_PrintFloat(72, 20, 25.5f, 1);
    OLED_Print(0, 40, "Celsius");
    OLED_Update(&oled);

    while(1) {}
}
🧠 แนวคิด: เรียก OLED_SetFont เพื่อเปลี่ยนฟอนต์ระหว่างไทยกับอังกฤษก่อนพิมพ์แต่ละครั้ง — ฟอนต์ไทยเป็น 16x16 พิกเซล อังกฤษ 6x8 — ต้องคำนวณตำแหน่ง x ให้ไม่ทับกัน — ใช้ OLED_PrintFloat สำหรับตัวเลขทศนิยม — ถ้าต้องการสลับภาษาแบบไดนามิก ให้เก็บ state ของภาษาที่ใช้อยู่แล้วเรียก SetFont ก่อนพิมพ์เสมอ

ข้อ 2: วาดกราฟแท่งแสดงค่า ADC

#include "oled_i2c.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    OLED_Handle oled;
    OLED_Init(&oled, OLED_128x64, 0x3C);
    ADC_SimpleInit(ADC_CH0);

    while(1) {
        uint16_t adc = ADC_Read(ADC_CH0);
        uint8_t bar_height = (uint8_t)((uint32_t)adc * 48 / 4095);

        OLED_Clear();
        OLED_DrawRect(10, 5, 30, 50, 1);       // กรอบนอก
        OLED_DrawRect(11, 5 + 50 - bar_height,
                      28, bar_height, 1);        // แท่งข้างใน
        OLED_SetFont(FONT_6X8);
        OLED_Print(45, 20, "ADC:");
        OLED_PrintNum(45, 30, adc);
        OLED_Update(&oled);

        Delay_Ms(100);
    }
}
🧠 แนวคิด: อ่านค่า ADC 0-4095 แล้ว map เป็นความสูงแท่ง 0-48 พิกเซล — OLED_DrawRect วาดกรอบนอกก่อน จากนั้นวาด filled rect ข้างในโดยคำนวณ y จากด้านล่างขึ้นบน — (5 + 50 - bar_height) ทำให้แท่งโตจากล่างขึ้นบน — ถ้าต้องการกราฟหลายแท่ง (เช่น ADC หลายช่อง) ให้เพิ่ม loop และเลื่อน x ไปทีละ 35 พิกเซล

ข้อ 3: Animation ลูกบอลกระเด้ง

#include "oled_i2c.h"

#define BALL_R  3

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    OLED_Handle oled;
    OLED_Init(&oled, OLED_128x64, 0x3C);

    int16_t x = 32, y = 24;
    int16_t vx = 2, vy = 1;
    const uint8_t gravity = 1;

    while(1) {
        // ลบบอลเก่า
        OLED_DrawCircle(x, y, BALL_R, 0);

        // อัปเดตตำแหน่งด้วยฟิสิกส์
        vy += gravity;
        x += vx;
        y += vy;

        // ชนขอบ — กลับทิศ + ลดพลังงาน
        if(x - BALL_R <= 0)   { x = BALL_R;     vx = -vx * 8 / 10; }
        if(x + BALL_R >= 127) { x = 127 - BALL_R; vx = -vx * 8 / 10; }
        if(y + BALL_R >= 63)  { y = 63 - BALL_R;  vy = -vy * 8 / 10; }
        if(y - BALL_R <= 0)   { y = BALL_R;       vy = -vy * 8 / 10; }

        // ถ้าความเร็วน้อยมาก → หยุด
        if(abs(vx) < 1 && abs(vy) < 1 && y >= 60) { vx = 0; vy = 0; }

        // วาดบอลใหม่
        OLED_DrawCircle(x, y, BALL_R, 1);
        OLED_Update(&oled);
        Delay_Ms(20);
    }
}
🧠 แนวคิด: จำลองฟิสิกส์ง่าย ๆ — มีแรงโน้มถ่วงเพิ่ม vy ทุกเฟรม — เมื่อชนขอบจะกลับทิศและลดพลังงานลง 80% (*8/10) เพื่อให้ลูกบอลค่อย ๆ หยุด — เทคนิค "ลบก่อนวาด" คือวาดวงกลมสีดำที่ตำแหน่งเก่าแล้ววาดสีขาวที่ตำแหน่งใหม่ — ถ้าต้องการหลายลูกบอล ให้เก็บ state ใน array of struct แล้ว loop วาด
