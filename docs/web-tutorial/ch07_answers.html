📝 เฉลย — บทที่ 7: I2C

ข้อ 1: สแกน I2C bus แสดง address ทั้งหมด

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include "SimpleHAL.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
    I2C_SimpleInit(I2C_100KHZ, I2C_PINS_DEFAULT);

    USART_Print("=== I2C Scanner ===\r\n");

    uint8_t found = 0;

    for (uint8_t addr = 1; addr < 128; addr++) {
        if (I2C_IsDeviceReady(addr)) {
            USART_Print("Device at 0x");
            USART_PrintHex(addr, 1);
            USART_Print("\r\n");
            found++;
        }
    }

    USART_Print("Total devices: ");
    USART_PrintNum(found);
    USART_Print("\r\n=== Done ===\r\n");

    while (1);
}
🧠 แนวคิด: I2C_IsDeviceReady(addr) ส่ง START + address + ตรวจสอบ ACK — ถ้ามีอุปกรณ์ต่ออยู่จะตอบ ACK — address I2C อยู่ในช่วง 0x03-0x77 (1-127) — address 0x00-0x02 เป็น reserved — ต้องต่อ Pull-up 4.7kΩ ที่ SDA/SCL! — อุปกรณ์บางตัวมีหลาย address (เช่น BH1750: 0x23 หรือ 0x5C แล้วแต่ ADDR pin)

ข้อ 2: อ่านค่าแสง BH1750

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include "SimpleHAL.h"

#define BH1750_ADDR  0x23   // ADDR pin LOW = 0x23

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
    I2C_SimpleInit(I2C_100KHZ, I2C_PINS_DEFAULT);

    // เช็คว่า BH1750 ต่ออยู่
    if (!I2C_IsDeviceReady(BH1750_ADDR)) {
        USART_Print("BH1750 not found!\r\n");
        while (1);
    }

    // ตั้งโหมด Continuous High-Res (1 lux resolution)
    I2C_WriteReg(BH1750_ADDR, 0x10, 0x00);
    USART_Print("BH1750 Ready\r\n");

    while (1) {
        Delay_Ms(180);  // รอ measurement time

        uint8_t buf[2];
        I2C_Status st = I2C_ReadRegMulti(BH1750_ADDR, 0x00, buf, 2);

        if (st == I2C_OK) {
            uint16_t raw = ((uint16_t)buf[0] << 8) | buf[1];
            uint16_t lux = raw / 1.2;  // High-Res mode

            USART_Print("Light: ");
            USART_PrintNum(lux);
            USART_Print(" lux\r\n");
        } else {
            USART_Print("I2C Error!\r\n");
        }

        Delay_Ms(500);
    }
}
🧠 แนวคิด: BH1750 = เซ็นเซอร์วัดแสง I2C — I2C_WriteReg(0x23, 0x10, 0x00) สั่ง Continuous High-Res Mode — รอ 180ms ให้วัดเสร็จ — I2C_ReadRegMulti() อ่าน 2 bytes — รวมเป็น 16-bit แล้วหาร 1.2 เป็น lux — เช็ค I2C_Status ทุกครั้ง — address เปลี่ยนได้ตาม ADDR pin (0x23 หรือ 0x5C)

ข้อ 3: เปลี่ยนไปใช้ I2C_PINS_PARTIAL_REMAP (PD2/PD1)

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include "SimpleHAL.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);

    // PARTIAL_REMAP: SCL=PD2, SDA=PD1
    // ใช้ได้เฉพาะ TSSOP-20/SOP-16/QFN-20 (มี PD2) — SOP-8 ไม่รองรับ (I2C_SimpleInit no-op)
    I2C_SimpleInit(I2C_100KHZ, I2C_PINS_PARTIAL_REMAP);

    USART_Print("I2C Scanner (REMAP: PD2/PD1)\r\n");

    uint8_t found = 0;

    for (uint8_t addr = 1; addr < 128; addr++) {
        if (I2C_IsDeviceReady(addr)) {
            USART_Print("Found: 0x");
            USART_PrintHex(addr, 1);
            USART_Print("\r\n");
            found++;
        }
    }

    if (found == 0) {
        USART_Print("No devices found.\r\n");
        USART_Print("Check pull-up and wiring!\r\n");
    }

    USART_Print("Done.\r\n");
    while (1);
}
🧠 แนวคิด: I2C_PINS_PARTIAL_REMAP เปลี่ยน SCL=PD2, SDA=PD1 — รองรับเฉพาะ TSSOP-20/SOP-16/QFN-20 (มี PD2) — SOP-8 ไม่รองรับ เพราะไม่มี PD2 → I2C_SimpleInit() จะ no-op เงียบ (ดู SimpleI2C.c:67-68) — SOP-8 ใช้ I2C_PINS_DEFAULT (PC1/PC2) หรือใช้ Software I2C แทน — I2C_PINS_FULL_REMAP (PD0/PD1) ก็ไม่รองรับ SOP-8 และ SOP-16 เพราะไม่มี PD0 — ต้องเป็น TSSOP-20/QFN-20 เท่านั้น
