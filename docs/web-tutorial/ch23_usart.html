ex01_Hello_World — ส่งข้อความ Hello World

📖 แนวคิด

โปรแกรมแรกของ USART — ใช้ USART_SimpleInit กำหนด Baud Rate 115200 — ส่งข้อความ "Hello from CH32V003!" ผ่าน USART_Print — ต้องต่อ USB-to-Serial แบบ Cross-connect

🔌 วงจร

    CH32V003              USB-Serial (CH340G/CP2102/FT232)
    PD5 (TX) ------------> RX
    PD6 (RX) <------------ TX
    GND ------------------ GND

    - Cross-connect: MCU-TX → USB-RX, MCU-RX → USB-TX
    - Baud Rate 115200 ใน Serial Monitor
    - GND ต้องต่อร่วมกันเสมอ!

💻 โค้ด + อธิบายทีละบรรทัด

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include <SimpleHAL.h>

int main(void)
{
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);

    while (1)
    {
        USART_Print("Hello from CH32V003!\r\n");
        Delay_Ms(1000);
    }
}
  • USART_SimpleInit(baud, pinConfig): เริ่ม USART 1 ช็อต — ตั้ง Baud Rate, pin TX/RX, 8N1 — USART_PINS_DEFAULT = TX=PD5, RX=PD6
  • USART_Print(str): ส่ง string (\r\n = ขึ้นบรรทัดใหม่ + carriage return)
  • BAUD_115200: ความเร็ว 115200 bps — ต้องตรงกับ serial monitor ทุกประการ

⚠️ จุดควรระวัง

  • ลืม SystemCoreClockUpdate() → USART baud rate คำนวณผิด → ข้อความเพี้ยน
  • ต่อสายผิด: TX→TX, RX→RX = ไม่มีข้อมูล — ต้อง Cross-connect เสมอ
  • Baud Rate ใน serial monitor ต้องตรงกับ BAUD_115200

✅ ตรวจสอบก่อนรัน

  • ต่อสาย Cross-connect ถูกต้อง? MCU-TX → USB-RX, MCU-RX → USB-TX
  • GND ต่อร่วมกัน? (MCU GND ↔ USB-Serial GND)
  • Serial monitor ตั้ง Baud 115200, 8N1 หรือไม่

ex02_Print_Formatted — พิมพ์เลข ฐานสิบ ฐานสิบหก และไบต์

📖 แนวคิด

ใช้ USART_PrintNum แสดงเลขจำนวนเต็ม int32_t — USART_PrintHex แสดงเลขฐานสิบหก — USART_WriteByte ส่ง 1 ไบต์ — ใช้ \r\n ขึ้นบรรทัดใหม่

🔌 วงจร

    CH32V003              USB-Serial
    PD5 (TX) ------------> RX
    PD6 (RX) <------------ TX
    GND ------------------ GND
    (เหมือน ex01 — ต่อแบบ Cross-connect, Baud 115200)

💻 โค้ด

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include <SimpleHAL.h>

int main(void)
{
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);

    while (1)
    {
        USART_Print("Decimal: ");
        USART_PrintNum(12345);
        USART_Print("\r\n");

        USART_Print("Hex: 0x");
        USART_PrintHex(12345, 1);
        USART_Print("\r\n");

        USART_Print("Negative: ");
        USART_PrintNum(-999);
        USART_Print("\r\n");

        USART_Print("Byte: 0x");
        USART_WriteByte(0x55);
        USART_Print("\r\n");

        Delay_Ms(2000);
    }
}
  • USART_PrintNum(int32_t): พิมพ์จำนวนเต็มมีเครื่องหมาย — แสดงเลขลบได้
  • USART_PrintHex(value, showPrefix): แสดงเลขฐานสิบหก — showPrefix=1 ต่อ "0x" อัตโนมัติ (แต่ ex02 ใช้เอง)
  • USART_WriteByte(byte): ส่ง 1 ไบต์ดิบ ๆ — ไม่แปลงเป็นข้อความ — ดูบน serial เป็นอักขระตาม ASCII

⚠️ จุดควรระวัง

  • USART_PrintNum รับ int32_t — ส่ง uint32_t ที่มีค่ามากกว่า 2^31-1 → แสดงเป็นลบ
  • USART_PrintHex รับ uint32_t — ค่า 0-FFFF แสดงถูก แต่ 100000 ก็แสดงได้
  • USART_WriteByte(0x55) = 'U' ใน ASCII — ถ้าต้องการดูเลข ต้อง PrintHex แทน

ex03_Echo_Command — สั่งงาน LED ผ่าน Serial

📖 แนวคิด

ใช้ USART_Available ตรวจสอบข้อมูลขาเข้า — USART_Read อ่าน 1 ไบต์ — สะสมเป็นคำสั่งจนเจอ \r\n — ใช้ strcmp เปรียบเทียบ — สั่ง LED=ON / LED=OFF

🔌 วงจร

    CH32V003              USB-Serial
    PD5 (TX) ------------> RX
    PD6 (RX) <------------ TX
    GND ------------------ GND

    PC0 --- 220Ω ---|>|--- GND    (LED เอาต์พุต)

💻 โค้ด

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include <SimpleHAL.h>

int main(void)
{
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
    pinMode(PC0, PIN_MODE_OUTPUT);

    char cmd[16];
    int  idx = 0;
    char ch;

    while (1)
    {
        if (USART_Available() > 0)
        {
            ch = USART_Read();

            if ((ch == '\r') || (ch == '\n'))
            {
                cmd[idx] = '\0';

                if (strcmp(cmd, "LED=ON") == 0)
                {
                    digitalWrite(PC0, HIGH);
                    USART_Print("LED ON\r\n");
                }
                else if (strcmp(cmd, "LED=OFF") == 0)
                {
                    digitalWrite(PC0, LOW);
                    USART_Print("LED OFF\r\n");
                }
                else
                {
                    USART_Print("You said: ");
                    USART_Print(cmd);
                    USART_Print("\r\n");
                }
                idx = 0;
            }
            else
            {
                if (idx < 15) { cmd[idx] = ch; idx++; }
            }
        }
    }
}
  • USART_Available(): คืนจำนวนไบต์ที่รออ่านใน RX buffer — ใช้ตรวจสอบก่อน USART_Read
  • USART_Read(): อ่าน 1 ไบต์ (blocking!) — ต้อง Ensure Available() > 0 ก่อน
  • strcmp: เปรียบเทียบ string — case-sensitive — "LED=ON" กับ "led=on" ไม่ตรงกัน

⚠️ จุดควรระวัง

  • บัฟเฟอร์ cmd[16] — พิมพ์เกิน 15 ตัวอักษร → buffer overflow — ต้องเพิ่มขนาดหรือ truncate
  • Serial Monitor ต้องตั้ง Line Ending เป็น "\r\n" หรือ "Both NL & CR" — ไม่ตั้ง = ไม่เจอ delimiter
  • USART_Read blocking — ถ้าไม่ตรวจ Available ก่อน โปรแกรมค้างรอ

ex04_USART_Read_Bytes — อ่านข้อมูลตามจำนวนที่กำหนด

📖 แนวคิด

ใช้ USART_ReadBytes(buffer, length) อ่านข้อมูลตามจำนวนไบต์ที่ต้องการ — blocking จนกว่าจะครบ — แสดงผลเป็น hex ทีละไบต์ — ใช้ print "0"+PrintHex สำหรับ Padding

💻 โค้ด

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include <SimpleHAL.h>

int main(void)
{
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);

    uint8_t buffer[5];
    int i;

    while (1)
    {
        USART_Print("Send 5 bytes:\r\n");
        USART_ReadBytes(buffer, 5);

        USART_Print("Received: ");
        for (i = 0; i < 5; i++)
        {
            if (buffer[i] < 16) USART_Print("0");
            USART_PrintHex(buffer[i], 1);
            if (i < 4) USART_Print(" ");
        }
        USART_Print("\r\n");
    }
}
  • USART_ReadBytes(buf, len): blocking read รอจนครบ len ไบต์ — เหมาะกับโปรโตคอลที่รู้ขนาดแน่นอน
  • Padding ด้วย "0": buffer[i] < 16 → เพิ่ม "0" ข้างหน้าให้เป็น 2 หลัก (0F แทน F)
  • USART_PrintHex(byte, 1): แสดงค่า hex ไม่มี prefix "0x" — ex04 จัดการ prefix เอง

⚠️ จุดควรระวัง

  • USART_ReadBytes blocking จนกว่าครบ — ถ้าส่งไม่ครบ 5 ไบต์ โปรแกรมค้างตลอด
  • ควรผสม USART_Available + USART_Read ถ้าต้องการ non-blocking หรือ timeout
  • buffer[5] เล็กเกินไปสำหรับข้อมูลเยอะ — ต้องปรับขนาดตาม use case

ex05_DMA_USART_Transmit — ส่งข้อมูลแบบไม่รอ CPU

📖 แนวคิด

ใช้ DMA_USART_Send ส่งข้อมูลผ่าน DMA โดย CPU ไม่ต้องรอ — ข้อความต้องเป็น global/static array — DMA_CH2 → USART — CPU สามารถทำงานอื่นระหว่างส่ง

💻 โค้ด

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include <SimpleHAL.h>

static const char msg[] = "Hello from DMA!\r\n";

int main(void)
{
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);

    while (1)
    {
        DMA_USART_Send(DMA_CH2, msg, sizeof(msg) - 1);
        Delay_Ms(2000);
    }
}
  • DMA_USART_SEND(chan, data, len): เริ่ม DMA transfer — return ทันที — CPU ทำงานต่อได้
  • ข้อความต้องเป็น global/static: DMA อ่านจาก RAM โดยตรง — ถ้าเป็น local variable หายเมื่อฟังก์ชันจบ → ส่งข้อมูลผิด
  • sizeof(msg)-1: ไม่ส่ง null terminator — ส่งแค่ "Hello from DMA!\r\n" 16 ตัว

⚠️ จุดควรระวัง

  • DMA channel ชนกับผู้ใช้รายอื่น — CH32V003 มี DMA 1 ชุด แชร์ระหว่าง USART, SPI, ADC
  • DMA_USART_Send return ทันที — ต้องมั่นใจว่าข้อมูลไม่ถูกเขียนทับก่อนส่งเสร็จ
  • global array ต้องคงอยู่ตลอด — ถ้าเปลี่ยนค่าขณะ DMA กำลังส่ง → ส่งผิด

ex06_USART_Remap — เปลี่ยนขา TX/RX

📖 แนวคิด

CH32V003 มี 4 pin configs สำหรับ USART: DEFAULT (PD5/PD6), REMAP1 (PD0/PD1), REMAP2 (PD6/PD5 สลับ), FULL_REMAP (PD6/PD5 รวมบิต) — ใช้ test_pin_config เพื่อลองทุกแบบ

💻 โค้ด

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include "SimpleHAL.h"

static void test_pin_config(USART_PinConfig config, const char* name) {
    USART_SimpleInit(BAUD_115200, config);
    USART_Print("\r\n=== ");
    USART_Print(name);
    USART_Print(" ===\r\n");
    USART_Print("CH32V003 USART Remap Test - OK\r\n");
}

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    pinMode(PC0, PIN_MODE_OUTPUT);

    test_pin_config(USART_PINS_DEFAULT, "DEFAULT  TX=PD5 RX=PD6");
    Delay_Ms(500);

    test_pin_config(USART_PINS_REMAP2, "REMAP2   TX=PD6 RX=PD5");
    Delay_Ms(500);

#if CH32V003_HAS_PD0
    test_pin_config(USART_PINS_REMAP1, "REMAP1   TX=PD0 RX=PD1");
    Delay_Ms(500);
#endif

    test_pin_config(USART_PINS_FULL_REMAP, "FULL     TX=PD6 RX=PD5");
    Delay_Ms(500);

    USART_Print("\r\n\r\nAll remap tests complete!\r\n");

    while (1) {
        digitalToggle(PC0);
        Delay_Ms(1000);
    }
}
  • USART_PINS_DEFAULT: TX=PD5, RX=PD6 — ใช้ได้ทุกแพ็กเกจ
  • USART_PINS_REMAP2: TX=PD6, RX=PD5 — สลับขั้ว — ใช้ได้ทุกแพ็กเกจ
  • USART_PINS_REMAP1: TX=PD0, RX=PD1 — ต้องมี PD0 — มีเฉพาะ TSSOP-20/QFN-20 — ใช้ CH32V003_HAS_PD0 ตรวจสอบ

⚠️ จุดควรระวัง

  • REMAP1 ใช้ PD0 — SOP-8/SOP-16 ไม่มี PD0 → compile error (#error) ถ้าไม่ใช้ #if guard
  • REMAP1 กับ FULL_REMAP ต่างกันที่ bit remap register — ดู datasheet
  • เปลี่ยน pin config แล้ว USART_SimpleInit ใหม่ — ไม่ต้อง power cycle

ex07_USART_Package_Aware — รองรับทุกแพ็กเกจ

📖 แนวคิด

ใช้ CH32V003_HAS_PD0, CH32V003_IS_SOP8, CH32V003_IS_SOP16 ปรับการทำงานตามแพ็กเกจ — SOP-8/SOP-16 ใช้ USART_PINS_DEFAULT หรือ REMAP2 เท่านั้น — TSSOP-20/QFN-20 ใช้ทุกแบบ

💻 โค้ด

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
// #define CH32V003_PACKAGE  PACKAGE_SOP16
// #define CH32V003_PACKAGE  PACKAGE_SOP8

#include "SimpleHAL.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();

#if CH32V003_HAS_PD0
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
    USART_Print("Package: TSSOP20/QFN20 — all pin configs available\r\n");
    USART_Print("USART_PINS_DEFAULT : TX=PD5 RX=PD6\r\n");
    USART_Print("USART_PINS_REMAP1  : TX=PD0 RX=PD1\r\n");
    USART_Print("USART_PINS_REMAP2  : TX=PD6 RX=PD5\r\n");
    USART_Print("USART_FULL_REMAP   : TX=PD6 RX=PD5\r\n");
#else
    USART_SimpleInit(BAUD_115200, USART_PINS_REMAP2);
    USART_Print("Package: SOP8/SOP16 — no PD0, using REMAP2\r\n");
    USART_Print("USART_PINS_DEFAULT : TX=PD5 RX=PD6\r\n");
    USART_Print("USART_PINS_REMAP2  : TX=PD6 RX=PD5\r\n");
    USART_Print("(REMAP1 requires PD0 — not available)\r\n");
#endif

#if CH32V003_IS_SOP8
    USART_Print("\r\nNote: SPI HW and I2C HW not available on SOP-8.\r\n");
    USART_Print("Use shiftOut/shiftIn and SimpleI2C_Soft instead.\r\n");
#elif CH32V003_IS_SOP16
    USART_Print("\r\nNote: SPI HW and I2C HW available with DEFAULT pins only.\r\n");
#endif

    pinMode(PC0, PIN_MODE_OUTPUT);
    while (1) {
        USART_Print(".");
        digitalToggle(PC0);
        Delay_Ms(1000);
    }
}
  • CH32V003_HAS_PD0: TRUE สำหรับ TSSOP-20/QFN-20 เท่านั้น — SOP-8/SOP-16 FALSE
  • CH32V003_IS_SOP8 / _IS_SOP16: macros ตรวจสอบแพ็กเกจ — ใช้ปรับข้อความและฟีเจอร์
  • SOP-8: ไม่มี SPI HW, I2C HW — ต้องใช้ Software bit-bang

⚠️ จุดควรระวัง

  • เปลี่ยนแค่ #define CH32V003_PACKAGE — rebuild — โค้ดปรับอัตโนมัติ ไม่ต้องแก้ไขส่วนอื่น
  • USART_PINS_REMAP1 ใน SOP-8 → #error ตอน compile — ต้องใช้ #if guard
  • ข้อความ "using REMAP2" ใน SOP-8/SOP-16 — REMAP2 ใช้ได้ทุกแพ็กเกจจริง
