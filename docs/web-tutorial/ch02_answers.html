📝 เฉลย — บทที่ 2: GPIO — Digital Input/Output

ข้อ 1: LED 3 ดวงวิ่ง Knight Rider (PC0→PC1→PC2→PC1→PC0)

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include "SimpleHAL.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();

    uint8_t leds[] = {PC0, PC1, PC2};
    pinModeMultiple(leds, PIN_MODE_OUTPUT);

    int idx = 0, dir = 1;

    while (1) {
        digitalWrite(PC0, LOW);
        digitalWrite(PC1, LOW);
        digitalWrite(PC2, LOW);
        digitalWrite(leds[idx], HIGH);

        Delay_Ms(150);

        idx += dir;
        if (idx >= 3) { idx = 1; dir = -1; }
        if (idx < 0)  { idx = 1; dir =  1; }
    }
}
🧠 แนวคิด: ใช้ pinModeMultiple() ตั้ง LED ทั้ง 3 ดวงเป็น output พร้อมกัน — array leds[] เก็บ pin — วน idx ไปข้างหน้า 0→1→2 แล้วกลับทิศ 2→1→0 — ก่อนเปลี่ยนดวงต้องดับทุกดวงด้วย digitalWrite(LOW) ก่อน แล้วติดเฉพาะดวงปัจจุบัน

ข้อ 2: ปุ่มกด PC1 เปลี่ยนทิศทางการวิ่ง (ใช้ interrupt)

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include "SimpleHAL.h"

uint8_t leds[] = {PC0, PC1, PC2};
volatile int idx = 0, dir = 1;

void btn_isr(void) {
    dir = -dir;  // กลับทิศ
}

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();

    pinModeMultiple(leds, PIN_MODE_OUTPUT);
    pinMode(PC1, PIN_MODE_INPUT_PULLUP);
    attachInterrupt(PC1, btn_isr, FALLING);

    while (1) {
        digitalWrite(PC0, LOW);
        digitalWrite(PC1, LOW);
        digitalWrite(PC2, LOW);
        digitalWrite(leds[idx], HIGH);

        Delay_Ms(200);

        idx += dir;
        if (idx >= 3) { idx = 1; dir = -dir; }
        if (idx < 0)  { idx = 1; dir = -dir; }
    }
}
🧠 แนวคิด: attachInterrupt(PC1, btn_isr, FALLING) — เมื่อกดปุ่ม (Active LOW → FALLING edge) จะเรียก btn_isr() ที่กลับค่า dir ทันที — ISR ต้องสั้นและเร็ว — ใช้ volatile กับตัวแปรที่แชร์ระหว่าง ISR กับ main loop

ข้อ 3: ใช้ pulseIn วัดระยะทางด้วย HC-SR04

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include "SimpleHAL.h"

#define TRIG  PC0
#define ECHO  PC1

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);

    pinMode(TRIG, PIN_MODE_OUTPUT);
    pinMode(ECHO, PIN_MODE_INPUT_PULLUP);

    while (1) {
        // ส่ง Trigger pulse 10us
        digitalWrite(TRIG, HIGH);
        Delay_Us(10);
        digitalWrite(TRIG, LOW);

        // วัดความกว้าง pulse ECHO (timeout 30ms = ~500cm)
        uint32_t pulse = pulseIn(ECHO, HIGH, 30000);

        if (pulse > 0) {
            uint32_t cm = pulse / 58;   // ความเร็วเสียง 343m/s
            USART_Print("Distance: ");
            USART_PrintNum(cm);
            USART_Print(" cm\r\n");
        } else {
            USART_Print("Out of range\r\n");
        }

        Delay_Ms(500);
    }
}
🧠 แนวคิด: HC-SR04 ทำงานโดยส่ง pulse 10μs ที่ TRIG → โมดูลส่งคลื่น ultrasonic → echo กลับที่ ECHO — pulseIn(ECHO, HIGH, 30000) วัดความกว้าง pulse หน่วย μs — แปลงเป็น cm ด้วย pulse/58 (343m/s ÷ 2 = 171.5m/s = 58μs/cm) — timeout 30ms (~500cm) ป้องกันการค้าง
