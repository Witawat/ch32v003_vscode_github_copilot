SHT3x — High-Accuracy Temp+Hum I2C

ระดับสูง

🧭 แผนภาพการทำงาน

flowchart TD
    A["SHT3x_Init(&sht, addr)"] --> B["SHT3x_Read(&sht, &t, &h)"]
    B --> C["float t (°C), float h (%)"]
    C --> B

🔌 การต่อวงจร

        SHT3x (I2C)
┌────────────────────────────────────┐
│ PC1 ──── SDA                       │
│ PC2 ──── SCL                       │
│ 3.3V ─── VCC                       │
│ GND  ─── GND                       │
│                                    │
│ ADDR ─── GND → addr = 0x44         │
│ ADDR ─── VCC → addr = 0x45         │
└────────────────────────────────────┘
MCU Pin→อุปกรณ์ขาหมายเหตุ
PC1↔SDA-สายข้อมูล I2C — ต้องมี R 4.7kΩ pull-up ไป 3.3V
PC2↔SCL-สายนาฬิกา I2C — ต้องมี R 4.7kΩ pull-up ไป 3.3V
3.3V→VCC-ไฟเลี้ยง 3.3V
GND→GND-กราวด์ร่วม; ADDR→GND=0x44, ADDR→VCC=0x45

📚 หลักการทำงาน

SHT3x เซนเซอร์อุณหภูมิ+ความชื้นความแม่นยำสูง — accuracy ±0.3°C / ±2%RH — range -40°C ถึง +125°C — มี heater ในตัวไล่ condensation

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Init I2C + SHT3x

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "SHT3x.h"

SHT3x_Instance sht;

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    I2C_SimpleInit(I2C_SPEED_100K, I2C_PINS_DEFAULT);

    SHT3x_Init(&sht, 0x44);    // ADDR=GND → 0x44
    USART_Println("SHT3x Ready!");

🔍 SHT3x_Init(&sht, addr) — instance-based, address 0x44 หรือ 0x45

ขั้นที่ 2: อ่านค่า

    float t, h;
    if(SHT3x_Read(&sht, &t, &h)) {
        USART_Print("T: "); USART_PrintFloat(t, 2);
        USART_Print("C  H: "); USART_PrintFloat(h, 2);
        USART_Println("%");
    }

🔍 SHT3x_Read(&sht, &t, &h) — single-shot measurement (~15ms) — เร็วกว่า AHT10 มาก — คืนค่า TRUE ถ้าสำเร็จ

ขั้นที่ 3: Loop + Heater

    Timer_t tmr;
    Start_Timer(&tmr, 1000, 1);

    float t, h;
    while(1) {
        if(Is_Timer_Expired(&tmr)) {
            if(SHT3x_Read(&sht, &t, &h)) {
                USART_Print("T:"); USART_PrintFloat(t, 2);
                USART_Print(" H:"); USART_PrintFloat(h, 2);
                USART_Println("%");
            }
        }
        IWDG_Feed();
    }
}

🔍 SHT3x_Read วัดครั้งเดียว ~15ms — ใช้ Non-Blocking Timer ควบคุม sampling rate

🛠️ สร้าง Product: Weather Station

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "SHT3x.h"
#include "oled_i2c.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    I2C_SimpleInit(I2C_SPEED_100K, I2C_PINS_DEFAULT);

    SHT3x_Instance sht;
    SHT3x_Init(&sht, 0x44);

    OLED_Handle oled;
    OLED_Init(&oled, OLED_128x64, 0x3C);

    Timer_t tmr;
    Start_Timer(&tmr, 1000, 1);

    float t, h;
    while(1) {
        if(Is_Timer_Expired(&tmr)) {
            if(SHT3x_Read(&sht, &t, &h)) {
                OLED_Clear(&oled);
                char buf[64];
                snprintf(buf, sizeof(buf), "T: %.1fC", t);
                OLED_DrawStringAlign(&oled, 0, 0, buf, ALIGN_LEFT);
                snprintf(buf, sizeof(buf), "H: %.1f%%", h);
                OLED_DrawStringAlign(&oled, 0, 20, buf, ALIGN_LEFT);
                OLED_Update(&oled);
            }
        }
    }
}
