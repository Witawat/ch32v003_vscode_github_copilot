📚 หลักการทำงาน

TCS34725 เป็น RGB color light-to-digital converter — photodiode 4 ช่อง: Red, Green, Blue, Clear — กรองแสงผ่าน IR filter

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Init + Setup

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "TCS34725.h"

TCS34725_Instance sensor;

int main(void) {
    SystemCoreClockUpdate();
    USART_Init(USART1, BAUD_115200);
    I2C_Init(I2C1, I2C_SPEED_STANDARD);

    TCS34725_Init(&sensor, TCS34725_GAIN_1X, TCS34725_INTEG_24MS);
    USART_Println("TCS34725 Ready!");

🔍 TCS34725_Init(&sensor, gain, integTime) — instance-based, gain: 1X/4X/16X/60X, integ: 2.4MS—700MS

ขั้นที่ 2: อ่านค่า RGB + Clear

    uint16_t r, g, b, c;
    if(TCS34725_ReadRGBC(&sensor, &r, &g, &b, &c)) {
        USART_Print("R:"); USART_PrintInt(r);
        USART_Print(" G:"); USART_PrintInt(g);
        USART_Print(" B:"); USART_PrintInt(b);
        USART_Print(" C:"); USART_PrintInt(c);
        USART_Println("");
    }

🔍 TCS34725_ReadRGBC(&sensor, &r, &g, &b, &c) — อ่านทั้ง 4 ช่องพร้อมกัน — คืน TRUE ถ้าสำเร็จ

ขั้นที่ 3: ตรวจจับสีเด่น

    while(1) {
        uint16_t r, g, b, c;
        if(!TCS34725_ReadRGBC(&sensor, &r, &g, &b, &c)) {
            USART_Println("Read error");
            Delay_Ms(100);
            continue;
        }

        if(c < 100)                     USART_Println(">>> DARK");
        else if(r > g && r > b)         USART_Println(">>> RED");
        else if(g > r && g > b)         USART_Println(">>> GREEN");
        else if(b > r && b > g)         USART_Println(">>> BLUE");

        Delay_Ms(500);
    }
}

🔍 ตรวจจับสีเด่นจากการเปรียบเทียบค่า RGB — c < 100 = มืดเกินไป

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
อ่านค่าได้ 0 ทุกช่องGain ต่ำเกินไป / LED ปิดเพิ่ม gain เป็น 16x หรือ 60x
ค่า Clear ถึง 65535 (saturated)แสงมากเกินไปลด gain เป็น 1x หรือลด integration time

🛠️ สร้าง Product: เครื่องคัดแยกสีวัตถุ

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "TCS34725.h"
#include "oled_i2c.h"

int main(void)
{
    SystemCoreClockUpdate();
    USART_Init(USART1, BAUD_115200);
    I2C_Init(I2C1, I2C_SPEED_STANDARD);

    OLED_Handle oled;
    OLED_Init(&oled, OLED_128x64, 0x3C);

    TCS34725_Instance sensor;
    TCS34725_Init(&sensor, TCS34725_GAIN_16X, TCS34725_INTEG_100MS);
    TCS34725_SetLED(&sensor, 1);

    while(1)
    {
        uint16_t r, g, b, c;
        if(!TCS34725_ReadRGBC(&sensor, &r, &g, &b, &c)) {
            Delay_Ms(50); continue;
        }

        OLED_Clear(&oled);

        const char *color;
        if(c < 150) color = "DARK";
        else if(r > g*1.2 && r > b*1.2) color = "RED";
        else if(g > r*1.2 && g > b*1.2) color = "GREEN";
        else if(b > r*1.2 && b > g*1.2) color = "BLUE";
        else color = "OTHER";

        char buf[64];
        snprintf(buf, sizeof(buf), "%s", color);
        OLED_DrawStringAlign(&oled, 0, 0, buf, ALIGN_LEFT);
        OLED_Update(&oled);
        Delay_Ms(300);
    }
}

✏️ ลองทำดู

  1. ทำ color identifier — แสดงชื่อสี 12 สีจากอัตราส่วน RGB
  2. ต่อ NeoPixel — ให้ LED strip เปลี่ยนสีตามวัตถุที่วางบนเซนเซอร์
  3. ทำ color sorter — ใช้ servo แยกวัตถุตามสีลงกล่อง 3 ช่อง
