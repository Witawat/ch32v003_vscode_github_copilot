    │ PC2─────────SCL          │ LED=IR   │

PC1─SDA, PC2─SCL — I2C address 0x29 (fix), มี white LED ในตัวช่วยส่องวัตถุ, มี IR filter บนชิป

📚 หลักการทำงาน

TCS34725 เป็น RGB color light-to-digital converter — ภายในมี photodiode 4 ช่อง: Red, Green, Blue, Clear — กรองแสงผ่าน IR filter ทำให้อ่านสีได้แม่นยำ ไม่ถูกรบกวนด้วย IR

Gain และ Integration Time:

  • Gain: 1x, 4x, 16x, 60x — ใช้ gain สูงในที่มืด, gain ต่ำในที่สว่าง
  • Integration Time: 2.4ms – 700ms — ตั้งเวลาสะสมแสง ยิ่งนานยิ่งไวแสง แต่ช้าลง

คำนวณอุณหภูมิสี (Color Temperature): ใช้สูตรจาก RGB → CIE XYZ → chromaticity → color temp ในหน่วย Kelvin (1000K–20000K)

Lux: คำนวณจากช่อง Clear + ค่าสัมประสิทธิ์ R/G/B — ได้ค่าความสว่างเช่นเดียวกับ BH1750

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Init + Setup

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "TCS34725.h"

int main(void) {
    SystemCoreClockUpdate();
    USART_Init(USART1, BAUD_115200);
    I2C_Init(I2C1, I2C_SPEED_STANDARD);

    // gain=16x, integration=100ms (เหมาะแสงในห้อง)
    TCS34725_Init(TCS34725_GAIN_16X, TCS34725_INT_100MS);

    // เปิด LED ในตัวเพื่อส่องวัตถุ
    TCS34725_SetLED(1);

🔍 TCS34725_Init(gain, integrationTime) — เลือก gain ตามสภาพแสง: 1x กลางแจ้ง, 16x ในห้อง, 60x มืด — integration time ยาว = แม่นยำกว่าแต่ช้า

ขั้นที่ 2: อ่าน/เขียนค่า

    uint16_t r, g, b, c;
    if(TCS34725_ReadRGB(&r, &g, &b, &c)) {
        USART_Print("R:"); USART_PrintInt(r);
        USART_Print(" G:"); USART_PrintInt(g);
        USART_Print(" B:"); USART_PrintInt(b);
        USART_Print(" C:"); USART_PrintInt(c);

        // คำนวณอัตราส่วนสี
        float ratio[3];
        TCS34725_GetColorRatio(r, g, b, ratio);

        // คำนวณอุณหภูมิสี (Kelvin)
        uint16_t colorTemp = TCS34725_CalcColorTemp(r, g, b);
        USART_Print(" CT:"); USART_PrintInt(colorTemp); USART_Println("K");
    }

🔍 TCS34725_ReadRGB() อ่านทั้ง 4 ช่องพร้อมกัน — c (Clear) ใช้คำนวณ lux และเช็คความอิ่มตัวของเซนเซอร์

ขั้นที่ 3: Loop + แสดงผล

    while(1) {
        uint16_t r, g, b, c;
        if(!TCS34725_ReadRGB(&r, &g, &b, &c)) {
            USART_Println("Read error");
            Delay_Ms(100);
            continue;
        }

        // ตรวจจับสีเด่น
        if(r > g && r > b && r > c/3)      USART_Println(">>> RED");
        else if(g > r && g > b)            USART_Println(">>> GREEN");
        else if(b > r && b > g)            USART_Println(">>> BLUE");
        else if(c < 100)                   USART_Println(">>> DARK");

        Delay_Ms(500);
    }
}

🔍 ตรวจจับสีเด่นจากการเปรียบเทียบค่า RGB — r > c/3 ป้องกัน false positive ตอนมืด — ใช้ค่านี้คัดแยกวัตถุตามสีได้

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
อ่านค่าได้ 0 ทุกช่องGain ต่ำเกินไป / LED ปิดเพิ่ม gain เป็น 16x หรือ 60x, เปิด LED ในตัว
ค่าสีเพี้ยน (ขาวทั้งที่ควรเป็นแดง)Integration time น้อยไป + gain สูงเพิ่ม integration time ลด noise, ลด gain
ค่า Clear channel ถึง 65535 (saturated)แสงมากเกินไปสำหรับ gain ปัจจุบันลด gain เป็น 1x หรือลด integration time
Color Temp เพี้ยนตอนมืดS/N ratio ต่ำ → IR noise ครอบงำเปิด LED, เพิ่ม integration time, หลีกเลี่ยงวัดสีตอน c < 200
อ่านค่าได้แต่ไม่ตรงกับสีจริงยังไม่ calibrate white balanceวัดค่าจากวัตถุขาว ใช้เป็น reference สำหรับปรับอัตราส่วน

🛠️ สร้าง Product: เครื่องคัดแยกสีวัตถุ

TCS34725 อ่านสี → แยกวัตถุแดง/เขียว/น้ำเงิน พร้อมแสดงบน OLED

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "TCS34725.h"
#include "OLED.h"

int main(void)
{
    SystemCoreClockUpdate();
    USART_Init(USART1, BAUD_115200);
    I2C_Init(I2C1, I2C_SPEED_STANDARD);

    OLED_t oled;
    OLED_Init(&oled, 0x3C, OLED_128x64);

    TCS34725_Init(TCS34725_GAIN_16X, TCS34725_INT_100MS);
    TCS34725_SetLED(1);

    while(1)
    {
        uint16_t r, g, b, c;
        if(!TCS34725_ReadRGB(&r, &g, &b, &c)) { Delay_Ms(50); continue; }

        OLED_Clear(&oled);
        OLED_SetCursor(&oled, 0, 0);

        const char *color;
        if(c < 150) color = "DARK";
        else if(r > g*1.2 && r > b*1.2) color = "RED";
        else if(g > r*1.2 && g > b*1.2) color = "GREEN";
        else if(b > r*1.2 && b > g*1.2) color = "BLUE";
        else color = "OTHER";

        OLED_Print(&oled, color);
        OLED_SetCursor(&oled, 0, 2);
        OLED_PrintFloat(&oled, "R:", (float)r, 0);
        OLED_PrintFloat(&oled, " G:", (float)g, 0);
        OLED_PrintFloat(&oled, " B:", (float)b, 0);
        OLED_Display(&oled);

        Delay_Ms(300);
    }
}

✏️ ลองทำดู

  1. ทำ color identifier — แสดงชื่อสี 12 สี (แดง, เขียว, น้ำเงิน, เหลือง, ม่วง, ฟ้า, ส้ม, ชมพู, ฯลฯ) จากอัตราส่วน RGB
  2. ต่อ NeoPixel — ให้ LED strip เปลี่ยนสีตามวัตถุที่วางบนเซนเซอร์ (copy สีวัตถุ)
  3. ทำ color sorter — ใช้ servo แยกวัตถุตามสีลงกล่อง 3 ช่อง
  4. วัด Color Temperature ของหลอดไฟ — เทียบ warm white vs cool white vs daylight
