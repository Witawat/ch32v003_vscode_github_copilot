📝 เฉลย — บทที่ 1: เริ่มต้นอย่างถูกต้อง

ข้อ 1: แก้ไขให้ LED กระพริบถี่ขึ้น (100ms)

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include "SimpleHAL.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    pinMode(PC0, PIN_MODE_OUTPUT);

    while (1) {
        digitalToggle(PC0);
        Delay_Ms(100);  // เปลี่ยนจาก 500 → 100ms
    }
}
🧠 แนวคิด: แค่เปลี่ยนตัวเลขใน Delay_Ms() — ค่าน้อย = กระพริบเร็ว ค่ามาก = กระพริบช้า digitalToggle สลับสถานะ HIGH↔LOW อัตโนมัติ

ข้อ 2: เพิ่ม LED 2 ดวงบน PC0 และ PC1 กระพริบสลับกัน

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include "SimpleHAL.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    pinMode(PC0, PIN_MODE_OUTPUT);
    pinMode(PC1, PIN_MODE_OUTPUT);

    while (1) {
        digitalWrite(PC0, HIGH);
        digitalWrite(PC1, LOW);
        Delay_Ms(300);

        digitalWrite(PC0, LOW);
        digitalWrite(PC1, HIGH);
        Delay_Ms(300);
    }
}
🧠 แนวคิด: ใช้ digitalWrite ควบคุมแต่ละ pin แยกกัน — เมื่อ PC0=HIGH ก็ตั้ง PC1=LOW (สลับกัน) วงจร: LED แต่ละดวงต่อกับขาของตัวเอง ผ่าน R 220Ω ลง GND

ข้อ 3: เปลี่ยนไปใช้ PACKAGE_SOP8 — โค้ดยังทำงานไหม?

#define CH32V003_PACKAGE  PACKAGE_SOP8   // เปลี่ยนแค่บรรทัดนี้!
#include "SimpleHAL.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();

    // SOP-8 มี PC1, PC2 — PC0 ไม่มี! ต้องเปลี่ยน pin
    pinMode(PC2, PIN_MODE_OUTPUT);  // PC2 มีบน SOP-8 (Pin 6)

    while (1) {
        digitalToggle(PC2);
        Delay_Ms(500);
    }
}
🧠 แนวคิด: SOP-8 มี 6 pins: PA1,PA2,PC1,PC2,PC4,PD6 — PC0 ไม่มี! ต้องเปลี่ยนเป็น pin ที่มีอยู่ ใช้ IS_VALID_PIN(pin) ตรวจสอบก่อน หรือดูตาราง pinout ในบทที่ 17
