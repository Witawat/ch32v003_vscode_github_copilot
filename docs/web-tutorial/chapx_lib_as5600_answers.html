📝 เฉลย — AS5600 Magnetic Rotary Encoder

ข้อ 1: Knob → Serial Angle Output

void as5600_knob_serial(void) {
    AS5600_Instance enc;
    AS5600_Init(&enc);

    uint16_t last_angle = 0xFFFF;

    while(1) {
        uint16_t angle = AS5600_ReadRawAngle(&enc);

        if(angle != last_angle) {
            float degrees = AS5600_ReadAngleDegrees(&enc);

            USART_Print("Angle: ");
            USART_PrintNum(angle);
            USART_Print(" raw (");
            USART_PrintFloat(degrees, 1);
            USART_Println(" deg)");

            // Optional: map 0–4095 → 0–100%
            uint8_t percent = (uint8_t)(angle * 100 / 4096);
            USART_Print("Percent: ");
            USART_PrintNum(percent);
            USART_Println("%");

            last_angle = angle;
        }
        Delay_Ms(10);
    }
}
🧠 แนวคิด: AS5600 ให้ค่า 0–4095 (12-bit) ต่อรอบ — แปลงเป็นองศา: angle × 360 / 4096 — ส่งเฉพาะเมื่อค่าเปลี่ยน ลด serial spam — แปลงเป็นเปอร์เซ็นต์สำหรับ knob UI — I2C address 0x36 fixed — ใช้แม่เหล็ก diametrically magnetized ให้ได้ linear output

ข้อ 2: Zero Position Setting

uint16_t as5600_zero_offset = 0;

void as5600_set_zero(AS5600_Instance *enc) {
    // อ่านตำแหน่งปัจจุบันมาเป็น zero
    uint16_t current = AS5600_ReadRawAngle(enc);

    // Burn ลง OTP (ทำครั้งเดียว!)
    // วิธีทางซอฟต์แวร์ — เก็บ offset ใน software ดีกว่า
    as5600_zero_offset = current;

    USART_Print("Zero set at raw angle: ");
    USART_PrintNum(as5600_zero_offset);
    USART_Println("");
}

uint16_t as5600_read_angle_zeroed(AS5600_Instance *enc) {
    uint16_t raw = AS5600_ReadRawAngle(enc);

    // Handle wraparound
    if(raw >= as5600_zero_offset) {
        return raw - as5600_zero_offset;
    } else {
        return 4096 - as5600_zero_offset + raw;
    }
}

float as5600_read_degrees(AS5600_Instance *enc) {
    return as5600_read_angle_zeroed(enc) * 360.0f / 4096.0f;
}
🧠 แนวคิด: เก็บ offset ในซอฟต์แวร์แทนการ burn OTP (แก้ไขภายหลังได้) — handle wraparound: ถ้า raw < offset แสดงว่าหมุนข้าม 0 → บวก 4096 ชดเชย — AS5600 มี zero position register ในฮาร์ดแวร์ด้วย แต่ burn OTP ทำครั้งเดียว ไม่ควรใช้ระหว่างพัฒนา

ข้อ 3: Magnet Detect Validation

typedef enum {
    MAGNET_OK,
    MAGNET_TOO_WEAK,
    MAGNET_TOO_STRONG,
    MAGNET_NOT_DETECTED
} magnet_status_t;

magnet_status_t as5600_check_magnet(AS5600_Instance *enc) {
    uint8_t status = AS5600_ReadStatus(enc);

    // Bit 3 = magnet detected, Bit 4 = magnet too weak, Bit 5 = magnet too strong
    if(!(status & 0x08)) return MAGNET_NOT_DETECTED;
    if(status & 0x10)    return MAGNET_TOO_WEAK;
    if(status & 0x20)    return MAGNET_TOO_STRONG;
    return MAGNET_OK;
}

void as5600_magnet_monitor(void) {
    AS5600_Instance enc;
    AS5600_Init(&enc);
    pinMode(PC0, PIN_MODE_OUTPUT);

    while(1) {
        magnet_status_t ms = as5600_check_magnet(&enc);

        switch(ms) {
            case MAGNET_OK:
                digitalWrite(PC0, HIGH);
                uint16_t angle = AS5600_ReadRawAngle(&enc);
                USART_Print("OK — Angle: ");
                USART_PrintNum(angle);
                USART_Println("");
                break;
            case MAGNET_TOO_WEAK:
                digitalWrite(PC0, HIGH); Delay_Ms(200);
                digitalWrite(PC0, LOW);  Delay_Ms(200);
                USART_Println("MAGNET TOO WEAK — bring closer");
                break;
            case MAGNET_TOO_STRONG:
                digitalWrite(PC0, HIGH); Delay_Ms(500);
                digitalWrite(PC0, LOW);  Delay_Ms(500);
                USART_Println("MAGNET TOO STRONG — move away");
                break;
            case MAGNET_NOT_DETECTED:
                digitalWrite(PC0, LOW);
                USART_Println("NO MAGNET");
                break;
        }
        Delay_Ms(300);
    }
}
🧠 แนวคิด: AS5600 status register บอกสถานะแม่เหล็ก — ระยะห่างที่เหมาะสม ~0.5–3mm ระหว่างแม่เหล็กกับ IC — แรงเกิน → saturation, แรงน้อยเกิน → noise — LED blink pattern บอกสถานะ: ติดค้าง=OK, กระพริบเร็ว=อ่อน, กระพริบช้า=แรง, ดับ=ไม่มีแม่เหล็ก
