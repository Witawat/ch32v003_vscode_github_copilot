📝 เฉลย — BH1750 เซนเซอร์วัดแสง

ข้อ 1: Lux Logger — บันทึกค่า lux ทุก N วินาทีลง Flash

#define LOG_INTERVAL_MS  5000
#define FLASH_SAVE_ADDR  0x08003C00

float last_lux = 0;
BH1750_Instance bh;

void bh1750_logger(void) {
    float lux = BH1750_ReadLux(&bh);
    if(lux < 0) {
        USART_Println("BH1750 read error");
        return;
    }

    USART_Print("Lux: ");
    USART_PrintFloat(lux, 1);
    USART_Println(" lx");

    // บันทึกลง Flash เฉพาะเมื่อค่าเปลี่ยนเกิน 5%
    if(fabsf(lux - last_lux) > last_lux * 0.05f) {
        FLASH_Unlock();
        FLASH_ErasePage(FLASH_SAVE_ADDR);
        FLASH_ProgramHalfWord(FLASH_SAVE_ADDR, (uint16_t)(lux * 10));
        FLASH_Lock();
        last_lux = lux;
    }

    Delay_Ms(LOG_INTERVAL_MS);
}
🧠 แนวคิด: ใช้ BH1750_ReadLux() อ่านค่าแล้ว USART_PrintFloat ออก serial — บันทึกเฉพาะเมื่อค่าเปลี่ยนเกิน threshold ป้องกัน Flash wear — แปลง float → uint16_t โดยคูณ 10 เก็บเป็น fixed-point ทศนิยม 1 ตำแหน่ง

ข้อ 2: Auto Night Light — PWM LED ตามค่า lux

#define LED_PIN       PA2
#define PWM_CHANNEL   1
#define LUX_DARK      50
#define LUX_BRIGHT    500

void bh1750_night_light(void) {
    BH1750_Instance bh;
    BH1750_Init(&bh, 0x23);
    PWM_SimpleInit(LED_PIN, PWM_CHANNEL, 1000, 0);

    while(1) {
        Delay_Ms(150);

        float lux = BH1750_ReadLux(&bh);
        if(lux < 0) continue;

        uint16_t duty;
        if(lux <= LUX_DARK) {
            duty = 1000;  // มืด → LED สว่างสุด
        } else if(lux >= LUX_BRIGHT) {
            duty = 0;     // สว่าง → LED ดับ
        } else {
            // map lux→duty แบบผกผัน: lux 50→1000, lux 500→0
            duty = 1000 - (uint16_t)((lux - LUX_DARK) * 1000 / (LUX_BRIGHT - LUX_DARK));
        }

        PWM_SetDuty(LED_PIN, PWM_CHANNEL, duty);
    }
}
🧠 แนวคิด: map เชิงเส้นแบบผกผัน — มืด LED สว่าง, สว่าง LED ดับ — ใช้ PWM_SimpleInit ตั้งความถี่ 1kHz แล้ว PWM_SetDuty อัปเดต duty cycle ตามค่า lux — clamp duty 0–1000 ป้องกัน overflow

ข้อ 3: OLED Lux Display — แสดงค่า lux บน OLED

void bh1750_oled_display(void) {
    OLED_Handle oled;
    OLED_Init(&oled, OLED_128x64, 0x3C);

    BH1750_Instance bh;
    BH1750_Init(&bh, 0x23);

    OLED_Clear(&oled);
    char buf[64];
    snprintf(buf, sizeof(buf), "Lux Meter");
    OLED_DrawStringAlign(&oled, 0, 0, buf, ALIGN_LEFT);
    OLED_Update(&oled);

    while(1) {
        Delay_Ms(200);

        float lux = BH1750_ReadLux(&bh);
        if(lux < 0) continue;

        snprintf(buf, sizeof(buf), "Lux: %.1f lx", lux);
        OLED_DrawStringAlign(&oled, 0, 10, buf, ALIGN_LEFT);

        // แสดงสถานะความสว่าง
        const char* status;
        if(lux < 50)       status = "Dark";
        else if(lux < 500) status = "Indoor";
        else if(lux < 5000)status = "Bright";
        else               status = "Outdoor";
        snprintf(buf, sizeof(buf), "%s      ", status);
        OLED_DrawStringAlign(&oled, 0, 20, buf, ALIGN_LEFT);
        OLED_Update(&oled);
    }
}
🧠 แนวคิด: ใช้ OLED_DrawStringAlign แสดงค่า lux — เพิ่มบรรทัดบอกสถานะ (Dark/Indoor/Bright/Outdoor) — snprintf จัดรูปแบบข้อความก่อนส่งให้ OLED_DrawStringAlign ระบุตำแหน่ง x,y — ช่องว่างท้ายสตริงลบตัวอักษรเก่าที่อาจค้าง — เรียก OLED_Update เพื่อส่ง buffer ลงจอ
