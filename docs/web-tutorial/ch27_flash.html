ex01_Read_Write_Basics — อ่าน/เขียน Flash พื้นฐาน

📖 แนวคิด

เรียนรู้การเขียนและอ่าน Flash ของ CH32V003 — ขนาดหน้า 64 bytes — ต้องลบ (erase) ก่อนเขียนทุกครั้ง — ใช้ Flash_Init เปิด clock, ErasePage ลบทีละหน้า, WriteByte เขียนทีละ byte, ReadWord อ่าน 32-bit

🧭 ผังการทำงาน

flowchart TD
    A["Flash_Init()"] --> B["Flash_ErasePage(0)"]
    B --> C["Flash_WriteByte(addr, 0x42)"]
    C --> D["Flash_WriteWord(addr, 0x12345678)"]
    D --> E["Flash_ReadByte(addr)"]
    E --> F["Flash_ReadWord(addr)"]
    F --> G["Print values + verify"]
    G --> H["while(1)"]

💻 โค้ด + อธิบายทีละบรรทัด

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include <SimpleHAL.h>

#define STORE_ADDR  0x08006000    // ใช้หน้า Flash ที่ 24 (ปลายๆ, ห่างจาก code)

int main(void)
{
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
    Flash_Init();                      // ★ เปิด clock + unlock Flash

    // ลบหน้าก่อนเขียน
    Flash_ErasePage(STORE_ADDR);       // ลบ 64 bytes ที่ address นี้ → ค่า = 0xFF

    // เขียนข้อมูล
    Flash_WriteByte(STORE_ADDR,     0x42);   // เขียน byte แรก
    Flash_WriteByte(STORE_ADDR + 1, 0xA5);   // เขียน byte ที่สอง

    // อ่านข้อมูล
    uint32_t word = Flash_ReadWord(STORE_ADDR);  // อ่าน 32-bit
    USART_Print("Read word: 0x");
    USART_PrintHex(word, 8);
    USART_Print("\r\n");
    USART_Print("Byte0: 0x");
    USART_PrintHex(Flash_ReadByte(STORE_ADDR), 2);
    USART_Print("\r\n");

    while(1) { IWDG_Feed(); }
}
  • Flash_Init(): เปิด clock ให้ Flash controller + unlock (เขียน KEY1/KEY2 ไปยัง FLASH_KEYR) — ต้องเรียกก่อนเขียน/ลบ
  • Flash_ErasePage(addr): ลบ 1 page (64 bytes) — addr ต้อง align กับ page boundary (0x0800XX00) — เขียน 0xFF ทุก byte
  • Flash_WriteByte(addr, data): เขียน 1 byte — addr ต้องอยู่ในพื้นที่ที่ถูกลบแล้ว — ลบก่อนเขียน!
  • Flash_ReadWord(addr): อ่าน 32-bit (4 bytes) — ไม่ต้อง unlock — อ่านได้ตลอดเวลา

⚠️ จุดควรระวัง

  • Flash CH32V003 = 16KB — 0x08000000 ถึง 0x08003FFF — ห้ามเขียนทับ code ที่กำลังรัน! (เลือก addr > 0x08002000 ปลอดภัย)
  • ลืม ErasePage ก่อนเขียน → เขียนไม่สำเร็จ (Flash บังคับลบก่อนเขียน — bit 1→0 ได้เท่านั้น)
  • Flash Write/Erase ใช้กระแสสูง (~5-10mA) — ถ้า VDD ตกขณะเขียนอาจ corrupt

✅ ตรวจสอบก่อนรัน

  • addr อยู่ในพื้นที่ Flash เปล่า? ใช้ linker script แยกพื้นที่ user data หรือเลือก addr หลัง code
  • Flash_Init() เรียกก่อน Write/Erase ทุกครั้ง?
  • Interrupt ปลอดภัย — SimpleFlash.c ปิด IRQ ให้อัตโนมัติรอบทุก erase/program แล้ว (หลัง commit 5293ae3) ไม่ต้อง disable interrupt เอง

ex02_Config_With_CRC — บันทึก/โหลด Config พร้อม CRC

📖 แนวคิด

ใช้ Flash_ErasePage + Flash_WriteByte + Flash_ReadWord เพื่อสร้างระบบ Save/Load Config — เพิ่ม CRC-8 ตรวจสอบความถูกต้องตอนโหลด — ใช้ Flash_SaveConfig บันทึก struct config และ Flash_LoadConfig อ่านกลับ — IsConfigValid เช็ค CRC

🧭 ผังการทำงาน

    B -->|"No"| C["Flash_SaveConfig(default)"]
    B -->|"Yes"| D["Flash_LoadConfig(&cfg)"]

    D --> E

💻 โค้ด

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include <SimpleHAL.h>

#define CFG_ADDR  0x08006000

typedef struct {
    uint8_t  brightness;     // 0-100
    uint8_t  mode;           // 0=auto, 1=manual
    uint16_t threshold;      // 0-4095
    uint8_t  crc;            // CRC-8 checksum (คำนวณจาก成员ข้างบน)
} __attribute__((packed)) DeviceConfig;

static DeviceConfig g_cfg;

// CRC-8 แบบง่าย (XOR)
static uint8_t calc_crc(const DeviceConfig *cfg) {
    uint8_t c = 0;
    const uint8_t *p = (const uint8_t*)cfg;
    for(uint8_t i = 0; i < sizeof(DeviceConfig) - 1; i++) {  // ไม่รวม crc เอง
        c ^= p[i];
    }
    return c;
}

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
    Flash_Init();

    // ตั้งค่าเริ่มต้น
    g_cfg.brightness = 75;
    g_cfg.mode       = 1;
    g_cfg.threshold  = 2048;
    g_cfg.crc        = calc_crc(&g_cfg);

    // save
    Flash_SaveConfig(CFG_ADDR, (uint8_t*)&g_cfg, sizeof(g_cfg));
    USART_Print("Config saved!\r\n");

    // load
    DeviceConfig loaded;
    Flash_LoadConfig(CFG_ADDR, (uint8_t*)&loaded, sizeof(loaded));

    if(IsConfigValid((uint8_t*)&loaded, sizeof(loaded))) {
        USART_Print("Config valid: brightness=");
        USART_PrintNum(loaded.brightness);
        USART_Print("\r\n");
    } else {
        USART_Print("Config CORRUPT! Using defaults.\r\n");
        loaded.brightness = 50;  // fallback
    }

    while(1) { IWDG_Feed(); }
}
  • Flash_SaveConfig(addr, data, len): ลบ page → เขียนทีละ byte — len ต้อง ≤ 64
  • Flash_LoadConfig(addr, data, len): อ่านตรง ๆ — ไม่ลบ — ใช้ uint8_t* อ่านทีละ byte
  • IsConfigValid(data, len): ตรวจสอบว่าไม่ใช่ 0xFF ทั้งหมด (uninitialized) + CRC match
  • CRC-8: XOR ทุก byte ยกเว้น byte สุดท้าย — ตรวจจับ data corruption พื้นฐาน

⚠️ จุดควรระวัง

  • Flash_ErasePage ใช้ ~20-40ms — ถ้ามี Watchdog ต้อง IWDG_Feed ก่อน/หลัง
  • CRC แบบ XOR กัน corruption เบื้องต้น — ถ้าต้องการ robustness ใช้ CRC-16/CRC-32 (ดู ex06)
  • Config struct ต้อง __attribute__((packed)) — กัน padding ที่ทำให้ CRC คำนวณผิด

ex03_String_And_Struct — เขียน/อ่าน String และ Struct

📖 แนวคิด

ใช้ WriteString และ ReadString สำหรับบันทึกข้อความ, WriteStruct สำหรับ struct ที่ซับซ้อนขึ้น — แสดงวิธีเก็บ device name + calibration data

🧭 ผังการทำงาน

    B --> C["Flash_WriteString(DATA_ADDR, str)"]
    C --> D["Flash_WriteStruct(DATA_ADDR+32, &cal)"]
    D --> E["Flash_ReadString(DATA_ADDR, buf)"]
    E --> F["Flash_ReadStruct(DATA_ADDR+32, &loaded)"]
    F --> G["Print device + calibration"]

💻 โค้ด

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include <SimpleHAL.h>

#define DATA_ADDR  0x08006000

typedef struct {
    float    gain;            // ค่า gain
    int16_t  offset;          // ค่า offset
    uint16_t serial_num;      // หมายเลขเครื่อง
} CalibrationData;

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
    Flash_Init();

    // เขียน string
    Flash_ErasePage(DATA_ADDR);
    Flash_WriteString(DATA_ADDR, "MyDevice v2.1");

    // เขียน struct ต่อจาก string
    CalibrationData cal = {1.05f, -12, 0x7B};
    Flash_WriteStruct(DATA_ADDR + 32, &cal, sizeof(cal));  // ใช้ offset 32

    // อ่านกลับ
    char buf[32];
    Flash_ReadString(DATA_ADDR, buf, sizeof(buf));
    USART_Print("Device: ");
    USART_Print(buf);
    USART_Print("\r\n");

    CalibrationData loaded;
    Flash_ReadStruct(DATA_ADDR + 32, &loaded, sizeof(loaded));
    USART_Print("Gain: ");
    USART_PrintFloat(loaded.gain, 3);
    USART_Print("  Offset: ");
    USART_PrintNum(loaded.offset);
    USART_Print("\r\n");

    while(1) { IWDG_Feed(); }
}
  • Flash_WriteString(addr, str): เขียน string พร้อม null-terminator — addr ต้องลบแล้ว
  • Flash_ReadString(addr, buf, maxLen): อ่านจนเจอ '\0' หรือครบ maxLen — ใส่ null-terminator
  • Flash_WriteStruct(addr, data, size): เขียน struct ใด ๆ — ใช้ memcpy ภายใน — size ต้องพอดี
  • Flash_ReadStruct(addr, data, size): อ่าน struct — ไม่ตรวจสอบความถูกต้อง (ใช้ CRC เองถ้าต้องการ)

⚠️ จุดควรระวัง

  • Flash เขียนทีละ byte — ถ้า struct มี padding → เขียน area ที่ยังไม่ลบ → ต้อง ErasePage ก่อนทุกครั้ง
  • String ยาวกว่า Flash page (64 bytes) → ต้องเขียนข้ามหลาย page — วางแผน layout
  • float ใน struct: CH32V003 รองรับ float ใน RAM แต่ Flash เก็บเป็น byte — อ่านกลับมาได้ปกติ

ex04_AutoErase_Operations — เขียนแบบลบอัตโนมัติ

📖 แนวคิด

ฟังก์ชัน WriteByteWithErase — รวม erase + write ในคำสั่งเดียว — เช็คว่า addr ถูก erase แล้วหรือยัง ถ้ายัง = ลบก่อนเขียน — ป้องกันการลืม ErasePage

🧭 ผังการทำงาน

flowchart TD
    A["Flash_Init()"] --> B["Flash_WriteByteWithErase(addr, 0xAA)"]
    B --> C["Flash_WriteByteWithErase(addr+1, 0xBB)"]
    C --> D["Flash_ReadByte(addr)"]
    D --> E["Flash_ReadByte(addr+1)"]
    E --> F["Print bytes"]
    F --> G["Flash_WriteByteWithErase(addr, 0xCC)"]
    G --> H["Print after rewrite"]
    H --> I["while(1)"]

💻 โค้ด

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include <SimpleHAL.h>

#define LOG_ADDR  0x08006000

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
    Flash_Init();

    // เขียนโดยไม่ต้อง ErasePage ล่วงหน้า
    Flash_WriteByteWithErase(LOG_ADDR,      0xAA);
    Flash_WriteByteWithErase(LOG_ADDR + 1,  0xBB);

    // อ่าน
    uint8_t b0 = Flash_ReadByte(LOG_ADDR);
    uint8_t b1 = Flash_ReadByte(LOG_ADDR + 1);
    USART_Print("Bytes: 0x");
    USART_PrintHex(b0, 2);
    USART_Print(" 0x");
    USART_PrintHex(b1, 2);
    USART_Print("\r\n");

    // เขียนซ้ำอีกครั้ง — จะ auto-erase อีก (ลบทั้ง page! ข้อมูล byte อื่นหาย)
    Flash_WriteByteWithErase(LOG_ADDR, 0xCC);
    uint8_t b2 = Flash_ReadByte(LOG_ADDR);
    USART_Print("After rewrite: 0x");
    USART_PrintHex(b2, 2);
    USART_Print("\r\n");  // b1 หายเพราะถูกลบทั้ง page

    while(1) { IWDG_Feed(); }
}
  • Flash_WriteByteWithErase(addr, data): ถ้า addr มีค่า != 0xFF → ErasePage ก่อน → แล้ว WriteByte
  • ข้อดี: สะดวก ไม่ต้องคิดเรื่อง erase — เหมาะกับ log ที่เขียน append ต่อ
  • ข้อเสีย: เขียนครั้งแรกโอเค — ครั้งที่ 2 ลบทั้ง page = ข้อมูลข้างเคียงหาย!

⚠️ จุดควรระวัง

  • Flash เขียน bit 1→0 ได้เท่านั้น — ถ้าต้องการเปลี่ยน 0→1 ต้อง ErasePage เสมอ
  • WriteByteWithErase ลบทั้ง page (64 bytes) — ถ้ามีข้อมูลสำคัญใน page เดียวกัน → หาย
  • สำหรับ data logger ที่เขียนต่อเนื่อง → ใช้หลาย page สลับกัน (ping-pong buffer)

ex05_Factory_Reset — กดปุ่ม 3 วินาทีลบ Flash ทั้งหมด

📖 แนวคิด

จำลอง factory reset — กดปุ่มค้าง 3 วินาที → ตรวจสอบด้วย ELAPSED_TIME → เรียก EraseAll ลบ Flash ทั้งหมดยกเว้น bootloader — ประโยค: "กดปุ่ม 3 วิ = reset กลับเป็นค่าโรงงาน"

🔌 วงจร

    PC1 -- Button -- GND    (ACTIVE_LOW, internal pull-up)
    PC0 ---R---|LED|---GND  (LED แสดงสถานะ factory reset)

🧭 ผังการทำงาน

flowchart TD
    A["Flash_Init()"] --> B["while(1)"]
    B --> C{"Button pressed?"}
    C -->|"No"| B
    C -->|"Yes"| D["Start timer"]
    D --> E{"Still held ≥ 3s?"}
    E -->|"No"| B
    E -->|"Yes"| F["Flash_EraseAll()"]
    F --> G["Print 'Factory Reset Done'"]
    G --> H["NVIC_SystemReset()"]

💻 โค้ด

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include <SimpleHAL.h>

#define BTN_PIN     PC1
#define LED_PIN     PC0
#define HOLD_MS     3000    // 3 วินาที

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
    Flash_Init();
    pinMode(BTN_PIN, PIN_MODE_INPUT_PULLUP);
    pinMode(LED_PIN, PIN_MODE_OUTPUT);

    USART_Print("Hold button 3s for Factory Reset...\r\n");

    while(1) {
        if(digitalRead(BTN_PIN) == LOW) {
            uint32_t start = Get_CurrentMs();
            // กดค้าง
            while(digitalRead(BTN_PIN) == LOW) {
                if(ELAPSED_TIME(start, Get_CurrentMs()) >= HOLD_MS) {
                    digitalWrite(LED_PIN, HIGH);
                    USART_Print("FACTORY RESET! Erasing all Flash...\r\n");

                    Flash_EraseAll();          // ลบ Flash ทั้งหมด (ยกเว้น bootloader)

                    USART_Print("Done! Restarting...\r\n");
                    Delay_Ms(100);
                    NVIC_SystemReset();        // รีสตาร์ท MCU
                }
                IWDG_Feed();
            }
            digitalWrite(LED_PIN, LOW);
        }
        IWDG_Feed();
    }
}
  • Flash_EraseAll(): ลบทุก page 0x08000000-0x08003FFF — ยกเว้น option bytes และ system memory
  • NVIC_SystemReset(): รีเซ็ต MCU โดยทาง software — หลังจากลบเสร็จให้เริ่มต้นใหม่
  • button hold: ใช้ polling + ELAPSED_TIME — ถ้าปล่อยก่อน 3 วิ → ยกเลิก ไม่ลบ

⚠️ จุดควรระวัง

  • EraseAll ใช้เวลา ~200-500ms — ต้อง disable interrupt + IWDG feed ก่อน
  • ถ้ามี bootloader ใน Flash — ตรวจสอบว่า EraseAll ลบเฉพาะ user area ไม่ลบ bootloader
  • ทดสอบ factory reset ด้วยความระมัดระวัง — ไม่มี undo!

ex06_CRC16_Checksum — ตรวจสอบความถูกต้องของ Flash

📖 แนวคิด

ใช้ CalculateCRC16 คำนวณ CRC-16 (CCITT) สำหรับข้อมูลใน Flash — ตรวจสอบความถูกต้องด้วย IsAddressValid — เหมาะกับ firmware integrity check หรือ data validation

🧭 ผังการทำงาน

flowchart TD
    A["Flash_Init()"] --> B["Flash_ErasePage(DATA_ADDR)"]
    B --> C["Write data bytes"]
    C --> D["CalculateCRC16(data, len)"]
    D --> E["IsAddressValid(addr, len)"]
    E --> F["Print CRC + validity"]
    F --> G["Write CRC to Flash"]
    G --> H["Verify CRC match"]
    H --> I["Print result"]
    I --> J["while(1)"]

💻 โค้ด

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include <SimpleHAL.h>

#define DATA_ADDR    0x08006000
#define DATA_LEN     32

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
    Flash_Init();

    // เตรียมข้อมูล
    Flash_ErasePage(DATA_ADDR);
    for(uint8_t i = 0; i < DATA_LEN; i++) {
        Flash_WriteByte(DATA_ADDR + i, i * 3 + 0x10);
    }

    // คำนวณ CRC-16
    uint16_t crc = CalculateCRC16((uint16_t*)DATA_ADDR, DATA_LEN / 2);
    USART_Print("CRC-16: 0x");
    USART_PrintHex(crc, 4);
    USART_Print("\r\n");

    // ตรวจสอบว่า address นี้ valid (ไม่ถูกลบ)
    if(IsAddressValid(DATA_ADDR, DATA_LEN)) {
        USART_Print("Address valid — data OK\r\n");
    } else {
        USART_Print("Address invalid — data corrupted!\r\n");
    }

    // เขียน CRC ต่อท้าย
    Flash_WriteByte(DATA_ADDR + DATA_LEN,     (crc >> 8) & 0xFF);
    Flash_WriteByte(DATA_ADDR + DATA_LEN + 1,  crc & 0xFF);

    // ตรวจสอบอีกครั้ง
    uint16_t crc2 = CalculateCRC16((uint16_t*)DATA_ADDR, DATA_LEN / 2);
    uint16_t stored = (Flash_ReadByte(DATA_ADDR + DATA_LEN) << 8) |
                       Flash_ReadByte(DATA_ADDR + DATA_LEN + 1);
    if(crc2 == stored) {
        USART_Print("CRC match! Data intact.\r\n");
    } else {
        USART_Print("CRC MISMATCH! Data corrupted.\r\n");
    }

    while(1) { IWDG_Feed(); }
}
  • CalculateCRC16(data, halfWords): คำนวณ CRC-16 CCITT (polynomial 0x1021) — รับข้อมูลเป็น uint16_t* — เหมาะกับการตรวจสอบ block ข้อมูล
  • IsAddressValid(addr, len): ตรวจสอบว่า addr อยู่ในช่วง Flash + ข้อมูลไม่เป็น 0xFF ทั้งหมด (unwritten)
  • CRC-16: ดีกว่า CRC-8 (XOR) — ตรวจจับ error ได้ถึง 99.998% — polynomial 0x1021 (CCITT)

⚠️ จุดควรระวัง

  • CalculateCRC16 อ่านทีละ 16-bit — ข้อมูลต้อง align 2 bytes
  • เก็บ CRC แยกต่างหากจากข้อมูล — ไม่รวม CRC ใน checksum
  • CRC-16 มี collision probability ~0.002% — สำหรับ safety-critical ใช้ CRC-32 หรือ SHA
