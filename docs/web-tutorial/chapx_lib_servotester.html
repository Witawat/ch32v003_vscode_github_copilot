รู้จักขา

  • PC0 (BTN1): ปุ่มเปลี่ยนโหมด Sweep / Center / Manual / Step
  • PC1 (BTN2): ปุ่มยืนยันหรือเลื่อนทีละขั้นในโหมด Step
  • PA1 (POT): อ่านค่าจาก potentiometer ใช้อะนาล็อกปรับมุม manual
  • PD2 (Signal): ส่งสัญญาณ PWM 50Hz ไปควบคุม servo
  • SDA (PC1): สายข้อมูล I2C ไปยัง OLED
  • SCL (PC2): สายนาฬิกา I2C ไปยัง OLED

วิธีต่อทีละขั้น

  1. ต่อ PC0 ไปขาหนึ่งปุ่ม BTN1 อีกขาไป GND — เลือกโหมด
  2. ต่อ PC1 ไปขาหนึ่งปุ่ม BTN2 อีกขาไป GND — ยืนยันหรือ step
  3. ต่อ PA1 ไปขากลางของ potentiometer — อ่านค่าอะนาล็อก
  4. ต่อ PD2 ไป Signal ของ Servo — ส่ง PWM 50Hz
  5. ต่อ PC1 ไป SDA ของ OLED และ PC2 ไป SCL ของ OLED — แสดงผล
  6. ต่อ GND ร่วมทุกอุปกรณ์

Station สำหรับทดสอบและ calibrate servo — มี 4 โหมด: Sweep อัตโนมัติ, Center หาจุดกึ่งกลาง, Manual ปรับด้วย Pot, Step ทีละ 10° — แสดงผลผ่าน OLED จอ 128x64

📚 หลักการทำงาน

ServoTester เป็น utility library สำหรับตรวจสอบว่า servo แต่ละตัวทำงานถูกต้องหรือไม่ — ใช้ก่อนติดตั้ง servo ลงใน project จริง

4 โหมดการทำงาน: (1) Sweep Mode — กวาด 0°→180°→0° อัตโนมัติ ใช้ตรวจสอบ smoothness; (2) Center Mode — สั่ง 90° เพื่อ calibrate horn ให้ตรงกลาง; (3) Manual Mode — ควบคุมด้วย potentiometer ปรับมุมละเอียด; (4) Step Mode — กดปุ่มเพิ่ม/ลดทีละ 10° วัด min/max angle

OLED Display แสดง: โหมดปัจจุบัน, มุม servo, duty cycle %, pulse width μs — ทำให้ calibrate ได้โดยไม่ต้องต่อ serial monitor

ใช้ Non-Blocking ทั้งปุ่มและ servo ทำให้ UI ตอบสนองทันที ไม่มีค้าง

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Init อุปกรณ์ทั้งหมด

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "ServoTester.h"
typedef enum {SWEEP, CENTER, MANUAL, STEP} Mode_t;
Mode_t mode = SWEEP;
uint8_t angle = 90;

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    ServoTester_Instance tester;
    ServoTester_Init(&tester, PWM1_CH4);
    OLED_Init_I2C(0x3C, PC2, PC1);
    ADC_Init_Single(PA1);
    pinMode(PC0, PIN_MODE_INPUT_PULLUP);  // BTN1 Mode
    pinMode(PC3, PIN_MODE_INPUT_PULLUP);  // BTN2 Step

🔍 ตั้งค่า OLED I²C, ADC สำหรับ Pot, ปุ่มแบบ INPUT_PULLUP (กด = LOW), และ PWM 50Hz บน PD2

ขั้นที่ 2: State Machine 4 โหมด

    void update_servo() {
        switch(mode) {
            case SWEEP: {
                static uint8_t dir = 1;
                angle += dir;
                if(angle >= 180) dir = -1;
                if(angle == 0) dir = 1;
                ServoTester_Sweep(&tester, angle, angle, 1);
                break;
            }
            case CENTER: ServoTester_FindCenter(&tester); angle = 90; break;
            case MANUAL: {
                uint16_t adc = ADC_Read(PA1);
                angle = (uint8_t)(adc * 180 / 4095);
                uint16_t pulse = 1000 + (uint16_t)((uint32_t)angle * 1000 / 180);
                ServoTester_SetPulse(&tester, pulse);
                break;
            }
            case STEP: break;
        }
    }

🔍 แต่ละโหมดคำนวณ angle ต่างกัน — Sweep ใช้ static dir กลับทิศ, Center ตั้ง 90° ตลอด, Manual อ่าน ADC เปลี่ยนเป็นมุม 0–180, Step รอปุ่มใน loop

ขั้นที่ 3: Main Loop + OLED Display

    Timer_t tmr_display;
    Start_Timer(&tmr_display, 100, 1);
    while(1) {
        if(!digitalRead(PC0)) {
            mode = (mode + 1) % 4;
            Delay_Ms(200);
        }
        if(mode == STEP && !digitalRead(PC3)) {
            angle = (angle + 10) % 190; if(angle > 180) angle = 0;
            uint16_t pulse = 1000 + (uint16_t)((uint32_t)angle * 1000 / 180);
            ServoTester_SetPulse(&tester, pulse);
            Delay_Ms(150);
        }
        update_servo();
        if(Is_Timer_Expired(&tmr_display)) {
            OLED_Clear();
            OLED_Printf(0,0,"Mode:%d Ang:%d", mode, angle);
            OLED_Update();
        }
        Delay_Ms(5);
    }
}

🔍 Main loop ตรวจสอบปุ่มเปลี่ยนโหมดและ step, เรียก update_servo() ทุก loop, OLED refresh ทุก 100ms ด้วย Non-Blocking Timer — Debounce ใช้ Delay_Ms สั้น ๆ

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
กดปุ่มแล้วโหมดไม่เปลี่ยนDebounce ไม่พอ — อ่านค่าซ้ำหลายครั้งเพิ่ม Delay_Ms(150–200) หลัง detect press
Sweep สะดุดกลางทางservo ถึง limit แล้วพยายามหมุนต่อตรวจสอบ boundary — if(angle >= 180) dir = -1
Manual Mode angle กระโดดADC noise จาก pot ราคาถูกใช้ Low-Pass Filter: angle = angle*0.9 + new*0.1 หรือเพิ่ม cap 100nF
OLED ไม่แสดงผลI²C address ผิด หรือลืมต่อ pull-upตรวจสอบ address (ปกติ 0x3C) และต่อ R pull-up 4.7kΩ ที่ SDA/SCL
Servo ไม่เคลื่อนที่ Step Modeไม่ได้กดปุ่มค้างนานพอใช้ auto-repeat — กดค้างขยับต่อเนื่องทุก 50ms

🛠️ สร้าง Product: Servo Diagnostic Station

เครื่องมือทดสอบ servo แบบพกพา — 4 โหมดครบ + OLED แสดงผล + เก็บค่า min/max angle + ทดสอบ deadband

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "ServoTester.h"

typedef enum {SWEEP=0, CENTER, MANUAL, STEP, SPEED_TEST} Mode_t;
Mode_t mode = SWEEP;
uint8_t angle = 90, step_size = 10;
uint8_t min_angle = 180, max_angle = 0;

int main(void) {
    SystemCoreClockUpdate(); Timer_Init();
    ServoTester_Instance tester;
    ServoTester_Init(&tester, PWM1_CH4);
    OLED_Init_I2C(0x3C, PC2, PC1);
    ADC_Init_Single(PA1);
    pinMode(PC0, PIN_MODE_INPUT_PULLUP);
    pinMode(PC3, PIN_MODE_INPUT_PULLUP);
    pinMode(PC4, PIN_MODE_INPUT_PULLUP);

    uint8_t btn_prev = 1, btn2_prev = 1, btn3_prev = 1;
    while(1) {
        uint8_t b1 = digitalRead(PC0); // Mode
        uint8_t b2 = digitalRead(PC3); // Step+
        uint8_t b3 = digitalRead(PC4); // Step-
        if(!b1 && btn_prev) { mode = (mode+1)%5; Delay_Ms(200); }
        if(!b2 && btn2_prev) { if(mode==STEP){angle=angle+step_size>180?180:angle+step_size;} Delay_Ms(150); }
        if(!b3 && btn3_prev) { if(mode==STEP){angle=angle=180)d=-1; if(angle==0)d=1; break; }
            case CENTER: ServoTester_FindCenter(&tester); angle=90; break;
            case MANUAL: { angle=(uint8_t)(ADC_Read(PA1)*180/4095);
                ServoTester_SetPulse(&tester, 1000+(uint16_t)((uint32_t)angle*1000/180)); break; }
            case SPEED_TEST: { static uint8_t phase=0; if(phase==0){angle=0;phase=1;} else{angle=180;phase=0;} break; }
        }
        uint16_t pulse = 1000 + (uint16_t)((uint32_t)angle * 1000 / 180);
        ServoTester_SetPulse(&tester, pulse);
        if(anglemax_angle)max_angle=angle;

        OLED_Clear();
        OLED_Printf(0,0,"M:%d A:%3d", mode, angle);
        OLED_Printf(0,2,"min:%d max:%d", min_angle, max_angle);
        OLED_Update();
        Delay_Ms(mode==SPEED_TEST?500:10);
    }
}

✏️ ลองทำดู

  1. เพิ่มโหมด Deadband Test — สั่ง angle=90 แล้วลองขยับทีละ 1° ดูว่า servo เริ่มขยับที่กี่องศา
  2. เพิ่มฟีเจอร์บันทึกค่า Calibration ลง Flash — ให้ station จำ min/max ของ servo แต่ละตัวได้หลังปิดเครื่อง
  3. ทำ Auto-Calibration — กดปุ่มเดียวให้ station สแกน min/max อัตโนมัติโดยใช้ current sensing
