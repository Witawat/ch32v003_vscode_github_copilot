บทที่ 9: DMA — ถ่ายโอนข้อมูลโดยไม่ใช้ CPU

ระดับสูง

🧭 แผนภาพการทำงาน

flowchart TD
    A["DMA_SimpleInit(channel, dst, src, size)"] --> B["DMA_Start(channel)"]
    B --> C["Transfer อัตโนมัติ (CPU ว่าง)"]
    C --> D{Transfer Complete?}
    D -->|ยังไม่เสร็จ| C
    D -->|เสร็จ| E["Callback / DMA_GetStatus()"]
    E --> F["ประมวลผลข้อมูล"]
    F --> B

🔌 การต่อวงจร

    DMA เป็น peripheral ภายใน — ไม่ต้องต่อวงจรภายนอก
    แต่ใช้ร่วมกับ peripheral อื่นเพื่อย้ายข้อมูล:

    DMA + ADC (Continuous Sampling)
    ┌─────────────────────────────────────────┐
    │  CH32V003                               │
    │                                         │
    │  ADC (PD2) ←─── [Sensor Signal]         │
    │     │                                   │
    │     ▼  DMA Channel 1                    │
    │  [buffer[128]] ←─── ถ่ายโอนอัตโนมัติ    │
    │     │                                   │
    │     ▼  (CPU ทำงานอื่นไปพร้อมกัน)         │
    │  USART ──────→ PC (แสดงผล)             │
    └─────────────────────────────────────────┘

    DMA + USART TX (ส่งข้อมูลบล็อกใหญ่)
    ┌─────────────────────────────────────────┐
    │  CH32V003                               │
    │                                         │
    │  RAM buffer[256] ──→ DMA Ch2 → USART_TX │
    │     (256 bytes ส่งโดย CPU ไม่ต้องวนลูป)   │
    └─────────────────────────────────────────┘

    CH32V003 มี 3 DMA Channels:
    Ch1: ADC, TIM1_CH1/2/3
    Ch2: USART1_TX, TIM1_CH4, TIM1_COM
    Ch3: TIM2_CH3/4, (reserved)

สำคัญ: DMA ทำงานโดยไม่ใช้ CPU — ขณะถ่ายโอน CPU ว่างไปทำอย่างอื่นได้เต็มที่ — เหมาะสำหรับระบบ real-time และ low-power

📚 หลักการทำงาน

DMA (Direct Memory Access) — ตัวควบคุมที่ถ่ายโอนข้อมูลระหว่าง memory ↔ peripheral โดยอัตโนมัติ โดยไม่ให้ CPU ต้องวนลูปอ่าน/เขียนทีละ byte

  • Memory-to-Memory: ก็อปปี้ข้อมูลจาก array หนึ่งไปอีก array — เร็วกว่า memcpy() เพราะ CPU ไม่ต้อง fetch/store ทีละ word
  • Peripheral-to-Memory: อ่านจาก ADC/USART/SPI ต่อเนื่อง — เหมาะสำหรับการเก็บตัวอย่างข้อมูลความเร็วสูง
  • Memory-to-Peripheral: ส่งข้อมูลไป USART/SPI — ส่ง buffer ขนาดใหญ่โดย CPU ไม่ต้องรอ

Transfer Size: DMA ของ CH32V003 รองรับ 8-bit / 16-bit / 32-bit ต่อ transfer

Callback: เมื่อ transfer เสร็จ DMA จะเรียก callback function — ให้คุณประมวลผลข้อมูลต่อได้ทันที

ข้อควรระวัง: DMA channel แต่ละตัวใช้เฉพาะ peripheral บางตัว — เช็คตาราง channel mapping ก่อนใช้

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Memory Copy ด้วย DMA

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"

uint8_t src[64] = {0xAA, 0xBB, 0xCC};
uint8_t dst[64] = {0};

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);

    DMA_SimpleInit(DMA_CH1, dst, src, 64);  // 64 bytes
    DMA_Start(DMA_CH1);

🔍 DMA_SimpleInit(ch, dst, src, size) — กำหนด DMA channel, ปลายทาง, ต้นทาง, ขนาด — รองรับสูงสุด 65535 bytes

ขั้นที่ 2: รอ Transfer เสร็จ

    while(DMA_GetStatus(DMA_CH1) == 0) {
        // CPU ว่าง — ทำอย่างอื่นได้!
        Delay_Ms(1);
    }
    USART_Print("DMA copy complete!\r\n");

    // ตรวจสอบผลลัพธ์
    USART_Print("dst[0]=0x");
    USART_PrintHex(dst[0], 1);
    USART_Print(" dst[1]=0x");
    USART_PrintHex(dst[1], 1);
    USART_Print("\r\n");

🔍 DMA_GetStatus(ch) — คืนค่า 0=กำลัง transfer, 1=เสร็จ — ในระหว่าง transfer CPU ว่าง 100% — ไม่ต้อง busy-wait

ขั้นที่ 3: DMA + ADC Continuous Sampling

    uint16_t adc_buf[128];
    ADC_SimpleInit();

    DMA_ADC_Init(DMA_CH1, adc_buf, 128);  // ADC → buffer
    ADC_StartContinuous(ADC_CH_PD2);       // เริ่ม ADC ต่อเนื่อง
    DMA_Start(DMA_CH1);                    // DMA เริ่มถ่ายโอน

    // CPU อิสระ — ทำงานอื่นได้!
    while(DMA_GetStatus(DMA_CH1) == 0) {
        // เช่น: ประมวลผลข้อมูลเก่า, สื่อสาร USART
    }

    // หาค่าเฉลี่ย
    uint32_t sum = 0;
    for(int i=0; i<128; i++) sum += adc_buf[i];
    uint16_t avg = sum / 128;

🔍 DMA_ADC_Init(ch, buf, n) — ตั้งค่า DMA อ่านจาก ADC ต่อเนื่อง — CPU รับข้อมูลทั้งหมดทีเดียวหลัง transfer เสร็จ

ขั้นที่ 4: DMA + USART ส่ง Buffer ใหญ่

    const char *msg = "Hello from DMA! 1234567890\r\n";
    DMA_USART_Transmit(DMA_CH2, (uint8_t*)msg, strlen(msg));

    USART_Print("Sending via DMA...\r\n");
    // CPU ทำอย่างอื่นต่อได้ — ไม่ต้องรอ USART send complete!

🔍 DMA_USART_Transmit(ch, data, len) — ส่งข้อมูลผ่าน USART โดย DMA — CPU ไม่ต้องวนลูปรอ TXE flag

ขั้นที่ 5: ใช้ DMA Callback

void on_dma_done(void) {
    USART_Print("DMA transfer complete!\r\n");
}

int main(void) {
    // ...
    DMA_SetCallback(DMA_CH1, on_dma_done);
    DMA_SimpleInit(DMA_CH1, dst, src, 64);
    DMA_Start(DMA_CH1);

    while(1) {
        // CPU ทำงานปกติ — callback จะถูกเรียกอัตโนมัติ
        Delay_Ms(100);
    }
}

🔍 DMA_SetCallback(ch, func) — ลงทะเบียน callback — ฟังก์ชันจะถูกเรียกเมื่อ DMA transfer เสร็จ — ไม่ต้อง poll

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
DMA transfer ไม่เริ่มลืมเรียก DMA_Start()ต้องเรียก DMA_SimpleInit() แล้ว DMA_Start()
ข้อมูลไม่ครบ/เสียหายBuffer ใน stack ถูก deallocate ก่อน transfer เสร็จประกาศ buffer เป็น global หรือ static
DMA ชนกับ peripheral อื่นใช้ DMA channel ที่ถูกจองแล้วเช็ค channel mapping — แต่ละ channel ใช้ peripheral เฉพาะ
ADC DMA รับค่าผิดลืมเรียก ADC_StartContinuous()ADC ต้องอยู่ใน continuous mode จึงจะ trigger DMA ได้
USART DMA ส่งไม่หมดเรียก DMA_Stop() ก่อน transfer เสร็จเช็ค DMA_GetStatus() ก่อนเรียก stop

🛠️ สร้าง Product: อัดเสียง/ข้อมูลเร็วโดย CPU ว่างทำงานอื่น

ใช้ DMA อ่าน ADC ต่อเนื่อง 128 samples ที่ 8kHz พร้อมกับส่งข้อมูลเก่า USART — CPU ว่างระหว่าง sampling

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"

#define BUF_SIZE 128

uint16_t adc_buf[BUF_SIZE];  // ต้องเป็น global!
volatile uint8_t dma_done = 0;

void dma_callback(void) {
    dma_done = 1;
}

int main(void)
{
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
    ADC_SimpleInit();

    DMA_ADC_Init(DMA_CH1, adc_buf, BUF_SIZE);
    DMA_SetCallback(DMA_CH1, dma_callback);

    while(1)
    {
        // เริ่ม sampling ชุดใหม่
        dma_done = 0;
        ADC_StartContinuous(ADC_CH_PD2);
        DMA_Start(DMA_CH1);

        // CPU ว่าง — ทำงานอื่นได้!
        uint32_t t = Get_CurrentMs();
        while(!dma_done) {
            // LED Blink แสดงว่า CPU ไม่ได้ busy-wait
            if(Get_CurrentMs() - t > 50) {
                // ทำอะไรก็ได้ระหว่างรอ
                t = Get_CurrentMs();
            }
        }

        // ประมวลผลข้อมูล
        uint32_t sum = 0;
        for(int i=0; i<BUF_SIZE; i++) sum += adc_buf[i];
        uint16_t avg = (uint16_t)(sum / BUF_SIZE);
        float v = ADC_ToVoltage(avg, 3.3f);

        USART_Print("ADC avg: "); USART_PrintNum((int32_t)avg);
        USART_Print(" ("); USART_PrintNum((int32_t)(v*1000));
        USART_Print("mV)\r\n");

        Delay_Ms(100);
    }
}

✏️ ลองทำดู

  1. แก้ไขให้ใช้ DMA_MemCopy() ก็อปปี้ข้อมูล ADC buffer ไปยัง buffer สำรอง แล้วประมวลผล buffer สำรองขณะ DMA กำลัง sampling รอบต่อไป (double-buffering)
  2. เปลี่ยนจาก polling DMA_GetStatus() เป็น callback — เปิด LED เมื่อ transfer เสร็จ
  3. ใช้ DMA_USART_Transmit() ส่งข้อความยาว 256 ตัวอักษร — วัดว่าเร็วกว่าส่งทีละ byte ด้วย USART_Print() กี่เท่า
