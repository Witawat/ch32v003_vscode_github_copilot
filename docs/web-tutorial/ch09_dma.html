บทที่ 9: DMA — ถ่ายโอนข้อมูลโดยไม่ใช้ CPU

🔌 การต่อวงจร

⚠️ ไม่ต้องต่อวงจรภายนอก: เป็น peripheral ภายในชิป CH32V003 — ทำงานทั้งหมดภายใน MCU ไม่ต้องต่ออุปกรณ์ใด ๆ เพิ่มเติม

    DMA เป็น peripheral ภายใน — ไม่ต้องต่อวงจรภายนอก
    แต่ใช้ร่วมกับ peripheral อื่นเพื่อย้ายข้อมูล:

    DMA + ADC (Continuous Sampling)
    ┌─────────────────────────────────────────┐
    │  CH32V003                               │
    │                                         │
    │  ADC (PD2) ←─── [Sensor Signal]         │
    │     │                                   │
    │     ▼  DMA Channel 1                    │
    │  [buffer[128]] ←─── ถ่ายโอนอัตโนมัติ    │
    │     │                                   │
    │     ▼  (CPU ทำงานอื่นไปพร้อมกัน)         │
    │  USART ──────→ PC (แสดงผล)             │
    └─────────────────────────────────────────┘

    DMA + USART TX (ส่งข้อมูลบล็อกใหญ่)
    ┌─────────────────────────────────────────┐
    │  CH32V003                               │
    │                                         │
    │  RAM buffer[256] ──→ DMA Ch2 → USART_TX │
    │     (256 bytes ส่งโดย CPU ไม่ต้องวนลูป)   │
    └─────────────────────────────────────────┘

    CH32V003 มี DMA controller 1 ตัว / 7 channels:
    Ch1: ADC1, TIM1_CH1/2/3, TIM2_CH2 (บาง map)
    Ch2: USART1_TX, TIM1_CH4, TIM1_COM
    Ch3: USART1_RX, TIM2_CH3/4
    Ch4: I2C1_TX, SPI1_TX (บาง config), TIM1_CH4 alt
    Ch5: I2C1_RX, TIM1 Update
    Ch6: TIM1_CH3 alt
    Ch7: TIM2_TRIG/UPDATE
    (fixed mapping ขึ้นกับ peripheral — ดู RCC DMA remap)

สำคัญ: DMA ทำงานโดยไม่ใช้ CPU — ขณะถ่ายโอน CPU ว่างไปทำอย่างอื่นได้เต็มที่ — เหมาะสำหรับระบบ real-time และ low-power

📚 หลักการทำงาน

DMA (Direct Memory Access) — ตัวควบคุมที่ถ่ายโอนข้อมูลระหว่าง memory ↔ peripheral โดยอัตโนมัติ โดยไม่ให้ CPU ต้องวนลูปอ่าน/เขียนทีละ byte

  • Memory-to-Memory: ก็อปปี้ข้อมูลจาก array หนึ่งไปอีก array — เร็วกว่า memcpy() เพราะ CPU ไม่ต้อง fetch/store ทีละ word
  • Peripheral-to-Memory: อ่านจาก ADC/USART/SPI ต่อเนื่อง — เหมาะสำหรับการเก็บตัวอย่างข้อมูลความเร็วสูง
  • Memory-to-Peripheral: ส่งข้อมูลไป USART/SPI — ส่ง buffer ขนาดใหญ่โดย CPU ไม่ต้องรอ

Transfer Size: DMA ของ CH32V003 รองรับ 8-bit / 16-bit / 32-bit ต่อ transfer (DMA_SIZE_BYTE / HALFWORD / WORD)

Callback: SimpleHAL แยก callback เป็น 3 ตัว — DMA_SetTransferCompleteCallback() (เสร็จ), DMA_SetHalfTransferCallback() (ครึ่งทาง — ping-pong), DMA_SetErrorCallback() (error) — เรียกใช้แยกกัน ไม่มี setter รวม

Status: DMA_GetStatus() คืนค่าเป็น enum DMA_Status: DMA_STATUS_IDLE / BUSY / COMPLETE / ERROR — ไม่ใช่ 0/1

ข้อควรระวัง: DMA channel แต่ละตัวใช้เฉพาะ peripheral บางตัว — เช็คตาราง channel mapping ก่อนใช้

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Memory Copy แบบ Blocking (Sync)

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"

uint8_t src[64] = {0xAA, 0xBB, 0xCC};
uint8_t dst[64] = {0};

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);

    // DMA_MemCopy — all-in-one: init + start + wait (blocking)
    DMA_MemCopy(dst, src, 64);
    // มาถึงบรรทัดนี้ dst ถูกก็อปปี้เรียบร้อย

    USART_Print("DMA MemCopy 64 bytes — DONE!\r\n");
    USART_Print("dst[0]=0x"); USART_PrintHex(dst[0], 1);
    USART_Print(" dst[1]=0x"); USART_PrintHex(dst[1], 1);
    USART_Print("\r\n");
    while (1);
}

🔍 DMA_MemCopy(dst, src, size) เป็น wrapper blocking — เรียกครั้งเดียวจบ — ภายในทำ init + start + wait ให้เอง — ไม่ต้องสร้าง DMA_Config_t เอง เหมาะกับงาน copy แบบ synchronous

ขั้นที่ 2: Memory Copy แบบ Non-blocking (Async)

uint8_t src[64], dst[64];

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);

    // วิธี A: helper แบบ async (เน้นง่าย)
    DMA_MemCopyAsync(DMA_CH1, dst, src, 64);
    while (DMA_GetStatus(DMA_CH1) == DMA_STATUS_BUSY) {
        // CPU ว่าง — ทำอย่างอื่นได้!
    }
    USART_Print("Async copy done via helper\r\n");

    // วิธี B: ใช้ DMA_Config_t + DMA_SimpleInit แบบเต็ม (คุมทุก field)
    DMA_Config_t cfg = {
        .channel          = DMA_CH1,
        .direction        = DMA_DIR_MEM_TO_MEM,
        .priority         = DMA_PRIORITY_HIGH,
        .data_size        = DMA_SIZE_BYTE,
        .mode             = DMA_MODE_NORMAL,
        .mem_increment    = 1,
        .periph_increment = 1,            // M2M ต้องเปิดทั้งสองฝั่ง
        .periph_addr      = (uint32_t)src, // ฝั่ง source
        .mem_addr         = (uint32_t)dst, // ฝั่ง destination
        .buffer_size      = 64
    };
    DMA_SimpleInit(&cfg);
    DMA_Start(DMA_CH1);

    // รอแบบ polling enum (ไม่ใช่ 0/1!)
    while (DMA_GetStatus(DMA_CH1) == DMA_STATUS_BUSY) {
        // blink LED / ทำงานอื่น
    }

    // หรือใช้ DMA_WaitComplete แบบ blocking มี timeout
    if (DMA_WaitComplete(DMA_CH1, 1000)) {
        USART_Print("Transfer done within 1000 ms\r\n");
    } else {
        USART_Print("Transfer TIMEOUT!\r\n");
    }
}

🔍 DMA_SimpleInit() รับ pointer ไปยัง DMA_Config_t ตัวเดียว — ไม่ใช่ 4 scalar — เลือกทิศทาง/priority/data_size/mode/increment ได้ครบ — ต้องตามด้วย DMA_Start(channel) เสมอ — ตรวจสถานะด้วย enum DMA_STATUS_BUSY / DMA_STATUS_COMPLETE (ไม่ใช่ 0/1)

ขั้นที่ 3: DMA + USART Transmit (ส่ง Buffer ใหญ่)

uint8_t tx_buf[256];

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);

    // เตรียมข้อความลง tx_buf
    const char *msg = "Hello from DMA! 1234567890 ABCDEFGHIJKLM\r\n";
    for (int i = 0; msg[i]; i++) tx_buf[i] = (uint8_t)msg[i];
    uint16_t len = strlen(msg);

    // ผูก DMA_CH2 กับ USART1_TX (1 ครั้งพอ)
    DMA_USART_InitTx(DMA_CH2, tx_buf, 256);

    // ส่งผ่าน DMA — CPU ไม่ต้องวนลูปรอ TXE
    DMA_USART_Transmit(DMA_CH2, tx_buf, len);

    // CPU ทำอย่างอื่นต่อได้ — เช็คสถานะด้วย enum
    while (DMA_GetStatus(DMA_CH2) == DMA_STATUS_BUSY) {
        // ...
    }
    USART_Print("USART DMA TX done\r\n");

    // หรือยิง one-shot: init + transmit + wait ครบจบในฟังก์ชันเดียว
    DMA_USART_Send(DMA_CH2, (const uint8_t*)"Goodbye DMA!\r\n", 13);
}

🔍 DMA_USART_InitTx(ch, buf, size) ผูก channel กับ USART1_TX — ไม่มี args circular (TX มักเป็น one-shot) — DMA_USART_Transmit() เริ่มส่งจริง — ใช้ DMA_USART_Send() ถ้าอยากครบจบในบรรทัดเดียว (blocking)

ขั้นที่ 4: DMA + SPI Transfer (Full-Duplex)

uint8_t spi_tx[32], spi_rx[32];

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    SPI_SimpleInit(SPI_MODE_MASTER, SPI_CLK_4MHZ);

    // ผูก TX/RX channel คู่กัน (1 ครั้งพอ)
    DMA_SPI_Init(DMA_CH4, DMA_CH5);  // TX=CH4, RX=CH5

    // เตรียม tx
    for (int i = 0; i < 32; i++) spi_tx[i] = (uint8_t)i;

    // Transfer full-duplex: ส่ง 32 และรับ 32 พร้อมกัน
    DMA_SPI_TransferBuffer(DMA_CH4, DMA_CH5, spi_tx, spi_rx, 32);

    // รอเสร็จ
    if (DMA_WaitComplete(DMA_CH4, 100) && DMA_WaitComplete(DMA_CH5, 100)) {
        USART_Print("SPI DMA done:\r\n");
        for (int i = 0; i < 8; i++) {
            USART_PrintHex(spi_rx[i], 1); USART_Print(" ");
        }
        USART_Print("\r\n");
    }
}

🔍 DMA_SPI_Init(tx_ch, rx_ch) config คู่ TX/RX ไว้ทีเดียว — DMA_SPI_TransferBuffer() active ทั้งสอง channel พร้อมกัน (full-duplex) — รอทั้งสองเสร็จด้วย DMA_WaitComplete() แยก channel

ขั้นที่ 5: DMA + TIM PWM Waveform (สร้าง sine / arbitrary wave)

// ตาราง sine 64 samples คำนวณล่วงหน้า (0..1000 duty)
const uint16_t sine_table[64] = { /* 64 ค่า */ };

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    // เปิด TIM1 CH1 PWM ก่อน (SimpleHAL TIM API แยก)
    // ... TIM_PWM_Init(TIM1, 1, 1000);

    // ให้ DMA เขียน duty cycle ใหม่ทุกครั้งที่ TIM1 update
    uint32_t ccr1 = DMA_TIM_GetCCRAddress(TIM1, 1);  // &TIM1->CH1CVR
    DMA_TIM_UpdatePWM(DMA_CH5, TIM1, ccr1, sine_table, 64, /*circular=*/1);

    // เปิด DMA trigger ของ TIM1 (TIM_DMA_Update)
    TIM_DMACmd(TIM1, TIM_DMA_Update, ENABLE);
    DMA_Start(DMA_CH5);

    // ตอนนี้ PWM output ออก sine wave ต่อเนื่องโดย CPU ไม่ต้องยุ่ง
    while (1);
}

🔍 DMA_TIM_UpdatePWM(ch, TIMx, ccr_addr, waveform, length, circular) ผูก DMA กับ CCR register ของ TIM — ทุก update event DMA จะโหลด duty ถัดไปจาก lookup table — circular=1 ทำให้วนซ้ำ forever (sine/triangle/sawtooth ต่อเนื่อง) — DMA_TIM_GetCCRAddress() ช่วยหา address ของ CHxCVR โดยไม่ต้องจำ

ขั้นที่ 6: DMA + I2C Transfer

uint8_t i2c_tx[32], i2c_rx[32];

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    I2C_SimpleInit(I2C_ADDR_7BIT, 100000);  // 100 kHz

    // I2C1_TX fixed-map → DMA_CH4, I2C1_RX fixed-map → DMA_CH5
    DMA_I2C_InitTx(DMA_CH4);
    DMA_I2C_InitRx(DMA_CH5);

    // เตรียม tx (register address + payload)
    i2c_tx[0] = 0x10;  // register address
    for (int i = 1; i < 32; i++) i2c_tx[i] = (uint8_t)i;

    // ส่ง + รับผ่าน DMA (ต้อง set slave/reg ผ่าน I2C_WriteReg ก่อนในบาง peripheral)
    DMA_I2C_Transfer(DMA_CH4, DMA_CH5, i2c_tx, i2c_rx, 32);

    if (DMA_WaitComplete(DMA_CH4, 100) && DMA_WaitComplete(DMA_CH5, 100)) {
        USART_Print("I2C DMA transfer complete\r\n");
    }
}

🔍 DMA_I2C_InitTx(ch) / DMA_I2C_InitRx(ch) — CH32V003 fixed-map I2C1_TX ที่ DMA_CH4 และ I2C1_RX ที่ DMA_CH5 ต้องใช้ channel ตรงตามนี้ — DMA_I2C_Transfer() ทำ transfer TX+RX พร้อมกัน — ระวังว่าบาง slave ต้องส่ง slave address + register ผ่าน I2C_WriteReg() ก่อน

ขั้นที่ 7: DMA + ADC Circular Sampling + TC Callback

#define BUF_SIZE 128
uint16_t adc_buf[BUF_SIZE];
volatile uint8_t adc_ready = 0;

// Callback สำหรับ Transfer Complete (receiver มี signature
// void f(DMA_Channel channel))
void on_adc_done(DMA_Channel channel) {
    adc_ready = 1;
}

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
    ADC_SimpleInit();
    pinMode(PC0, PIN_MODE_OUTPUT);

    // DMA_ADC_Init — รับ 4 args (สุดท้าย circular=1)
    DMA_ADC_Init(DMA_CH1, adc_buf, BUF_SIZE, /*circular=*/1);

    // ตั้ง Transfer Complete callback (แยกจาก Error/Half)
    DMA_SetTransferCompleteCallback(DMA_CH1, on_adc_done);

    // เริ่ม ADC continuous + DMA
    ADC_StartContinuous(ADC_CH_PD2);
    DMA_Start(DMA_CH1);

    while (1) {
        if (adc_ready) {
            adc_ready = 0;

            // สถิติ 128 samples ล่าสุด
            uint32_t sum = 0;
            for (int i = 0; i < BUF_SIZE; i++) sum += adc_buf[i];
            uint16_t avg = (uint16_t)(sum / BUF_SIZE);

            USART_Print("ADC avg: "); USART_PrintNum((int32_t)avg);
            USART_Print("\r\n");
        }
        // CPU ว่าง — ทำอย่างอื่นได้ตลอด
        digitalToggle(PC0);
        Delay_Ms(50);
    }
}

🔍 DMA_ADC_Init(ch, buf, size, circular) รับ 4 args — arg สุดท้าย circular=1 ทำให้ DMA วนเขียน buffer ตลอด (ring buffer) — ต้องใช้ DMA_SetTransferCompleteCallback() ที่เป็น setter แยก (รับ DMA_Channel) — ไม่มี setter แบบรวม generic ใน API

ขั้นที่ 8: Advanced — Multi-channel ADC + Ping-Pong

// 3 ADC channels × 10 samples each = 30 slots ต่อ half-buffer
#define N_CHAN   3
#define N_SAMP   10
#define HALF     (N_CHAN * N_SAMP)   // 30
uint16_t adc_multi[HALF * 2];        // 60 — ping + pong

volatile uint8_t half_ready = 0;     // 0 = first half พร้อม, 1 = second half
volatile uint8_t full_ready = 0;

void on_half(DMA_Channel channel) {
    // ถึงครึ่งทาง → buffer 0..29 (first half) พร้อมอ่าน
    half_ready = 1;
}

void on_full(DMA_Channel channel) {
    // จบรอบ → buffer 30..59 (second half) พร้อมอ่าน
    full_ready = 1;
}

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
    ADC_SimpleInit();

    // Multi-channel ADC DMA: 3 channels, 10 samples each
    DMA_ADC_InitMultiChannel(DMA_CH1, adc_multi, N_CHAN, N_SAMP * 2);

    // ใช้ 2 callback แยก — Half สำหรับ ping-pong, TC สำหรับจบรอบ
    DMA_SetHalfTransferCallback(DMA_CH1, on_half);
    DMA_SetTransferCompleteCallback(DMA_CH1, on_full);

    ADC_StartContinuous(ADC_CH_PD2);  // +เปิดหลาย channel ใน ADC config
    DMA_Start(DMA_CH1);

    while (1) {
        if (half_ready) {
            // process &adc_multi[0..29]
            half_ready = 0;
        }
        if (full_ready) {
            // process &adc_multi[30..59]
            full_ready = 0;
        }
        // CPU ทำอย่างอื่นได้ตลอด
    }
}

🔍 DMA_ADC_InitMultiChannel(ch, buf, num_channels, samples_per_channel) ผูก DMA กับ multi-channel ADC (interleaved: ch1,ch2,ch3,ch1,ch2,...) — ping-pong ใช้ DMA_SetHalfTransferCallback() แจ้งเมื่อถึงครึ่ง → ประมวลผลครึ่งแรกได้ทันที ขณะ DMA เขียนครึ่งหลัง — TC callback แจ้งจบรอบ — สาม callback แยกอิสระกัน

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
Transfer ไม่เริ่มลืมเรียก DMA_Start(channel)ต้องเรียกทั้ง DMA_SimpleInit(&config) และ DMA_Start(ch)
เช็คสถานะผิดเสมอเทียบค่า DMA_GetStatus() กับตัวเลข 0 หรือ 1 แทน enumใช้ enum DMA_STATUS_BUSY / DMA_STATUS_COMPLETE เท่านั้น
Callback ทำงานไม่ครบใช้ setter แบบรวม generic ซึ่งไม่มีใน APIใช้ DMA_SetTransferCompleteCallback(), DMA_SetErrorCallback(), DMA_SetHalfTransferCallback() แยก
ADC DMA init ผิดเรียก DMA_ADC_Init(ch, buf, n) 3 argsAPI จริงรับ 4 args มี circular ท้าย — DMA_ADC_Init(ch, buf, n, 1)
SimpleInit พารามิเตอร์ผิดส่ง 4 scalar DMA_SimpleInit(ch, dst, src, size)API จริงรับ pointer ตัวเดียว — สร้าง DMA_Config_t แล้วส่ง &config
ข้อมูลไม่ครบ/เสียหายBuffer ใน stack ถูก dealloc ก่อน transfer เสร็จประกาศ buffer เป็น global หรือ static
DMA ชนกับ peripheral อื่นใช้ channel ผิด mappingเช็ค channel mapping — I2C1_TX fixed CH4, I2C1_RX fixed CH5 ฯลฯ
ADC DMA รับค่าผิดลืม ADC_StartContinuous()ADC ต้อง continuous mode จึงจะ trigger DMA ได้
USART DMA ส่งไม่หมดเรียก DMA_Stop(ch) ก่อนเสร็จเช็ค DMA_GetStatus(ch) หรือใช้ DMA_WaitComplete() ก่อน stop

🛠️ สร้าง Product: อัดเสียง/ข้อมูลเร็วโดย CPU ว่างทำงานอื่น

ใช้ DMA อ่าน ADC ต่อเนื่อง 128 samples ที่ circular mode พร้อมกับส่งข้อมูลเก่า USART — CPU ว่างระหว่าง sampling

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"

#define BUF_SIZE 128

uint16_t adc_buf[BUF_SIZE];   // ต้องเป็น global!
volatile uint8_t dma_ready = 0;

// TC callback (signature: void f(DMA_Channel))
void on_adc_tc(DMA_Channel channel) {
    dma_ready = 1;
}

int main(void)
{
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
    ADC_SimpleInit();
    pinMode(PC0, PIN_MODE_OUTPUT);

    // 4 args — ตัวสุดท้าย circular=1 ให้ DMA วนเขียนตลอด
    DMA_ADC_Init(DMA_CH1, adc_buf, BUF_SIZE, 1);
    DMA_SetTransferCompleteCallback(DMA_CH1, on_adc_tc);

    ADC_StartContinuous(ADC_CH_PD2);
    DMA_Start(DMA_CH1);

    while (1)
    {
        if (dma_ready) {
            dma_ready = 0;

            // ประมวลผล 128 samples
            uint32_t sum = 0;
            for (int i = 0; i < BUF_SIZE; i++) sum += adc_buf[i];
            uint16_t avg = (uint16_t)(sum / BUF_SIZE);
            float v = ADC_ToVoltage(avg, 3.3f);

            USART_Print("ADC avg: "); USART_PrintNum((int32_t)avg);
            USART_Print(" ("); USART_PrintNum((int32_t)(v * 1000));
            USART_Print("mV)\r\n");
        }
        // CPU ว่าง — toggle LED ตามใจ
        digitalToggle(PC0);
        Delay_Ms(50);
    }
}

✏️ ลองทำดู

  1. ใช้ DMA_MemCopyAsync() ก็อปปี้ข้อมูล ADC buffer ไปยัง buffer สำรอง — ประมวลผล buffer สำรองขณะ DMA sampling รอบต่อไป (double-buffering) โดยใช้ DMA_SetHalfTransferCallback() แจ้งครึ่งทาง
  2. เปลี่ยนจาก polling DMA_GetStatus() เป็น DMA_SetTransferCompleteCallback() — เปิด LED เมื่อ transfer เสร็จ
  3. ใช้ DMA_USART_Send() ส่งข้อความยาว 256 ตัวอักษร — วัดว่าเร็วกว่าส่งทีละ byte ด้วย USART_Print() กี่เท่า
