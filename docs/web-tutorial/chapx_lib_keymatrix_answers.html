📝 เฉลย — Key Matrix

ข้อ 1: Calculator — +−×÷ = บน LCD 16×2

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "KeyMatrix.h"
#include "lcd1602_i2c.h"

KeyMatrix_Instance kp;
#define KEYMAP "123+456-789*C0=/"  // 4×4 matrix

int32_t num1 = 0, num2 = 0;
char op = 0;
bool new_input = true;
char display[17];

void calc_process(char key) {
    if(key >= '0' && key <= '9') {
        if(new_input) { num1 = key - '0'; new_input = false; }
        else num1 = num1 * 10 + (key - '0');
    }
    else if(key == 'C') { num1 = num2 = 0; op = 0; new_input = true; }
    else if(key == '=') {
        if(op == '+') num1 = num2 + num1;
        if(op == '-') num1 = num2 - num1;
        if(op == '*') num1 = num2 * num1;
        if(op == '/') num1 = num2 / num1;
        op = 0; new_input = true;
    }
    else {  // operator + - * /
        num2 = num1; op = key; new_input = true;
    }
}

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    GPIO_Pin rows[] = {PA0, PA1, PA2, PA3};
    GPIO_Pin cols[] = {PA4, PA5, PA6, PA7};
    KeyMatrix_Init(&kp, rows, 4, cols, 4, KEYMAP);
    I2C_LCD_Init(0x27, 16, 2);
    I2C_LCD_Backlight(1);

    while(1) {
        char key = KeyMatrix_GetKey(&kp);
        if(key != 0) {
            calc_process(key);
            I2C_LCD_Clear();
            if(op) snprintf(display, 17, "%ld%c", num2, op);
            else   snprintf(display, 17, "%ld", num1);
            I2C_LCD_SetCursor(0, 0); I2C_LCD_Print(display);
            if(!new_input) {
                snprintf(display, 17, "%ld", num1);
                I2C_LCD_SetCursor(0, 1); I2C_LCD_Print(display);
            }
            while(KeyMatrix_GetKey(&kp) != 0);  // debounce wait
        }
    }
}
🧠 แนวคิด: แยก state: new_input flag บอกว่าเป็นเลขตัวแรกหลัง operator — กดเลข → ต่อ string ทศนิยม (×10+) — กด operator → เก็บ num2+op แล้วรอเลขถัดไป — กด = → switch ตาม op คำนวณ num2 op num1 — แสดงบรรทัดบนเป็นสูตร บรรทัดล่างเป็นเลขปัจจุบัน — C ล้างทั้งหมด — หารใช้ integer หารเอาเฉพาะผลลัพธ์

ข้อ 2: Password Input — ป้อน 4 หลัก แสดง ***

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "KeyMatrix.h"
#include "lcd1602_i2c.h"
#include "AT24Cxx.h"

KeyMatrix_Instance kp;
#define KEYMAP  "123A456B789C*0#D"
#define PASSWD  "2847"
#define MAXLEN  4

char input[5] = "";
uint8_t pos = 0;

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    GPIO_Pin rows[] = {PA0, PA1, PA2, PA3};
    GPIO_Pin cols[] = {PA4, PA5, PA6, PA7};
    KeyMatrix_Init(&kp, rows, 4, cols, 4, KEYMAP);
    I2C_LCD_Init(0x27, 16, 2);
    I2C_LCD_Backlight(1);
    I2C_LCD_SetCursor(0, 0); I2C_LCD_Print("Enter Password:");

    while(1) {
        char key = KeyMatrix_GetKey(&kp);
        if(key == 0) continue;

        if(key >= '0' && key <= '9' && pos < MAXLEN) {
            input[pos++] = key; input[pos] = '\0';
            I2C_LCD_SetCursor(4, 1);
            for(uint8_t i = 0; i < pos; i++) I2C_LCD_Print("*");
            for(uint8_t i = pos; i < MAXLEN; i++) I2C_LCD_Print(" ");
        }

        if(key == 'C' || key == '#') {  // clear
            pos = 0; input[0] = '\0';
            I2C_LCD_SetCursor(4, 1); I2C_LCD_Print("    ");
        }

        if(key == '#' && pos == MAXLEN) {
            if(strcmp(input, PASSWD) == 0) {
                I2C_LCD_Clear(); I2C_LCD_Print("ACCESS GRANTED");
                GPIO_WriteBit(PC0, 1); Delay_Ms(2000); GPIO_WriteBit(PC0, 0);
            } else {
                I2C_LCD_Clear(); I2C_LCD_Print("ACCESS DENIED");
                Delay_Ms(1500);
            }
            pos = 0; input[0] = '\0';
            I2C_LCD_Clear(); I2C_LCD_SetCursor(0, 0); I2C_LCD_Print("Enter Password:");
        }

        Delay_Ms(80);  // debounce
    }
}
🧠 แนวคิด: รับเฉพาะ '0'-'9' สะสมใน input[] ถึง MAXLEN — แสดง * เพื่ออำพรางรหัส — ปุ่ม C ล้าง input — ปุ่ม # ยืนยันเมื่อครบ 4 หลัก → strcmp เทียบกับรหัสลับ — ถ้าผ่าน → เปิด relay/LED, ถ้าไม่ → แจ้ง DENIED — เพิ่ม delay หลัง GetKey กัน debounce ซ้ำ — สำหรับระบบจริง ควรเก็บรหัสใน EEPROM ไม่ใช่ hardcode

ข้อ 3: Ghost Key Detection — ตรวจจับ phantom key ใน matrix

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "KeyMatrix.h"

KeyMatrix_Instance kp;
#define ROWS 4
#define COLS 4
#define KEYMAP "123A456B789C*0#D"

uint8_t row_pins[] = {PA0, PA1, PA2, PA3};
uint8_t col_pins[] = {PA4, PA5, PA6, PA7};

uint8_t ghost_detect(uint8_t *keys, uint8_t count) {
    if(count < 3) return 0;  // ghost เกิดเฉพาะเมื่อกด ≥ 3 ปุ่ม

    uint8_t row_mask = 0, col_mask = 0;
    for(uint8_t i = 0; i < count; i++) {
        uint8_t r = (keys[i] - 1) / COLS;
        uint8_t c = (keys[i] - 1) % COLS;
        row_mask |= (1 << r);
        col_mask |= (1 << c);
    }

    // ถ้ามี row ใดถูกกด ≥ 2 ปุ่ม และ col เหล่านั้นมีอีกปุ่มอยู่ → ghost
    uint8_t rows_used = 0, cols_used = 0;
    for(uint8_t i = 0; i < ROWS; i++) if(row_mask & (1<= 2 && cols_used >= 2);
}

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_Init(115200);
    KeyMatrix_Init(&kp, row_pins, 4, col_pins, 4, KEYMAP);

    uint8_t pressed[16], pcount = 0;
    Timer_t scan_tmr;
    Start_Timer(&scan_tmr, 20, 1);

    while(1) {
        if(Is_Timer_Expired(&scan_tmr)) {
            // Note: API จริงรองรับ single key เท่านั้น (KeyMatrix_GetCurrentKey)
            // ghost key detection ต้องการ multi-key scan ซึ่ง library ไม่รองรับ
            char key = KeyMatrix_GetCurrentKey(&kp);
            if(key) {
                pressed[0] = key;
                pcount = 1;
            } else {
                pcount = 0;
            }

            if(ghost_detect(pressed, pcount)) {
                USART_Println("⚠ GHOST KEY DETECTED — data unreliable!");
            } else {
                USART_Printf("Keys: %d → ", pcount);
                for(uint8_t i = 0; i < pcount; i++)
                    USART_Printf("%c ", pressed[i]);
                USART_Println("");
            }
        }
    }
}
🧠 แนวคิด: Ghost key เกิดเมื่อกด ≥ 3 ปุ่มใน matrix ที่ไม่มี diode — วงจรไหลย้อนกลับทำให้อ่านค่าผิด — วิธีตรวจจับ: นับจำนวน row/col ที่ active — ถ้า count < rows_used × cols_used แสดงว่ามีตำแหน่งที่ "ว่าง" แต่ matrix อ่านเจอ — วิธีแก้ ghost จริง: ใส่ diode 1N4148 อนุกรมทุกสวิตช์ (hardware fix) — ถ้าตรวจพบ ghost ให้ ignore การกดทั้งหมด ป้องกัน false trigger
