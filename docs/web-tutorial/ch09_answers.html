📝 เฉลย — บทที่ 9: DMA

ข้อ 1: DMA_MemCopyAsync + blink LED ขณะ DMA ทำงาน

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include "SimpleHAL.h"

uint8_t  g_src[128];
uint8_t  g_dst[128];  // ต้อง global — DMA ใช้ pointer ตรง

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
    pinMode(PC0, PIN_MODE_OUTPUT);

    // เติม src ด้วย pattern
    for (int i = 0; i < 128; i++) g_src[i] = (uint8_t)(i * 2);

    // เริ่ม async copy — return ทันที ไม่รอ
    DMA_MemCopyAsync(DMA_CH1, g_dst, g_src, 128);

    // ขณะ DMA ทำงาน → CPU blink LED (พิสูจน์ว่าไม่ busy-wait)
    // สำคัญ: เช็คสถานะด้วย enum ไม่ใช่ 0/1
    while (DMA_GetStatus(DMA_CH1) == DMA_STATUS_BUSY) {
        digitalToggle(PC0);
        Delay_Ms(25);
    }

    // ตรวจสถานะสุดท้าย — อาจเป็น COMPLETE หรือ ERROR
    if (DMA_GetStatus(DMA_CH1) == DMA_STATUS_COMPLETE) {
        USART_Print("DMA MemCopy 128 bytes — DONE!\r\n");
        USART_Print("Verify: ");
        for (int i = 0; i < 8; i++) {
            USART_PrintHex(g_dst[i], 1);
            USART_Print(" ");
        }
        USART_Print("\r\n");
    } else {
        USART_Print("DMA failed!\r\n");
    }

    while (1);
}
🧠 แนวคิด: DMA_MemCopyAsync(ch, dst, src, size) เริ่ม transfer แล้ว return ทันที — ใช้ DMA_GetStatus(ch) คืนค่าเป็น enum DMA_Status ตรวจสถานะ — ต้องเทียบกับ DMA_STATUS_BUSY / DMA_STATUS_COMPLETE / DMA_STATUS_ERROR เท่านั้น — ห้ามใช้ == 0 หรือ == 1 เพราะ DMA_STATUS_IDLE=0 และ DMA_STATUS_BUSY=1 แตกต่างจากการตีความเดิม (running/done) — buffer ต้องเป็น global/static เพราะ DMA เข้าถึงผ่าน pointer ไม่รู้เรื่อง stack frame — ระหว่าง loop คุณสามารถทำงาน CPU-intensive อย่างอื่นได้โดยไม่บล็อก

ข้อ 2: DMA_USART_InitTx + TC callback ส่งข้อความโดยไม่บล็อก CPU

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include "SimpleHAL.h"

uint8_t g_tx_buf[256];
const char *g_msg = "Hello from DMA! CPU is free :)\r\n";
volatile uint8_t g_tx_done = 0;

// TC callback — signature เป็น void f(DMA_Channel channel)
void on_tx_complete(DMA_Channel channel) {
    g_tx_done = 1;
}

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
    pinMode(PC0, PIN_MODE_OUTPUT);

    // คัดลอกข้อความลง tx_buf (DMA ต้องการ buffer ที่อยู่ใน memory ตลอด)
    uint16_t len = 0;
    for (; g_msg[len]; len++) g_tx_buf[len] = (uint8_t)g_msg[len];

    // ผูก DMA_CH2 กับ USART1_TX (1 ครั้งพอ)
    DMA_USART_InitTx(DMA_CH2, g_tx_buf, 256);

    // ตั้ง TC callback แยก (ไม่ใช้ setter รวมแบบ generic)
    DMA_SetTransferCompleteCallback(DMA_CH2, on_tx_complete);

    USART_Print("=== DMA USART Send Demo ===\r\n");

    while (1) {
        // เริ่มส่ง — CPU ไม่ต้องรอ
        g_tx_done = 0;
        DMA_USART_Transmit(DMA_CH2, g_tx_buf, len);

        // CPU ว่าง — blink LED ได้ตามปกติ
        uint32_t start = Get_CurrentMs();
        while (!g_tx_done) {
            digitalToggle(PC0);
            Delay_Ms(50);
            // ปลอดภัยเพราะ DMA ส่งเอง ไม่เกี่ยวกับ CPU
        }

        USART_Print("  (sent in ");
        USART_PrintNum((int32_t)(Get_CurrentMs() - start));
        USART_Print("ms)\r\n");

        Delay_Ms(1000);
    }
}
🧠 แนวคิด: DMA_USART_InitTx(ch, buf, size) ผูก channel กับ USART1_TX — 3 args ไม่มี circular (TX เป็น one-shot) — DMA_USART_Transmit() เริ่มส่งจริง — ใช้ DMA_SetTransferCompleteCallback() ซึ่งเป็น setter แยก (รับ DMA_Channel เป็นพารามิเตอร์) ไม่มี setter แบบ generic รวมใน API — ระหว่างรอ callback, CPU blink LED ได้ตามปกติ — ข้อควรระวัง: buffer ต้องอยู่ใน memory จนกว่า DMA จะส่งเสร็จ (global/static)

ข้อ 3: DMA_ADC_Init (4 args) + circular + statistics + ping-pong

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include "SimpleHAL.h"

#define BUF_SIZE 128

uint16_t g_adc_buf[BUF_SIZE];   // ต้อง global!
volatile uint8_t g_half_ready   = 0;  // ping-pong: ครึ่งแรก
volatile uint8_t g_full_ready   = 0;  // ping-pong: ครึ่งหลัง

// Half-Transfer callback — ถึงครึ่งทาง = first half พร้อมอ่าน
void on_half(DMA_Channel channel) {
    g_half_ready = 1;
}

// Transfer Complete callback — จบรอบ = second half พร้อมอ่าน
void on_full(DMA_Channel channel) {
    g_full_ready = 1;
}

// ฟังก์ชันสถิติใช้ซ้ำกับทั้ง first / second half
static void compute_stats(uint16_t *p, uint16_t n,
                          uint16_t *avg, uint16_t *vmin, uint16_t *vmax) {
    uint32_t sum = 0;
    uint16_t lo = 4095, hi = 0;
    for (uint16_t i = 0; i < n; i++) {
        sum += p[i];
        if (p[i] < lo) lo = p[i];
        if (p[i] > hi) hi = p[i];
    }
    *avg  = (uint16_t)(sum / n);
    *vmin = lo;
    *vmax = hi;
}

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
    ADC_SimpleInit();
    pinMode(PC0, PIN_MODE_OUTPUT);

    // DMA_ADC_Init — 4 args! arg สุดท้าย circular=1
    DMA_ADC_Init(DMA_CH1, g_adc_buf, BUF_SIZE, 1);

    // ใช้ 2 callback แยก: Half (ping-pong) + TC (จบรอบ)
    DMA_SetHalfTransferCallback(DMA_CH1, on_half);
    DMA_SetTransferCompleteCallback(DMA_CH1, on_full);

    ADC_StartContinuous(ADC_CH_PD2);
    DMA_Start(DMA_CH1);

    USART_Print("DMA ADC Circular + Ping-Pong — 128 samples\r\n");

    while (1) {
        // first half (samples 0..63) พร้อม
        if (g_half_ready) {
            g_half_ready = 0;
            uint16_t avg, vmin, vmax;
            compute_stats(&g_adc_buf[0], BUF_SIZE / 2, &avg, &vmin, &vmax);
            USART_Print("H1 Avg="); USART_PrintNum(avg);
            USART_Print(" Min=");    USART_PrintNum(vmin);
            USART_Print(" Max=");    USART_PrintNum(vmax);
            USART_Print("\r\n");
        }

        // second half (samples 64..127) พร้อม
        if (g_full_ready) {
            g_full_ready = 0;
            uint16_t avg, vmin, vmax;
            compute_stats(&g_adc_buf[BUF_SIZE / 2], BUF_SIZE / 2, &avg, &vmin, &vmax);
            float v = ADC_ToVoltage(avg, 3.3f);
            USART_Print("H2 Avg="); USART_PrintNum(avg);
            USART_Print(" (");      USART_PrintNum((int32_t)(v * 1000));
            USART_Print("mV)\r\n");
        }

        // CPU ว่าง — toggle LED ตามใจ
        digitalToggle(PC0);
        Delay_Ms(20);
    }
}
🧠 แนวคิด: DMA_ADC_Init(ch, buf, n, circular) รับ 4 args — arg สุดท้าย circular=1 เปิด ring buffer — สำหรับ ping-pong ต้องตั้งทั้ง DMA_SetHalfTransferCallback() (แจ้งถึงครึ่งทาง) และ DMA_SetTransferCompleteCallback() (แจ้งจบรอบ) — เป็น 2 setter แยกอิสระ ไม่ใช่ setter รวมแบบ generic — เมื่อ half callback มา ให้อ่าน buf[0..63] ขณะที่ DMA เขียน buf[64..127] และกลับกัน — ลด lag เพราะประมวลผลได้ทันทีที่ครึ่งแรกเสร็จ ไม่ต้องรอเต็ม 128 — สถิติ avg/min/max ใน callback หรือใน main loop ก็ได้ — ถ้าเกิด error ให้ตรวจ DMA_GetStatus(ch) == DMA_STATUS_ERROR และตั้ง DMA_SetErrorCallback เพิ่มเติม
