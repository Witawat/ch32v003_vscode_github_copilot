YL-83 — Rain Sensor ตรวจจับน้ำฝน ADC + GPIO

ระดับกลาง

YL-83 ใช้แผ่น PCB แบบ interleaved traces — เมื่อมีหยดน้ำตกลงบนแผ่น → ความต้านทานลดลง → ADC สูงขึ้น — มี LM393 Comparator สำหรับ Digital Output — Pot ปรับ threshold

📚 หลักการทำงาน

YL-83 Rain Sensor วัดความชื้นบนแผ่นเซนเซอร์ — แผ่นเป็นลายทองแดงสลับฟันปลา — เมื่อน้ำเชื่อมรอยต่อ ความต้านทานลด → ADC สูง — แห้ง = ความต้านทานสูง → ADC ต่ำ

2 Outputs: Analog (A0) — วัดปริมาณน้ำที่ตก (0=แห้ง, 1023=น้ำท่วม) — Digital (D0) — LM393 comparator — HIGH เมื่อน้ำเกิน threshold — ใช้ขัดจังหวะ MCU ได้

Corrosion Warning: แผ่นเซนเซอร์เป็นทองแดง — ถ้าเปิดไฟเลี้ยงตลอดเวลา จะเกิด electrolysis กัดกร่อนเร็ว — ใช้ GPIO ควบคุม VCC เปิดเฉพาะตอนวัด (~100ms) — ปิดทันทีเมื่อวัดเสร็จ

การติดตั้ง: วางแผ่นราบ แต่เอียงเล็กน้อย (~15°) ให้น้ำไหลออก ไม่ขัง — หลีกเลี่ยงแดดจัดเพราะทำให้แห้งเร็ว — ใช้วัดปริมาณน้ำฝนแบบ analog ไม่ใช่แค่ on/off

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Init + VCC Control

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "RainSensor.h"

#define RAIN_VCC PC0
#define RAIN_ADC PD2
#define RAIN_DIG PC3

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    ADC_Init(RAIN_ADC, ADC_CH0);
    pinMode(RAIN_VCC, PIN_MODE_OUTPUT);
    pinMode(RAIN_DIG, PIN_MODE_INPUT);
    digitalWrite(RAIN_VCC, LOW);
    USART_Init(115200);

    USART_Print("Rain Sensor Ready\r\n");

🔍 VCC ควบคุมด้วย GPIO — เปิดเฉพาะตอนวัด — ป้องกันแผ่นเซนเซอร์กัดกร่อน — Digital อ่านสถานะเปียก/แห้ง

ขั้นที่ 2: อ่านค่า + แปลผลระดับฝน

    digitalWrite(RAIN_VCC, HIGH);
    Delay_Ms(50);

    uint16_t adc = ADC_Read(RAIN_ADC);
    uint8_t dig = GPIO_Read(RAIN_DIG);

    digitalWrite(RAIN_VCC, LOW); // ปิดทันที!

    USART_Print("Rain ADC:"); USART_PrintDec(adc);

    if(dig) {
        USART_Print(" HEAVY RAIN!");
    } else if(adc > 700) {
        USART_Print(" Moderate rain");
    } else if(adc > 400) {
        USART_Print(" Light rain / Drizzle");
    } else {
        USART_Print(" Dry");
    }
    USART_Print("\r\n");

🔍 เปิด VCC → รอ 50ms → อ่าน ADC + Digital → ปิด VCC ทันที — แบ่งระดับ: >700=ฝนปานกลาง, >400=ฝนเบา, <400=แห้ง — Digital=ฝนหนัก (น้ำท่วมแผ่น)

ขั้นที่ 3: ระบบแจ้งเตือนฝนตก + กันขโมย

    pinMode(PC1, PIN_MODE_OUTPUT); // Buzzer
    pinMode(PC2, PIN_MODE_OUTPUT); // Servo ปิดหลังคา
    Timer_t tmr;
    Start_Timer(&tmr, 5000, 1);

    uint8_t rain_streak = 0;

    while(1) {
        if(Is_Timer_Expired(&tmr)) {
            digitalWrite(RAIN_VCC, HIGH);
            Delay_Ms(50);
            uint16_t adc = ADC_Read(RAIN_ADC);
            uint8_t dig = GPIO_Read(RAIN_DIG);
            digitalWrite(RAIN_VCC, LOW);

            if(dig && adc > 800) {
                rain_streak++;
                if(rain_streak >= 3) {
                    USART_Print("RAIN CONFIRMED!\r\n");
                    digitalWrite(PC1, HIGH); // Buzzer
                    digitalWrite(PC2, HIGH); // ปิดหลังคา
                }
            } else if(adc < 300) {
                if(rain_streak > 0) rain_streak--;
                if(rain_streak == 0) {
                    digitalWrite(PC1, LOW);
                    digitalWrite(PC2, LOW); // เปิดหลังคา
                }
            }

            USART_Print("ADC:"); USART_PrintDec(adc);
            USART_Print(" S:"); USART_PrintDec(rain_streak);
            USART_Print("\r\n");
        }
    }
}

🔍 ใช้ Debounce 3 ครั้ง (15 วิ) ยืนยันฝนตก — ป้องกัน false alarm จากละออง — เมื่อยืนยัน → Buzzer + Servo ปิดหลังคา — เมื่อแห้ง → กลับสู่ปกติ

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
แผ่นเซนเซอร์เป็นสนิมเขียวเปิดไฟเลี้ยงค้าง — electrolysis กัดทองแดงใช้ GPIO เปิดเฉพาะตอนวัด — ปิดทันทีหลังอ่าน — เคลือบแผ่นด้วย conformal coating
ตรวจจับฝนทั้งที่ไม่มีฝนน้ำค้าง / ความชื้นสูง / ฝุ่นบนแผ่นใช้ debounce — threshold สูงพอ — ทำความสะอาดแผ่นสม่ำเสมอ
Digital ไม่เคย HIGHPot ตั้ง threshold สูงเกินหมุน Pot — ใช้หยดน้ำทดสอบ — ดู LED บนโมดูล
น้ำขังบนแผ่นวางแผ่นราบเกินไปเอียงแผ่น ~15° — ให้น้ำไหลทิ้ง — เจาะรูระบายน้ำ
ADC อ่านค่าแปลกแผ่น corrode จนลายทองแดงขาดตรวจสอบลายทองแดง — เปลี่ยนแผ่นใหม่ — ใช้แผ่นเคลือบทอง

🛠️ สร้าง Product: ระบบกันฝนอัตโนมัติ

YL-83 + Servo + Buzzer — ตรวจจับฝน → ปิดหลังคา/หน้าต่าง — เมื่อแห้ง → เปิดกลับ

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "RainSensor.h"
#include "Servo.h"

#define RAIN_VCC PC0
#define RAIN_ADC PD2
#define RAIN_DIG PC3
#define SERVO_PIN PC4
#define BUZZER PC1

int main(void)
{
    SystemCoreClockUpdate();
    Timer_Init();
    ADC_Init(RAIN_ADC, ADC_CH0);
    pinMode(RAIN_VCC, PIN_MODE_OUTPUT);
    pinMode(RAIN_DIG, PIN_MODE_INPUT);
    pinMode(BUZZER, PIN_MODE_OUTPUT);
    Servo_Init(SERVO_PIN);
    USART_Init(115200);

    digitalWrite(RAIN_VCC, LOW);
    digitalWrite(BUZZER, LOW);
    Servo_SetAngle(SERVO_PIN, 0); // เปิด

    uint8_t roof_open = 1;
    uint8_t rain_confirmed = 0;
    Timer_t tmr;
    Start_Timer(&tmr, 3000, 1);

    while(1)
    {
        if(Is_Timer_Expired(&tmr))
        {
            digitalWrite(RAIN_VCC, HIGH);
            Delay_Ms(50);
            uint16_t adc = ADC_Read(RAIN_ADC);
            uint8_t dig = GPIO_Read(RAIN_DIG);
            digitalWrite(RAIN_VCC, LOW);

            if(dig && adc > 700) {
                rain_confirmed++;
                if(rain_confirmed >= 3 && roof_open) {
                    USART_Print("RAIN! Closing roof\r\n");
                    digitalWrite(BUZZER, HIGH);
                    Servo_SetAngle(SERVO_PIN, 90); // ปิดหลังคา
                    roof_open = 0;
                }
            } else if(adc < 300) {
                if(rain_confirmed > 0) rain_confirmed--;
                if(rain_confirmed == 0 && !roof_open) {
                    USART_Print("Dry. Opening roof\r\n");
                    digitalWrite(BUZZER, LOW);
                    Servo_SetAngle(SERVO_PIN, 0); // เปิด
                    roof_open = 1;
                }
            }

            USART_Print("ADC:"); USART_PrintDec(adc);
            USART_Print(" Roof:"); USART_Print(roof_open ? "OPEN" : "CLOSED");
            USART_Print("\r\n");
        }
    }
}
