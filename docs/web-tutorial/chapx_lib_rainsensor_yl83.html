MCU Pin→อุปกรณ์ขาหมายเหตุ
3.3V→YL-83VCCจ่ายไฟ 3.3V (library จัดการเปิด/ปิดอัตโนมัติ)
PC4 (ADC)←YL-83A0Analog Output — ความชื้น
PC1 (GPIO)←YL-83D0Digital Output — ตรวจจับน้ำ
GND→YL-83GNDGround ร่วม

YL-83 ใช้แผ่น PCB แบบ interleaved traces — เมื่อมีหยดน้ำตกลงบนแผ่น → ความต้านทานลดลง → ADC สูงขึ้น — มี LM393 Comparator สำหรับ Digital Output

📚 หลักการทำงาน

YL-83 Rain Sensor วัดความชื้นบนแผ่นเซนเซอร์ — แผ่นเป็นลายทองแดงสลับฟันปลา — เมื่อน้ำเชื่อมรอยต่อ ความต้านทานลด → ADC สูง — แห้ง = ความต้านทานสูง → ADC ต่ำ

Corrosion Warning: แผ่นเซนเซอร์เป็นทองแดง — ถ้าเปิดไฟเลี้ยงตลอดเวลา จะเกิด electrolysis กัดกร่อนเร็ว — ตัว library จัดการ VCC control ให้อัตโนมัติ (เปิดเฉพาะตอนวัด)

การติดตั้ง: วางแผ่นราบ แต่เอียงเล็กน้อย (~15°) ให้น้ำไหลออก ไม่ขัง — หลีกเลี่ยงแดดจัดเพราะทำให้แห้งเร็ว

🔍 RainSensor_Init(&r, ADC_CH_PC4) — ตั้งค่า ADC + GPIO — library จัดการ VCC control ให้อัตโนมัติเพื่อป้องกัน corrosion

ขั้นที่ 2: ตรวจสอบฝนตก

    if(RainSensor_IsRaining(&r)) {
        USART_Println("RAINING!");
    } else {
        USART_Println("Dry");
    }

🔍 RainSensor_IsRaining(&r) — ตรวจสอบทั้ง Analog + Digital — คืน true เมื่อตรวจพบน้ำฝน — จัดการ VCC เปิด/ปิด อัตโนมัติ

ขั้นที่ 3: ระบบแจ้งเตือนฝนตก + Debounce

    pinMode(PC1, PIN_MODE_OUTPUT); // Buzzer
    pinMode(PC2, PIN_MODE_OUTPUT); // Servo ปิดหลังคา
    Timer_t tmr;
    Start_Timer(&tmr, 5000, 1);

    uint8_t rain_streak = 0;

    while(1) {
        if(Is_Timer_Expired(&tmr)) {
            if(RainSensor_IsRaining(&r)) {
                rain_streak++;
                if(rain_streak >= 3) {
                    USART_Println("RAIN CONFIRMED!");
                    digitalWrite(PC1, HIGH); // Buzzer
                    digitalWrite(PC2, HIGH); // ปิดหลังคา
                }
            } else {
                if(rain_streak > 0) rain_streak--;
                if(rain_streak == 0) {
                    digitalWrite(PC1, LOW);
                    digitalWrite(PC2, LOW); // เปิดหลังคา
                }
            }

            USART_Print("Streak: "); USART_PrintDec(rain_streak);
            USART_Println("");
        }
    }
}

🔍 ใช้ Debounce 3 ครั้ง (15 วิ) ยืนยันฝนตก — ป้องกัน false alarm จากละออง — เมื่อยืนยัน → Buzzer + Servo ปิดหลังคา

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
แผ่นเซนเซอร์เป็นสนิมเขียวเปิดไฟเลี้ยงค้าง — electrolysis กัดทองแดงใช้ library จัดการ VCC อัตโนมัติ — เปิดเฉพาะตอนวัด
ตรวจจับฝนทั้งที่ไม่มีฝนน้ำค้าง / ความชื้นสูง / ฝุ่นบนแผ่นใช้ debounce — threshold สูงพอ — ทำความสะอาดแผ่นสม่ำเสมอ
น้ำขังบนแผ่นวางแผ่นราบเกินไปเอียงแผ่น ~15° — ให้น้ำไหลทิ้ง — เจาะรูระบายน้ำ

🛠️ สร้าง Product: ระบบกันฝนอัตโนมัติ

YL-83 + Servo + Buzzer — ตรวจจับฝน → ปิดหลังคา/หน้าต่าง

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "RainSensor.h"
#include "Servo.h"

#define SERVO_PIN PC4
#define BUZZER PC1

int main(void)
{
    SystemCoreClockUpdate();
    Timer_Init();
    USART_Init(115200);

    RainSensor_Instance r;
    RainSensor_Init(&r, ADC_CH_PC4);

    pinMode(BUZZER, PIN_MODE_OUTPUT);
    Servo_Init(SERVO_PIN);

    digitalWrite(BUZZER, LOW);
    Servo_SetAngle(SERVO_PIN, 0); // เปิด

    uint8_t roof_open = 1;
    uint8_t rain_confirmed = 0;
    Timer_t tmr;
    Start_Timer(&tmr, 3000, 1);

    while(1)
    {
        if(Is_Timer_Expired(&tmr))
        {
            if(RainSensor_IsRaining(&r)) {
                rain_confirmed++;
                if(rain_confirmed >= 3 && roof_open) {
                    USART_Println("RAIN! Closing roof");
                    digitalWrite(BUZZER, HIGH);
                    Servo_SetAngle(SERVO_PIN, 90); // ปิดหลังคา
                    roof_open = 0;
                }
            } else {
                if(rain_confirmed > 0) rain_confirmed--;
                if(rain_confirmed == 0 && !roof_open) {
                    USART_Println("Dry. Opening roof");
                    digitalWrite(BUZZER, LOW);
                    Servo_SetAngle(SERVO_PIN, 0); // เปิด
                    roof_open = 1;
                }
            }
        }
    }
}
