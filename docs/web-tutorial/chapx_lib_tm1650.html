TM1650 เป็น I2C variant ของ TM1637 — address 0x24-0x27 — ต้องมี pull-up 4.7kΩ บน SDA/SCL

รู้จักขา

  • VCC: ไฟเลี้ยง 3.3V
  • GND: ต่อกราวด์ร่วมกับ MCU
  • SDA: ขาข้อมูล I2C ต่อกับ PC1
  • SCL: ขาสัญญาณนาฬิกา I2C ต่อกับ PC2

วิธีต่อทีละขั้น

  1. ต่อ PC1 ไป SDA — ข้อมูล I2C (ต้องมี pull-up 4.7kΩ)
  2. ต่อ PC2 ไป SCL — นาฬิกา I2C (ต้องมี pull-up 4.7kΩ)
  3. ต่อ 3.3V ไป VCC — จ่ายไฟ
  4. ต่อ GND ไป GND — กราวด์ร่วม

📚 หลักการทำงาน

TM1650 เป็น I2C variant ของ TM1637 — ใช้ I2C มาตรฐาน — TM1650_Handle instance — address 0x24-0x27

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Init + Setup

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "TM1650.h"

TM1650_Handle h;

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    I2C_SimpleInit(I2C_SPEED_100K, I2C_PINS_DEFAULT);

    TM1650_Init(&h, 0x24);       // instance-based, address 0x24
    TM1650_SetBrightness(&h, 5);

🔍 TM1650_Init(&h, addr) — instance-based, ต้องเริ่ม I2C ก่อน — address 0x24-0x27

ขั้นที่ 2: แสดงผลรูปแบบต่างๆ

    TM1650_DisplayNumber(&h, 5678);
    Delay_Ms(2000);

    TM1650_DisplayTime(&h, 8, 30);
    Delay_Ms(2000);

    TM1650_DisplayFloat(&h, 3.14f, 2);
    Delay_Ms(2000);

    TM1650_Clear(&h);

🔍 API รับ &h เป็นพารามิเตอร์ — DisplayNumber, DisplayTime, DisplayFloat — Clear() ล้างหน้าจอ

ขั้นที่ 3: Loop แสดง RTC

    Timer_t tmr;
    Start_Timer(&tmr, 1000, 1);
    uint8_t h = 8, m = 30, s = 0;

    while(1) {
        if(Is_Timer_Expired(&tmr)) {
            s++;
            if(s >= 60) { s = 0; m++; }
            if(m >= 60) { m = 0; h++; }
            if(h >= 24) { h = 0; }

            TM1650_DisplayTime(&h, m);
        }
    }
}

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
I2C scan ไม่เจอ TM1650Address ผิดตรวจสอบ jumper A1/A2 — 0x24-0x27

🛠️ สร้าง Product: Real-Time Clock Display

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "TM1650.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    I2C_SimpleInit(I2C_SPEED_100K, I2C_PINS_DEFAULT);

    TM1650_Handle h;
    TM1650_Init(&h, 0x24);
    TM1650_SetBrightness(&h, 5);

    uint8_t hr = 8, min = 30, sec = 0;
    Timer_t tmr;
    Start_Timer(&tmr, 1000, 1);

    while(1) {
        if(Is_Timer_Expired(&tmr)) {
            sec++;
            if(sec >= 60) { sec = 0; min++; }
            if(min >= 60) { min = 0; hr++; }
            if(hr >= 24) { hr = 0; }
            TM1650_DisplayTime(&h, hr, min);
        }
    }
}
