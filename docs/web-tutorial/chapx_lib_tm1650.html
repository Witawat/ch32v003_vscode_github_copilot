TM1650 ใช้ I2C มาตรฐาน — address 0x24-0x27 เลือกได้ด้วย jumper A1/A2 — API คล้าย TM1637 แต่ใช้ I2C ไม่ต้อง bit-bang

📚 หลักการทำงาน

I2C Variant ของ TM1637: TM1650 ใช้โปรโตคอล I2C มาตรฐานแทน bit-bang GPIO — ทำให้ไม่ต้องกังวลเรื่อง interrupt ขัดจังหวะ — เร็วกว่าและน่าเชื่อถือกว่า TM1637

Address Selection: 4 address ให้เลือกผ่าน jumper A1 และ A2 — 0x24 (A1=GND, A2=GND), 0x25 (A1=VCC, A2=GND), 0x26 (A1=GND, A2=VCC), 0x27 (A1=VCC, A2=VCC) — สามารถต่อหลายตัวบน I2C bus เดียวกัน

Compatible API: มีฟังก์ชันเดียวกับ TM1637 — DisplayNumber, DisplayTime, DisplayFloat — เปลี่ยนจาก TM1637 เป็น TM1650 แค่เปลี่ยน Init และใส่ address

Brightness: 8 ระดับ (0-7) — ควบคุมผ่าน I2C command — ระดับ 5-6 เหมาะสำหรับใช้งานปกติ

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Init + Setup

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "TM1650.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    I2C_SimpleInit(I2C_SPEED_100K, I2C_PINS_DEFAULT);

    TM1650_Init(0x24);           // address 0x24 (default)
    TM1650_SetBrightness(5);     // 5/7 brightness

🔍 ต้องเริ่ม I2C ก่อน — TM1650_Init(address) ระบุ address — ใช้ I2C_SimpleInit แทน GPIO bit-bang — SetBrightness เหมือน TM1637

ขั้นที่ 2: แสดงผลรูปแบบต่างๆ

    TM1650_DisplayNumber(5678);     // แสดง 5678
    Delay_Ms(2000);

    TM1650_DisplayTime(8, 30);      // แสดง 08:30
    Delay_Ms(2000);

    TM1650_DisplayFloat(3.14f, 2);  // แสดง 3.14
    Delay_Ms(2000);

    TM1650_Clear();                 // ล้างหน้าจอ

🔍 API เหมือน TM1637 ทุกประการ — DisplayNumber, DisplayTime, DisplayFloat — Clear() ล้างหน้าจอ — ใช้ Delay_Ms หน่วงระหว่างตัวอย่าง

ขั้นที่ 3: Loop แสดง RTC

    Timer_t tmr;
    Start_Timer(&tmr, 1000, 1);
    uint8_t h = 8, m = 30, s = 0;

    while(1) {
        if(Is_Timer_Expired(&tmr)) {
            s++;
            if(s >= 60) { s = 0; m++; }
            if(m >= 60) { m = 0; h++; }
            if(h >= 24) { h = 0; }

            TM1650_DisplayTime(h, m);
        }
    }
}

🔍 นาฬิกาแบบ Non-Blocking — อัปเดตทุก 1 วิ — แสดง hh:mm — ใช้ Timer_t ควบคุม — เรียบง่าย ไม่ต้องใช้ RTC จริง

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
I2C scan ไม่เจอ TM1650Address ผิด — ใช้ jumper เลือก addressตรวจสอบ jumper A1/A2 — ใช้ I2C_Scan() หา address (0x24-0x27)
จอไม่แสดงผลลืมเริ่ม I2C ก่อน TM1650_Initต้องเรียก I2C_SimpleInit() ก่อน TM1650_Init()
DisplayTime แสดงแค่ 3 หลักใช้ DisplayNumber แทน DisplayTimeDisplayTime จัดการ leading zero และ colon ให้ — ใช้ DisplayTime สำหรับเวลา
Colon ติดตลอดเวลาTM1650 มี colon fix — ควบคุมไม่ได้แยกเป็นข้อจำกัดของ chip — TM1652 มี colon control ถ้าต้องการแยก
ต่อหลายตัวบน I2C busAddress ชนกันตั้ง jumper A1/A2 ให้ต่างกัน — รองรับสูงสุด 4 ตัวบน bus เดียว

🛠️ สร้าง Product: Real-Time Clock Display

TM1650 + DS3231 RTC — แสดงเวลา 24 ชม. — ใช้ Non-Blocking Timer อัปเดตทุก 1 วินาที

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "TM1650.h"

RTC_DS3231_t rtc;

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    I2C_SimpleInit(I2C_SPEED_100K, I2C_PINS_DEFAULT);

    TM1650_Init(0x24);
    TM1650_SetBrightness(5);

    if(DS3231_Init(&rtc) != DS3231_OK) {
        TM1650_DisplayNumber(0xEEEE);  // Error display
        while(1);
    }

    RTC_DateTime_t dt;
    Timer_t tmr;
    Start_Timer(&tmr, 1000, 1);

    while(1) {
        if(Is_Timer_Expired(&tmr)) {
            DS3231_GetDateTime(&rtc, &dt);
            TM1650_DisplayTime(dt.hour, dt.minute);
        }
    }
}
