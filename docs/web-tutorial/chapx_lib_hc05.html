HC-05 — Bluetooth USART, AT Mode, Pairing

ระดับกลาง

🧭 แผนภาพการทำงาน

flowchart TD
    A["HC05_Init(USART,baud)"] --> B{"ต้องการ AT Mode?"}

🔌 การต่อวงจร

                  HC-05 Bluetooth Module
    ┌──────────────────────────────────────────────┐
    │ CH32V003                          HC-05      │
    │ PD5(TX)──────────RX               ┌─────────┤
    │ PD6(RX)──────────TX               │         │

    └──────────────────────────────────────────────┘

HC-05 TX 5V output → ควรใช้ voltage divider (1k+2k) ลดเหลือ 3.3V ก่อนเข้า CH32V003 RX เพื่อความปลอดภัย, EN pin HIGH = AT Mode (38400 baud), EN pin LOW/ลอย = Data Mode

📚 หลักการทำงาน

HC-05 เป็น Bluetooth SPP (Serial Port Profile) module — เมื่อจับคู่กับสมาร์ทโฟน/PC แล้ว จะทำงานเหมือน serial cable ไร้สาย: ข้อมูลที่ CH32V003 ส่งผ่าน USART จะไปออกที่ Bluetooth client และข้อมูลที่ client ส่งมาก็จะมาออกที่ขา TX ของ HC-05

2 โหมดการทำงาน:

  • Data Mode (default, baud 9600): สื่อสารปกติ — ส่งข้อมูลไปมาระหว่าง MCU ↔ Bluetooth client
  • AT Mode (EN=HIGH ตอน power on, baud 38400): ตั้งค่า module — เปลี่ยนชื่อ, PIN, baud rate, master/slave

Master vs Slave: Default HC-05 เป็น Slave — รอให้อุปกรณ์อื่นมาเชื่อมต่อ ถ้าต้องการให้ HC-05 เป็นฝ่ายเชื่อมต่อไปหา device อื่น ต้องตั้งเป็น Master ด้วย AT+ROLE=1

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Init + Setup

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "HC05.h"

int main(void) {
    SystemCoreClockUpdate();
    USART_Init(USART1, BAUD_115200);       // debug
    USART_Init(USART2, BAUD_9600);         // HC-05 default baud
    pinMode(PC0, PIN_MODE_OUTPUT);         // EN/KEY pin
    digitalWrite(PC0, LOW);                 // normal mode
    HC05_Init(USART2, BAUD_9600);
    USART_Println("HC-05 Ready!");

🔍 HC-05 default baud = 9600, AT Mode baud = 38400 — ต้องใช้ให้ตรง, ตั้ง EN=LOW เพื่อเข้าสู่ Data Mode ปกติ, USART1 แสดง debug, USART2 ต่อกับ HC-05

ขั้นที่ 2: AT Mode — ตั้งค่า HC-05

    // === เข้า AT Mode ===
    digitalWrite(PC0, HIGH);          // EN=HIGH
    Delay_Ms(100);
    HC05_EnterATMode(USART2);         // เปลี่ยน baud เป็น 38400 ชั่วคราว

    HC05_SendAT("AT");                // test
    HC05_SendAT("AT+NAME=CH32V003");  // เปลี่ยนชื่อ Bluetooth
    HC05_SendAT("AT+PSWD=4321");      // เปลี่ยน PIN
    HC05_SendAT("AT+UART=9600,0,0");  // ตั้ง baud=9600

    digitalWrite(PC0, LOW);           // กลับ Data Mode
    Delay_Ms(500);

🔍 HC05_SendAT(cmd) — ส่ง AT command + รอ OK response, EN=HIGH แล้ว power-on ถึงเข้า AT Mode (ถ้า EN สลับระหว่างรันไทม์ อาจต้องรีเซ็ต HC-05 ก่อน)

ขั้นที่ 3: รับ-ส่งข้อมูลใน Loop

    char rx_buf[64];
    uint8_t idx = 0;
    while(1) {
        if(USART_Available(USART2)) {   // มีข้อมูลจาก HC-05
            char c = USART_Read(USART2);
            USART_Write(USART1, c);     // echo ไป debug
            rx_buf[idx++] = c;
            if(c == '\n' || idx >= 63) {
                rx_buf[idx] = '\0';
                // รับคำสั่ง: "LED1" → toggle PC1
                if(strstr(rx_buf, "LED1")) digitalToggle(PC1);
                idx = 0;
            }
        }
    }
}

🔍 ใช้ USART_Available() เช็คก่อนอ่าน — non-blocking, รับทีละ char สะสมใน buffer, เมื่อเจอ \n หรือเต็ม buffer → ประมวลผลคำสั่ง, สามารถใช้ Serial Bluetooth Terminal app บนมือถือทดสอบได้

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
AT ไม่ตอบEN ไม่ได้ HIGH ตอน power onกดปุ่มค้างไว้ตอนเสียบไฟ หรือต่อ EN กับ VCC แล้วรีเซ็ต
AT ตอบแต่บอดไม่ตรงAT Mode ใช้ baud 38400 เสมอต้องเปลี่ยน USART baud เป็น 38400 ก่อนเข้า AT Mode
Pair ได้แต่ส่งข้อมูลไม่ได้RX/TX สลับข้างMCU TX → HC-05 RX, MCU RX → HC-05 TX — ไขว้สาย
ข้อมูลเพี้ยน/ขยะHC-05 TX 5V เข้า CH32V003 RX เกินใช้ voltage divider 1k+2k ลดจาก 5V → 3.3V
เชื่อมต่อได้ครั้งเดียวแล้วหลุดState pin ต่อผิด หรือไฟไม่เสถียรเช็ค STATE pin, เพิ่ม capacitor 10µF ที่ VCC

🛠️ สร้าง Product: Bluetooth LED Controller

ควบคุม LED 4 ดวงผ่าน Bluetooth จากสมาร์ทโฟน — ส่งข้อความ "LED1 ON", "LED2 OFF" ฯลฯ

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "HC05.h"
#include 

int main(void)
{
    SystemCoreClockUpdate();
    USART_Init(USART1, BAUD_115200);
    USART_Init(USART2, BAUD_9600);
    HC05_Init(USART2, BAUD_9600);

    pinMode(PC0, PIN_MODE_OUTPUT);  // LED1
    pinMode(PC1, PIN_MODE_OUTPUT);  // LED2
    pinMode(PC2, PIN_MODE_OUTPUT);  // LED3
    pinMode(PC3, PIN_MODE_OUTPUT);  // LED4

    char buf[32]; uint8_t idx = 0;
    USART_Println("BT LED Controller Ready");

    while(1)
    {
        if(USART_Available(USART2)) {
            char c = USART_Read(USART2);
            if(c == '\n' || c == '\r') {
                buf[idx] = '\0';
                if(strstr(buf, "LED1 ON"))  digitalWrite(PC0, HIGH);
                if(strstr(buf, "LED1 OFF")) digitalWrite(PC0, LOW);
                if(strstr(buf, "LED2 ON"))  digitalWrite(PC1, HIGH);
                if(strstr(buf, "LED2 OFF")) digitalWrite(PC1, LOW);
                if(strstr(buf, "LED3 ON"))  digitalWrite(PC2, HIGH);
                if(strstr(buf, "LED3 OFF")) digitalWrite(PC2, LOW);
                if(strstr(buf, "LED4 ON"))  digitalWrite(PC3, HIGH);
                if(strstr(buf, "LED4 OFF")) digitalWrite(PC3, LOW);
                if(strstr(buf, "ALL ON")) {
                    digitalWrite(PC0, HIGH); digitalWrite(PC1, HIGH);
                    digitalWrite(PC2, HIGH); digitalWrite(PC3, HIGH);
                }
                if(strstr(buf, "ALL OFF")) {
                    digitalWrite(PC0, LOW); digitalWrite(PC1, LOW);
                    digitalWrite(PC2, LOW); digitalWrite(PC3, LOW);
                }
                idx = 0;
            } else if(idx < 31) {
                buf[idx++] = c;
            }
        }
    }
}
