HC-05 — Bluetooth USART, AT Mode, Pairing

ระดับกลาง

🧭 แผนภาพการทำงาน

flowchart TD
    A["HC05_Init(&bt, BAUD_38400)"] --> B{"ต้องการ AT Mode?"}
    B -->|"ใช่"| C["กดปุ่มค้าง + ตอน power on"]
    B -->|"ไม่"| D["Normal Mode — เชื่อมต่อ Bluetooth"]
    C --> E["HC05_SendString(&bt, cmd)"]
    D --> F["สมาร์ทโฟน/PC ค้นหา HC-05"]
    F --> G["Pair (PIN: 1234)"]
    G --> H["ส่ง/รับข้อมูลผ่าน USART"]

🔌 การต่อวงจร

                  HC-05 Bluetooth Module
    ┌──────────────────────────────────────────────┐
    │ CH32V003                          HC-05      │
    │ PD5(TX)──────────RX               ┌─────────┤
    │ PD6(RX)──────────TX               │         │
    │ 5V───────────────VCC              │         │
    │ GND──────────────GND              │         │
    └──────────────────────────────────────────────┘
MCU Pin→อุปกรณ์ขาหมายเหตุ
PD5 (TX)→HC-05RXMCU→HC-05
PD6 (RX)←HC-05TXHC-05→MCU
5V↔HC-05VCCจ่ายไฟ 5V
GND↔HC-05GNDกราวด์ร่วม

รู้จักขา

  • VCC: ไฟเลี้ยง 5V (⚠️ HC-05 ใช้ 5V ไม่ใช่ 3.3V)
  • GND: กราวด์
  • RX: รับข้อมูลจาก MCU — ต่อขา PD5 (TX) ของ MCU
  • TX: ส่งข้อมูลไป MCU — ต่อขา PD6 (RX) ของ MCU

วิธีต่อทีละขั้น

  1. ต่อ VCC HC-05 ไป 5V ของ MCU
  2. ต่อ GND HC-05 ไป GND ของ MCU
  3. ต่อ RX HC-05 ไปขา PD5 (TX) ของ MCU
  4. ต่อ TX HC-05 ไปขา PD6 (RX) ของ MCU

📚 หลักการทำงาน

HC-05 เป็น Bluetooth SPP module — เมื่อจับคู่กับสมาร์ทโฟน/PC แล้ว ทำงานเหมือน serial cable ไร้สาย — HC05_SendString() ส่งข้อความ

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Init + Setup

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "HC05.h"

HC05_Instance bt;

int main(void) {
    SystemCoreClockUpdate();
    USART_Init(USART1, BAUD_115200);       // debug

    HC05_Init(&bt, BAUD_38400);
    USART_Println("HC-05 Ready!");

🔍 HC05_Init(&bt, baud) — instance-based, HC-05 default baud = 38400

ขั้นที่ 2: ส่งข้อความ

    HC05_SendString(&bt, "AT\r\n");
    Delay_Ms(500);

    HC05_SendString(&bt, "Hello from CH32V003!\r\n");

🔍 HC05_SendString(&bt, str) — ส่ง string ผ่าน Bluetooth — ใช้\r\n เป็นตัวจบบรรทัด

ขั้นที่ 3: รับ-ส่งข้อมูลใน Loop

    char rx_buf[64];
    uint8_t idx = 0;
    while(1) {
        if(HC05_Available(&bt)) {
            uint8_t c; HC05_ReadByte(&bt, &c);
            USART_Write(USART1, c);     // echo to debug
            rx_buf[idx++] = c;
            if(c == '\n' || idx >= 63) {
                rx_buf[idx] = '\0';
                if(strstr(rx_buf, "LED1")) digitalToggle(PC1);
                idx = 0;
            }
        }
    }
}

🔍 HC05_Available(&bt) เช็คก่อนอ่าน — non-blocking, รับทีละ char สะสมใน buffer

📋 API Reference

ฟังก์ชันคำอธิบายตัวอย่าง
HC05_Init(&bt, baud)เริ่มต้น HC-05HC05_Init(&bt, BAUD_38400)
HC05_SendString(&bt, str)ส่ง stringHC05_SendString(&bt, "AT\r\n")

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
AT ไม่ตอบไม่ได้เข้า AT Modeกดปุ่มค้างไว้ตอนเสียบไฟ
Pair ได้แต่ส่งข้อมูลไม่ได้RX/TX สลับข้างMCU TX → HC-05 RX, MCU RX → HC-05 TX

🛠️ สร้าง Product: Bluetooth LED Controller

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "HC05.h"
#include 

int main(void)
{
    SystemCoreClockUpdate();
    USART_Init(USART1, BAUD_115200);

    HC05_Instance bt;
    HC05_Init(&bt, BAUD_38400);

    pinMode(PC0, PIN_MODE_OUTPUT);  // LED1
    pinMode(PC1, PIN_MODE_OUTPUT);  // LED2

    char buf[32]; uint8_t idx = 0;
    USART_Println("BT LED Controller Ready");

    while(1)
    {
        if(HC05_Available(&bt)) {
            uint8_t c; HC05_ReadByte(&bt, &c);
            if(c == '\n' || c == '\r') {
                buf[idx] = '\0';
                if(strstr(buf, "LED1 ON"))  digitalWrite(PC0, HIGH);
                if(strstr(buf, "LED1 OFF")) digitalWrite(PC0, LOW);
                if(strstr(buf, "LED2 ON"))  digitalWrite(PC1, HIGH);
                if(strstr(buf, "LED2 OFF")) digitalWrite(PC1, LOW);
                idx = 0;
            } else if(idx < 31) {
                buf[idx++] = c;
            }
        }
    }
}
