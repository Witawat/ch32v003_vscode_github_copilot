รู้จักขา

  • VCC: ไฟเลี้ยง 3.3V
  • GND: กราวด์
  • RX: รับข้อมูลจาก MCU — ต่อขา PD5 (TX) ของ MCU
  • TX: ส่งข้อมูลไป MCU — ต่อขา PD6 (RX) ของ MCU

วิธีต่อทีละขั้น

  1. ต่อ VCC NEO-6M ไป 3.3V ของ MCU
  2. ต่อ GND NEO-6M ไป GND ของ MCU
  3. ต่อ RX NEO-6M ไปขา PD5 (TX) ของ MCU
  4. ต่อ TX NEO-6M ไปขา PD6 (RX) ของ MCU

📚 หลักการทำงาน

GPS NEO-6M รับสัญญาณจากดาวเทียม GPS — ส่งข้อมูลออกเป็น NMEA sentences ผ่าน USART ที่ 9600 baud (default) — GPS_Update() parse NMEA, GPS_GetLatitude()/GPS_GetLongitude() คืนค่าพิกัด

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Init + Setup

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "GPS.h"

GPS_Instance gps;

int main(void) {
    SystemCoreClockUpdate();
    USART_Init(USART1, BAUD_115200);  // debug

    GPS_Init(&gps, BAUD_9600, USART_PINS_DEFAULT);
    USART_Println("GPS Starting...");

🔍 GPS_Init(&gps, baud, pins) — instance-based, baud 9600 default, pins=USART_PINS_DEFAULT

ขั้นที่ 2: รอ Fix + อ่านพิกัด

    while(!GPS_IsFixValid(&gps)) {
        GPS_Update(&gps);
        USART_Println("Waiting for GPS fix...");
        Delay_Ms(1000);
    }
    USART_Println("GPS Fix acquired!");

    float lat = GPS_GetLatitude(&gps);
    float lon = GPS_GetLongitude(&gps);
    USART_Print("Lat: "); USART_PrintFloat(lat, 6);
    USART_Print(" Lon: "); USART_PrintFloat(lon, 6);
    USART_Println("");

🔍 GPS_Update(&gps) — ประมวลผล NMEA แบบ char-by-char, GPS_GetLatitude(&gps) / GPS_GetLongitude(&gps) — คืนค่าพิกัด decimal

ขั้นที่ 3: Loop — อ่านและแสดงผล

    while(1) {
        GPS_Update(&gps);

        if(GPS_IsFixValid(&gps)) {
            float lat = GPS_GetLatitude(&gps);
            float lon = GPS_GetLongitude(&gps);
            float spd = GPS_GetSpeed(&gps);
            uint8_t sat = GPS_GetSatellites(&gps);

            USART_Print("Lat:"); USART_PrintFloat(lat, 6);
            USART_Print(" Lon:"); USART_PrintFloat(lon, 6);
            USART_Print(" Spd:"); USART_PrintFloat(spd, 1);
            USART_Print(" Sat:"); USART_PrintInt(sat);
            USART_Println("");
        }
        Delay_Ms(500);
    }
}

🔍 GPS_Update() ต้องถูกเรียกตลอดเวลาเพื่อ parse NMEA — GPS_IsFixValid() เช็คว่ามี GPS fix หรือยัง

🛠️ สร้าง Product: GPS Speedometer

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "GPS.h"

int main(void)
{
    SystemCoreClockUpdate();
    USART_Init(USART1, BAUD_115200);

    GPS_Instance gps;
    GPS_Init(&gps, BAUD_9600, USART_PINS_DEFAULT);

    while(!GPS_IsFixValid(&gps)) { GPS_Update(&gps); Delay_Ms(500); }

    while(1)
    {
        GPS_Update(&gps);
        if(GPS_IsFixValid(&gps)) {
            float lat = GPS_GetLatitude(&gps);
            float lon = GPS_GetLongitude(&gps);
            float spd = GPS_GetSpeed(&gps);
            USART_Print("Speed:"); USART_PrintFloat(spd, 1);
            USART_Print(" km/h Lat:"); USART_PrintFloat(lat, 6);
            USART_Print(" Lon:"); USART_PrintFloat(lon, 6);
            USART_Println("");
        }
        Delay_Ms(500);
    }
}

✏️ ลองทำดู

  1. ทำ GPS data logger — บันทึก lat/lon/time ลง EEPROM ทุก 10 วิ
  2. ทำ geofence alarm — ถ้า GPS อยู่นอกขอบเขตที่กำหนด → LED ติด
  3. ทำ GPS clock — ใช้ $GPRMC time stamp ตั้งเวลาให้ DS3231
