GPS module ส่งข้อมูลออกทาง TX ตลอดเวลาที่ 9600 baud (default) — ต่อ RX ของ CH32V003 เข้ากับ TX ของ GPS ก็พอ (ขา TX ของ CH32V003 ไม่จำเป็นต้องต่อ ยกเว้นต้องการส่งคำสั่ง config)

📚 หลักการทำงาน

GPS NEO-6M รับสัญญาณจากดาวเทียม GPS อย่างน้อย 3 ดวงเพื่อระบุตำแหน่ง 2D fix, 4+ ดวงเพื่อ 3D fix + altitude — ส่งข้อมูลออกเป็น NMEA sentences ผ่าน USART ที่ 9600 baud (default)

NMEA Sentences สำคัญ:

  • $GPRMC — Recommended Minimum: time, status, lat, lon, speed, date — สำคัญที่สุด
  • $GPGGA — Fix data: quality, satellites, altitude, HDOP
  • $GPVTG — Course and speed over ground

Fix Quality: 0=no fix, 1=GPS fix, 2=DGPS fix — ต้องมี fix ก่อนค่าพิกัดถึงจะน่าเชื่อถือ

Baud Rate: Default 9600 — เปลี่ยนได้ด้วย UBX commands แต่ 9600 เพียงพอสำหรับ 1Hz update

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Init + Setup

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "GPS_NEO6M.h"

GPS_Data gps;  // struct: lat, lon, speed, time, fix

int main(void) {
    SystemCoreClockUpdate();
    USART_Init(USART1, BAUD_115200);     // debug
    USART_Init(USART2, BAUD_9600);       // GPS TX connected to MCU RX
    GPS_Init(USART2, BAUD_9600);
    USART_Println("GPS Starting...");

🔍 GPS_Init(uart, baud) — เริ่ม USART สำหรับรับ NMEA, struct GPS_Data เก็บค่าที่ parse แล้ว, GPS ต้องอยู่กลางแจ้ง view ท้องฟ้าเพื่อรับสัญญาณ

ขั้นที่ 2: Parse NMEA + ตรวจสอบ Fix

    while(!GPS_HasFix()) {          // รอจนได้ fix
        if(USART_Available(USART2)){
            GPS_Process();           // feed NMEA char-by-char
        }
        USART_Println("Waiting for GPS fix...");
        Delay_Ms(1000);
    }
    USART_Println("GPS Fix acquired!");

🔍 GPS_Process() — อ่าน USART buffer ทีละ char, parse $GPRMC sentence, GPS_HasFix() — เช็ค Status='A' (Active/Va​lid) ต้อง 'A' เท่านั้นถึงจะเชื่อถือพิกัดได้

ขั้นที่ 3: Loop — อ่านและแสดงผล

    while(1) {
        if(USART_Available(USART2)) GPS_Process();

        GPS_GetData(&gps);  // read latest parsed data
        if(gps.fix) {
            USART_Print("Lat: ");  USART_PrintFloat(gps.lat, 6);
            USART_Print(" Lon: "); USART_PrintFloat(gps.lon, 6);
            USART_Print(" Spd: "); USART_PrintFloat(gps.speed_kmh, 1);
            USART_Print(" km/h Sat: "); USART_PrintInt(gps.satellites);
            USART_Print(" Time: "); USART_Println(gps.time_str);
        }
        Delay_Ms(500);
    }
}

🔍 GPS_GetData(&gps) — คัดลอกข้อมูล GPS ล่าสุดเข้า struct, latitude/longitude เป็นองศา (ddmm.mmmm format → แปลงเป็น decimal), speed เป็น km/h, time เป็น HHMMSS format

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
ไม่มี NMEA ออกมาเลยBaud rate ไม่ตรง (default 9600)เช็ค GPS_Init(USART2, BAUD_9600), ลอง baud อื่น
fix ไม่เคยได้อยู่ภายในอาคาร — สัญญาณ GPS ทะลุไม่ถึงต้องอยู่กลางแจ้งหรือใกล้หน้าต่าง, cold start อาจใช้เวลา 5-15 นาที
ค่า lat/lon เพี้ยนใช้ Status='V' (Invalid)เช็ค GPS_HasFix() ก่อน — ต้อง 'A' เท่านั้น
GPS_Process() นานเกินบัฟเฟอร์ล้น — อ่านไม่ทันเรียก GPS_Process() บ่อยๆ ใน loop, อย่า delay นานเกิน 100ms
ตำแหน่งกระโดดMultipath หรือสัญญาณอ่อนใช้ averaging หรือ Kalman filter, รอให้ดาวเทียม ≥6 ดวง

🛠️ สร้าง Product: GPS Speedometer + Logger

อ่านความเร็วจาก GPS + แสดงผ่าน USART + บันทึกตำแหน่งล่าสุด

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "GPS_NEO6M.h"
#include 

GPS_Data gps;
float total_dist = 0;
float prev_lat = 0, prev_lon = 0;

float haversine(float lat1, float lon1, float lat2, float lon2) {
    float R = 6371000; // metres
    float dlat = (lat2-lat1) * 3.14159/180;
    float dlon = (lon2-lon1) * 3.14159/180;
    float a = sin(dlat/2)*sin(dlat/2) +
              cos(lat1*3.14159/180)*cos(lat2*3.14159/180)*sin(dlon/2)*sin(dlon/2);
    return R * 2 * atan2(sqrt(a), sqrt(1-a));
}

int main(void)
{
    SystemCoreClockUpdate();
    USART_Init(USART1, BAUD_115200);
    USART_Init(USART2, BAUD_9600);
    GPS_Init(USART2, BAUD_9600);

    while(!GPS_HasFix()) { GPS_Process(); Delay_Ms(500); }
    GPS_GetData(&gps);
    prev_lat = gps.lat; prev_lon = gps.lon;

    while(1)
    {
        GPS_Process();
        GPS_GetData(&gps);
        if(gps.fix) {
            float dist = haversine(prev_lat, prev_lon, gps.lat, gps.lon);
            if(dist < 100) total_dist += dist; // filter glitch
            prev_lat = gps.lat; prev_lon = gps.lon;

            USART_Print("Speed:"); USART_PrintFloat(gps.speed_kmh,1);
            USART_Print(" km/h Dist:"); USART_PrintFloat(total_dist/1000,3);
            USART_Print(" km Sat:"); USART_PrintInt(gps.satellites);
            USART_Println("");
        }
        Delay_Ms(500);
    }
}

✏️ ลองทำดู

  1. ทำ GPS data logger — บันทึก lat/lon/time ลง EEPROM ทุก 10 วินาที → ดึงข้อมูลผ่าน USART ทีหลัง
  2. ทำ geofence alarm — ถ้า GPS อยู่นอกขอบเขตที่กำหนด (รั้วเสมือน) → LED ติด/ส่ง SMS ผ่าน ESP8266
  3. ทำ GPS clock — ใช้ $GPRMC time stamp ตั้งเวลาให้ DS3231 RTC อัตโนมัติ
