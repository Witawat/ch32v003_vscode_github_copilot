📝 เฉลย — YL-83 Rain Sensor

ข้อ 1: Rain Alert — แจ้งเตือนเมื่อฝนตกด้วย Buzzer + LED

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "RainSensor.h"
#include "oled_i2c.h"

RainSensor_Instance r;
#define RAIN_VCC PC0
#define RAIN_ADC PD2
#define RAIN_DIG PC3
#define BUZZER   PC1
#define LED_WARN PC2

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    I2C_Init(PC1, PC2, I2C_SPEED_STANDARD);
    RainSensor_Init(&r, ADC_CH0);
    GPIO_SimpleInit(RAIN_VCC, GPIO_OUT_PP);
    GPIO_SimpleInit(RAIN_DIG, GPIO_IN_PU);
    GPIO_SimpleInit(BUZZER, GPIO_OUT_PP);
    GPIO_SimpleInit(LED_WARN, GPIO_OUT_PP);
    USART_Init(115200);

    GPIO_WriteBit(RAIN_VCC, 0);
    GPIO_WriteBit(BUZZER, 0);
    GPIO_WriteBit(LED_WARN, 0);

    OLED_Handle oled;
    OLED_Init(&oled, OLED_128x64, 0x3C);
    OLED_SetFont(&oled, FONT_8x16);

    uint8_t rain_streak = 0;
    uint8_t alert_active = 0;
    uint8_t last_dry_streak = 0;
    Timer_t tmr;
    Start_Timer(&tmr, 5000, 1);

    USART_Print("Rain Alert System Ready\r\n");

    while(1) {
        if(Is_Timer_Expired(&tmr)) {
            GPIO_WriteBit(RAIN_VCC, 1);
            Delay_Ms(50);
            uint16_t adc = ADC_Read(RAIN_ADC);
            uint8_t dig = GPIO_ReadBit(RAIN_DIG);
            GPIO_WriteBit(RAIN_VCC, 0);

            const char* status;
            if(dig && adc > 800) {
                rain_streak++;
                last_dry_streak = 0;
                status = "HEAVY RAIN!";

                if(rain_streak >= 3 && !alert_active) {
                    alert_active = 1;
                    GPIO_WriteBit(BUZZER, 1);
                    GPIO_WriteBit(LED_WARN, 1);
                    USART_Print("RAIN ALERT! Bring laundry in!\r\n");
                }
            } else if(adc > 500) {
                rain_streak++;
                last_dry_streak = 0;
                status = "Rain";
            } else if(adc > 300) {
                status = "Drizzle";
                rain_streak = 0;
                last_dry_streak++;
            } else {
                status = "Dry";
                rain_streak = 0;
                last_dry_streak++;
            }

            if(last_dry_streak >= 6 && alert_active) {
                alert_active = 0;
                GPIO_WriteBit(BUZZER, 0);
                GPIO_WriteBit(LED_WARN, 0);
                USART_Print("Rain stopped, alert cleared\r\n");
            }

            OLED_Clear(&oled);
            OLED_SetCursor(&oled, 0, 0);
            OLED_Print(&oled, "Rain Alert");

            OLED_SetCursor(&oled, 0, 2);
            OLED_Print(&oled, status);
            OLED_Print(&oled, "  ADC:");
            OLED_PrintDec(&oled, adc);

            OLED_SetCursor(&oled, 0, 4);
            OLED_Print(&oled, "Streak: ");
            OLED_PrintDec(&oled, rain_streak);
            OLED_Print(&oled, "/3");

            OLED_SetCursor(&oled, 0, 6);
            OLED_Print(alert_active ? "<< ALERT >>" : "Status: OK");
            OLED_Update(&oled);

            USART_Print(status); USART_Print(" ADC:");
            USART_PrintDec(adc); USART_Print(" D:");
            USART_PrintDec(dig); USART_Print(" S:");
            USART_PrintDec(rain_streak); USART_Print("\r\n");
        }
    }
}
🧠 แนวคิด: วัดทุก 5 วิ — 3 ระดับ: Heavy Rain (dig+adc>800), Rain (>500), Drizzle (>300), Dry (<300) — debounce 3 ครั้ง (15 วิ) ยืนยันฝนตก — last_dry_streak 6 ครั้ง (30 วิ) ยืนยันฝนหยุด → reset alert — VCC เปิดเฉพาะตอนวัด + ปิดทันที — OLED แสดงสถานะแบบ real-time — ใช้เตือนให้เก็บผ้า, ปิดหน้าต่าง, หรือแจ้งเตือนทั่วไป

ข้อ 2: Smart Garden Combo — Rain Sensor + Soil Moisture

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "RainSensor.h"

RainSensor_Instance r;
SoilMoisture_Instance s;
#define RAIN_VCC   PC0
#define RAIN_ADC   PD2
#define RAIN_DIG   PC3
#define SOIL_VCC   PC4
#define SOIL_ADC   PD4
#define SOIL_CH    ADC_CH1
#define PUMP_PIN   PC5
#define ALERT_LED  PC1

uint16_t read_sensor_raw(uint8_t vcc_pin, uint8_t adc_pin) {
    GPIO_WriteBit(vcc_pin, 1);
    Delay_Ms(60);
    uint16_t val = ADC_Read(adc_pin);
    GPIO_WriteBit(vcc_pin, 0);
    return val;
}

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    ADC_Init(RAIN_ADC, ADC_CH0);
    ADC_Init(SOIL_ADC, SOIL_CH);
    RainSensor_Init(&r, ADC_CH0);
    SoilMoisture_Init(&s, SOIL_CH);
    GPIO_SimpleInit(RAIN_VCC, GPIO_OUT_PP);
    GPIO_SimpleInit(RAIN_DIG, GPIO_IN_PU);
    GPIO_SimpleInit(SOIL_VCC, GPIO_OUT_PP);
    GPIO_SimpleInit(PUMP_PIN, GPIO_OUT_PP);
    GPIO_SimpleInit(ALERT_LED, GPIO_OUT_PP);
    USART_Init(115200);

    GPIO_WriteBit(RAIN_VCC, 0);
    GPIO_WriteBit(SOIL_VCC, 0);
    GPIO_WriteBit(PUMP_PIN, 0);
    GPIO_WriteBit(ALERT_LED, 0);

    Timer_t tmr;
    Start_Timer(&tmr, 30000, 1);

    uint8_t pump_blocked = 0;
    uint8_t rain_today = 0;

    USART_Print("Smart Garden: Rain+Soil Combo\r\n");

    while(1) {
        if(Is_Timer_Expired(&tmr)) {
            uint16_t rain_adc = read_sensor_raw(RAIN_VCC, RAIN_ADC);
            uint8_t  rain_dig = GPIO_ReadBit(RAIN_DIG);
            float soil_moisture = SoilMoisture_Read(&s);

            uint8_t is_raining = RainSensor_IsRaining(&r);
            if(is_raining) rain_today = 1;

            if(soil_moisture < 0) soil_moisture = 0;
            if(soil_moisture > 100) soil_moisture = 100;

            USART_Print("Rain:"); USART_PrintDec(rain_adc);
            USART_Print("/"); USART_PrintDec(rain_dig);
            USART_Print(" Soil:"); USART_PrintFloat(soil_moisture, 1);
            USART_Print("%");

            if(is_raining && !pump_blocked) {
                pump_blocked = 1;
                USART_Print(" RAIN - Pump blocked\r\n");
                GPIO_WriteBit(ALERT_LED, 1);
            } else if(!is_raining && pump_blocked && !rain_today) {
                pump_blocked = 0;
                USART_Print(" Dry - Pump unblocked\r\n");
                GPIO_WriteBit(ALERT_LED, 0);
            }

            if(!is_raining && soil_moisture < 30.0f && !pump_blocked) {
                USART_Print(" WATERING\r\n");
                GPIO_WriteBit(PUMP_PIN, 1);
                Delay_Ms(5000);
                GPIO_WriteBit(PUMP_PIN, 0);
            } else {
                USART_Print("\r\n");
            }
        }
    }
}
🧠 แนวคิด: รวม Rain Sensor + Soil Moisture — ตรรกะอัจฉริยะ: ถ้าฝนตก → ห้ามรดน้ำ (pump_blocked) แม้ดินจะแห้ง — rain_today flag บอกว่าวันนี้ฝนตกแล้ว — ฝนหยุด + ไม่มีฝนตกวันนี้ → ปลดล็อกปั๊ม — ฟังก์ชัน read_sensor() reuse สำหรับทั้งสองเซนเซอร์ (เปิด VCC→อ่าน→ปิด) — ลดการใช้น้ำเปลืองเมื่อฝนตก — ถ้าดินเปียกจากฝน → ไม่ต้องรดน้ำ → ประหยัดน้ำ

ข้อ 3: Rain Gauge — วัดปริมาณน้ำฝนสะสม

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "RainSensor.h"

RainSensor_Instance r;
#define RAIN_VCC  PC0
#define RAIN_ADC  PD2
#define RAIN_DIG  PC3

/*
  การวัดปริมาณน้ำฝนโดยประมาณ:
  - ADC เพิ่มขึ้น 1 หน่วย ≈ 0.01mm (ต้อง calibrate)
  - สะสมค่าต่างตั้งแต่เริ่ม
*/

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    RainSensor_Init(&r, ADC_CH0);
    GPIO_SimpleInit(RAIN_VCC, GPIO_OUT_PP);
    GPIO_SimpleInit(RAIN_DIG, GPIO_IN_PU);
    USART_Init(115200);

    GPIO_WriteBit(RAIN_VCC, 0);

    GPIO_WriteBit(RAIN_VCC, 1);
    Delay_Ms(50);
    uint16_t baseline_adc = ADC_Read(RAIN_ADC);
    GPIO_WriteBit(RAIN_VCC, 0);

    float total_rain_mm = 0;
    uint16_t prev_adc = baseline_adc;
    Timer_t tmr;
    Start_Timer(&tmr, 10000, 1);

    USART_Print("Rain Gauge Ready\r\n");
    USART_Print("Baseline ADC: ");
    USART_PrintDec(baseline_adc);
    USART_Print("\r\n");

    while(1) {
        if(Is_Timer_Expired(&tmr)) {
            GPIO_WriteBit(RAIN_VCC, 1);
            Delay_Ms(50);
            uint16_t adc = ADC_Read(RAIN_ADC);
            uint8_t dig = GPIO_ReadBit(RAIN_DIG);
            GPIO_WriteBit(RAIN_VCC, 0);

            if(adc > prev_adc + 5) {
                float rain_added = (adc - prev_adc) * 0.015f;
                total_rain_mm += rain_added;
            }
            prev_adc = adc;

            USART_Print("ADC:"); USART_PrintDec(adc);
            USART_Print(" Rain:"); USART_PrintFloat(total_rain_mm, 2);
            USART_Print("mm D:"); USART_PrintDec(dig);

            if(dig) USART_Print(" HEAVY");

            if(total_rain_mm > 50.0f) {
                USART_Print(" FLOOD RISK!");
            }

            USART_Print("\r\n");
        }
    }
}
🧠 แนวคิด: วัด baseline ADC ตอนแห้ง — เมื่อ ADC เพิ่มขึ้น → มีน้ำฝน — สะสมส่วนต่าง ADC × conversion factor (0.015mm/unit) — ต้อง calibrate จริงด้วยการเทน้ำปริมาตรที่ทราบบนแผ่น — ใช้ prev_adc ติดตามการเปลี่ยนแปลง — threshold +5 ป้องกัน noise — แจ้งเตือนเมื่อฝนสะสม >50mm (flood risk) — ปิด VCC ระหว่างวัดเพื่อลด corrosion
