📚 หลักการทำงาน

OPAMP (Operational Amplifier) — วงจรขยายสัญญาณอนาล็อก — CH32V003 บางรุ่นมี OPAMP ในตัว — ไม่ต้องต่อ IC ภายนอก

Mode การทำงาน

  • Voltage Follower (Buffer): Gain = 1 — อินพุต impedance สูงมาก (~100MΩ) — เอาต์พุต impedance ต่ำ — ใช้กับเซ็นเซอร์ที่มี output impedance สูง (pH probe, piezo)
  • PGA (Programmable Gain Amplifier): Gain = 2, 4, 8, 16 — ขยายสัญญาณเล็กให้ ADC อ่านได้ — เช่น Thermocouple (mV) → ขยาย 16x → ADC อ่านได้แม่น
  • Non-Inverting Amplifier: ใช้ R ภายนอกตั้ง gain — Gain = 1 + Rf/Rg
  • Inverting Amplifier: ใช้ R ภายนอก — Gain = -Rf/Rin — กลับเฟสสัญญาณ

ข้อควรระวัง: OPAMP ภายใน CH32V003 มี bandwidth ~1MHz, input common-mode range 0-VDD, output range ~0.1V ถึง VDD-0.1V — ไม่ใช่ rail-to-rail

Power: OPAMP กินกระแส ~200µA เมื่อเปิด — ถ้าไม่ใช้ควรปิด (OPAMP_Disable())

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: OPAMP Voltage Follower พื้นฐาน

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
    ADC_SimpleInit();

    // Voltage Follower: PA4(input) → PA5(output)
    OPAMP_SimpleInit(OPAMP_CH1, OPAMP_MODE_FOLLOWER);

    // เปิด OPAMP
    OPAMP_Enable(OPAMP_CH1);

🔍 OPAMP_SimpleInit(ch, mode) — ตั้งค่า OPAMP — OPAMP_MODE_FOLLOWER — ทำหน้าที่เป็น buffer — OPAMP_Enable() — จ่ายไฟให้ OPAMP

ขั้นที่ 2: อ่านค่าผ่าน OPAMP Buffer

    while(1) {
        // อ่าน ADC จาก OPAMP output (PA5)
        // PA5 = ADC_CH5 (เช็ค datasheet)
        uint16_t raw = ADC_Read(ADC_CH_PA5);
        float voltage = ADC_ToVoltage(raw, 3.3f);

        USART_Print("OPAMP output: ");
        USART_PrintNum((int32_t)(voltage * 1000));
        USART_Print(" mV\r\n");

        Delay_Ms(500);
    }

🔍 OPAMP buffer ทำให้ ADC อ่านค่าได้แม่นยำขึ้น — โดยเฉพาะเมื่อเซ็นเซอร์มี output impedance สูง — ถ้าไม่มี buffer, ADC sample/hold capacitor จะดึงกระแสจากเซ็นเซอร์ทำให้แรงดันตก

ขั้นที่ 3: PGA — ขยายสัญญาณเล็ก

    // ตัวอย่าง: Thermocouple ให้สัญญาณ ~0-50mV
    // ขยาย 16x → 0-800mV → ADC อ่านได้

    OPAMP_SimpleInit(OPAMP_CH1, OPAMP_MODE_PGA);
    OPAMP_ConfigPGA(OPAMP_CH1, OPAMP_PGA_16);   // Gain = 16x
    OPAMP_Enable(OPAMP_CH1);

    while(1) {
        uint16_t raw = ADC_Read(ADC_CH_PA5);
        // actual voltage = (raw * Vref / 1024) / 16
        float v_input = ADC_ToVoltage(raw, 3.3f) / 16.0f;
        USART_Print("Input: ");
        USART_PrintNum((int32_t)(v_input * 1000000));
        USART_Print(" uV\r\n");
        Delay_Ms(500);
    }

🔍 OPAMP_ConfigPGA(ch, gain) — ตั้ง gain 2/4/8/16 — ข้อควรระวัง: output ต้องไม่เกิน VDD — Input 50mV × 16 = 800mV ยังปลอดภัย

ขั้นที่ 4: เปิด/ปิด OPAMP เพื่อประหยัดไฟ

    // OPAMP กินไฟ ~200µA — ปิดเมื่อไม่ใช้
    OPAMP_Enable(OPAMP_CH1);
    Delay_Ms(10);  // รอ OPAMP settle

    // อ่านค่า...
    uint16_t val = ADC_Read(ADC_CH_PA5);

    // ปิด OPAMP เพื่อประหยัดไฟ
    OPAMP_Disable(OPAMP_CH1);

    // Sleep หรืองานอื่น...
    Delay_Ms(5000);

🔍 OPAMP_Disable(ch) — ปิด OPAMP — ลดกระแส — เหมาะกับระบบ battery-powered ที่อ่านค่าเป็นช่วงๆ

🛠️ สร้าง Product: เครื่องวัด pH

ใช้ OPAMP Buffer + ADC อ่านค่า pH probe — แสดงผลผ่าน USART

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"

// pH Calibration Constants
#define PH_SLOPE     (-59.16f)   // mV/pH @25°C (Nernst)
#define PH7_MV       0.0f        // Offset at pH 7 (calibrated)
#define ADC_VREF     3.3f

int main(void)
{
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
    ADC_SimpleInit();

    // pH probe มี impedance สูงมาก (~100MΩ)
    // ต้องใช้ Voltage Follower เพื่อ impedance matching
    OPAMP_SimpleInit(OPAMP_CH1, OPAMP_MODE_FOLLOWER);
    OPAMP_Enable(OPAMP_CH1);

    USART_Print("pH Meter Ready\r\n");
    USART_Print("Connect pH probe to PA4\r\n");

    while(1)
    {
        // อ่าน ADC จาก OPAMP output
        uint16_t raw = ADC_Read(ADC_CH_PA5);
        float voltage = ADC_ToVoltage(raw, ADC_VREF);

        // pH probe output: ~ -414mV ถึง +414mV (pH 0-14 @ 25°C)
        // ค่าเป็นลบ — ต้อง bias เข้า range 0-3.3V ด้วย R divider

        // คำนวณ pH จากแรงดัน
        // pH = 7.0 + (voltage_mv - PH7_MV) / PH_SLOPE
        float mv = (voltage * 1000.0f) - 1650.0f;  // Remove 1.65V bias
        float ph = 7.0f + (mv - PH7_MV) / PH_SLOPE;

        USART_Print("pH: ");
        USART_PrintNum((int32_t)ph);
        USART_Print(".");
        USART_PrintNum((int32_t)((ph - (int32_t)ph) * 100));
        USART_Print("  ADC: ");
        USART_PrintNum(raw);
        USART_Print("\r\n");

        Delay_Ms(2000);
    }
}

หมายเหตุ: pH probe ให้สัญญาณ bipolar (±414mV) — ต้องใช้ voltage divider bias ให้เป็น 0-3.3V — ตัวอย่างนี้สมมติว่ามีวงจร bias ภายนอก

🛠️ สร้าง Product: ขยายสัญญาณ PIR Motion Sensor (RE200B / D202S)

ใช้ OPAMP ขยายสัญญาณเล็กๆ จากเซนเซอร์ตรวจจับความร้อน RE200B หรือ D202S (~0.5–2mV เวลามีคนเดินผ่าน) ให้ใหญ่ขึ้น ~100 เท่า ก่อนส่งเข้า ADC

รู้จัก RE200B / D202S ก่อน

เซนเซอร์ PIR แบบนี้เป็น element ตรวจจับความร้อนล้วนๆ — ต่างจากโมดูล HC-SR501 ที่มีวงจรเปรียบเทียบในตัว — RE200B/D202S ให้สัญญาณออกมาเป็นแรงดันไฟฟ้าที่ เล็กมาก (~0.5–2mV) ถ้าไม่ขยาย ADC จะอ่านไม่ออก

  • RE200B: มี 2 ขา (Drain=ขา1, Source=ขา2) — ใช้คู่กับ Fresnel lens
  • D202S/D203S (Murata): มี 3 ขา (Drain=ขา1, Source=ขา2, Ground=ขา3) — เหมือน RE200B แต่เพิ่มขา Ground ป้องกันสัญญาณรบกวน

วิธีต่อทีละขั้น

  1. ต่อไฟเลี้ยง PIR: ขา Source (ขา 2) ของ RE200B/D202S ต่อ R 10kΩ → ไปหาไฟกึ่งกลาง 1.65V (VDD/2)
  2. ต่อกราวด์ PIR: ขา Drain (ขา 1) ต่อ GND — ถ้าเป็น D202S ให้ต่อขา 3 (Ground) ไป GND ด้วย เพื่อลดสัญญาณรบกวน
  3. ตัดไฟกระแสตรง (DC block): ต่อ C 1µF (ตัวเก็บประจุ) ระหว่างขา Source กับขา PD5 ของ MCU — ตัวเก็บประจุนี้ทำหน้าที่ กั้นไฟกระแสตรงไม่ให้เข้า OPAMP แต่ยอมให้สัญญาณเคลื่อนไหวผ่านไปได้ เหมือนประตูที่เปิดให้เฉพาะคนขาเข้าเท่านั้น
  4. ตั้งจุดอ้างอิง: ต่อ R 1MΩ จาก PD5 ลง GND — ทำหน้าที่ดึงแรงดันที่ PD5 ให้กลับมาที่ 0V เวลาไม่มีสัญญาณ
  5. ตั้งอัตราขยาย:
    • ต่อ R 1MΩ (Rf) ระหว่าง PA5 (OPAMP output) ↔ PA6 (OPAMP input ลบ)
    • ต่อ R 10kΩ (Rg) จาก PA6 ลง GND
    • Rf+Rg นี้กำหนดว่า OPAMP จะขยายสัญญาณกี่เท่า — สูตร: ขยาย = 1 + Rf/Rg = 1 + 1MΩ/10kΩ = 101 เท่า
    • เช่น สัญญาณเข้า 1mV → ได้ออกมา ~101mV
  6. ต่อไฟกึ่งกลาง 1.65V (VDD/2):
    • ใช้ R 10kΩ ต่อจาก 3.3V ต่อกับ R 10kΩ อีกตัว ต่อลง GND — จุดต่อระหว่าง R สองตัวนี้จะได้ไฟ ~1.65V
    • ไฟ 1.65V นี้ใช้จ่ายให้ R_load ของ PIR (ข้อ 1)
  7. อ่านค่าด้วย ADC: PA5 (OPAMP output) ต่อภายในกับ ADC อยู่แล้ว — ไม่ต้องต่อสาย — เขียนโค้ดอ่าน ADC_Read() ได้เลย

ทำไมต้องมีอุปกรณ์พวกนี้?

อุปกรณ์หน้าที่ (ภาษาชาวบ้าน)ถ้าไม่มีจะเกิดอะไรขึ้น
C 1µFกันไฟกระแสตรง — ให้เฉพาะสัญญาณ "การเคลื่อนไหว" ผ่านOPAMP จะขยายไฟนิ่ง 1.65V ไปด้วย → output ติดเพดาน 3.3V ตลอด
R 1MΩดึงแรงดันที่ขา OPAMP กลับลง GND เวลาไม่มีสัญญาณแรงดันที่ขา OPAMP จะลอย ไม่แน่นอน
Rf=1MΩ, Rg=10kΩกำหนดอัตราขยาย 101 เท่าสัญญาณเล็กเกินไป → ADC อ่านไม่ออก
ไฟ 1.65Vจ่ายไฟเลี้ยงให้ PIR element ทำงานที่จุดกึ่งกลางPIR element ไม่มีไฟเลี้ยง → ไม่ทำงาน

สรุปภาพรวม

สัญญาณจากการเคลื่อนไหว (~1mV) → C 1µF ตัดไฟนิ่งทิ้ง → OPAMP ขยาย 101 เท่า → ได้ ~100mV → ADC อ่านค่า → ถ้าค่าเปลี่ยนไปเรื่อยๆ แปลว่ามีการเคลื่อนไหว

RE200B (2 ขา) และ D202S (3 ขา) ใช้แทนกันได้ — D202S มีขา Ground เพิ่มมาป้องกันสัญญาณรบกวน — วงจรเดียวกัน ต่อขา 3 เพิ่มลง GND

💻 โค้ดอ่านสัญญาณ PIR ผ่าน OPAMP

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"

// ค่า threshold สำหรับตรวจจับ motion
// RE200B + OPAMP gain 101x → output ~50-200mV peak
#define MOTION_THRESHOLD_MV  30   // mV เปลี่ยนแปลง

void setup_bias(void) {
    // สร้าง VDD/2 = 1.65V สำหรับ bias RE200B
    // ใช้ PWM + RC filter หรือ R divider ภายนอก
    pinMode(PC1, PIN_MODE_OUTPUT);
    digitalWrite(PC1, HIGH);  // ใช้ digital output + R divider
}

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
    ADC_SimpleInit();

    // OPAMP Non-Inverting — gain = 1 + Rf/Rg
    OPAMP_SimpleInit(OPAMP1, OPAMP_MODE_NON_INVERTING);
    OPAMP_Enable(OPAMP1);

    USART_Print("PIR RE200B Amplifier Ready\r\n");

    uint16_t last_raw = 0;
    uint16_t baseline = 2048;  // ADC 12-bit, VDD/2 ≈ 2048
    uint8_t motion = 0;

    while(1) {
        // อ่าน ADC จาก OPAMP output
        uint16_t raw = ADC_Read(ADC_CH_PA5);
        float mv = (float)raw * 3300.0f / 4095.0f;

        if(last_raw > 0) {
            int16_t diff = (int16_t)raw - (int16_t)last_raw;
            int16_t diff_mv = (int16_t)(mv - (last_raw * 3300.0f / 4095.0f));

            if(abs(diff) > (MOTION_THRESHOLD_MV * 4095 / 3300)) {
                if(!motion) {
                    motion = 1;
                    USART_Print("🔥 Motion detected! (");
                    USART_PrintNum(diff_mv);
                    USART_Print(" mV change)\r\n");
                }
            } else {
                motion = 0;
            }
        }

        USART_Print("ADC: ");
        USART_PrintNum(raw);
        USART_Print(" (");
        USART_PrintFloat(mv, 1);
        USART_Print(" mV)");
        USART_Print(motion ? " 🔥\r\n" : "\r\n");

        last_raw = raw;
        Delay_Ms(50);  // sample 20Hz
    }
}

หลักการ: ADC อ่านค่า amplified signal ทุก 50ms — เปรียบเทียบกับค่าก่อนหน้า — ถ้าเปลี่ยนแปลงเกิน threshold → มี motion — RE200B ให้สัญญาณ AC เมื่อมีการเคลื่อนไหวเท่านั้น (ถ้านั่งนิ่ง output จะกลับมาที่ baseline)

ปรับปรุงเพิ่มเติม: ใช้ค่าเฉลี่ย running average ลด noise — ใช้ HPF digital cutoff 0.5Hz — ปรับ threshold ตามสภาพแวดล้อมอัตโนมัติ (adaptive threshold)
