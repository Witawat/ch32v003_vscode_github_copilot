🔌 การต่อวงจร

               AT24Cxx EEPROM I2C — External Memory
    ┌──────────────────────────────────────────────┐

    │ PC1(SDA)────────SDA    ┌────────────────────┤
    │ PC2(SCL)────────SCL    │                    │

    │ A0,A1,A2─GND = 0x50   │                    │
    │ WP─GND = write enable  └────────────────────┤
    └──────────────────────────────────────────────┘

AT24Cxx ใช้ I2C address 0x50 (A0=A1=A2=GND) — ต่อ A0-A2 เป็น HIGH/LOW เพื่อเปลี่ยน address ได้สูงสุด 8 ตัวบนบัสเดียว, WP pin ต่อ GND = เขียนได้, ต่อ VCC = write protect

📚 หลักการทำงาน

AT24Cxx เป็น EEPROM แบบ I2C — เก็บข้อมูลได้แม้ไม่มีไฟ, endurance 1,000,000 write cycles ต่อ cell, data retention >100 ปี, ใช้แรงดัน 1.8V-5.5V (CH32V003 3.3V ใช้งานได้)

Page Boundary สำคัญ! การเขียนแบบ sequential (page write) เขียนได้ไม่เกินขอบเขตของ page — AT24C02 page=8 bytes, AT24C256 page=64 bytes — ถ้าเขียนข้าม page boundary จะ wrap กลับไปทับต้น page เดิม!

Write Cycle Time: หลังเขียนแต่ละครั้งต้องรอ ~5ms (self-timed write cycle) — EEPROM จะไม่ตอบ ACK ระหว่างนี้ ต้อง poll หรือ delay

Wear Leveling: กระจายการเขียนไปยัง address ต่างๆ แทนที่จะเขียนที่เดิมซ้ำๆ — ยืดอายุ EEPROM จากเขียนทุกวินาทีที่เดียว → พังใน 11 วัน → กระจาย 1000 address → อยู่ได้ 30 ปี

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Init + Setup

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "AT24Cxx.h"

int main(void) {
    SystemCoreClockUpdate();
    I2C_Init(I2C1, I2C_SPEED_STANDARD);
    AT24C_Init(I2C1, AT24C02, 0x50);  // 256 bytes, addr 0x50
    USART_Init(USART1, BAUD_115200);
    USART_Println("AT24C02 Ready (256 bytes)");

🔍 AT24C_Init(i2c, type, addr) — เลือกเบอร์ชิป (AT24C01/02/04/08/16/32/64/128/256/512) และ I2C address, ไม่ต้องต่อ pull-up เพิ่มถ้าใช้ module สำเร็จรูป

ขั้นที่ 2: เขียน/อ่าน ไบต์ และจัดการ Page

    // === Byte Write/Read ===
    AT24C_WriteByte(0x00, 0xAB);  // write 0xAB to addr 0
    Delay_Ms(5);                  // wait write cycle
    uint8_t val = AT24C_ReadByte(0x00);  // read back → 0xAB

    // === Page Write === (AT24C02 page=8 bytes)
    uint8_t buf[8] = {0,1,2,3,4,5,6,7};
    AT24C_WritePage(0x00, buf, 8);  // write 8 bytes at page0
    Delay_Ms(10);

    // ⚠️ Write ข้าม page boundary — WRONG!
    // AT24C_WritePage(0x06, buf, 8);
    // addr 6+7=page0, addr 8+9=WRAP to addr0,1!

🔍 AT24C_WriteByte(addr, val) — เขียน 1 ไบต์ + auto delay 5ms, AT24C_WritePage(addr, buf, len) — เขียนหลายไบต์ใน page เดียวกัน, ⚠️ ห้ามเขียนข้าม page boundary — ต้องคำนวณก่อน

ขั้นที่ 3: Sequential Read + Wear Leveling

    // === Sequential Read ===
    uint8_t rbuf[32];
    AT24C_ReadSeq(0x00, rbuf, 32);  // read 32 bytes from addr 0

    // === Wear Leveling — เขียนไปเรื่อยๆ วนลูป ===
    uint16_t write_addr = 0;
    uint16_t eeprom_size = 256;  // AT24C02
    while(1) {
        uint8_t data = (uint8_t)(write_addr & 0xFF);
        AT24C_WriteByte(write_addr, data);
        Delay_Ms(5);

        write_addr++;
        if(write_addr >= eeprom_size) write_addr = 0;  // wrap
        Delay_Ms(1000);  // write 1 byte/sec
    }
}

🔍 AT24C_ReadSeq(addr, buf, len) — อ่านหลายไบต์ต่อเนื่อง (ไม่มี page boundary ในการอ่าน), wear leveling: เขียนวนไปเรื่อยๆ ทั่วพื้นที่ แทนที่จะเขียนที่เดิมซ้ำ

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
อ่านค่าผิด/0xFF ตลอดลืม delay หลัง write (write cycle)ต้องDelay_Ms(5) หรือ poll AT24C_IsReady() หลังเขียนทุกครั้ง
เขียนแล้วค่าที่ต้น page หายเขียนข้าม page boundary — wrap ทับเช็ค page size ก่อนเขียน — AT24C02=8B/page, AT24C256=64B/page
I2C ไม่เจอ deviceAddress A0,A1,A2 ไม่ตรง หรือลืม VCCเช็ค A0-A2 ว่าต่อ GND หรือ VCC ตรงกับ address ที่ตั้ง
EEPROM พังเร็วเขียนที่ address เดิมซ้ำๆ เกิน 1M cyclesใช้ wear leveling กระจายการเขียน, หรือใช้ ring buffer
ค่าที่อ่านเก่าไปไม่ได้เขียนทับก่อน — EEPROM ไม่ล้างเองค่าที่อ่านได้คือค่าที่เขียนล่าสุด — ถ้าต้องการให้ว่างเขียน 0xFF

🛠️ สร้าง Product: Config Store + Wear-Leveled Logger

เก็บ configuration ที่ address 0-63 + log ข้อมูลหมุนเวียนที่ address 64-255

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "AT24Cxx.h"

#define CONFIG_ADDR 0
#define LOG_START   64
#define LOG_SIZE    192
uint8_t log_index = 0;

int main(void)
{
    SystemCoreClockUpdate();
    I2C_Init(I2C1, I2C_SPEED_STANDARD);
    AT24C_Init(I2C1, AT24C02, 0x50);
    USART_Init(USART1, BAUD_115200);

    // Load config
    uint8_t brightness = AT24C_ReadByte(CONFIG_ADDR);
    if(brightness == 0xFF) brightness = 128;  // default
    USART_Print("Brightness: "); USART_PrintInt(brightness);

    while(1)
    {
        // Simulate sensor reading + log
        uint16_t adc = ADC_Read(ADC1, ADC_CH0);
        uint8_t log_data[2] = {adc & 0xFF, (adc >> 8) & 0xFF};

        uint16_t log_addr = LOG_START + log_index;
        AT24C_WritePage(log_addr & 0xF8, log_data, 2);  // align to page
        Delay_Ms(5);

        log_index = (log_index + 2) % LOG_SIZE;  // ring buffer
        Delay_Ms(5000);
    }
}

✏️ ลองทำดู

  1. ทำ settings manager — เก็บ WiFi SSID, password, device name ใน EEPROM → อ่านตอน boot
  2. ทำ event counter — นับจำนวนครั้งที่กดปุ่ม → เก็บใน EEPROM (wear leveled) → ไม่หายแม้ไฟดับ
  3. ต่อ AT24C02 2 ตัวบนบัส I2C เดียวกัน — A0 ต่างกัน → address 0x50 และ 0x51 → อ่าน/เขียนสลับ
