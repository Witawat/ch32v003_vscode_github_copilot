📝 เฉลย — บทที่ 4: USART

ข้อ 1: รับคำสั่ง "LED1=ON", "LED1=OFF" ทาง Serial

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include <string.h>

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
    pinMode(PC0, PIN_MODE_OUTPUT);  // LED1

    USART_Print("Ready. Commands: LED1=ON, LED1=OFF\r\n");

    char buf[32];
    uint8_t idx = 0;

    while (1) {
        if (USART_Available()) {
            char ch = USART_Read();
            if (ch == '\r' || ch == '\n') {
                if (idx > 0) {
                    buf[idx] = '\0';
                    idx = 0;

                    if (strcmp(buf, "LED1=ON") == 0) {
                        digitalWrite(PC0, HIGH);
                        USART_Print("LED1: ON\r\n");
                    } else if (strcmp(buf, "LED1=OFF") == 0) {
                        digitalWrite(PC0, LOW);
                        USART_Print("LED1: OFF\r\n");
                    } else {
                        USART_Print("Unknown: ");
                        USART_Print(buf);
                        USART_Print("\r\n");
                    }
                }
            } else if (idx < sizeof(buf) - 1) {
                buf[idx++] = ch;
            }
        }
    }
}
🧠 แนวคิด: รับอักขระทีละตัว สะสมใน buffer buf[] — เมื่อเจอ \r หรือ \n แสดงว่าจบคำสั่ง — ใช้ strcmp() เปรียบเทียบ string แบบ case-sensitive — LED1=ON → HIGH, LED1=OFF → LOW — รวม string.h เพื่อใช้ strcmp()

ข้อ 2: ส่งค่า ADC แบบ real-time ทุก 100ms

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include "SimpleHAL.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
    ADC_SimpleInit();

    Timer_t tmr_adc;
    Start_Timer(&tmr_adc, 100, 1);  // ทุก 100ms

    while (1) {
        if (Is_Timer_Expired(&tmr_adc)) {
            uint16_t adc = ADC_Read(ADC_CH_PD2);
            uint16_t mv = (uint32_t)adc * 3300 / 1023;

            USART_Print("ADC: ");
            USART_PrintNum(adc);
            USART_Print(" (");
            USART_PrintNum(mv);
            USART_Print("mV)\r\n");
        }

        // ทำงานอื่นใน main loop ได้ — ไม่หยุดรอ
    }
}
🧠 แนวคิด: ใช้ Timer_t แบบ non-blocking หมดทุก 100ms — ไม่ใช้ Delay_Ms() เพราะจะหยุดโปรแกรม — แปลง ADC (0-1023) เป็น mV ด้วย adc*3300/1023 (integer math เร็วกว่า float) — main loop ยังว่างทำอย่างอื่นได้

ข้อ 3: เปลี่ยนไปใช้ USART_PINS_REMAP2 แล้วส่งข้อมูล

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include "SimpleHAL.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();

    // REMAP2: TX=PD6, RX=PD5 (ตรงข้ามกับ DEFAULT)
    USART_SimpleInit(BAUD_115200, USART_PINS_REMAP2);

    USART_Print("USART REMAP2 Active!\r\n");
    USART_Print("TX=PD6, RX=PD5\r\n");

    uint32_t counter = 0;

    while (1) {
        USART_Print("Count: ");
        USART_PrintNum(counter++);
        USART_Print("\r\n");

        Delay_Ms(1000);
    }
}
🧠 แนวคิด: แค่เปลี่ยน USART_PINS_DEFAULT → USART_PINS_REMAP2 — TX/RX สลับกัน (TX=PD6, RX=PD5 แทน TX=PD5, RX=PD6) — มีประโยชน์เมื่อต้องการ cross-connect โดยไม่ต้องสลับสาย hardware — REMAP1 (TX=PD0, RX=PD1) มีเฉพาะ TSSOP-20/QFN-20 — REMAP2 ใช้ได้ทุกแพ็กเกจรวมถึง SOP-8
