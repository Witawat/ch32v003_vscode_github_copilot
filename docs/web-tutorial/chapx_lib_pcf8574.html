PCF8574 เพิ่ม GPIO 8 ขาผ่าน I2C — quasi-bidirectional (ไม่ต้อง config input/output), เขียน = 1 (HIGH) เพื่อตั้งเป็น input (ลอย), เขียน = 0 (LOW) เพื่อขับ output เป็น LOW, I2C address 0x20-0x27 เลือกด้วย A0-A2

📚 หลักการทำงาน

PCF8574 เป็น I2C 8-bit IO expander — quasi-bidirectional หมายความว่าแต่ละ pin เป็นทั้ง input และ output โดยไม่ต้องตั้งค่า register — ทำให้ใช้ง่ายมาก: เขียน byte = output, อ่าน byte = input

Quasi-Bidirectional ทำงานอย่างไร: Output HIGH จริงๆ คือ weak pull-up (~100µA) — ถ้า pin ถูกดึงลง LOW จากภายนอก (เช่นกดปุ่ม) ก็จะอ่านได้เป็น LOW — ทำให้ pin เดียวกันใช้เป็น input และ output ได้สลับกัน

วิธีใช้เป็น Input: เขียน 1 ไปที่ pin นั้นก่อน (ตั้งเป็น HIGH/output off) → pin จะลอย → อ่านค่าได้

วิธีใช้เป็น Output: เขียน 0 (LOW) หรือ 1 (HIGH) → pin จะขับ output ตามนั้น (แต่อย่าลืมว่า HIGH ไม่แรง — ~100µA เท่านั้น ถ้าต้องการกระแสมากใช้ external pull-up)

Interrupt (INT): เมื่อ input เปลี่ยนสถานะ PCF8574 จะดึง INT pin LOW — ใช้แจ้ง MCU ว่ามีการเปลี่ยนแปลงโดยไม่ต้อง poll I2C ตลอด

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Init + Setup

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "PCF8574.h"

int main(void) {
    SystemCoreClockUpdate();
    I2C_Init(I2C1, I2C_SPEED_STANDARD);
    PCF8574_Init(I2C1, 0x20);  // A0=A1=A2=GND → 0x20
    USART_Init(USART1, BAUD_115200);

    // ตั้งค่าเริ่มต้น: All HIGH (input mode)
    PCF8574_Write(0xFF);
    USART_Println("PCF8574 Ready! 8 extra GPIO pins");

🔍 PCF8574_Init(i2c, addr) — address 0x20-0x27, default = 0x20, เขียน 0xFF = ตั้งทุก pin เป็น input (HIGH = weak pull-up)

ขั้นที่ 2: Output — ควบคุม LED/Relay

    // LED on P0, P1, P7 — active LOW (LOW = LED ON)
    // Relay on P4, P5 — active HIGH (HIGH = ON)

    PCF8574_Write(0x30);  // P4,P5=HIGH(relay on), others=LOW
    Delay_Ms(1000);

    // Toggle P0 (LED1)
    uint8_t state = PCF8574_Read();  // read current state
    state ^= 0x01;                    // toggle bit 0
    PCF8574_Write(state);

    // ใช้ฟังก์ชันตั้งทีละบิต
    PCF8574_WritePin(0, 0);  // P0=LOW (LED1 ON)
    PCF8574_WritePin(1, 1);  // P1=HIGH (LED2 OFF)
    PCF8574_WritePin(7, 0);  // P7=LOW (LED3 ON)
    Delay_Ms(1000);
    PCF8574_WritePin(0, 1);  // P0=HIGH (LED1 OFF)

🔍 PCF8574_Write(value) — เขียน 8-bit พร้อมกัน (1 byte = 8 pins), PCF8574_WritePin(pin, val) — ตั้งทีละ pin (0-7), ⚠️ quasi-bidirectional: HIGH = ~100µA pull-up เท่านั้น

ขั้นที่ 3: Input — อ่านปุ่ม + Interrupt

    // P2, P3 เป็น input (ปุ่มกด) — ต้องเขียน 1 ก่อน
    // P2=button1 (active LOW, external pull-up)
    // P3=button2 (active LOW)

    // ตั้ง P2,P3 เป็น input (HIGH) ที่เหลือ output
    PCF8574_Write(0x0C);  // 0b00001100 = P2,P3=HIGH

    // ต่อ INT pin → PC3 (falling edge interrupt)
    pinMode(PC3, PIN_MODE_INPUT_PULLUP);

    while(1) {
        if(!digitalRead(PC3)) {  // INT pin LOW → มีการเปลี่ยนแปลง
            uint8_t pins = PCF8574_Read();

            if(PCF8574_ReadPin(2) == 0) {  // button1 pressed
                USART_Println("Button 1 pressed!");
                PCF8574_WritePin(0, 0);  // LED1 ON
            }
            if(PCF8574_ReadPin(3) == 0) {  // button2 pressed
                USART_Println("Button 2 pressed!");
                PCF8574_WritePin(0, 1);  // LED1 OFF
            }

            // INT จะ clear อัตโนมัติเมื่ออ่าน
        }
        Delay_Ms(10);
    }
}

🔍 PCF8574_Read() — อ่านสถานะ 8-bit, PCF8574_ReadPin(n) — อ่านทีละ pin, INT pin จะ LOW เมื่อ input เปลี่ยน → อ่าน I2C เพื่อ clear, ข้อดี: ไม่ต้อง poll I2C ตลอด ใช้ INT interrupt แทน

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
อ่าน input ได้ 0 ตลอดลืมเขียน 1 ก่อนอ่าน (ตั้งเป็น input)ต้องเขียน 1 ที่ pin ที่ต้องการอ่านก่อน → PCF8574_WritePin(2, 1)
LED ติดแต่หรี่มากQuasi-bidir HIGH = ~100µA เท่านั้นต่อ LED แบบ active-LOW (MCU→LOW=ON), หรือเพิ่ม external pull-up
INT ไม่เคย activeInput pin ไม่ได้ถูกดึงเป็น LOW/HIGH ชัดเจนต้องมี external pull-up 10kΩ, หรือใช้ PCF8574_Init() ตั้ง pull-up ภายใน
PCF8574 vs PCF8574A สับสนAddress ต่างกัน: PCF8574=0x20-0x27, PCF8574A=0x38-0x3Fเช็คเบอร์ชิป — ใช้ I2C scanner ถ้าไม่แน่ใจ
I2C address ไม่เจอA0-A2 ต่อผิด หรือใช้ address ผิดA0-A2 ต่อ GND = 0x20, ต่อ VCC = 0x27 — เช็คการต่อ

🛠️ สร้าง Product: 8-Channel Relay Board + 4 Button Input

PCF8574 ขยาย GPIO — P0-P3=output (relay), P4-P7=input (buttons) — interrupt-driven

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "PCF8574.h"

int main(void)
{
    SystemCoreClockUpdate();
    I2C_Init(I2C1, I2C_SPEED_STANDARD);
    PCF8574_Init(I2C1, 0x20);
    USART_Init(USART1, BAUD_115200);

    // P0-P3: output (relay), P4-P7: input (buttons)
    PCF8574_Write(0xF0);  // 0b11110000

    pinMode(PC3, PIN_MODE_INPUT_PULLUP);  // INT pin

    uint8_t relay_state = 0x00;  // all relays OFF
    PCF8574_Write(0xF0 | relay_state);

    while(1)
    {
        if(!digitalRead(PC3)) {  // INT active
            uint8_t pins = PCF8574_Read();

            // Button P4 → toggle relay P0
            if(!(pins & 0x10)) {  // P4=LOW (button pressed)
                relay_state ^= 0x01;
                USART_Println("Toggle Relay 0");
            }
            if(!(pins & 0x20)) {  // P5
                relay_state ^= 0x02;
                USART_Println("Toggle Relay 1");
            }
            if(!(pins & 0x40)) {  // P6
                relay_state ^= 0x04;
                USART_Println("Toggle Relay 2");
            }
            if(!(pins & 0x80)) {  // P7
                relay_state = 0x00;  // ALL OFF
                USART_Println("All Relays OFF");
            }

            // Update outputs
            PCF8574_Write(0xF0 | relay_state);
        }
        Delay_Ms(10);
    }
}

✏️ ลองทำดู

  1. ต่อ PCF8574 2 ตัวบนบัส I2C เดียวกัน — address 0x20 และ 0x21 — รวม 16 GPIO → ควบคุม LED matrix 4x4
  2. ทำ keypad 4x4 — ใช้ PCF8574 1 ตัว (P0-P3=row outputs, P4-P7=col inputs) — สแกน matrix
  3. ทำ interrupt-driven encoder — ต่อ rotary encoder กับ PCF8574 P0,P1 → INT แจ้งเมื่อหมุน
