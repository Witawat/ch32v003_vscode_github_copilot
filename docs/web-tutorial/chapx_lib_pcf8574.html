📚 หลักการทำงาน

PCF8574 เป็น I2C 8-bit IO expander — quasi-bidirectional (ไม่ต้อง config input/output) — เขียน byte = output, อ่าน byte = input

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Init + Setup

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "PCF8574.h"

PCF8574_Instance pcf;

int main(void) {
    SystemCoreClockUpdate();
    I2C_Init(I2C1, I2C_SPEED_STANDARD);
    PCF8574_Init(&pcf, 0x20);  // A0=A1=A2=GND → 0x20
    USART_Init(USART1, BAUD_115200);

    // ตั้งค่าเริ่มต้น: All HIGH (input mode)
    PCF8574_WritePort(&pcf, 0xFF);
    USART_Println("PCF8574 Ready!");

🔍 PCF8574_Init(&pcf, addr) — instance-based, address 0x20-0x27

ขั้นที่ 2: Output — ควบคุม LED/Relay

    PCF8574_WritePort(&pcf, 0x30);  // P4,P5=HIGH, others=LOW
    Delay_Ms(1000);

    // Toggle P0
    uint8_t state = PCF8574_ReadPort(&pcf);
    state ^= 0x01;
    PCF8574_WritePort(&pcf, state);

    // ตั้งทีละบิต
    PCF8574_WritePin(&pcf, 0, 0);  // P0=LOW (LED1 ON)
    PCF8574_WritePin(&pcf, 1, 1);  // P1=HIGH (LED2 OFF)

🔍 PCF8574_WritePort(&pcf, value) — เขียน 8-bit พร้อมกัน (1 byte = 8 pins)

ขั้นที่ 3: Input — อ่านปุ่ม + Interrupt

    // ตั้ง P2,P3 เป็น input (HIGH) ที่เหลือ output
    PCF8574_WritePort(&pcf, 0x0C);  // 0b00001100 = P2,P3=HIGH

    pinMode(PC3, PIN_MODE_INPUT_PULLUP);  // INT pin

    while(1) {
        if(!digitalRead(PC3)) {
            uint8_t pins = PCF8574_ReadPort(&pcf);

            if(PCF8574_ReadPin(&pcf, 2) == 0) {
                USART_Println("Button 1 pressed!");
                PCF8574_WritePin(&pcf, 0, 0);
            }
            if(PCF8574_ReadPin(&pcf, 3) == 0) {
                USART_Println("Button 2 pressed!");
                PCF8574_WritePin(&pcf, 0, 1);
            }
        }
        Delay_Ms(10);
    }
}

🔍 PCF8574_ReadPort(&pcf) — อ่านสถานะ 8-bit, PCF8574_ReadPin(&pcf, n) — อ่านทีละ pin

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
อ่าน input ได้ 0 ตลอดลืมเขียน 1 ก่อนอ่านต้องเขียน 1 ที่ pin ที่ต้องการอ่านก่อน
LED ติดแต่หรี่มากQuasi-bidir HIGH = ~100µA เท่านั้นต่อ LED แบบ active-LOW

🛠️ สร้าง Product: 8-Channel Relay Board + 4 Button Input

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "PCF8574.h"

int main(void)
{
    SystemCoreClockUpdate();
    I2C_Init(I2C1, I2C_SPEED_STANDARD);

    PCF8574_Instance pcf;
    PCF8574_Init(&pcf, 0x20);
    USART_Init(USART1, BAUD_115200);

    PCF8574_WritePort(&pcf, 0xF0);  // P0-P3: output, P4-P7: input
    pinMode(PC3, PIN_MODE_INPUT_PULLUP);

    uint8_t relay_state = 0x00;

    while(1)
    {
        if(!digitalRead(PC3)) {
            uint8_t pins = PCF8574_ReadPort(&pcf);

            if(!(pins & 0x10)) { relay_state ^= 0x01; }  // P4 → toggle P0
            if(!(pins & 0x20)) { relay_state ^= 0x02; }  // P5 → toggle P1
            if(!(pins & 0x80)) { relay_state = 0x00; }   // P7 → all OFF

            PCF8574_WritePort(&pcf, 0xF0 | relay_state);
        }
        Delay_Ms(10);
    }
}

✏️ ลองทำดู

  1. ต่อ PCF8574 2 ตัวบนบัส I2C เดียวกัน — address 0x20 และ 0x21 — รวม 16 GPIO
  2. ทำ keypad 4x4 — ใช้ PCF8574 1 ตัว (P0-P3=row, P4-P7=col)
