📚 หลักการทำงาน

DRV8825 เป็น stepper motor driver ที่ใช้ STEP/DIR interface — ต่างจาก ULN2003 ตรงที่ไม่ต้องเขียน step sequence เอง — ส่ง pulse ที่ STEP pin 1 ครั้ง = 1 microstep — ความถี่ของ pulse กำหนดความเร็ว

Microstepping 1/1 ถึง 1/32 — ตั้งค่าด้วย M0,M1,M2 3 pins — NEMA17 ทั่วไป 1.8°/step (200 steps/rev ใน full-step) — 1/32 microstepping = 6400 pulses/rev — การเคลื่อนที่ละเอียดและเงียบขึ้นมาก

Current Limit: ตั้งกระแสสูงสุดด้วยตัวต้านทานปรับค่า (pot) บนบอร์ด — สูตร: I_max = Vref × 2 — วัด Vref ที่ test point — NEMA17 ทั่วไป 1-2A/phase ตั้ง Vref = 0.5-1.0V

Enable (EN): LOW = เปิดใช้งาน, HIGH = ปิด (ปล่อย motor) — ใช้ LOW ตลอดเวลาเพื่อให้มอเตอร์ hold ตำแหน่ง

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Init GPIO + Microstepping Pins

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"

#define STEP PC0
#define DIR  PC1
#define EN   PC2
#define M0   PC3
#define M1   PC4
#define M2   PC5

typedef enum {FULL=0, HALF=1, QUARTER=2, EIGHTH=3, SIXTEENTH=4, THIRTYSEC=5} Microstep_t;

void drv8825_set_microstep(Microstep_t ms) {
    digitalWrite(M0, ms & 0x01);
    digitalWrite(M1, (ms >> 1) & 0x01);
    digitalWrite(M2, (ms >> 2) & 0x01);
}

🔍 M0-M2 ใช้ 3-bit binary เลือก microstep mode — เขียน GPIO ตามตาราง datasheet — LOW=LOW=LOW → Full-Step, HIGH=HIGH=HIGH → 1/32

ขั้นที่ 2: Pulse Generation + Speed

void drv8825_step(uint32_t pulses, uint16_t delay_us, uint8_t dir) {
    digitalWrite(DIR, dir);  // 1=CW, 0=CCW
    for(uint32_t i = 0; i < pulses; i++) {
        digitalWrite(STEP, HIGH);
        Delay_Us(2);       // DRV8825 ต้องการ pulse > 1.9μs
        digitalWrite(STEP, LOW);
        Delay_Us(delay_us);  // ความเร็ว = delay ระหว่าง pulses
    }
}

int main(void) {
    SystemCoreClockUpdate();
    pinMode(STEP, PIN_MODE_OUTPUT);
    pinMode(DIR, PIN_MODE_OUTPUT);
    pinMode(EN, PIN_MODE_OUTPUT);
    pinMode(M0, PIN_MODE_OUTPUT);
    pinMode(M1, PIN_MODE_OUTPUT);
    pinMode(M2, PIN_MODE_OUTPUT);

    digitalWrite(EN, LOW);    // Enable driver
    drv8825_set_microstep(SIXTEENTH);  // 1/16 = 3200 pulses/rev

🔍 Pulse ต้องกว้าง ≥1.9μs — EN=LOW เพื่อเปิด — 1/16 microstepping ให้ความละเอียด 3200 pulses/รอบ — delay_us ยิ่งน้อยยิ่งเร็ว (100μs = ~187 RPM)

ขั้นที่ 3: Acceleration Profile + Loop

    while(1) {
        // หมุน 10 รอบที่ 1/16 = 32000 pulses
        // Ramp up from 500μs to 100μs
        for(uint32_t i = 0; i < 32000; i++) {
            uint16_t spd;
            if(i < 5000) spd = 500 - (i * 400 / 5000);       // ramp up
            else if(i > 27000) spd = 100 + ((i-27000) * 400 / 5000); // ramp down
            else spd = 100;  // constant speed

            digitalWrite(STEP, HIGH); Delay_Us(2);
            digitalWrite(STEP, LOW);  Delay_Us(spd);
        }
        Delay_Ms(1000);

        // เปลี่ยนทิศทาง
        digitalWrite(DIR, !digitalRead(DIR)); // toggle
    }
}

🔍 Acceleration แบบ linear ramp — 3 ช่วง: เร่ง 5k pulses, คงที่ 22k pulses, ผ่อน 5k pulses — ป้องกัน motor stall ที่รอบสูง — 1/16 microstepping = 3200 pulses/rev → 32000 = 10 รอบ

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
Motor ไม่หมุน — DRV8825 ร้อนจัดต่อ VMOT กลับขั้ว หรือ Current Limit สูงเกินตรวจสอบขั้ว VMOT — ปรับ pot ทวนเข็มเพื่อลด Vref — วัด Vref ≤ 1V
Motor สั่นแต่ไม่หมุนpulse frequency สูงเกิน หรือ microstep mode ไม่ตรงเริ่มจากความเร็วต่ำ (delay 1000μs) — ตรวจสอบ M0-M2 logic กับตาราง datasheet
Motor skip steps ที่ความเร็วสูงไม่มี acceleration — หรือ current limit ต่ำไปใช้ ramp-up profile — เพิ่ม Vref (ระวังความร้อน) — ลด microstep level
STEP pin ไม่สร้าง pulseลืมตั้ง pinMode เป็น OUTPUTตรวจสอบ pinMode(STEP, PIN_MODE_OUTPUT) — ใช้ oscilloscope วัด
นับรอบผิด — จำนวน pulses ไม่ตรงลืมเปลี่ยน steps/rev ตาม microstep modeFull=200, 1/16=3200, 1/32=6400 — คำนวณสูตร: 200 × microstep_level

🛠️ สร้าง Product: Precision Linear Slide

NEMA17 + DRV8825 + Lead Screw GT2 — เคลื่อนที่ตำแหน่ง mm ที่แม่นยำ — ใช้ Endstop + Home sequence

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"

#define STEP PC0
#define DIR  PC1
#define EN   PC2
#define M0   PC3
#define M1   PC4
#define M2   PC5
#define LIMIT PC6  // Endstop (NC, LOW=triggered)

#define STEPS_PER_MM 25600  // 1/32 * 200 steps * (1rev/2.5mm) * 32
int32_t position_mm = 0;

void move_to_mm(int32_t target_mm) {
    int32_t delta = target_mm - position_mm;
    uint8_t dir = (delta > 0) ? 1 : 0;
    uint32_t pulses = (delta > 0) ? delta * STEPS_PER_MM : -delta * STEPS_PER_MM;
    digitalWrite(DIR, dir);
    for(uint32_t i = 0; i < pulses; i++) {
        if(!digitalRead(LIMIT) && dir == 0) break; // Limit switch stop
        digitalWrite(STEP, HIGH); Delay_Us(2);
        digitalWrite(STEP, LOW);  Delay_Us(300);
    }
    position_mm = target_mm;
}

void home_sequence() {
    digitalWrite(DIR, 0);  // Move towards limit
    while(digitalRead(LIMIT)) {  // Until triggered
        digitalWrite(STEP, HIGH); Delay_Us(2);
        digitalWrite(STEP, LOW);  Delay_Us(500);
    }
    position_mm = 0;  // Home = 0mm
}

int main(void) {
    SystemCoreClockUpdate();
    pinMode(STEP, PIN_MODE_OUTPUT); pinMode(DIR, PIN_MODE_OUTPUT);
    pinMode(EN, PIN_MODE_OUTPUT);
    pinMode(M0,PIN_MODE_OUTPUT); pinMode(M1,PIN_MODE_OUTPUT); pinMode(M2,PIN_MODE_OUTPUT);
    pinMode(LIMIT, PIN_MODE_INPUT_PULLUP);
    digitalWrite(EN, LOW);
    digitalWrite(M0,HIGH); digitalWrite(M1,HIGH); digitalWrite(M2,HIGH); // 1/32

    home_sequence();
    while(1) {
        move_to_mm(100); Delay_Ms(1000);
        move_to_mm(0);   Delay_Ms(1000);
    }
}
