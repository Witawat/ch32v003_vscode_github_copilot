⚠️ สำคัญ: Servo กินกระแสสูงสุด 500mA+ — ต้องใช้ไฟ 5V แยก ไม่ใช้ไฟ 3.3V จาก MCU ตรง ๆ! อย่าลืมต่อ GND ร่วมกันระหว่าง MCU กับ External Supply

📚 หลักการทำงาน

PWM 50Hz คือความถี่มาตรฐานของ RC Servo — 20ms ต่อ 1 คาบ (1/50 = 0.02s) ภายใน 1 คาบ จะมี pulse ความกว้าง 0.5ms–2.5ms คุมมุม 0°–180° โดย center = 1.5ms = 90°

SG90 Specifications: แรงดัน 4.8V, แรงบิด 1.2 kg·cm, ความเร็ว 0.12s/60°, มุมทำงาน 0–180°, น้ำหนัก 9g, ขนาด 23×12×29mm

Servo_Init + Attach/Detach: Servo_Init() กำหนด PWM channel — ต้องเรียก Servo_Attach() ก่อนใช้งานจึงจะมีสัญญาณ — Servo_Detach() หยุดสร้างสัญญาณ (ประหยัดไฟ, กัน servo ร้อน)

Servo_Write / Servo_WriteMicroseconds: Servo_Write() รับมุม 0-180° — Servo_WriteMicroseconds() รับค่า pulse โดยตรงหน่วย μs (500-2500) — Servo_SetPulseRange() ปรับ range ได้ ใช้กับ servo ที่ pulse ต่างจากมาตรฐาน

ข้อควรระวัง: Servo ต้อง Attach ก่อน — ถ้าไม่ได้ Attach Servo_Write จะไม่ทำงาน — หลังจบการใช้งานควร Detach เพื่อไม่ให้ servo สั่นค้างตำแหน่ง

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Init + Attach

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "Servo.h"

Servo_Instance servo;

int main(void) {
    SystemCoreClockUpdate();

    Servo_Init(&servo, PWM1_CH4);
    Servo_Attach(&servo);

🔍 Servo_Init(&servo, channel) กำหนด PWM channel (PWM1_CH4 ตรงกับขา PD2) — Servo_Attach() เริ่มสร้างสัญญาณ — ต้องเรียก Attach ก่อนจึงจะควบคุม servo ได้

ขั้นที่ 2: ควบคุมตำแหน่ง Servo

    // หมุนไป 0°
    Servo_Write(&servo, 0);
    Delay_Ms(1000);
    // หมุนไป 90° (กึ่งกลาง)
    Servo_Write(&servo, 90);
    Delay_Ms(1000);
    // หมุนไป 180°
    Servo_Write(&servo, 180);
    Delay_Ms(1000);
    // ใช้ pulse โดยตรง (1500μs = 90°)
    Servo_WriteMicroseconds(&servo, 1500);
    Delay_Ms(1000);

🔍 Servo_Write(&servo, angle) — รับมุม 0-180° — Servo_WriteMicroseconds() รับค่า μs โดยตรง — หน่วง 1 วิให้ servo มีเวลาหมุนถึงตำแหน่ง

ขั้นที่ 3: Sweep Loop

    while(1) {
        // Sweep 0° → 180°
        for(uint8_t a = 0; a <= 180; a++) {
            Servo_Write(&servo, a);
            Delay_Ms(15);
        }
        // Sweep 180° → 0°
        for(uint8_t a = 180; a > 0; a--) {
            Servo_Write(&servo, a);
            Delay_Ms(15);
        }
    }

    // เมื่อเลิกใช้:
    Servo_Detach(&servo);
}

🔍 Sweep ไป-กลับ 0–180° — Delay_Ms(15) ให้ servo มีเวลาขยับทีละองศา — หลังใช้งานควรเรียก Servo_Detach() หยุดสัญญาณ

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
Servo กระตุก/ไม่หมุนไฟ 5V จ่ายกระแสไม่พอ (servo กิน 500mA+)ใช้ External 5V supply แยก — ห้ามใช้ 3.3V จาก MCU
Servo ไม่ตอบสนองลืมเรียก Servo_Attach() หลัง Initต้องเรียก Attach() ก่อนจึงจะมีสัญญาณ PWM ออก
Servo สั่นค้างตำแหน่งไม่มี load หรือ PWM timing ผิดเรียก Servo_Detach() เมื่อไม่ใช้งาน — หรือเช็ค channel init
Servo หมุนเกิน 180°pulse width เกิน 0.5–2.5msใช้ Servo_SetPulseRange() จำกัดช่วง pulse ให้เหมาะสม
GND ไม่ร่วมกันลืมต่อ GND MCU กับ External 5Vต่อ GND ของ MCU, External 5V, และ servo เข้าด้วยกัน
หลายตัวควบคุมไม่แม่นCH32V003 มี PWM channels จำกัดใช้ PCA9685 I²C Servo Driver สำหรับ ≥4 ตัว

🛠️ สร้าง Product: Pan-Tilt 2-Axis Camera Mount

ใช้ Servo 2 ตัว + Joystick (2 Potentiometer ADC) ควบคุมกล้อง — แกน X (Pan) PWM1_CH4, แกน Y (Tilt) PWM1_CH3

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "Servo.h"

Servo_Instance pan_servo, tilt_servo;

int main(void) {
    SystemCoreClockUpdate();
    ADC_Init_Single(PA1); ADC_Init_Single(PA2);

    Servo_Init(&pan_servo, PWM1_CH4);
    Servo_Attach(&pan_servo);
    Servo_Init(&tilt_servo, PWM1_CH3);
    Servo_Attach(&tilt_servo);

    uint8_t pan = 90, tilt = 90;
    while(1) {
        uint16_t adc_x = ADC_Read(PA1);
        uint16_t adc_y = ADC_Read(PA2);
        pan = (uint8_t)(adc_x * 180 / 4095);
        tilt = (uint8_t)(adc_y * 180 / 4095);
        Servo_Write(&pan_servo, pan);
        Servo_Write(&tilt_servo, tilt);
        Delay_Ms(20);
    }
}

✏️ ลองทำดู

  1. ทำ Pan-Tilt tracker ที่ขยับตามแสง — ใช้ LDR 2 ตัวอ่านค่า ADC แล้วปรับ servo ให้หันเข้าหาแสงที่สว่างที่สุด
  2. ทำ servo sweep ด้วย Non-Blocking Timer — ให้ main loop ยังทำงานอื่นได้ขณะ servo กำลังกวาด
  3. ใช้ Servo + Laser Pointer — ติด laser บน servo แล้วเขียนโปรแกรมให้ฉายจุดเลเซอร์เป็นรูปสี่เหลี่ยมบนผนัง
