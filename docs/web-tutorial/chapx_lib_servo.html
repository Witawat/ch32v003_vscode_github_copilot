📚 หลักการทำงาน

PWM 50Hz คือความถี่มาตรฐานของ RC Servo — 20ms ต่อ 1 คาบ (1/50 = 0.02s) ภายใน 1 คาบ จะมี pulse ความกว้าง 0.5ms–2.5ms คุมมุม 0°–180° โดย center = 1.5ms = 90°

SG90 Specifications: แรงดัน 4.8V, แรงบิด 1.2 kg·cm, ความเร็ว 0.12s/60°, มุมทำงาน 0–180°, น้ำหนัก 9g, ขนาด 23×12×29mm

การคำนวณ Duty Cycle: ที่ 50Hz (20ms) → duty = 0.5ms/20ms × 100 = 2.5% (0°) ถึง 2.5ms/20ms × 100 = 12.5% (180°) → สูตร: duty = 2.5 + (angle × 10.0 / 180.0)

ข้อควรระวัง: Servo สั่งงานด้วย pulse width — ไม่ใช่ duty cycle ของ PWM ทั่วไป ถ้าเปลี่ยนความถี่ PWM ต้องคำนวณ duty ใหม่

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Init PWM + Setup

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "Servo.h"

int main(void) {
    SystemCoreClockUpdate();
    // เริ่ม PWM1 ที่ PD2, ความถี่ 50Hz
    PWM_SimpleInit(1, PD2, 50);
    PWM_Start(1);

🔍 PWM_SimpleInit(ch, pin, freq_hz) ตั้งค่า PWM channel 1 บน PD2 ที่ 50Hz (มาตรฐาน servo) — PWM_Start() เริ่มสร้างสัญญาณ

ขั้นที่ 2: ควบคุมตำแหน่ง Servo

    // หมุนไป 0° (duty ~2.5%)
    PWM_SetDuty(1, 2.5f);
    Delay_Ms(1000);
    // หมุนไป 90° (duty ~7.5%)
    PWM_SetDuty(1, 7.5f);
    Delay_Ms(1000);
    // หมุนไป 180° (duty ~12.5%)
    PWM_SetDuty(1, 12.5f);
    Delay_Ms(1000);

🔍 PWM_SetDuty(ch, percent) รับค่า % — 2.5% = 0°, 7.5% = 90°, 12.5% = 180° ที่ 50Hz — หน่วง 1 วิให้ servo มีเวลาหมุนถึงตำแหน่ง

ขั้นที่ 3: Sweep Loop + แปลงมุม

    while(1) {
        // Sweep 0° → 180°
        for(uint8_t a = 0; a <= 180; a++) {
            float duty = 2.5f + (a * 10.0f / 180.0f);
            PWM_SetDuty(1, duty);
            Delay_Ms(15);
        }
        // Sweep 180° → 0°
        for(uint8_t a = 180; a > 0; a--) {
            float duty = 2.5f + (a * 10.0f / 180.0f);
            PWM_SetDuty(1, duty);
            Delay_Ms(15);
        }
    }
}

🔍 Sweep ไป-กลับ 0–180° — Delay_Ms(15) ให้ servo มีเวลาขยับทีละองศา (0.12s/60° ≈ 2ms/°) — ใช้ float เพื่อความแม่นยำของ duty

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
Servo กระตุก/ไม่หมุนไฟ 5V จ่ายกระแสไม่พอ (servo กิน 500mA+)ใช้ External 5V supply แยก — ห้ามใช้ 3.3V จาก MCU
Servo หมุนเกิน 180°pulse width เกิน 0.5–2.5msจำกัด duty ระหว่าง 2.5%–12.5% ที่ 50Hz
Servo ไม่ตอบสนองGND ไม่ร่วมกันระหว่าง MCU กับ servoต่อ GND ของ MCU, External 5V, และ servo เข้าด้วยกัน
หลายตัวควบคุมไม่แม่นCH32V003 มี PWM channels จำกัดใช้ PCA9685 I²C Servo Driver สำหรับ ≥4 ตัว
Servo สั่นค้างตำแหน่งPWM ความถี่ไม่ใช่ 50Hzตรวจสอบว่า freq=50 — ถ้าใช้ 1kHz pulse จะแคบเกินไป

🛠️ สร้าง Product: Pan-Tilt 2-Axis Camera Mount

ใช้ Servo 2 ตัว + Joystick (2 Potentiometer ADC) ควบคุมกล้อง — แกน X (Pan) PD2, แกน Y (Tilt) PD3

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "Servo.h"

float angle_to_duty(uint8_t angle) {
    if(angle > 180) angle = 180;
    return 2.5f + (angle * 10.0f / 180.0f);
}

int main(void) {
    SystemCoreClockUpdate();
    ADC_Init_Single(PA1); ADC_Init_Single(PA2);
    PWM_SimpleInit(1, PD2, 50); PWM_Start(1);
    PWM_SimpleInit(2, PD3, 50); PWM_Start(2);

    uint8_t pan = 90, tilt = 90;
    while(1) {
        uint16_t adc_x = ADC_Read(PA1);
        uint16_t adc_y = ADC_Read(PA2);
        pan = (uint8_t)(adc_x * 180 / 4095);
        tilt = (uint8_t)(adc_y * 180 / 4095);
        PWM_SetDuty(1, angle_to_duty(pan));
        PWM_SetDuty(2, angle_to_duty(tilt));
        Delay_Ms(20);
    }
}

✏️ ลองทำดู

  1. ทำ Pan-Tilt tracker ที่ขยับตามแสง — ใช้ LDR 2 ตัวอ่านค่า ADC แล้วปรับ servo ให้หันเข้าหาแสงที่สว่างที่สุด
  2. ทำ servo sweep ด้วย Non-Blocking Timer — ให้ main loop ยังทำงานอื่นได้ขณะ servo กำลังกวาด
  3. ใช้ Servo + Laser Pointer — ติด laser บน servo แล้วเขียนโปรแกรมให้ฉายจุดเลเซอร์เป็นรูปสี่เหลี่ยมบนผนัง
