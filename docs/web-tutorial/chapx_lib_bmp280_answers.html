📝 เฉลย — BMP280

ข้อ 1: Weather Station Plot — ส่งข้อมูลผ่าน USART สำหรับกราฟ

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "BMP280.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_Init(115200);

    BMP280_Instance bmp;
    BMP280_Init(&bmp, BMP280_ADDR_LOW);
    USART_Println("T(C),P(hPa),Alt(m)");  // header สำหรับ plot

    Timer_t tmr;
    Start_Timer(&tmr, 500, 1);

    while(1) {
        if(Is_Timer_Expired(&tmr)) {
            float t, p;
            BMP280_Read(&bmp, &t, &p);
            float alt = BMP280_GetAltitude(&bmp, 1013.25f);

            USART_PrintFloat(t, 1);
            USART_Print(",");
            USART_PrintFloat(p, 1);
            USART_Print(",");
            USART_PrintFloat(alt, 1);
            USART_Println("");

            // เปิด Arduino Serial Plotter → Tools → Serial Plotter
            // จะเห็น 3 กราฟ: อุณหภูมิ, ความดัน, ความสูง
        }
    }
}
🧠 แนวคิด: ใช้ format value1,value2,value3 แบบ CSV — Arduino Serial Plotter จะวาดกราฟให้อัตโนมัติ — BMP280_CalcAltitude(1013.25f) คำนวณความสูงจากความดันเทียบกับ sea-level (1013.25 hPa) — แต่ละค่าคั่นด้วย , — ขึ้นบรรทัดใหม่ด้วย USART_Println("") — Timer 500ms ให้กราฟ smooth

ข้อ 2: Altitude Logger — บันทึกความสูง พร้อมปรับ sea-level pressure

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "SimpleFlash.h"
#include "BMP280.h"

#define LOG_ADDR    0x0800F800
#define MAX_ENTRIES 256

typedef struct {
    uint32_t timestamp;
    float temperature;
    float pressure;
    float altitude;
} AltLog_t;

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_Init(115200);

    BMP280_Instance bmp;
    BMP280_Init(&bmp, BMP280_ADDR_LOW);

    float sea_level_hPa = 1013.25f;  // ตั้งค่าได้จากสภาพอากาศวันนั้น

    Timer_t tmr;
    Start_Timer(&tmr, 1000, 1);
    uint8_t idx = 0;

    while(1) {
        if(Is_Timer_Expired(&tmr)) {
            AltLog_t log;
            log.timestamp = Timer_GetMs();
            float t, p;
            BMP280_Read(&bmp, &t, &p);
            log.temperature = t;
            log.pressure = p;
            log.altitude = BMP280_GetAltitude(&bmp, sea_level_hPa);

            USART_Print("Alt:"); USART_PrintFloat(log.altitude, 1);
            USART_Print("m P:"); USART_PrintFloat(log.pressure, 1);
            USART_Print("hPa T:"); USART_PrintFloat(log.temperature, 1);
            USART_Println("C");

            Flash_WriteLog(&log, sizeof(AltLog_t), idx);
            idx = (idx + 1) % MAX_ENTRIES;
        }
    }
}
🧠 แนวคิด: BMP280_CalcAltitude(sea_level_hPa) — ความสูงขึ้นกับความดันอ้างอิงระดับทะเล (เปลี่ยนตามสภาพอากาศได้) — ใช้ sea_level_hPa ตัวแปรที่ calibrate ได้ — ถ้าอยู่ที่สูงจากระดับน้ำทะเลจริง เช่น 500m ควรปรับ sea_level จน altitude ≈ 500 — Flash_WriteLog บันทึก timestamp+temp+pressure+altitude ทุก 1 วิ — อ่านย้อนหลังวิเคราะห์การปีนเขาหรือขึ้นลิฟต์ได้

ข้อ 3: เปรียบเทียบ BMP280 2 ตัวบน I2C บัสเดียวกัน

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "BMP280.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_Init(115200);

    BMP280_Instance bmp1, bmp2;
    BMP280_Init(&bmp1, BMP280_ADDR_LOW);   // SDO → GND = 0x76
    BMP280_Init(&bmp2, BMP280_ADDR_HIGH);  // SDO → VCC = 0x77

    USART_Println("Comparing BMP280 #1 (0x76) vs #2 (0x77)");
    USART_Println("T1,P1,A1,T2,P2,A2,dT,dP");

    Timer_t tmr;
    Start_Timer(&tmr, 1000, 1);

    while(1) {
        if(Is_Timer_Expired(&tmr)) {
            float t1, p1, t2, p2;
            BMP280_Read(&bmp1, &t1, &p1);
            BMP280_Read(&bmp2, &t2, &p2);
            float a1 = BMP280_GetAltitude(&bmp1, 1013.25f);
            float a2 = BMP280_GetAltitude(&bmp2, 1013.25f);

            USART_PrintFloat(t1, 1); USART_Print(",");
            USART_PrintFloat(p1, 1); USART_Print(",");
            USART_PrintFloat(a1, 1); USART_Print(",");
            USART_PrintFloat(t2, 1); USART_Print(",");
            USART_PrintFloat(p2, 1); USART_Print(",");
            USART_PrintFloat(a2, 1); USART_Print(",");
            USART_PrintFloat(t1 - t2, 2); USART_Print(",");
            USART_PrintFloat(p1 - p2, 2); USART_Println("");
        }
    }
}
🧠 แนวคิด: BMP280 ต่อ 2 ตัวบนบัส I2C เดียวได้ด้วยการตั้งขา SDO — ตัวแรก: SDO ต่อ GND = addr 0x76 — ตัวที่สอง: SDO ต่อ VCC = addr 0x77 — ฟังก์ชัน _Addr() ระบุ address ชัดเจน — พิมพ์ค่าเปรียบเทียบ + ค่า Δ (ผลต่าง) — ใช้ตรวจสอบความแม่นยำของเซนเซอร์ ก่อนนำไปใช้งานจริง
