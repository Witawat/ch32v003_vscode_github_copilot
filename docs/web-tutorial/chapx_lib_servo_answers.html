📝 เฉลย — Servo Motor

ข้อ 1: Pan-Tilt Tracker ด้วยปุ่ม ADC 2 ตัว

#include "simple_servo.h"
#include "simple_adc.h"

Servo_t pan, tilt;

static uint8_t map_adc_to_angle(uint16_t adc) {
    return (uint8_t)((uint32_t)adc * 180 / 4095);
}

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    ADC_SimpleInit(ADC_CH0 | ADC_CH1);
    Servo_Attach(&pan, GPIOA, GPIO_Pin_0);
    Servo_Attach(&tilt, GPIOA, GPIO_Pin_1);

    while (1) {
        uint8_t pan_angle  = map_adc_to_angle(ADC_Read(ADC_CH0));
        uint8_t tilt_angle = map_adc_to_angle(ADC_Read(ADC_CH1));
        Servo_Write(&pan,  pan_angle);
        Servo_Write(&tilt, tilt_angle);
        Delay_Ms(20);
    }
}
🧠 แนวคิด: อ่านค่า ADC 2 ช่อง เชื่อมกับ pot 2 ตัว — map_adc_to_angle แปลง 0–4095 → 0–180 องศา — เรียก Servo_Write ทุก 20ms เพื่ออัปเดตทั้ง 2 เซอร์โวพร้อมกัน — Pan ควบคุมแกนนอน (ซ้าย-ขวา) Tilt ควบคุมแกนตั้ง (บน-ล่าง) — ถ้าต้องการลดอาการกระตุก ให้ใช้ low-pass filter หาค่าเฉลี่ย ADC ก่อนส่งเข้า map

ข้อ 2: Servo Sweep 0→180→0 อัตโนมัติ

#include "simple_servo.h"

Servo_t servo;

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    Servo_Attach(&servo, GPIOA, GPIO_Pin_0);

    uint8_t angle = 0;
    int8_t  step  = 1;

    while (1) {
        Servo_Write(&servo, angle);

        angle += step;
        if (angle == 180) step = -1;
        if (angle == 0)   step =  1;

        Delay_Ms(15);  // smooth movement
    }
}
🧠 แนวคิด: ใช้ตัวแปร step บอกทิศทาง (+1 หรือ -1) — เมื่อแตะ 180 ก็กลับทิศลง เมื่อแตะ 0 ก็กลับทิศขึ้น — delay 15ms ต่อ step ทำให้เคลื่อนที่ประมาณ 67 step/วินาที — วิธีนี้ไม่ใช้ for-loop ซ้อน ทำให้ code กระชับและปรับความเร็วได้จากการเปลี่ยน delay — ถ้าต้องการ sweep แบบ sinusoid ให้ใช้ sin() และ map ค่า -1..1 → 0..180 แทน

ข้อ 3: ควบคุม Laser Pointer ด้วย Servo

#include "simple_servo.h"

#define LASER_PIN  GPIO_Pin_2
#define LASER_PORT GPIOA

Servo_t servo;

void laser_on(void)  { GPIO_WriteBit(LASER_PORT, LASER_PIN, Bit_SET); }
void laser_off(void) { GPIO_WriteBit(LASER_PORT, LASER_PIN, Bit_RESET); }

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    Servo_Attach(&servo, GPIOA, GPIO_Pin_0);

    GPIO_InitTypeDef g = {0};
    g.GPIO_Pin   = LASER_PIN;
    g.GPIO_Mode  = GPIO_Mode_Out_PP;
    g.GPIO_Speed = GPIO_Speed_50MHz;
    GPIO_Init(LASER_PORT, &g);

    uint8_t targets[] = {0, 45, 90, 135, 180};
    uint8_t idx = 0;

    while (1) {
        Servo_Write(&servo, targets[idx]);

        if (targets[idx] == 90) laser_on();
        else                      laser_off();

        idx = (idx + 1) % 5;
        Delay_Ms(1000);
    }
}
🧠 แนวคิด: ใช้ GPIO ขาควบคุม laser module — ตรรกะ: เลเซอร์ติดเฉพาะเมื่อ servo ชี้ไปที่ 90° (กึ่งกลาง) — หมุนวน 5 ตำแหน่ง หน่วงเวลา 1 วินาทีต่อตำแหน่ง — ถ้าต้องการให้เลเซอร์กวาดเป็นเส้นตรง (line scan) ให้เปลี่ยนมุมทีละน้อยใน loop และเปิดเลเซอร์ตลอด — GPIO_WriteBit ควบคุมเลเซอร์ ON/OFF แบบ digital — ในทางปฏิบัติ ควรใช้ MOSFET หรือ transistor ขับเลเซอร์เพราะ GPIO ให้กระแสจำกัด

ข้อ 4: ควบคุม 2 Servo พร้อมกันด้วย Sweep แบบ Phase-Shift

#include "simple_servo.h"
#include <math.h>

Servo_t s1, s2;

#ifndef M_PI
#define M_PI 3.14159265358979323846f
#endif

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    Servo_Attach(&s1, GPIOA, GPIO_Pin_0);
    Servo_Attach(&s2, GPIOA, GPIO_Pin_1);

    float phase = 0;
    while (1) {
        uint8_t a1 = (uint8_t)(90 + 90 * sinf(phase));
        uint8_t a2 = (uint8_t)(90 + 90 * sinf(phase + M_PI));

        Servo_Write(&s1, a1);
        Servo_Write(&s2, a2);

        phase += 0.05f;
        if (phase > 2 * M_PI) phase -= 2 * M_PI;

        Delay_Ms(10);
    }
}
🧠 แนวคิด: ใช้ฟังก์ชัน sinf สร้างการเคลื่อนที่แบบ sine wave — servo ตัวที่สองมี phase offset 180° (+M_PI) ทำให้เคลื่อนที่สวนทางกันพอดี — การเคลื่อนที่แบบ smooth sinusoid ดูเป็นธรรมชาติกว่า linear sweep — ปรับ phase += 0.05f ให้ใหญ่ขึ้นเพื่อความเร็ว — ถ้าใช้ CH32V003 ที่ไม่มี FPU ให้ใช้ lookup table แบบ pre-calculated แทนการคำนวณ float
