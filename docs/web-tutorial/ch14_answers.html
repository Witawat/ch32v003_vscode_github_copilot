📝 เฉลย — บทที่ 14: 1-Wire

ข้อ 1: อ่านอุณหภูมิ DS18B20 ตัวเดียว

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include "SimpleHAL.h"

#define OW_PIN  PC0

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
    pinMode(PC1, PIN_MODE_OUTPUT);

    OneWire_Init(OW_PIN);

    USART_Print("=== DS18B20 Single Device ===\r\n");
    USART_Print("PC0 = DQ (with 4.7k pull-up to 3.3V)\r\n");

    uint8_t presence = OneWire_Reset();
    if (!presence) {
        USART_Print("ERROR: No DS18B20 detected!\r\n");
        while (1) {
            digitalWrite(PC1, !digitalRead(PC1));
            Delay_Ms(250);
        }
    }
    USART_Print("DS18B20 found!\r\n\r\n");

    uint32_t reading = 0;

    while (1) {
        reading++;
        digitalWrite(PC1, HIGH);

        // Step 1: Start Conversion (0x44)
        OneWire_Reset();
        OneWire_SkipROM();
        OneWire_WriteByte(0x44);

        // Step 2: Wait for conversion (750ms for 12-bit)
        Delay_Ms(750);

        // Step 3: Read Scratchpad (0xBE)
        OneWire_Reset();
        OneWire_SkipROM();
        OneWire_WriteByte(0xBE);

        uint8_t sp[9];
        for (int i = 0; i < 9; i++) {
            sp[i] = OneWire_ReadByte();
        }

        // Step 4: CRC Check
        uint8_t crc = OneWire_CRC8(sp, 8);
        if (crc != sp[8]) {
            USART_Print("[CRC ERR] ");
        }

        // Step 5: Parse Temperature
        int16_t raw = (int16_t)(sp[0] | (sp[1] << 8));
        float temp_c = raw / 16.0f;

        USART_Print("#");
        USART_PrintNum((int32_t)reading);
        USART_Print(" Temp=");
        USART_PrintNum((int32_t)temp_c);
        USART_Print(".");

        int16_t dec = (int16_t)((temp_c - (int32_t)temp_c) * 100);
        if (dec < 0) dec = -dec;
        if (dec < 10) USART_Print("0");
        USART_PrintNum(dec);

        USART_Print(" C  CRC=");
        USART_Print(crc == sp[8] ? "OK" : "FAIL");
        USART_Print("\r\n");

        digitalWrite(PC1, LOW);
        Delay_Ms(2000);
    }
}
🧠 แนวคิด: DS18B20 ตัวเดียวใช้ OneWire_SkipROM() ข้าม ROM ID — Flow: Reset → SkipROM → 0x44 (Convert) → รอ 750ms → Reset → SkipROM → 0xBE (Read) → อ่าน 9 bytes — Temperature: signed 16-bit, LSB = 1/16 C — temp = raw / 16.0 — Byte 8 = CRC8 — ต้องตรวจสอบเสมอ — Pull-up 4.7kΩ สำคัญ — 1-Wire เป็น open-drain — Timing: CH32V003 ใช้ software bit-bang (ไม่มี HW 1-Wire)

ข้อ 2: ค้นหา devices หลายตัวบน bus

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include "SimpleHAL.h"

#define OW_PIN       PC0
#define MAX_DEVICES  8

uint64_t g_roms[MAX_DEVICES];
uint8_t  g_count = 0;

void print_rom_id(uint64_t rom) {
    USART_Print("0x");
    for (int i = 0; i < 8; i++) {
        USART_PrintHex((rom >> (i * 8)) & 0xFF, 1);
        if (i < 7) USART_Print(":");
    }
}

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);

    OneWire_Init(OW_PIN);

    USART_Print("=== 1-Wire Bus Scan ===\r\n");

    g_count = OneWire_SearchROM(g_roms, MAX_DEVICES);

    USART_Print("Found ");
    USART_PrintNum(g_count);
    USART_Print(" device(s)\r\n\r\n");

    if (g_count == 0) {
        USART_Print("No devices - check wiring & pull-up\r\n");
        while (1);
    }

    // Show all ROM IDs with family code
    for (uint8_t d = 0; d < g_count; d++) {
        uint8_t family = (uint8_t)(g_roms[d] & 0xFF);
        USART_Print("[");
        USART_PrintNum(d);
        USART_Print("] ROM=");
        print_rom_id(g_roms[d]);
        USART_Print(" Family=0x");
        USART_PrintHex(family, 1);

        // Family: 0x28 = DS18B20, 0x22 = DS1822, 0x10 = DS18S20
        if (family == 0x28) {
            USART_Print(" (DS18B20)");
        } else if (family == 0x22) {
            USART_Print(" (DS1822)");
        } else {
            USART_Print(" (Unknown)");
        }
        USART_Print("\r\n");
    }

    USART_Print("\r\nReading temperatures...\r\n");

    while (1) {
        // Start all conversions simultaneously (SkipROM + Convert)
        OneWire_Reset();
        OneWire_SkipROM();
        OneWire_WriteByte(0x44);

        Delay_Ms(750);  // Wait for all

        // Read each device individually
        for (uint8_t d = 0; d < g_count; d++) {
            OneWire_Reset();
            OneWire_MatchROM(g_roms[d]);
            OneWire_WriteByte(0xBE);

            uint8_t sp[9];
            for (int i = 0; i < 9; i++) sp[i] = OneWire_ReadByte();

            if (OneWire_CRC8(sp, 8) != sp[8]) {
                USART_Print("[ERR] ");
                continue;
            }

            int16_t raw = (int16_t)(sp[0] | (sp[1] << 8));
            float t = raw / 16.0f;

            USART_Print("Dev");
            USART_PrintNum(d);
            USART_Print(": ");
            USART_PrintNum((int32_t)t);
            USART_Print(" C  ");
        }
        USART_Print("\r\n");

        Delay_Ms(2000);
    }
}
🧠 แนวคิด: OneWire_SearchROM(roms, max) ใช้ algorithm ค้นหาแบบ binary tree บน 1-Wire bus — คืนจำนวนที่พบ — แต่ละอุปกรณ์มี 64-bit ROM ID: Family(8) + Serial(48) + CRC8 — Family 0x28 = DS18B20 — เทคนิคเร่งความเร็ว: ส่ง Convert T แบบ SkipROM → เริ่มวัดทุกตัวพร้อม → รอ 750ms ครั้งเดียว → อ่านทีละตัวด้วย MatchROM — ลดเวลาจาก N×750ms เป็น 750ms + N×เวลาอ่าน

ข้อ 3: OneWire_MatchROM เลือก device เฉพาะ

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include "SimpleHAL.h"

#define OW_PIN  PC0

// ROM ID ของ DS18B20 ตัวที่ต้องการ (จากข้อ 2)
// ตัวอย่าง: 0x28FF641E0F2B03A2
#define TARGET_ROM  0x28FF641E0F2B03A2ULL

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);

    OneWire_Init(OW_PIN);

    USART_Print("=== OneWire MatchROM ===\r\n");
    USART_Print("Target ROM: 0x28FF641E0F2B03A2\r\n\r\n");

    while (1) {
        // Reset + MatchRom เลือก target device เฉพาะ
        OneWire_Reset();
        OneWire_MatchROM(TARGET_ROM);

        // Command ส่งถึง target device เท่านั้น
        OneWire_WriteByte(0x44);  // Convert T

        Delay_Ms(750);

        OneWire_Reset();
        OneWire_MatchROM(TARGET_ROM);
        OneWire_WriteByte(0xBE);

        uint8_t sp[9];
        for (int i = 0; i < 9; i++) sp[i] = OneWire_ReadByte();

        if (OneWire_CRC8(sp, 8) != sp[8]) {
            USART_Print("CRC Error!\r\n");
        } else {
            int16_t raw = (int16_t)(sp[0] | (sp[1] << 8));
            float temp = raw / 16.0f;

            USART_Print("Target temp: ");
            USART_PrintNum((int32_t)temp);
            USART_Print(" C\r\n");
        }

        // การใช้ MatchROM + SearchROM ร่วมกัน:
        // 1. OneWire_SearchROM() -> get all ROM IDs
        // 2. OneWire_MatchROM(roms[i]) -> select specific
        // 3. Send command -> only that device responds

        Delay_Ms(2000);
    }
}
🧠 แนวคิด: OneWire_MatchROM(rom) ส่ง ROM ID บนบัส — เฉพาะอุปกรณ์ที่มี ROM ตรงกันเท่านั้นที่จะตอบสนอง — ใช้คู่กับ OneWire_SearchROM() — Flow ทั่วไป: SearchROM → เก็บ ROM IDs → MatchROM(device[i]) → Command — ข้อควรระวัง: ROM ID เป็น 64-bit — ต้องใช้ uint64_t — Family code 0x28 = DS18B20 — Read Scratchpad 9 bytes ต่อครั้ง — ตรวจสอบ CRC8 ทุกครั้ง — MatchROM ใช้เมื่อมีหลายตัว — ถ้ามีตัวเดียว SkipROM ง่ายกว่า
