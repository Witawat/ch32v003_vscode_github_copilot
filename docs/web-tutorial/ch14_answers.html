📝 เฉลย — บทที่ 14: 1-Wire

ข้อ 1: อ่านอุณหภูมิ DS18B20 ตัวเดียว (SkipROM + VerifyCRC)

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include "SimpleHAL.h"

#define OW_PIN  PC0

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
    pinMode(PC1, PIN_MODE_OUTPUT);

    // เก็บ bus handle จาก OneWire_Init (factory pattern)
    OneWire_Bus* bus = OneWire_Init(OW_PIN);
    if (bus == NULL) {
        USART_Print("ERROR: bus slot เต็ม (max 4) หรือ pin ผิด\r\n");
        while (1);
    }

    USART_Print("=== DS18B20 Single Device ===\r\n");
    USART_Print("PC0 = DQ (with 4.7k pull-up to 3.3V)\r\n");

    // ตรวจ presence — ส่ง bus arg
    if (!OneWire_Reset(bus)) {
        USART_Print("ERROR: No DS18B20 detected!\r\n");
        while (1) {
            digitalWrite(PC1, !digitalRead(PC1));
            Delay_Ms(250);
        }
    }
    USART_Print("DS18B20 found!\r\n\r\n");

    uint32_t reading = 0;

    while (1) {
        reading++;
        digitalWrite(PC1, HIGH);

        // Step 1: Start Conversion (0x44) — ทุกฟังก์ชันส่ง bus
        OneWire_Reset(bus);
        OneWire_SkipROM(bus);          // 0xCC — มีตัวเดียว
        OneWire_WriteByte(bus, 0x44);   // Convert T

        // Step 2: Wait for conversion (750ms for 12-bit)
        Delay_Ms(750);

        // Step 3: Read Scratchpad (0xBE)
        OneWire_Reset(bus);
        OneWire_SkipROM(bus);
        OneWire_WriteByte(bus, 0xBE);

        uint8_t sp[9];
        OneWire_ReadBytes(bus, sp, 9);  // อ่าน 9 bytes ทีเดียว

        // Step 4: CRC Check (VerifyCRC รวม CRC byte ท้ายสุด — pure function ไม่ต้องส่ง bus)
        bool crc_ok = OneWire_VerifyCRC(sp, 9);
        if (!crc_ok) {
            USART_Print("[CRC ERR] ");
        }

        // Step 5: Parse Temperature — signed 16-bit, LSB = 1/16 C
        int16_t raw = (int16_t)(sp[0] | (sp[1] << 8));
        // 85°C = power-on default — skip
        if (raw == 0x0550) {
            USART_Print("(still powering up)\r\n");
            continue;
        }
        float temp_c = raw / 16.0f;

        USART_Print("#");
        USART_PrintNum((int32_t)reading);
        USART_Print(" Temp=");
        USART_PrintNum((int32_t)temp_c);
        USART_Print(".");

        int16_t dec = (int16_t)((temp_c - (int32_t)temp_c) * 100);
        if (dec < 0) dec = -dec;
        if (dec < 10) USART_Print("0");
        USART_PrintNum(dec);

        USART_Print(" C  CRC=");
        USART_Print(crc_ok ? "OK" : "FAIL");
        USART_Print("\r\n");

        digitalWrite(PC1, LOW);
        Delay_Ms(2000);
    }
}
🧠 แนวคิด: ดึง OneWire_Bus* bus จาก OneWire_Init(pin) แล้วส่ง bus ให้ทุก OneWire_* — DS18B20 ตัวเดียวใช้ OneWire_SkipROM(bus) ข้าม ROM ID — Flow: Reset → SkipROM → 0x44 (Convert) → รอ 750ms → Reset → SkipROM → 0xBE (Read) → OneWire_ReadBytes(bus, sp, 9) อ่าน 9 bytes ทีเดียว — Temperature: signed 16-bit, LSB = 1/16 C — temp = raw / 16.0 — CRC: ใช้ OneWire_VerifyCRC(sp, 9) ตรวจทั้ง 9 bytes (รวม CRC byte ท้ายสุด) — pure function ไม่ต้องส่ง bus — Pull-up 4.7kΩ สำคัญ — 1-Wire เป็น open-drain — Timing: CH32V003 ใช้ software bit-bang (ไม่มี HW 1-Wire) — SimpleHAL ปิด interrupt ใน critical section ให้แล้ว

ข้อ 2: ค้นหา devices หลายตัวบน bus (Triad Search + Jump Temp)

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include "SimpleHAL.h"

#define OW_PIN       PC0
#define MAX_DEVICES  8

// ROM IDs เก็บเป็น uint8_t[8] — ตาม Simple1Wire.h (OneWire_GetAddress เขียน uint8_t[8])
uint8_t g_roms[MAX_DEVICES][8];
uint8_t g_count = 0;

void print_rom_id(const uint8_t* rom) {
    USART_Print("0x");
    for (int i = 0; i < 8; i++) {
        USART_PrintHex(rom[i], 1);
        if (i < 7) USART_Print(":");
    }
}

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);

    OneWire_Bus* bus = OneWire_Init(OW_PIN);
    if (bus == NULL) {
        USART_Print("Bus init failed\r\n");
        while (1);
    }

    USART_Print("=== 1-Wire Bus Scan ===\r\n");

    // Triad Search: ResetSearch → Search → GetAddress — ไม่มี search ROM สำเร็จรูป
    OneWire_ResetSearch(bus);
    while (OneWire_Search(bus) && g_count < MAX_DEVICES) {
        OneWire_GetAddress(bus, g_roms[g_count]);  // ดึง ROM 8 bytes ออกมา
        g_count++;
    }

    USART_Print("Found ");
    USART_PrintNum(g_count);
    USART_Print(" device(s)\r\n\r\n");

    if (g_count == 0) {
        USART_Print("No devices - check wiring & pull-up\r\n");
        while (1);
    }

    // Show all ROM IDs with family code
    for (uint8_t d = 0; d < g_count; d++) {
        uint8_t family = g_roms[d][0];   // byte 0 = family code
        USART_Print("[");
        USART_PrintNum(d);
        USART_Print("] ROM=");
        print_rom_id(g_roms[d]);
        USART_Print(" Family=0x");
        USART_PrintHex(family, 1);

        if (family == 0x28) {
            USART_Print(" (DS18B20)");
        } else if (family == 0x22) {
            USART_Print(" (DS1822)");
        } else if (family == 0x10) {
            USART_Print(" (DS18S20)");
        } else {
            USART_Print(" (Unknown)");
        }
        USART_Print("\r\n");
    }

    USART_Print("\r\nReading temperatures (Jump Temp mode)...\r\n");

    while (1) {
        // === Jump Temp: สั่ง Convert T ทุกตัวพร้อมกันด้วย SkipROM ===
        OneWire_Reset(bus);
        OneWire_SkipROM(bus);
        OneWire_WriteByte(bus, 0x44);

        Delay_Ms(750);  // รอครั้งเดียว — แม้จะมี N ตัว

        // === อ่านทีละตัวด้วย MatchROM ===
        for (uint8_t d = 0; d < g_count; d++) {
            OneWire_Reset(bus);
            OneWire_MatchROM(bus, g_roms[d]);   // 0x55 + 8 bytes ROM
            OneWire_WriteByte(bus, 0xBE);       // Read Scratchpad

            uint8_t sp[9];
            OneWire_ReadBytes(bus, sp, 9);

            if (!OneWire_VerifyCRC(sp, 9)) {
                USART_Print("[CRC ERR] ");
                continue;
            }

            int16_t raw = (int16_t)(sp[0] | (sp[1] << 8));
            if (raw == 0x0550) continue;  // สำรอง 85°C power-on
            float t = raw / 16.0f;

            USART_Print("Dev");
            USART_PrintNum(d);
            USART_Print(": ");
            USART_PrintNum((int32_t)t);
            USART_Print(" C  ");
        }
        USART_Print("\r\n");

        Delay_Ms(2000);
    }
}
🧠 แนวคิด: ใช้ triad แทนฟังก์ชัน search ROM สำเร็จรูปที่ไม่มีใน Simple1Wire.h — OneWire_ResetSearch(bus) รีเซ็ต state → OneWire_Search(bus) หา device ถัดไป (ROM เก็บใน bus->rom) → OneWire_GetAddress(bus, rom) ดึงออกมา — ROM ID เก็บเป็น uint8_t[8] ไม่ใช่ uint64_t — byte 0 = family code (0x28 = DS18B20) — เทคนิค Jump Temp: ส่ง Convert T แบบ SkipROM → เริ่มวัดทุกตัวพร้อม → รอ 750ms ครั้งเดียว → อ่านทีละตัวด้วย OneWire_MatchROM(bus, g_roms[d]) — ลดเวลาจาก N × 750ms เป็น 750ms + N × เวลาอ่าน scratchpad (ราว 5ms) — CRC ตรวจด้วย OneWire_VerifyCRC(sp, 9) (pure function ไม่ต้องส่ง bus)

ข้อ 3: OneWire_MatchROM เลือก device เฉพาะ (หรือใช้ OneWire_Select)

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include "SimpleHAL.h"

#define OW_PIN  PC0

// ROM ID ของ DS18B20 ตัวเป้าหมาย — เก็บเป็น uint8_t[8] (ตาม API)
// ตัวอย่าง: 28 FF 64 1E 0F 2B 03 A2  (Family=0x28 DS18B20, CRC=A2)
uint8_t target_rom[8] = {0x28, 0xFF, 0x64, 0x1E, 0x0F, 0x2B, 0x03, 0xA2};

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);

    OneWire_Bus* bus = OneWire_Init(OW_PIN);
    if (bus == NULL) {
        USART_Print("Bus init failed\r\n");
        while (1);
    }

    USART_Print("=== OneWire MatchROM ===\r\n");
    USART_Print("Target ROM: ");
    for (int i = 0; i < 8; i++) {
        USART_PrintHex(target_rom[i], 1);
        if (i < 7) USART_Print(":");
    }
    USART_Print("\r\n\r\n");

    // ตรวจ CRC ของ target_rom เองก่อน — VerifyCRC ไม่ต้องส่ง bus
    if (!OneWire_VerifyCRC(target_rom, 8)) {
        USART_Print("WARN: target_rom CRC ไม่ตรง — ตรวจ byte ท้ายสุด\r\n");
    }

    while (1) {
        // === ทางที่ 1: Reset + MatchROM แยก ===
        // Reset + MatchROM เลือก target device เฉพาะ
        OneWire_Reset(bus);
        OneWire_MatchROM(bus, target_rom);   // 0x55 + 8 bytes ROM

        // Command ส่งถึง target device เท่านั้น
        OneWire_WriteByte(bus, 0x44);  // Convert T

        Delay_Ms(750);

        OneWire_Reset(bus);
        OneWire_MatchROM(bus, target_rom);
        OneWire_WriteByte(bus, 0xBE);  // Read Scratchpad

        uint8_t sp[9];
        OneWire_ReadBytes(bus, sp, 9);

        if (!OneWire_VerifyCRC(sp, 9)) {
            USART_Print("CRC Error!\r\n");
        } else {
            int16_t raw = (int16_t)(sp[0] | (sp[1] << 8));
            if (raw == 0x0550) {
                USART_Print("(still powering up)\r\n");
            } else {
                float temp = raw / 16.0f;

                USART_Print("Target temp: ");
                USART_PrintNum((int32_t)temp);
                USART_Print(" C\r\n");
            }
        }

        Delay_Ms(2000);

        // === ทางที่ 2: ใช้ OneWire_Select รวม Reset + MatchROM (สั้นกว่า) ===
        // if (OneWire_Select(bus, target_rom)) {
        //     OneWire_WriteByte(bus, 0x44);
        //     Delay_Ms(750);
        //     OneWire_Select(bus, target_rom);
        //     OneWire_WriteByte(bus, 0xBE);
        //     OneWire_ReadBytes(bus, sp, 9);
        // }
    }
}
🧠 แนวคิด: OneWire_MatchROM(bus, rom) รับ const uint8_t* rom (pointer ไปยัง 8 bytes) — ส่ง 0x55 + ROM 8 bytes — เฉพาะอุปกรณ์ที่มี ROM ตรงกันเท่านั้นที่จะตอบสนอง — ต้องเรียก OneWire_Reset(bus) ก่อนเสมอ — หรือใช้ OneWire_Select(bus, rom) ที่รวม Reset + MatchROM ในฟังก์ชันเดียว (คืน false ถ้าไม่มี device) — การใช้งานจริง: Search triad → เก็บ ROM IDs → OneWire_MatchROM(bus, roms[i]) อ่านทีละตัว → ส่ง command — ข้อควรระวัง: ROM ID 8 bytes ไม่ใช่ uint64_t — byte 0 = family code (0x28 = DS18B20) — byte 7 = CRC8 — ตรวจด้วย OneWire_VerifyCRC(rom, 8) ก่อนใช้ก็ได้ — อ่าน scratchpad 9 bytes + CRC ทุกครั้ง — MatchROM ใช้เมื่อมีหลายตัว — ถ้ามีตัวเดียว OneWire_SkipROM(bus) ง่ายกว่า
