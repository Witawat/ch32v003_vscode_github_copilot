📝 เฉลย — MQ Gas เซนเซอร์แก๊ส

ข้อ 1: MQ-2 Smoke Alarm — LED+Buzzer เตือนควัน

#define MQ2_PIN       PA1
#define LED_RED       PC0
#define BUZZER_PIN    PC3
#define SMOKE_THRESH  400

void mq2_smoke_alarm(void) {
    pinMode(MQ2_PIN, PIN_MODE_INPUT);
    pinMode(LED_RED, PIN_MODE_OUTPUT);
    pinMode(BUZZER_PIN, PIN_MODE_OUTPUT);

    while(1) {
        uint16_t adc = analogRead(MQ2_PIN);
        float voltage = adc * 3.3f / 1023.0f;
        float rs = (5.0f - voltage) / voltage * 10000.0f;  // RL=10k
        float ppm = MQGAS_CalcPPM(MQGAS_MQ2, rs);

        USART_Print("ADC:"); USART_PrintNum(adc);
        USART_Print(" PPM:"); USART_PrintFloat(ppm, 1);
        USART_Println("");

        if(ppm > SMOKE_THRESH) {
            digitalWrite(LED_RED, HIGH);
            digitalWrite(BUZZER_PIN, HIGH);
            Delay_Ms(100);
            digitalWrite(BUZZER_PIN, LOW);
            Delay_Ms(100);
        } else {
            digitalWrite(LED_RED, LOW);
            digitalWrite(BUZZER_PIN, LOW);
        }
        Delay_Ms(500);
    }
}
🧠 แนวคิด: อ่าน ADC → คำนวณ Rs (ความต้านทานเซนเซอร์) ผ่าน voltage divider RL=10k — MQGAS_CalcPPM แปลง Rs→PPM — Buzzer+LED เตือนแบบกระพริบเมื่อเกิน threshold — ค่า PPM สูง = ควันมาก = อันตราย

ข้อ 2: Calibrate Ro ในอากาศสะอาด

float mq2_calibrate_ro(void) {
    const int samples = 50;
    float rs_sum = 0;

    USART_Println("Calibrating Ro in clean air...");
    USART_Println("Wait 60s for sensor warm-up");

    // รอเซนเซอร์อุ่น 1 นาที
    for(int i = 0; i < 60; i++) {
        Delay_Ms(1000);
        pinMode(PC5, PIN_MODE_OUTPUT);
        digitalToggle(PC5);  // heartbeat LED
    }

    // เก็บตัวอย่าง 50 ครั้ง
    for(int i = 0; i < samples; i++) {
        uint16_t adc = analogRead(MQ2_PIN);
        float voltage = adc * 3.3f / 1023.0f;
        float rs = (5.0f - voltage) / voltage * 10000.0f;
        rs_sum += rs;
        Delay_Ms(100);
    }

    float rs_avg = rs_sum / samples;
    float ro = rs_avg / 9.8f;  // MQ-2: Rs/Ro ≈ 9.8 ในอากาศสะอาด

    USART_Print("Ro = ");
    USART_PrintFloat(ro, 1);
    USART_Println(" Ohm — saved");

    return ro;
}
🧠 แนวคิด: ในอากาศสะอาด MQ-2 มี Rs/Ro ≈ 9.8 — วัด Rs เฉลี่ย 50 ตัวอย่าง → Ro = Rs_avg / 9.8 — ต้อง warm-up อย่างน้อย 60 วินาทีก่อน calibrate ไม่งั้นค่าจะผิด — เก็บ Ro ไว้ใน Flash หรือ global variable เพื่อใช้ต่อ

ข้อ 3: Threshold-Based Alert — หลายระดับการเตือน

#define LEVEL_SAFE   100
#define LEVEL_WARN   400
#define LEVEL_DANGER 800

void mq2_multi_level_alert(float ppm) {
    if(ppm < LEVEL_SAFE) {
        // Safe: LED เขียวติด, Buzzer เงียบ
        digitalWrite(PC0, HIGH);
        digitalWrite(PC1, LOW);
        digitalWrite(PC3, LOW);
    } else if(ppm < LEVEL_WARN) {
        // Warning: LED เหลืองกระพริบ, Buzzer ดังช้า
        digitalWrite(PC0, LOW);
        digitalWrite(PC1, HIGH);
        digitalWrite(PC3, HIGH); Delay_Ms(300);
        digitalWrite(PC3, LOW);  Delay_Ms(700);
    } else if(ppm < LEVEL_DANGER) {
        // Danger: LED แดงกระพริบเร็ว, Buzzer ดังเร็ว
        digitalWrite(PC0, LOW);
        digitalWrite(PC1, LOW);
        digitalWrite(PC3, HIGH); Delay_Ms(100);
        digitalWrite(PC3, LOW);  Delay_Ms(100);
    } else {
        // Critical: LED แดง+Buzzer ติดค้าง
        digitalWrite(PC0, LOW);
        digitalWrite(PC1, LOW);
        digitalWrite(PC3, HIGH);
    }
}
🧠 แนวคิด: 4 ระดับ: Safe / Warning / Danger / Critical — LED แยกสี (เขียว/เหลือง/แดง) + Buzzer ความถี่ต่างกัน — pattern กระพริบที่ต่างกันช่วยให้รู้ระดับโดยไม่ต้องดู serial — ใช้ pinMode หลายขาพร้อมกันจัดการง่าย
