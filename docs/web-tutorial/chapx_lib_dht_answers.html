📝 เฉลย — DHT11/DHT22

ข้อ 1: LED แสดงสถานะ — เขียวปกติ, แดงร้อน >35°C

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "DHT.h"

#define LED_GREEN PC1
#define LED_RED   PC2

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    pinMode(LED_GREEN, PIN_MODE_OUTPUT);
    pinMode(LED_RED, PIN_MODE_OUTPUT);
    DHT_Init(PC0, DHT11);

    Timer_t tmr;
    Start_Timer(&tmr, 2000, 1);

    while(1) {
        if(Is_Timer_Expired(&tmr)) {
            float t, h;
            if(DHT_Read(&t, &h) == DHT_OK) {
                if(t > 35.0f) {
                    digitalWrite(LED_RED, HIGH);
                    digitalWrite(LED_GREEN, LOW);
                } else {
                    digitalWrite(LED_GREEN, HIGH);
                    digitalWrite(LED_RED, LOW);
                }
            }
        }
    }
}
🧠 แนวคิด: ใช้ if(t > 35.0f) ตัดสินสถานะ — สั่ง digitalWrite แยกขา LED เขียว/แดง พร้อมกัน — เมื่อไฟแดงติด ไฟเขียวต้องดับ (และกลับกัน) — ใช้ Timer_t อ่านทุก 2 วิ ไม่บล็อก loop

ข้อ 2: Data Logger บันทึกลง Flash ทุก 5 นาที

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "SimpleFlash.h"
#include "DHT.h"

#define LOG_ADDR   0x0800F800
#define MAX_ENTRIES 72

typedef struct {
    uint32_t timestamp;
    float temp;
    float hum;
} LogEntry_t;

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_Init(115200);
    DHT_Init(PC0, DHT11);

    Timer_t tmr;
    Start_Timer(&tmr, 300000, 1);   // 5 min = 300,000 ms

    uint8_t idx = 0;
    while(1) {
        if(Is_Timer_Expired(&tmr)) {
            float t, h;
            if(DHT_Read(&t, &h) == DHT_OK) {
                LogEntry_t entry;
                entry.timestamp = Timer_GetMs();
                entry.temp = t;
                entry.hum = h;
                Flash_WriteLog(&entry, sizeof(LogEntry_t), idx);
                idx = (idx + 1) % MAX_ENTRIES;
                USART_Print("Log #"); USART_PrintNum(idx);
                USART_Print(" T:"); USART_PrintFloat(t, 1);
                USART_Print(" H:"); USART_PrintFloat(h, 1);
                USART_Println("%");
            }
        }
        IWDG_Feed();
    }
}
🧠 แนวคิด: ใช้ Timer_t ตั้ง 300,000ms (5 นาที) — Flash_WriteLog() จาก SimpleFlash.h เขียน struct LogEntry_t ลง Flash (มี timestamp+temp+hum) — วน buffer ด้วย modulo idx % MAX_ENTRIES — Flash_ReadLog() อ่านกลับมาทีหลังได้

ข้อ 3: เทียบ DHT11 vs DHT22 บน 2 pin

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "DHT.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_Init(115200);

    DHT_Init(PC0, DHT11);
    DHT_Init(PC1, DHT22);

    Timer_t tmr;
    Start_Timer(&tmr, 2500, 1);  // DHT22 ต้อง >= 2000ms

    while(1) {
        if(Is_Timer_Expired(&tmr)) {
            float t11, h11, t22, h22;
            uint8_t r11 = DHT_Read(PC0, &t11, &h11);
            uint8_t r22 = DHT_Read(PC1, &t22, &h22);

            USART_Println("---------");
            if(r11 == DHT_OK) {
                USART_Print("DHT11 → T:"); USART_PrintFloat(t11, 1);
                USART_Print(" H:"); USART_PrintFloat(h11, 1);
                USART_Println("%");
            } else USART_Println("DHT11 FAIL");

            if(r22 == DHT_OK) {
                USART_Print("DHT22 → T:"); USART_PrintFloat(t22, 1);
                USART_Print(" H:"); USART_PrintFloat(h22, 1);
                USART_Println("%");
            } else USART_Println("DHT22 FAIL");

            if(r11 == DHT_OK && r22 == DHT_OK) {
                USART_Print("ΔT:"); USART_PrintFloat(t22 - t11, 2);
                USART_Print(" ΔH:"); USART_PrintFloat(h22 - h11, 2);
                USART_Println("");
            }
        }
    }
}
🧠 แนวคิด: DHT_Init 2 ตัวบนคนละขา (PC0=DHT11, PC1=DHT22) — DHT_Read(pin, &t, &h) อ่านแยกกัน — ตั้ง Timer 2500ms เพราะ DHT22 ต้องเว้น ≥ 2 วิ — พิมพ์ค่าเปรียบเทียบและค่า Δ (ผลต่าง) ดู accuracy ของทั้ง 2 รุ่น — DHT22 แม่นยำกว่าแต่ช้ากว่า
