สำคัญ: IWDG เมื่อเปิดแล้ว ปิดไม่ได้ จนกว่าจะ reset — การออกแบบนี้ป้องกัน bug ที่อาจปิด watchdog โดยไม่ตั้งใจ

📚 หลักการทำงาน

Watchdog — ตัวจับเวลาถอยหลัง — ถ้าไม่ "feed" (รีเซ็ต timer) ภายในเวลาที่กำหนด → MCU จะถูกรีเซ็ต — ป้องกันระบบค้างจาก bug

IWDG (Independent Watchdog)

  • ใช้ LSI (40kHz) เป็น clock — อิสระจาก system clock — ทำงานแม้ HSI หยุด
  • Timebase: prescaler ÷ LSI = timeout — เลือกได้ 4ms ถึง 26 วินาที
  • เมื่อเปิดแล้ว ปิดไม่ได้ — hardware fuse — ออกแบบมาเพื่อความปลอดภัยสูงสุด
  • ถ้า main loop ค้าง → ไม่มีการ feed → IWDG timeout → reset → ระบบกลับมาทำงานใหม่

WWDG (Window Watchdog)

  • ใช้ PCLK1 เป็น clock — แม่นยำกว่า IWDG
  • Window concept: ต้อง feed ในช่วงเวลาที่กำหนดเท่านั้น — feed เร็วเกินก็ reset, feed ช้าเกินก็ reset
  • มี EWI (Early Wakeup Interrupt) — เตือนก่อน timeout — ให้โอกาสบันทึกข้อมูลสำคัญก่อน reset
  • Timeout: ~51µs ถึง ~28ms — เหมาะกับระบบ real-time

Watchdog Strategy: วาง IWDG_Feed() จุดเดียวใน main loop — ห้าม feed ใน ISR — ถ้า loop ทำงานถูกต้อง จะ feed ทันเวลา — ถ้าค้าง = reset

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: IWDG พื้นฐาน — Feed ใน Main Loop

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"

int main(void) {
    SystemCoreClockUpdate();

    // เช็คว่าตื่นจาก IWDG reset หรือไม่
    if(IWDG_CheckReset()) {
        // ตื่นเพราะ Watchdog timeout!
        // ระบบค้างมาก่อน — log หรือแจ้งเตือน
        Timer_Init();
        USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
        USART_Print("WARNING: Woke from Watchdog reset!\r\n");
        pinMode(PC0, PIN_MODE_OUTPUT);
        // กระพริบ LED เร็ว 3 ครั้ง แสดงว่าเกิด reset
        for(int i=0; i<3; i++) {
            digitalWrite(PC0, HIGH); Delay_Ms(100);
            digitalWrite(PC0, LOW);  Delay_Ms(100);
        }
    }

    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);

    // เริ่ม IWDG — timeout 2 วินาที
    IWDG_SimpleInit(2000);

    USART_Print("System running with IWDG (2s timeout)\r\n");
    pinMode(PC0, PIN_MODE_OUTPUT);

🔍 IWDG_CheckReset() — เช็ค flag ว่า reset ครั้งนี้เกิดจาก IWDG timeout หรือไม่ — IWDG_SimpleInit(2000) — เริ่ม IWDG timeout 2000ms — หลังจากนี้ต้อง feed ภายใน 2s มิฉะนั้น reset!

ขั้นที่ 2: Feed ใน Main Loop

    uint32_t counter = 0;

    while(1) {
        // ทำงานปกติ ...
        digitalWrite(PC0, !digitalRead(PC0));  // Toggle LED
        Delay_Ms(500);

        counter++;
        USART_Print("Loop #");
        USART_PrintNum((int32_t)counter);
        USART_Print(" — feeding watchdog\r\n");

        // Feed watchdog — รีเซ็ต timer กลับไป 2s
        IWDG_Feed();

        // ถ้า loop ใช้เวลาน้อยกว่า 2s — OK
        // ถ้ามี bug ทำให้ loop ค้าง → ไม่ feed → 2s → reset
    }

🔍 IWDG_Feed() — รีเซ็ต IWDG counter กลับไปค่า timeout — ต้องเรียกก่อน timeout — กฎสำคัญ: feed ใน main loop เท่านั้น — ห้าม feed ใน ISR (เพราะ ISR อาจยังทำงานแม้ main loop ค้าง)

ขั้นที่ 3: WWDG — Window Watchdog

    // WWDG — ต้อง feed ใน window 50%-100% ของ timeout
    // ถ้า feed เร็วเกิน (ก่อน 50%) → reset
    // ถ้า feed ช้าเกิน (เกิน 100%) → reset

    WWDG_SimpleInit(
        WWDG_TIMEOUT_28MS,  // timeout
        50                   // window: 50% (14ms-28ms)
    );

    while(1) {
        // ... ทำงานอื่น ...
        Delay_Ms(20);  // ให้เวลาเลย 50% window

        WWDG_Refresh();  // feed ใน window 14-28ms
    }

🔍 WWDG_SimpleInit(timeout, window_pct) — เริ่ม WWDG — Window: ต้อง refresh ในช่วง window_pct%-100% ของ timeout — ป้องกัน loop เร็วเกิน (bug infinite loop แบบไม่ทำอะไร) และช้าเกิน (loop ค้าง)

ขั้นที่ 4: ทดสอบ Watchdog (จงใจทำให้ค้าง)

    // จงใจทำให้ระบบค้าง — Watchdog ควร reset
    if(counter == 5) {
        USART_Print("Intentionally freezing... Watchdog should reset!\r\n");
        Delay_Ms(100);  // รอ USART ส่งเสร็จ
        while(1) {
            // ค้าง! — ไม่มี IWDG_Feed()
            // 2 วินาทีต่อมา → RESET
        }
    }

🔍 จงใจค้างเพื่อทดสอบ watchdog — หลังจาก counter=5 — เข้า infinite loop โดยไม่มี feed — 2 วินาทีต่อมา MCU จะ reset — ดู LED กระพริบเร็ว 3 ครั้งจากการตรวจจับ IWDG reset

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
MCU รีเซ็ตตลอดเวลาIWDG timeout สั้นเกินไป — loop ใช้เวลาเกิน timeoutเพิ่ม timeout หรือลดงานใน loop — เช็คด้วย IWDG_CheckReset()
IWDG ไม่ทำงานDebugger หยุด IWDG อัตโนมัติใน debug modeทดสอบโดยไม่ใช้ debugger — IWDG จะทำงานเฉพาะรันจริง
Feed IWDG แล้วก็ยัง resetลืม feed หลังจาก Delay_Ms() ยาวถ้า delay นานกว่า timeout — ต้อง feed ก่อน delay หรือแยก delay เป็นช่วงสั้นๆ
WWDG reset ตั้งแต่เริ่มFeed ก่อนถึง window (เร็วเกิน)WWDG ต้อง feed ใน window — ถ้า feed เร็วเกิน = reset — หน่วงเวลาให้เลย window ก่อน feed แรก
IWDG_CheckReset() เป็นจริงแต่ไม่ได้ค้างPower glitch หรือ noise trigger IWDGเช็คแหล่งจ่ายไฟ — IWDG อาจ reset จากไฟตกชั่วขณะ

🛠️ สร้าง Product: ระบบที่ไม่เคยค้าง

ระบบ production-grade — IWDG 4s timeout + ตรวจจับ reset + log การค้าง

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"

volatile uint32_t g_wdg_reset_count = 0;

int main(void)
{
    SystemCoreClockUpdate();

    // เช็คสาเหตุ reset
    if(IWDG_CheckReset())
    {
        // ตื่นจาก Watchdog — เกิดการค้าง!
        // หน่วงเวลาเล็กน้อยก่อน init — ป้องกัน infinite reset loop
        for(volatile uint32_t i=0; i<100000; i++);

        g_wdg_reset_count++;
    }

    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
    pinMode(PC0, PIN_MODE_OUTPUT);

    if(g_wdg_reset_count > 0)
    {
        USART_Print("!!! WATCHDOG RESET #");
        USART_PrintNum((int32_t)g_wdg_reset_count);
        USART_Print(" !!!\r\n");

        // ถ้ารีเซ็ตติดกันเกิน 5 ครั้ง — safe mode
        if(g_wdg_reset_count >= 5)
        {
            USART_Print("Entering SAFE MODE\r\n");
            while(1)
            {
                digitalWrite(PC0, HIGH);
                Delay_Ms(200);
                digitalWrite(PC0, LOW);
                Delay_Ms(800);
                // Safe mode: ไม่ทำอะไรเสี่ยง — แค่กระพริบ LED
                IWDG_Feed();
            }
        }
    }

    // เริ่ม IWDG — 4 วินาที
    IWDG_SimpleInit(4000);
    USART_Print("System running (IWDG 4s)\r\n");

    uint32_t tick = 0;
    while(1)
    {
        // งานที่อาจค้าง... (ตัวอย่าง: เช็ค I2C device)
        digitalWrite(PC0, !digitalRead(PC0));

        USART_Print("Tick: ");
        USART_PrintNum((int32_t)++tick);
        USART_Print(" OK\r\n");

        Delay_Ms(1000);  // < 4s — ปลอดภัย

        IWDG_Feed();  // Feed ทุกรอบ loop
    }
}

✏️ ลองทำดู

  1. ลด IWDG timeout เป็น 500ms — เพิ่ม Delay_Ms(2000) ใน loop — สังเกตการ reset
  2. ใช้ WWDG แทน IWDG — ตั้ง window 60% — ทดสอบ feed เร็วเกิน (30%) และ feed ช้าเกิน (110%) — ดูผลต่าง
  3. สร้างระบบที่ใช้ทั้ง IWDG + WWDG พร้อมกัน — IWDG เป็น safety net (4s) — WWDG ตรวจจับ loop anomaly (28ms window)
