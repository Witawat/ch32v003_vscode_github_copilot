🔌 การต่อวงจร

          MAX7219 8x8 LED Matrix SPI
    ┌────────────────────────────────┐
    │ MCU                      MAX   │
    │ 5V  ─────────────── VCC        │
    │ GND ─────────────── GND        │
    │ PC4 ─────────────── CS         │

MAX7219 สื่อสารผ่าน SPI — ใช้ 5V จ่ายไฟมอดูลละ ~330mA (8×8 LEDs on) — ต่อพ่วงหลายโมดูลได้โดย DOUT → DIN

📚 หลักการทำงาน

SPI Chainable: MAX7219 แต่ละตัวมี shift register ภายใน 16-bit — ส่งข้อมูลผ่าน SPI (CS, SCK, DIN) — ข้อมูลเลื่อนจาก DIN → DOUT ไปตัวถัดไป

8×8 Per Module: แต่ละโมดูลมี LED 8×8 = 64 ดวง — MAX7219 ควบคุม 8 แถว × 8 คอลัมน์ ด้วยการ scan multiplex

MAX7219_Handle Pointer: MAX7219_Init() คืน Handle pointer — ทุกฟังก์ชันรับ handle เป็นพารามิเตอร์ตัวแรก — MAX7219_Update() ส่งข้อมูลทั้งหมดขึ้นจอ

Intensity Control: 16 ระดับความสว่าง (0-15) — ควบคุมด้วย PWM ภายใน — MAX7219_SetIntensity(display, 5)

Scroll Text Engine: MAX7219_StartScrollText(display, str, speed_ms, direction) เลื่อนข้อความอัตโนมัติ — direction: SCROLL_LEFT/SCROLL_RIGHT

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Init + Setup

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "MAX7219.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    SPI_Init(SPI_MODE0, SPI_SPEED_8M);

    MAX7219_Handle* display = MAX7219_Init(PC5, PC6, PC4, 1);
    MAX7219_SetIntensity(display, 5);
    MAX7219_Clear(display);

🔍 MAX7219_Init(CLK, MOSI, CS, count) — PC5=CLK, PC6=MOSI, PC4=CS, 1 module — คืน Handle pointer — MAX7219_SetIntensity() 0-15 — MAX7219_Clear() ล้างทุกโมดูล

ขั้นที่ 2: Scroll Text + SetPixel

    // เลื่อนข้อความ (speed 100ms ต่อเฟรม, SCROLL_LEFT)
    MAX7219_StartScrollText(display, "HELLO CH32V003", 100, SCROLL_LEFT);

    // หรือเปิด pixel ทีละจุด
    MAX7219_SetPixel(display, 3, 3, 1);
    MAX7219_SetPixel(display, 4, 3, 1);
    MAX7219_SetPixel(display, 3, 4, 1);
    MAX7219_SetPixel(display, 4, 4, 1);

    MAX7219_Update(display);

🔍 ScrollText(handle, str, speed, direction) เลื่อนข้อความ — SetPixel(handle, x, y, on/off) เปิด/ปิด 1 pixel — MAX7219_Update() ส่งข้อมูลขึ้นจอ

ขั้นที่ 3: Loop + Heart Animation

    Timer_t tmr;
    Start_Timer(&tmr, 500, 1);
    uint8_t frame = 0;

    while(1) {
        if(Is_Timer_Expired(&tmr)) {
            MAX7219_Clear(display);
            if(frame == 0) {
                // Heart frame 1
                MAX7219_SetPixel(display, 3, 1, 1);
                MAX7219_SetPixel(display, 4, 1, 1);
            } else {
                // Heart frame 2
                MAX7219_SetPixel(display, 1, 2, 1);
                MAX7219_SetPixel(display, 6, 2, 1);
            }
            MAX7219_Update(display);
            frame = !frame;
        }
    }
}

🔍 ใช้ Non-Blocking Timer เปลี่ยน frame ทุก 500ms — Clear() + SetPixel() + Update() pattern — อนิเมชันหัวใจ 2 เฟรม

📋 API Reference

ฟังก์ชันคำอธิบายตัวอย่าง
MAX7219_Init(CLK, MOSI, CS, count)เริ่มต้น n โมดูล — คืน Handle*MAX7219_Handle* d = MAX7219_Init(PC5, PC6, PC4, 1)
MAX7219_SetIntensity(h, v)ความสว่าง 0-15MAX7219_SetIntensity(display, 5)
MAX7219_Clear(h)ล้างทุกโมดูลMAX7219_Clear(display)
MAX7219_StartScrollText(h, str, ms, dir)เลื่อนข้อความ — dir: SCROLL_LEFT/SCROLL_RIGHTMAX7219_StartScrollText(display, "HELLO", 100, SCROLL_LEFT)
MAX7219_SetPixel(h, x, y, v)เปิด/ปิด 1 pixel (v=0/1)MAX7219_SetPixel(display, 3, 3, 1)
MAX7219_Update(h)ส่งข้อมูลขึ้นจอทั้งหมดMAX7219_Update(display)

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
LED ไม่ติดเลยSPI pin ผิด — ต้อง PC5=CLK, PC6=MOSIตรวจสอบ pin mapping — CH32V003 SPI1 ใช้ PC5/PC6
LEDs dim / สว่างไม่พอIntensity ต่ำกว่า 3 หรือไฟเลี้ยงไม่พอ (5V)SetIntensity(≥5) — ใช้ external 5V จ่ายไฟโดยตรง
Scroll กระตุก / ไม่ smoothDelay น้อยเกินจน MAX7219 ตามไม่ทันspeed parameter ≥ 50ms ต่อเฟรม — ลอง 100ms
ต่อ 2 โมดูลแล้วไม่ทำงานที่ 2Init ระบุแค่ 1 โมดูลใช้ MAX7219_Init(PC5, PC6, PC4, 2)
LED บางดวงติดค้างSPI noise หรือ CS toggle ผิดจังหวะต่อ pull-up 10k ที่ CS — ลด SPI speed เหลือ 1MHz

🛠️ สร้าง Product: ป้ายไฟวิ่งภาษาไทย 4 โมดูล

MAX7219 4 โมดูลต่อพ่วง — แสดงข้อความเลื่อนภาษาไทย

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "MAX7219.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    SPI_Init(SPI_MODE0, SPI_SPEED_8M);

    MAX7219_Handle* display = MAX7219_Init(PC5, PC6, PC4, 4);
    MAX7219_SetIntensity(display, 5);
    MAX7219_Clear(display);

    while(1) {
        MAX7219_StartScrollText(display, "สวัสดี CH32V003 ", 80, SCROLL_LEFT);
        Delay_Ms(2000);
        MAX7219_StartScrollText(display, "MAX7219 4 Module ", 80, SCROLL_LEFT);
        Delay_Ms(2000);
    }
}
