เซนเซอร์ DS18B20 อ่านอุณหภูมิ (Input) → PID คำนวณ Output → PWM/Relay ควบคุม Heater — closed-loop feedback ต่อเนื่อง

📚 หลักการทำงาน

PID คืออะไร: Proportional-Integral-Derivative — ระบบควบคุมแบบป้อนกลับที่คำนวณค่า error (setpoint − input) แล้วปรับ output เพื่อลด error ให้เหลือน้อยที่สุด

Kp (Proportional): ตอบสนองตาม error ปัจจุบัน — ยิ่ง Kp สูง ยิ่งตอบสนองเร็ว — แต่มากเกินจะเกิด oscillation (แกว่ง) — P = Kp × error

Ki (Integral): สะสม error ในอดีต — กำจัด steady-state error (offset) — I += Ki × error × dt — มากเกินจะเกิด overshoot

Kd (Derivative): ทำนาย error ในอนาคตจากอัตราการเปลี่ยนแปลง — ลด overshoot — D = Kd × d(error)/dt — ไวต่อ noise มาก

dt (Sample Time): ระบุเป็นวินาทีตอน Init — ต้องเรียก PID_Compute() ในช่วงเวลาที่สม่ำเสมอ — default 1.0 วินาที — ใช้ Timer_t จับเวลา

Anti-Windup: เมื่อ output ถูก clamp ด้วยขีดจำกัด — integral term จะหยุดสะสมป้องกัน "windup" ที่ทำให้ระบบตอบสนองช้าเมื่อกลับมาใน range

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Init + ตั้งค่า Kp/Ki/Kd/dt + Setpoint

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "PID.h"

PID_Controller pid;

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_Init(115200);

    PID_Init(&pid, 2.0f, 5.0f, 1.0f, 1.0f);  // Kp, Ki, Kd, dt(s)
    PID_SetSetpoint(&pid, 40.0f);  // Target 40°C

🔍 PID_Init(&pid, Kp, Ki, Kd, dt) — ตั้งค่า gain และ sample time ในขั้นตอนเดียว — dt=1.0f หมายถึง sample ทุก 1 วิ — PID_SetSetpoint() ตั้งค่าเป้าหมาย

ขั้นที่ 2: อ่าน Sensor + Compute + Apply Output

    pinMode(PC1, PIN_MODE_OUTPUT);  // Heater relay
    Timer_t tmr;
    Start_Timer(&tmr, 1000, 1);  // Sample every 1s

    while(1) {
        if(Is_Timer_Expired(&tmr)) {
            float input = read_temperature();  // จาก DS18B20
            float output = PID_Compute(&pid, input);

            if(output > 50) digitalWrite(PC1, HIGH);   // Heater ON
            else digitalWrite(PC1, LOW);               // Heater OFF

            USART_Print("Temp: "); USART_PrintFloat(input, 1);
            USART_Print(" Out: "); USART_PrintFloat(output, 1);
            USART_Println("");
        }
        IWDG_Feed();
    }
}

🔍 PID_Compute(&pid, input) — รับค่าปัจจุบันจากเซนเซอร์ คืนค่า output ที่ปรับแล้ว — ต้องเรียกใน sample time ที่สม่ำเสมอตามค่า dt ที่ตั้งไว้ — ถ้า sample time ไม่คงที่ PID จะคำนวณผิดพลาด

ขั้นที่ 3: Auto-Tune + Ziegler-Nichols

    // Ziegler-Nichols Tuning Method:
    // 1. Set Ki=0, Kd=0
    // 2. เพิ่ม Kp จนระบบ oscillate สม่ำเสมอ → Ku
    // 3. วัดคาบการแกว่ง → Tu (วินาที)
    // 4. คำนวณ:
    //    Kp = 0.6 × Ku
    //    Ki = 2 × Kp / Tu
    //    Kd = Kp × Tu / 8

    float Ku = 4.0f, Tu = 10.0f;
    PID_Init(&pid, 0.6f*Ku, 2.0f*0.6f*Ku/Tu, 0.6f*Ku*Tu/8.0f, Tu/10.0f);

🔍 Ziegler-Nichols เป็นวิธี manual tune ที่ใช้กันแพร่หลาย — หา Ku (ultimate gain) และ Tu (oscillation period) โดยทดลอง — แล้วคำนวณ Kp, Ki, Kd จากสูตร — เรียก PID_Init ใหม่ด้วยค่าที่คำนวณได้

📋 API Reference

ฟังก์ชันคำอธิบายตัวอย่าง
PID_Init(&pid, Kp, Ki, Kd, dt)เริ่มต้น PID พร้อมตั้ง gain + dtPID_Init(&pid, 2.0f, 3.0f, 0.5f, 1.0f)
PID_SetSetpoint(&pid, sp)ตั้งค่าเป้าหมาย (target)PID_SetSetpoint(&pid, 40.0f)
PID_Compute(&pid, input)คำนวณ output จาก input ปัจจุบันfloat out = PID_Compute(&pid, temp)
PID_Reset(&pid)รีเซ็ต integral + derivative สะสมPID_Reset(&pid)

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
อุณหภูมิแกว่ง (oscillate) ไม่หยุดKp สูงเกิน — proportional term overreactลด Kp ลง 50% — หรือเพิ่ม Kd เพื่อ dampen การแกว่ง
อุณหภูมิไม่ถึง setpoint (offset)Ki ต่ำเกิน — integral ไม่พอชดเชย steady-state errorเพิ่ม Ki ทีละ 0.1
Overshoot มาก — พุ่งเกินแล้วค่อยกลับKi สูงเกิน — integral windup สะสมเกินลด Ki ลง — ใช้ dt ที่ถี่ขึ้น
PID Compute คืนค่าเท่าเดิมทุกครั้งSample time ไม่คงที่ — หรือไม่ได้ call Computeใช้ Timer_t จับ sample time ให้เป๊ะ — อย่าใช้ Delay_Ms ใน loop
Noise ทำให้ derivative พุ่งเซนเซอร์มี noise — Kd ขยาย noiseกรอง input ด้วย moving average 3-5 ตัวอย่าง — หรือลด Kd ลง

🛠️ สร้าง Product: Temperature Controller 40°C ±0.5°C

PID + DS18B20 + Relay Heater — ควบคุมอุณหภูมิให้คงที่ 40°C — แสดงผล OLED — มีโหมด Auto-Tune

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "PID.h"
#include "DS18B20.h"
#include "oled_i2c.h"

PID_Controller pid;
DS18B20_Handle sensor;
OLED_Handle oled;
Timer_t tmr_sample, tmr_display;

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_Init(115200);

    DS18B20_SearchDevices(&sensor, PC0, DS18B20_RES_12BIT);
    PID_Init(&pid, 2.0f, 5.0f, 1.0f, 1.0f);
    PID_SetSetpoint(&pid, 40.0f);
    pinMode(PC1, PIN_MODE_OUTPUT);  // Heater relay

    OLED_Init(&oled, OLED_128x64, 0x3C);
    OLED_SetFont(&oled, FONT_8x16);
    Start_Timer(&tmr_sample, 1000, 1);
    Start_Timer(&tmr_display, 500, 1);

    while(1) {
        if(Is_Timer_Expired(&tmr_sample)) {
            float temp;
            if(DS18B20_Read(&sensor, &temp) == DS18B20_OK) {
                float out = PID_Compute(&pid, temp);
                if(out > 50) digitalWrite(PC1, HIGH);
                else digitalWrite(PC1, LOW);
            }
        }
        if(Is_Timer_Expired(&tmr_display)) {
            OLED_Clear(&oled);
            OLED_SetCursor(&oled,0,0); OLED_PrintFloat(&oled,"T:",temp,1);
            OLED_SetCursor(&oled,0,2); OLED_PrintFloat(&oled,"P:",out,1);
            OLED_SetCursor(&oled,0,4); OLED_Print(&oled,relay?"HEAT":" IDLE");
            OLED_Update(&oled);
        }
        IWDG_Feed();
    }
}

✏️ ลองทำดู

  1. เปลี่ยนเป็น PWM Heater — ใช้ PWM_SetDuty() แทน relay on/off — PID output 0-100% → PWM duty — ละเอียดกว่าแบบ on/off
  2. ทำ Profile Ramp — อุ่นจาก 25°C ไป 40°C ใน 5 นาที — ค่อยๆเปลี่ยน setpoint ทีละ 0.05°C ต่อ loop
  3. PID Auto-Tune — ใช้ Ziegler-Nichols หาค่า Kp,Ki,Kd ให้อัตโนมัติ — แล้วบันทึกค่าลง Flash ด้วย Flash_Write()
