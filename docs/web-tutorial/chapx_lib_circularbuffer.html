ไม่ต้องต่อวงจรเพิ่ม — ใช้ USART ที่มีอยู่แล้ว — จุดสำคัญคือ pattern ISR-write + main-read ที่ต้องแยกกันด้วย buffer

📚 หลักการทำงาน

FIFO (First-In First-Out): ข้อมูลที่เขียนก่อนจะถูกอ่านก่อน — เหมือนคิว — เหมาะสำหรับ buffer ข้อมูลที่ผลิตไม่สม่ำเสมอ (เช่น USART, SPI, ADC DMA)

Head/Tail Mechanism: Head = ตำแหน่งเขียนถัดไป — Tail = ตำแหน่งอ่านถัดไป — เมื่อ head == tail: buffer ว่าง — เมื่อ head+1 == tail: buffer เต็ม — wrap-around อัตโนมัติ

ISR-Write + Main-Read Thread-Safe: ISR เขียน (interrupt context) — main loop อ่าน (foreground) — คนละ context แต่ไม่ชนกันเพราะคนละ pointer — ไม่ต้องใช้ mutex หรือ critical section

Overflow Detection: ถ้า buffer เต็ม มี 2 กลยุทธ์: (1) Drop — ไม่เขียนข้อมูลใหม่ คืน 0 (safe, data loss) — (2) Overwrite — เขียนทับข้อมูลเก่าสุด (tail++) — เหมาะกับ real-time stream

0 RAM Overhead: ตัว buffer struct ใช้แค่ head, tail, size, buffer pointer (~12 bytes) — หน่วยความจำทั้งหมดมาจาก array ที่ผู้ใช้ประกาศ — ไม่มี dynamic allocation

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Init + ประกาศ Buffer

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "CircularBuffer.h"

#define BUF_SIZE 128
CircularBuffer cb;
uint8_t rx_buf[BUF_SIZE];
char line[64];
uint8_t line_idx = 0;

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_Init(115200);
    USART_EnableInterrupt(USART1, USART_IT_RXNE);

    CircularBuffer_Init(&cb, rx_buf, BUF_SIZE);
    USART_Println("=== Circular Buffer RX Demo ===");

🔍 CircularBuffer_Init(&cb, buffer_array, size) — buffer_array ต้องเป็น static/global — ห้ามใช้ stack variable — size ควรเป็น power of 2 เพื่อให้ wrap เร็วขึ้น — เปิด RX interrupt เพื่อให้ ISR เขียนเข้า buffer ได้

ขั้นที่ 2: ISR Handler — เขียนเข้า Buffer

void USART1_IRQHandler(void) __attribute__((interrupt)) {
    if(USART_GetFlagStatus(USART1, USART_FLAG_RXNE)) {
        char c = USART_ReceiveData(USART1);
        CircularBuffer_Push(&cb, c);  // Thread-safe!
    }
}

🔍 ISR ต้องสั้นที่สุด — แค่อ่าน RXNE และเขียนเข้า buffer — CircularBuffer_Push() ออกแบบให้ปลอดภัยใน ISR เพราะเขียนเฉพาะ head (main loop ไม่แตะ head) — ถ้า buffer เต็ม ข้อมูลจะถูก drop (return 0) — ไม่บล็อก ISR

ขั้นที่ 3: Main Loop — อ่านจาก Buffer + Process

    while(1) {
        while(CircularBuffer_Available(&cb)) {
            uint8_t c;
            CircularBuffer_Pop(&cb, &c);

            if(c == '\n' || c == '\r') {
                if(line_idx > 0) {
                    line[line_idx] = '\0';
                    USART_Print("ECHO: "); USART_Println(line);
                    process_command(line);  // parse & execute
                    line_idx = 0;
                }
            } else if(line_idx < sizeof(line)-1) {
                line[line_idx++] = c;
            }
        }
        IWDG_Feed();
    }
}

🔍 CircularBuffer_Available() — เช็คว่ามีข้อมูลรออ่านหรือไม่ — CircularBuffer_Pop(&cb, &data) — อ่าน 1 byte จาก tail (ผ่าน out-param) คืน CIRCULAR_BUFFER_OK หรือ EMPTY — main loop อ่านเฉพาะ tail (ISR ไม่แตะ tail) — สร้าง line buffer แยกเพื่อ parse command แบบ line-by-line

📋 API Reference

ฟังก์ชันคำอธิบายตัวอย่าง
CircularBuffer_Init(&cb, buf, size)เริ่มต้น buffer — buf ต้องเป็น array คงที่CircularBuffer_Init(&cb, rx_buf, 128)
CircularBuffer_Push(&cb, data)เขียน 1 byte — คืน CIRCULAR_BUFFER_OK หรือ FULLif(CircularBuffer_Push(&cb, c) != CIRCULAR_BUFFER_OK){/*overflow*/}
CircularBuffer_Pop(&cb, &data)อ่าน 1 byte (out-param) — คืน statusuint8_t c; CircularBuffer_Pop(&cb, &c)
CircularBuffer_Available(&cb)จำนวน byte ที่รออ่านwhile(CircularBuffer_Available(&cb)){...}
CircularBuffer_IsFull(&cb)เช็คว่า buffer เต็มหรือไม่if(CircularBuffer_IsFull(&cb)){...}
CircularBuffer_IsEmpty(&cb)เช็คว่า buffer ว่างหรือไม่if(!CircularBuffer_IsEmpty(&cb)){...}
CircularBuffer_Flush(&cb)ล้าง buffer — รีเซ็ต head=tail=0CircularBuffer_Flush(&cb)
CircularBuffer_Remaining(&cb)จำนวน byte ที่ยังเขียนได้uint16_t free = CircularBuffer_Remaining(&cb)

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
Buffer overflow — ข้อมูลหายMain loop อ่านไม่ทัน — ISR เขียนเร็วกว่าเพิ่ม buffer size → 256 — หรือลด baud rate — หรือใช้ DMA + buffer ใหญ่ขึ้น
ข้อมูลเพี้ยน — อ่านออกมาแปลกๆBuffer ใช้ stack variable — ถูก overwriteประกาศ buffer เป็น global หรือ static — ห้ามใช้ local array
ISR ค้าง — ไม่ตอบสนองเรียก CircularBuffer_Pop() ใน ISR (เขียน+อ่านพร้อมกัน)ISR ควร Write เท่านั้น — Main loop Read เท่านั้น — ห้ามสลับหน้าที่
ข้อมูลเก่าค้างใน bufferไม่ได้ Flush buffer ตอนเปลี่ยนโหมดเรียก CircularBuffer_Flush(&cb) ก่อนเริ่มรับข้อมูลชุดใหม่
Available() คืนค่าไม่ถูกต้องInterrupt เกิดขณะกำลัง Read ใน main — race conditionไม่จำเป็นต้องห่อ __disable_irq() เอง — library ปิด IRQ ให้อัตโนมัติรอบ Push/Pop แล้ว — ใช้ขนาด buffer เป็น power of 2 ช่วยให้ wrap เร็วขึ้น
เขียนสำเร็จแต่ Read ไม่ได้อะไรใช้คนละ handle หรือ buffer array คนละตัวเช็คว่า ISR กับ main loop ใช้ &cb ตัวเดียวกัน — และ buf array ตัวเดียวกัน

🛠️ สร้าง Product: USART RX Buffer + Command Parser

รับคำสั่งผ่าน Serial ด้วย CircularBuffer — แยก command แบบ line-by-line — รองรับ LED ON/OFF, STATUS, HELP — ห้ามพลาดทุกตัวอักษร

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "CircularBuffer.h"

#define BUF_SIZE 128
CircularBuffer cb;
uint8_t rx_buf[BUF_SIZE];
char line[64]; uint8_t idx = 0;

void process(char *cmd) {
    if(!strcmp(cmd, "LED ON"))  { pinMode(PC0,PIN_MODE_OUTPUT); digitalWrite(PC0,HIGH); USART_Println("OK"); }
    else if(!strcmp(cmd, "LED OFF")) { digitalWrite(PC0,LOW); USART_Println("OK"); }
    else if(!strcmp(cmd, "STATUS")) { USART_Println("CH32V003 Online"); }
    else if(!strcmp(cmd, "HELP"))  { USART_Println("LED ON|OFF, STATUS, HELP"); }
    else { USART_Print("Unknown: "); USART_Println(cmd); }
}

void USART1_IRQHandler(void) __attribute__((interrupt)) {
    if(USART_GetFlagStatus(USART1, USART_FLAG_RXNE))
        CircularBuffer_Push(&cb, USART_ReceiveData(USART1));
}

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_Init(115200);
    USART_EnableInterrupt(USART1, USART_IT_RXNE);
    CircularBuffer_Init(&cb, rx_buf, BUF_SIZE);
    USART_Println("Shell Ready. Type HELP");
    while(1) {
        while(CircularBuffer_Available(&cb)) {
            uint8_t c;
            CircularBuffer_Pop(&cb, &c);
            if(c == '\n' || c == '\r') {
                if(idx > 0) { line[idx] = '\0'; process(line); idx = 0; }
            } else if(idx < sizeof(line)-1) { line[idx++] = c; }
        }
        IWDG_Feed();
    }
}

✏️ ลองทำดู

  1. Binary Protocol — แพ็กเกจแบบไบนารี (STX + length + payload + CRC) — ใช้ buffer เก็บแล้ว parse state machine
  2. Dual Buffer — 2 buffers: 1 สำหรับ RX (ISR→main), 1 สำหรับ TX (main→ISR) — ทำ full-duplex buffered UART
  3. Overflow LED — ถ้า buffer เต็ม ให้กระพริบ LED เตือน — ใช้ CircularBuffer_IsFull() ใน main loop
