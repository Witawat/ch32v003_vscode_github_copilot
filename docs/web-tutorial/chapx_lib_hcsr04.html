HC-SR04 — เซนเซอร์วัดระยะอัลตราโซนิค

ระดับกลาง

🧭 แผนภาพการทำงาน

flowchart TD
    A["Trigger 10µs HIGH"] --> B["ส่งคลื่น 40kHz 8 pulses"]
    B --> C["Echo goes HIGH"]
    C --> D["pulseIn(Echo)
= pulse width (µs)"]
    D --> E["Distance = (time × 0.034) / 2"]
    E --> F["Display / Decision"]
    F --> A

🔌 การต่อวงจร

            HC-SR04 — Ultrasonic Distance Sensor
    ┌────────────────────────────────────┐
    │ MCU                       HC-SR04  │
    │ PC0──────TRIG        ┌────────────┤
    │ PC1──────ECHO        │ 40kHz      │
    │ 5V───────VCC         │ Transducer │
    │ GND──────GND         └────────────┤
    │                                    │
    │ Range: 2cm – 400cm                │
    │ Accuracy: ±3mm                    │
    │ Angle: 15° cone                   │
    └────────────────────────────────────┘

PC0─TRIG (output), PC1─ECHO (input) — ใช้ไฟ 5V แต่ TRIG รับ 3.3V logic ได้, ECHO output เป็น 5V → ใช้ voltage divider 2 ตัว (1k+2k) ลดเหลือ ~3.3V

📚 หลักการทำงาน

HC-SR04 วัดระยะทางโดยการส่งพัลส์อัลตราโซนิค 40 kHz จำนวน 8 ลูก แล้วจับเวลาที่คลื่นสะท้อนกลับ — หลักการเดียวกับค้างคาวและเรดาร์

สูตรคำนวณระยะทาง:

Distance (cm) = (Pulse_Width_µs × Speed_of_Sound) / 2

Speed of Sound = 0.0343 cm/µs (@ 20°C)
= pulse_µs / 58 สำหรับ cm
= pulse_µs / 148 สำหรับ inch

Timeout: ถ้าไม่มีวัตถุในระยะ (< 2cm หรือ > 400cm) — ECHO จะค้าง HIGH นานเกิน — ต้องตั้ง timeout ~38ms (เกินเวลาที่คลื่นเดินทางไปกลับ 4 เมตร) เพื่อป้องกันโปรแกรมค้าง

ผลของอุณหภูมิ: ความเร็วเสียงเปลี่ยนตามอุณหภูมิ — v = 331.3 + 0.606×T (°C) — ที่ 40°C คลาดเคลื่อน ~3.5% — แก้โดยวัดอุณหภูมิแล้วชดเชยในสูตร

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Init + Setup

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "HCSR04.h"

HCSR04_Instance s;

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();  // สำหรับ pulseIn
    USART_Init(USART1, BAUD_115200);

    HCSR04_Init(&s, PC0, PC1);
    //              inst  trig echo
    // timeout = 38ms = ระยะสูงสุด ~650cm (ภายใน lib)

🔍 HCSR04_Init(&s, trig, echo) — timeout ป้องกันโปรแกรมค้างถูกตั้งภายใน lib (~38ms) — CH32V003 3.3V อาจต้องใช้ level shifter ที่ ECHO

ขั้นที่ 2: อ่าน/เขียนค่า

    float distanceCm = HCSR04_MeasureCm(&s);

    if(distanceCm < 0) {
        USART_Println("Out of range (TOO_CLOSE / TOO_FAR)");
    } else {
        USART_Print("Distance: ");
        USART_PrintFloat(distanceCm, 1);
        USART_Println(" cm");
    }

🔍 HCSR04_MeasureCm(&s) — คืนค่าลบ (-1) ถ้า timeout หรือพัลส์ผิดปกติ — ใช้ pulseIn(ECHO_PIN, HIGH, timeout) วัดความกว้างพัลส์ขาขึ้นในหน่วย µs

ขั้นที่ 3: Loop + แสดงผล

    OLED_Handle oled;
    OLED_Init(&oled, OLED_128x64, 0x3C);

    char buf[64];
    while(1) {
        float dist = HCSR04_MeasureCm(&s);

        OLED_Clear(&oled);

        if(dist < 0) {
            snprintf(buf, sizeof(buf), "Out of Range");
            OLED_DrawStringAlign(&oled, 0, 0, buf, ALIGN_LEFT);
        } else {
            snprintf(buf, sizeof(buf), "Distance: %.1f cm", dist);
            OLED_DrawStringAlign(&oled, 0, 0, buf, ALIGN_LEFT);

            if(dist < 10)       snprintf(buf, sizeof(buf), "TOO CLOSE!");
            else if(dist < 50)  snprintf(buf, sizeof(buf), "Close");
            else if(dist < 200) snprintf(buf, sizeof(buf), "Medium");
            else                snprintf(buf, sizeof(buf), "Far");
            OLED_DrawStringAlign(&oled, 0, 20, buf, ALIGN_LEFT);
        }

        OLED_Update(&oled);
        Delay_Ms(100);  // 10 readings/sec
    }
}

🔍 แสดงระยะทางบน OLED แบบ real-time — ใช้ Timer_t + Non-Blocking ก็ได้ถ้าต้องการทำอย่างอื่นพร้อมกัน — HCSR04_MeasureCm() ใช้ Delay_Us() ภายใน — blocking ~40ms max

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
อ่านได้ 0 cm / ค่าลบตลอดECHO 5V → CH32V003 3.3V ไม่มี level shifterใช้ voltage divider (R1=1k, R2=2k) ที่ ECHO
วัดระยะผิดอย่างมากวัตถุเอียงหรืออยู่นอก cone 15°หันเซนเซอร์ตั้งฉากกับวัตถุ ใช้พื้นผิวเรียบแข็ง
วัดไม่ได้ระยะ < 2-3 cmDead zone — Echo กลับเร็วกว่าเซนเซอร์จะพร้อมรับถอยเซนเซอร์ห่างวัตถุเกิน 3 cm
ค่า distance แกว่ง ±2-3 cmการรบกวนระหว่าง Trigger→Echo, noise จากแหล่งจ่ายเฉลี่ย 5-10 readings, ต่อ capacitor 100µF ที่ VCC
ติด timeout ตลอด (>38ms)ไม่มีวัตถุในระยะ หรือวัตถุดูดซับเสียง (ผ้า/ฟองน้ำ)เช็คระยะด้วยมือ, ใช้วัตถุแข็งผิวเรียบ

🛠️ สร้าง Product: เครื่องวัดระดับน้ำในถัง

HC-SR04 ติดบนฝาถัง วัดลงไปที่ผิวน้ำ — แสดง % ความจุที่เหลือ

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "HCSR04.h"
#include "oled_i2c.h"

#define TANK_HEIGHT  150.0f  // ความสูงถัง 150cm
#define SENSOR_OFFSET  5.0f  // เซนเซอร์สูงจากฝา 5cm

int main(void)
{
    SystemCoreClockUpdate();
    Timer_Init();
    USART_Init(USART1, BAUD_115200);
    I2C_Init(I2C1, I2C_SPEED_STANDARD);

    OLED_Handle oled;
    OLED_Init(&oled, OLED_128x64, 0x3C);

    HCSR04_Instance s;
    HCSR04_Init(&s, PC0, PC1);

    char buf[64];
    while(1)
    {
        float dist = HCSR04_MeasureCm(&s);
        if(dist < 0) { Delay_Ms(200); continue; }

        float waterLevel = TANK_HEIGHT - dist + SENSOR_OFFSET;
        if(waterLevel < 0) waterLevel = 0;
        if(waterLevel > TANK_HEIGHT) waterLevel = TANK_HEIGHT;

        float percent = (waterLevel / TANK_HEIGHT) * 100.0f;

        OLED_Clear(&oled);
        snprintf(buf, sizeof(buf), "Water: %.1f cm", waterLevel);
        OLED_DrawStringAlign(&oled, 0, 0, buf, ALIGN_LEFT);

        snprintf(buf, sizeof(buf), "Level: %.0f %%", percent);
        OLED_DrawStringAlign(&oled, 0, 20, buf, ALIGN_LEFT);

        // Draw bar graph
        int barWidth = (int)(percent * 128 / 100);
        OLED_DrawRect(&oled, 0, 48, barWidth, 15, 1);

        OLED_Update(&oled);
        Delay_Ms(500);
    }
}

✏️ ลองทำดู

  1. ทำ parking sensor — วัดระยะจากรถถึงกำแพง — ไฟเขียว (>50cm), เหลือง (20-50), แดง (<20cm) + Buzzer ถี่ขึ้น
  2. ทำเซนเซอร์วัดระดับน้ำ — ติด HC-SR04 ด้านบนถัง วัดลงหาผิวน้ำ — log ระดับน้ำทุก 10 นาที
  3. ใช้ NTC ชดเชยอุณหภูมิ — อ่านอุณหภูมิจาก NTC → แก้ speed of sound → ระยะแม่นขึ้นที่อุณหภูมิสูง/ต่ำ
