🔌 การต่อวงจรพื้นฐาน (ทุกตัวอย่างใช้ร่วมกัน)

TJC HMI Display เป็นจอสัมผัส serial ที่คุยกับ MCU ผ่าน UART เท่านั้น — ทุกตัวอย่างในบทนี้ต่อวงจรพื้นฐานแบบเดียวกัน ส่วนอุปกรณ์เสริม (LED/Relay/Sensor/ปุ่มกด) จะระบุแยกในแต่ละตัวอย่าง

    CH32V003              TJC HMI Display
    PD5 (TX) ------------> RX
    PD6 (RX) <------------ TX
    VDD ------------------ VCC (5V หรือ 3.3V ตามสเปคจอ)
    GND ------------------ GND
MCU Pin→อุปกรณ์ขาหมายเหตุ
PD5→TJC DisplayRXMCU TX → จอ RX (USART_PINS_DEFAULT)
PD6←TJC DisplayTXจอ TX → MCU RX
TJC (VCC)←5V/3.3V—เช็คสเปคจอรุ่นที่ใช้ก่อนจ่ายไฟ
TJC (GND)→GND—ต่อร่วมกับ GND ของ MCU เสมอ
💡 v2.1 — ไม่ต้องเขียน IRQ Handler เอง: SimpleUSART.c เป็นเจ้าของ USART1_IRQHandler() อยู่แล้ว และเรียก weak hook USART_RxByteHook() ให้ทุกครั้งที่รับ byte ใหม่ — TJC.c override hook นี้เพื่อรับข้อมูลเข้า rx_buffer ของตัวเองอัตโนมัติ ห้ามเพิ่ม USART1_IRQHandler() ของตัวเองที่เรียก TJC_UART_IRQHandler() (ตามเอกสารรุ่นเก่า) — จะได้ linker error "multiple definition of USART1_IRQHandler" ทันที

ex01_Basic_Command — คำสั่งพื้นฐานควบคุมจอ

📖 แนวคิด

ส่งคำสั่งพื้นฐานไปยังจอ TJC ผ่าน TJC_SendCommand() — เปลี่ยนหน้า (page), ตั้งข้อความ (t0.txt), ตั้งตัวเลข (n0.val), progress bar (j0.val), ซ่อน/แสดง component (vis), จำลองการกด (click), ปรับความสว่างจอ (dim) — และรับ error response ผ่าน callback

🧭 ผังการทำงาน

flowchart TD
    A["TJC_Init(BAUD_115200, ...)"] --> B["TJC_RegisterErrorCallback(OnError)"]
    B --> C["TJC_SendCommand หลายคำสั่ง"]
    C --> D["while(1)"]
    D --> E["TJC_ProcessResponse()"]
    E --> D

💻 โค้ด + อธิบายทีละบรรทัด

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include <SimpleHAL.h>
#include "TJC.h"

void OnError(uint8_t error_code) {
    if (error_code != TJC_ERR_SUCCESS) {
        USART_Print("TJC Error: ");
        USART_Print(TJC_GetErrorString(error_code));
        USART_Print("\r\n");
    }
}

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();

    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
    TJC_Init(BAUD_115200, USART_PINS_DEFAULT);
    TJC_RegisterErrorCallback(OnError);

    Delay_Ms(100);

    TJC_SendCommand("bkcmd=3");             // เปิด response ทุกคำสั่ง (ต้องตั้งก่อนถึงจะเห็น error)
    TJC_SendCommand("page 0");              // ไปหน้า 0
    TJC_SendCommand("t0.txt=\"Hello TJC!\"");  // ตั้งข้อความ component t0
    TJC_SendCommand("n0.val=42");           // ตั้งตัวเลข component n0
    TJC_SendCommand("j0.val=75");           // progress bar j0 = 75%
    TJC_SendCommand("dim=50");              // ความสว่างจอ 50%

    TJC_SendCommand("vis b0,1");            // แสดง component b0
    Delay_Ms(2000);
    TJC_SendCommand("vis b0,0");            // ซ่อน component b0

    TJC_SendCommand("click b0,1");          // จำลองกดปุ่ม b0
    Delay_Ms(100);
    TJC_SendCommand("click b0,0");          // จำลองปล่อยปุ่ม b0

    while (1) {
        TJC_ProcessResponse();              // ★ ต้องเรียกทุกรอบ loop — ประมวลผล UART response
    }
}
  • TJC_Init(baud, pins): รับ USART_BaudRate enum และ USART_PinConfig enum (ไม่ใช่ค่าตัวเลขตรง หรือ TJC_PINS_*)
  • bkcmd=3: ต้องตั้งก่อนเสมอถ้าอยากได้ error/response กลับมา — ค่า default ของจอมักไม่ตอบกลับ
  • TJC_ProcessResponse(): ★ ต้องเรียกใน while(1) เสมอ — ถ้าไม่เรียก callback ทั้งหมดจะไม่ทำงานเลย เพราะข้อมูลค้างอยู่ใน rx_buffer

⚠️ จุดควรระวัง

  • ต้องตั้ง bkcmd=3 ก่อนถึงจะได้ response กลับมาให้ OnError เห็น
  • ทุกคำสั่งจบด้วย 0xFF 0xFF 0xFF อัตโนมัติ — ไม่ต้องเติมเองใน string
  • ลืมเรียก TJC_ProcessResponse() ใน loop = จอไม่ตอบสนอง event ใด ๆ เลย

ex02_Touch_Event — รับ Touch Event ควบคุม LED จริง

📖 แนวคิด

รับ touch event (press/release) จากจอผ่าน callback — เทียบ page_id/component_id เพื่อรู้ว่าใครถูกกด แล้วสั่งเปิด/ปิด LED จริงบน MCU หรือเปลี่ยนหน้าจอ

🔌 อุปกรณ์เสริม

MCU Pin→อุปกรณ์หมายเหตุ
PD3→LED (ผ่าน R 220Ω) → GNDActive-HIGH

TJC Editor: สร้างปุ่ม b0 และ b1 บน page 0 แล้วเปิด Touch Event สำหรับทั้งคู่

💻 โค้ด

#define LED_PIN PD3

void OnTouch(TJC_TouchEvent_t *event) {
    USART_Print("Touch: page=");
    USART_PrintNum(event->page_id);
    USART_Print(" comp=");
    USART_PrintNum(event->component_id);
    USART_Print(" type=");
    USART_PrintNum(event->event_type);
    USART_Print("\r\n");

    if (event->page_id == 0 && event->component_id == 0) {
        digitalWrite(LED_PIN, event->event_type == 0x01);  // 0x01=Press, 0x00=Release
    }

    if (event->page_id == 0 && event->component_id == 1 && event->event_type == 0x01) {
        TJC_SendCommand("page 1");
    }
}

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
    pinMode(LED_PIN, PIN_MODE_OUTPUT);

    TJC_Init(BAUD_115200, USART_PINS_DEFAULT);
    TJC_RegisterTouchEventCallback(OnTouch);

    Delay_Ms(100);
    TJC_SendCommand("page 0");

    while (1) {
        TJC_ProcessResponse();
    }
}
  • TJC_TouchEvent_t: มี page_id, component_id, event_type (0x01=Press, 0x00=Release)
  • TJC_RegisterTouchEventCallback: ต้องเรียกก่อน loop — callback จะถูกเรียกเองจาก TJC_ProcessResponse()

⚠️ จุดควรระวัง

  • ต้องเปิด "Touch Event" ให้ component นั้น ๆ ใน TJC Editor ก่อน ไม่งั้น callback ไม่ทำงานเลย
  • ระวังสลับ event_type — Press=0x01, Release=0x00 (ตรงข้ามกับสัญชาตญาณของหลายคน)

ex03_Touch_Coordinate — รับพิกัด X,Y การสัมผัส

📖 แนวคิด

เปิดโหมดส่งพิกัด (sendxy=1) แล้วรับค่า X,Y ผ่าน callback — ส่งกลับไปแสดงผลบนจอเองในรูปแบบ (x,y)

💻 โค้ด

#include <stdio.h>

void OnTouchCoord(TJC_TouchCoord_t *coord) {
    USART_Print("Touch XY: (");
    USART_PrintNum(coord->x);
    USART_Print(", ");
    USART_PrintNum(coord->y);
    USART_Print(") ");
    USART_Print(coord->event_type == 0x01 ? "Press" : "Release");
    USART_Print("\r\n");

    char buf[32];
    snprintf(buf, sizeof(buf), "t_coord.txt=\"(%d,%d)\"", coord->x, coord->y);
    TJC_SendCommand(buf);
}

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);

    TJC_Init(BAUD_115200, USART_PINS_DEFAULT);
    TJC_RegisterTouchCoordCallback(OnTouchCoord);

    Delay_Ms(100);
    TJC_SendCommand("sendxy=1");    // ★ ต้องเปิดก่อนถึงจะได้พิกัด
    TJC_SendCommand("page 0");

    while (1) {
        TJC_ProcessResponse();
    }
}
  • sendxy=1: คำสั่งฝั่งจอที่บอกให้ส่งพิกัดทุกครั้งที่สัมผัส — ถ้าไม่ส่ง callback นี้จะไม่ถูกเรียกเลย
  • snprintf: สร้างคำสั่งแบบ dynamic ก่อนส่งด้วย TJC_SendCommand() — buffer ต้องใหญ่พอ

⚠️ จุดควรระวัง

  • ลืมส่ง sendxy=1 = ไม่มีพิกัดส่งมาเลย แม้จะสัมผัสจอ
  • buf[32] พอสำหรับพิกัด 4 หลัก — ถ้าจอความละเอียดสูงกว่านี้ควรขยาย buffer

ex04_Numeric_String_Callback — get command + Page ID

📖 แนวคิด

ขอค่าจาก component บนจอด้วยคำสั่ง get แล้วรอรับผ่าน callback แบบ async (ไม่ใช่ return ค่าตรง ๆ) — โพลทุก 3 วินาที พร้อมรับ page ID ปัจจุบันด้วย sendme

💻 โค้ด

#include <string.h>

void OnNumeric(uint32_t value) {
    USART_Print("Numeric received: ");
    USART_PrintNum(value);
    USART_Print("\r\n");
}

void OnString(const char *str, uint16_t len) {
    USART_Print("String received (len=");
    USART_PrintNum(len);
    USART_Print("): ");
    USART_Print(str);
    USART_Print("\r\n");

    if (strcmp(str, "ready") == 0) {
        USART_Print(">> TJC is ready!\r\n");
    }
}

void OnPageId(uint8_t page_id) {
    USART_Print("Page changed: ");
    USART_PrintNum(page_id);
    USART_Print("\r\n");
}

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);

    TJC_Init(BAUD_115200, USART_PINS_DEFAULT);
    TJC_RegisterNumericCallback(OnNumeric);
    TJC_RegisterStringCallback(OnString);
    TJC_RegisterPageIdCallback(OnPageId);

    Delay_Ms(100);
    TJC_SendCommand("page 0");

    uint32_t last_query = 0;

    while (1) {
        TJC_ProcessResponse();

        uint32_t now = Get_CurrentMs();
        if (now - last_query >= 3000) {
            last_query = now;
            TJC_SendCommand("get n0.val");   // ตอบกลับผ่าน OnNumeric
            TJC_SendCommand("get t0.txt");   // ตอบกลับผ่าน OnString
            TJC_SendCommand("sendme");       // ตอบกลับผ่าน OnPageId
        }
    }
}
  • get n0.val / get t0.txt: ขอค่าจาก component — คำตอบมาแบบ async ผ่าน OnNumeric/OnString ไม่ใช่ return ค่าจากฟังก์ชัน
  • sendme: ขอ page ID ปัจจุบัน — ตอบกลับผ่าน OnPageId

⚠️ จุดควรระวัง

  • ส่ง get แล้วผลลัพธ์ไม่มาทันที ต้องรอ callback ในรอบ TJC_ProcessResponse() ถัดไป — อย่าเขียนโค้ดคาดหวัง return ค่าตรง
  • ถ้าส่ง get ถี่เกินไปโดยยังไม่ได้รับ response เดิม อาจสับสนว่าค่าไหนตอบคำถามไหน

ex05_System_Event — จัดการ Startup/Sleep/Wake + Error

📖 แนวคิด

รับ system event จากจอเอง (ไม่ใช่จากผู้ใช้กด) — เมื่อจอ startup / auto sleep (ตาม thsp) / auto wake / SD upgrade — พร้อมแสดง error code เป็น hex ผ่าน OnError

💻 โค้ด

static uint8_t tjc_is_awake = 1;

void OnSystemEvent(uint8_t event_type) {
    switch (event_type) {
        case TJC_RET_STARTUP:
            tjc_is_awake = 1;
            TJC_SendCommand("bkcmd=1");
            TJC_SendCommand("page 0");
            TJC_SendCommand("t0.txt=\"Ready\"");
            TJC_SendCommand("dim=100");
            break;
        case TJC_RET_AUTO_SLEEP:
            tjc_is_awake = 0;
            break;
        case TJC_RET_AUTO_WAKE:
            tjc_is_awake = 1;
            TJC_SendCommand("page 0");
            TJC_SendCommand("dim=100");
            break;
        case TJC_RET_SD_UPGRADE:
            USART_Print(">> TJC SD Upgrade\r\n");
            break;
    }
}

void OnError(uint8_t error_code) {
    if (error_code == TJC_ERR_SUCCESS) return;
    USART_Print("TJC Error 0x");
    if (error_code < 0x10) USART_Print("0");
    USART_PrintHex(error_code, 1);
    USART_Print(": ");
    USART_Print(TJC_GetErrorString(error_code));
    USART_Print("\r\n");
}

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);

    TJC_Init(BAUD_115200, USART_PINS_DEFAULT);
    TJC_RegisterSystemEventCallback(OnSystemEvent);
    TJC_RegisterErrorCallback(OnError);

    Delay_Ms(100);
    TJC_SendCommand("bkcmd=3");
    TJC_SendCommand("thsp=30");   // sleep อัตโนมัติหลัง 30 วินาทีไม่มีการสัมผัส

    while (1) {
        TJC_ProcessResponse();
        if (!tjc_is_awake) {
            Delay_Ms(5000);       // ประหยัดพลังงานตอนจอหลับ
        }
    }
}
  • TJC_RET_STARTUP / AUTO_SLEEP / AUTO_WAKE / SD_UPGRADE: enum event ที่จอส่งมาเอง ไม่ต้อง poll
  • thsp=30: ตั้งเวลา auto-sleep ของจอ (วินาที) — ปรับตามการใช้พลังงานของโปรเจกต์

⚠️ จุดควรระวัง

  • header ตัวอย่างต้นฉบับกล่าวถึง buzzer (PD2) สำหรับ beep ตอน startup แต่โค้ดตัวอย่างนี้ยังไม่ได้ใส่ PWM จริง — เป็นจุดต่อเสริมเผื่อขยายเอง ไม่ใช่บั๊ก
  • OnSystemEvent ถูกเรียกจาก TJC_ProcessResponse() เท่านั้น — ถ้าลืมเรียกจะไม่รู้เลยว่าจอ sleep/wake เมื่อไร

ex06_Custom_Command — TJC สั่งกลับมาหา MCU

📖 แนวคิด

สลับทิศทาง — ให้ปุ่มบนจอ TJC ส่งคำสั่งกลับมาหา MCU เองด้วย prints "cmd|param;" ที่ตั้งไว้ใน Touch Press Event ของ TJC Editor — MCU parse คำสั่งด้วย TJC_RegisterCommandCallback แล้วสั่งงาน LED/Relay/PWM/เปลี่ยนหน้า/รายงานสถานะ

🔌 อุปกรณ์เสริม

MCU Pin→อุปกรณ์หมายเหตุ
PD3→LED (ผ่าน R 220Ω)Active-HIGH
PD4→Relay (ผ่าน R 1kΩ + ทรานซิสเตอร์)ห้ามขับ relay ตรงจาก pin
PD2→Buzzer (PWM)PWM1_CH1

TJC Editor: ตั้งปุ่มแต่ละตัวให้ prints "led|1;", prints "relay|1;", prints "goto|1;", prints "status;" — slider ส่ง prints "pwm|"; prints h0.val,0; prints ";"

💻 โค้ด

#include <string.h>
#include <stdlib.h>
#include <stdio.h>

#define LED_PIN    PD3
#define RELAY_PIN  PD4
#define BUZZER_PWM PWM1_CH1

static uint16_t current_pwm_duty = 0;

void OnTJCCommand(TJC_ReceivedCommand_t *cmd) {
    if (strcmp(cmd->command, "led") == 0 && cmd->param_count >= 1) {
        uint8_t state = (uint8_t)atoi(cmd->params[0]);
        digitalWrite(LED_PIN, state);
        TJC_SendCommand(state ? "t_led.txt=\"ON\"" : "t_led.txt=\"OFF\"");
    }
    else if (strcmp(cmd->command, "relay") == 0 && cmd->param_count >= 1) {
        digitalWrite(RELAY_PIN, (uint8_t)atoi(cmd->params[0]));
    }
    else if (strcmp(cmd->command, "pwm") == 0 && cmd->param_count >= 1) {
        current_pwm_duty = (uint16_t)atoi(cmd->params[0]);
        if (current_pwm_duty > 100) current_pwm_duty = 100;
        PWM_SetDutyCycle(BUZZER_PWM, (uint8_t)current_pwm_duty);
    }
    else if (strcmp(cmd->command, "goto") == 0 && cmd->param_count >= 1) {
        char buf[32];
        snprintf(buf, sizeof(buf), "page %s", cmd->params[0]);
        TJC_SendCommand(buf);
    }
    else if (strcmp(cmd->command, "status") == 0) {
        char buf[64];
        snprintf(buf, sizeof(buf), "t_status.txt=\"%s\"",
                 digitalRead(LED_PIN) ? "LED=ON" : "LED=OFF");
        TJC_SendCommand(buf);
    }
    else {
        TJC_SendCommand("t_err.txt=\"Unknown CMD\"");
    }
}

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);

    pinMode(LED_PIN, PIN_MODE_OUTPUT);
    pinMode(RELAY_PIN, PIN_MODE_OUTPUT);
    PWM_Init(BUZZER_PWM, 1000);
    PWM_Start(BUZZER_PWM);

    TJC_Init(BAUD_115200, USART_PINS_DEFAULT);
    TJC_RegisterCommandCallback(OnTJCCommand);

    Delay_Ms(100);
    TJC_SendCommand("bkcmd=1");
    TJC_SendCommand("page 0");

    while (1) {
        TJC_ProcessResponse();
    }
}
  • TJC_ReceivedCommand_t: มี command (ชื่อคำสั่ง) และ params[]/param_count (พารามิเตอร์ที่แยกด้วย |)
  • atoi: แปลง string param เป็นตัวเลข — ไม่มี validation ว่าเป็นตัวเลขจริง ต้องระวังถ้า UI ส่งค่าผิดรูปแบบ

⚠️ จุดควรระวัง

  • รูปแบบ "cmd|param;" ต้องตั้งใน TJC Editor เอง — MCU ไม่รู้จักปุ่มไหนสั่งอะไรจนกว่าจะได้รับ string นี้
  • ห้ามขับ Relay coil ตรงจาก GPIO pin — ต้องผ่านทรานซิสเตอร์/ULN2003 เสมอ กระแสเกิน 8mA/pin จะทำ MCU เสียหาย
  • ไม่มี strcmp ตรงกับคำสั่งใดเลย = ตกไปเคส "Unknown command" เงียบ ๆ ตรวจสอบชื่อคำสั่งให้ตรงกับที่ตั้งใน TJC Editor

ex07_Full_Features — รวมทุก Callback + Sensor จริง

📖 แนวคิด

ตัวอย่างระดับ production — เปิดใช้ทุก callback พร้อมกัน (System Event, Error, Touch, Command, Numeric, String) ร่วมกับอ่านค่าจริงจาก DHT11 (1-Wire แบบ bit-bang) และ BMP280 (I2C) ทุก 2 วินาที แล้วอัปเดตขึ้นจอ

🔌 อุปกรณ์เสริม

MCU Pin→อุปกรณ์หมายเหตุ
PD3→LED (ผ่าน R 220Ω)—
PD4→Relayผ่านทรานซิสเตอร์
PC0↔DHT11 (DATA)1-Wire bit-bang

💻 โครงสร้างหลัก + อธิบาย

โค้ดเต็มยาว ~330 บรรทัด (ดูต้นฉบับที่ User/Examples/16_TJC_HMI_Display/ex07_Full_Features.c) — ส่วนสำคัญที่เป็นแก่นของบทเรียนคือการรวม callback ทั้งหมดเข้าด้วยกัน:

/* ต่อ callback ครบทุกประเภทในโปรแกรมเดียว */
TJC_Init(BAUD_115200, USART_PINS_DEFAULT);
TJC_RegisterSystemEventCallback(OnSystemEvent);
TJC_RegisterErrorCallback(OnError);
TJC_RegisterTouchEventCallback(OnTouch);
TJC_RegisterCommandCallback(OnTJCCommand);
TJC_RegisterNumericCallback(OnNumeric);
TJC_RegisterStringCallback(OnString);

uint32_t last_sensor = 0;
while (1) {
    TJC_ProcessResponse();               // ★ ประมวลผล callback ทั้งหมดข้างบน

    uint32_t now = Get_CurrentMs();
    if (now - last_sensor >= 2000) {     // non-blocking 2s tick — ไม่ใช้ Delay_Ms คั่น loop
        last_sensor = now;
        UpdateSensorDisplay();           // อ่าน DHT11 + BMP280 แล้วส่งขึ้นจอ
    }
}

UpdateSensorDisplay() อ่าน DHT11 (temp/humidity) ด้วย bit-bang 1-Wire แบบเดียวกับที่สอนในบทที่ 14, และอ่าน BMP280 ด้วย I2C แล้วคำนวณผ่านสูตร compensation ตาม datasheet (คนละเรื่องกับ TJC โดยตรง — ดูรายละเอียดเต็มที่ chapx_lib_bmp280.html) ก่อนส่งผลลัพธ์ขึ้นจอด้วย TJC_SendCommand("t0.txt=...")

void UpdateSensorDisplay(void) {
    char buf[48];
    int16_t dht_temp; uint16_t dht_hum;
    if (DHT11_Read(&dht_temp, &dht_hum) == 0) {
        snprintf(buf, sizeof(buf), "t0.txt=\"%d.%dC\"", dht_temp / 10, dht_temp % 10);
        TJC_SendCommand(buf);
        snprintf(buf, sizeof(buf), "t1.txt=\"%d%%\"", dht_hum / 10);
        TJC_SendCommand(buf);
    }
    /* ... อ่าน + คำนวณค่าความดันจาก BMP280 (ตัดมาแสดงย่อ) ... */
    snprintf(buf, sizeof(buf), "t2.txt=\"%d.%dhPa\"", (int)(pressure / 100), (int)(pressure % 100));
    TJC_SendCommand(buf);
}
  • รวม callback หลายตัว: ทุก callback ทำงานอิสระจากกัน — TJC_ProcessResponse() ตัวเดียวจัดการ dispatch ให้ครบ
  • non-blocking sensor tick: ใช้ Get_CurrentMs() เทียบ delta แทน Delay_Ms() เพื่อไม่ให้ block การรับ touch event ระหว่างรอ sensor

⚠️ จุดควรระวัง

  • ยิ่งลงทะเบียน callback เยอะ ยิ่งต้องระวังให้แต่ละ callback สั้นและไม่ blocking — ทุกตัวถูกเรียกจาก loop เดียวกัน
  • DHT11 bit-bang อ่อนไหวต่อ timing — ถ้ามี interrupt อื่นแทรกระหว่างอ่านอาจได้ค่าผิด (checksum fail คืน error code 3)
  • I2C กับ 1-Wire ใช้ pin คนละกลุ่ม (PC0 vs PC1/PC2) — อย่าต่อชนกัน

💻 โค้ด — Debounce + Repeat State Machine

#define DEBOUNCE_MS     50
#define REPEAT_START    500
#define REPEAT_INTERVAL 150

typedef struct {
    uint8_t pin, prev_state, curr_state, repeat_triggered;
    uint32_t last_change_ms, press_start_ms;
} Button_t;

static uint8_t Button_Update(Button_t *btn) {
    uint8_t raw = digitalRead(btn->pin);
    uint32_t now = Get_CurrentMs();
    btn->prev_state = btn->curr_state;

    if (raw != btn->curr_state && now - btn->last_change_ms >= DEBOUNCE_MS) {
        btn->curr_state = raw;
        btn->last_change_ms = now;
        if (raw == 0) { btn->press_start_ms = now; btn->repeat_triggered = 0; }
    }
    return btn->curr_state;
}

static uint8_t Button_JustPressed(Button_t *btn) {
    return (btn->prev_state == 1 && btn->curr_state == 0);
}

static uint8_t Button_Repeat(Button_t *btn) {
    if (btn->curr_state != 0) return 0;         // ไม่ได้กดอยู่
    uint32_t held = Get_CurrentMs() - btn->press_start_ms;
    if (!btn->repeat_triggered && held >= REPEAT_START) {
        btn->repeat_triggered = 1; btn->press_start_ms = Get_CurrentMs();
        return 1;
    }
    if (btn->repeat_triggered && held >= REPEAT_INTERVAL) {
        btn->press_start_ms = Get_CurrentMs();
        return 1;
    }
    return 0;
}
/* main loop (ย่อ) */
while (1) {
    TJC_ProcessResponse();
    Button_Update(&btn_up); Button_Update(&btn_down);
    Button_Update(&btn_left); Button_Update(&btn_right);

    if (Button_JustPressed(&btn_up) || Button_Repeat(&btn_up)) {
        counter++; if (counter > 100) counter = 100;
        UpdateDisplay();
    }
    if (Button_JustPressed(&btn_left)) {
        GoToPage(current_page > 0 ? current_page - 1 : max_page);
    }
    Delay_Ms(10);
}
  • Debounce: เพิกเฉยการเปลี่ยนสถานะภายใน 50ms หลังการเปลี่ยนครั้งก่อน — กันสัญญาณกระเพื่อมจากหน้าสัมผัสปุ่ม
  • JustPressed vs Repeat: แยก edge-detect (กดครั้งแรก) ออกจาก auto-repeat (กดค้าง) — UX เหมือนรีโมททีวี
  • UP+DOWN พร้อมกัน: ใช้เป็น combo พิเศษ (toggle LED) — เช็คก่อนเช็ค individual press เสมอ

⚠️ จุดควรระวัง

  • ต้องเรียก Button_Update() ทุกปุ่มทุกรอบ loop ก่อนเช็ค JustPressed/Repeat — สลับลำดับจะทำให้ state ค้าง
  • combo UP+DOWN ต้องเช็คก่อนเงื่อนไข individual — ไม่งั้นจะเข้า UP action ก่อนที่จะทันเช็ค combo
  • ไม่มี touchscreen เกี่ยวข้องเลยในตัวอย่างนี้ — ใช้ TJC เป็นจอแสดงผลอย่างเดียว เหมาะกับ enclosure ที่ปุ่มกดทนทานกว่าจอสัมผัส
