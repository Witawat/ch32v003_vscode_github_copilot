⚠️ TMC5160 รองรับมอเตอร์กระแสสูงสุด 10A — VMOT 8-60V — SPI Mode 3 (CPOL=1, CPHA=1) ที่ 4MHz+ — CSN LOW ก่อนส่ง HIGH หลังเสร็จ — ต่อ capacitor 1000μF ขนาน VMOT-GND

📚 หลักการทำงาน

TMC5160 คือ stepper motion controller + driver ในตัวเดียว — ไม่ต้องใช้ MCU สร้าง STEP pulse — สั่งงานผ่าน SPI ด้วยคำสั่งระดับสูง (เคลื่อนที่ไปตำแหน่ง X) — TMC5160 จัดการ acceleration profile เองทั้งหมด

6-Point Ramp Generator (built-in): A1 (start accel) -> V1 (start velocity) -> AMAX (max accel) -> VMAX (max velocity) -> DMAX (max decel) -> VSTOP (stop velocity) — motion profile ราบรื่นโดยไม่ใช้ CPU

Features: StealthChop2 (เงียบ), SpreadCycle (แรงบิด), CoolStep (ปรับกระแสตามโหลด), StallGuard2 (sensorless stall), dcStep (velocity control แม้ stall), Encoder Interface (ABN + SPI encoder) — ครบทุกฟีเจอร์ในชิปเดียว

SPI Registers: GCONF (0x00), IHOLD_IRUN (0x10), RAMPMODE (0x20), XTARGET (0x2D), VMAX (0x27), AMAX (0x24), VACTUAL (0x22), RAMP_STAT (0x35) — เขียน/อ่าน 40-bit datagram: 8-bit addr + 32-bit data

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: สร้าง Instance + Init

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "TMC5160.h"

TMC5160_Instance drv;

int main(void) {
    SystemCoreClockUpdate();

    // Init: EN=PC2, DIR=PC1, STEP=PC0, CSN=PC4, 200 steps/rev
    TMC5160_Init(&drv, PC2, PC1, PC0, PC4, 200);
    USART_Println("TMC5160 Ready!");

🔍 TMC5160_Init(&drv, en, dir, step, csn, stepsPerRev) — เริ่ม SPI + ตั้ง STEP/DIR/EN pins — CSN=PC4 สำหรับ SPI chip select — ไลบรารีจัดการ SPI Mode 3 อัตโนมัติ — 200 steps/rev สำหรับ NEMA23 ทั่วไป

ขั้นที่ 2: ตั้งความเร็ว + Enable

    // ตั้งความเร็ว
    TMC5160_SetSpeed(&drv, 200);   // 200 RPM

    // เปิด driver (StealthChop + ramp generator)
    TMC5160_Enable(&drv);

    // StallGuard threshold
    TMC5160_SetStallThreshold(&drv, 50);

🔍 TMC5160_SetSpeed(&drv, rpm) ตั้งความเร็ว — TMC5160_Enable(&drv) เปิด StealthChop + internal ramp generator — 6-point ramp จัดการในตัว — ไม่ต้องใช้ MCU สร้าง pulse

ขั้นที่ 3: เคลื่อนที่ + Status

    // Move 5000 microsteps (~25 rev at 1/16)
    TMC5160_Move(&drv, 5000);
    USART_Println("Move complete!");

    Delay_Ms(1000);

    // Move back
    TMC5160_Move(&drv, -5000);

    // Sensorless homing
    if(TMC5160_IsStalled(&drv)) {
        USART_Println("Stall! At home.");
        TMC5160_Stop(&drv);
    }

    while(1) {
        TMC5160_Move(&drv, 10000);
        Delay_Ms(500);
        TMC5160_Move(&drv, -10000);
        Delay_Ms(500);
    }
}

🔍 TMC5160_Move(&drv, steps) สั่งให้ TMC5160 จัดการ ramp เอง — blocking move — TMC5160_IsStalled ตรวจสอบ stallGuard — sensorless homing ทำได้โดยเคลื่อนที่จน stall

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
SPI ไม่ตอบสนอง — TMC5160 silentSPI mode ผิด (ต้อง Mode 3) หรือ CSN ไม่ toggleตั้ง SPI Mode 3 (CPOL=1,CPHA=1) — เช็ค CSN LOW/HIGH ด้วย oscilloscope
Motor สั่นแต่ไม่ขยับตาม rampRAMPMODE ไม่ได้ set เป็น position modeเขียน TMC5160_Enable(&drv) — RAMPMODE=0 = hold mode (ไม่เคลื่อนที่)
Motor เร่งแล้วหยุดทันที — RAMP_STAT ค้างAMAX หรือ VMAX ตั้งสูงเกิน — มอเตอร์ stallลด AMAX, VMAX ลง 50% — เช็ค StallGuard2 value — OR เปิด dcStep
TMC5160 ร้อนจน thermal shutdownIRUN สูงเกิน + heatsink ไม่พอ — 10A peakใช้ heatsink + active fan — เปิด CoolStep (TCOOLTHRS) — ลด IRUN
หลาย SPI device บนบัส — CSN แย่งกันใช้ CSN เดียวกันหลายอุปกรณ์CSN ต้องแยกต่อ device — TMC5160 1 ตัวต่อ 1 CSN pin — ห้าม share

🛠️ สร้าง Product: High-Torque CNC Spindle Axis

ใช้ TMC5160 + NEMA23 — เคลื่อนที่ด้วย 6-point ramp — G-Code position mode — StallGuard2 sensorless homing

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "TMC5160.h"

TMC5160_Instance drv;
int32_t pos = 0;
#define STEPS_PER_MM 3200

void home(void) {
    TMC5160_SetSpeed(&drv, 30);  // slow for homing
    TMC5160_Move(&drv, -200000);  // move towards home
    if(TMC5160_IsStalled(&drv)) {
        pos = 0;
        USART_Println("Homed!");
    }
    TMC5160_SetSpeed(&drv, 200); // restore speed
}

void move_to_mm(float mm) {
    int32_t target = (int32_t)(mm * STEPS_PER_MM);
    int32_t delta = target - pos;
    TMC5160_Move(&drv, delta);
    pos = target;
}

int main(void) {
    SystemCoreClockUpdate();
    USART1_Init(115200);
    TMC5160_Init(&drv, PC2, PC1, PC0, PC4, 200);
    TMC5160_SetSpeed(&drv, 200);
    TMC5160_Enable(&drv);
    TMC5160_SetStallThreshold(&drv, 30);
    home();

    while(1) {
        if(USART1_Available()) {
            char cmd = USART1_Read();
            if(cmd == 'H') home();
            if(cmd == '1') move_to_mm(50);
            if(cmd == '2') move_to_mm(0);
        }
    }
}
