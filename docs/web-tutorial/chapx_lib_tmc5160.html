📚 หลักการทำงาน

TMC5160 คือ stepper motion controller + driver ในตัวเดียว — ไม่ต้องใช้ MCU สร้าง STEP pulse — สั่งงานผ่าน SPI ด้วยคำสั่งระดับสูง (เคลื่อนที่ไปตำแหน่ง X) — TMC5160 จัดการ acceleration profile เองทั้งหมด

6-Point Ramp Generator (built-in): A1 (start accel) -> V1 (start velocity) -> AMAX (max accel) -> VMAX (max velocity) -> DMAX (max decel) -> VSTOP (stop velocity) — motion profile ราบรื่นโดยไม่ใช้ CPU

Features: StealthChop2 (เงียบ), SpreadCycle (แรงบิด), CoolStep (ปรับกระแสตามโหลด), StallGuard2 (sensorless stall), dcStep (velocity control แม้ stall), Encoder Interface (ABN + SPI encoder) — ครบทุกฟีเจอร์ในชิปเดียว

SPI Registers: GCONF (0x00), IHOLD_IRUN (0x10), RAMPMODE (0x20), XTARGET (0x2D), VMAX (0x27), AMAX (0x24), VACTUAL (0x22), RAMP_STAT (0x35) — เขียน/อ่าน 40-bit datagram: 8-bit addr + 32-bit data

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: SPI Init + Register R/W

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"

#define CSN PC4
#define REG_IHOLD_IRUN 0x10
#define REG_XTARGET    0x2D
#define REG_VMAX       0x27
#define REG_AMAX       0x24
#define REG_RAMP_STAT  0x35

void tmc5160_write(uint8_t reg, uint32_t val) {
    digitalWrite(CSN, LOW);
    SPI_Transfer((reg | 0x80) & 0xFF);
    SPI_Transfer((val >> 24) & 0xFF);
    SPI_Transfer((val >> 16) & 0xFF);
    SPI_Transfer((val >> 8) & 0xFF);
    SPI_Transfer(val & 0xFF);
    digitalWrite(CSN, HIGH);
}

uint32_t tmc5160_read(uint8_t reg) {
    digitalWrite(CSN, LOW);
    SPI_Transfer(reg & 0x7F);  // Read
    uint32_t v = (uint32_t)SPI_Transfer(0) << 24;
    v |= (uint32_t)SPI_Transfer(0) << 16;
    v |= (uint32_t)SPI_Transfer(0) << 8;
    v |= SPI_Transfer(0);
    digitalWrite(CSN, HIGH);
    return v;
}

🔍 TMC5160 ใช้ SPI 40-bit datagram: 8-bit address (W=0x80) + 32-bit data — CSN LOW ก่อน-หลังทุก transaction — SPI mode 3, CPOL=1, CPHA=1

ขั้นที่ 2: Configure Ramp + Current

int main(void) {
    SystemCoreClockUpdate();
    SPI_Init(PC5, PC6, PC7, SPI_MODE3);
    pinMode(CSN, PIN_MODE_OUTPUT);
    digitalWrite(CSN, HIGH);

    // Enable StealthChop + internal ramp generator
    tmc5160_write(0x00, 0x00000004); // GCONF: stealthChop
    tmc5160_write(0x20, 0x00000001); // RAMPMODE: position mode

    // Current: IHOLD=8/31 (~0.4A), IRUN=16/31 (~0.8A)
    tmc5160_write(REG_IHOLD_IRUN, 0x00021008);

    // 6-Point Ramp parameters (in microsteps per second)
    tmc5160_write(0x24, 50000);  // AMAX: max accel
    tmc5160_write(0x26, 10000);  // DMAX: max decel
    tmc5160_write(REG_VMAX, 200000); // VMAX: max velocity
    tmc5160_write(0x2A, 10);     // A1: start accel
    tmc5160_write(0x2B, 500);    // V1: start velocity
    tmc5160_write(0x2C, 100);    // VSTOP

🔍 RAMPMODE=1 เปิด position mode — set IHOLD/IRUN ตาม spec มอเตอร์ — 6-point ramp ใส่ค่าใน microsteps/sec — ไม่ต้องใช้ MCU จัดการ pulse อีกต่อไป

ขั้นที่ 3: Motion Commands + Status Poll

    while(1) {
        // Move to position 100000 microsteps (~50 rev at 1/16)
        tmc5160_write(REG_XTARGET, 100000);
        // Wait until position reached
        while(1) {
            uint32_t stat = tmc5160_read(REG_RAMP_STAT);
            if(stat & 0x04) break;  // position_reached flag
            Delay_Ms(10);
        }
        Delay_Ms(1000);

        // Move back to 0
        tmc5160_write(REG_XTARGET, 0);
        while(1) {
            uint32_t stat = tmc5160_read(REG_RAMP_STAT);
            if(stat & 0x04) break;
            Delay_Ms(10);
        }
        Delay_Ms(1000);

        // Read actual velocity for monitoring
        uint32_t vactual = tmc5160_read(0x22);
        // VACTUAL in microsteps/sec (signed int24)
        int32_t speed = (int32_t)(vactual << 8) >> 8;
    }
}

🔍 tmc5160_write(REG_XTARGET, pos) สั่งให้ TMC5160 จัดการ ramp เอง — poll RAMP_STAT bit 2 (position_reached) — VACTUAL เป็น signed 24-bit แสดงความเร็วจริง

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
SPI ไม่ตอบสนอง — TMC5160 silentSPI mode ผิด (ต้อง Mode 3) หรือ CSN ไม่ toggleตั้ง SPI Mode 3 (CPOL=1,CPHA=1) — เช็ค CSN LOW/HIGH ด้วย oscilloscope
Motor สั่นแต่ไม่ขยับตาม rampRAMPMODE ไม่ได้ set เป็น position modeเขียน tmc5160_write(0x20, 1) — RAMPMODE=0 = hold mode (ไม่เคลื่อนที่)
Motor เร่งแล้วหยุดทันที — RAMP_STAT ค้างAMAX หรือ VMAX ตั้งสูงเกิน — มอเตอร์ stallลด AMAX, VMAX ลง 50% — เช็ค StallGuard2 value — OR เปิด dcStep
TMC5160 ร้อนจน thermal shutdownIRUN สูงเกิน + heatsink ไม่พอ — 10A peakใช้ heatsink + active fan — เปิด CoolStep (TCOOLTHRS) — ลด IRUN
หลาย SPI device บนบัส — CSN แย่งกันใช้ CSN เดียวกันหลายอุปกรณ์CSN ต้องแยกต่อ device — TMC5160 1 ตัวต่อ 1 CSN pin — ห้าม share

🛠️ สร้าง Product: High-Torque CNC Spindle Axis

ใช้ TMC5160 + NEMA23 — เคลื่อนที่ด้วย 6-point ramp — G-Code position mode — StallGuard2 sensorless homing

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"

int32_t pos = 0;
#define STEPS_PER_MM 3200

void move_to_mm(float mm) {
    int32_t target = (int32_t)(mm * STEPS_PER_MM);
    TMC5160_WriteReg(0x2D, target);
    while(1) {
        uint32_t st = TMC5160_ReadReg(0x35);
        if(st & 0x04) break;  // position reached
        // Check stallGuard for safety
        uint32_t sg = TMC5160_ReadReg(0x52);
        if(sg < 10) { TMC5160_WriteReg(0x2D, pos); break; } // stall -> stop
    }
    pos = target;
}

void home() {
    TMC5160_WriteReg(0x27, 50000);  // slow VMAX for homing
    TMC5160_WriteReg(0x2D, -200000); // move towards limit
    while(1) {
        uint32_t sg = TMC5160_ReadReg(0x52);
        if(sg < 15) {  // Stall = home
            TMC5160_WriteReg(0x2D, TMC5160_ReadReg(0x22)/256);
            break;
        }
    }
    pos = 0; TMC5160_WriteReg(0x2D, 0);  // set as zero
    TMC5160_WriteReg(0x27, 200000); // restore VMAX
}

int main(void) {
    SystemCoreClockUpdate();
    USART1_Init(115200);
    TMC5160_Init(PC4);
    TMC5160_WriteReg(0x00, 0x04);   // StealthChop
    TMC5160_WriteReg(0x20, 0x01);   // Position mode
    TMC5160_WriteReg(0x10, 0x00021008); // Current
    TMC5160_WriteReg(0x24, 50000);  // AMAX
    TMC5160_WriteReg(0x27, 200000); // VMAX
    home();

    while(1) {
        if(USART1_Available()) {
            char cmd = USART1_Read();
            if(cmd == 'H') home();
            if(cmd == '1') move_to_mm(50);
            if(cmd == '2') move_to_mm(0);
        }
    }
}
