🔌 การต่อวงจร

            BH1750 — เซนเซอร์วัดแสง I2C
    ┌────────────────────────────────────┐
    │ MCU                         BH1750 │
    │ PC1──────SDA                ┌──────┤
    │ PC2──────SCL                │      │
    │ 3.3V─────VCC    ADDR─GND    │0x23  │
    │ GND──────GND                └──────┤
    │                                    │
    │ ADDR=LOW → 0x23  ADDR=HIGH → 0x5C │
    └────────────────────────────────────┘
MCU Pin→อุปกรณ์ขาหมายเหตุ
PC1↔SDA-สายข้อมูล I2C — ต้องมี R 4.7kΩ pull-up ไป 3.3V
PC2→SCL-สายนาฬิกา I2C — ต้องมี R 4.7kΩ pull-up ไป 3.3V
3.3V→VCC-ไฟเลี้ยง 3.3V
GND→GND-กราวด์ร่วม
GND→ADDR-ต่อ GND = address 0x23, ต่อ VCC = 0x5C

📚 หลักการทำงาน

BH1750 เป็นเซนเซอร์วัดความสว่างแบบ I2C ambient light sensor ย่านวัด 1–65535 lux — การตอบสนองสเปกตรัมใกล้เคียงกับตามนุษย์

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Init + Setup

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "BH1750.h"

BH1750_Instance bh;

int main(void) {
    SystemCoreClockUpdate();
    USART_Init(USART1, BAUD_115200);
    I2C_Init(I2C1, I2C_SPEED_STANDARD);

    BH1750_Init(&bh, 0x23);  // addr=0x23 (ADDR=GND)
    USART_Println("BH1750 Ready!");

🔍 BH1750_Init(&bh, addr) — instance-based, address 0x23 หรือ 0x5C

ขั้นที่ 2: อ่านค่า lux

    uint16_t lux = BH1750_ReadLux(&bh);
    if(lux == 0xFFFF) {
        USART_Println("Sensor error!");
    } else {
        USART_Print("Light: ");
        USART_PrintInt(lux);
        USART_Println(" lux");
    }

🔍 BH1750_ReadLux(&bh) — คืนค่า 1–65535 lux — 0xFFFF = I2C error

ขั้นที่ 3: Loop — ไฟอัตโนมัติ

    while(1) {
        uint16_t lux = BH1750_ReadLux(&bh);
        if(lux != 0xFFFF) {
            uint8_t duty = (uint8_t)((uint32_t)lux * 255 / 1000);
            if(duty > 255) duty = 255;
            PWM_SetDuty(TIM2_CH1, duty);
        }
        Delay_Ms(200);
    }
}

🔍 ใช้ lux ปรับ LED brightness อัตโนมัติ — มืด→LED สว่าง, สว่าง→LED หรี่

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
อ่านได้ 0 lux ตลอดAddress ผิด (0x23 vs 0x5C)เช็ค ADDR pin — ต่อ GND ได้ 0x23
ค่า overflow (65535)แสงแดดจัด >65535 luxเพิ่มตัวกรองแสง (ND filter)

🛠️ สร้าง Product: ไฟอัตโนมัติ

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "BH1750.h"

int main(void)
{
    SystemCoreClockUpdate();
    USART_Init(USART1, BAUD_115200);
    I2C_Init(I2C1, I2C_SPEED_STANDARD);
    PWM_Init(TIM2, TIM2_CH1, PC4, 1000);

    BH1750_Instance bh;
    BH1750_Init(&bh, 0x23);

    while(1)
    {
        uint16_t lux = BH1750_ReadLux(&bh);
        if(lux == 0xFFFF) { Delay_Ms(1000); continue; }

        uint8_t duty;
        if(lux <= 50)      duty = 255;
        else if(lux < 500) duty = (uint8_t)((500 - lux) * 255 / 450);
        else               duty = 0;

        PWM_SetDuty(TIM2_CH1, duty);

        USART_Print("Lux: "); USART_PrintInt(lux);
        USART_Print(" | PWM: "); USART_PrintInt(duty); USART_Println("");
        Delay_Ms(200);
    }
}

✏️ ลองทำดู

  1. ทำ lux logger — บันทึกค่า lux ทุก 1 วิ ผ่าน USART
  2. ทำไฟกลางคืนอัตโนมัติ — lux < 30 → LED ติด, lux > 50 → LED ดับ
