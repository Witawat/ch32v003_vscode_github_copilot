📚 หลักการทำงาน

Relay Module ใช้ GPIO ควบคุมสวิตช์กล — ภายในประกอบด้วย Optocoupler (แยกสัญญาณไฟฟ้า), Transistor Driver (ขยายกระแส), Flyback Diode (ป้องกันแรงดันย้อนกลับจาก coil), และ Relay (สวิตช์กล)

Active LOW vs HIGH: Relay module ส่วนใหญ่เป็น Active LOW — GPIO=0 → Relay ON (COM เชื่อมกับ NO), GPIO=1 → Relay OFF — ต้องเช็ค datasheet ของ module ก่อน — บางรุ่นเป็น Active HIGH ใช้ jumper เลือกได้

Optocoupler แยกวงจร MCU ออกจากวงจร Relay ด้วยแสง — ป้องกัน noise และ voltage spike จาก coil ไม่ให้กลับมาทำลาย MCU

Flyback Diode ต่อขนานกับ coil — เมื่อ transistor ปิด coil จะคายพลังงานกลับ — diode ลัดวงจรพลังงานนี้ไม่ให้ spike ทำลาย transistor

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: สร้าง Instance + Init

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "Relay.h"

Relay_Instance relay1;
Relay_Instance relay2;

int main(void) {
    SystemCoreClockUpdate();

    // Active LOW: LOW=ON, HIGH=OFF
    Relay_Init(&relay1, PC0, RELAY_ACTIVE_LOW);
    Relay_Init(&relay2, PC1, RELAY_ACTIVE_LOW);

    // เริ่มต้นปิดทุก relay
    Relay_Off(&relay1);
    Relay_Off(&relay2);

🔍 Relay_Init(&relay, pin, activeLevel) ใช้ instance-based — RELAY_ACTIVE_LOW หรือ RELAY_ACTIVE_HIGH — เริ่มต้นด้วย Off() เพื่อให้ relay OFF — ฟังเสียง "click" เพื่อยืนยัน

ขั้นที่ 2: ON/OFF + Toggle

    // ทดสอบเปิด-ปิด relay ด้วย Non-Blocking Timer
    Timer_t tmr_relay;
    Start_Timer(&tmr_relay, 2000, 1);  // toggle ทุก 2 วิ

    uint8_t relay_state = 0;
    while(1) {
        if(Is_Timer_Expired(&tmr_relay)) {
            relay_state = !relay_state;
            if(relay_state) {
                Relay_On(&relay1);
                Relay_On(&relay2);
            } else {
                Relay_Off(&relay1);
                Relay_Off(&relay2);
            }
        }
        // main loop ทำงานอื่นได้อิสระ
    }
}

🔍 Relay_On(&relay) / Relay_Off(&relay) ใช้เปิดปิด — Relay_Toggle(&relay) สลับสถานะ — Relay_IsOn(&relay) อ่านสถานะ — relay เป็น mechanical device อายุใช้งาน ~100,000 ครั้ง

ขั้นที่ 3: Multi-Relay + Button + Safety

// 2-channel relay + 2 buttons (toggle)
int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    Relay_Init(&relay1, PC0, RELAY_ACTIVE_LOW);
    Relay_Init(&relay2, PC1, RELAY_ACTIVE_LOW);
    Relay_Off(&relay1);
    Relay_Off(&relay2);

    pinMode(PC3, PIN_MODE_INPUT_PULLUP);  // BTN1
    pinMode(PC4, PIN_MODE_INPUT_PULLUP);  // BTN2

    uint8_t btn1_prev = 1, btn2_prev = 1;

    while(1) {
        uint8_t b1 = digitalRead(PC3);
        uint8_t b2 = digitalRead(PC4);

        // กดปุ่ม toggle relay (falling edge)
        if(!b1 && btn1_prev) {
            Relay_Toggle(&relay1);
            Delay_Ms(50); // debounce
        }
        if(!b2 && btn2_prev) {
            Relay_Toggle(&relay2);
            Delay_Ms(50);
        }
        btn1_prev = b1; btn2_prev = b2;

        // Safety: auto-off after 30 seconds if both on
        static Timer_t tmr_safety;
        static uint8_t safety_started = 0;
        if(Relay_IsOn(&relay1) && Relay_IsOn(&relay2) && !safety_started) {
            Start_Timer(&tmr_safety, 30000, 0);
            safety_started = 1;
        }
        if(safety_started && Is_Timer_Expired(&tmr_safety)) {
            Relay_Off(&relay1);
            Relay_Off(&relay2);
            safety_started = 0;
        }
        if(!Relay_IsOn(&relay1) && !Relay_IsOn(&relay2)) safety_started = 0;
    }
}

🔍 Toggle relay ด้วยปุ่ม — debounce 50ms — Relay_IsOn(&relay) อ่านสถานะ — Safety timer: ถ้า relay ทั้งสอง ON พร้อมกันเกิน 30 วิ ให้ปิดอัตโนมัติ — ป้องกันอุปกรณ์เสียหายหรือไฟไหม้

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
Relay ไม่ ON — ไม่มีเสียง clickGPIO logic กลับด้าน — module เป็น Active HIGHลองเปลี่ยน RELAY_ACTIVE_LOW เป็น RELAY_ACTIVE_HIGH — หรือใช้ jumper บนบอร์ดเลือกโหมด
Relay ON ค้าง ปิดไม่ได้MCU GPIO ไม่พอจ่ายกระแส — transistor ไม่ fully offตรวจสอบ GPIO output voltage — ใช้ pull-down resistor 10kΩ ที่ IN pin
MCU รีเซ็ตเองเวลา relay เปลี่ยนสถานะFlyback spike จาก relay coil รบกวนไฟ MCUใช้ capacitor 100μF + 100nF ขนาน VCC-GND ใกล้ MCU — แยก supply relay กับ MCU
Relay ติด ๆ ดับ ๆ (chattering)GPIO noise — สัญญาณรบกวนจาก AC lineใช้ shielded cable — ต่อ snubber circuit (100Ω + 100nF series) ขนาน contact — แยกสายไฟ AC ห่างจาก MCU
ไฟดูด — บาดเจ็บจับขั้ว relay ขณะต่อ AC 220V⚡ ใส่กล่องพลาสติกปิดมิด — ใช้ screw terminal — ถอดปลั๊กก่อนแก้ไขทุกครั้ง — ห้ามทำคนเดียว

🛠️ สร้าง Product: Smart Home Timer Switch

ใช้ Relay + OLED + RTC DS1307 — ตั้งเวลาเปิด-ปิดไฟอัตโนมัติ — ปุ่มปรับเวลา — Non-Blocking timer

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "Relay.h"

Relay_Instance relay;
uint8_t schedule_h = 18, schedule_m = 30; // เปิด 18:30
uint8_t duration_min = 120;  // ปิดหลัง 120 นาที
uint8_t relay_on = 0;
uint32_t auto_off_time = 0;

int main(void) {
    SystemCoreClockUpdate(); Timer_Init();
    OLED_Init_I2C(0x3C, PC2, PC1);
    Relay_Init(&relay, PC0, RELAY_ACTIVE_LOW);
    Relay_Off(&relay);

    pinMode(PC3, PIN_MODE_INPUT_PULLUP); // BTN: manual toggle

    uint8_t btn_prev = 1;
    uint8_t hour = 12, min = 0;  // simulated time (use RTC in real)
    Timer_t tmr_clock;
    Start_Timer(&tmr_clock, 1000, 1);  // tick every "minute"

    while(1) {
        // Simulate clock (replace with DS1307_ReadTime in real)
        if(Is_Timer_Expired(&tmr_clock)) {
            min++; if(min >= 60) { min = 0; hour++; if(hour >= 24) hour = 0; }
        }

        // Auto ON at schedule time
        if(hour == schedule_h && min == schedule_m && !relay_on) {
            Relay_On(&relay);
            relay_on = 1;
            auto_off_time = Get_CurrentMs() + (uint32_t)duration_min * 60000;
        }

        // Auto OFF after duration
        if(relay_on && Get_CurrentMs() >= auto_off_time) {
            Relay_Off(&relay);
            relay_on = 0;
        }

        // Manual toggle button
        uint8_t btn = digitalRead(PC3);
        if(!btn && btn_prev) {
            Relay_Toggle(&relay);
            relay_on = Relay_IsOn(&relay);
            Delay_Ms(50);
        }
        btn_prev = btn;

        // Update OLED
        OLED_Clear();
        OLED_Printf(0,0,"Time: %02d:%02d", hour, min);
        OLED_Printf(0,2,"Relay: %s", relay_on ? "ON " : "OFF");
        OLED_Printf(0,4,"Sched: %02d:%02d +%dm", schedule_h, schedule_m, duration_min);
        OLED_Update();
        Delay_Ms(100);
    }
}
