YF-S201 — Water Flow Sensor นับพัลส์วัดอัตราการไหล

ระดับกลาง

🧭 แผนภาพการทำงาน

flowchart TD
    A["Init GPIO + Interrupt (PC3)"] --> B["น้ำไหล → Hall Sensor → พัลส์"]
    B --> C["ISR: pulse_count++"]
    C --> D["Timer 1 วิ: คำนวณ flow rate"]
    D --> E["Flow(L/min) = pulses / K-factor"]
    E --> F["Total Volume += Flow × Δt"]
    F --> G["แสดง Flow Rate + Total"]
    G --> D

YF-S201 ใช้ Hall Effect Sensor + Magnet ในกังหัน — เมื่อน้ำไหล กังหันหมุน → Hall Sensor สร้างพัลส์ — K-factor = 450 pulse/Liter — ต่อตรง GPIO ใช้ Interrupt นับ

📚 หลักการทำงาน

YF-S201 Water Flow Sensor ใช้กังหันที่มีแม่เหล็กติดอยู่ — เมื่อน้ำไหลผ่าน กังหันหมุน — Hall Effect Sensor ตรวจจับแม่เหล็ก → สร้างพัลส์สัญญาณ (LOW-HIGH-LOW) — ความถี่ของพัลส์แปรผันตามอัตราการไหล

K-Factor (Pulses/Liter): แต่ละรุ่นมีค่า K-factor ต่างกัน — YF-S201 = 450 pulse/L, YF-S401 = 5880 pulse/L — Flow Rate (L/min) = (pulse_count / K_factor) × 60 — หรือใช้ library ที่คำนวณให้

Interrupt-Based Counting: ใช้ EXTI Interrupt ที่ขา GPIO — ทุกครั้งที่มี pulse (falling edge) → ISR เพิ่มตัวนับ — Timer อ่านค่าตัวนับทุก 1 วินาที → คำนวณ flow rate + total volume — แม่นยำกว่า polling มาก

การคำนวณปริมาตรรวม: Total Volume (L) = Total Pulses / K-factor — หรือสะสมจาก flow rate ทุกวินาที: Volume += (Flow_L_min / 60) × 1sec

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Init GPIO + Interrupt

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"

#define FLOW_PIN PC3
volatile uint32_t pulse_count = 0;

void EXTI_IRQ(void) {
    if(EXTI_GetITStatus(EXTI_Line3)) {
        pulse_count++;
        EXTI_ClearITPendingBit(EXTI_Line3);
    }
}

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_Init(115200);

    pinMode(FLOW_PIN, PIN_MODE_INPUT_PULLUP);
    EXTI_Init(FLOW_PIN, EXTI_Trigger_Falling);
    NVIC_EnableIRQ(EXTI7_0_IRQn);

    USART_Print("Flow Sensor Ready\r\n");

🔍 ใช้ EXTI Interrupt แบบ Falling Edge — ทุกพัลส์จาก flow sensor → ISR เพิ่ม pulse_count — ใช้ Internal Pull-Up — volatile เพราะแก้ใน ISR — K-factor 450 pulse/L

ขั้นที่ 2: คำนวณ Flow Rate + Total Volume

    float K_FACTOR = 450.0f; // YF-S201
    float total_liters = 0;
    uint32_t last_pulse = 0;
    Timer_t tmr;
    Start_Timer(&tmr, 1000, 1);

    while(1) {
        if(Is_Timer_Expired(&tmr)) {
            uint32_t pulses_this_sec;
            NVIC_DisableIRQ(EXTI7_0_IRQn);
            pulses_this_sec = pulse_count - last_pulse;
            last_pulse = pulse_count;
            NVIC_EnableIRQ(EXTI7_0_IRQn);

            float flow_l_min = (pulses_this_sec / K_FACTOR) * 60.0f;
            total_liters += (float)pulses_this_sec / K_FACTOR;

            USART_Print("Flow: "); USART_PrintFloat(flow_l_min, 1);
            USART_Print(" L/min Total: ");
            USART_PrintFloat(total_liters, 1);
            USART_Print(" L\r\n");
        }
    }
}

🔍 ปิด interrupt ชั่วคราวเพื่ออ่านค่า atomic — คำนวณ flow rate จาก pulses ใน 1 วิ — Flow(L/min) = (pulses/K_factor) × 60 — Total = pulses สะสม / K_factor

ขั้นที่ 3: วาล์วควบคุม + แจ้งเตือน overflow

    pinMode(PC0, PIN_MODE_OUTPUT); // Solenoid Valve
    digitalWrite(PC0, LOW);

    float flow_limit = 10.0f; // L/min max
    float daily_limit = 100.0f; // L/day max

    while(1) {
        if(Is_Timer_Expired(&tmr)) {
            NVIC_DisableIRQ(EXTI7_0_IRQn);
            uint32_t pps = pulse_count - last_pulse;
            last_pulse = pulse_count;
            NVIC_EnableIRQ(EXTI7_0_IRQn);

            float flow = (pps / K_FACTOR) * 60.0f;
            total_liters += pps / K_FACTOR;

            USART_Print("Flow:"); USART_PrintFloat(flow, 1);
            USART_Print(" Total:"); USART_PrintFloat(total_liters, 1);

            if(flow > flow_limit) {
                USART_Print(" OVERFLOW!");
                digitalWrite(PC0, HIGH); // ปิดวาล์ว
            }
            if(total_liters > daily_limit) {
                USART_Print(" DAILY LIMIT!");
                digitalWrite(PC0, HIGH);
            }
            USART_Print("\r\n");
        }
    }
}

🔍 ตรวจจับ overflow เมื่อ flow rate เกิน 10 L/min — ปิด Solenoid Valve ทันที — จำกัดปริมาณน้ำต่อวัน 100 L — ป้องกันน้ำรั่ว/ท่อแตก

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
นับพัลส์ไม่ได้เลยไม่ใช้ Interrupt / Pull-up หายใช้ EXTI Falling Edge — ต่อ Pull-up 10kΩ ถ้าไม่มีในตัว — ใช้ Oscilloscope ดูสัญญาณ
Flow Rate อ่านผิดK-factor ไม่ตรงรุ่นเซนเซอร์YF-S201=450, YF-S401=5880, YF-B1=396 — เช็ค sticker — calibrate เอง
ค่ากระโดดไปมาน้ำไหลไม่สม่ำเสมอ / ฟองอากาศใช้ averaging 5 วินาที — ติดตั้งแนวนอน — ไล่ฟองอากาศในท่อ
Interrupt รบกวน USARTISR ใช้เวลานาน / priority conflictISR ให้เร็วที่สุด — แค่ pulse_count++ — งานหนักทำใน main loop
น้ำไหลแต่พัลส์น้อยกว่าที่ควรFlow rate ต่ำกว่า minimum (1 L/min)YF-S201 วัดต่ำสุด ~1 L/min — ใช้ YF-S401 สำหรับ flow ต่ำ (0.3 L/min)

🛠️ สร้าง Product: Smart Water Meter

YF-S201 + OLED + Solenoid Valve — วัดปริมาณน้ำ ควบคุมวาล์ว — แสดง Flow Rate + Total

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "OLED.h"

#define FLOW_PIN PC3
#define VALVE_PIN PC0
#define K_FACTOR 450.0f

volatile uint32_t pulse_cnt = 0;

void EXTI_IRQ(void) {
    if(EXTI_GetITStatus(EXTI_Line3)) {
        pulse_cnt++;
        EXTI_ClearITPendingBit(EXTI_Line3);
    }
}

int main(void)
{
    SystemCoreClockUpdate();
    Timer_Init();
    I2C_Init(PC1, PC2, I2C_SPEED_STANDARD);
    USART_Init(115200);

    pinMode(FLOW_PIN, PIN_MODE_INPUT_PULLUP);
    pinMode(VALVE_PIN, PIN_MODE_OUTPUT);
    EXTI_Init(FLOW_PIN, EXTI_Trigger_Falling);
    NVIC_EnableIRQ(EXTI7_0_IRQn);
    digitalWrite(VALVE_PIN, LOW);

    OLED_t oled;
    OLED_Init(&oled, 0x3C, OLED_128x64);
    OLED_SetFont(&oled, FONT_8x16);

    float total_L = 0;
    uint32_t last = 0;
    Timer_t tmr;
    Start_Timer(&tmr, 1000, 1);

    while(1)
    {
        if(Is_Timer_Expired(&tmr))
        {
            NVIC_DisableIRQ(EXTI7_0_IRQn);
            uint32_t pps = pulse_cnt - last;
            last = pulse_cnt;
            NVIC_EnableIRQ(EXTI7_0_IRQn);

            float flow_lpm = (pps / K_FACTOR) * 60.0f;
            total_L += pps / K_FACTOR;

            OLED_Clear(&oled);
            OLED_SetCursor(&oled, 0, 0);
            OLED_Print(&oled, "Flow Rate:");
            OLED_SetCursor(&oled, 0, 2);
            OLED_PrintFloat(&oled, "", flow_lpm, 1);
            OLED_Print(&oled, " L/min");

            OLED_SetCursor(&oled, 0, 4);
            OLED_Print(&oled, "Total:");
            OLED_SetCursor(&oled, 0, 6);
            OLED_PrintFloat(&oled, "", total_L, 1);
            OLED_Print(&oled, " L");

            if(flow_lpm > 15.0f) {
                digitalWrite(VALVE_PIN, HIGH);
                OLED_SetCursor(&oled, 80, 0);
                OLED_Print(&oled, "MAX!");
            }

            OLED_Display(&oled);
        }
    }
}

✏️ ลองทำดู

  1. Calibrate K-factor — ปล่อยน้ำ 1 ลิตร — นับพัลส์ — ใช้ค่านั้นแทน 450 เพื่อความแม่นยำ
  2. เพิ่ม Water Temperature Sensor — วัดอุณหภูมิน้ำ — ปิดวาล์วถ้าร้อนเกิน 60°C
  3. ทำ Water Dispenser — กดปุ่ม → เปิดวาล์วจนครบปริมาณที่ตั้ง (เช่น 250ml) → ปิดอัตโนมัติ
  4. บันทึกการใช้น้ำรายวันลง EEPROM — คำนวณค่าใช้จ่าย — สร้างกราฟการใช้น้ำ
