YF-S201 — Water Flow Sensor นับพัลส์วัดอัตราการไหล

ระดับกลาง

🧭 แผนภาพการทำงาน

flowchart TD
    A["WaterFlow_Init(&w, PC0, 450.0f)"] --> B["น้ำไหล → Hall Sensor → พัลส์"]
    B --> C["ISR: pulse_count++ (auto)"]
    C --> D["WaterFlow_GetFlowRate(&w)"]
    D --> E["Flow(L/min) = pulses / K-factor"]
    E --> F["WaterFlow_GetTotalVolume(&w)"]
    F --> G["แสดง Flow Rate + Total"]
    G --> D

YF-S201 ใช้ Hall Effect Sensor + Magnet ในกังหัน — เมื่อน้ำไหล กังหันหมุน → Hall Sensor สร้างพัลส์ — K-factor = 450 pulse/Liter — ต่อตรง GPIO ใช้ Interrupt นับ

📚 หลักการทำงาน

YF-S201 Water Flow Sensor ใช้กังหันที่มีแม่เหล็กติดอยู่ — เมื่อน้ำไหลผ่าน กังหันหมุน — Hall Effect Sensor ตรวจจับแม่เหล็ก → สร้างพัลส์สัญญาณ — ความถี่ของพัลส์แปรผันตามอัตราการไหล

K-Factor (Pulses/Liter): แต่ละรุ่นมีค่า K-factor ต่างกัน — YF-S201 = 450 pulse/L, YF-S401 = 5880 pulse/L — Flow Rate (L/min) = (pulse_count / K_factor) × 60

Interrupt-Based Counting: Library จัดการ EXTI Interrupt ให้อัตโนมัติ — ไม่ต้องเขียน ISR เอง — WaterFlow_GetFlowRate() และ WaterFlow_GetTotalVolume() อ่านค่าได้ทันที

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Init WaterFlow

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "WaterFlow.h"

WaterFlow_Instance w;

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_Init(115200);

    WaterFlow_Init(&w, PC0, 450.0f); // pin, K-factor
    USART_Println("Water Flow Sensor Ready!");

🔍 WaterFlow_Init(&w, pin, kfactor) — ตั้งค่า GPIO + EXTI Interrupt อัตโนมัติ — K-factor 450 สำหรับ YF-S201 — library จัดการ ISR ภายใน

ขั้นที่ 2: อ่าน Flow Rate + Total Volume

    float lpm = WaterFlow_GetFlowRate(&w);   // L/min
    float total = WaterFlow_GetTotalVolume(&w); // L

    USART_Print("Flow: ");
    USART_PrintFloat(lpm, 1);
    USART_Print(" L/min  |  Total: ");
    USART_PrintFloat(total, 1);
    USART_Println(" L");

🔍 WaterFlow_GetFlowRate(&w) — อัตราการไหล L/min — WaterFlow_GetTotalVolume(&w) — ปริมาตรรวม L — library คำนวณให้อัตโนมัติ

ขั้นที่ 3: วาล์วควบคุม + แจ้งเตือน overflow

    pinMode(PC1, PIN_MODE_OUTPUT); // Solenoid Valve
    digitalWrite(PC1, LOW);

    float flow_limit = 10.0f; // L/min max
    float daily_limit = 100.0f; // L/day max
    Timer_t tmr;
    Start_Timer(&tmr, 1000, 1);

    while(1) {
        if(Is_Timer_Expired(&tmr)) {
            float lpm = WaterFlow_GetFlowRate(&w);
            float total = WaterFlow_GetTotalVolume(&w);

            USART_Print("Flow:"); USART_PrintFloat(lpm, 1);
            USART_Print(" Total:"); USART_PrintFloat(total, 1);

            if(lpm > flow_limit) {
                USART_Print(" OVERFLOW!");
                digitalWrite(PC1, HIGH); // ปิดวาล์ว
            }
            if(total > daily_limit) {
                USART_Print(" DAILY LIMIT!");
                digitalWrite(PC1, HIGH);
            }
            USART_Println("");
        }
    }
}

🔍 ตรวจจับ overflow เมื่อ flow rate เกิน 10 L/min — ปิด Solenoid Valve ทันที — จำกัดปริมาณน้ำต่อวัน 100 L — ป้องกันน้ำรั่ว/ท่อแตก

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
นับพัลส์ไม่ได้เลยสายสัญญาณหลุด / Pull-up หายใช้ Pull-up 10kΩ ถ้าไม่มีในตัว — ใช้ Oscilloscope ดูสัญญาณ
Flow Rate อ่านผิดK-factor ไม่ตรงรุ่นเซนเซอร์YF-S201=450, YF-S401=5880, YF-B1=396 — เช็ค sticker
ค่ากระโดดไปมาน้ำไหลไม่สม่ำเสมอ / ฟองอากาศใช้ averaging — ติดตั้งแนวนอน — ไล่ฟองอากาศในท่อ
น้ำไหลแต่พัลส์น้อยกว่าที่ควรFlow rate ต่ำกว่า minimum (1 L/min)YF-S201 วัดต่ำสุด ~1 L/min — ใช้ YF-S401 สำหรับ flow ต่ำ

🛠️ สร้าง Product: Smart Water Meter

YF-S201 + OLED + Solenoid Valve — วัดปริมาณน้ำ ควบคุมวาล์ว

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "WaterFlow.h"
#include "oled_i2c.h"

#define VALVE_PIN PC1

int main(void)
{
    SystemCoreClockUpdate();
    Timer_Init();
    I2C_Init(PC1, PC2, I2C_SPEED_STANDARD);
    USART_Init(115200);

    WaterFlow_Instance w;
    WaterFlow_Init(&w, PC0, 450.0f);

    pinMode(VALVE_PIN, PIN_MODE_OUTPUT);
    digitalWrite(VALVE_PIN, LOW);

    OLED_Handle oled;
    OLED_Init(&oled, OLED_128x64, 0x3C);
    OLED_SetFont(&oled, FONT_8x16);

    Timer_t tmr;
    Start_Timer(&tmr, 1000, 1);

    while(1)
    {
        if(Is_Timer_Expired(&tmr))
        {
            float lpm = WaterFlow_GetFlowRate(&w);
            float total = WaterFlow_GetTotalVolume(&w);

            OLED_Clear(&oled);
            OLED_SetCursor(&oled, 0, 0);
            OLED_Print(&oled, "Flow Rate:");
            OLED_SetCursor(&oled, 0, 2);
            OLED_PrintFloat(&oled, "", lpm, 1);
            OLED_Print(&oled, " L/min");

            OLED_SetCursor(&oled, 0, 4);
            OLED_Print(&oled, "Total:");
            OLED_SetCursor(&oled, 0, 6);
            OLED_PrintFloat(&oled, "", total, 1);
            OLED_Print(&oled, " L");

            if(lpm > 15.0f) {
                digitalWrite(VALVE_PIN, HIGH);
                OLED_SetCursor(&oled, 80, 0);
                OLED_Print(&oled, "MAX!");
            }

            OLED_Update(&oled);
        }
    }
}

✏️ ลองทำดู

  1. ทำ Water Dispenser — กดปุ่ม → เปิดวาล์วจนครบปริมาณที่ตั้ง (เช่น 250ml) → ปิดอัตโนมัติ
  2. เพิ่ม Water Temperature Sensor — วัดอุณหภูมิน้ำ — ปิดวาล์วถ้าร้อนเกิน 60°C
