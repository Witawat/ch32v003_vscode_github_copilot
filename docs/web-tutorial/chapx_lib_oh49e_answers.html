📝 เฉลย — OH49E

ข้อ 1: Magnet Polarity Detection — ตรวจจับขั้วแม่เหล็ก North/South

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"

#define HALL_PIN PD2
#define HALL_CH  ADC_CH0
#define SENSITIVITY 1.4f

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    ADC_Init(HALL_PIN, HALL_CH);
    USART_Init(115200);

    float zero_v = HALL_Calibrate(HALL_PIN, 100);
    pinMode(PC0, PIN_MODE_OUTPUT);
    pinMode(PC1, PIN_MODE_OUTPUT);
    pinMode(PC2, PIN_MODE_OUTPUT);

    USART_Print("Magnet Polarity Detector\r\n");
    USART_Print("Zero: "); USART_PrintFloat(zero_v, 3); USART_Print("V\r\n");

    while(1) {
        uint16_t adc = ADC_Read(HALL_PIN);
        float voltage = adc * (3.3f / 1024.0f);
        float gauss = ((voltage - zero_v) * 1000.0f) / SENSITIVITY;

        digitalWrite(PC0, LOW);
        digitalWrite(PC1, LOW);
        digitalWrite(PC2, LOW);

        USART_Print("ADC:"); USART_PrintDec(adc);
        USART_Print(" V:"); USART_PrintFloat(voltage, 3);
        USART_Print(" G:"); USART_PrintFloat(gauss, 1);

        if(gauss > 30) {
            USART_Print(" << SOUTH POLE >>");
            digitalWrite(PC0, HIGH);
        } else if(gauss > 10) {
            USART_Print(" South (weak)");
            digitalWrite(PC0, HIGH);
        } else if(gauss < -30) {
            USART_Print(" << NORTH POLE >>");
            digitalWrite(PC1, HIGH);
        } else if(gauss < -10) {
            USART_Print(" North (weak)");
            digitalWrite(PC1, HIGH);
        } else {
            USART_Print(" ~ No Magnet");
            digitalWrite(PC2, HIGH);
        }

        USART_Print("\r\n");
        Delay_Ms(100);
    }
}
🧠 แนวคิด: Calibrate zero ด้วย HALL_Calibrate() (100 samples) — แยก 5 ระดับ: South strong (>30G), South weak (>10G), Neutral (0±10G), North weak (<-10G), North strong (<-30G) — LED 3 ดวงแสดงสถานะ: PC0=South, PC1=North, PC2=No Magnet — ใช้ขั้วแม่เหล็กต่างกันเพื่อตรวจจับทิศทาง — OH49E ด้านมีตัวอักษร → South pole = output สูงกว่า zero

ข้อ 2: Position Sensor — วัดระยะทางจากแม่เหล็ก

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"

#define HALL_PIN PD2
#define HALL_CH  ADC_CH0
#define SENSITIVITY 1.4f

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    ADC_Init(HALL_PIN, HALL_CH);
    USART_Init(115200);

    USART_Print("Calibrating...\r\n");
    float zero_v = HALL_Calibrate(HALL_PIN, 100);
    USART_Print("Zero: "); USART_PrintFloat(zero_v, 3); USART_Print("V\r\n");

    USART_Print("Place magnet at 5mm distance...\r\n");
    USART_Print("Press any key when ready\r\n");
    while(!USART_Available());
    while(USART_Available()) USART_Read();

    float sum = 0;
    for(int i = 0; i < 50; i++) {
        uint16_t adc = ADC_Read(HALL_PIN);
        float v = adc * (3.3f / 1024.0f);
        sum += ((v - zero_v) * 1000.0f) / SENSITIVITY;
        Delay_Ms(10);
    }
    float gauss_5mm = sum / 50.0f;
    float scale = 5.0f / gauss_5mm;

    USART_Print("Gauss at 5mm: ");
    USART_PrintFloat(gauss_5mm, 1);
    USART_Print(" Scale: ");
    USART_PrintFloat(scale, 3);
    USART_Print("\r\nReady\r\n");

    while(1) {
        uint16_t adc = ADC_Read(HALL_PIN);
        float voltage = adc * (3.3f / 1024.0f);
        float gauss = ((voltage - zero_v) * 1000.0f) / SENSITIVITY;
        float distance_mm = gauss * scale;

        if(fabs(gauss) < 3) distance_mm = 999;

        USART_Print("G:"); USART_PrintFloat(gauss, 1);
        USART_Print(" Dist:"); USART_PrintFloat(distance_mm, 1);
        USART_Print("mm");

        if(distance_mm < 5)  USART_Print(" CLOSE!");
        if(distance_mm > 50) USART_Print(" FAR");

        USART_Print("\r\n");
        Delay_Ms(200);
    }
}
🧠 แนวคิด: Calibrate 2 จุด: zero (ไม่มีแม่เหล็ก) และที่ 5mm — สนามแม่เหล็กลดลงตามระยะทาง (inverse square law โดยประมาณ) — scale factor = 5mm / gauss_at_5mm — ระยะทาง ≈ gauss × scale — แม่เหล็ก Neodymium ให้สัญญาณแรงกว่า — ใช้สำหรับ linear position, door sensor, หรือ piston stroke measurement

ข้อ 3: Field Strength Measurement — Gauss Meter

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "OLED.h"

#define HALL_PIN PD2
#define HALL_CH  ADC_CH0
#define SENSITIVITY 1.4f

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    I2C_Init(PC1, PC2, I2C_SPEED_STANDARD);
    ADC_Init(HALL_PIN, HALL_CH);
    USART_Init(115200);

    OLED_t oled;
    OLED_Init(&oled, 0x3C, OLED_128x64);
    OLED_SetFont(&oled, FONT_8x16);

    float zero_v = HALL_Calibrate(HALL_PIN, 100);

    float peak_north = 0, peak_south = 0;

    while(1) {
        uint32_t adc_sum = 0;
        for(int i = 0; i < 8; i++) {
            adc_sum += ADC_Read(HALL_PIN);
            Delay_Ms(5);
        }
        uint16_t adc = adc_sum / 8;
        float voltage = adc * (3.3f / 1024.0f);
        float gauss = ((voltage - zero_v) * 1000.0f) / SENSITIVITY;

        if(gauss > peak_south) peak_south = gauss;
        if(-gauss > peak_north) peak_north = -gauss;
        float abs_gauss = fabs(gauss);

        OLED_Clear(&oled);

        OLED_SetCursor(&oled, 0, 0);
        OLED_Print(&oled, "=== GAUSS METER ===");

        OLED_SetCursor(&oled, 0, 1);
        OLED_PrintFloat(&oled, "Live: ", gauss, 1);
        OLED_Print(&oled, " G");

        OLED_SetCursor(&oled, 60, 1);
        if(gauss > 5)      OLED_Print(&oled, "(S)");
        else if(gauss < -5) OLED_Print(&oled, "(N)");
        else                OLED_Print(&oled, "--");

        int bar = (int)(abs_gauss * 0.5f);
        if(bar > 64) bar = 64;
        OLED_SetCursor(&oled, 0, 3);
        OLED_Print(&oled, "[");
        for(int i = 0; i < bar; i++) {
            OLED_Print(&oled, abs_gauss > 50 ? "#" : "=");
        }
        OLED_Print(&oled, "]");

        OLED_SetCursor(&oled, 0, 5);
        OLED_PrintFloat(&oled, "Peak S:", peak_south, 0);
        OLED_Print(&oled, " ");
        OLED_PrintFloat(&oled, "N:", peak_north, 0);
        OLED_Print(&oled, " G");

        OLED_SetCursor(&oled, 0, 7);
        OLED_PrintFloat(&oled, "V:", voltage, 3);
        OLED_Print(&oled, " ADC:");
        OLED_PrintDec(&oled, adc);

        OLED_Display(&oled);
        Delay_Ms(50);
    }
}
🧠 แนวคิด: Gauss Meter บน OLED — averaging 8 samples (40ms) ลด noise — Bar Graph แสดงความแรงสนามแม่เหล็ก (สูงสุด 64 ตัวอักษร) — Peak Hold แยก South/North — สัญลักษณ์ (S)/(N) บอกขั้ว — เปลี่ยน "#" เมื่อ >50G (สนามแรงมาก) — ใช้ calibrate แม่เหล็ก, ตรวจสอบ magnetizer/demagnetizer, หรือวัดสนามรบกวน
