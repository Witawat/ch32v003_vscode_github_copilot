📝 เฉลย — บทที่ 6: ADC

ข้อ 1: อ่าน potentiometer + แสดงผลเป็น mV ด้วย analogReadMilliVolts

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include "SimpleHAL.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
    ADC_SimpleInit();

    while (1) {
        // Arduino-style: อ่านค่า ADC เป็น mV โดยตรง
        float vdd = ADC_GetVDD();
        uint16_t raw = analogRead(PD2);
        uint16_t mv = (uint32_t)raw * (uint32_t)(vdd * 1000) / 1023;

        USART_Print("ADC: ");
        USART_PrintNum(raw);
        USART_Print(" (");
        USART_PrintNum(mv);
        USART_Print("mV) VDD=");
        USART_PrintNum((uint16_t)(vdd * 1000));
        USART_Print("mV\r\n");

        Delay_Ms(500);
    }
}
🧠 แนวคิด: analogRead(PD2) อ่านค่า ADC (0-1023) — ADC_GetVDD() ใช้ internal 1.2V reference คำนวณ VDD จริง — แปลงเป็น mV ด้วย raw * (vdd*1000) / 1023 แบบ integer (เร็วกว่า float) — VDD จริงอาจเป็น 3.2V หรือ 3.4V ไม่ใช่ 3.3V เสมอ — การใช้ VDD จริงทำให้ค่า mV แม่นยำกว่า

ข้อ 2: วัดอุณหภูมิ NTC + แสดง °C

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include <math.h>

#define NTC_RFIX  10000.0f   // R คงที่ 10kΩ
#define NTC_R25   10000.0f   // R NTC ที่ 25°C
#define NTC_B     3950.0f    // B-constant

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
    ADC_SimpleInit();

    while (1) {
        float vdd = ADC_GetVDD();

        uint16_t adc = ADC_ReadAverage(ADC_CH_PD3, 10);
        float v_ntc = ADC_ToVoltage(adc, vdd);

        // คำนวณ R NTC จาก voltage divider
        float r_ntc = NTC_RFIX * v_ntc / (vdd - v_ntc);

        // Steinhart-Hart (B-parameter equation)
        float temp_k = 1.0f / (1.0f/298.15f + logf(r_ntc/NTC_R25)/NTC_B);
        float temp_c = temp_k - 273.15f;

        USART_Print("NTC: ");
        USART_PrintNum(adc);
        USART_Print(" -> ");
        USART_PrintNum((int32_t)(temp_c * 10));
        USART_Print(" (x0.1C)\r\n");

        Delay_Ms(1000);
    }
}
🧠 แนวคิด: NTC = ตัวต้านทานที่ค่าเปลี่ยนตามอุณหภูมิ — วงจร voltage divider: 3.3V → R_fix → ADC_pin → NTC → GND — อุณหภูมิสูง → R_NTC ลด → แรงดันที่ ADC ลด — สูตร Steinhart-Hart ใช้ B-constant 3950 (ค่าทั่วไป) — ADC_ReadAverage(n, 10) ลด noise — ใช้ logf() จาก math.h

ข้อ 3: ใช้ ADC_ReadAverage ลด noise 16 samples

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include "SimpleHAL.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
    ADC_SimpleInit();

    while (1) {
        // 16 samples — noise ต่ำกว่าแต่ช้าลง (16x)
        uint16_t avg = ADC_ReadAverage(ADC_CH_PD2, 16);
        uint16_t mv = (uint32_t)avg * 3300 / 1023;

        // เทียบกับค่าเดียว (อ่าน 1 ครั้ง)
        uint16_t single = ADC_Read(ADC_CH_PD2);
        uint16_t mv_single = (uint32_t)single * 3300 / 1023;

        USART_Print("Single: ");
        USART_PrintNum(mv_single);
        USART_Print("mV | Avg16: ");
        USART_PrintNum(mv);
        USART_Print("mV\r\n");

        Delay_Ms(500);
    }
}
🧠 แนวคิด: ADC_ReadAverage(ch, 16) อ่าน 16 ครั้งแล้วหาค่าเฉลี่ย — ยิ่ง samples มาก noise ยิ่งต่ำ (standard deviation ลดลง √N เท่า) — แต่ใช้เวลานานขึ้น (16 ครั้ง × ~3.5μs/ครั้ง ≈ 56μs) — แสดงทั้งค่าเดียวและค่าเฉลี่ยเทียบกัน — ค่าเฉลี่ยจะนิ่งกว่าเห็นชัด
