📝 เฉลย — 74HC595 Shift Register

ข้อ 1: Binary Counter 8-bit พร้อม LED

#include "ShiftReg595.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    ShiftReg_Init(SHIFTREG_PINS_DEFAULT);

    uint8_t counter = 0;

    while(1) {
        ShiftReg_Write(counter);
        ShiftReg_Update();
        counter++;
        Delay_Ms(300);

        if(counter == 0) {  // overflow
            // กระพริบทุก LED 3 ครั้ง
            for(uint8_t i = 0; i < 3; i++) {
                ShiftReg_Write(0xFF);
                ShiftReg_Update();
                Delay_Ms(200);
                ShiftReg_Write(0x00);
                ShiftReg_Update();
                Delay_Ms(200);
            }
        }
    }
}
🧠 แนวคิด: 74HC595 เป็น shift register 8-bit รับข้อมูลผ่าน SPI (SER, SRCLK, RCLK) — ShiftReg_Write(data) shift 8 bits เข้า register — ShiftReg_Update latch ข้อมูลไป output ด้วย RCLK pulse — counter เพิ่มทีละ 1 แสดงผลเป็น binary บน LED 8 ดวง — เมื่อ overflow (กลับ 0) กระพริบทุก LED 3 ครั้ง — ถ้าต้องการนับถอยหลัง ให้ใช้ counter-- แทน — 74HC595 ขับกระแสได้ ~6mA ต่อขา รวม ~70mA ต่อ IC

ข้อ 2: Knight Rider (ไฟวิ่งกลับไปกลับมา)

#include "ShiftReg595.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    ShiftReg_Init(SHIFTREG_PINS_DEFAULT);

    while(1) {
        // วิ่งไปทางขวา
        for(uint8_t i = 0; i < 7; i++) {
            ShiftReg_Write(1 << i);
            ShiftReg_Update();
            Delay_Ms(80);
        }

        // วิ่งกลับทางซ้าย
        for(int8_t i = 7; i >= 0; i--) {
            ShiftReg_Write(1 << i);
            ShiftReg_Update();
            Delay_Ms(80);
        }
    }
}
🧠 แนวคิด: Knight Rider ใช้ bit shift 1< เพื่อให้ LED ติดทีละดวง — วน i 0→7 วิ่งขวา, แล้ววน 7→0 วิ่งซ้าย — delay 80ms ต่อ step ให้เห็นชัด — ShiftReg_Write รับค่า 8-bit ส่งตรงกับ Q0-Q7 — สำหรับเอฟเฟกต์สมจริงขึ้น เพิ่มหาง LED 2-3 ดวงด้วย PWM (ต้องใช้ PWM-capable shift register หรือ bit-bang) — ถ้าต้องการ Knight Rider แบบ 16 LED ให้ chain 74HC595 2 ตัว

ข้อ 3: Chain 2 Chips — ขับ 16 Outputs

#include "ShiftReg595.h"

// เขียนข้อมูล 16 บิต ผ่าน 74HC595 2 ตัวแบบ daisy-chain
void ShiftReg_Write16(uint16_t data) {
    for(int16_t i = 15; i >= 0; i--) {
        digitalWrite(SRCLK_PIN, LOW);
        digitalWrite(SER_PIN, (data >> i) & 1);
        digitalWrite(SRCLK_PIN, HIGH);
    }
    // Latch ทั้ง 2 ตัวพร้อมกัน
    digitalWrite(RCLK_PIN, LOW);
    digitalWrite(RCLK_PIN, HIGH);
}

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();

    // Setup pins
    pinMode(SER_PIN, PIN_MODE_OUTPUT);
    pinMode(SRCLK_PIN, PIN_MODE_OUTPUT);
    pinMode(RCLK_PIN, PIN_MODE_OUTPUT);

    uint8_t pattern = 0x01;
    uint8_t dir = 0;  // 0=shift left, 1=shift right

    while(1) {
        // สร้าง 16-bit pattern: left chip + right chip
        uint16_t out;
        if(dir == 0) {
            out = ((uint16_t)pattern << 8) | pattern;
            pattern <<= 1;
            if(pattern == 0) { pattern = 0x80; dir = 1; }
        } else {
            out = ((uint16_t)pattern << 8) | pattern;
            pattern >>= 1;
            if(pattern == 0) { pattern = 0x01; dir = 0; }
        }

        ShiftReg_Write16(out);
        Delay_Ms(100);
    }
}
🧠 แนวคิด: ต่อ 74HC595 2 ตัวแบบ daisy-chain — Q7' ของตัวแรกต่อเข้า SER ของตัวที่สอง — ข้อมูล 16 บิต shift เข้า MSB ก่อน — ตัวแรกจะได้ high byte ตัวที่สองได้ low byte — Latch ทั้งสองตัวพร้อมกันด้วย RCLK pulse เดียว — pattern เลื่อนซ้าย-ขวาแสดงผลซ้ำกันทั้ง 2 chips — chain ได้ไม่จำกัดจำนวน เพียงเพิ่มจำนวนบิตและเวลา shift — สำหรับ SPI hardware ให้ใช้ 16-bit transfer แทน bit-bang เพื่อความเร็ว
