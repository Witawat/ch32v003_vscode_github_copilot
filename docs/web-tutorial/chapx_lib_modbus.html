Modbus RTU Master — DMA + USART สองโหมดขนส่ง

ระดับสูง

🔌 การต่อวงจร

   ┌────────────────────────────────────────────┐
   │  MCU (Master)        Slave (Modbus RTU)    │
   │  PD5─TX ───────────── RX                   │
   │  PD6─RX ───────────── TX                   │
   │  GND ──────────────── GND                  │
   │                                            │
   │  RS-485 (ระยะไกล):                        │
   │  PD5─DI  MAX485 ── A ───── A (485+)       │
   │  PD6─RO  MAX485 ── B ───── B (485-)       │
   │  PC0─DE  (DE+RE รวมกัน)                   │
   └────────────────────────────────────────────┘
MCU Pin→อุปกรณ์ขาหมายเหตุ
PD5 (TX)→SlaveRXUSART1 ส่งคำขอ Modbus (9600 8N1)
PD6 (RX)←SlaveTXUSART1 รับ response
GND→SlaveGNDGround ร่วม (จำเป็นมาก)
PC0 (ทางเลือก)→MAX485DE+RERS-485 — 1=ส่ง, 0=รับ

⚠️ Modbus RTU — 9600 8N1 — ระยะห่างเฟรม ≥ 3.5 char time (~4ms) — ใช้ได้ทั้งต่อตรงและผ่าน MAX485

รู้จักขา

  • TX (PD5): ส่งคำขอ [SlaveAddr][FC][Data][CRC]
  • RX (PD6): รับ response จาก slave
  • DE/RE (PC0): ควบคุม MAX485 — ส่ง = HIGH, รับ = LOW (เฉพาะบัส RS-485)
  • A/B: สาย 485+ / 485- — จบสายด้วย 120Ω ที่ปลายทั้งสอง

วิธีต่อทีละขั้น

  1. ต่อ PD5 (TX) ไป RX ของ slave และ PD6 (RX) ไป TX
  2. ต่อ GND ร่วม — ขาดไม่ได้ ถ้าต่อสาย RS-485 ต้องใช้สาย twisted pair
  3. เช็ค slave address และ baud rate ของ slave (default 9600 8N1)
  4. ถ้าใช้ MAX485: ต่อ DE กับ RE รวมกันที่ GPIO แล้วควบคุมก่อน/หลังส่ง

📚 หลักการทำงาน

Modbus RTU เป็นโปรโตคอลอุตสาหกรรมแบบ request/response — Master ส่งคำขอ → Slave ตอบกลับ — แต่ละเฟรมมี CRC-16 กันข้อมูลเสีย — รองรับ 8 function codes ในไลบรารีนี้

เฟรม Request: [SlaveAddr][FC][Data...][CRC_LO][CRC_HI] — Response: เหมือนกัน + ถ้า error จะตอบ FC|0x80 + รหัสเหตุผล (exception)

FCความหมายฟังก์ชันในไลบรารี
0x01Read CoilsMODBUS_ReadCoils(&mb, coil, count, buf)
0x02Read Discrete InputsMODBUS_ReadDiscreteInputs(&mb, input, count, buf)
0x03Read Holding RegistersMODBUS_ReadHoldingRegisters(&mb, reg, count, buf)
0x04Read Input RegistersMODBUS_ReadInputRegisters(&mb, reg, count, buf)
0x05Write Single CoilMODBUS_WriteSingleCoil(&mb, coil, value)
0x06Write Single RegisterMODBUS_WriteSingleRegister(&mb, reg, value)
0x0FWrite Multiple CoilsMODBUS_WriteMultipleCoils(&mb, coil, count, buf)
0x10Write Multiple RegistersMODBUS_WriteMultipleRegisters(&mb, reg, count, buf)

2 โหมดขนส่ง: MODBUS_TRANSPORT_USART — รับข้อมูลผ่าน interrupt ring buffer ของ SimpleUSART (ง่ายสุด) — MODBUS_TRANSPORT_DMA — DMA RX circular (DMA_CH3) + จับจบเฟรมด้วย USART IDLE interrupt + ส่งด้วย DMA TX (DMA_CH2) — CPU ไม่ถูก interrupt ทุก byte

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Init (โหมด USART)

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "Modbus.h"

Modbus mb;
uint16_t regs[10];

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();

    MODBUS_Init(&mb, 1, MODBUS_TRANSPORT_USART, USART_PINS_DEFAULT);
    // slave address = 1, transport = USART ring buffer
    USART_Println("Modbus Ready!");

🔍 MODBUS_Init(&mb, addr, transport, pins) — เปิด USART 9600 8N1 + ตั้งค่าโปรโตคอล — โหมด USART ใช้ interrupt RXNE ของ SimpleUSART เดิม

ขั้นที่ 2: อ่าน Holding Registers + จัดการ Error

    Timer_t tmr;
    Start_Timer(&tmr, 1000, 1);

    while(1) {
        if(Is_Timer_Expired(&tmr)) {
            Modbus_Status st = MODBUS_ReadHoldingRegisters(&mb, 0x0000, 10, regs);

            if(st == MODBUS_OK) {
                USART_Print("reg0=");
                USART_PrintNum(regs[0]);
                USART_Print(" reg1=");
                USART_PrintNum(regs[1]);
                USART_Println("");
            } else {
                USART_Print("Error: ");
                USART_Print(MODBUS_StatusStr(st));
                if(st == MODBUS_ERROR_EXCEPT) {
                    USART_Print(" ex=0x");
                    USART_PrintHex(MODBUS_GetLastException(&mb), 1);
                }
                USART_Println("");
            }
        }
    }
}

🔍 อ่าน 10 registers (0x0000-0x0009) ทุก 1 วินาที — ค่า 16-bit ถูกแปลง Big-endian → host order ให้แล้ว — MODBUS_StatusStr() แปลง error เป็นข้อความ — exception บอกสาเหตุจาก slave (เช่น 0x02 = register เกินช่วง)

ขั้นที่ 3: เขียน Register / Coil

    // เขียนค่า 1000 ไป register 0x0000
    MODBUS_WriteSingleRegister(&mb, 0x0000, 1000);

    // เปิด coil 3 (สั่งงาน relay ฝั่ง slave)
    MODBUS_WriteSingleCoil(&mb, 3, 1);

    // เขียน 4 registers พร้อมกัน
    uint16_t batch[4] = {10, 20, 30, 40};
    MODBUS_WriteMultipleRegisters(&mb, 0x0010, 4, batch);

🔍 ทุกคำสั่งเขียน return Modbus_Status — ตรวจ == MODBUS_OK ก่อนเชื่อผล — MODBUS_GetLastError(&mb) อ่าน error ของคำขอล่าสุดได้เช่นกัน

ขั้นที่ 4: เปลี่ยนเป็นโหมด DMA — แค่บรรทัดเดียว

    // เดิม: MODBUS_TRANSPORT_USART → เปลี่ยนเป็น DMA
    MODBUS_Init(&mb, 1, MODBUS_TRANSPORT_DMA, USART_PINS_DEFAULT);
    // DMA CH3: RX circular buffer (256B) + IDLE interrupt จับจบเฟรม
    // DMA CH2: TX — ส่งคำขอแบบ DMA
    // CPU แทบไม่ถูก interrupt ระหว่างรอ response

🔍 โหมด DMA ใช้ USART_IdleHook() (weak hook ใน SimpleUSART) — เมื่อสายเงียบเกิน 1 char time จะคัดลอกเฟรมออกจาก DMA buffer ใน ISR — API ภายนอกเหมือนโหมด USART ทุกอย่าง — ⚠️ DMA_CH2/CH3 ถูกยึด ห้ามใช้กับ DMA อื่น

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
Timeout ตลอดสาย TX/RX กลับด้าน หรือ baud ไม่ตรงเช็ค TX→RX, RX→TX — 9600 8N1 — ต่อ GND ร่วม
Timeout บางครั้ง / CRC Errorสัญญาณรบกวน สายยาวใช้ MAX485 + twisted pair + 120Ω จบสาย
Exception 0x02register เกินช่วงของ slaveเปิดคู่มือ slave ดู register map — อ่านครั้งละ ≤ 125 regs
โหมด DMA ไม่งานDMA_CH2/3 ถูกใช้ หรือ RAM ไม่พอปลด DMA อื่น — ใช้โหมด USART แทนได้ทันที
IRQ ขัดแย้งมี lib อื่น override USART_IdleHookoverride ได้ตัวเดียว — ปัจจุบันมีแค่ Modbus
เฟรมเสียหาย / CRC Error ขณะโหมด DMAพิมพ์ debug (USART_Print) ระหว่างรอ responseห้าม Print ระหว่างคำขอ — พิมพ์หลังคำขอเสร็จ หรือใช้ SoftUART

🛠️ สร้าง Product: อ่าน VFD แสดงผล OLED

Modbus Master อ่านความถี่ + กระแสจาก VFD (slave addr 1) แล้วแสดงบน OLED — โหมด DMA

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "Modbus.h"
#include "oled_i2c.h"

Modbus mb;
uint16_t regs[8];

int main(void)
{
    SystemCoreClockUpdate();
    Timer_Init();
    I2C_Init(PC1, PC2, I2C_SPEED_STANDARD);

    MODBUS_Init(&mb, 1, MODBUS_TRANSPORT_DMA, USART_PINS_DEFAULT);

    OLED_Handle oled;
    OLED_Init(&oled, OLED_128x64, 0x3C);
    OLED_SetFont(&oled, FONT_8x16);

    Timer_t tmr;
    Start_Timer(&tmr, 500, 1);

    while(1)
    {
        if(Is_Timer_Expired(&tmr))
        {
            char buf[32];
            if(MODBUS_ReadHoldingRegisters(&mb, 0x2000, 4, regs) == MODBUS_OK) {
                snprintf(buf, sizeof(buf), "Freq: %d Hz", regs[0] / 100);
                OLED_Clear(&oled);
                OLED_DrawStringAlign(&oled, 0, 0, buf, ALIGN_LEFT);

                snprintf(buf, sizeof(buf), "Cur:  %d.%02d A", regs[2] / 100, regs[2] % 100);
                OLED_DrawStringAlign(&oled, 0, 20, buf, ALIGN_LEFT);

                OLED_Update(&oled);
            } else {
                OLED_Clear(&oled);
                OLED_DrawStringAlign(&oled, 0, 0, "COMM ERROR", ALIGN_LEFT);
                OLED_Update(&oled);
            }
        }
    }
}

✏️ ลองทำดู

  2. เขียนโปรแกรมอ่าน Input Registers จาก PZEM-004T v3 (addr 1) ด้วยไลบรารีนี้ แทน PZEM004Tv3 lib
