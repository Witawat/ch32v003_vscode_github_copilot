DS18B20 — เซนเซอร์อุณหภูมิ 1-Wire กันน้ำ

1-Wire ต่อได้หลายร้อยตัวบนบัสเส้นเดียว — แต่ละตัวมี unique 64-bit ROM ID

📚 หลักการทำงาน

1-Wire Protocol: สื่อสารผ่านสายเส้นเดียว (DQ) — ใช้ timing แบบ bit-bang โดย MCU สร้างพัลส์ความกว้างต่างๆ เพื่อส่งคำสั่งและอ่านข้อมูล

Unique 64-bit ROM: DS18B20 ทุกตัวมีรหัส 64-bit ที่ไม่ซ้ำกันจากโรงงาน — ทำให้ต่อหลายตัวบนบัสเดียวกันได้ (Multi-drop)

ความละเอียด 9-12 bit: เลือกได้ — 9bit = 0.5°C/resolution, 12bit = 0.0625°C/resolution

DS18B20_Device Pointer: DS18B20_Init(pin) คืน DS18B20_Device* — ใช้ pointer นี้ในการอ่านค่า — DS18B20_ReadTemperatureBlocking() อ่านแบบบล็อก (รอจนเสร็จ) — DS18B20_ReadTemperature() อ่านแบบ non-blocking

Parasite Power: ใช้ไฟจากบัส DQ ระหว่าง conversion — ประหยัดสาย 1 เส้น (2-wire แทน 3-wire) แต่ต้องใช้ strong pull-up

Accuracy: ±0.5°C, ช่วง -55°C ถึง +125°C — กันน้ำได้ถ้าใช้รุ่นสาย (probe)

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Init 1-Wire (Single Sensor)

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "DS18B20.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();

    DS18B20_Device* sensor = DS18B20_Init(PD2);
    if(sensor == NULL) {
        USART_Println("No DS18B20 found!");
        while(1);
    }
    USART_Println("DS18B20 Ready!");

🔍 DS18B20_Init(pin) คืน DS18B20_Device* — ถ้าไม่เจอเซนเซอร์จะคืน NULL — ไม่ต้องเรียก OneWire_Init แยก — library จัดการให้เอง

ขั้นที่ 2: อ่านอุณหภูมิ (Blocking)

    float temp = DS18B20_ReadTemperatureBlocking(sensor);
    USART_Print("Temp: ");
    USART_PrintFloat(temp, 2);
    USART_Println(" C");

🔍 DS18B20_ReadTemperatureBlocking(sensor) — สั่งวัดและรอจนเสร็จ (บล็อก ~750ms สำหรับ 12-bit) — คืนค่าอุณหภูมิ °C โดยตรง — ไม่ต้องเรียก RequestTemperatures แยก

ขั้นที่ 3: Loop (Multi-Sensor + Non-Blocking)

    // สำหรับหลายเซนเซอร์: ใช้ SearchDevices
    DS18B20_SearchDevices(sensor);
    // อ่านทีละตัวแบบ non-blocking:
    float t = DS18B20_ReadTemperature(sensor);

    Timer_t tmr;
    Start_Timer(&tmr, 2000, 1);

    while(1) {
        if(Is_Timer_Expired(&tmr)) {
            float temp = DS18B20_ReadTemperatureBlocking(sensor);
            USART_Print("Temp: ");
            USART_PrintFloat(temp, 2);
            USART_Println(" C");
        }
        IWDG_Feed();
    }
}

🔍 DS18B20_SearchDevices(sensor) ค้นหาทุกตัวบนบัส — DS18B20_ReadTemperature(sensor) เป็น non-blocking — DS18B20_ReadTemperatureBlocking(sensor) เป็น blocking — ใช้ Timer_t กำหนดช่วงอ่าน

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
DS18B20_Init คืน NULLไม่มี pull-up resistor หรือสายหลวมต่อ R 4.7kΩ ระหว่าง DQ ↔ 3.3V, เช็คสายทุกเส้น
อ่านได้ 85°C ตลอดยังไม่ได้ทำ power-on reset หรือเซนเซอร์ยังไม่พร้อมเรียก DS18B20_Init ใหม่ — รอให้เซนเซอร์พร้อมหลังเปิดไฟ
Parasite mode ไม่อ่านค่าขาด strong pull-up ระหว่าง conversionใช้ MOSFET/transistor strong pull-up — หรือใช้โหมด 3-wire (VCC จริง)
ต่อหลายตัวแต่ไม่เจอPower-on ใหม่โดยไม่ Init ซ้ำเรียก DS18B20_SearchDevices() ใหม่ทุกครั้งหลังเปิดไฟ
อ่านค่าเพี้ยน ±1-2°Cสายยาวเกิน 3m หรือ noise บนบัสใช้ shielded cable, twisted pair, ลด pull-up เหลือ 2.2kΩ สำหรับสายยาว

🛠️ สร้าง Product: เครื่องวัดอุณหภูมิ 3 จุด

DS18B20 3 ตัวแยกพอร์ต — แสดงผลบน LCD1602 I2C

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "DS18B20.h"
#include "lcd1602_i2c.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    I2C_SimpleInit(I2C_SPEED_100K, I2C_PINS_DEFAULT);

    DS18B20_Device* s1 = DS18B20_Init(PD2);
    DS18B20_Device* s2 = DS18B20_Init(PC0);
    DS18B20_Device* s3 = DS18B20_Init(PC1);

    LCD1602_Handle lcd;
    LCD_Init(&lcd, 0x27, LCD_16x2);
    LCD_Backlight(&lcd, 1);

    Timer_t tmr;
    Start_Timer(&tmr, 2000, 1);
    char buf[16];

    while(1) {
        if(Is_Timer_Expired(&tmr)) {
            float t1 = DS18B20_ReadTemperatureBlocking(s1);
            float t2 = DS18B20_ReadTemperatureBlocking(s2);
            float t3 = DS18B20_ReadTemperatureBlocking(s3);

            LCD_Clear(&lcd);
            SimpleHAL_Sprintf(buf, "S1:%.1f S2:%.1f", t1, t2);
            LCD_SetCursor(&lcd, 0, 0);
            LCD_Print(&lcd, buf);
            SimpleHAL_Sprintf(buf, "S3:%.1f C", t3);
            LCD_SetCursor(&lcd, 1, 0);
            LCD_Print(&lcd, buf);
        }
        IWDG_Feed();
    }
}

✏️ ลองทำดู

  1. แสดง ROM ID ของเซนเซอร์ที่เจอบนบัส — ใช้ USART_PrintHex แสดง 8 byte ROM ID
  2. ทดลอง Parasite Power mode — ต่อ DS18B20 แบบ 2-wire (DQ+GND เท่านั้น) เปรียบเทียบกับ 3-wire ปกติ
  3. ทำ Data Logger + Flash — บันทึกอุณหภูมิจาก 3 จุดทุก 1 นาที เก็บลง Flash แล้วอ่านกลับมาดูภายหลังผ่าน USART
