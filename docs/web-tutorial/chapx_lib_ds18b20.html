1-Wire ต่อได้หลายร้อยตัวบนบัสเส้นเดียว — แต่ละตัวมี unique 64-bit ROM ID

📚 หลักการทำงาน

1-Wire Protocol: สื่อสารผ่านสายเส้นเดียว (DQ) — ใช้ timing แบบ bit-bang โดย MCU สร้างพัลส์ความกว้างต่างๆ เพื่อส่งคำสั่งและอ่านข้อมูล

Unique 64-bit ROM: DS18B20 ทุกตัวมีรหัส 64-bit ที่ไม่ซ้ำกันจากโรงงาน — ทำให้ต่อหลายตัวบนบัสเดียวกันได้ (Multi-drop)

ความละเอียด 9-12 bit: เลือกได้ — 9bit = 0.5°C/resolution, 12bit = 0.0625°C/resolution

Conversion time: 93.75ms (9bit) ถึง 750ms (12bit) — ต้องรอให้เสร็จก่อนอ่านค่า

Parasite Power: ใช้ไฟจากบัส DQ ระหว่าง conversion — ประหยัดสาย 1 เส้น (2-wire แทน 3-wire) แต่ต้องใช้ strong pull-up

Accuracy: ±0.5°C, ช่วง -55°C ถึง +125°C — กันน้ำได้ถ้าใช้รุ่นสาย (probe)

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Init 1-Wire + ค้นหา Devices

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "DS18B20.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    OneWire_Init(PC0);         // ตั้งขา DQ
    DS18B20_Init();            // ค้นหาทุกตัวบนบัส
    uint8_t count = DS18B20_GetDeviceCount();
    USART_Print("Found: "); USART_PrintNum(count);
    USART_Println(" devices");

🔍 OneWire_Init(PC0) กำหนดขา DQ, DS18B20_Init() จะ auto-scan ค้นหาทุกตัวที่ต่ออยู่บนบัส — DS18B20_GetDeviceCount() บอกจำนวนที่เจอ

ขั้นที่ 2: สั่งวัด + รอ + อ่านผล

    DS18B20_RequestTemperatures();   // สั่ง convert ทั้งหมด
    Delay_Ms(750);                    // รอ 12-bit conversion
    float t = DS18B20_GetTempCByIndex(0);  // อ่านตัวแรก
    USART_Print("Temp: ");
    USART_PrintFloat(t, 2);
    USART_Println(" C");

🔍 RequestTemperatures() สั่งให้ทุกตัวเริ่มวัดพร้อมกัน, Delay_Ms(750) ต้องรอให้วัดเสร็จ (ใช้เวลา 93.75ms ที่ 9bit ถึง 750ms ที่ 12bit), GetTempCByIndex(0) อ่านเซนเซอร์ตัวแรก

ขั้นที่ 3: วนอ่านหลายตัว + Loop

    Timer_t tmr;
    Start_Timer(&tmr, 1000, 1);  // อ่านทุก 1 วิ

    while(1) {
        if(Is_Timer_Expired(&tmr)) {
            DS18B20_RequestTemperatures();
            Delay_Ms(750);
            uint8_t n = DS18B20_GetDeviceCount();
            for(uint8_t i = 0; i < n; i++) {
                DS18B20_ROM_t rom;
                DS18B20_GetROM(i, &rom);
                float ti = DS18B20_GetTempC(&rom);
                USART_Print("Sensor "); USART_PrintNum(i);
                USART_Print(": "); USART_PrintFloat(ti, 2);
                USART_Println(" C");
            }
        }
        IWDG_Feed();
    }
}

🔍 วนอ่านทุกตัวด้วย GetROM(i, &rom) และ GetTempC(&rom) — ใช้ ROM ID อ้างอิงแต่ละตัว, IWDG_Feed() ป้องกัน watchdog reset ระหว่างรอ 750ms

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
อ่านได้ 85°C ตลอดยังไม่ได้เรียก RequestTemperatures() หรือรอไม่ครบ 750msต้องเรียก RequestTemperatures() ก่อน — แล้ว Delay_Ms(750) ก่อนอ่าน
GetDeviceCount() คืน 0ไม่มี pull-up resistor หรือสายหลวมต่อ R 4.7kΩ ระหว่าง DQ ↔ 3.3V, เช็คสายทุกเส้น
Parasite mode ไม่อ่านค่าขาด strong pull-up ระหว่าง conversionใช้ MOSFET/transistor strong pull-up ช่วงรอ 750ms หรือใช้โหมด 3-wire (VCC จริง)
ต่อหลายตัวแต่ ROM ซ้ำ / หายบางตัวPower-on ใหม่โดยไม่ Init ซ้ำเรียก DS18B20_Init() ใหม่ทุกครั้งหลังเปิดไฟ — รายการ ROM อาจเปลี่ยน
อ่านค่าเพี้ยน ±1-2°Cสายยาวเกิน 3m หรือ noise บนบัสใช้ shielded cable, twisted pair, ลด pull-up เหลือ 2.2kΩ สำหรับสายยาว
Conversion ไม่เสร็จ — อ่านค่าค้างใช้ความละเอียด 12bit แต่รอแค่ 93msรอ 750ms สำหรับ 12bit — หรือใช้ SetResolution(9) ลดเวลารอ

🛠️ สร้าง Product: เครื่องวัดอุณหภูมิ 3 จุด

DS18B20 3 ตัวบน 1-Wire บัสเดียว — แสดงผลบน LCD1602 I2C

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "DS18B20.h"
#include "LCD1602_I2C.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    I2C_SimpleInit(I2C_SPEED_100K, I2C_PINS_DEFAULT);

    OneWire_Init(PC0);
    DS18B20_Init();
    LCD1602_I2C_Init(0x27);

    Timer_t tmr;
    Start_Timer(&tmr, 2000, 1);

    while(1) {
        if(Is_Timer_Expired(&tmr)) {
            DS18B20_RequestTemperatures();
            Delay_Ms(750);
            uint8_t n = DS18B20_GetDeviceCount();
            for(uint8_t i = 0; i < n && i < 3; i++) {
                DS18B20_ROM_t rom;
                DS18B20_GetROM(i, &rom);
                float t = DS18B20_GetTempC(&rom);
                LCD1602_I2C_SetCursor(i, 0);
                LCD1602_I2C_Print("S"); LCD1602_I2C_PrintNum(i+1);
                LCD1602_I2C_Print(":"); LCD1602_I2C_PrintFloat(t, 1);
                LCD1602_I2C_Print("C   ");
            }
        }
        IWDG_Feed();
    }
}

✏️ ลองทำดู

  1. แสดง ROM ID ทั้งหมดของทุกเซนเซอร์ — ใช้ USART_PrintHex(rom.bytes, 8) พิมพ์ 8 byte ROM ID
  2. ทดลอง Parasite Power mode — ต่อ DS18B20 แบบ 2-wire (DQ+GND เท่านั้น) เปรียบเทียบกับ 3-wire ปกติ
  3. ทำ Data Logger + Flash — บันทึกอุณหภูมิจาก 3 จุดทุก 1 นาที เก็บลง Flash แล้วอ่านกลับมาดูภายหลังผ่าน USART
