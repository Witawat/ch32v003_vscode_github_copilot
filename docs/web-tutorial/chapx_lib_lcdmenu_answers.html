📝 เฉลย — LCD Menu

ข้อ 1: WiFi SSID Edit — แก้ไขชื่อ WiFi ผ่านปุ่มกด

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "LCDMenu.h"
#include "lcd1602_i2c.h"

LCD1602_Handle lcd;
Button_Instance btn_up, btn_down, btn_enter, btn_back;
LCDMenu_Handle menu;

char ssid[33] = "MyWiFi_2.4G";
uint8_t cursor_pos = 0;

const char charset[] = " ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789-_.";
#define CHARSET_LEN (sizeof(charset) - 1)

void ssid_edit_up(void) {
    char *p = strchr(charset, ssid[cursor_pos]);
    if(p) { int idx = (p - charset + 1) % CHARSET_LEN;
            ssid[cursor_pos] = charset[idx]; }
}

void ssid_edit_down(void) {
    char *p = strchr(charset, ssid[cursor_pos]);
    if(p) { int idx = (p - charset - 1 + CHARSET_LEN) % CHARSET_LEN;
            ssid[cursor_pos] = charset[idx]; }
}

void ssid_edit_right(void) { if(cursor_pos < 31) cursor_pos++; }
void ssid_edit_left(void)  { if(cursor_pos > 0)  cursor_pos--; }

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    I2C_LCD_Init(0x27, 16, 2);
    I2C_LCD_Backlight(1);

    LCDMenu_Init(&menu, &lcd, &btn_up, &btn_down, &btn_enter, &btn_back);
    LCDMenu_AddCallback(&menu, menu.root, "WiFi SSID", ssid_edit_up);

    while(1) {
        LCDMenu_Update(&menu);

        I2C_LCD_SetCursor(0, 0); I2C_LCD_Print("SSID:");
        I2C_LCD_SetCursor(0, 1); I2C_LCD_Print(ssid + max(0, cursor_pos - 15));
        I2C_LCD_SetCursor(min(cursor_pos, 15), 1);
        I2C_LCD_Write(0x5F);  // underscore for cursor

        Delay_Ms(50);
    }
}
🧠 แนวคิด: ใช้ charset[] เป็นชุดอักขระที่ยอมรับ — ปุ่ม up/down วนเลือกตัวอักษรใน charset ด้วย modulo — left/right เลื่อน cursor — แสดงผลโดย offset cursor_pos-15 ให้ auto-scroll เมื่อชื่อยาวเกิน 16 ตัว — ใช้ 0x5F (underscore) แทน cursor กะพริบ — LCDMENU_TYPE_CALLBACK บอกว่า menu item นี้เป็น action (ENTER เรียก callback) — เมื่อบันทึกค่า ควรเขียนลง EEPROM ด้วย AT24Cxx_Write — charset ควรแยกเป็นกลุ่ม (uppercase/lowercase/numbers/symbols) เพื่อให้แก้ไขเร็วขึ้น

ข้อ 2: Setpoint Adjust — ปรับค่า setpoint ด้วยปุ่ม up/down

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "LCDMenu.h"
#include "lcd1602_i2c.h"

LCD1602_Handle lcd;
Button_Instance btn_up, btn_down, btn_enter, btn_back;
LCDMenu_Handle menu;

static int32_t setpoint = 40;

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    I2C_LCD_Init(0x27, 16, 2);
    I2C_LCD_Backlight(1);

    LCDMenu_Init(&menu, &lcd, &btn_up, &btn_down, &btn_enter, &btn_back);
    LCDMenu_AddValue(&menu, menu.root, "Setpoint", &setpoint, 10, 80, 1, NULL);

    char buf[17];
    while(1) {
        LCDMenu_Update(&menu);

        I2C_LCD_SetCursor(0, 0); I2C_LCD_Print("TARGET TEMP");
        I2C_LCD_SetCursor(3, 1);
        snprintf(buf, 17, "%ld C    ", (long)setpoint);
        I2C_LCD_Print(buf);

        Delay_Ms(80);
    }
}
🧠 แนวคิด: LCDMenu_AddValue รับ int32_t* + min/max/step — ENTER เข้า edit mode, UP/DOWN ปรับ ±step, ENTER ยืนยัน, BACK ยกเลิก — clamp ด้วย min/max ป้องกันค่าผิดปกติ — แสดงด้วย %ld — space ท้ายสตริงลบตัวอักษรเก่าที่อาจค้าง — ควรเก็บค่า setpoint ลง EEPROM ทุกครั้งที่เปลี่ยน — สำหรับระบบจริงควรเพิ่ม callback on_exit เพื่อบันทึกค่าเฉพาะเมื่อออกจากเมนู

ข้อ 3: Toggle ON/OFF + Save ลง EEPROM

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "LCDMenu.h"
#include "lcd1602_i2c.h"
#include "AT24Cxx.h"

LCD1602_Handle lcd;
Button_Instance btn_up, btn_down, btn_enter, btn_back;
LCDMenu_Handle menu;
AT24Cxx_Instance eeprom;
#define EEPROM_ADDR  0x00

typedef struct {
    bool power_on;
    bool backlight;
    bool buzzer;
    float   setpoint;
} Config_t;

Config_t cfg;

void toggle_power(void)    { cfg.power_on  = !cfg.power_on; }
void toggle_backlight(void) { cfg.backlight = !cfg.backlight; }
void toggle_buzzer(void)   { cfg.buzzer    = !cfg.buzzer; }

void save_config(void) {
    AT24Cxx_Write(&eeprom, EEPROM_ADDR, (uint8_t*)&cfg, sizeof(Config_t));
}

void load_config(void) {
    AT24Cxx_Read(&eeprom, EEPROM_ADDR, (uint8_t*)&cfg, sizeof(Config_t));
    // ตรวจสอบ magic number / CRC
    if(cfg.setpoint < 0 || cfg.setpoint > 100) {
        memset(&cfg, 0, sizeof(Config_t));
        cfg.backlight = 1; cfg.power_on = 1; cfg.setpoint = 40.0f;
        save_config();
    }
}

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    I2C_LCD_Init(0x27, 16, 2);
    AT24Cxx_Init(&eeprom, AT24C32, 0x50);
    load_config();

    LCDMenu_Init(&menu, &lcd, &btn_up, &btn_down, &btn_enter, &btn_back);
    LCDMenu_AddToggle(&menu, menu.root, "Power", &cfg.power_on, toggle_power);
    LCDMenu_AddToggle(&menu, menu.root, "Backlight", &cfg.backlight, toggle_backlight);
    LCDMenu_AddToggle(&menu, menu.root, "Buzzer", &cfg.buzzer, toggle_buzzer);

    I2C_LCD_Print("Config Loaded");
    Delay_Ms(1000);

    while(1) {
        LCDMenu_Update(&menu);

        static Config_t last_cfg;
        if(memcmp(&cfg, &last_cfg, sizeof(Config_t)) != 0) {
            I2C_LCD_Clear();
            I2C_LCD_SetCursor(0, 0); I2C_LCD_Printf("PWR:%s BL:%s BZ:%s",
                cfg.power_on ? "ON " : "OFF",
                cfg.backlight ? "ON " : "OFF",
                cfg.buzzer ? "ON " : "OFF");
            I2C_LCD_SetCursor(0, 1); I2C_LCD_Print("→ saving...");
            save_config();
            last_cfg = cfg;
        }

        Delay_Ms(50);
    }
}
🧠 แนวคิด: LCDMENU_TYPE_TOGGLE สำหรับ boolean settings — เก็บ config ใน Config_t struct — save_config() เขียน struct ทั้งหมดลง AT24Cxx EEPROM — load_config() อ่านตอนบูท พร้อม validation: ถ้าค่าผิดปกติ → reset เป็น default แล้ว save — ใช้ memcmp เช็คการเปลี่ยนแปลงก่อน save ลด EEPROM wear — ควรเพิ่ม CRC8/CRC16 ใน struct เพื่อตรวจจับ EEPROM corruption — สำหรับ production ควรมี factory reset (กดปุ่มค้างตอนบูท) เพื่อ restore default
