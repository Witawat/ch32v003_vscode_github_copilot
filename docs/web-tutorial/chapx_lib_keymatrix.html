MCU Pin→อุปกรณ์ขาหมายเหตุ
PC0→Keypad 4×4ROW1Output active LOW
PC1→Keypad 4×4ROW2Output active LOW
PC2→Keypad 4×4ROW3Output active LOW
PC3→Keypad 4×4ROW4Output active LOW
PC4←Keypad 4×4COL1Input พร้อม internal pull-up
PC5←Keypad 4×4COL2Input พร้อม internal pull-up
PC6←Keypad 4×4COL3Input พร้อม internal pull-up
PC7←Keypad 4×4COL4Input พร้อม internal pull-up

Rows เป็น Output (active LOW), Columns เป็น Input พร้อม internal pull-up — เมื่อกดปุ่ม Column ที่ตรงกันจะถูกดึงลง LOW

รู้จักขา

  • ROW1–ROW4: ขาแถว (Output) ต่อกับ PC0–PC3 — MCU จะสแกนทีละแถว
  • COL1–COL4: ขาแนวตั้ง (Input pull-up) ต่อกับ PC4–PC7 — อ่านว่าปุ่มไหนถูกกด

วิธีต่อทีละขั้น

  1. ต่อ PC0–PC3 ไป ROW1–ROW4 — MCU จะดึงลงทีละแถว (Active LOW)
  2. ต่อ PC4–PC7 ไป COL1–COL4 — อ่านค่าเมื่อปุ่มถูกกด (ใช้ internal pull-up)
  3. ไม่ต้องต่อ R ภายนอก — ใช้ internal pull-up ที่ Columns

📚 หลักการทำงาน

Row Scan: เริ่มจาก Row1=0 (Active) ส่วน Row2-4=1 — อ่านค่าที่ Column ทั้ง 4 ถ้า Column ไหนอ่านได้ 0 แสดงว่าปุ่มที่ intersection นั้นถูกกด แล้วเลื่อนไป Row ถัดไป — สแกนวนครบ 4 Row

Ghost Key Prevention: เมื่อกด 3 ปุ่มพร้อมกันในตำแหน่งมุมของสี่เหลี่ยมผืนผ้า จะเกิด "phantom key" — KeyMatrix ตรวจจับและกรองได้ด้วยการเช็คจำนวน Column ที่ Active ต่อ Row

Debounce ~20ms: ใช้หลักการเดียวกับ Button แต่เร็วกว่าเพราะ keypad มักเด้งน้อยกว่าปุ่มเดี่ยว — ปรับได้ตามคุณภาพ keypad

Key Mapping Matrix: char keys[4][4] กำหนดอักขระให้แต่ละตำแหน่ง — สามารถเปลี่ยนเป็นตัวเลขหรืออักขระพิเศษได้อิสระ — รองรับสูงสุด 8×8 (64 ปุ่ม)

Rollover: รองรับการกด 2 ปุ่มพร้อมกัน (2-key rollover) — ถ้ากดมากกว่านี้อาจเกิด ghost ได้ในบาง keypad layout

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: กำหนด Pins + Key Map + Init

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "KeyMatrix.h"

KeyMatrix_Instance kp;
GPIO_Pin rows[] = {PC0, PC1, PC2, PC3};
GPIO_Pin cols[] = {PC4, PC5, PC6, PC7};
char map[4][KEYMATRIX_MAX_COLS] = {
    {'1','2','3','A'},
    {'4','5','6','B'},
    {'7','8','9','C'},
    {'*','0','#','D'}
};

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_Init(115200);

    KeyMatrix_Init(&kp, rows, 4, cols, 4, map);
    USART_Println("Keypad 4x4 Ready!");

🔍 KeyMatrix_Init(&kp, rows, numRows, cols, numCols, keyMap) — ใช้ GPIO_Pin array — rows/cols ละ 4 pin รองรับสูงสุด 8×8 — ใช้ internal pull-up อัตโนมัติ

ขั้นที่ 2: Update + GetKey ใน Loop

    char last_key = 0;

    while(1) {
        KeyMatrix_Update(&kp);
        char key = KeyMatrix_GetKey(&kp);
        if(key && key != last_key) {
            USART_WriteByte(key);
            USART_Println("");
            last_key = key;
        }
        if(!key) last_key = 0;
        Delay_Ms(20);  // scan rate ~50Hz
    }
}

🔍 KeyMatrix_GetKey(&kp) คืนอักขระที่กดล่าสุด หรือ 0 ถ้าไม่มี — ใช้ last_key ป้องกันการส่งซ้ำ — scan rate 20ms (~50Hz) เร็วพอสำหรับการพิมพ์ปกติ

ขั้นที่ 3: Multi-Key + State Machine

    char buf[16]; uint8_t idx = 0;

    while(1) {
        KeyMatrix_Update(&kp);
        char key = KeyMatrix_GetKey(&kp);
        if(key) {
            if(key == '#') {  // Enter
                buf[idx] = '\0';
                USART_Print("INPUT: "); USART_Println(buf);
                idx = 0;
            } else if(key == '*') {  // Clear
                idx = 0;
                USART_Println("CLEAR");
            } else if(idx < 15) {
                buf[idx++] = key;
                USART_WriteByte(key);
            }
            Delay_Ms(200);  // debounce release
        }
        IWDG_Feed();
    }
}

🔍 ใช้งานจริงต้องกันการกดซ้ำ — Delay_Ms(200) หลังได้ key ป้องกันการอ่านซ้ำตอนปล่อย — สร้าง state machine ง่ายๆ: * = clear, # = enter

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
กดปุ่มแล้วขึ้นผิดตัวอักษรKey map สลับ row/col — หรือต่อสายผิดตรวจสอบ key map กับ pin mapping — row กับ col ต้องตรงกับสายที่ต่อ
Phantom key — กด 1 ปุ่มแต่ขึ้น 3 ตัวGhost key จาก layout รูปสี่เหลี่ยมกดครั้งละไม่เกิน 2 ปุ่ม — หรือใช้ diode กัน ghost (hardware fix)
ปุ่มหลุด ไม่ตอบสนองบางครั้งDebounce น้อยไป หรือ scan rate ช้าเกินเพิ่ม debounce เป็น 25-30ms — ลด delay ใน loop ให้ต่ำกว่า 30ms
Column อ่านได้ 0 ตลอดเวลาลืมเปิด pull-up — หรือดึงลงด้วยฮาร์ดแวร์เช็ค pinMode(col, INPUT_PULLUP) — ต่อ R 10kΩ pull-up ภายนอกสำรอง
Row ไม่ยอม activeตั้ง Row เป็น input แทน outputRow ต้องเป็น output — ไลบรารีตั้งให้อัตโนมัติ แต่ถ้าใช้ pinMode เองต้องระวัง
Short circuit เมื่อกด 2 ปุ่มRow output 2 แถว active พร้อมกันไม่ควรเกิดกับไลบรารี — แต่ถ้าเกิดให้เช็คว่ามีโค้ดอื่นเปลี่ยน pin โดยตรงหรือไม่

🛠️ สร้าง Product: Calculator 4×4 + LCD1602

เครื่องคิดเลขพื้นฐาน บวก ลบ คูณ หาร แสดงผลบน LCD1602 I2C — ใช้ KeyMatrix รับอินพุต

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "KeyMatrix.h"
#include "lcd1602_i2c.h"

KeyMatrix_Instance kp;
GPIO_Pin rows[] = {PC0,PC1,PC2,PC3};
GPIO_Pin cols[] = {PC4,PC5,PC6,PC7};
char map[4][KEYMATRIX_MAX_COLS] = {{'1','2','3','+'},{'4','5','6','-'},{'7','8','9','*'},{'C','0','=','/'}};
int32_t a=0, b=0, result=0; char op=0, disp[17]; uint8_t state=0;

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_Init(115200);
    LCD_Init(0x27);
    KeyMatrix_Init(&kp, rows, 4, cols, 4, map);
    LCD_SetCursor(0,0); LCD_Print("Calc Ready!");
    while(1) {
        KeyMatrix_Update(&kp);
        char key = KeyMatrix_GetKey(&kp);
        if(key) {
            if(key>='0' && key<='9') {
                if(state==0) a=a*10+(key-'0');
                else b=b*10+(key-'0');
            } else if(key=='C') { a=b=result=op=state=0; }
            else if(key=='=') {
                switch(op){case'+':result=a+b;break;case'-':result=a-b;break;
                case'*':result=a*b;break;case'/':result=b? a/b:0;break;}
                a=result; b=0; op=0; state=0;
            } else { op=key; state=1; }
            sprintf(disp,"%ld %c %ld=%ld",a,op?op:' ',b,result);
            LCD_SetCursor(0,0); LCD_Print(disp);
            Delay_Ms(250);
        }
        IWDG_Feed();
    }
}

✏️ ลองทำดู

  1. Password Lock — รับรหัส 4 หลัก เปรียบเทียบกับรหัสที่เก็บไว้ ถ้าถูกต้อง LED ติด — ใช้ strcmp()
  2. Key Repeat — กดค้างแล้วพิมพ์ซ้ำอัตโนมัติ เหมือนคีย์บอร์ด — ใช้ Timer_t จับเวลา 500ms แล้ว repeat ทุก 100ms
  3. MIDI Controller — 4x4 pad ส่งโน้ต MIDI ผ่าน USART — map แต่ละปุ่มเป็น note number (C4=60)
