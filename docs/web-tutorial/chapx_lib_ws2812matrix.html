WS2812Matrix — 8×8 RGB Matrix GPIO

WS2812B 8×8 Matrix มี LED 64 ดวงเรียงแบบ Serpentine (S-curve) — ใช้ protocol เดียวกับ NeoPixel — ต้องต่อ 5V แยก ≥ 2A — Level Shifter จำเป็น

รู้จักขา

  • VCC: ไฟเลี้ยง 5V อย่างน้อย 2A (64 LEDs × 60mA peak)
  • DIN: ขาข้อมูลเข้า ต่อกับ PC0 ผ่าน level shifter
  • GND: ต่อกราวด์ร่วมกับ MCU

วิธีต่อทีละขั้น

  1. ต่อ PC0 ผ่าน Level Shifter 3.3V→5V ไป DIN — จำเป็น! WS2812B ใช้ 5V logic
  2. ต่อ 5V ภายนอก (≥2A) ไป VCC — อย่าใช้ไฟจาก MCU
  3. ต่อ GND MCU และไฟแยก ไป GND — กราวด์ร่วม

📚 หลักการทำงาน

Instance-Based API: WS2812M_Instance ใช้จอง buffer + จัดการ LED แบบ instance — รองรับ matrix ขนาดต่างๆ — ระบุ pin, width, height, และ wiring mode

Serpentine Mapping: 8×8 Matrix เรียง LED แบบซิกแซก (แถว 0 ซ้าย→ขวา, แถว 1 ขวา→ซ้าย, ...) — ใช้ WS2812M_ZIGZAG flag — ระบุตำแหน่งด้วย (x, y) แบบ Cartesian ปกติ

Effects Engine: มี effect ในตัว — Rainbow, Sparkle, ColorWipe — ใช้ Non-Blocking state machine ในแต่ละ effect

Color Support: 24-bit RGB ต่อ LED — WS2812M_SetPixel(&matrix, x, y, r, g, b) — 16.7 ล้านสี — แต่ละ LED กินกระแสสูงสุด 60mA (RGB full white)

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: สร้าง Instance + Init

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "WS2812Matrix.h"

WS2812M_Instance matrix;

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();

    WS2812M_Init(&matrix, PC0, 8, 8, WS2812M_ZIGZAG);
    WS2812M_Clear(&matrix);
    WS2812M_Show(&matrix);

🔍 WS2812M_Init(&matrix, pin, w, h, wiring) — ใช้ WS2812M_ZIGZAG สำหรับ serpentine layout — รองรับ matrix ขนาดต่างๆ — Clear+Show ล้างหน้าจอ

ขั้นที่ 2: แสดง Pixel + Effect

    // ตั้ง pixel ทีละดวง
    WS2812M_SetPixel(&matrix, 0, 0, 255, 0, 0);   // Red at (0,0)
    WS2812M_SetPixel(&matrix, 3, 3, 0, 255, 0);   // Green at (3,3)
    WS2812M_SetPixel(&matrix, 7, 7, 0, 0, 255);   // Blue at (7,7)
    WS2812M_Show(&matrix);
    Delay_Ms(1000);

    // ล้างหน้าจอ
    WS2812M_Clear(&matrix);
    WS2812M_Show(&matrix);

    // Effect: Rainbow
    WS2812M_Rainbow(&matrix, 30);

🔍 WS2812M_SetPixel(&matrix, x, y, r, g, b) ตั้งค่าสีแบบ RGB แยก — WS2812M_Show(&matrix) ส่งข้อมูลไป LED — WS2812M_Clear(&matrix) ล้าง buffer — WS2812M_Rainbow effect รุ้งวน

ขั้นที่ 3: Loop สลับ Effect

    Timer_t tmr;
    Start_Timer(&tmr, 5000, 1);
    uint8_t fx = 0;

    while(1) {
        if(Is_Timer_Expired(&tmr)) {
            WS2812M_Clear(&matrix);
            if(fx == 0) {
                for(uint8_t x=0; x<8; x++)
                    WS2812M_SetPixel(&matrix, x, 0, 255, 0, 0);
                WS2812M_Show(&matrix);
            } else if(fx == 1) {
                WS2812M_Rainbow(&matrix, 30);
            }
            fx = (fx + 1) % 2;
        }
    }
}

🔍 สลับ effect ทุก 5 วินาที — ใช้ Non-Blocking Timer — WS2812M_Rainbow ทำงานต่อเนื่อง — WS2812M_Show ส่ง buffer ไป hardware

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
Matrix ไม่ติดเลยขาด Level Shifter — 3.3V logic ไม่พอใช้ 74HCT125 — หรือใช้โมดูลที่มี level shifter ในตัว
Matrix ติดบางส่วน สีผิดเพี้ยนไฟเลี้ยงไม่พอ — ต้องการ 5V ≥ 2Aใช้ external PSU 5V 2A+ — ต่อไฟเข้าขั้ว Matrix โดยตรง
ข้อความแสดงตำแหน่งผิดSerpentine mapping ไม่ตรงกับโมดูลตรวจสอบ wiring flag — WS2812M_ZIGZAG หรือ WS2812M_PROGRESSIVE
สีจางเมื่อเปิดหลาย LEDVoltage drop — สายไฟเล็กเกินใช้สาย AWG 20-22 — ต่อ 5V + GND ทั้งด้านเข้าและออก
Effect กระตุกShow() ใช้เวลานานสำหรับ 64 LEDs64 LEDs = ~1.9ms — ใช้ Non-Blocking effect variant ที่มี state machine

🛠️ สร้าง Product: ป้ายไฟ RGB 3 Effect สลับ

8×8 Matrix แสดง 3 โหมด — Rainbow → Sparkle → Pixel Art — กดปุ่มเปลี่ยนโหมด

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "WS2812Matrix.h"

WS2812M_Instance matrix;

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    WS2812M_Init(&matrix, PC0, 8, 8, WS2812M_ZIGZAG);

    pinMode(PC3, PIN_MODE_INPUT_PULLUP);
    uint8_t mode = 0;

    while(1) {
        if(digitalRead(PC3) == 0) {
            Delay_Ms(50);
            if(digitalRead(PC3) == 0) {
                mode = (mode + 1) % 3;
                while(digitalRead(PC3) == 0);
            }
        }

        switch(mode) {
            case 0:
                WS2812M_Rainbow(&matrix, 30);
                break;
            case 1:
                WS2812M_Clear(&matrix);
                WS2812M_SetPixel(&matrix, 4, 4, 255, 255, 0);
                WS2812M_Show(&matrix);
                break;
            case 2:
                WS2812M_Clear(&matrix);
                for(uint8_t x=0; x<8; x++) {
                    WS2812M_SetPixel(&matrix, x, 0, 0, 255, 0);
                    WS2812M_SetPixel(&matrix, x, 7, 0, 255, 0);
                }
                WS2812M_Show(&matrix);
                break;
        }
    }
}
