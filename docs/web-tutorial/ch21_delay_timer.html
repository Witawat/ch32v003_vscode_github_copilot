ex01_NonBlocking_Timer — Non-blocking Timer ควบคุม LED

📖 แนวคิด

ใช้ Timer_t + Start_Timer + Is_Timer_Expired ควบคุม LED blink โดยไม่บล็อก main loop — ใช้ SysTick millis counter (ไม่ใช่ hardware timer) — main loop ยังทำงานอื่นได้ระหว่างรอ

💻 โค้ด + อธิบายทีละบรรทัด

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include <SimpleHAL.h>

static Timer_t led_timer;          // ตัวแปร timer สำหรับควบคุม LED
static Timer_t debounce_timer;     // ตัวแปร timer สำหรับ debounce ปุ่ม
static uint8_t led_state = 0;      // สถานะ LED (0=OFF, 1=ON)
static uint8_t timer_running = 1;  // สถานะการทำงานของ timer
static uint8_t btn_pending = 0;    // flag: มีปุ่มกดรอประมวลผล

int main(void)
{
    SystemCoreClockUpdate();            // อัปเดตความถี่ระบบ
    Timer_Init();                       // เริ่ม SysTick timer (1ms resolution)
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);  // USART 115200 baud

    pinMode(PC0, PIN_MODE_OUTPUT);      // PC0 output → LED
    pinMode(PC1, PIN_MODE_INPUT_PULLUP);  // PC1 input pull-up → ปุ่มกด

    Start_Timer(&led_timer, 500, 1);    // เริ่ม timer 500ms, repeat=1 (วนซ้ำ)
    timer_running = 1;

    uint32_t last_print = 0;

    while (1)
    {
        // === อ่านปุ่ม non-blocking debounce ===
        if (digitalRead(PC1) == LOW) {
            if (!debounce_timer.active) {
                Start_Timer(&debounce_timer, 50, 0);  // debounce 50ms
            }
            if (Is_Timer_Expired(&debounce_timer) && !btn_pending) {
                btn_pending = 1;
            }
        } else {
            if (btn_pending) {
                btn_pending = 0;
                if (timer_running) {
                    Stop_Timer(&led_timer);
                    timer_running = 0;
                    digitalWrite(PC0, LOW);
                    USART_Print("Timer Stopped\r\n");
                } else {
                    Start_Timer(&led_timer, 500, 1);
                    timer_running = 1;
                    USART_Print("Timer Started\r\n");
                }
            }
            if (debounce_timer.active) {
                Stop_Timer(&debounce_timer);
            }
        }

        // === ตรวจสอบ Timer หมดเวลาหรือยัง ===
        if (timer_running) {
            if (Is_Timer_Expired(&led_timer))  // ไม่บล็อก — คืนทันที
            {
                led_state = !led_state;
                digitalWrite(PC0, led_state);
                USART_Print(led_state ? "LED ON\r\n" : "LED OFF\r\n");
            }
        }

        // === แสดงว่า main loop ยังทำงาน ===
        if (ELAPSED_TIME(last_print, Get_CurrentMs()) >= 100)
        {
            last_print = Get_CurrentMs();
            USART_Print("loop\r\n");  // พิมพ์ทุก 100ms แสดงว่าไม่ถูกบล็อก
        }
    }
}
  • Timer_t: struct ที่เก็บเวลาที่ตั้ง + สถานะ — ใช้ร่วมกับ Start_Timer / Is_Timer_Expired / Stop_Timer
  • Is_Timer_Expired: คืนค่า 1 ทันทีเมื่อ timer หมดเวลา — ไม่บล็อก — ใช้ SysTick millis เปรียบเทียบ
  • repeat=1: timer restart อัตโนมัติ — repeat=0 ต้อง Start_Timer ใหม่ทุกครั้ง
  • debounce_timer.active: เช็คว่า timer กำลังทำงานอยู่ — ใช้เช็คสถานะแทนตัวแปรแยก

⚠️ จุดควรระวัง

  • ต้องเรียก Start_Timer() ก่อน Is_Timer_Expired() เสมอ — ค่าเริ่มต้นของ Timer_t ไม่แน่นอน
  • Timer นี้ใช้ SysTick counter ไม่ใช่ hardware timer — แม่นยำ ±1ms
  • ห้ามใช้ timer ค่า 0ms — จะทำงานไม่ถูกต้อง (expired ทันที)
  • repeat=0 หมดแล้วไม่ restart — ต้อง Start_Timer ใหม่

✅ ตรวจสอบก่อนรัน

  • LED ต่อ R 220Ω ระหว่าง PC0 กับแอโนด? แคโทด→GND?
  • ปุ่มต่อ PC1→GND ใช้ internal pull-up? (กด=LOW, ปล่อย=HIGH)
  • Serial monitor 115200 baud? แสดง "loop" ทุก 100ms?

ex02_Blocking_Delay — Blocking Delay ควบคุม LED Sequence

📖 แนวคิด

ใช้ Delay_Ms() และ Delay_Us() แบบ blocking — หน่วงเวลาแม่นยำแต่ CPU หยุดรอ — แสดง sequence ON/OFF หลายช่วง รวมถึงทดสอบ Delay_Us 1µs

💻 โค้ด + อธิบายทีละบรรทัด

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include <SimpleHAL.h>

static uint8_t led_pin = PC0;

int main(void)
{
    SystemCoreClockUpdate();            // อัปเดตความถี่ระบบ
    Timer_Init();                       // เริ่ม SysTick (จำเป็นสำหรับ Delay)
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);

    pinMode(led_pin, PIN_MODE_OUTPUT);

    USART_Print("Blocking Delay Example\r\n");

    while (1)
    {
        // Sequence 1: ON 200ms, OFF 100ms
        USART_Print("Sequence: ON 200ms\r\n");
        digitalWrite(led_pin, HIGH);
        Delay_Ms(200);                  // บล็อก 200ms

        USART_Print("Sequence: OFF 100ms\r\n");
        digitalWrite(led_pin, LOW);
        Delay_Ms(100);                  // บล็อก 100ms

        // Sequence 2: ON 300ms, OFF 1000ms
        USART_Print("Sequence: ON 300ms\r\n");
        digitalWrite(led_pin, HIGH);
        Delay_Ms(300);                  // บล็อก 300ms

        USART_Print("Sequence: OFF 1000ms\r\n");
        digitalWrite(led_pin, LOW);
        Delay_Ms(1000);                 // บล็อก 1 วินาที

        // ทดสอบ Delay_Us แม่นยำระดับ microseconds
        USART_Print("Testing Delay_Us 1us for 10 cycles\r\n");
        for (uint8_t i = 0; i < 10; i++)
        {
            digitalWrite(led_pin, HIGH);
            Delay_Us(1);                // บล็อก 1µs
            digitalWrite(led_pin, LOW);
            Delay_Us(1);                // บล็อก 1µs
        }
        USART_Print("Delay_Us precision test complete!\r\n\r\n");
    }
}
  • Delay_Ms(ms): blocking delay ระดับ millisecond — ใช้ SysTick busy-wait — แม่นยำ ±1ms
  • Delay_Us(us): blocking delay ระดับ microsecond — ใช้ instruction cycle counting — แม่นยำ ±2-3 cycles
  • Sequence: ควบคุม LED ON/OFF หลายช่วงเวลา — 200ms, 100ms, 300ms, 1000ms
  • Delay_Us(1): เร็วเกินตาเห็น แต่พิสูจน์ความสามารถระดับ µs ของ MCU

⚠️ จุดควรระวัง

  • Delay_Ms / Delay_Us BLOCKING — CPU ยุ่งรออยู่ ไม่ทำงานอื่น
  • ระหว่าง delay interrupt อื่นยังทำงานได้ (SysTick, TIM) แต่ main loop หยุด
  • Delay_Us(0) หรือ Delay_Ms(0) → return ทันที
  • ไม่เหมาะกับงาน real-time ที่ต้องตอบสนองตลอด — ใช้ non-blocking timer แทน

✅ ตรวจสอบก่อนรัน

  • LED ต่อ R 220Ω ถูกต้อง?
  • Serial monitor แสดง sequence ตามที่ตั้ง?
  • สังเกตว่า main loop ถูกบล็อก — กดปุ่มระหว่าง delay ไม่มีผล

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include <SimpleHAL.h>

static uint8_t led_pin = PC0;
static uint32_t loop_count = 0;

int main(void)
{
    SystemCoreClockUpdate();
    Timer_Init();                       // เริ่ม SysTick (1ms resolution)
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);

    pinMode(led_pin, PIN_MODE_OUTPUT);

    USART_Print("Time Reading Example\r\n");
    USART_Print("========================\r\n");

    uint32_t last_report = 0;
    uint32_t start_time = Get_CurrentMs();   // จับเวลาเริ่มต้น
    uint32_t start_us = Get_CurrentUs();     // จับเวลาเริ่มต้น (µs)

    while (1)
    {
        uint32_t current_ms = Get_CurrentMs();
        uint32_t current_us = Get_CurrentUs();
        uint32_t tick_us = Get_TickMicros();     // 0-999 ใน tick ปัจจุบัน

        if (ELAPSED_TIME(last_report, current_ms) >= 1000)
        {
            last_report = current_ms;

            digitalWrite(led_pin, HIGH);
            Delay_Ms(50);
            digitalWrite(led_pin, LOW);

            // แสดงค่าปัจจุบัน
            USART_Print("Milliseconds: ");
            USART_PrintNum(current_ms);
            USART_Print(", Microseconds: ");
            USART_PrintNum(current_us);
            USART_Print("\r\n");

            // แสดง Tick Micros
            USART_Print("TickMicros: ");
            USART_PrintNum(tick_us);
            USART_Print(" (within current SysTick)\r\n");

            // Elapsed time แบบ overflow-safe
            uint32_t elapsed = ELAPSED_TIME(start_time, current_ms);
            USART_Print("Elapsed since start: ");
            USART_PrintNum(elapsed);
            USART_Print(" ms\r\n");

            // Elapsed microseconds
            uint32_t elapsed_us = current_us - start_us;
            USART_Print("Elapsed micros: ");
            USART_PrintNum(elapsed_us);
            USART_Print(" us\r\n");

            loop_count++;
            USART_Print("Loop count: ");
            USART_PrintNum(loop_count);
            USART_Print("\r\n\r\n");
        }

        // วัดเวลา USART_Print
        uint32_t t1 = Get_CurrentUs();
        USART_Print("");
        uint32_t t2 = Get_CurrentUs();
        uint32_t usart_time = t2 - t1;
        if (usart_time > 100) {
            USART_Print("Note: USART took ");
            USART_PrintNum(usart_time);
            USART_Print(" us\r\n");
        }
    }
}
  • Get_CurrentMs(): คืนค่า millisecond ปัจจุบัน (SysTick counter) — uint32_t — overflow ~49 วัน
  • Get_CurrentUs(): คืนค่า microsecond ปัจจุบัน — uint32_t — overflow ~71 นาที
  • ELAPSED_TIME(start, now): คำนวณ elapsed time แบบ overflow-safe — ใช้ unsigned arithmetic wrap-around
  • Get_TickMicros(): คืนค่า 0-999 — microsecond ใน SysTick ปัจจุบัน
  • วัดเวลา USART: จับ t1 ก่อน → USART_Print → t2 หลัง = เวลาที่ใช้

⚠️ จุดควรระวัง

  • Get_CurrentUs() overflow หลังจาก ~71 นาที — Get_CurrentMs() overflow หลังจาก ~49 วัน
  • ห้ามลบค่า millis/micros โดยตรง — ใช้ ELAPSED_TIME macro ป้องกัน overflow error
  • Get_CurrentUs() ปิด interrupt ชั่วคราวเพื่ออ่านค่า atomic — อาจกระทบ timing เล็กน้อย
  • ค่าเวลามีไว้เปรียบเทียบ — ไม่ใช่เวลาจริงของโลก (อาจคลาดเคลื่อนจาก crystal)

✅ ตรวจสอบก่อนรัน

  • Serial monitor แสดง millis, micros, tick micros เพิ่มขึ้นเรื่อย ๆ?
  • Elapsed time ตรงกับเวลาจริง? (นาฬิกาจับเวลา 10 วิ → แสดง ~10000 ms)
  • LED กระพริบทุกครั้งที่รายงาน?

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include <SimpleHAL.h>

static volatile uint8_t led_state = 0;   // volatile — เปลี่ยนใน ISR
static uint8_t led_pin = PC0;
static uint8_t button_pin = PC1;

void timer_isr_callback(void)
{
    led_state = !led_state;              // สลับสถานะ
    digitalWrite(led_pin, led_state);    // เขียนค่า LED (register-level)
}

int main(void)
{
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);

    pinMode(led_pin, PIN_MODE_OUTPUT);
    pinMode(button_pin, PIN_MODE_INPUT_PULLUP);

    // ใช้ TIM2 (TIM1 สงวนให้ SimplePWM)
    TIM_SimpleInit(TIM_2, 2);                  // TIM2 2Hz = 500ms period
    TIM_AttachInterrupt(TIM_2, timer_isr_callback);  // กำหนด callback
    TIM_Start(TIM_2);                          // เริ่ม timer

    USART_Print("Timer Interrupt Example\r\n");
    USART_Print("TIM2 configured at 2Hz (500ms)\r\n");
    USART_Print("Press PC1 button to start/stop\r\n");

    uint32_t last_print = 0;

    while (1)
    {
        // === Non-blocking button state machine ===
        static uint8_t timer_running = 1;
        static uint8_t btn_state = 0;
        static Timer_t btn_timer;

        if (btn_state == 0 && digitalRead(button_pin) == LOW) {
            btn_state = 1;
            Start_Timer(&btn_timer, 50, 0);
        }
        if (btn_state == 1 && Is_Timer_Expired(&btn_timer)) {
            if (digitalRead(button_pin) == LOW) {
                btn_state = 2;
                timer_running = !timer_running;
                if (timer_running) {
                    TIM_Start(TIM_2);
                    USART_Print("Timer Started\r\n");
                } else {
                    TIM_Stop(TIM_2);
                    digitalWrite(led_pin, LOW);
                    USART_Print("Timer Stopped\r\n");
                }
            } else {
                btn_state = 0;
            }
        }
        if (btn_state == 2 && digitalRead(button_pin) == HIGH) {
            btn_state = 0;
        }

        // === Main loop ทุก 2 วิ ===
        if (ELAPSED_TIME(last_print, Get_CurrentMs()) >= 2000)
        {
            last_print = Get_CurrentMs();
            USART_Print("Main loop running, LED state: ");
            USART_Print(led_state ? "ON\r\n" : "OFF\r\n");

            uint16_t counter = Simple_TIM_GetCounter(TIM_2);
            USART_Print("TIM2 Counter: ");
            USART_PrintNum(counter);
            USART_Print("\r\n");
        }
    }
}
  • TIM_SimpleInit(TIM_2, freq_hz): ตั้งค่า timer ที่ความถี่ Hz — 2Hz = 500ms ต่อรอบ
  • TIM_AttachInterrupt: กำหนด callback ที่จะถูกเรียกใน ISR ทุกครั้งที่ timer overflow
  • TIM_Start / TIM_Stop: เริ่ม/หยุด timer — ใช้ปุ่มกด toggle
  • volatile: จำเป็นสำหรับตัวแปรที่เปลี่ยนค่าใน ISR — ป้องกัน optimization
  • Non-blocking button: state machine 3 state (wait → debounce → confirm) ใช้ Timer_t debounce

⚠️ จุดควรระวัง

  • TIM1 สงวนสำหรับ SimplePWM! — ใช้ TIM2 สำหรับ timer interrupt เสมอ
  • ห้าม Delay_Ms() หรือ USART_Print() ใน ISR — blocking + priority inversion → deadlock
  • ISR ควรสั้นที่สุด — ตั้ง flag แล้วให้ main loop จัดการ
  • ตัวแปร共用 ISR↔main loop ต้อง volatile

✅ ตรวจสอบก่อนรัน

  • LED กระพริบ 500ms โดยอัตโนมัติ (ISR จัดการ)?
  • กด PC1 → LED หยุดกระพริบ (TIM_Stop)? กดอีก → กลับมากระพริบ?
  • Serial monitor แสดง LED state และ counter ทุก 2 วิ?

💻 โค้ด + อธิบายทีละบรรทัด

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include <SimpleHAL.h>

static uint8_t led_pin = PC0;
static uint16_t last_counter = 0;        // ค่า counter ก่อนหน้า (ตรวจจับ wrap)

int main(void)
{
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);

    pinMode(led_pin, PIN_MODE_OUTPUT);

    // ตั้งค่า: 10kHz ด้วย PSC=239, Period=9 @24MHz
    // Frequency = 24000000 / ((239+1)*(9+1)) = 24000000/2400 = 10000 Hz
    uint16_t prescaler = 239;
    uint16_t period = 9;
    TIM_Mode mode = TIM_MODE_UP;

    TIM_AdvancedInit(TIM_1, prescaler, period, mode);
    TIM_Start(TIM_1);

    uint32_t last_report = 0;
    uint8_t freq_changed = 0;

    while (1)
    {
        // อ่าน counter และ period ปัจจุบัน
        uint16_t counter = Simple_TIM_GetCounter(TIM_1);
        uint16_t current_period = TIM_GetPeriod(TIM_1);

        // ตรวจจับ counter wrap (overflow)
        if (counter < last_counter)
        {
            USART_Print("Counter wrapped! Period=");
            USART_PrintNum(current_period);
            USART_Print(", Counter=");
            USART_PrintNum(counter);
            USART_Print("\r\n");
            digitalWrite(led_pin, !digitalRead(led_pin));  // toggle LED ทุก wrap
        }
        last_counter = counter;

        // รายงานทุก 2 วิ
        if (ELAPSED_TIME(last_report, Get_CurrentMs()) >= 2000)
        {
            last_report = Get_CurrentMs();
            USART_Print("Counter=");
            USART_PrintNum(counter);
            USART_Print(", Period=");
            USART_PrintNum(current_period);
            USART_Print(", PSC=");
            USART_PrintNum(Simple_TIM_GetPrescaler(TIM_1));
            USART_Print("\r\n");

            // เปลี่ยน prescaler หลังจาก 5 วิ
            if (!freq_changed && (Get_CurrentMs() > 5000))
            {
                freq_changed = 1;
                uint16_t new_psc = 119;  // ความถี่ใหม่ 20kHz

                TIM_SetPrescaler(TIM_1, new_psc);
                TIM_SetMode(TIM_1, TIM_MODE_DOWN);  // เปลี่ยนเป็น DOWN mode
                TIM_GenerateUpdate(TIM_1);  // บังคับ reload ทันที

                uint32_t new_freq = SystemCoreClock / ((uint32_t)(new_psc + 1) * (period + 1));
                USART_Print("New Frequency: ");
                USART_PrintNum(new_freq);
                USART_Print(" Hz, Mode=DOWN\r\n");
            }
        }
    }
}
  • TIM_AdvancedInit(tim, psc, period, mode): ตั้งค่า timer ครบทุกพารามิเตอร์ — PSC, ARR, Mode
  • TIM_SetPrescaler: เปลี่ยน prescaler ขณะรัน — มีผลหลังจาก next update event
  • TIM_SetMode: เปลี่ยน UP/DOWN mode — ส่งผลทันทีหลังจาก GenerateUpdate
  • TIM_GenerateUpdate: บังคับให้ register reload ทันที — prescaler ใหม่生效
  • Frequency = SystemCoreClock / ((PSC+1) * (Period+1)) — สูตรคำนวณความถี่ timer

⚠️ จุดควรระวัง

  • 16-bit timer: PSC และ Period มีค่าสูงสุด 65535
  • ที่ 24MHz: ความถี่ต่ำสุด ~0.366Hz (PSC=65535, Period=65535), สูงสุด 24MHz (PSC=0, Period=0)
  • TIM_SetPrescaler() รอจนถึง next update event — ถ้าต้องการ生效ทันที เรียก TIM_GenerateUpdate()
  • ที่ 48MHz (HSE) ค่า PSC/Period ต้องปรับตาม

✅ ตรวจสอบก่อนรัน

  • Serial แสดง Counter, Period, PSC ทุก 2 วิ? หลังจาก 5 วิ PSC เปลี่ยน?
  • LED กระพริบทุกครั้งที่ counter wrap? (toggle)
  • Mode เปลี่ยนจาก UP เป็น DOWN หลัง 5 วิ?

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include <SimpleHAL.h>

static uint8_t btn_start_stop = PC0;
static uint8_t btn_reset = PC1;

int main(void)
{
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);

    pinMode(btn_start_stop, PIN_MODE_INPUT_PULLUP);
    pinMode(btn_reset, PIN_MODE_INPUT_PULLUP);

    Stopwatch_Init();                   // เริ่ม Stopwatch (TIM2 1000Hz)

    USART_Print("Stopwatch Example\r\n");
    USART_Print("PC0: Start/Stop\r\n");
    USART_Print("PC1: Reset\r\n");
    USART_Print("Status: Stopped\r\n");
    USART_Print("Time: 00:00:00\r\n");

    uint32_t last_display = 0;
    uint8_t button1_prev = HIGH;
    uint8_t button2_prev = HIGH;

    while (1)
    {
        uint8_t btn1_current = digitalRead(btn_start_stop);
        uint8_t btn2_current = digitalRead(btn_reset);

        // === Button1: Start/Stop state machine ===
        static uint8_t btn1_state = 0;
        static Timer_t btn1_timer;

        if (btn1_state == 0 && button1_prev == HIGH && btn1_current == LOW) {
            Start_Timer(&btn1_timer, 50, 0);
            btn1_state = 1;
        }
        if (btn1_state == 1 && Is_Timer_Expired(&btn1_timer)) {
            if (digitalRead(btn_start_stop) == LOW) {
                btn1_state = 2;
                if (Stopwatch_IsRunning()) {
                    Stopwatch_Stop();
                    USART_Print("Status: Paused\r\n");
                } else {
                    Stopwatch_Start();
                    USART_Print("Status: Running\r\n");
                }
            } else {
                btn1_state = 0;
            }
        }
        if (btn1_state == 2 && digitalRead(btn_start_stop) == HIGH) {
            btn1_state = 0;
        }

        // Button2: Reset (falling edge)
        if (button2_prev == HIGH && btn2_current == LOW) {
            Delay_Ms(50);  // blocking debounce
            if (digitalRead(btn_reset) == LOW) {
                Stopwatch_Reset();
                USART_Print("Status: Reset\r\n");
            }
        }

        button1_prev = btn1_current;
        button2_prev = btn2_current;

        // แสดงเวลาทุก 100ms
        if (ELAPSED_TIME(last_display, Get_CurrentMs()) >= 100)
        {
            last_display = Get_CurrentMs();
            char time_buf[32];
            Stopwatch_GetTimeString(time_buf, TIME_FORMAT_HHMMSS, TIME_DISPLAY_NORMALIZED);
            USART_Print("Time: ");
            USART_Print(time_buf);
            uint32_t total_sec = Stopwatch_GetTotalSeconds();
            USART_Print(" (");
            USART_PrintNum(total_sec);
            USART_Print(" sec)\r\n");
        }
    }
}
  • Stopwatch_Init(): เริ่มต้น stopwatch subsystem — ใช้ TIM2 ที่ 1000Hz (resolution 1ms)
  • Stopwatch_Start/Stop: เริ่ม/หยุดจับเวลา — ทำงานแบบ accumulate (ไม่ reset เมื่อ start ใหม่)
  • Stopwatch_Reset: ตั้งเวลาเป็น 00:00:00 — หยุดก่อน reset
  • Stopwatch_GetTimeString(buf, format, display): แปลงเวลาเป็น string — HH:MM:SS หรือ MM:SS.ms
  • Stopwatch_IsRunning(): เช็คสถานะ — ใช้ตัดสินใจ Start vs Stop
  • Stopwatch_GetTotalSeconds(): คืนค่า elapsed time เป็นวินาที (uint32_t)

⚠️ จุดควรระวัง

  • Stopwatch ใช้ TIM2 — ถ้ากำลังใช้ TIM2 กับ PWM หรือ timer interrupt อื่น → conflict
  • ไม่สามารถใช้ Stopwatch + Countdown (ex07) พร้อมกัน (ใช้ TIM2 ทั้งคู่)
  • ความละเอียด 1ms — เวลาสูงสุด ~49 วัน (overflow uint32_t ms)
  • ต้องเรียก Stopwatch_Init() ก่อนใช้ฟังก์ชันอื่น

✅ ตรวจสอบก่อนรัน

  • กด PC0 → "Running" → เวลาเพิ่มขึ้น?
  • กด PC0 อีกครั้ง → "Paused" → เวลาหยุด?
  • กด PC1 → "Reset" → เวลา 00:00:00?

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include <SimpleHAL.h>

static uint8_t led_alarm = PC0;
static uint8_t btn_start_pause = PC1;
static volatile uint8_t alarm_triggered = 0;  // flag alarm

void alarm_callback(void)
{
    alarm_triggered = 1;        // ตั้ง flag — main loop จะจัดการ
}

int main(void)
{
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);

    pinMode(led_alarm, PIN_MODE_OUTPUT);
    pinMode(btn_start_pause, PIN_MODE_INPUT_PULLUP);

    Countdown_Init(0, 0, 10);               // ตั้งเวลา 10 วินาที
    Countdown_SetAlarmCallback(alarm_callback);  // กำหนด callback

    USART_Print("Countdown Example\r\n");
    USART_Print("Countdown: 10 seconds\r\n");
    USART_Print("Press PC1 to Start/Pause\r\n");
    USART_Print("Status: Stopped\r\n");

    uint32_t last_report = 0;
    uint8_t button_prev = HIGH;

    while (1)
    {
        uint8_t btn_current = digitalRead(btn_start_pause);

        // Edge detection: HIGH → LOW
        if (button_prev == HIGH && btn_current == LOW)
        {
            Delay_Ms(50);  // blocking debounce
            if (digitalRead(btn_start_pause) == LOW)
            {
                if (Countdown_IsRunning()) {
                    Countdown_Stop();
                    USART_Print("Status: Paused\r\n");
                } else {
                    if (Countdown_IsFinished()) {
                        Countdown_Reset();       // reset เป็น 10s
                        alarm_triggered = 0;
                        digitalWrite(led_alarm, LOW);
                        USART_Print("Countdown Reset to 10s\r\n");
                    }
                    Countdown_Start();
                    USART_Print("Status: Running\r\n");
                }
            }
        }
        button_prev = btn_current;

        // รายงานทุก 1 วินาที
        if (Countdown_IsRunning()) {
            if (ELAPSED_TIME(last_report, Get_CurrentMs()) >= 1000) {
                last_report = Get_CurrentMs();
                uint32_t remaining = Countdown_GetRemainingSeconds();
                USART_Print("Remaining: ");
                USART_PrintNum(remaining);
                USART_Print(" seconds\r\n");
            }
        }

        // จัดการ alarm
        if (alarm_triggered) {
            alarm_triggered = 0;
            USART_Print("\r\n*** Time's up! ***\r\n");
            USART_Print("Countdown finished!\r\n");
            for (uint8_t i = 0; i < 10; i++) {  // LED กระพริบ 10 รอบ
                digitalWrite(led_alarm, HIGH);
                Delay_Ms(100);
                digitalWrite(led_alarm, LOW);
                Delay_Ms(100);
            }
            USART_Print("Press PC1 to reset and start again\r\n");
        }
    }
}
  • Countdown_Init(h, m, s): ตั้งเวลาเริ่มต้น — ชั่วโมง, นาที, วินาที
  • Countdown_Start/Stop: เริ่ม/หยุดนับถอยหลัง — Stop = Pause (เวลาเหลือ)
  • Countdown_IsRunning(): เช็คว่ากำลังนับถอยหลัง
  • Countdown_IsFinished(): เช็คว่านับถึง 0 แล้วหรือยัง
  • Countdown_Reset(): ตั้งเวลากลับเป็นค่า Init เดิม
  • Countdown_SetAlarmCallback: callback ถูกเรียกใน main loop context (ไม่ใช่ ISR) — ปลอดภัยกว่า
  • Countdown_GetRemainingSeconds(): คืนค่าวินาทีที่เหลือ

⚠️ จุดควรระวัง

  • Countdown ใช้ TIM2 (base timer) — conflict กับ Stopwatch (ex06) และ PWM2
  • Alarm callback ทำงานใน main loop context (ไม่ใช่ ISR) — ใช้ Delay_Ms ได้ แต่ไม่ควรนานเกิน
  • Countdown_Stop() = pause (เวลาเหลือ) — Countdown_Reset() ตั้งเวลาใหม่
  • เมื่อ countdown หมดแล้วกด Start → จะไม่เริ่ม ต้อง Reset ก่อน

  • LED กระพริบ 10 รอบเมื่อหมดเวลา?
  • กด PC1 ขณะนับ → "Paused"? กดอีก → ต่อ?
  • กด PC1 หลังจาก alarm → Reset → 10s พร้อมเริ่มใหม่?

ex08_Timer_Resource_Management — หลีกเลี่ยง TIM1/TIM2 Conflict

📖 แนวคิด

CH32V003 มี 2 hardware timers: TIM1 (ใช้โดย PWM1) และ TIM2 (ใช้โดย PWM2/Stopwatch/Countdown) — ถ้า TIM1 ถูก PWM ใช้แล้ว ต้องใช้ SysTick-based timer (Timer_t) แทน — ตัวอย่างนี้ TIM1 → PWM fade, SysTick → 3 non-blocking timers, TIM2 = ว่าง

     TIM1 → PD2 (PWM1_CH1) → LED fade (hardware PWM)

     PC0 → LED toggle (software — SysTick timer)

     USART (TX=PD5, RX=PD6) → แสดงสถานะ

     (ไม่ใช้ TIM2 เลย — หลีกเลี่ยง conflict)

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include "SimpleHAL.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
    pinMode(PC0, PIN_MODE_OUTPUT);

    // TIM1 → PWM LED fade (ใช้ TIM1 hardware)
    PWM_Init(PWM1_CH1, 1000);  // PD2, 1kHz
    PWM_Start(PWM1_CH1);

    // SysTick → Non-blocking timers (ไม่ใช้ TIM1/TIM2 เลย!)
    Timer_t print_timer;
    Timer_t led_timer;
    Timer_t pwm_timer;

    Start_Timer(&print_timer, 2000, 1);  // ทุก 2 วิ — แจ้งสถานะ
    Start_Timer(&led_timer, 500, 1);     // ทุก 0.5 วิ — LED toggle
    Start_Timer(&pwm_timer, 50, 1);      // ทุก 50ms — PWM step

    uint8_t pwm_val = 0;
    int8_t pwm_dir = 5;

    USART_Print("\r\n=== Timer Resource Management ===\r\n");
    USART_Print("TIM1 = PWM (PD2 fade)\r\n");
    USART_Print("SysTick = 3 non-blocking timers\r\n");
    USART_Print("TIM2 = free (available for user)\r\n\r\n");

    while (1) {
        // PWM fade — ใช้ SysTick timing แทน TIM2 interrupt
        if (Is_Timer_Expired(&pwm_timer)) {
            pwm_val += pwm_dir;
            if (pwm_val >= 100 || pwm_val <= 0) pwm_dir = -pwm_dir;
            PWM_SetDutyCycle(PWM1_CH1, pwm_val);
        }

        // LED toggle
        if (Is_Timer_Expired(&led_timer)) {
            digitalToggle(PC0);
        }

        // Status print
        if (Is_Timer_Expired(&print_timer)) {
            USART_Print("PWM=");
            USART_PrintNum(pwm_val);
            USART_Print("%  TIM2=free  SysTick=OK\r\n");
        }

        // yield() — เรียก IWDG feed ถ้าใช้ IWDG
        yield();
    }
}
  • Timer allocation: TIM1 → PWM, SysTick → general timing, TIM2 = free
  • 3 Timer_t instances: print_timer (2s), led_timer (500ms), pwm_timer (50ms) — ทั้งหมดใช้ SysTick
  • PWM fade: ปรับ duty cycle ทุก 50ms — เพิ่ม 5% → 100% → ลด → 0% วนไป
  • digitalToggle(PC0): สลับ LED ทุก 500ms — โชว์ว่า SysTick timer ทำงานปกติ
  • yield(): เรียก IWDG_Feed ถ้าเปิด IWDG ไว้ — ป้องกัน watchdog reset

⚠️ จุดควรระวัง

  • TIM1 สงวนสำหรับ SimplePWM (PWM1_CH1-CH4) — ใช้ TIM2 สำหรับ user timer ถ้าต้องการ hardware timer
  • TIM2 สงวนสำหรับ SimplePWM (PWM2_CH1-CH4), Stopwatch และ Countdown — ใช้พร้อมกันไม่ได้
  • SysTick-based timer (Timer_t) แม่นยำ ±1ms — ไม่มี hardware timer drift
  • Timer_t อาศัย SysTick interrupt — ถ้า SysTick ถูก Disable → timer หยุดทำงาน

  • PD2 (PWM1_CH1) ต่อ LED — สว่าง fade ขึ้นลง?
  • PC0 กระพริบทุก 500ms?
  • Serial แสดง "PWM=xx% TIM2=free SysTick=OK" ทุก 2 วิ?
  • TIM2 ไม่ถูกใช้ — ว่างให้ผู้ใช้
