📝 เฉลย — W25Qxx SPI Flash

ข้อ 1: Data Logger — บันทึก 1000 readings แบบ append

#define FLASH_LOG_SECTOR   (4 * 4096)  // sector 4
#define MAX_READINGS       1000
#define READING_SIZE       8

typedef struct {
    uint32_t timestamp;
    int16_t  temp;     // ×10 (°C)
    uint16_t hum;      // ×10 (%)
} sensor_reading_t;

uint16_t log_index = 0;   // จำนวนที่บันทึกแล้ว

void flash_logger_init(void) {
    W25Q_Init();

    // อ่าน index จาก address สุดท้ายของ sector
    W25Q_Read(FLASH_LOG_SECTOR + MAX_READINGS * READING_SIZE,
              (uint8_t *)&log_index, 2);

    if(log_index > MAX_READINGS) log_index = 0;

    USART_Print("Logger: ");
    USART_PrintNum(log_index);
    USART_Println(" existing readings");
}

void flash_log_append(int16_t temp, uint16_t hum) {
    if(log_index >= MAX_READINGS) {
        USART_Println("[LOG] Full — erase and restart");
        W25Q_SectorErase(FLASH_LOG_SECTOR);
        log_index = 0;
    }

    sensor_reading_t r;
    r.timestamp = Timer_GetMs() / 1000;
    r.temp = temp;
    r.hum = hum;

    uint32_t addr = FLASH_LOG_SECTOR + log_index * READING_SIZE;
    W25Q_Write(addr, (uint8_t *)&r, READING_SIZE);

    // บันทึก index
    log_index++;
    W25Q_Write(FLASH_LOG_SECTOR + MAX_READINGS * READING_SIZE,
               (uint8_t *)&log_index, 2);

    USART_Print("Logged #");
    USART_PrintNum(log_index);
    USART_Print(" @0x");
    USART_PrintHex(addr);
    USART_Println("");
}

void flash_logger_loop(void) {
    flash_logger_init();

    uint32_t last_log = 0;
    while(1) {
        if(Timer_GetMs() - last_log > 1000) {
            int16_t t = (int16_t)(DHT_ReadTemp() * 10);
            uint16_t h = (uint16_t)(DHT_ReadHum() * 10);
            flash_log_append(t, h);
            last_log = Timer_GetMs();
        }
        Delay_Ms(100);
    }
}
🧠 แนวคิด: W25Qxx ต้อง erase ก่อนเขียน (sector=4KB, block=32/64KB) — ใช้ sector 4 สำหรับ data log — เขียนแบบ append: index บอกจำนวน record, เขียนต่อท้ายไปเรื่อย ๆ — เก็บ index ไว้หลังข้อมูล (address=FLASH_LOG_SECTOR + 1000*8) — เมื่อเต็ม 1000 records → erase sector แล้วเริ่มใหม่ — 8 bytes/record = 8000 bytes ใช้ 2 sectors — W25Qxx เขียนได้เฉพาะ bit 1→0, ต้อง erase เพื่อ reset เป็น 1

ข้อ 3: Sector Erase + Rewrite — จัดการข้อมูลเก่า

#define CONFIG_SECTOR    (8 * 4096)  // sector 8, 4KB
#define CONFIG_MAGIC     0xBEEF

typedef struct {
    uint16_t magic;
    uint16_t version;
    uint8_t  dev_name[16];
    uint8_t  baudrate_idx;
    uint8_t  i2c_addr;
    uint8_t  padding[10];  // align 32 bytes
} flash_config_t;

flash_config_t flash_cfg;

void flash_config_load(void) {
    W25Q_Init();
    W25Q_Read(CONFIG_SECTOR, (uint8_t *)&flash_cfg,
              sizeof(flash_cfg));

    if(flash_cfg.magic != CONFIG_MAGIC) {
        USART_Println("[CONFIG] No config — load defaults");
        strcpy((char *)flash_cfg.dev_name, "CH32V003");
        flash_cfg.baudrate_idx = 7;  // 115200
        flash_cfg.i2c_addr = 0x68;
        flash_cfg.version = 1;
        flash_cfg.magic = CONFIG_MAGIC;
        flash_config_save();
    }
    USART_Print("Config v");
    USART_PrintNum(flash_cfg.version);
    USART_Print(" loaded: ");
    USART_Println((char *)flash_cfg.dev_name);
}

void flash_config_save(void) {
    flash_cfg.magic = CONFIG_MAGIC;
    flash_cfg.version++;

    USART_Println("[CONFIG] Erasing sector...");
    W25Q_SectorErase(CONFIG_SECTOR);
    Delay_Ms(100);  // รอ erase จริง (W25Qxx ~30-100ms)

    USART_Println("[CONFIG] Writing new config...");
    W25Q_Write(CONFIG_SECTOR, (uint8_t *)&flash_cfg,
               sizeof(flash_cfg));

    USART_Println("[CONFIG] Saved OK");
}

void flash_sector_verify(void) {
    USART_Println("Verifying...");
    flash_config_t verify;

    W25Q_Read(CONFIG_SECTOR, (uint8_t *)&verify,
              sizeof(verify));

    if(memcmp(&flash_cfg, &verify, sizeof(flash_cfg)) == 0)
        USART_Println("✓ Verify PASS");
    else
        USART_Println("✗ Verify FAIL");
}
🧠 แนวคิด: W25Qxx ต้อง erase ทั้ง sector ก่อนเขียนใหม่ — sector erase ใช้เวลา 30-100ms (ขึ้นกับรุ่น) — หลัง erase ทุก byte เป็น 0xFF — เขียน config ใหม่หลัง erase — magic number ป้องกันอ่านค่า random ตอน Flash ว่าง — version field ตามการเปลี่ยนแปลง — verify อ่านกลับมาเทียบว่าเขียนถูกต้อง — sector เดียวเก็บ config ได้หลายเวอร์ชันโดยใช้ offset ต่างกัน
