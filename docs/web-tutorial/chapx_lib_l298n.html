L298N เป็น Dual H-Bridge — ขับมอเตอร์ DC ได้ 2 ตัว (Motor A, Motor B) หรือ Stepper 1 ตัว — รับแรงดัน VMS 5V–35V, กระแส 2A ต่อช่อง — ใช้ PWM ที่ ENA/ENB ปรับความเร็ว — ต่อ GND ของ MCU, L298N logic, และ Motor Supply รวมกัน

รู้จักขา

  • IN1/IN2: ควบคุมทิศทาง Motor A (HIGH/LOW)
  • IN3/IN4: ควบคุมทิศทาง Motor B
  • ENA: PWM ปรับความเร็ว Motor A
  • ENB: PWM ปรับความเร็ว Motor B
  • OUT1/OUT2: ขาต่อ Motor A
  • OUT3/OUT4: ขาต่อ Motor B
  • VMS: จ่ายไฟให้มอเตอร์ 5–35V
  • VCC/VSS: จ่ายไฟ logic 5V
  • GND: Ground ร่วม

วิธีต่อทีละขั้น

  1. ต่อ IN1-IN4 ไป GPIO (PC0-PC3) — ควบคุมทิศทาง
  2. ต่อ ENA/ENB ไป GPIO พร้อม PWM (PD2/PD3) — ปรับความเร็ว
  3. ต่อ VMS กับ แหล่งจ่าย 5–35V — จ่ายไฟมอเตอร์
  4. ต่อ VCC/VSS ไป 5V และ GND ไป GND — จ่ายไฟ logic
  5. ต่อ มอเตอร์ DC เข้ากับ OUT1/OUT2 (Motor A) และ OUT3/OUT4 (Motor B)

📚 หลักการทำงาน

H-Bridge คือวงจรสวิตช์ 4 ตัว (2×High-side + 2×Low-side) ต่อเป็นรูปตัว H — ควบคุมทิศทางกระแสที่ไหลผ่านมอเตอร์ → หมุนเดินหน้า/ถอยหลัง

4 States: (1) Forward — IN1=HIGH, IN2=LOW → กระแสไหล A→B; (2) Reverse — IN1=LOW, IN2=HIGH → กระแสไหล B→A; (3) Brake — IN1=IN2=HIGH → ลัดวงจรขั้วมอเตอร์ → หยุดเร็ว; (4) Coast — IN1=IN2=LOW → ปล่อยให้มอเตอร์หมุนอิสระ

PWM Speed Control: ปรับ duty cycle ที่ ENA pin — 0% = หยุด, 100% = สุด — ใช้ PWM_SimpleInit() ที่ ~1-20kHz (เหนือเสียง) เพื่อไม่ให้มีเสียงหวีด

2 Channels อิสระ: Motor A ใช้ IN1,IN2,ENA; Motor B ใช้ IN3,IN4,ENB — ควบคุมแยกกันได้ทั้งทิศทางและความเร็ว

ขั้นที่ 2: Forward/Reverse/Brake/Coast

    void motor_forward(uint8_t speed) {
        L298N_SetDirection(&motor_a, L298N_FWD);
        L298N_SetSpeed(&motor_a, speed);  // 0-100%
        L298N_Run(&motor_a);
    }
    void motor_reverse(uint8_t speed) {
        L298N_SetDirection(&motor_a, L298N_REV);
        L298N_SetSpeed(&motor_a, speed);
        L298N_Run(&motor_a);
    }
    void motor_stop(void) {
        L298N_Stop(&motor_a);
    }

🔍 L298N_SetDirection(&motor, dir) — ตั้ง FWD/REV — L298N_Run(&motor) เริ่มหมุน — L298N_SetSpeed(&motor, pct) ปรับความเร็ว — L298N_Stop(&motor) หยุด

ขั้นที่ 3: Speed Ramp + Loop Demo

    while(1) {
        // Motor A: ramp up forward
        for(uint8_t s = 0; s <= 100; s += 5) {

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
มอเตอร์ไม่หมุนเลยลืมต่อ VMS (Motor Supply) หรือไม่ได้เรียก L298N_Runตรวจสอบ 12V/VMS ขั้ว L298N — L298N_Run(&motor) — วัดไฟที่ OUT1/OUT2 ด้วย multimeter
มอเตอร์หมุนทิศทางเดียวเสมอIN1/IN2 สลับ — หรือต่อสายมอเตอร์กลับขั้วตรวจสอบ logic: IN1=1,IN2=0 = Forward — ลองสลับสายมอเตอร์ที่ OUT1/OUT2
L298N ร้อนมาก (thermal shutdown)มอเตอร์กินกระแสเกิน 2A — ไม่มี heatsinkใช้ heatsink + fan — หรือเปลี่ยนเป็น driver ที่ทนกระแสสูงกว่าเช่น BTS7960
ความเร็วไม่ linear — 0-30% ไม่ขยับDeadband ของมอเตอร์ — ต้องใช้แรงดันขั้นต่ำก่อนหมุนใช้ lookup table ปรับ throttle curve — หรือใช้ closed-loop PID
PWM ทำให้เสียงหวีดใช้ความถี่ PWM ในช่วงการได้ยิน (< 8kHz)ตั้ง PWM frequency ≥ 10kHz — อยู่เหนือย่านการได้ยินของมนุษย์

🛠️ สร้าง Product: 2-Wheel Differential Drive Robot Base

มอเตอร์ 2 ตัว L298N + Bluetooth Control — Forward/Reverse/Left/Right ด้วย PWM speed แยกซ้าย-ขวา

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "L298N.h"

L298N_Instance motor_left, motor_right;

void drive(int16_t left, int16_t right) {
    if(left > 0) {
        L298N_SetDirection(&motor_left, L298N_FWD);
        L298N_SetSpeed(&motor_left, left);
    } else if(left < 0) {
        L298N_SetDirection(&motor_left, L298N_REV);
        L298N_SetSpeed(&motor_left, -left);
    } else { L298N_Stop(&motor_left); }
    L298N_Run(&motor_left);

    if(right > 0) {
        L298N_SetDirection(&motor_right, L298N_FWD);
        L298N_SetSpeed(&motor_right, right);
    } else if(right < 0) {
        L298N_SetDirection(&motor_right, L298N_REV);
        L298N_SetSpeed(&motor_right, -right);
    } else { L298N_Stop(&motor_right); }
    L298N_Run(&motor_right);
}

int main(void) {
    SystemCoreClockUpdate();
    USART1_Init(115200);
    L298N_Init(&motor_left, PC0, PC1, PWM1_CH4);
    L298N_Init(&motor_right, PC2, PC3, PWM1_CH3);

    while(1) {
        if(USART1_Available()) {
            char cmd = USART1_Read();
            switch(cmd) {
                case 'F': drive(80, 80); break;
                case 'B': drive(-80, -80); break;
                case 'L': drive(-50, 50); break;
                case 'R': drive(50, -50); break;
                case 'S': drive(0, 0); break;
            }
        }
    }
}
