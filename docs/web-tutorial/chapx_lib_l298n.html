L298N — DC Motor H-Bridge Driver

ระดับกลาง

🧭 แผนภาพการทำงาน

flowchart TD
    A["Motor_Init(IN1,IN2,ENA)"] --> B{"Command"}
    B -->|"Forward"| C["IN1=HIGH, IN2=LOW, ENA=PWM"]
    B -->|"Reverse"| D["IN1=LOW, IN2=HIGH, ENA=PWM"]
    B -->|"Brake"| E["IN1=HIGH, IN2=HIGH"]
    B -->|"Coast"| F["IN1=LOW, IN2=LOW, ENA=0"]

    F --> J["Motor coasts freely"]

    H --> B
    I --> B
    J --> B

L298N เป็น Dual H-Bridge — ขับมอเตอร์ DC ได้ 2 ตัว (Motor A, Motor B) หรือ Stepper 1 ตัว — รับแรงดัน VMS 5V–35V, กระแส 2A ต่อช่อง — ใช้ PWM ที่ ENA/ENB ปรับความเร็ว — ต่อ GND ของ MCU, L298N logic, และ Motor Supply รวมกัน

📚 หลักการทำงาน

H-Bridge คือวงจรสวิตช์ 4 ตัว (2×High-side + 2×Low-side) ต่อเป็นรูปตัว H — ควบคุมทิศทางกระแสที่ไหลผ่านมอเตอร์ → หมุนเดินหน้า/ถอยหลัง

4 States: (1) Forward — IN1=HIGH, IN2=LOW → กระแสไหล A→B; (2) Reverse — IN1=LOW, IN2=HIGH → กระแสไหล B→A; (3) Brake — IN1=IN2=HIGH → ลัดวงจรขั้วมอเตอร์ → หยุดเร็ว; (4) Coast — IN1=IN2=LOW → ปล่อยให้มอเตอร์หมุนอิสระ

PWM Speed Control: ปรับ duty cycle ที่ ENA pin — 0% = หยุด, 100% = สุด — ใช้ PWM_SimpleInit() ที่ ~1-20kHz (เหนือเสียง) เพื่อไม่ให้มีเสียงหวีด

2 Channels อิสระ: Motor A ใช้ IN1,IN2,ENA; Motor B ใช้ IN3,IN4,ENB — ควบคุมแยกกันได้ทั้งทิศทางและความเร็ว

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Init Motor Channels + PWM

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"

#define IN1 PC0
#define IN2 PC1
#define ENA PD2  // PWM channel 1
#define IN3 PC2
#define IN4 PC3
#define ENB PD3  // PWM channel 2

int main(void) {
    SystemCoreClockUpdate();
    // Motor A
    pinMode(IN1, PIN_MODE_OUTPUT);
    pinMode(IN2, PIN_MODE_OUTPUT);
    PWM_SimpleInit(1, ENA, 10000);  // 10kHz
    PWM_Start(1);
    // Motor B
    pinMode(IN3, PIN_MODE_OUTPUT);
    pinMode(IN4, PIN_MODE_OUTPUT);
    PWM_SimpleInit(2, ENB, 10000);
    PWM_Start(2);

🔍 10kHz PWM — หูคนไม่ได้ยิน — 2 channels อิสระ — แต่ละ channel มี IN pins 2 pins + ENA PWM pin

ขั้นที่ 2: Forward/Reverse/Brake/Coast

    void motor_a_forward(uint8_t speed) {
        digitalWrite(IN1, HIGH);
        digitalWrite(IN2, LOW);
        PWM_SetDuty(1, speed);  // 0-100%
    }
    void motor_a_reverse(uint8_t speed) {
        digitalWrite(IN1, LOW);
        digitalWrite(IN2, HIGH);
        PWM_SetDuty(1, speed);
    }
    void motor_a_brake() {
        digitalWrite(IN1, HIGH);
        digitalWrite(IN2, HIGH);  // Short brake
        PWM_SetDuty(1, 0);
    }
    void motor_a_coast() {
        digitalWrite(IN1, LOW);
        digitalWrite(IN2, LOW);   // Free spin
        PWM_SetDuty(1, 0);
    }

🔍 Forward: IN1 high, IN2 low + PWM — Reverse: สลับ logic — Brake: ทั้งคู่ high (มอเตอร์กลายเป็น generator, พลังงานถูก short → หยุดเร็ว) — Coast: ทั้งคู่ low (ปล่อยอิสระ)

ขั้นที่ 3: Speed Ramp + Loop Demo

    while(1) {
        // Motor A: ramp up forward
        for(uint8_t s = 0; s <= 100; s += 5) {
            motor_a_forward(s);
            Delay_Ms(100);
        }
        Delay_Ms(1000);
        motor_a_brake();  // หยุดแบบ brake
        Delay_Ms(2000);

        // Motor A: ramp up reverse
        for(uint8_t s = 0; s <= 80; s += 5) {
            motor_a_reverse(s);
            Delay_Ms(100);
        }
        Delay_Ms(1000);
        motor_a_coast();  // ปล่อย coast
        Delay_Ms(3000);

        // Motor B: ไปพร้อมกัน
        motor_a_forward(60);
        motor_b_forward(30);  // ต้อง implement คล้าย motor_a
        Delay_Ms(3000);
        motor_a_brake();
        motor_b_brake();
        Delay_Ms(2000);
    }
}

🔍 Ramp speed ขึ้นทีละ 5% — ป้องกันไฟกระชาก — Brake vs Coast ใช้ต่างสถานการณ์: brake ตอนต้องการหยุดแม่น, coast ตอนปล่อยอิสระ (เช่น robot coasting)

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
มอเตอร์ไม่หมุนเลยลืมต่อ VMS (Motor Supply) หรือ ENA PWM ไม่ startตรวจสอบ 12V/VMS ขั้ว L298N — PWM_Start(ch) — วัดไฟที่ OUT1/OUT2 ด้วย multimeter
มอเตอร์หมุนทิศทางเดียวเสมอIN1/IN2 สลับ — หรือต่อสายมอเตอร์กลับขั้วตรวจสอบ logic: IN1=1,IN2=0 = Forward — ลองสลับสายมอเตอร์ที่ OUT1/OUT2
L298N ร้อนมาก (thermal shutdown)มอเตอร์กินกระแสเกิน 2A — ไม่มี heatsinkใช้ heatsink + fan — หรือเปลี่ยนเป็น driver ที่ทนกระแสสูงกว่าเช่น BTS7960
ความเร็วไม่ linear — 0-30% ไม่ขยับDeadband ของมอเตอร์ — ต้องใช้แรงดันขั้นต่ำก่อนหมุนใช้ lookup table ปรับ throttle curve — หรือใช้ closed-loop PID
PWM ทำให้เสียงหวีดใช้ความถี่ PWM ในช่วงการได้ยิน (< 8kHz)ตั้ง PWM frequency ≥ 10kHz — อยู่เหนือย่านการได้ยินของมนุษย์

🛠️ สร้าง Product: 2-Wheel Differential Drive Robot Base

มอเตอร์ 2 ตัว L298N + Bluetooth Control — Forward/Reverse/Left/Right ด้วย PWM speed แยกซ้าย-ขวา

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"

void motor_left(int16_t speed) {  // -100..0..+100
    if(speed > 0) { digitalWrite(PC0,HIGH); digitalWrite(PC1,LOW); PWM_SetDuty(1,speed); }
    else if(speed < 0) { digitalWrite(PC0,LOW); digitalWrite(PC1,HIGH); PWM_SetDuty(1,-speed); }
    else { digitalWrite(PC0,LOW); digitalWrite(PC1,LOW); PWM_SetDuty(1,0); }
}
void motor_right(int16_t speed) {
    if(speed > 0) { digitalWrite(PC2,LOW); digitalWrite(PC3,HIGH); PWM_SetDuty(2,speed); }
    else if(speed < 0) { digitalWrite(PC2,HIGH); digitalWrite(PC3,LOW); PWM_SetDuty(2,-speed); }
    else { digitalWrite(PC2,LOW); digitalWrite(PC3,LOW); PWM_SetDuty(2,0); }
}

int main(void) {
    SystemCoreClockUpdate();
    USART1_Init(115200);  // Bluetooth HC-05
    pinMode(PC0,PIN_MODE_OUTPUT); pinMode(PC1,PIN_MODE_OUTPUT);
    pinMode(PC2,PIN_MODE_OUTPUT); pinMode(PC3,PIN_MODE_OUTPUT);
    PWM_SimpleInit(1,PD2,10000); PWM_Start(1);
    PWM_SimpleInit(2,PD3,10000); PWM_Start(2);

    while(1) {
        if(USART1_Available()) {
            char cmd = USART1_Read();
            switch(cmd) {
                case 'F': motor_left(80); motor_right(80); break;
                case 'B': motor_left(-80); motor_right(-80); break;
                case 'L': motor_left(-50); motor_right(50); break;
                case 'R': motor_left(50); motor_right(-50); break;
                case 'S': motor_left(0); motor_right(0); break;
            }
        }
    }
}
