RotaryEncoder — ตัวหมุนเข้ารหัส

ระดับกลาง

🔌 การต่อวงจร

    Rotary Encoder Module (KY-040)
┌────────────────────────────────────┐
│ MCU                         CH32V003│
│ PC0 ─── CLK  (A)                   │
│ PC1 ─── DT   (B)                   │
│ PC2 ─── SW   (Button)              │
│ 3.3V ── VCC                        │
│ GND  ── GND                        │
│                                    │
│ ใช้ internal pull-up ทั้ง 3 ขา      │
│ 20 pulses/rev (typical)            │
└────────────────────────────────────┘

Rotary Encoder มาตรฐานมี 3 ขา (CLK, DT, SW) — CLK/DT สร้างสัญญาณ quadrature — SW เป็นปุ่มในตัว

📚 หลักการทำงาน

Quadrature Decoding: CLK และ DT มีเฟสห่างกัน 90° — เมื่อหมุนตามเข็ม (CW) CLK จะนำ DT — เมื่อหมุนทวนเข็ม (CCW) DT จะนำ CLK — ตรวจจับ edge การเปลี่ยนแปลงเพื่อนับ step

RotaryEncoder Struct: RotaryEncoder encoder; — Rotary_Init(&encoder, CLK, DT, SW) — ใช้ polling โดยเรียก Rotary_GetPosition(&encoder)

Callback System: Rotary_SetCallback(&encoder, ROTARY_EVENT_ROTATE, my_callback) — callback รับ RotaryEncoder* e — ภายใน callback อ่านตำแหน่งด้วย Rotary_GetPosition(e)

Debounce: Rotary_SetDebounceTime(&encoder, 30) — ตั้งค่า debounce ในหน่วย ms — default 30ms เหมาะกับ encoder ทั่วไป

ไม่มี Interrupt Mode แยก: API ใช้ polling ผ่าน Rotary_GetPosition — ไม่มี Rotary_InitInt — ไม่มี acceleration

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Init + Set Callbacks

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "RotaryEncoder.h"

RotaryEncoder encoder;
int32_t position = 0;

void on_rotate(RotaryEncoder* e) {
    position = Rotary_GetPosition(e);
    USART_Print("Pos: "); USART_PrintNum(position); USART_Println("");
}

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_Init(115200);

    Rotary_Init(&encoder, PC0, PC1, PC2);
    Rotary_SetDebounceTime(&encoder, 30);
    Rotary_SetCallback(&encoder, ROTARY_EVENT_ROTATE, on_rotate);
    USART_Println("Encoder Ready!");

🔍 Rotary_Init(&encoder, CLK, DT, SW) — ถ้าไม่มีปุ่ม ส่ง NO_PIN แทน SW — Rotary_SetDebounceTime() ตั้ง debounce — Rotary_SetCallback() ใช้ event type — callback รับ RotaryEncoder* e

ขั้นที่ 2: Loop (Polling Mode)

    while(1) {
        // Callback ถูกเรียกอัตโนมัติเมื่อมีการหมุน
        // หรือ poll ตำแหน่งเอง:
        int32_t pos = Rotary_GetPosition(&encoder);
        IWDG_Feed();
    }
}

🔍 Rotary_GetPosition(&encoder) อ่านตำแหน่งสะสม — callback จะถูกเรียกอัตโนมัติเมื่อหมุน — สามารถ poll ตำแหน่งเองใน loop ได้

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
หมุน 1 step แต่ค่าขึ้น 2-3Bounce หรือ noise บนสาย CLK/DTใช้ capacitor 100nF คร่อม CLK-GND และ DT-GND — เพิ่ม debounce time
พลาด step เมื่อหมุนเร็วPoll rate ไม่พอใช้ callback แทน — callback จะถูกเรียกอัตโนมัติเมื่อหมุน
ทิศทางกลับด้าน (CW ขึ้นเป็นลบ)สลับสาย CLK กับ DTสลับขา CLK/DT ในโค้ด
Callback ไม่ทำงานลืม Rotary_SetCallback()ต้องตั้ง callback ด้วย event type ที่ถูกต้อง
ปุ่มไม่ทำงานSW pin ต่อผิด หรือลืมตั้ง callback สำหรับปุ่มเช็ค SW pin mapping — ตั้ง callback แยกสำหรับปุ่ม

🛠️ สร้าง Product: Volume Control + PWM

Rotary Encoder ควบคุม PWM duty cycle (0-100%) — หมุน=ปรับ, กด=mute toggle

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "RotaryEncoder.h"

RotaryEncoder encoder;
int32_t volume = 50;  // 0-100
uint8_t muted = 0;

void on_rotate(RotaryEncoder* e) {
    int32_t pos = Rotary_GetPosition(e);
    if(!muted) {
        volume = (pos % 101 + 101) % 101;  // clamp 0-100
        PWM_SetDuty(TIM2_CH1, (uint8_t)volume);
        USART_Print("Vol: "); USART_PrintNum((uint32_t)volume); USART_Println("%");
    }
}
void on_button(RotaryEncoder* e) {
    muted = !muted;
    PWM_SetDuty(TIM2_CH1, muted ? 0 : (uint8_t)volume);
    USART_Println(muted ? "MUTED" : "UNMUTED");
}

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_Init(115200);
    PWM_Init(TIM2_CH1, PC4, 10000, 50);
    Rotary_Init(&encoder, PC0, PC1, PC2);
    Rotary_SetDebounceTime(&encoder, 30);
    Rotary_SetCallback(&encoder, ROTARY_EVENT_ROTATE, on_rotate);
    USART_Println("=== Volume Knob ===");
    while(1) { IWDG_Feed(); }
}

✏️ ลองทำดู

  2. Multi-turn Counter — นับตำแหน่งแบบ absolute หมุนกี่รอบก็ได้ — แสดงจำนวนรอบ
  3. Game Paddle — หมุนเพื่อขยับ paddle ในเกม Pong — สื่อสารผ่าน USART
