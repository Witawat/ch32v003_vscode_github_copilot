Rotary Encoder มาตรฐานมี 3 ขา (CLK, DT, SW) บวก VCC/GND — CLK และ DT สร้างสัญญาณ quadrature — ขา SW เป็นปุ่มในตัว

📚 หลักการทำงาน

Quadrature Decoding: CLK และ DT มีเฟสห่างกัน 90° — เมื่อหมุนตามเข็ม (CW) CLK จะนำ DT — เมื่อหมุนทวนเข็ม (CCW) DT จะนำ CLK — ตรวจจับ edge การเปลี่ยนแปลงเพื่อนับ step

Step Counting: 1 รอบหมุนมี 20-24 steps (pulses) — ไลบรารีคืนค่า +1 หรือ -1 ต่อ step — ผู้ใช้สะสมค่าเองเพื่อทำตำแหน่ง

Interrupt vs Polling: Interrupt mode (RotaryEncoder_InitInt) ใช้ GPIO interrupt จับ edge — ไม่ต้องเรียก Update() ใน loop — เหมาะกับหมุนเร็วหรือ CPU busy — Polling mode ต้องเรียก Rotary_Update() บ่อยๆ

Acceleration: เมื่อหมุนเร็ว ไลบรารีจะคูณ step เพื่อให้ขยับไวขึ้น — เช่น หมุนช้า step=1, หมุนเร็ว step=4 — threshold และ multiplier ปรับได้

Button Press: ปุ่มในตัว rotary encoder (กดลงไป) — ใช้ debounce 50ms — callback แยกออกจาก rotate — มักใช้เป็น Enter หรือ Reset

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Init + Set Callbacks

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "RotaryEncoder.h"

int32_t position = 0;

void on_rotate(int8_t dir) {
    position += dir;
    USART_Print("Pos: "); USART_PrintNum(position); USART_Println("");
}
void on_button(void) {
    position = 0;
    USART_Println("Reset!");
}

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_Init(115200);

    Rotary_Init(PC0, PC1, PC2);  // CLK, DT, SW
    Rotary_SetCallback(on_rotate);
    Rotary_SetCallback(on_button);
    USART_Println("Encoder Ready!");

🔍 Rotary_Init(CLK, DT, SW) — ถ้าไม่มีปุ่มให้ส่ง NO_PIN แทน SW — Rotary_SetCallback(cb) รับ callback ที่มี parameter int8_t dir (+1 หรือ -1)

ขั้นที่ 2: Loop (Polling Mode)

    while(1) {
        Rotary_Update();
        IWDG_Feed();
    }
}

🔍 Polling mode ต้องเรียก Rotary_Update() ใน main loop ทุกรอบ — ต้องเรียกบ่อยพอ (อย่างน้อยทุก 1ms) ไม่เช่นนั้นจะพลาด step — ถ้า main loop ยุ่งมากให้ใช้ interrupt mode แทน

ขั้นที่ 3: Interrupt Mode + Acceleration

    // แทน Init+Update ด้วย InitInt — ไม่ต้อง poll
    RotaryEncoder_InitInt(PC0, PC1, PC2);
    Rotary_SetCallback(on_rotate);
    Rotary_SetCallback(on_button);

    // ตั้งค่า acceleration
    RotaryEncoder_SetAccel(2, 4);  // ถ้าหมุน >2 steps/50ms → คูณ 4

    while(1) {
        // ไม่ต้องเรียก Update() — interrupt จัดการให้
        IWDG_Feed();
    }
}

🔍 RotaryEncoder_InitInt() ใช้ GPIO interrupt — CPU ไม่ต้อง poll — SetAccel(threshold, multiplier) ทำให้หมุนเร็วขยับไว — callback ถูกเรียกจาก ISR โดยตรง ดังนั้นต้องสั้นและไม่ใช้ Delay

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
หมุน 1 step แต่ค่าขึ้น 2-3Bounce หรือ noise บนสาย CLK/DTใช้ capacitor 100nF คร่อม CLK-GND และ DT-GND — หรือใช้ HW debounce RC
พลาด step เมื่อหมุนเร็ว (Polling)Update() ไม่ถูกเรียกถี่พอเปลี่ยนเป็น Interrupt mode (InitInt) — หรือลดงานใน main loop
ทิศทางกลับด้าน (CW ขึ้นเป็นลบ)สลับสาย CLK กับ DTสลับขา CLK/DT ในโค้ด หรือป้อนค่าลบให้ callback
Callback ทำงานใน ISR แล้ว delay ไม่ได้Callback ถูกเรียกจาก interruptCallback ต้องสั้น — ถ้าต้องการ delay ให้ set flag แล้ว main loop จัดการ
ปุ่มไม่ทำงานลืม Rotary_SetCallback() — หรือ SW ต่อผิดขาเช็ค SW pin mapping — ใช้ digitalRead(SW) ทดสอบว่าอ่านค่าถูก
Step หายเมื่อหมุนกลับไปกลับมาเร็วMCU จับ edge ไม่ทัน — polling rate ช้าใช้ Interrupt mode + เพิ่มความสำคัญของ ISR — หรือใช้ hardware quadrature decoder

🛠️ สร้าง Product: Volume Control + PWM Audio

Rotary Encoder ควบคุม PWM duty cycle (0-100%) สำหรับ LED/motor — หมุน=ปรับ, กด=mute toggle — แสดงผล OLED

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "RotaryEncoder.h"
#include "PWM.h"

uint8_t volume = 50;  // 0-100%
uint8_t muted = 0;

void on_rotate(int8_t dir) {
    if(!muted) {
        int16_t v = volume + dir * 5;
        if(v < 0) v = 0; if(v > 100) v = 100;
        volume = (uint8_t)v;
        PWM_SetDuty(TIM2_CH1, volume);
        USART_Print("Vol: "); USART_PrintNum(volume); USART_Println("%");
    }
}
void on_button(void) {
    muted = !muted;
    PWM_SetDuty(TIM2_CH1, muted ? 0 : volume);
    USART_Println(muted ? "MUTED" : "UNMUTED");
}

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_Init(115200);
    PWM_Init(TIM2_CH1, PC4, 10000, 50);  // 10kHz, 50% initial
    RotaryEncoder_InitInt(PC0, PC1, PC2);
    Rotary_SetCallback(on_rotate);
    Rotary_SetCallback(on_button);
    USART_Println("=== Volume Knob ===");
    while(1) { IWDG_Feed(); }
}

  1. Menu Navigator — หมุน=เลื่อนเมนู, กด=เลือก — ใช้ร่วมกับ LCDMenu หรือ OLED แสดงชื่อเมนูปัจจุบัน
  2. Multi-turn Counter — นับตำแหน่งแบบ absolute หมุนกี่รอบก็ได้ — ใช้ OLED แสดงจำนวนรอบและองศา (360°/20steps = 18°ต่อ step)
  3. Game Paddle — หมุนเพื่อขยับ paddle ในเกม Pong — สื่อสารผ่าน USART ไป PC โดยส่งเฉพาะ deltas
