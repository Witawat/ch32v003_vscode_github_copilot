📝 เฉลย — DS18B20

ข้อ 1: แสดง ROM ID ทั้งหมดของทุกเซนเซอร์

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "DS18B20.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_Init(115200);

    DS18B20_Device* dev = DS18B20_Init(PC0);

    USART_Print("Device initialized\r\n");
    USART_Print("DS18B20 ROM: ");
    uint8_t rom[8];
    DS18B20_GetROM(dev, rom);
    for(uint8_t j = 0; j < 8; j++) {
        USART_PrintHex(rom[j]);
        USART_Print(" ");
    }
    USART_Println("");

    if(rom[0] == 0x28) USART_Println("  → DS18B20");
    else if(rom[0] == 0x22) USART_Println("  → DS1822");
    else USART_Println("  → Unknown");

    while(1) {
        float t = DS18B20_ReadTemperatureBlocking(dev);
        USART_Print("T:"); USART_PrintFloat(t, 1);
        USART_Println("C");
        Delay_Ms(2000);
    }
}
🧠 แนวคิด: DS18B20_GetDeviceCount() นับจำนวนเซนเซอร์บนบัส — วน loop ตามจำนวน device — DS18B20_GetROM(i, &rom) ดึง 8-byte ROM ID ของแต่ละตัว — USART_PrintHex() พิมพ์ทีละ byte — เช็ค rom.family เพื่อระบุรุ่น — ใช้ DS18B20_GetTempCByROM() ด้วย ROM ได้ถ้าต้องการอ่านเฉพาะตัว

ข้อ 2: Parasite Power Mode (2-wire)

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "DS18B20.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_Init(115200);

    DS18B20_Device* dev = DS18B20_Init(PC0);
    DS18B20_SetParasiteMode(dev, 1);   // เปิด parasite (2-wire)
    USART_Println("Parasite Power Mode Enabled");
    USART_Println("Wiring: DQ+PC0, VCC+GND tied to GND");

    Timer_t tmr;
    Start_Timer(&tmr, 2000, 1);

    while(1) {
        if(Is_Timer_Expired(&tmr)) {
            float t = DS18B20_ReadTemperatureBlocking(dev);
            USART_Print("T: "); USART_PrintFloat(t, 1);
            USART_Println("C");
        }
    }
}
🧠 แนวคิด: Parasite mode ใช้แค่ 2 เส้น (DQ+GND) — VCC ต้องต่อ GND — DS18B20_SetParasiteMode(1) เปิด strong pull-up บน pin DQ ระหว่าง convert — ต้องต่อ MOSFET หรือใช้ GPIO push-pull strong pull-up ถ้าบัสไม่มี pull-up ในตัว — Delay_Ms(1000) นานกว่าปกติ (750ms) เพราะ parasite ต้องชาร์จ capacitor ภายในชิป — ถ้าอ่านไม่ขึ้น ให้เพิ่มเวลา หรือกลับไปใช้ 3-wire

ข้อ 3: Data Logger + Flash บันทึกอุณหภูมิ 3 จุด

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "SimpleFlash.h"
#include "DS18B20.h"

#define LOG_ADDR    0x0800F800
#define MAX_ENTRIES 144

typedef struct {
    uint32_t timestamp;
    float temps[3];
    uint8_t count;
} LogEntry_t;

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_Init(115200);

    DS18B20_Device* dev1 = DS18B20_Init(PC0);
    DS18B20_Device* dev2 = DS18B20_Init(PC1);
    DS18B20_Device* dev3 = DS18B20_Init(PC2);

    DS18B20_Device* devs[3] = {dev1, dev2, dev3};
    uint8_t n = 3;

    Timer_t tmr;
    Start_Timer(&tmr, 60000, 1);  // ทุก 1 นาที
    uint8_t idx = 0;

    while(1) {
        if(Is_Timer_Expired(&tmr)) {
            LogEntry_t entry;
            entry.timestamp = Timer_GetMs();
            entry.count = n;

            for(uint8_t i = 0; i < entry.count; i++) {
                entry.temps[i] = DS18B20_ReadTemperatureBlocking(devs[i]);
                USART_Print("#"); USART_PrintNum(i+1);
                USART_Print(":"); USART_PrintFloat(entry.temps[i], 1);
                USART_Print("C ");
            }
            USART_Println("");

            Flash_WriteLog(&entry, sizeof(LogEntry_t), idx);
            idx = (idx + 1) % MAX_ENTRIES;
        }
        IWDG_Feed();
    }
}
🧠 แนวคิด: รองรับ 3 จุดวัด — RequestTemperatures สั่ง convert พร้อมกันทุกตัว — Flash_WriteLog บันทึก struct ที่มี timestamp + array อุณหภูมิ — เลือกอ่านเฉพาะ ROM 3 ตัวแรก (หรือใช้ DS18B20_GetTempCByIndex) — อ่านกลับด้วย Flash_ReadLog เมื่อต้องการ dump ข้อมูลผ่าน USART
