📚 หลักการทำงาน

ESC (Electronic Speed Control) รับสัญญาณ PWM แบบเดียวกับ servo (50Hz, 1–2ms pulse) — แต่ ESC ใช้ควบคุมความเร็วมอเตอร์แบบไร้แปรงถ่าน (BLDC) โดย 1ms = 0% throttle, 2ms = 100% throttle

Calibration Sequence: ESC ต้องรู้ค่า min/max pulse ก่อนใช้งาน — (1) จ่ายไฟ ESC, (2) ส่ง max throttle ~2000μs, (3) รอเสียง beep, (4) ส่ง min throttle ~1000μs, (5) รอเสียง beep ยืนยัน — ทำครั้งเดียวหรือเมื่อเปลี่ยน ESC

Arm/Disarm: หลัง calibration ต้อง "arm" โดยส่ง throttle 0% ค้างไว้ ~3 วินาที — ESC บางรุ่นต้องใช้ combination (เช่น throttle min + รอ beep) — "disarm" โดยลด throttle ลง 0%

Throttle 0–100% แปลงเป็น pulse width: 1000μs (0%) ถึง 2000μs (100%) — duty cycle ที่ 50Hz: 5%–10%

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Init PWM + ESC Setup

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "ESC.h"

int main(void) {
    SystemCoreClockUpdate();
    ESC_Instance esc;
    ESC_Init(&esc, PWM1_CH1, 1000, 2000);  // channel, min 1000µs, max 2000µs

    ESC_Arm(&esc);  // รอ arm 3 วินาที

🔍 ESC_Init(&esc, PWM1_CH1, 1000, 2000) — ตั้งช่อง PWM, min pulse 1000µs, max pulse 2000µs — ESC_Arm(&esc) รอ 3 วิ

ขั้นที่ 2: Throttle Control + Safety

    // ไล่ throttle จาก 0% → 50% → 0%
    for(uint8_t t = 0; t <= 50; t += 5) {
        ESC_SetThrottle(&esc, t);
        Delay_Ms(1000);
    }
    for(uint8_t t = 50; t > 0; t -= 5) {
        ESC_SetThrottle(&esc, t);
        Delay_Ms(500);
    }

🔍 ESC_SetThrottle(&esc, t) — เปลี่ยน throttle โดยตรงเป็น % — ramp ช้า ๆ ป้องกัน ESC ตัดการทำงาน

ขั้นที่ 3: Calibration + Safe Loop

    // ===== ESC Calibration (ทำครั้งเดียว) =====
    ESC_SetThrottle(&esc, 100);  // Max throttle
    Delay_Ms(3000);              // รอ ESC beep
    ESC_SetThrottle(&esc, 0);    // Min throttle
    Delay_Ms(2000);              // รอ ESC confirmation beep
    // ===== Calibration Complete =====

    while(1) {
        for(uint8_t t = 0; t <= 80; t++) {
            ESC_SetThrottle(&esc, t);
            Delay_Ms(50);
        }
        Delay_Ms(3000);
        for(uint8_t t = 80; t > 0; t--) {
            ESC_SetThrottle(&esc, t);
            Delay_Ms(50);
        }
        ESC_SetThrottle(&esc, 0);
        ESC_Disarm(&esc);
        Delay_Ms(5000);
        ESC_Arm(&esc);
    }
}

🔍 Calibration sequence ต้องทำก่อนใช้งานจริง — Max→Min→Done — ถ้าเสียบแบตแล้ว ESC beep ถี่ ๆ แสดงว่ายังไม่ calibrate — ใช้ ESC_Disarm(&esc) + ESC_Arm(&esc) รีเซ็ตรอบ

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
มอเตอร์ไม่หมุน — ESC beep ถี่ESC ยังไม่ได้ Calibrate หรือ Throttle ไม่ใช่ 0% ตอนเริ่มทำ Calibration sequence: Max→Min→Done — หรือตั้ง throttle เป็น 0% แล้วรอ 3 วิ
มอเตอร์กระตุก ไม่ smoothPWM ความถี่ไม่ใช่ 50Hzตรวจสอบ PWM frequency — ต้องเป็น 50Hz เท่านั้น (20ms period)
มอเตอร์เร่งไม่สุดPulse width ไม่ถึง 2msตั้ง duty = 10% ที่ 50Hz (2000μs) — เช็คด้วย oscilloscope
ESC ร้อนเกินไปBattery voltage เกิน spec ESCตรวจสอบ LiPo cell count — 2S=7.4V, 3S=11.1V, 4S=14.8V — ใช้ให้ตรง ESC rating
ถอดใบพัดไม่ได้ → อุบัติเหตุละเลย safety protocol⚠️ ตั้งกฎ: "No props on bench" — ใช้เทปพันเพลาแทนการทดสอบ

🛠️ สร้าง Product: BLDC Thrust Tester

ใช้ Load Cell (HX711) + ESC วัดแรงขับของมอเตอร์ที่ throttle ต่าง ๆ — แสดงผลผ่าน OLED

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "ESC.h"

int main(void) {
    SystemCoreClockUpdate(); Timer_Init();
    OLED_Init_I2C(0x3C, PC2, PC1);
    HX711_Init(PD3, PD4);

    ESC_Instance esc;
    ESC_Init(&esc, PWM1_CH1, 1000, 2000);
    ESC_Arm(&esc);

    uint8_t throttle = 0;
    while(1) {
        throttle += 10; if(throttle > 100) throttle = 0;
        ESC_SetThrottle(&esc, throttle);
        Delay_Ms(2000);

        float grams = HX711_ReadGrams();
        OLED_Clear();
        OLED_Printf(0,0,"Throttle: %d%%", throttle);
        OLED_Printf(0,2,"Thrust: %.1fg", grams);
        OLED_Update();

        if(throttle == 0) { ESC_Disarm(&esc); Delay_Ms(3000); ESC_Arm(&esc); }
    }
}

✏️ ลองทำดู

  1. ทำ ESC Tester แบบพกพา — ใช้ POT ควบคุม throttle + OLED แสดงผล — ใส่ในกล่องมีปุ่ม ARM/DISARM
  2. ทำ Closed-Loop Speed Control — ใช้ hall sensor อ่าน RPM แล้วปรับ throttle อัตโนมัติให้ได้ความเร็วคงที่
  3. วัด KV Rating ของมอเตอร์ — ใช้ ESC + Tachometer หาค่า RPM/Volt โดยอัตโนมัติ
