ข้อ 2: เขียนข้อมูลลงการ์ด Mifare Classic

RC522_Instance rfid;
#define MFRC_SECTOR    1     // sector ที่จะเขียน
#define MFRC_BLOCK     (MFRC_SECTOR * 4)  // block แรกของ sector
#define MFRC_DEFAULT_KEY  {0xFF,0xFF,0xFF,0xFF,0xFF,0xFF}

uint8_t key_a[6] = MFRC_DEFAULT_KEY;

typedef struct {
    char    name[16];
    uint32_t balance;     // หน่วยสตางค์
    uint16_t issued;      // YYMM
    uint8_t  checksum;
} card_data_t;

void rc522_write_card_data(void) {
    RC522_Init(&rfid, PC4, PC3);

    USART_Println("[RC522] Place card to write...");

    while(1) {
        if(!RC522_IsCardPresent(&rfid)) {
            Delay_Ms(200);
            continue;
        }

        uint8_t uid[10];
        uint8_t uid_len = RC522_ReadUID(&rfid, uid);
        if(uid_len == 0) { Delay_Ms(200); continue; }

        USART_Println("Card detected. Writing...");

        // Authenticate sector ด้วย Key A
        if(!RC522_Auth(&rfid, MFRC_PICC_AUTHENT1A, MFRC_BLOCK,
                       key_a, uid)) {
            USART_Println("Auth FAIL");
            RC522_Halt(&rfid);
            Delay_Ms(500);
            continue;
        }

        // เตรียมข้อมูล
        card_data_t data;
        strcpy(data.name, "CH32V003 User");
        data.balance = 10000;  // 100.00 บาท
        data.issued = 2406;    // มิ.ย. 2024
        data.checksum = rc522_calc_checksum((uint8_t *)&data,
                                            sizeof(data)-1);

        // เขียน block (16 bytes)
        RC522_WriteBlock(&rfid, MFRC_BLOCK, (uint8_t *)&data);

        USART_Println("Write OK");

        // Verify: อ่านกลับ
        uint8_t verify[16];
        RC522_ReadBlock(&rfid, MFRC_BLOCK, verify);
        USART_Print("Verify: ");
        for(uint8_t i=0; i<16; i++) {
            USART_PrintHexByte(verify[i]);
            USART_Print(" ");
        }
        USART_Println("");

        RC522_Halt(&rfid);
        Delay_Ms(1000);
    }
}

uint8_t rc522_calc_checksum(uint8_t *d, uint8_t len) {
    uint8_t sum = 0;
    while(len--) sum ^= *d++;
    return sum;
}
🧠 แนวคิด: Mifare Classic 1K มี 16 sectors, แต่ละ sector มี 4 blocks (block0-3) — Block 3 ของแต่ละ sector คือ sector trailer (keyA + access bits + keyB) — ต้อง authenticate ด้วย key ก่อนอ่าน/เขียน — เขียน 16 bytes/block — เขียนข้อมูลผู้ใช้ลง block 0 ของ sector 1 — checksum XOR ง่ายสำหรับตรวจ integrity — อ่านกลับ verify ว่าเขียนถูก — sector trailer ห้ามเขียนทับโดยไม่ระวัง (จะล็อก sector ถาวร)

ข้อ 3: Sector Key Management — เปลี่ยนคีย์ A/B

RC522_Instance rfid;
uint8_t new_key_a[6] = {0xA0, 0xA1, 0xA2, 0xA3, 0xA4, 0xA5};
uint8_t new_key_b[6] = {0xB0, 0xB1, 0xB2, 0xB3, 0xB4, 0xB5};

void rc522_change_keys(void) {
    RC522_Init(&rfid, PC4, PC3);

    USART_Println("[KEY] Place card to change keys...");

    while(1) {
        if(!RC522_IsCardPresent(&rfid)) { Delay_Ms(200); continue; }

        uint8_t uid[10];
        uint8_t uid_len = RC522_ReadUID(&rfid, uid);
        if(uid_len == 0) continue;

        uint8_t sector = 1;
        uint8_t trailer = sector * 4 + 3;  // block 7

        // 1. Auth ด้วย key เดิม (default: FFFFFF)
        uint8_t default_key[6] = MFRC_DEFAULT_KEY;
        if(!RC522_Auth(&rfid, MFRC_PICC_AUTHENT1A, trailer,
                       default_key, uid)) {
            USART_Println("Auth FAIL — wrong old key?");
            RC522_Halt(&rfid); Delay_Ms(500); continue;
        }

        // 2. สร้าง sector trailer ใหม่
        uint8_t new_trailer[16];
        memcpy(new_trailer, new_key_a, 6);           // Key A
        new_trailer[6] = 0xFF; new_trailer[7] = 0x07;
        new_trailer[8] = 0x80;  // access bits
        memcpy(new_trailer + 10, new_key_b, 6);      // Key B

        // 3. เขียน trailer ใหม่
        RC522_WriteBlock(&rfid, trailer, new_trailer);

        USART_Println("[KEY] Sector 1 keys changed!");

        // 4. Verify ด้วย key ใหม่
        if(RC522_Auth(&rfid, MFRC_PICC_AUTHENT1A, trailer,
                      new_key_a, uid)) {
            USART_Println("✓ New Key A verified OK");
        }
        if(RC522_Auth(&rfid, MFRC_PICC_AUTHENT1B, trailer,
                      new_key_b, uid)) {
            USART_Println("✓ New Key B verified OK");
        }

        // แสดง access bits
        uint8_t tr[16];
        RC522_ReadBlock(&rfid, trailer, tr);
        USART_Print("New trailer: ");
        for(uint8_t i=0; i<16; i++) {
            USART_PrintHexByte(tr[i]);
            USART_Print(" ");
        }
        USART_Println("");

        RC522_Halt(&rfid);
        break;
    }
}

// Access bits 0x78 → KeyA: read+write, KeyB: read
// Access bits 0x08 → KeyA: read, KeyB: read+write
// Access bits 0x0F → transport config (default)
// คำเตือน: ห้ามตั้ง access bits ผิด → sector ถูกล็อกถาวร!
🧠 แนวคิด: sector trailer = block สุดท้ายของแต่ละ sector — 6 bytes Key A + 4 bytes access bits + 6 bytes Key B — เปลี่ยนคีย์: auth ด้วยคีย์เก่า → เขียน trailer ใหม่ → verify ด้วยคีย์ใหม่ — access bits 0xFF0780 = KeyA ใช้ auth ได้, KeyB อ่าน/เขียนได้ — อันตราย: ตั้ง access bits ผิดทำให้ sector ล็อกถาวร — ทดสอบกับ Mifare Classic 1K blank card ก่อนใช้จริง
