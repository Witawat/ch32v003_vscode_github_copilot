📝 เฉลย — RC522 RFID (UID Detection เท่านั้น)

ข้อ 1: UID Reader + Version Debug — อ่าน UID แสดงเป็น hex และ debug เมื่อ version ผิด

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "RC522.h"

RC522_Instance rfid;

// แปลง version byte เป็น string สำหรับ debug
const char* rc522_version_str(uint8_t v) {
    switch (v) {
        case 0x91: return "MFRC522 v1.0";
        case 0x92: return "MFRC522 v2.0";
        case 0x00:
        case 0xFF: return "⚠️ NOT FOUND (SPI/wiring/power?)";
        default:   return "Unknown (noise?)";
    }
}

// โปรแกรมหลัก — อ่าน UID และแสดงเป็น hex
void rc522_uid_reader(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);

    // ⚠️ SPI ต้อง init ก่อน RC522_Init เสมอ
    SPI_SimpleInit(SPI_MODE0, SPI_4MHZ, SPI_PINS_DEFAULT);

    RC522_Status st = RC522_Init(&rfid, PC4, PD2);   // CS=PC4, RST=PD2
    if (st != RC522_OK) {
        USART_Println("RC522_Init FAIL — ตรวจ CS/RST pin");
        while (1);
    }

    // ตรวจ version เป็น heart-beat ของ SPI
    uint8_t ver = RC522_GetVersion(&rfid);
    USART_Printf("RC522 version = 0x%02X (%s)\r\n", ver, rc522_version_str(ver));

    if (ver != 0x91 && ver != 0x92) {
        USART_Println("=== DIAGNOSTIC ===");
        USART_Println("  1. เช็ค SPI_SimpleInit(SPI_MODE0, SPI_4MHZ, SPI_PINS_DEFAULT) ก่อน RC522_Init");
        USART_Println("  2. เช็คสาย: PC5=SCK, PC6=MOSI, PC7=MISO, PC4=CS, PD2=RST");
        USART_Println("  3. เช็คไฟ = 3.3V (ห้าม 5V)");
        USART_Println("  4. เช็ค CS ลอยหรือไม่ — ต้อง LOW ตอนสื่อสาร");

        // ลอง reset แล้วอ่านใหม่ — บางที power-on ไม่ clean
        USART_Println("Try RC522_Reset + re-read version...");
        RC522_Reset(&rfid);
        Delay_Ms(50);
        ver = RC522_GetVersion(&rfid);
        USART_Printf("Version after reset = 0x%02X (%s)\r\n",
                     ver, rc522_version_str(ver));
        if (ver != 0x91 && ver != 0x92) {
            USART_Println("ยังผิด — ลองต่อวงจรใหม่อีกครั้ง");
            while (1);
        }
        USART_Println("✓ OK after reset");
    }

    USART_Println("Waiting for card...");
    while (1) {
        if (!RC522_IsCardPresent(&rfid)) {
            Delay_Ms(150);
            continue;
        }

        // ⚠️ 3 args: rfid, uid buffer, &len (out-param)
        uint8_t uid[RC522_UID_MAX_LEN];
        uint8_t len = 0;
        RC522_Status r = RC522_ReadUID(&rfid, uid, &len);

        if (r == RC522_OK) {
            USART_Printf("UID(%uB): ", (unsigned)len);
            for (uint8_t i = 0; i < len; i++) {
                USART_Printf("%02X ", uid[i]);
            }
            USART_Println("");
        } else {
            // พิมพ์ error code ตาม RC522_Status enum
            USART_Printf("ReadUID err=%d (", (int)r);
            switch (r) {
                case RC522_ERROR_NOCARD:    USART_Println("NOCARD)"); break;
                case RC522_ERROR_COLLISION: USART_Println("COLLISION)"); break;
                case RC522_ERROR_TIMEOUT:   USART_Println("TIMEOUT)"); break;
                case RC522_ERROR_CRC:       USART_Println("CRC — antenna weak/noise)"); break;
                default:                    USART_Println("OTHER)"); break;
            }
        }

        RC522_Halt(&rfid);   // ⚠️ ต้อง Halt ทุกครั้ง กัน UID ซ้ำซ้อน
    }
}
🧠 แนวคิด:
  • ลำดับที่ถูก: SPI_SimpleInit(SPI_MODE0, SPI_4MHZ, SPI_PINS_DEFAULT) → RC522_Init(&rfid, CS, RST) → RC522_GetVersion(&rfid) → polling RC522_IsCardPresent → RC522_ReadUID(&rfid, uid, &len) → RC522_Halt(&rfid)
  • RC522_ReadUID มี 3 args — &len เป็น out-param รับจำนวน bytes จริง — ห้ามเขียนแบบบทเดิมที่มี 2 args
  • RC522_GetVersion คืน uint8_t ใช้เป็น heart-beat ตรวจ SPI โดยไม่ต้องมีการ์ด — 0x91=v1.0, 0x92=v2.0, 0x00/0xFF = ไม่ได้ต่อ/SPI ผิด/ไฟผิด
  • ถ้า version ผิด → ลอง RC522_Reset(&rfid) + Delay 50ms แล้วอ่านใหม่ — บางที power-on ramp ไม่ clean — ถ้ายังผิด = wiring ผิดจริง
  • ต้อง Halt หลัง ReadUID ทุกครั้ง — ไม่งั้นการ์ดยังอยู่ในสนามจะถูกวนอ่านซ้ำ
  • ไม่มี ฟังก์ชัน Self-Test / Select Tag — ใช้ GetVersion แทนฟังก์ชัน self-test แบบบทเดิม, wrapper นี้อ่าน UID ได้อย่างเดียว

ข้อ 2: UID Whitelist (5 cards) + Relay + Buzzer Feedback

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "RC522.h"
#include <string.h>

RC522_Instance rfid;

#define RELAY_PIN   PC0
#define BUZZER_PIN  PC1
#define LED_OK      PC2
#define LED_DENY    PC3

// Whitelist — 5 การ์ด (เปลี่ยน UID เป็นที่อ่านได้จากข้อ 1)
#define WHITELIST_COUNT  5
uint8_t whitelist[WHITELIST_COUNT][RC522_UID_MAX_LEN] = {
    {0xA1, 0xB2, 0xC3, 0xD4, 0,0,0,0,0,0},   // card 1 (4 bytes — MIFARE Classic)
    {0xDE, 0xAD, 0xBE, 0xEF, 0x12, 0x34, 0,0,0,0},  // card 2 (7 bytes)
    {0x12, 0x34, 0x56, 0x78, 0,0,0,0,0,0},   // card 3 (4 bytes)
    {0xAA, 0xBB, 0xCC, 0xDD, 0xEE, 0xFF, 0x11,0,0,0}, // card 4 (7 bytes)
    {0x99, 0x88, 0x77, 0x66, 0,0,0,0,0,0},   // card 5 (4 bytes)
};
uint8_t whitelist_len[WHITELIST_COUNT] = {4, 7, 4, 7, 4};

// เล่นเสียง buzzer — frequency + duration
void buzzer(uint16_t freq, uint16_t ms) {
    PWM_SimpleInit(BUZZER_PIN, 1, freq, 500);
    Delay_Ms(ms);
    PWM_SetDuty(BUZZER_PIN, 1, 0);
}

void door_grant(void) {
    USART_Println("  → ✓ ACCESS GRANTED");
    GPIO_WriteBit(LED_OK,   1);
    GPIO_WriteBit(LED_DENY, 0);
    GPIO_WriteBit(RELAY_PIN, 1);   // เปิดล็อก
    buzzer(2000, 150);             // เสียงสูง 1 beep = ผ่าน
    Delay_Ms(3000);                // เปิด 3 วิ
    GPIO_WriteBit(RELAY_PIN, 0);   // ล็อกกลับ
    GPIO_WriteBit(LED_OK,   0);
}

void door_deny(void) {
    USART_Println("  → ✗ ACCESS DENIED");
    GPIO_WriteBit(LED_DENY, 1);
    GPIO_WriteBit(LED_OK,   0);
    buzzer(500, 400);              // เสียงต่ำ 1 beep = ไม่ผ่าน
    GPIO_WriteBit(LED_DENY, 0);
    Delay_Ms(1000);                // หน่วงกัน brute-force
}

// ค้น whitelist — คืน 1 ถ้า UID อยู่ใน list
uint8_t whitelist_match(const uint8_t* uid, uint8_t len) {
    for (uint8_t c = 0; c < WHITELIST_COUNT; c++) {
        if (len == whitelist_len[c] &&
            memcmp(uid, whitelist[c], len) == 0) {
            return 1;
        }
    }
    return 0;
}

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
    SPI_SimpleInit(SPI_MODE0, SPI_4MHZ, SPI_PINS_DEFAULT);

    if (RC522_Init(&rfid, PC4, PD2) != RC522_OK) {
        USART_Println("RC522 init fail");
        while (1);
    }

    // version check — heart-beat ก่อนลงมือ
    uint8_t ver = RC522_GetVersion(&rfid);
    if (ver != 0x91 && ver != 0x92) {
        USART_Println("⚠️ version ผิด — door ทำงานต่อไม่ได้");
        while (1);
    }
    USART_Println("[DOOR] Ready, waiting for card...");

    // ตั้งขา relay/LED เป็น output
    GPIO_SimpleInit(RELAY_PIN, GPIO_OUT_PP);
    GPIO_SimpleInit(LED_OK,    GPIO_OUT_PP);
    GPIO_SimpleInit(LED_DENY,  GPIO_OUT_PP);
    GPIO_WriteBit(RELAY_PIN, 0);

    while (1) {
        if (!RC522_IsCardPresent(&rfid)) {
            Delay_Ms(150);
            continue;
        }

        uint8_t uid[RC522_UID_MAX_LEN];
        uint8_t len = 0;
        RC522_Status r = RC522_ReadUID(&rfid, uid, &len);
        if (r != RC522_OK) {
            RC522_Halt(&rfid);
            continue;
        }

        // พิมพ์ UID ตาม debug
        USART_Print("Card ");
        for (uint8_t i = 0; i < len; i++) USART_Printf("%02X ", uid[i]);

        // ค้น whitelist
        if (whitelist_match(uid, len)) door_grant();
        else                            door_deny();

        RC522_Halt(&rfid);    // ⚠️ Halt หลังอ่านทุกครั้ง
        Delay_Ms(300);
    }
}
🧠 แนวคิด:
  • Whitelist เก็บ UID 5 ใบ — ทั้ง UID bytes และความยาวแยก array — เปรียบเทียบ len ก่อน memcmp กัน match ผิดถ้า UID มี padding zeros
  • Relay active HIGH เปิด 3 วิ → ปิดกลับ — ใช้ GPIO_SimpleInit(pin, GPIO_OUT_PP) + GPIO_WriteBit(pin, val)
  • Buzzer ใช้ PWM_SimpleInit(pin, ch, freq, duty) — frequency สูง (2000 Hz) = ผ่าน, ต่ำ (500 Hz) = ไม่ผ่าน — เล่นแล้ว PWM_SetDuty(0) ปิดเสียงทันที
  • ทุก call ส่ง &rfid (handle-based) — RC522_ReadUID(&rfid, uid, &len) มี 3 args + out-param — RC522_Halt(&rfid) ทุกครั้งหลังอ่าน
  • หน่วง 1 วิเมื่อ deny กัน brute-force — ตรวจ version ก่อนเปิดบริการ ถ้าผิด = รวงเลย (ไม่เสียความปลอดภัย)
  • Library อ่าน UID ได้อย่างเดียว — door lock ใช้แค่นี้พอ — ถ้าต้องการเก็บ balance หรือข้อมูลในบัตรต้องใช้ standard peripheral (Auth + ReadBlock/WriteBlock ไม่มีใน wrapper นี้)

ข้อ 3: Card Swap Detector — ตรวจจับการเปลี่ยนการ์ดระหว่างแตะ + ทดสอบ Halt/Reset

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "RC522.h"
#include <string.h>

RC522_Instance rfid;

// เก็บ UID ล่าสุดเพื่อเทียบกับ UID ใหม่
uint8_t last_uid[RC522_UID_MAX_LEN];
uint8_t last_len = 0;
uint8_t has_last = 0;
uint32_t swap_count = 0;

// พิมพ์ UID เป็น hex
void print_uid(const uint8_t* uid, uint8_t len) {
    for (uint8_t i = 0; i < len; i++) USART_Printf("%02X ", uid[i]);
}

// เปรียบเทียบ UID สองค่า — คืน 1 ถ้าเหมือนกัน
uint8_t uid_equal(const uint8_t* a, uint8_t alen,
                  const uint8_t* b, uint8_t blen) {
    if (alen != blen) return 0;
    return (memcmp(a, b, alen) == 0) ? 1 : 0;
}

// ทดสอบพฤติกรรม RC522_Halt vs RC522_Reset
// mode 0 = Halt เท่านั้น (default) — การ์ด sleep, ออกสนามก่อนค่อยอ่านใหม่
// mode 1 = Reset — รีเซ็ต MFRC522 ทั้งหมดทุกครั้ง (ช้ากว่า)
// mode 2 = ไม่ Halt เลย — จะเห็น UID ซ้ำซ้อน (bug demo)
uint8_t test_mode = 0;

void rc522_card_swap_test(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
    SPI_SimpleInit(SPI_MODE0, SPI_4MHZ, SPI_PINS_DEFAULT);

    if (RC522_Init(&rfid, PC4, PD2) != RC522_OK) {
        USART_Println("RC522 init fail");
        while (1);
    }

    uint8_t ver = RC522_GetVersion(&rfid);
    USART_Printf("RC522 version = 0x%02X\r\n", ver);
    if (ver != 0x91 && ver != 0x92) {
        USART_Println("⚠️ version ผิด — เช็ค SPI/wiring ก่อน");
        while (1);
    }

    USART_Println("Card Swap Detector ready");
    USART_Println("mode 0 = Halt, 1 = Reset, 2 = no-Halt (bug demo)");
    USART_Println("แตะการ์ดสลับกัน → นับจำนวนครั้งที่เปลี่ยนการ์ด");
    USART_Println("");
    USART_Println("การทดสอบ: แตะการ์ด A → ยกออก → แตะการ์ด B");
    USART_Println("           (ถ้าไม่ยกออก = L1 → L2 conflict จะ COLLISION)");

    while (1) {
        if (!RC522_IsCardPresent(&rfid)) {
            Delay_Ms(120);
            continue;
        }

        uint8_t uid[RC522_UID_MAX_LEN];
        uint8_t len = 0;
        RC522_Status r = RC522_ReadUID(&rfid, uid, &len);
        if (r != RC522_OK) {
            // อาจเป็น COLLISION ถ้าอ่านระหว่างเปลี่ยนการ์ด
            USART_Printf("ReadUID err=%d", (int)r);
            if (r == RC522_ERROR_COLLISION)
                USART_Println(" (เปลี่ยนการ์ดระหว่างอ่าน?)");
            else if (r == RC522_ERROR_CRC)
                USART_Println(" (antenna weak?)");
            RC522_Halt(&rfid);
            Delay_Ms(200);
            continue;
        }

        // พิมพ์ UID ปัจจุบัน
        USART_Print("UID: ");
        print_uid(uid, len);

        // เทียบกับครั้งก่อน → ตรวจ card swap
        if (has_last) {
            if (!uid_equal(last_uid, last_len, uid, len)) {
                swap_count++;
                USART_Printf("  → ⚡ CARD SWAP! (count=%lu)",
                             (unsigned long)swap_count);
                USART_Print("  prev was: ");
                print_uid(last_uid, last_len);
            }
        } else {
            USART_Print("  → first card");
        }

        // เก็บ UID ปัจจุบันเป็น last
        memcpy(last_uid, uid, len);
        last_len = len;
        has_last = 1;
        USART_Println("");

        // ทดสอบพฤติกรรม Halt vs Reset ตาม mode
        switch (test_mode) {
            case 0:
                RC522_Halt(&rfid);       // ปกติ — การ์ด Halt, อ่าน UID ครั้งต่อไปได้
                break;
            case 1:
                RC522_Reset(&rfid);      // รีเซ็ต module ทั้งหมด — ช้ากว่า
                Delay_Ms(50);             // ต้อง Delay หลัง Reset
                break;
            case 2:
                // ไม่ Halt — จะเห็น UID ซ้ำซ้อนทุกรอบ (demo bug)
                USART_Println("  ⚠️ no-Halt: จะอ่าน UID ซ้ำรอบหน้า");
                break;
        }

        Delay_Ms(250);
    }
}

// สรุปพฤติกรรมที่สังเกตได้จากการทดสอบ 3 mode:
//
// mode 0 (Halt — แนะนำ):
//   - แตะการ์ด A → อ่าน UID ครั้งเดียว → Halt → การ์ด sleep
//   - ยกออกแล้วแตะ B → UID ใหม่ → เทียบกับ last → "CARD SWAP!"
//   - ไม่แตะออกแล้วแตะทับ B → REQA ไม่ได้ (A ยัง Halt อยู่) → ไม่ detect
//     → ต้องยก A ออกก่อน
//
// mode 1 (Reset — ช้ากว่า แต่ brute-force reset):
//   - เหมาะเวลา module ค้างจากการ์ดไม่ดี
//   - Delay_Ms(50) หลัง Reset → ทำให้ polling ห่างขึ้น
//
// mode 2 (no-Halt — bug demo):
//   - การ์ด A ยังอยู่ในสนาม → ReadUID ซ้ำได้ UID เดิมทุกรอบ
//   - ไม่มีการ "detect removal" → ไม่เหมาะสำหรับ card swap detection
//   - ยืนยันว่า Halt จำเป็นมาก
🧠 แนวคิด:
  • Card swap detection: เก็บ UID + ความยาวของครั้งก่อนไว้ใน last_uid/last_len — อ่าน UID ใหม่ → เทียบด้วย memcmp — ถ้าไม่เท่ากัน = "CARD SWAP" → นับ swap_count
  • เทียบ UID ต้องเช็คความยาวก่อน memcmp — ถ้า len ไม่เท่ากัน = การ์ดคนละประเภท → swap แน่นอน — memcmp ต้องส่ง len ที่ถูกต้อง ไม่งั้น umcmp bytes padding ที่เป็น zeros
  • พฤติกรรม RC522_Halt(&rfid): สั่งการ์ด HLTA → เข้า Halt state → REQA จะไม่ตอบ → "ต้องยกการ์ดออกแล้วแตะใหม่" → เหมาะสำหรับ card swap detection (เพราะต้อง remove/re-insert ก่อน)
  • พฤติกรรม RC522_Reset(&rfid): รีเซ็ต MFRC522 module ทั้งหมด (soft reset) — เว้น Delay_Ms(50) หลังเรียก — ช้ากว่า Halt ใช้เฉพาะ module ค้าง
  • bug demo (mode 2): ถ้าไม่ Halt → การ์ดยัง active → RC522_IsCardPresent + ReadUID จะได้ UID เดิมซ้ำๆ → ไม่ detect removal → ยืนยันว่า Halt จำเป็น
  • COLLISION/CRC error ระหว่างอ่าน = อาจเปลี่ยนการ์ดระหว่าง REQA → anticollision → handle error ด้วยการ Halt + retry — ไม่ใช่ bug ของ wiring
  • Library wrapper นี้ อ่าน UID ได้อย่างเดียว — ใช้ทำ attendance, door lock, card swap ได้ — ถ้าต้องการเก็บข้อมูลหรือเปลี่ยน key ต้องใช้ standard peripheral (ไม่มี SelectTag/Auth/ReadBlock/WriteBlock ใน wrapper)
