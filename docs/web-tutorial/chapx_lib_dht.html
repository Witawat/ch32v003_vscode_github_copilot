DHT11/DHT22 — เซนเซอร์อุณหภูมิ+ความชื้น GPIO

ระดับเริ่มต้น

🧭 แผนภาพการทำงาน

flowchart TD
    A["DHT_Init(&dht, PC4, DHT_TYPE_DHT22)"] --> B["Delay_Ms(2000)"]
    B --> C["DHT_Read(&dht)"]
    C --> D{"==DHT_OK?"}
    D -->|Yes| E["DHT_GetTemperature / DHT_GetHumidity"]
    E --> C
    D -->|No| F["DHT_StatusStr(dht.last_status)"]
    F --> C

🔌 การต่อวงจร

        DHT11 / DHT22 (1-Wire GPIO)
┌────────────────────────────────────┐
│ PC4 ──── DATA                      │
│ 3.3V ─── VCC                       │
│ GND  ─── GND                       │
│                                    │
│ R 4.7kΩ pull-up: DATA ↔ 3.3V      │
│ (โมดูลสำเร็จรูปมี pull-up ในตัว)     │
└────────────────────────────────────┘

สายสัญญาณเส้นเดียว (Single-wire bidirectional) — ใช้ GPIO แบบ bit-bang ห้ามลืม pull-up 4.7kΩ ถ้าไม่มี pull-up จะอ่านไม่ขึ้น

📚 หลักการทำงาน

Single-wire bidirectional: ใช้ขา GPIO เพียงขาเดียว ทั้งส่งคำสั่งและรับข้อมูลกลับ — ผลัดกันเป็น input/output

40-bit frame: ความชื้นจำนวนเต็ม 8bit + ความชื้นทศนิยม 8bit + อุณหภูมิจำนวนเต็ม 8bit + อุณหภูมิทศนิยม 8bit + checksum 8bit

DHT11: ความแม่นยำ ±2°C / ±5%RH, อ่านได้สูงสุด 1Hz (1 ครั้งต่อวินาที) — ใช้ type DHT_TYPE_DHT11

DHT22: ความแม่นยำ ±0.5°C / ±2%RH, อ่านได้สูงสุด 0.5Hz (1 ครั้งต่อ 2 วินาที), มี CRC 8-bit — ใช้ type DHT_TYPE_DHT22

Blocking ~25ms: ระหว่างอ่าน DHT_Read() จะบล็อกโปรแกรมประมาณ 25ms — ห้ามเรียกใน ISR โดยเด็ดขาด

DHT_Instance: ใช้ struct instance — DHT_Init(&dht, pin, type) → DHT_Read(&dht) → DHT_GetTemperature(&dht) / DHT_GetHumidity(&dht) — dht.last_status เก็บสถานะล่าสุด

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Init + Setup

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "DHT.h"

DHT_Instance dht;

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();

    DHT_Init(&dht, PC4, DHT_TYPE_DHT22);
    Delay_Ms(2000);  // ให้เซนเซอร์ stabilize
    USART_Println("DHT Ready!");

🔍 DHT_Init(&dht, pin, type) — type เป็น DHT_TYPE_DHT11 หรือ DHT_TYPE_DHT22 — ต้องหน่วง 2 วิหลังเปิดไฟให้เซนเซอร์ stabilize ก่อนอ่านครั้งแรก

ขั้นที่ 2: อ่านค่าอุณหภูมิ+ความชื้น

    if(DHT_Read(&dht) == DHT_OK) {
        float t = DHT_GetTemperature(&dht);
        float h = DHT_GetHumidity(&dht);
        USART_Print("Temp: ");
        USART_PrintFloat(t, 1);
        USART_Print("C  Hum: ");
        USART_PrintFloat(h, 1);
        USART_Println("%");
    } else {
        USART_Print("Error: ");
        USART_Println(DHT_StatusStr(dht.last_status));
    }

🔍 DHT_Read(&dht) — คืน DHT_OK เมื่ออ่านสำเร็จ — จากนั้นเรียก DHT_GetTemperature() และ DHT_GetHumidity() แยกกัน — dht.last_status + DHT_StatusStr() ใช้ดูข้อความ error

ขั้นที่ 3: Loop + Non-Blocking Timer

    Timer_t tmr;
    Start_Timer(&tmr, 2000, 1);  // อ่านทุก 2 วินาที

    while(1) {
        if(Is_Timer_Expired(&tmr)) {
            if(DHT_Read(&dht) == DHT_OK) {
                float t = DHT_GetTemperature(&dht);
                float h = DHT_GetHumidity(&dht);
                USART_Print("T:"); USART_PrintFloat(t, 1);
                USART_Print(" H:"); USART_PrintFloat(h, 1);
                USART_Println("%");
            }
        }
        IWDG_Feed();
    }
}

🔍 ใช้ Timer_t แบบ Non-Blocking ตั้งอ่านทุก 2 วิ — ไม่บล็อก main loop, IWDG_Feed() เลี้ยง watchdog ระหว่างรอรอบถัดไป

📋 API Reference

ฟังก์ชันคำอธิบายตัวอย่าง
DHT_Init(&dht, pin, type)กำหนด instance, ขา GPIO และชนิดDHT_Init(&dht, PC4, DHT_TYPE_DHT22)
Delay_Ms(2000)หน่วงให้เซนเซอร์ stabilize หลัง initDelay_Ms(2000)
DHT_Read(&dht)อ่านข้อมูล — คืน DHT_OK หรือ error codeif(DHT_Read(&dht) == DHT_OK){...}
DHT_GetTemperature(&dht)ดึงค่าอุณหภูมิหลัง Read สำเร็จ (°C)float t = DHT_GetTemperature(&dht)
DHT_GetHumidity(&dht)ดึงค่าความชื้นหลัง Read สำเร็จ (%)float h = DHT_GetHumidity(&dht)
DHT_StatusStr(status)แปลง status code เป็นข้อความDHT_StatusStr(dht.last_status)

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
อ่านได้ -999 หรือ error เสมอไม่มี pull-up resistor หรือสายยาวเกิน 1 เมตรต่อ R 4.7kΩ ระหว่าง DATA ↔ 3.3V, ลดความยาวสายให้สั้นกว่า 50cm
อ่านได้ครั้งแรกครั้งเดียว แล้วค้างอ่านถี่เกิน — DHT11 ห้าม < 1Hz, DHT22 ห้าม < 0.5Hzใช้ Timer_t ตั้งช่วงห่าง >= 2000ms สำหรับ DHT22, >= 1000ms สำหรับ DHT11
DHT22 ให้ค่าแปลกๆDHT_Init ใส่ type เป็น DHT_TYPE_DHT11 แต่เซนเซอร์เป็น DHT22เปลี่ยนเป็น DHT_Init(&dht, PC4, DHT_TYPE_DHT22)
ISR ทำงานแปลกหลังจากอ่าน DHTDHT_Read() บล็อก ~25ms — ไม่ควรเรียกใน ISRย้ายการอ่าน DHT ออกจาก ISR — อ่านใน main loop เท่านั้น ใช้ flag แจ้ง ISR
ค่าแกว่ง อ่านแล้วได้คนละค่าทุกครั้งไฟเลี้ยงไม่นิ่ง หรือ noise บนสาย DATAใช้ capacitor 100nF คร่อม VCC-GND ใกล้เซนเซอร์, ใช้สายสั้น, average 3-5 ครั้ง
อ่านไม่ขึ้นเลย — DHT_Read คืน errorสลับขา VCC กับ DATA หรือลืม Delay_Ms(2000)เช็คขั้วสาย — DHT11/22 pin1=VCC, pin2=DATA, pin3=NC, pin4=GND — รอ 2 วิหลัง init

🛠️ สร้าง Product: เครื่องวัดอุณหภูมิ+ความชื้นในห้อง

DHT22 + OLED SSD1306 แสดงผลทุก 2 วินาที ด้วย Timer Non-Blocking

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "DHT.h"
#include "oled_i2c.h"
#include "oled_graphics.h"
#include "oled_fonts.h"

DHT_Instance dht;
OLED_Handle oled;

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    I2C_SimpleInit(I2C_SPEED_100K, I2C_PINS_DEFAULT);

    DHT_Init(&dht, PC4, DHT_TYPE_DHT22);
    Delay_Ms(2000);

    OLED_Init(&oled, OLED_128x64, 0x3C);
    OLED_SetFont(&oled, FONT_8x16);

    Timer_t tmr;
    Start_Timer(&tmr, 2000, 1);
    char buf[32];

    while(1) {
        if(Is_Timer_Expired(&tmr)) {
            OLED_Clear(&oled);
            if(DHT_Read(&dht) == DHT_OK) {
                float t = DHT_GetTemperature(&dht);
                float h = DHT_GetHumidity(&dht);
                SimpleHAL_Sprintf(buf, "Temp: %.1f C", t);
                OLED_DrawStringAlign(&oled, 0, 0, buf, ALIGN_LEFT);
                SimpleHAL_Sprintf(buf, "Hum : %.1f %%", h);
                OLED_DrawStringAlign(&oled, 32, 0, buf, ALIGN_LEFT);
            } else {
                OLED_DrawStringAlign(&oled, 24, 0, "Sensor Error!", ALIGN_LEFT);
            }
            OLED_Update(&oled);
        }
        IWDG_Feed();
    }
}

✏️ ลองทำดู

  1. LED แสดงสถานะ: LED เขียวเมื่ออุณหภูมิปกติ (≤ 35°C), LED แดงเมื่อร้อน (> 35°C) — ใช้ digitalWrite
  2. ทำ Data Logger บันทึกอุณหภูมิ+ความชื้นลง Flash ทุก 5 นาที — ใช้ Flash_Write() และ ELAPSED_TIME()
  3. เทียบ DHT11 vs DHT22 บน MCU ตัวเดียว — ต่อคนละขา ใช้ 2 instances แสดงผลบน OLED พร้อมกัน
