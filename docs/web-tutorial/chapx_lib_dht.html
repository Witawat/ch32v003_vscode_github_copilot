DHT11/DHT22 — เซนเซอร์อุณหภูมิ+ความชื้น GPIO

ระดับเริ่มต้น

🧭 แผนภาพการทำงาน

flowchart TD
    A["DHT_Init(pin,type)"] --> B["DHT_Read(&t,&h)"]
    B --> C{"==DHT_OK?"}
    C -->|Yes| D["ใช้ค่า"]
    D --> B
    C -->|No| E["Delay 1000ms"]
    E --> B

🔌 การต่อวงจร

        DHT11 / DHT22 (1-Wire GPIO)
┌────────────────────────────────────┐
│ PC0 ──── DATA                      │
│ 3.3V ─── VCC                       │
│ GND  ─── GND                       │
│                                    │
│ R 4.7kΩ pull-up: DATA ↔ 3.3V      │
│ (โมดูลสำเร็จรูปมี pull-up ในตัว)     │
└────────────────────────────────────┘

สายสัญญาณเส้นเดียว (Single-wire bidirectional) — ใช้ GPIO แบบ bit-bang ห้ามลืม pull-up 4.7kΩ ถ้าไม่มี pull-up จะอ่านไม่ขึ้น

📚 หลักการทำงาน

Single-wire bidirectional: ใช้ขา GPIO เพียงขาเดียว ทั้งส่งคำสั่งและรับข้อมูลกลับ — ผลัดกันเป็น input/output

40-bit frame: ความชื้นจำนวนเต็ม 8bit + ความชื้นทศนิยม 8bit + อุณหภูมิจำนวนเต็ม 8bit + อุณหภูมิทศนิยม 8bit + checksum 8bit

DHT11: ความแม่นยำ ±2°C / ±5%RH, อ่านได้สูงสุด 1Hz (1 ครั้งต่อวินาที)

DHT22: ความแม่นยำ ±0.5°C / ±2%RH, อ่านได้สูงสุด 0.5Hz (1 ครั้งต่อ 2 วินาที), มี CRC 8-bit ตรวจสอบความถูกต้อง

Blocking ~25ms: ระหว่างอ่าน DHT_Read() จะบล็อกโปรแกรมประมาณ 25ms — ห้ามเรียกใน ISR โดยเด็ดขาด

ห้ามอ่านถี่เกิน: ถ้าอ่านเร็วกว่า 1Hz(DHT11) หรือ 0.5Hz(DHT22) จะได้ค่าเดิมซ้ำ หรือค่าเพี้ยน

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Init + Setup

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "DHT.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    DHT_Init(PC0, DHT11);  // หรือ DHT22
    USART_Println("DHT Ready!");

🔍 DHT_Init(pin, type) — type เป็น DHT11 หรือ DHT22 ต้องตรงกับเซนเซอร์ที่ต่ออยู่ ถ้า DHT22 แต่ตั้งเป็น DHT11 จะอ่านค่าแปลก

ขั้นที่ 2: อ่านค่าอุณหภูมิ+ความชื้น

    float t, h;
    if(DHT_Read(&t, &h) == DHT_OK) {
        USART_Print("Temp: ");
        USART_PrintFloat(t, 1);
        USART_Print("C  Hum: ");
        USART_PrintFloat(h, 1);
        USART_Println("%");
    } else {
        USART_Println("DHT Read Failed!");
    }

🔍 DHT_Read(&t, &h) — คืน DHT_OK เมื่ออ่านสำเร็จ หรือ DHT_FAIL เมื่อไม่สำเร็จ (pull-up หาย/สายหลวม/สายยาวเกิน 1m) ค่าที่อ่านได้จะถูกเก็บในตัวแปร t และ h ตามลำดับ

ขั้นที่ 3: Loop + Non-Blocking Timer

    Timer_t tmr;
    Start_Timer(&tmr, 2000, 1);  // อ่านทุก 2 วินาที

    while(1) {
        if(Is_Timer_Expired(&tmr)) {
            float t, h;
            if(DHT_Read(&t, &h) == DHT_OK) {
                USART_Print("T:"); USART_PrintFloat(t, 1);
                USART_Print(" H:"); USART_PrintFloat(h, 1);
                USART_Println("%");
            }
        }
        IWDG_Feed();  // เลี้ยงหมาถ้ามี
    }
}

🔍 ใช้ Timer_t แบบ Non-Blocking ตั้งอ่านทุก 2 วิ — ไม่บล็อก main loop, IWDG_Feed() เลี้ยง watchdog ระหว่างรอรอบถัดไป

📋 API Reference

ฟังก์ชันคำอธิบายตัวอย่าง
DHT_Init(pin, type)กำหนดขา GPIO และชนิดเซนเซอร์DHT_Init(PC0, DHT22)
DHT_Read(&t, &h)อ่านอุณหภูมิ+ความชื้น — คืน DHT_OK หรือ DHT_FAILif(DHT_Read(&t, &h) == DHT_OK){...}
DHT_ReadTempOnly(&t)อ่านเฉพาะอุณหภูมิDHT_ReadTempOnly(&temp)
DHT_ReadHumOnly(&h)อ่านเฉพาะความชื้นDHT_ReadHumOnly(&hum)

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
อ่านได้ -999 เสมอไม่มี pull-up resistor หรือสายยาวเกิน 1 เมตรต่อ R 4.7kΩ ระหว่าง DATA ↔ 3.3V, ลดความยาวสายให้สั้นกว่า 50cm
อ่านได้ครั้งแรกครั้งเดียว แล้วค้างอ่านถี่เกิน — DHT11 ห้าม < 1Hz, DHT22 ห้าม < 0.5Hzใช้ Timer_t ตั้งช่วงห่าง >= 2000ms สำหรับ DHT22, >= 1000ms สำหรับ DHT11
DHT22 ให้ค่าแปลกๆ (เช่น temp ติดลบทั้งที่ห้อง 30°C)DHT_Init ใส่ type เป็น DHT11 แต่เซนเซอร์เป็น DHT22เปลี่ยน DHT_Init(PC0, DHT11) → DHT_Init(PC0, DHT22)
ISR ทำงานแปลกหลังจากอ่าน DHTDHT_Read() บล็อก ~25ms — ไม่ควรเรียกใน ISRย้ายการอ่าน DHT ออกจาก ISR — อ่านใน main loop เท่านั้น ใช้ flag แจ้ง ISR
ค่าแกว่ง อ่านแล้วได้คนละค่าทุกครั้งไฟเลี้ยงไม่นิ่ง หรือ noise บนสาย DATAใช้ capacitor 100nF คร่อม VCC-GND ใกล้เซนเซอร์, ใช้สายสั้น, average 3-5 ครั้ง
อ่านไม่ขึ้นเลย — คืน DHT_FAIL ตลอดสลับขา VCC กับ DATAเช็คขั้วสาย — DHT11/22 pin1=VCC, pin2=DATA, pin3=NC, pin4=GND

🛠️ สร้าง Product: เครื่องวัดอุณหภูมิ+ความชื้นในห้อง

DHT11 + OLED SSD1306 แสดงผลทุก 2 วินาที ด้วย Timer Non-Blocking

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "DHT.h"
#include "OLED.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_Init(115200);
    USART_Println("=== DHT11 Room Monitor ===");

    DHT_Init(PC0, DHT11);

    OLED_t oled;
    OLED_Init(&oled, 0x3C, OLED_128x64);
    OLED_SetFont(&oled, FONT_8x16);

    Timer_t tmr;
    Start_Timer(&tmr, 2000, 1);

    while(1) {
        if(Is_Timer_Expired(&tmr)) {
            float t, h;
            OLED_Clear(&oled);
            if(DHT_Read(&t, &h) == DHT_OK) {
                OLED_SetCursor(&oled, 0, 0);
                OLED_PrintFloat(&oled, "Temp:", t, 1); OLED_Print(&oled, " C");
                OLED_SetCursor(&oled, 0, 2);
                OLED_PrintFloat(&oled, "Hum :", h, 1); OLED_Print(&oled, " %");
            } else {
                OLED_SetCursor(&oled, 0, 1);
                OLED_Print(&oled, "Sensor Error!");
            }
            OLED_Display(&oled);
        }
        IWDG_Feed();
    }
}

✏️ ลองทำดู

  1. LED แสดงสถานะ: LED เขียวเมื่ออุณหภูมิปกติ (≤ 35°C), LED แดงเมื่อร้อน (> 35°C) — ใช้ digitalWrite
  2. ทำ Data Logger บันทึกอุณหภูมิ+ความชื้นลง Flash ทุก 5 นาที — ใช้ Flash_Write() และ ELAPSED_TIME()
  3. เทียบ DHT11 vs DHT22 บน MCU ตัวเดียว — ต่อคนละขา (PC0= DHT11, PC1= DHT22) แสดงผลบน OLED พร้อมกัน
