OLED SSD1306 — จอ 128x64 I2C

3.3V─VCC  GND─GND  PC1─SDA  PC2─SCL   (I2C addr=0x3C)

#include "oled_i2c.h"
OLED_Handle oled;
OLED_Init(&oled, OLED_128x64, 0x3C);
OLED_Clear(&oled);
OLED_SetCursor(&oled, 0, 0); OLED_Print(&oled, "สวัสดี CH32V003!");
OLED_SetCursor(&oled, 0, 2); OLED_PrintFloat(&oled, "Temp:", 25.5f, 1);
OLED_DrawRect(&oled, 100, 0, 127, 63, 1);
OLED_Update(&oled);  // ★ ต้องเรียก — push buffer ลงจอ

⚠️ 3 ขนาด (128x64/128x32/64x48) → เลือกตอน Init, buffer ~1KB ใน RAM, I2C addr 0x3C(0x78<<1) หรือ 0x3D, Font 6x8/8x16 สลับด้วย OLED_SetFont(&oled, FONT_8x16)

API หลัก

OLED_Init(o, addr, size)Init + clearOLED_SetCursor(o, x, y)ตำแหน่ง text
OLED_Print(o, str)พิมพ์ข้อความOLED_PrintFloat(o, prefix, val, decimals)เลขทศนิยม
OLED_SetPixel(o, x, y)จุดOLED_DrawLine(o, x1,y1,x2,y2)เส้น
OLED_DrawRect(o, x,y,w,h, fill)สี่เหลี่ยมOLED_DrawCircle(o, cx,cy,r, fill)วงกลม
OLED_Display(o)Push buffer → จอOLED_Clear(o)ล้าง buffer

LCD1602 I2C — จอตัวอักษร 16x2

#include "lcd1602_i2c.h"
LCD_Init(0x27);  // หรือ 0x3F
LCD_SetCursor(0, 0); LCD_Print("Hello World!");
LCD_SetCursor(1, 0); LCD_Print("Line 2");
// Custom char: LCD_CreateChar(0, custom_pattern); LCD1602_I2C_WriteChar(0);

⚠️ 2 address หลัก: 0x27 vs 0x3F — ใช้ I2CScan ตรวจสอบ, backlight เปิด/ปิดด้วย jumper หรือคำสั่ง

MAX7219 — LED Matrix 8x8 (SPI)

PC4─CS  PC5─SCK  PC6─DIN  5V─VCC  GND─GND  (chain ได้หลาย module)

#include "MAX7219.h"
MAX7219_Init(PC5, PC6, PC4, 1);       // 1 module, CS=PC4
MAX7219_SetIntensity(5);    // 0-15
MAX7219_Clear();
MAX7219_StartScrollText("สวัสดี", 100);  // 100ms/frame
// Static char: MAX7219_DrawBitmap('A');
// Custom bitmap: MAX7219_DrawBitmap(custom_8x8);

⚠️ SPI pins ต้องเป็น PC5(SCK) PC6(MOSI) — CS เลือก pin ไหนก็ได้, ต่อหลาย module → MAX7219_Init(N, CS), ใช้ไฟ 5V ได้

TM1637 — 7-Segment 4 หลัก (GPIO)

#include "TM1637.h"
TM1637_Init(PC0, PC1);           // CLK, DIO — pin ไหนก็ได้
TM1637_SetBrightness(7);         // 0-7
TM1637_DisplayNumber(1234);      // "1234"
TM1637_DisplayTime(12, 30);      // "12:30"
TM1637_DisplayFloat(25.5f, 1);   // "25.5"

⚠️ GPIO bit-bang, เลือก pin อะไรก็ได้, 4-digit เท่านั้น, no colon control — colon ติดตลอดเวลา

TM1650 — 7-Segment I2C

#include "TM1650.h"
TM1650_Init(0x24);  // I2C addr 0x24-0x27
TM1650_SetBrightness(3);
TM1650_DisplayNumber(5678);
TM1650_DisplayTime(8, 30);  // "08:30"

⚠️ I2C 4 address (0x24/0x25/0x26/0x27) เลือกด้วย jumper, เหมือน TM1637 แต่ใช้ I2C แทน GPIO — ประหยัด pin

NeoPixel — WS2812B RGB Strip (GPIO)

PC0─DATA  5V─VCC  GND─GND  (ห้าม 3.3V! WS2812B ต้อง 5V logic → level shifter)

#include "NeoPixel.h"
#define LED_COUNT 8
NeoPixel_Init(GPIOC, GPIO_Pin_0, LED_COUNT);
NeoPixel_SetPixelColor(0, 255, 0, 0);    // LED 0 = Red
NeoPixel_SetPixelColor(1, 0, 255, 0);    // LED 1 = Green
NeoPixel_SetAll(0, 0, 255);         // All Blue
NeoPixel_Show();                     // Push to strip
// Rainbow: for(i=0;i

⚠️ WS2812B ใช้ 5V logic → CH32V003 3.3V → ต้องใช้ level shifter (74HCT125/SN74AHCT125) หรือ hack: ลดไฟ VCC strip เหลือ ~3.7V, timing-critical — ห้ามมี interrupt ขัดจังหวะตอน NeoPixel_Show()

WS2812Matrix — 8x8 RGB Matrix (GPIO)

#include "WS2812Matrix.h"
WS2812Matrix_Init(PC0, MATRIX_8x8);
WS2812Matrix_Clear();
WS2812Matrix_SetBrightness(50);  // 0-255
WS2812Matrix_PrintChar('A', 255,0,0);    // Red 'A'
WS2812Matrix_ScrollText("THAI", 100, 0,255,0);  // Green scroll
WS2812Matrix_DrawPixel(0, 0, 255, 255, 255);    // White pixel
WS2812Matrix_Show();

⚠️ ต่อยอดจาก NeoPixel, RAM ~512B (buffer+temp), มีฟอนต์ไทยในตัว, รองรับ effects: fade, flash, rotate

WS2815Matrix — 12V RGB Matrix

#include "WS2815Matrix.h"
// API เหมือน WS2812Matrix — เปลี่ยนแค่ 12V power + ระดับสัญญาณ
WS2815Matrix_Init(PC0, MATRIX_8x8);
// ... same API ...

⚠️ 12V — ใช้กับไฟเลี้ยง 12V, signal ยัง 5V (ใช้ level shifter), เหนือกว่า WS2812: ถ้า LED หลอดนึงพัง หลอดต่อไปยังทำงาน (break-point continue)

P10 — LED Panel 32x16 (GPIO+timer)

PC0─A  PC1─B  PC2─CLK  PC3─SCLK  PC4─R1  PC5─OE  5V─VCC

#include "P10.h"
P10_Init(PC2, PC4, PC0, PC1, PC3, PC5); // CLK, R1, A, B, SCLK, OE
P10_Clear();
P10_SetCursor(0, 0); P10_Print("P10 Panel");
P10_SetCursor(0, 8); P10_Print("32x16");
// P10_Update() — called by timer ISR (auto)

⚠️ ใช้ TIM2 สำหรับ scan rate (~2kHz), 6 GPIO pins, RAM 64B buffer, 5V logic — CH32V003 3.3V พอ (marginal), ใช้ external 5V power

TJC HMI — Touch Display (USART)

PD5(TX)─RX(TJC)  PD6(RX)─TX(TJC)  5V─VCC  GND─GND

#include "TJC.h"
TJC_Handle hmi;
TJC_Init(&hmi, USART1, BAUD_9600);
TJC_SendCommand(&hmi, "page 0");  // switch page
TJC_SetText(&hmi, "t0", "Hello");  // set text widget
TJC_SetValue(&hmi, "n0", 42);      // set number widget
// Callback:
void on_btn_press(TJC_Handle *h, const char *page, const char *id) {
    if(strcmp(id, "b0") == 0) { USART_Println("OK pressed!"); }
}
TJC_RegisterTouchCallback(&hmi, on_btn_press);
// In loop:
TJC_ProcessResponse(&hmi);  // ★ ต้องเรียกบ่อย — process RX buffer

⚠️ ใช้ USART 1 เดียวของ CH32V003 → แชร์กับ USART_Print ไม่ได้, ต้องออกแบบ UI ใน TJC Editor ก่อน (PC software), callback 5 แบบ: touch, numeric, command, page, error, ข้อความภาษาไทย → ใช้ font ใน TJC ไม่เกี่ยวกับ CH32V003

ShiftReg595 — 74HC595 (GPIO)

PC0─DATA  PC1─CLK  PC2─LATCH  3.3V─VCC  GND─GND

#include "ShiftReg595.h"
ShiftReg595_Init(PC0, PC1, PC2);  // DATA, CLK, LATCH
ShiftReg595_Write(0b10101010);     // 8-bit output
ShiftReg595_WriteBit(0, 1);        // bit 0 = HIGH
// Chain multiple: ShiftReg595_Write16(0xFF00);  // 16-bit
ShiftReg595_Update();              // push to outputs

⚠️ 74HC595 ขยาย 8 outputs ต่อ 1 ตัว — ใช้ 3 GPIO pins เท่านั้น — chain ได้ไม่จำกัด — ไม่มี PWM, output HIGH/LOW เท่านั้น — กระแส output ~6mA/pin
