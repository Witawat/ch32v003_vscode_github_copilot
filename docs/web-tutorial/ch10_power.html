สำคัญ: Standby mode รีเซ็ต MCU! — ตัวแปร RAM หายหมด — ต้องเก็บข้อมูลสำคัญใน Flash/BKP register ก่อนเข้า Standby

📚 หลักการทำงาน

Sleep Mode: CPU หยุดทำงาน แต่ peripheral (UART, Timer, GPIO) ยังทำงานต่อ — ตื่นได้จาก interrupt ใดๆ — ใช้เวลาตื่น ~1µs

Standby Mode: ทุกอย่างหยุดหมด — HSI, HSE, PLL ปิด — ใช้กระแสต่ำมาก (~5µA) — ตื่นได้จาก AWU timer หรือ reset pin เท่านั้น — RAM สูญหาย!

AWU (Auto Wake-Up Unit): ตัวจับเวลาในโหมด Standby — ตั้งเวลาได้ตั้งแต่ ~16ms ถึง ~30s — เมื่อครบกำหนดจะ reset MCU

CH32V003 ไม่มี PFIC (Power Controller) แยก — ใช้ register PWR_CTLR โดยตรง — SimpleHAL จัดการให้ผ่าน API เรียบง่าย

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Sleep Mode พื้นฐาน

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    pinMode(PC0, PIN_MODE_OUTPUT);

    while(1) {
        digitalWrite(PC0, HIGH);
        Delay_Ms(100);
        digitalWrite(PC0, LOW);

        // เข้า Sleep: CPU หยุด, Peripheral ยังทำงาน
        PWR_Sleep();  // ตื่นเมื่อ SysTick interrupt (1ms)
    }
}

🔍 PWR_Sleep() — CPU หยุดทำงาน — ตื่นได้จาก interrupt ใดๆ — รวมถึง SysTick — ดังนั้นในตัวอย่างนี้จะตื่นทุก 1ms โดยอัตโนมัติ

ขั้นที่ 2: Standby + AWU (ตื่นทุกนาที)

    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
    ADC_SimpleInit();

    // ตั้งค่า AWU — ตื่นทุก 30 วินาที
    PWR_ConfigureAWU(AWU_30S);

    // เข้า Standby — RAM หาย! — ตื่นด้วย AWU
    PWR_Standby();

    // —————— ตื่นจาก Standby (reset) ——————
    // ต้อง init ทุกอย่างใหม่!

    // อ่านเซ็นเซอร์
    uint16_t adc = ADC_Read(ADC_CH_PD5);
    float temp = ADC_ToVoltage(adc, 3.3f);

    USART_Print("Woke up! Sensor: ");
    USART_PrintNum((int32_t)(temp*100));
    USART_Print("\r\n");

🔍 PWR_ConfigureAWU(period) — ตั้ง AWU timer — รองรับ AWU_16MS ถึง AWU_30S — PWR_Standby() — เข้าโหมด Standby — MCU จะ reset เมื่อตื่น — ทุกอย่างเริ่มต้นใหม่

ขั้นที่ 3: ตรวจจับการตื่นจาก Standby

    if(PWR_WasStandbyWakeup()) {
        // ตื่นจาก Standby — ข้อมูลเก่าหายหมด
        USART_Print("Woke from Standby!\r\n");
    } else {
        // Power-on reset ครั้งแรก
        USART_Print("Fresh start!\r\n");
    }

    // เก็บข้อมูลข้าม Standby ด้วย BKP register
    PWR_WriteBackup(0, wake_count);  // BKP register 0
    // ... เข้า Standby ...
    // ตื่น:
    wake_count = PWR_ReadBackup(0) + 1;

🔍 PWR_WasStandbyWakeup() — เช็คว่าตื่นจาก Standby หรือ power-on จริง — BKP Registers: 10 registers (32-bit) ที่คงค่าผ่าน Standby — ใช้เก็บ counter, config

ขั้นที่ 4: Sensor Node — วัด + ส่ง + หลับ

    // วัด VDD (แบตเตอรี่)
    float vdd = ADC_GetVDD();
    uint32_t batt_pct = (uint32_t)PWR_CalculateBatteryLife(vdd, 2.7f, 3.3f);

    // ส่งข้อมูลผ่าน USART
    USART_Print("Batt: ");
    USART_PrintNum((int32_t)batt_pct);
    USART_Print("% VDD: ");
    USART_PrintNum((int32_t)(vdd*1000));
    USART_Print("mV\r\n");

    Delay_Ms(50);  // รอ USART ส่งเสร็จ

    // เข้า Standby — ตื่นอีกที 60 วินาที
    PWR_ConfigureAWU(AWU_60S);
    PWR_Standby();

🔍 PWR_CalculateBatteryLife(vdd, vmin, vmax) — คำนวณ % แบตเตอรี่จากแรงดัน — ใช้ร่วมกับ ADC_GetVDD()

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
ตื่นจาก Standby แล้วข้อมูลหายRAM ถูกล้างตอน Standbyใช้ PWR_WriteBackup() เก็บข้อมูลสำคัญก่อนเข้า Standby
Standby แล้วไม่ตื่นเลยไม่ได้ตั้ง AWU หรือไม่มี wake sourceเรียก PWR_ConfigureAWU() ก่อน PWR_Standby()
Sleep แล้ว USART ส่งข้อมูลไม่หมดCPU หยุดก่อน USART shift register ว่างรอ USART TX complete (while(USART_GetFlagStatus(USART1, USART_FLAG_TC) == RESET)) ก่อนเข้า Sleep
Batt % เกิน 100 หรือติดลบกำหนด vmin/vmax ไม่ถูกต้องvmin คือแรงดันต่ำสุดที่อุปกรณ์ทำงานได้, vmax คือแรงดันเต็ม
AWU ตื่นถี่/ห่างเกินคาดใช้ AWU period ผิดเช็คช่วง time: AWU_16MS, AWU_128MS, AWU_1S, AWU_2S, AWU_4S, AWU_8S, AWU_16S, AWU_30S

🛠️ สร้าง Product: อุปกรณ์ใช้ถ่านได้ 1 ปี

Wireless Sensor Node — ตื่นทุก 60 วินาที อ่านอุณหภูมิ + แรงดันแบต หลับต่อ — ใช้กระแสเฉลี่ย <10µA

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"

int main(void)
{
    SystemCoreClockUpdate();

    if(PWR_WasStandbyWakeup())
    {
        // ——— ตื่นจาก Standby ———
        Timer_Init();
        USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
        ADC_SimpleInit();

        // อ่านแรงดันแบตเตอรี่
        float vdd = ADC_GetVDD();
        uint32_t batt = (uint32_t)PWR_CalculateBatteryLife(vdd, 2.7f, 3.3f);

        // อ่าน wake count จาก BKP
        uint32_t count = PWR_ReadBackup(0) + 1;

        USART_Print("Wake #");
        USART_PrintNum((int32_t)count);
        USART_Print(" Batt: ");
        USART_PrintNum((int32_t)batt);
        USART_Print("%\r\n");

        Delay_Ms(30);  // รอ USART ส่งเสร็จ

        PWR_WriteBackup(0, count);

        // ถ้าแบตต่ำกว่า 10% — กระพริบ LED เตือน
        if(batt < 10) {
            pinMode(PC0, PIN_MODE_OUTPUT);
            for(int i=0; i<5; i++) {
                digitalWrite(PC0, HIGH);
                Delay_Ms(100);
                digitalWrite(PC0, LOW);
                Delay_Ms(100);
            }
        }
    }
    else
    {
        // ——— Power-on ครั้งแรก ———
        Timer_Init();
        USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
        USART_Print("Cold boot — entering Standby cycle\r\n");
        PWR_WriteBackup(0, 0);
        Delay_Ms(30);
    }

    // เข้า Standby — ตื่นอีกที 60 วินาที
    PWR_ConfigureAWU(AWU_60S);
    PWR_Standby();

    // ไม่มีทางมาถึงบรรทัดนี้ (Standby = reset)
}

คำนวณอายุถ่าน CR2032 (225mAh):

  • Active 50ms @ 4mA = 0.2 mAs ต่อครั้ง
  • Standby 60s @ 5µA = 0.3 mAs ต่อครั้ง
  • รวม 0.5 mAs/รอบ → 225mAh ÷ 0.5mAs × 60s ≈ 27,000,000 รอบ ≈ 51 ปี (ในทางทฤษฎี)
  • ในทางปฏิบัติ: self-discharge ของถ่าน + LDO quiescent current → ~1-2 ปี

✏️ ลองทำดู

  1. เปลี่ยนจาก Standby เป็น Sleep — วัดกระแสด้วยมัลติมิเตอร์ — เปรียบเทียบ Sleep vs Standby
  2. ใช้ Button (PA1) เป็น wake source จาก Sleep — กดปุ่มแล้ว LED ติด 2 วินาที แล้วหลับต่อ
  3. เก็บข้อมูลหลายชุดใน BKP registers (0-9) — เช่น timestamp, temperature, voltage — แล้วส่งทั้งหมดเมื่อตื่น
