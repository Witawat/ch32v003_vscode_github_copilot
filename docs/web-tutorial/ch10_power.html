บทที่ 10: จัดการพลังงาน — Sleep, Standby, AWU, PVD

ระดับสูง

🔍 บน CH32V003 Standby wake = reset ระบบ — RAM และ CPU state หายหมด — main() เริ่มต้นใหม่เหมือนเพิ่งจ่ายไฟ — เก็บข้อมูลสำคัญก่อนเข้า Standby ใน Flash (บทที่ 11)

วิธีต่อทีละขั้น

  1. ต่อ CR2032 ผ่าน LDO 3.3V ไป VDD — จ่ายไฟ MCU
  2. ต่อ PC0 → R 220Ω → LED → GND — LED กระพริบเมื่อตื่น
  3. ต่อ PA1 → ปุ่มกด → GND — ปลุกจาก Sleep/Standby ด้วย EXTI
  4. ต่อ PD5 → Sensor — อ่านค่า

สำคัญ: Standby mode รีเซ็ต MCU! — RAM หายหมด — SimpleHAL และ standard peripheral ch32v00x_pwr.h/ch32v00x_rcc.h ไม่มี BKP register wrapper — ต้องเก็บข้อมูลสำคัญใน Flash (บทที่ 11) ก่อนเข้า Standby

📚 หลักการทำงาน

Sleep: CPU หยุด แต่ peripheral (UART, Timer, GPIO, SysTick) ยังทำงาน — ตื่นจาก interrupt ใดๆ — เวลาตื่น ~1µs — RAM คงอยู่

Standby: ทุกอย่างปิด (HSI/HSE/PLL) — กระแสต่ำ ~2-5µA — ตื่นจาก AWU, external EXTI บางขา หรือ reset pin — RAM สูญหาย! ระบบ reset เริ่ม main() ใหม่

AWU (Auto Wake-Up): จับเวลาใน Standby — ตั้ง ~0.5ms ถึง ~30s — ครบกำหนดจะ reset MCU — คุมด้วย prescaler (15 ค่า) + window 6-bit (0-63)

PVD (Power Voltage Detector): ตรวจจับ VDD ต่ำกว่า threshold — 8 ระดับ (2.9V-4.4V) — เช็คสถานะด้วย PWR_GetPVDStatus()

⚠ Wake source ของ Standby จริงมีแค่: (1) AWU timer — ผ่าน PWR_Standby(ms) หรือ PWR_ConfigureAWU() + PWR_EnterStandbyMode() (2) External interrupt บาง pin (PA1-PA2, PC0-PC7, PD0-PD7 — ไม่มี PA0 ในทุกแพ็กเกจของ CH32V003) (3) Reset pin — ไม่ใช่ทุก pin ปลุกได้ ต้อง config EXTI ขานั้นก่อนเรียก PWR_StandbyUntilInterrupt()

💻 เขียน Code ทีละขั้น

ขั้นที่ 1 (ex01): Sleep Mode พื้นฐาน

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    pinMode(PC0, PIN_MODE_OUTPUT);
    while(1) {
        digitalWrite(PC0, HIGH);
        Delay_Ms(100);
        digitalWrite(PC0, LOW);
        // เข้า Sleep: CPU หยุด, Peripheral ยังทำงาน
        PWR_Sleep();  // ตื่นเมื่อ interrupt ใดๆ (SysTick 1ms ก็ได้)
    }
}

🔍 PWR_Sleep() ไม่รับ argument — CPU หยุดทันที — ตื่นด้วย interrupt ใดๆ ที่ enable (SysTick, GPIO EXTI, USART RXNE, Timer) — ในตัวอย่างตื่นทุก 1ms โดย SysTick — RAM และ state คงอยู่ ทำงานต่อได้

ขั้นที่ 2 (ex02): Standby + AWU อัตโนมัติ (5 วินาที)

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"

int main(void) {
    SystemCoreClockUpdate();

    // ตื่นแล้ว = reset ระบบ → main() เริ่มใหม่ที่นี่
    if (PWR_WasStandbyWakeup()) {
        // reset ครั้งนี้เกิดจาก standby wake (AWU หรือ EXTI)
        // ถ้าจะแยกว่า AWU หรือ EXTI ต้องเช็ค EXTI pending flag เอง
    }

    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
    USART_Print("Woke up — did reset!\r\n");
    Delay_Ms(30);  // รอ USART ส่งเสร็จก่อนเข้า Standby ใหม่

    PWR_Standby(5000);  // เข้า Standby 5 วินาที — RAM หายทั้งหมด — auto-select prescaler
    // ไม่มีทางมาถึง — PWR_Standby = reset เมื่อตื่น
    while(1);
}

🔍 PWR_Standby(uint32_t timeout_ms) รับ timeout 1-30000ms (สูงสุด 30s) — auto-select prescaler+window ให้ — ตื่นแล้ว reset — ถ้าต้องการนานกว่า 30s ต้อง wake หลายครั้งแล้วนับใน Flash

ขั้นที่ 3 (ex03): Standby + AWU ตั้งค่าเอง (~1 วินาที แม่น)

    // prescaler 1024 = ~31.25ms/count — คำนวณ window จาก timeout ที่ต้องการ
    uint8_t win = PWR_AWU_CALC_WINDOW(PWR_AWU_PRESCALER_1024, 1000);  // ~1s
    PWR_ConfigureAWU(PWR_AWU_PRESCALER_1024, win);
    USART_Print("Enter Standby ~1s\r\n");
    Delay_Ms(30);

    // เข้า Standby แบบ manual — AWU ที่ตั้งไว้จะ wake หลังครบเวลา
    PWR_EnterStandbyMode(PWR_ENTRY_WFI);
    // หรือเรียก PWR_Standby(1000) สั้นๆ ที่จะทำข้างบนให้อัตโนมัติ
    while(1);  // ไม่มาถึง — reset เมื่อ wake

🔍 PWR_AWU_CALC_WINDOW(prescaler, timeout_ms) คำนวณ window จาก timeout — สูตร window = (timeout_ms × 128000) / (prescaler × 1000) — ถ้า window เกิน PWR_AWU_MAX_WINDOW (63) ต้องเพิ่ม prescaler — PWR_GetAWUTimeout(prescaler, window) ตรวจค่า ms จริงได้

ขั้นที่ 4 (ex04): Standby + External Interrupt Wake

volatile uint8_t g_btn = 0;
void btn_isr(void) { g_btn = 1; }  // ISR (ตื่นแล้ว reset จริงๆ ISR ไม่ครบทำงาน)

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);

    // config PA1 เป็น EXTI Falling — กดปุ่ม = wake
    pinMode(PA1, PIN_MODE_INPUT_PULLUP);
    attachInterrupt(digitalPinToInterrupt(PA1), btn_isr, FALLING);

    USART_Print("Enter Standby — press PA1 to wake\r\n");
    Delay_Ms(30);

    // เข้า Standby จนกว่าจะมี external interrupt (PA1/PC0-7/PD0-7)
    PWR_StandbyUntilInterrupt();  // ไม่ set AWU — wake ด้วย EXTI เท่านั้น
    // ตื่น = reset → main() เริ่มใหม่
    while(1);
}

🔍 PWR_StandbyUntilInterrupt() ไม่รับ argument — เข้า Standby จนกว่าจะมี external interrupt บนขาที่ config EXTI — ต้อง config EXTI ก่อนเรียก ด้วย attachInterrupt() ไม่งั้น MCU จะหลับตลอดกาล — ฟังก์ชันนี้ไม่ set AWU ถ้าไม่มีใครกดปุ่มจะไม่ตื่นเลย — ห้ามใช้ PA0 ไม่มีในแพ็กเกจใดของ CH32V003

ขั้นที่ 5 (ex05): ตรวจจับการตื่นจาก Standby

    if (PWR_WasStandbyWakeup()) {
        // reset ครั้งนี้เกิดจาก standby wake (AWU หรือ EXTI)
        USART_Print("Woke from Standby!\r\n");
        PWR_ClearStandbyFlag();  // เคลียร์ flag ก่อนเข้าครั้งต่อไป
    } else {
        // Power-on reset ครั้งแรก
        USART_Print("Fresh power-on!\r\n");
    }

    // ⚠ ไม่มี BKP register wrapper ใน SimpleHAL และ standard peripheral
    //    เก็บข้อมูลข้าม reset ใน Flash แทน (บทที่ 11):
    //      Flash_LoadConfig(&cfg, sizeof(cfg));

🔍 PWR_WasStandbyWakeup() เช็คว่า reset ครั้งนี้เกิดจาก standby wake หรือ power-on — อย่าลืมเรียก PWR_ClearStandbyFlag() ไม่งั้น flag ค้าง — ไม่มี BKP register ใน SimpleHAL หรือ standard peripheral — เก็บข้อมูลข้าม reset ใน Flash (บทที่ 11) เท่านั้น

ขั้นที่ 6 (ex06): PVD Monitor — ตรวจจับแรงดันต่ำ

    // เปิด PVD ที่ 3.3V — ถ้า VDD ต่ำกว่า 3.3V flag จะติด
    PWR_EnablePVD(PWR_PVD_3V3);

    while (1) {
        if (PWR_GetPVDStatus()) {
            // VDD ต่ำกว่า 3.3V — แบตกำลังจะหมด!
            USART_Print("!!! LOW VOLTAGE — safe shutdown\r\n");
            Delay_Ms(30);
            // เก็บ state ลง Flash → เข้า Standby จนกว่าจ่ายไฟใหม่
            PWR_StandbyUntilInterrupt();  // ปลุกด้วยกด RESET/EXTI เท่านั้น
        }
        digitalWrite(PC0, !digitalRead(PC0));
        Delay_Ms(500);
    }

🔍 PWR_EnablePVD(level) เปิด voltage detector ที่ threshold 8 ระดับ — PWR_GetPVDStatus() คืน 1 ถ้า VDD ต่ำกว่า threshold — ใช้ทำ safe shutdown ก่อนแบตหมด — ปิดด้วย PWR_DisablePVD()

ขั้นที่ 7 (ex07): ประเมินอายุแบต (signature 4 args จริง)

    // CR2032 1000mAh, active 1% @20mA, standby 99% @5uA → คืน hours
    uint32_t hours = PWR_CalculateBatteryLife(1000, 1, 20, 5);
    USART_Print("Battery life = ");
    USART_PrintNum((int32_t)hours);
    USART_Print(" hours\r\n");

    // สูตรภายใน (คำนวณเป็นค่าเฉลี่ยถ่วงน้ำหนัก):
    //   weighted_current_uA = active_pct * active_mA * 1000
    //                       + (100 - active_pct) * standby_uA
    //   hours = (battery_mah * 1000) / weighted_current_uA
    //   = (1000*1000) / (1*20000 + 99*5) = 1000000 / 20495 ≈ 48 ชั่วโมง

    // ประมาณการกระแส standby ก่อนคำนวณอายุจริง:
    //   uint32_t ua = PWR_EstimateStandbyCurrent(0, 1);  // PVD off, AWU on → ~5uA

🔍 สำคัญ: PWR_CalculateBatteryLife(battery_mah, active_time_percent, active_current_ma, standby_current_ua) รับ 4 arguments และคืน ค่า hours (ชั่วโมง) — ไม่ใช่ signing 3 args (vdd, vmin, vmax) และไม่ใช่ % แบต — ถ้าต้องการ % แบตจากแรงดัน ใช้ ADC_GetBatteryPercent(vdd, vmin, vmax) ของ SimpleADC แทน

ขั้นที่ 8 (ex08): Wakeup Pin Limitation (CH32V003)

    // ⚠ PWR_EnableWakeupPin() บน CH32V003 ไม่มีผล!
    //    header warning: CH32V003 ทุกแพ็กเกจไม่มีขา PA0 —
    //    ฟังก์ชันนี้เป็น no-op ที่มีไว้เพื่อ API compatible
    PWR_EnableWakeupPin();   // ไม่ทำอะไรเลย
    PWR_DisableWakeupPin();  // ไม่ทำอะไรเลย

    // ทางเลือก: ใช้ external interrupt แทน — config pin + EXTI
    pinMode(PA1, PIN_MODE_INPUT_PULLUP);   // หรือ PC1, PD2
    attachInterrupt(digitalPinToInterrupt(PA1), btn_isr, FALLING);
    PWR_StandbyUntilInterrupt();           // ปลุกด้วย EXTI บน PA1

🔍 CH32V003 ไม่มีขา PA0 ในทุกแพ็กเกจ (SOP-8/SOP-16/TSSOP-20) — PWR_EnableWakeupPin()/PWR_DisableWakeupPin() เป็น stub ที่ไม่มีผลจริง — ต้องใช้ external interrupt บน PA1-PA2, PC0-PC7, PD0-PD7 แทน — ห้ามวางแผนใช้ PA0 wake

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
ตื่นจาก Standby แล้วข้อมูลหายRAM ถูกล้างตอน Standby (reset)เก็บข้อมูลสำคัญใน Flash (บทที่ 11) ก่อนเข้า Standby — ไม่มี BKP register ใน SimpleHAL
Standby แล้วไม่ตื่นเลยไม่ได้ config AWU/EXTI ก่อนใช้ PWR_Standby(5000) (auto AWU) หรือถ้าใช้ PWR_StandbyUntilInterrupt() ต้อง config EXTI ผ่าน attachInterrupt() ก่อน
Sleep แล้ว USART ส่งข้อมูลไม่หมดCPU หยุดก่อน USART shift register ว่างรอ USART TX complete (while(USART_GetFlagStatus(USART1, USART_FLAG_TC) == RESET)) ก่อนเข้า Sleep
Battery life ออกมาผิดเพี้ยนใช้ signature 3 args ปลอม (vdd, vmin, vmax)ใช้ signature จริง 4 args PWR_CalculateBatteryLife(mah, pct, mA, uA) — คืน hours ไม่ใช่ %
คาดหวัง % แบตจาก PWR_CalculateBatteryLifeฟังก์ชันนี้คืน hours ไม่ใช่ %% แบตใช้ ADC_GetBatteryPercent(vdd, vmin, vmax) ของ SimpleADC
AWU ตื่นถี่/ห่างเกินคาดเดา macro shorthand AWU รูปแบบ time-basedเลือก PWR_AWU_PRESCALER_* + window เอง — ตรวจด้วย PWR_GetAWUTimeout() — สูงสุดคือ prescaler 61440 + window 63 ≈ 30s
เรียก PWR_EnableWakeupPin() แล้วไม่ตื่นCH32V003 ไม่มี PA0 — ฟังก์ชัน no-opใช้ EXTI บน PA1/PC1/PD2 แทน — attachInterrupt() + PWR_StandbyUntilInterrupt()
ต้องการให้ Standby นานกว่า 30sPWR_Standby(timeout_ms) สูงสุด 30000msตื่นทุก 30s แล้วนับ counter ใน Flash — ครบเท่าไหร่ค่อยทำงานจริง

Wireless Sensor Node — ตื่นทุก 30s อ่านแบต หลับต่อ — กระแสเฉลี่ย <10µA — ปลุกด้วย AWU auto

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"

typedef struct {
    uint32_t magic;
    uint32_t wake_count;
    uint16_t crc;
} PersistState;

int main(void) {
    SystemCoreClockUpdate();

    // โหลด state จาก Flash — ไม่มี BKP register ให้ใช้
    PersistState st;
    if (!Flash_LoadConfig(&st, sizeof(st)) || st.magic != 0xBEEF1234) {
        st.magic = 0xBEEF1234;
        st.wake_count = 0;
    }

    if (PWR_WasStandbyWakeup()) {
        // ——— ตื่นจาก Standby (AWU/EXTI) ———
        Timer_Init();
        USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
        ADC_SimpleInit();
        pinMode(PC0, PIN_MODE_OUTPUT);
        PWR_ClearStandbyFlag();

        st.wake_count++;

        // วัดแรงดันแบตจริง
        float vdd = ADC_GetVDD();
        float batt_pct = ADC_GetBatteryPercent(vdd, 2.0f, 3.0f);  // CR2032: 2.0-3.0V

        USART_Print("Wake #");
        USART_PrintNum((int32_t)st.wake_count);
        USART_Print("  VDD=");
        USART_PrintNum((int32_t)(vdd * 1000));
        USART_Print("mV  Batt=");
        USART_PrintNum((int32_t)batt_pct);
        USART_Print("%\r\n");

        // ประเมินอายุแบตจาก signature จริง 4 args (CR2032 1000mAh)
        //   active 1% @ 20mA, standby 99% @ 5uA → hours
        uint32_t hours = PWR_CalculateBatteryLife(1000, 1, 20, 5);
        USART_Print("  Est battery life = ");
        USART_PrintNum((int32_t)hours);
        USART_Print(" hours (");
        USART_PrintNum((int32_t)(hours / 24));
        USART_Print(" days)\r\n");

        // เก็บ state ลง flash (ระวัง wear — ทน ~10,000-80,000 รอบ/page)
        Flash_SaveConfig(&st, sizeof(st));

        // ถ้าแบตต่ำกว่า 10% — กระพริบ LED เตือน
        if (batt_pct < 10.0f) {
            for (int i = 0; i < 5; i++) {
                digitalWrite(PC0, HIGH); Delay_Ms(100);
                digitalWrite(PC0, LOW);  Delay_Ms(100);
            }
        }
        Delay_Ms(30);  // รอ USART ส่งเสร็จ
    } else {
        // ——— Power-on ครั้งแรก ———
        Timer_Init();
        USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
        USART_Print("Cold boot — entering Standby cycle\r\n");
        st.wake_count = 0;
        Flash_SaveConfig(&st, sizeof(st));
        Delay_Ms(30);
    }

    // เข้า Standby — ตื่นอีกทีใน 30 วินาที (สูงสุดของ PWR_Standby)
    PWR_Standby(30000);

    // ไม่มีทางมาถึง (Standby = reset)
    while (1);
}

คำนวณอายุ CR2032 (220mAh) ด้วย PWR_CalculateBatteryLife:

  • Active 0.05s @ 4mA ทุก 30s → active_time_percent ≈ 0.17%
  • PWR_CalculateBatteryLife(220, 0, 4, 5) ≈ 220000 / 5 = 44,000 ชม. ≈ 5 ปี (ทฤษฎี)
  • ในทางปฏิบัติ: self-discharge CR2032 + LDO quiescent (~1-2µA) → อายุจริง ~1-2 ปี
  • PWR_EstimateStandbyCurrent(0, 1) ประมาณ standby current ~5µA เมื่อเปิด AWU เท่านั้น

✏️ ลองทำดู

  1. เปลี่ยนจาก Standby เป็น Sleep — วัดกระแสด้วยมัลติมิเตอร์ — เปรียบเทียบ Sleep (~1-2mA) vs Standby (~5µA)
  2. ใช้ปุ่ม PA1 + attachInterrupt() เป็น wake source จาก PWR_StandbyUntilInterrupt() — กดปุ่มแล้ว LED ติด 2 วินาที แล้วหลับต่อ
  3. ตั้ง AWU แม่น 1s ด้วย PWR_AWU_CALC_WINDOW(PWR_AWU_PRESCALER_1024, 1000) + PWR_EnterStandbyMode() — เทียบกับ PWR_Standby(1000) ดูค่าตรงกันไหม
  4. เปิด PVD 3.3V — ลดแรงดันจ่ายไฟทีละนิด — สังเกต PWR_GetPVDStatus() เปลี่ยนจาก 0 → 1 ตอนไหน
