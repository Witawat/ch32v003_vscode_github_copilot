HX711 เป็น ADC 24-bit สำหรับ Load Cell โดยเฉพาะ — ใช้สาย 2 เส้น (DT/SCK) สื่อสารแบบ Bit-Bang — Gain 128 (Ch A) หรือ 64 (Ch B)

📚 หลักการทำงาน

HX711 เป็น Precision 24-bit Analog-to-Digital Converter ออกแบบมาสำหรับ Load Cell (Wheatstone Bridge) — มี PGA (Programmable Gain Amplifier) ในตัว — Gain 128 สำหรับ Channel A

Load Cell เป็น Wheatstone Bridge — เมื่อมีแรงกด → ความต้านทานเปลี่ยน → แรงดัน differential เปลี่ยน (mV level) — HX711 ขยายและแปลงเป็น digital 24-bit

Tare (Zeroing): น้ำหนักเปล่า (ภาชนะ/ถาด) จะให้ค่า raw offset — Tare = เก็บ offset ไว้ → Weight = (Raw - Offset) / Calibration_Factor

Calibration: ใช้ตุ้มน้ำหนักที่ทราบค่า (เช่น 100g) → Cal_Factor = (Raw_with_weight - Raw_empty) / Known_Weight

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Init HX711 + Set Gain

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "HX711.h"

HX711_Instance hx;

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_Init(115200);

    HX711_Init(&hx, PC1, PC0);  // DT=PC1, SCK=PC0
    HX711_SetGain(&hx, HX711_GAIN_128);

    USART_Println("HX711 Ready!");

🔍 HX711_Init(&hx, dt_pin, sck_pin) — DT = Data, SCK = Clock — HX711_SetGain() ตั้ง gain แยกหลัง Init — Gain 128 สำหรับ Load Cell มาตรฐาน

ขั้นที่ 2: Tare + อ่านน้ำหนัก

    USART_Println("Place empty scale...");
    Delay_Ms(3000);

    HX711_Tare(&hx, 10); // average 10 readings
    USART_Println("Tare done!");

    USART_Println("Place weight...");
    Delay_Ms(3000);

    float kg = HX711_GetWeight(&hx);
    USART_Print("Weight: ");
    USART_PrintFloat(kg, 3);
    USART_Println(" kg");

    // Calibrate: ใช้ scale factor
    // HX711_SetScale(&hx, 420.5f);

🔍 HX711_Tare(&hx, samples) — อ่านค่าเฉลี่ย samples ครั้ง → เก็บเป็น offset — HX711_GetWeight(&hx) คืนน้ำหนักเป็นกิโลกรัม

ขั้นที่ 3: Loop + ปุ่ม Tare

    pinMode(PC2, PIN_MODE_INPUT_PULLUP); // ปุ่ม Tare

    while(1) {
        float kg = HX711_GetWeight(&hx);

        USART_Print("Weight: ");
        USART_PrintFloat(kg, 3);
        USART_Println(" kg");

        if(GPIO_Read(PC2) == 0) {
            Delay_Ms(50);
            HX711_Tare(&hx, 10);
            USART_Println("TARE pressed!");
            while(GPIO_Read(PC2) == 0);
        }
        Delay_Ms(200);
    }
}

🔍 HX711_GetWeight(&hx) — อ่านน้ำหนักหน่วย kg — กดปุ่ม Tare เพื่อลบน้ำหนักภาชนะ — อ่านทุก 200ms

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
อ่านค่าได้ 0 ตลอดต่อสายผิด / Load Cell ไม่ต่อ / DT/SCK สลับตรวจสอบสีสาย (Red=E+, Black=E-, Green=A+, White=A-) — DT=Data, SCK=Clock
ค่า Jump ไปมา ไม่นิ่งสายยาว / Noise / Sample rate สูงใช้ shield cable — ต่อ 0.1µF cap ที่ VCC
น้ำหนักไม่ linearCalibration ผิด / Load Cell เสียหายCalibrate ด้วยตุ้มน้ำหนักหลายค่า (50g, 100g, 200g)
Tare แล้วน้ำหนักไม่เป็น 0อ่าน samples น้อยเกิน / Load Cell ยังสั่นใช้ samples ≥ 10 — รอให้ stable ก่อน Tare
น้ำหนัก drift ตามเวลาLoad cell creep / อุณหภูมิเปลี่ยนใช้ Tare เป็นระยะ — ใช้ Load Cell ที่มี temperature compensation

🛠️ สร้าง Product: Digital Kitchen Scale

HX711 + Load Cell 5kg + OLED + ปุ่ม Tare

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "HX711.h"
#include "oled_i2c.h"

#define TARE_BTN PC2

int main(void)
{
    SystemCoreClockUpdate();
    Timer_Init();
    I2C_Init(PC1, PC2, I2C_SPEED_STANDARD);
    USART_Init(115200);

    HX711_Instance hx;
    HX711_Init(&hx, PC1, PC0);
    HX711_SetGain(&hx, HX711_GAIN_128);

    OLED_Handle oled;
    OLED_Init(&oled, OLED_128x64, 0x3C);
    OLED_SetFont(&oled, FONT_8x16);

    pinMode(TARE_BTN, PIN_MODE_INPUT_PULLUP);

    USART_Println("Auto-Tare...");
    Delay_Ms(1000);
    HX711_Tare(&hx, 15);
    USART_Println("Scale Ready");

    char buf[64];
    while(1)
    {
        float kg = HX711_GetWeight(&hx);

        OLED_Clear(&oled);
        snprintf(buf, sizeof(buf), "Weight:");
        OLED_DrawStringAlign(&oled, 0, 10, buf, ALIGN_LEFT);

        if(kg < 0.001f) {
            snprintf(buf, sizeof(buf), "  0.000 kg");
        } else {
            snprintf(buf, sizeof(buf), "%.3f kg", kg);
        }
        OLED_DrawStringAlign(&oled, 0, 30, buf, ALIGN_LEFT);

        snprintf(buf, sizeof(buf), "Press to TARE");
        OLED_DrawStringAlign(&oled, 0, 60, buf, ALIGN_LEFT);

        if(GPIO_Read(TARE_BTN) == 0) {
            Delay_Ms(100);
            HX711_Tare(&hx, 15);
            OLED_Update(&oled);
            Delay_Ms(500);
            while(GPIO_Read(TARE_BTN) == 0);
        }

        OLED_Update(&oled);
        Delay_Ms(200);
    }
}
