HX711 เป็น ADC 24-bit สำหรับ Load Cell โดยเฉพาะ — ใช้สาย 2 เส้น (DT/SCK) สื่อสารแบบ Bit-Bang — Gain 128 (Ch A) หรือ 64 (Ch B) — Sample Rate 10 SPS หรือ 80 SPS

📚 หลักการทำงาน

HX711 เป็น Precision 24-bit Analog-to-Digital Converter ออกแบบมาสำหรับ Load Cell (Wheatstone Bridge) โดยเฉพาะ — มี PGA (Programmable Gain Amplifier) ในตัว — Gain 128 สำหรับ Channel A (Load Cell หลัก) หรือ Gain 64 สำหรับ Channel B

Load Cell เป็น Wheatstone Bridge — เมื่อมีแรงกด → ความต้านทานเปลี่ยน → แรงดัน differential เปลี่ยน (mV level) — HX711 ขยายและแปลงเป็น digital 24-bit — ค่า raw อยู่ระหว่าง -8,388,608 ถึง 8,388,607

Tare (Zeroing): น้ำหนักเปล่า (ภาชนะ/ถาด) จะให้ค่า raw offset — Tare = เก็บ offset ไว้ → Weight = (Raw - Offset) / Calibration_Factor — กดปุ่ม Tare เพื่อลบน้ำหนักภาชนะ

Calibration: ใช้ตุ้มน้ำหนักที่ทราบค่า (เช่น 100g) — อ่าน raw difference → Cal_Factor = (Raw_with_weight - Raw_empty) / Known_Weight — เก็บ Cal_Factor ไว้ใช้คำนวณน้ำหนักจริง

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Init HX711 + Set Gain

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "HX711.h"

HX711_t scale;

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_Init(115200);

    HX711_Init(&scale, PC2, PC3, HX711_GAIN_128);
    // DT=PC2, SCK=PC3, Gain=128

    USART_Print("HX711 Ready\r\n");
    USART_Print("Place empty scale...\r\n");
    Delay_Ms(3000);

🔍 HX711_Init(dev, dt_pin, sck_pin, gain) — DT = Data Output (input to MCU) — SCK = Clock (output from MCU) — Gain 128 สำหรับ Load Cell มาตรฐาน

ขั้นที่ 2: Tare + Calibrate

    // Tare — วัดน้ำหนักเปล่า
    HX711_Tare(&scale, 10); // average 10 readings
    USART_Print("Tare done. Offset: ");
    USART_PrintDec(scale.offset);
    USART_Print("\r\n");

    // Calibrate — วางตุ้มน้ำหนัก 100g
    USART_Print("Place 100g weight...\r\n");
    Delay_Ms(3000);
    HX711_Calibrate(&scale, 100.0f, 10);
    USART_Print("Cal done. Factor: ");
    USART_PrintFloat(scale.cal_factor, 4);
    USART_Print("\r\n");

🔍 HX711_Tare(dev, samples) — อ่านค่าเฉลี่ย 10 ครั้ง → เก็บเป็น offset — HX711_Calibrate(dev, known_weight, samples) — ใช้ตุ้มน้ำหนักที่รู้ค่า (เช่น 100g) — คำนวณ calibration factor อัตโนมัติ

ขั้นที่ 3: อ่านน้ำหนัก + แสดงผล

    pinMode(PC0, PIN_MODE_INPUT_PULLUP); // ปุ่ม Tare

    while(1) {
        float weight = HX711_ReadWeight(&scale, 5); // avg 5 samples

        USART_Print("Weight: ");
        USART_PrintFloat(weight, 1);
        USART_Print(" g");

        if(weight < 0.5f) {
            USART_Print(" (empty)");
        }
        USART_Print("\r\n");

        // กดปุ่ม Tare
        if(GPIO_Read(PC0) == 0) {
            Delay_Ms(50);
            HX711_Tare(&scale, 10);
            USART_Print("TARE pressed!\r\n");
            while(GPIO_Read(PC0) == 0);
        }

        Delay_Ms(200);
    }
}

🔍 HX711_ReadWeight(dev, samples) — อ่านค่าเฉลี่ย samples ครั้ง → คำนวณน้ำหนักจาก (raw - offset) / cal_factor — แสดงหน่วยกรัม — ปุ่ม Tare สำหรับลบน้ำหนักภาชนะ

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
อ่านค่าได้ 0 หรือ -8388608 ตลอดต่อสายผิด / Load Cell ไม่ต่อ / DT/SCK สลับตรวจสอบสีสาย (Red=E+, Black=E-, Green=A+, White=A-) — DT=Data, SCK=Clock
ค่า Jump ไปมา ไม่นิ่งสายยาว / Noise / Sample rate สูงใช้ averaging 10-20 samples — ใช้ 10 SPS (RATE=LOW) — shield cable — ต่อ 0.1µF cap ที่ VCC
น้ำหนักไม่ linearCalibration ผิด / Load Cell เสียหายCalibrate ด้วยตุ้มน้ำหนักหลายค่า (50g, 100g, 200g) — ตรวจสอบ Overload
Tare แล้วน้ำหนักไม่เป็น 0อ่าน samples น้อยเกิน / Load Cell ยังสั่นใช้ samples ≥ 10 — รอให้ stable ก่อน Tare — ใช้ filter moving average
DT ค้าง HIGH ตลอดHX711 ยังไม่พร้อม (data not ready)รอจน DT = LOW (ready) — ถ้าค้าง HIGH > 100ms → power cycle HX711
น้ำหนัก drift ตามเวลาLoad cell creep / อุณหภูมิเปลี่ยนใช้ Tare เป็นระยะ — ใช้ Load Cell ที่มี temperature compensation

🛠️ สร้าง Product: Digital Kitchen Scale

HX711 + Load Cell 5kg + OLED + ปุ่ม Tare — แสดงน้ำหนัก g/kg — Auto Tare ตอนเปิด

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "HX711.h"
#include "OLED.h"

#define DT_PIN PC2
#define SCK_PIN PC3
#define TARE_BTN PC0
#define CAL_FACTOR 420.5f // calibrate เอง!

int main(void)
{
    SystemCoreClockUpdate();
    Timer_Init();
    I2C_Init(PC1, PC2, I2C_SPEED_STANDARD);
    USART_Init(115200);

    HX711_t scale;
    HX711_Init(&scale, DT_PIN, SCK_PIN, HX711_GAIN_128);
    HX711_SetScale(&scale, CAL_FACTOR);

    OLED_t oled;
    OLED_Init(&oled, 0x3C, OLED_128x64);
    OLED_SetFont(&oled, FONT_8x16);

    pinMode(TARE_BTN, PIN_MODE_INPUT_PULLUP);

    USART_Print("Auto-Tare...\r\n");
    Delay_Ms(1000);
    HX711_Tare(&scale, 15);
    USART_Print("Scale Ready\r\n");

    while(1)
    {
        float weight = HX711_ReadWeight(&scale, 8);

        OLED_Clear(&oled);
        OLED_SetCursor(&oled, 0, 1);
        OLED_Print(&oled, "Weight:");

        OLED_SetCursor(&oled, 0, 3);
        if(fabs(weight) < 0.5f) {
            OLED_Print(&oled, "  0.0 g");
        } else if(weight > 1000.0f) {
            OLED_PrintFloat(&oled, "", weight / 1000.0f, 3);
            OLED_Print(&oled, " kg");
        } else {
            OLED_PrintFloat(&oled, "", weight, 1);
            OLED_Print(&oled, " g");
        }

        OLED_SetCursor(&oled, 0, 6);
        OLED_Print(&oled, "Press to TARE");

        if(GPIO_Read(TARE_BTN) == 0) {
            Delay_Ms(100);
            HX711_Tare(&scale, 15);
            OLED_SetCursor(&oled, 0, 6);
            OLED_Print(&oled, "TARED!      ");
            OLED_Display(&oled);
            Delay_Ms(500);
            while(GPIO_Read(TARE_BTN) == 0);
        }

        OLED_Display(&oled);
        Delay_Ms(200);
    }
}

✏️ ลองทำดู

  1. Calibrate ด้วยตุ้มน้ำหนักหลายค่า — ตรวจสอบ linearity — ปรับใช้ piecewise calibration ถ้าไม่ linear
  2. ทำ Counting Scale — ชั่งน้ำหนัก 1 ชิ้น → คำนวณจำนวนชิ้นจากน้ำหนักรวม — แสดง count บน OLED
  3. ทำ Auto-Sleep — ถ้าไม่มีน้ำหนักเปลี่ยนเกิน 5 นาที → HX711_PowerDown() → ปลุกด้วยปุ่ม Tare
  4. บันทึกน้ำหนักลง W25Qxx Flash — เก็บ log ชั่งน้ำหนักย้อนหลัง — ทำกราฟน้ำหนักรายวัน
