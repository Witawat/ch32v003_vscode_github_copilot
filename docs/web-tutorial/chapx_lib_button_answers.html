📝 เฉลย — Button

ข้อ 1: Single-Button Menu — short=next, long=select, double=back

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "Button.h"

typedef enum { MENU_MAIN, MENU_LED, MENU_SENSOR, MENU_COUNT } Menu_t;

const char *items[] = {"1. LED Control", "2. Read Sensor", "3. Settings"};
uint8_t cur = 0;
Menu_t submenu = MENU_MAIN;
Button_Instance btn;

void btn_short_cb(void) { cur = (cur + 1) % MENU_COUNT; }
void btn_long_cb(void)  { submenu = cur; }
void btn_double_cb(void) { if(submenu != MENU_MAIN) submenu = MENU_MAIN; }

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_Init(115200);
    Button_Init(&btn, PC0, BUTTON_ACTIVE_LOW);

    Button_SetCallback(&btn, BUTTON_EVENT_PRESS,        btn_short_cb);
    Button_SetCallback(&btn, BUTTON_EVENT_LONG_PRESS,   btn_long_cb);
    Button_SetCallback(&btn, BUTTON_EVENT_DOUBLE_CLICK, btn_double_cb);

    USART_Println("Menu: short=next, long=select, double=back");

    while(1) {
        Button_Process();

        static uint8_t last_cur = 255, last_sub = 255;
        if(cur != last_cur || submenu != last_sub) {
            USART_Print("\nMenu> ");
            if(submenu == MENU_MAIN) for(uint8_t i = 0; i < MENU_COUNT; i++)
                USART_Printf("%s%s", cur == i ? " ->[" : "    ", items[i]);
            else USART_Printf("[%s selected — double=back]", items[submenu]);
            last_cur = cur; last_sub = submenu;
        }
    }
}
🧠 แนวคิด: ใช้ Button_AttachEvent ผูก callback 3 แบบ — short เลื่อน cursor, long เข้า submenu, double กลับ menu หลัก — ใช้ enum Menu_t เก็บสถานะ — พิมพ์เฉพาะเมื่อ state เปลี่ยนด้วยการเทียบ last_cur/last_sub ลด serial spam — Button_Process() ใน loop คอยอัปเดต state machine ภายใน

ข้อ 2: Double-Click Counter — นับจำนวนกด 2 ครั้งติด

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "Button.h"

volatile uint16_t dbl_cnt = 0;
Button_Instance btn;

void on_double(void) { dbl_cnt++; }

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_Init(115200);

    Button_Init(&btn, PC0, BUTTON_ACTIVE_LOW);
    Button_SetCallback(&btn, BUTTON_EVENT_DOUBLE_CLICK, on_double);

    Timer_t tmr;
    Start_Timer(&tmr, 1000, 1);

    USART_Println("Double-click to count. Press DEL to reset.");

    while(1) {
        if(Is_Timer_Expired(&tmr)) {
            USART_Printf("Double-clicks: %d\n", dbl_cnt);
        }

        if(USART_Available()) {
            char c = USART_Read();
            if(c == 127 || c == 'r') dbl_cnt = 0;  // DEL or 'r' resets
        }
    }
}
🧠 แนวคิด: BTN_EVENT_DOUBLE จับ double-click โดยตรง — counter เป็น volatile uint16_t เพราะถูกแก้ไขใน ISR — แสดงผลทุก 1 วิด้วย Is_Timer_Expired — เพิ่ม serial command 'r' หรือ DEL เพื่อรีเซ็ต counter — ถ้าต้องการหลายปุ่ม แค่ Init คนละขาแล้ว AttachEvent แยกกัน

ข้อ 3: Hold-to-Dim LED — กดค้างแล้ว LED ค่อย ๆ หรี่

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "Button.h"

#define LED_PIN   PC1
#define BTN_PIN   PC0
#define PWM_CH    1

volatile uint16_t duty = 1000;
volatile bool holding = false;
Button_Instance btn;

void btn_down(void)  { holding = true; }
void btn_up(void)    { holding = false; }

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    Button_Init(&btn, BTN_PIN, BUTTON_ACTIVE_LOW);

    Button_SetCallback(&btn, BUTTON_EVENT_DOWN, btn_down);
    Button_SetCallback(&btn, BUTTON_EVENT_UP,   btn_up);

    PWM_SimpleInit(LED_PIN, PWM_CH, 1000, 1000);  // เริ่มสว่างสุด
    pinMode(PA0, PIN_MODE_OUTPUT);
    digitalWrite(PA0, HIGH);  // LED on-board แสดงสถานะ

    Timer_t dim_tmr;
    Start_Timer(&dim_tmr, 50, 1);

    while(1) {
        if(holding && Is_Timer_Expired(&dim_tmr)) {
            if(duty > 0) duty -= 10;           // ลด duty ทีละ 1% ทุก 50ms
            PWM_SetDuty(LED_PIN, PWM_CH, duty);
        }

        if(!holding && duty < 1000) {
            duty = 1000;                         // ปล่อยปุ่ม → สว่างเต็มทันที
            PWM_SetDuty(LED_PIN, PWM_CH, duty);
        }
    }
}
🧠 แนวคิด: ใช้ BTN_EVENT_DOWN/BTN_EVENT_UP จับสถานะกด/ปล่อย — flag holding บอกว่ากำลังกดค้าง — Timer 50ms ลด duty cycle ทีละ 10 (1%) — จาก 1000→0 ใช้เวลา ~5 วิ — เมื่อปล่อยปุ่ม duty กลับ 100% ทันที — PWM_SimpleInit ตั้งความถี่ 1kHz แสงไม่กะพริบ

ข้อ 4: Combo Lock — ปลดล็อกด้วยลำดับกด (สั้น-ยาว-สั้น)

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "Button.h"

#define LOCK_PIN  PC2
#define BTN_PIN   PC0

const uint8_t combo[] = {BUTTON_EVENT_PRESS, BUTTON_EVENT_LONG_PRESS, BUTTON_EVENT_PRESS};
#define COMBO_LEN 3
uint8_t input_buf[COMBO_LEN];
uint8_t idx = 0;
Button_Instance btn;

void btn_event_cb(uint8_t event) {
    input_buf[idx++] = event;
    if(idx >= COMBO_LEN) idx = 0;  // wrap around
}

bool check_combo(void) {
    for(uint8_t i = 0; i < COMBO_LEN; i++)
        if(input_buf[i] != combo[i]) return false;
    return true;
}

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_Init(115200);
    Button_Init(&btn, BTN_PIN, BUTTON_ACTIVE_LOW);

    Button_SetCallback(&btn, BUTTON_EVENT_PRESS,      btn_event_cb);
    Button_SetCallback(&btn, BUTTON_EVENT_LONG_PRESS, btn_event_cb);

    pinMode(LOCK_PIN, PIN_MODE_OUTPUT);
    digitalWrite(LOCK_PIN, LOW);

    USART_Println("Enter combo: SHORT→LONG→SHORT");

    Timer_t tmr;
    Start_Timer(&tmr, 5000, 1);   // auto-reset ทุก 5 วิ

    while(1) {
        if(Is_Timer_Expired(&tmr)) {
            idx = 0;                              // หมดเวลา → รีเซ็ต input
            USART_Println("Timeout! Try again.");
        }

        if(idx == 0 && input_buf[2] != 0) {       // input เต็มแล้ววนกลับมา
            if(check_combo()) {
                digitalWrite(LOCK_PIN, HIGH);     // ปลดล็อก
                USART_Println("UNLOCKED!");
                Delay_Ms(3000);
                digitalWrite(LOCK_PIN, LOW);
            }
            memset(input_buf, 0, COMBO_LEN);
            idx = 0;
        }
    }
}
🧠 แนวคิด: เก็บ combo[] เป็น array ของ event type — btn_event_cb รับ event ทุกแบบมาเติม input_buf แบบ circular — เมื่อครบ 3 ครั้ง เรียก check_combo() เทียบทีละ element — ใช้ Timer 5 วิ auto-reset ถ้า user กดไม่ครบ — memset ล้าง buffer หลังจากเช็คแล้ว — ถ้าถูกต้อง เปิด relay/LED 3 วิแล้วปิด — สำหรับ combo ยาวขึ้น แค่เพิ่ม COMBO_LEN และขยาย array
