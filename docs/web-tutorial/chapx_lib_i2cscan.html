I2C Scanner — ตรวจจับอุปกรณ์บนบัส I2C

ระดับเริ่มต้น

I2C Scanner เป็นเครื่องมือพื้นฐานที่สุด — ใช้หา address ของอุปกรณ์ I2C ทุกตัวบนบัส, ใช้ debug ตอนต่อวงจรใหม่, ใช้ตรวจสอบว่าบัส I2C สื่อสารได้หรือไม่, ใช้เช็ค address conflict (2 อุปกรณ์ใช้ address เดียวกัน)

📚 หลักการทำงาน

I2C Scanner ทำงานโดยส่ง I2C Start condition + Address + R/W bit ไปยังทุก address ตั้งแต่ 1 ถึง 127 — ถ้าอุปกรณ์ตอบ ACK (ดึง SDA LOW ระหว่าง bit 9) แสดงว่ามีอุปกรณ์ที่ address นั้น, ถ้า NACK (SDA HIGH) แสดงว่าไม่มีอุปกรณ์

Address Range:

  • 0x00-0x07: Reserved — 10-bit address และ special purposes
  • 0x08-0x77: Normal device addresses (1-127)
  • 0x78-0x7F: Reserved — 10-bit address และ broadcast

7-bit vs 8-bit Address: I2C address จริงๆ คือ 7-bit (0-127) — แต่บาง datasheet แสดงเป็น 8-bit (shift ซ้าย 1 บิต + R/W bit) เช่น address 0x68 (7-bit) = 0xD0 (8-bit write), 0xD1 (8-bit read) — I2C Scanner มักแสดงค่าทั้งสองแบบ

ประโยชน์:

  • ตรวจสอบว่าอุปกรณ์ต่อถูกต้อง — ถ้า scan ไม่เจอ แสดงว่าสายหลุดหรือ address ไม่ตรง
  • ตรวจจับ address conflict — ถ้า 2 อุปกรณ์ใช้ address เดียวกัน I2C จะพัง
  • ทดสอบ pull-up resistors — ถ้าไม่มี pull-up, I2C จะไม่ทำงาน

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Init + Setup

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "I2C_Scanner.h"

int main(void) {
    SystemCoreClockUpdate();
    I2C_Init(I2C1, I2C_SPEED_STANDARD);  // 100kHz
    USART_Init(USART1, BAUD_115200);
    USART_Println("=== I2C Scanner ===");
    USART_Println("Scanning addresses 1-127...");

🔍 ใช้ I2C speed 100kHz (standard) — ปลอดภัยกับทุกอุปกรณ์, USART สำหรับแสดงผล, ไม่ต้องต่ออะไรพิเศษนอกจาก pull-up 4.7k ที่ SDA/SCL

ขั้นที่ 2: สแกนหาอุปกรณ์

    uint8_t found = 0;
    for(uint8_t addr = 1; addr < 128; addr++) {
        I2C_Start(I2C1);
        I2C_WriteByte(I2C1, (addr << 1) | 0);  // 7-bit addr + W=0
        if(I2C_CheckACK(I2C1)) {  // ACK = device present!
            USART_Print("Device found at 0x");
            USART_PrintHex(addr);  // 7-bit address
            USART_Print(" (8-bit: 0x");
            USART_PrintHex(addr << 1);  // 8-bit write
            USART_Print(" / 0x");
            USART_PrintHex((addr << 1) | 1);  // 8-bit read
            USART_Println(")");
            found++;
        }
        I2C_Stop(I2C1);
        Delay_Ms(1);  // brief pause between scans
    }

🔍 I2C_WriteByte(i2c, addr << 1) — shift 7-bit address ไปเป็น 8-bit (bit 0 = R/W), I2C_CheckACK(i2c) — ตรวจสอบว่าได้รับ ACK หรือไม่ (ACK = มีอุปกรณ์), delay 1ms ระหว่างแต่ละ scan เพื่อไม่ให้อุปกรณ์สับสน

ขั้นที่ 3: แสดงสรุป + I2C_Scan() ฟังก์ชันสำเร็จรูป

    USART_Print("Scan complete. Found ");
    USART_PrintInt(found); USART_Println(" device(s).");

    if(found == 0) {
        USART_Println("No I2C devices found!");
        USART_Println("Check:");
        USART_Println("  1. SDA/SCL wiring");
        USART_Println("  2. Pull-up resistors (4.7k)");
        USART_Println("  3. Power to devices (3.3V)");
    }

    // === ใช้ฟังก์ชัน I2C_Scan() สำเร็จรูป ===
    // I2C_Scan(I2C1);  // พิมพ์ผลลัพธ์อัตโนมัติผ่าน USART

    while(1) {
        // ทำงานอื่นต่อ...
        Delay_Ms(1000);
    }
}

🔍 I2C_Scan(I2C1) — สแกน I2C อัตโนมัติและพิมพ์ผลลัพธ์, ถ้าไม่พบอุปกรณ์เลย → เช็ค 3 อย่าง: สาย, pull-up, ไฟเลี้ยง, แนะนำให้รัน I2C Scanner เป็นขั้นตอนแรกทุกครั้งที่ต่ออุปกรณ์ I2C ใหม่

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
Scan ไม่เจออุปกรณ์เลยลืมต่อ pull-up resistors (4.7k)ต่อ 4.7kΩ จาก SDA→3.3V และ SCL→3.3V (ถ้าไม่มีบนโมดูล)
เจอ address ลึกลับที่ไม่รู้จักNoise หรือสายยาวเกินไปใช้สายสั้น (<20cm), ลด speed เหลือ 50kHz, เพิ่ม capacitor 100nF ที่ VCC
เจอแต่บาง address ไม่ครบBus capacitance สูง — SDA/SCL ขึ้นไม่ทันลด pull-up เหลือ 2.2k หรือ 1k, ลดความเร็ว I2C
Address conflict (2 devices)2 อุปกรณ์ใช้ address เดียวกันเปลี่ยน address ด้วย A0/A1/A2 pins — เช็ค datasheet ว่าปรับได้ไหม
Scan แล้ว I2C bus ค้างอุปกรณ์บางตัว SDA stuck LOWส่ง clock pulses 9-16 ครั้งเพื่อให้อุปกรณ์ปล่อย SDA, แล้วลองใหม่

🛠️ สร้าง Product: I2C Bus Monitor + Device Checker

สแกน I2C ทุก 5 วินาที — แสดงอุปกรณ์ที่ต่ออยู่, ตรวจจับเมื่ออุปกรณ์ถูกถอด/เพิ่ม

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"

int main(void)
{
    SystemCoreClockUpdate();
    I2C_Init(I2C1, I2C_SPEED_STANDARD);
    USART_Init(USART1, BAUD_115200);

    uint8_t prev_list[128] = {0};  // previous scan results
    uint8_t curr_list[128] = {0};

    USART_Println("I2C Bus Monitor");

    // Initial scan
    I2C_Scan(I2C1);

    while(1)
    {
        // Scan current devices
        uint8_t count = 0;
        for(uint8_t a=1; a<128; a++) {
            I2C_Start(I2C1);
            I2C_WriteByte(I2C1, (a << 1) | 0);
            if(I2C_CheckACK(I2C1)) curr_list[a] = 1;
            else curr_list[a] = 0;
            I2C_Stop(I2C1);
            Delay_Ms(1);
        }

        // Detect changes
        for(uint8_t a=1; a<128; a++) {
            if(curr_list[a] && !prev_list[a]) {
                USART_Print("NEW device: 0x");
                USART_PrintHex(a); USART_Println("");
            }
            if(!curr_list[a] && prev_list[a]) {
                USART_Print("REMOVED device: 0x");
                USART_PrintHex(a); USART_Println("");
            }
        }

        // Current status
        uint8_t dev_count = 0;
        for(uint8_t a=1; a<128; a++) if(curr_list[a]) dev_count++;
        USART_Print("Devices on bus: ");
        USART_PrintInt(dev_count); USART_Println("");

        // Save state for next round
        for(uint8_t a=1; a<128; a++) prev_list[a] = curr_list[a];

        Delay_Ms(5000);
    }
}

✏️ ลองทำดู

  1. ทำ I2C device enumerator — ต่ออุปกรณ์หลายตัว → scan → แสดงชื่ออุปกรณ์จาก lookup table (เช่น 0x68=DS3231, 0x3C=OLED)
  2. ทำ I2C conflict detector — เช็คว่ามี address ซ้ำกันไหม → แจ้งเตือนถ้า conflict
  3. ทำ I2C bus diagnostic — วัด bus capacitance, ตรวจจับ stuck bus, แนะนำ pull-up value ที่เหมาะสม
