📝 เฉลย — VL53L0X ToF Laser Distance

ข้อ 1: Continuous Measurement พร้อม Threshold Alert

#define DIST_NEAR    100   // mm
#define DIST_MEDIUM  500
#define DIST_FAR     1200

void vl53l0x_continuous_alert(void) {
    VL53L0X_Instance tof;
    VL53L0X_Init(&tof, 0x29);
    VL53L0X_StartContinuous(&tof);

    pinMode(PC4, PIN_MODE_OUTPUT);
    pinMode(PC5, PIN_MODE_OUTPUT);
    pinMode(PC3, PIN_MODE_OUTPUT);

    while(1) {
        uint16_t dist = VL53L0X_ReadRangeMM(&tof);

        if(dist > DIST_FAR || dist == 0) {
            USART_Println("Out of range");
            digitalWrite(PC4, LOW);
            digitalWrite(PC5, LOW);
            digitalWrite(PC3, HIGH); Delay_Ms(100);
            digitalWrite(PC3, LOW);
        } else {
            USART_Print("Distance: ");
            USART_PrintNum(dist);
            USART_Println(" mm");

            digitalWrite(PC4, dist <= DIST_NEAR);          // ใกล้ → แดง
            digitalWrite(PC5, dist > DIST_NEAR && dist <= DIST_MEDIUM); // ปานกลาง → เหลือง
            digitalWrite(PC3, dist > DIST_MEDIUM && dist <= DIST_FAR);  // ไกล → เขียว
        }
        Delay_Ms(50);
    }
}
🧠 แนวคิด: VL53L0X_StartContinuous เปิดโหมดวัดต่อเนื่อง — VL53L0X_ReadContinuous อ่านค่าโดยไม่ต้อง init รอบใหม่ เร็วกว่า — 3 LED ตามระยะ — dist=0 มักเกิดจาก target ไกลเกิน 2m หรือ surface มืดเกินไป

ข้อ 2: Threshold Alert พร้อม Hysteresis

#define ALARM_THRESHOLD   300  // mm
#define HYSTERESIS        20

uint8_t alarm_active = 0;

void vl53l0x_threshold_hysteresis(void) {
    VL53L0X_Instance tof;
    VL53L0X_Init(&tof, 0x29);
    VL53L0X_StartContinuous(&tof);
    pinMode(PC0, PIN_MODE_OUTPUT);  // alarm relay/LED

    while(1) {
        uint16_t dist = VL53L0X_ReadRangeMM(&tof);
        if(dist == 0) { Delay_Ms(30); continue; }

        if(!alarm_active && dist < ALARM_THRESHOLD - HYSTERESIS) {
            alarm_active = 1;
            digitalWrite(PC0, HIGH);
            USART_Print("ALARM ON  @ "); USART_PrintNum(dist); USART_Println("mm");
        } else if(alarm_active && dist > ALARM_THRESHOLD + HYSTERESIS) {
            alarm_active = 0;
            digitalWrite(PC0, LOW);
            USART_Print("ALARM OFF @ "); USART_PrintNum(dist); USART_Println("mm");
        }

        USART_PrintNum(dist); USART_Print("mm alarm="); USART_PrintNum(alarm_active);
        USART_Println("");
        Delay_Ms(100);
    }
}
🧠 แนวคิด: Hysteresis ป้องกัน alarm เปิด-ปิดกระพริบที่ค่า threshold — ON ที่ 280mm แต่ OFF ที่ 320mm สร้าง deadband 40mm — ใช้ state variable alarm_active แทนการเช็คค่าเดียว — รูปแบบนี้ใช้ได้กับเซนเซอร์ทุกชนิดที่มี noise ใกล้ threshold

ข้อ 3: Indoor vs Outdoor Accuracy

void vl53l0x_accuracy_test(void) {
    VL53L0X_Instance tof;
    VL53L0X_Init(&tof, 0x29);
    VL53L0X_StartContinuous(&tof);

    USART_Println("Place at known distances: 100, 200, 500, 1000 mm");
    USART_Println("Test in indoor & outdoor condition");
    USART_Println("Expected(mm), Indoor(mm), Outdoor(mm)");

    const uint16_t targets[] = {100, 200, 500, 1000, 1500};

    for(int t = 0; t < 5; t++) {
        USART_Print("Place target at ");
        USART_PrintNum(targets[t]);
        USART_Println(" mm — then press Enter");
        while(USART_Available() == 0) {
            // แสดงค่าปัจจุบันรอผู้ใช้กด
            uint16_t d = VL53L0X_ReadRangeMM(&tof);
            USART_PrintNum(d); USART_Print(" mm\r");
            Delay_Ms(100);
        }
        USART_Read();  // consume Enter

        // เก็บ 20 samples
        float sum = 0; int valid = 0;
        for(int i = 0; i < 20; i++) {
            uint16_t d = VL53L0X_ReadRangeMM(&tof);
            if(d > 0) { sum += d; valid++; }
            Delay_Ms(30);
        }
        float avg = valid > 0 ? sum / valid : 0;

        USART_PrintNum(targets[t]); USART_Print(", ");
        USART_PrintFloat(avg, 1);
        USART_Println("");
    }
    USART_Println("Done — repeat in other lighting, copy to Excel");
}
🧠 แนวคิด: VL53L0X ใช้ infrared 940nm — แสงแดดมี IR เยอะ → outdoor accuracy ลดลงที่ระยะไกล — indoor แม่นยำสูงสุด — ทดสอบหลายระยะและสภาพแสงแล้ว export CSV — 940nm ทะลุกระจกไม่ได้ ถ้าวัดผ่าน glass cover ต้อง calibrate ใหม่ — ข้อดี VL53L0X: ไม่สนสีและพื้นผิว target เท่า ultrasonic
