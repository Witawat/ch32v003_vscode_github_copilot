📝 เฉลย — AT24Cxx EEPROM

ข้อ 1: Config Store — เก็บ WiFi SSID/PASS ใน EEPROM (WriteArray + CRC16 + out-param)

#define EE_CFG_ADDR   0x0000
#define EE_MAGIC      0xA55A
#define SSID_MAX_LEN  32
#define PASS_MAX_LEN  32

typedef struct {
    uint16_t magic;
    uint16_t crc;
    char     ssid[SSID_MAX_LEN];
    char     pass[PASS_MAX_LEN];
    uint8_t  brightness;
    uint8_t  reserved[15];   // รวม 96 bytes
} ee_config_t;

ee_config_t cfg;
AT24Cxx_Instance eeprom;

uint16_t crc16(const uint8_t *data, uint16_t len) {
    uint16_t crc = 0xFFFF;
    while (len--) {
        crc ^= (uint16_t)(*data++) << 8;
        for (uint8_t i = 0; i < 8; i++)
            crc = (crc & 0x8000) ? (crc << 1) ^ 0x1021 : crc << 1;
    }
    return crc;
}

// บันทึก: WriteArray คืน AT24Cxx_Status — เช็ค error
AT24Cxx_Status cfg_save(void) {
    cfg.magic = EE_MAGIC;
    cfg.crc   = crc16((uint8_t *)&cfg + 4, sizeof(cfg) - 4);

    return AT24Cxx_WriteArray(&eeprom, EE_CFG_ADDR,
                              (uint8_t *)&cfg, sizeof(cfg));
    //  library รอ write cycle 5ms/segment ให้ภายใน (AT24CXX_WRITE_CYCLE_MS)
}

// โหลด: ReadArray คืน status + data ผ่าน OUT-PARAM
uint8_t cfg_load(void) {
    AT24Cxx_Status r = AT24Cxx_ReadArray(&eeprom, EE_CFG_ADDR,
                                         (uint8_t *)&cfg, sizeof(cfg));
    if (r != AT24CXX_OK) {
        USART_Print("Load fail: ");
        USART_Println(AT24Cxx_StatusStr(r));
        return 0;
    }

    if (cfg.magic == EE_MAGIC) {
        uint16_t calc = crc16((uint8_t *)&cfg + 4, sizeof(cfg) - 4);
        if (calc == cfg.crc) return 1;          // valid
        USART_Println("[EEPROM] CRC mismatch");
    }
    // fallback defaults
    memset(&cfg, 0, sizeof(cfg));
    strcpy(cfg.ssid, "DEFAULT_SSID");
    strcpy(cfg.pass, "DEFAULT_PASS");
    cfg.brightness = 128;
    return 0;
}
🧠 แนวคิด: struct เก็บ config มี magic + CRC-16 ตรวจ integrity — ทุก call ใช้รูปแบบ status + out-param ที่ถูกต้อง: AT24Cxx_WriteArray และ AT24Cxx_ReadArray คืน AT24Cxx_Status — data ผ่าน out-param pointer ไม่ใช่ return value — CRC ครอบข้อมูลตั้งแต่หลัง magic+crc เพื่อกันข้ามฟิลด์ — magic=0xA55A กันอ่านค่า random ตอน EEPROM ว่าง — library รอ write cycle 5ms ให้ภายในตัวไม่ต้อง Delay_Ms(5) เอง — ไม่มี AT24Cxx_Write/Read แบบบทเดิม (ชื่อผิด) — ใช้ WriteArray/ReadArray เท่านั้น

ข้อ 2: Wear Leveling — กระจายการเขียนข้ามหลาย slot

#define EE_WEAR_SLOTS   8
#define EE_SLOT_SIZE    16    // เท่ากับ page size AT24C32+ แนะหน่อย
#define EE_WEAR_START   0x40

typedef struct {
    uint32_t write_count;
    uint8_t  data[8];
    uint8_t  valid;   // 0xFF = ว่าง(erased), 0x00 = ใช้แล้ว
} ee_slot_t;

AT24Cxx_Instance eeprom;

// หา slot ว่างถัดไป — ใช้ ReadArray อ่านทุก slot
int8_t ee_find_free_slot(void) {
    ee_slot_t slot;
    for (uint8_t i = 0; i < EE_WEAR_SLOTS; i++) {
        AT24Cxx_Status r = AT24Cxx_ReadArray(&eeprom,
            EE_WEAR_START + i * EE_SLOT_SIZE,
            (uint8_t *)&slot, sizeof(slot));
        if (r != AT24CXX_OK) continue;
        if (slot.valid == 0xFF) return (int8_t)i;   // erased → free
    }

    // ทุก slot เต็ม → หาเก่าสุด reuse
    uint32_t oldest = 0xFFFFFFFF;
    int8_t   oldest_idx = 0;
    for (uint8_t i = 0; i < EE_WEAR_SLOTS; i++) {
        AT24Cxx_ReadArray(&eeprom, EE_WEAR_START + i * EE_SLOT_SIZE,
                          (uint8_t *)&slot, sizeof(slot));
        if (slot.write_count < oldest) {
            oldest = slot.write_count;
            oldest_idx = (int8_t)i;
        }
    }
    return oldest_idx;
}

void ee_wear_write(const uint8_t *data, uint8_t len) {
    int8_t idx = ee_find_free_slot();
    ee_slot_t slot = {0};

    AT24Cxx_ReadArray(&eeprom, EE_WEAR_START + idx * EE_SLOT_SIZE,
                      (uint8_t *)&slot, sizeof(slot));
    slot.write_count = (slot.valid == 0xFF) ? 1 : slot.write_count + 1;
    slot.valid = 0x00;
    memcpy(slot.data, data, len < 8 ? len : 8);

    // ใช้ WriteArray ข้าม page boundary ได้ และคืน status ให้เช็ค
    AT24Cxx_Status w = AT24Cxx_WriteArray(&eeprom,
        EE_WEAR_START + idx * EE_SLOT_SIZE,
        (uint8_t *)&slot, sizeof(slot));
    if (w != AT24CXX_OK) USART_Println(AT24Cxx_StatusStr(w));
}
🧠 แนวคิด: Wear leveling กระจายการเขียนไป 8 slots แทนเขียนที่เดิมซ้ำ เพิ่มอายุ EEPROM ~8 เท่า (LIMITATIONS.md: endurance ~1M cycles/cell) — แต่ละ slot มี write_count และ valid flag — 0xFF = erased state (EEPROM ใหม่) — เมื่อทุก slot เต็มเลือก write_count ต่ำสุด reuse — ใช้ ReadArray/WriteArray ที่คืน status + out-param เท่านั้น — ไม่ต้อง Delay_Ms(5) ระหว่างเขียนเพราะ library รอ write cycle ให้ภายในตัว

ข้อ 3: Byte Write vs Array Write — เปรียบเทียบความเร็ว

AT24Cxx_Instance eeprom;

void ee_speed_compare(void) {
    uint8_t data[16];
    for (uint8_t i = 0; i < 16; i++) data[i] = i;

    uint32_t t_start, t_elapsed;

    // === WriteByte 16 ครั้ง (แต่ละครั้ง library รอ write cycle 5ms ให้ภายใน) ===
    t_start = Timer_GetMs();
    for (uint8_t i = 0; i < 16; i++) {
        AT24Cxx_Status w = AT24Cxx_WriteByte(&eeprom, 0x00 + i, data[i]);
        if (w != AT24CXX_OK) USART_Println(AT24Cxx_StatusStr(w));
    }
    t_elapsed = Timer_GetMs() - t_start;
    USART_Printf("WriteByte x16: %lu ms\r\n", t_elapsed);

    // === WriteArray 1 ครั้ง (16 bytes ในคำสั่งเดียว — ข้าม page boundary) ===
    t_start = Timer_GetMs();
    AT24Cxx_Status w = AT24Cxx_WriteArray(&eeprom, 0x20, data, 16);
    if (w != AT24CXX_OK) USART_Println(AT24Cxx_StatusStr(w));
    t_elapsed = Timer_GetMs() - t_start;
    USART_Printf("WriteArray x1: %lu ms\r\n", t_elapsed);

    // === อ่านกลับด้วย ReadArray (out-param) เพื่อยืนยัน ===
    uint8_t rb[16] = {0};
    AT24Cxx_Status r = AT24Cxx_ReadArray(&eeprom, 0x20, rb, 16);
    if (r == AT24CXX_OK && memcmp(data, rb, 16) == 0)
        USART_Println("Verify OK");
    else
        USART_Println(AT24Cxx_StatusStr(r));

    USART_Println("---");
    USART_Printf("Capacity=%lu bytes\r\n", AT24Cxx_GetCapacity(&eeprom));
    USART_Println("WriteArray เร็วกว่าเพราารวม write cycle ในคำสั่งเดียว");
}

// page size ตามรุ่น (จาก	enum AT24Cxx_Type ใน header):
//   AT24C01/02 = 8B, AT24C04/08/16 = 16B, AT24C32/64 = 32B,
//   AT24C128/256 = 64B, AT24C512 = 128B
🧠 แนวคิด: ทุก EEPROM byte write ต้องรอ write cycle 5ms (tWR) — WriteByte 16 ครั้ง = 16 × overhead, WriteArray 16 bytes = ครั้งเดียว (library แบ่ง page boundary และรอ write cycle ให้ภายในตัว) — ทุก call เช็ค AT24Cxx_Status ที่คืนมา เพราะ I2C/timeout error เกิดขึ้นได้ — AT24Cxx_GetCapacity(&eeprom) คืน bytes ไม่ใช่ enum — page size ขึ้นกับรุ่น (ดู enum) — ไม่ใช้ AT24Cxx_WritePage (ชื่อบทเดิมผิด ไม่มีใน header) — ใช้ WriteArray แทนเพราะข้าม page boundary ให้เอง
