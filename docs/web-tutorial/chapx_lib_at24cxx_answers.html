📝 เฉลย — AT24Cxx EEPROM

ข้อ 1: Config Store — เก็บ WiFi SSID/PASS ใน EEPROM

#define EE_CONFIG_ADDR   0x0000
#define EE_MAGIC         0xA55A
#define SSID_MAX_LEN     32
#define PASS_MAX_LEN     32

typedef struct {
    uint16_t magic;
    uint16_t crc;
    char ssid[SSID_MAX_LEN];
    char pass[PASS_MAX_LEN];
    uint8_t  reserved[28];  // เผื่ออนาคต
} ee_config_t;  // รวม 96 bytes

ee_config_t ee_config;

uint16_t ee_crc16(const uint8_t *data, uint16_t len) {
    uint16_t crc = 0xFFFF;
    while(len--) {
        crc ^= (uint16_t)*data++ << 8;
        for(uint8_t i=0; i<8; i++)
            crc = (crc & 0x8000) ? (crc<<1)^0x1021 : crc<<1;
    }
    return crc;
}

void ee_config_save(void) {
    ee_config.magic = EE_MAGIC;
    ee_config.crc = ee_crc16((uint8_t *)&ee_config + 4,
                             sizeof(ee_config) - 4);

    AT24C_Write(EE_CONFIG_ADDR, (uint8_t *)&ee_config,
                sizeof(ee_config));
    USART_Println("[EEPROM] Config saved");
}

uint8_t ee_config_load(void) {
    AT24C_Read(EE_CONFIG_ADDR, (uint8_t *)&ee_config,
               sizeof(ee_config));

    if(ee_config.magic != EE_MAGIC) {
        USART_Println("[EEPROM] No valid config — use defaults");
        memset(&ee_config, 0, sizeof(ee_config));
        strcpy(ee_config.ssid, "DEFAULT_SSID");
        strcpy(ee_config.pass, "DEFAULT_PASS");
        return 0;
    }

    uint16_t calc = ee_crc16((uint8_t *)&ee_config + 4,
                             sizeof(ee_config) - 4);
    if(calc != ee_config.crc) {
        USART_Println("[EEPROM] CRC error — config corrupt");
        return 0;
    }

    USART_Print("SSID: ");
    USART_Println(ee_config.ssid);
    return 1;
}
🧠 แนวคิด: ใช้ struct เก็บ config มี magic number ตรวจว่าเคยเขียนหรือยัง — CRC-16 ตรวจ integrity (CCITT polynomial 0x1021) — AT24C_Write/Read เขียน/อ่านทั้ง struct ในครั้งเดียว — magic=0xA55A ป้องกันการอ่านค่า random ตอน EEPROM ว่าง — reserved bytes เผื่อขยาย struct ในอนาคตโดยไม่ต้องย้าย address — AT24C02 มี 256 bytes ใช้ 96 bytes ต่อ config เหลืออีก 160 bytes

ข้อ 3: Byte Write vs Page Write — เปรียบเทียบความเร็ว

void ee_speed_compare(void) {
    uint8_t data[16];
    uint32_t t_start, t_elapsed;

    // เตรียม test data
    for(uint8_t i=0; i<16; i++) data[i] = i;

    // === Byte Write: เขียนทีละ 1 byte 16 ครั้ง ===
    t_start = Timer_GetUs();
    for(uint8_t i=0; i<16; i++) {
        AT24C_WriteByte(0x00 + i, data[i]);
        Delay_Ms(5);  // AT24Cxx write cycle ~5ms
    }
    t_elapsed = Timer_GetUs() - t_start;

    USART_Print("Byte Write (16x): ");
    USART_PrintNum((uint16_t)(t_elapsed / 1000));
    USART_Println(" ms");

    // === Page Write: เขียน 16 bytes ในครั้งเดียว ===
    t_start = Timer_GetUs();
    AT24C_WritePage(0x10, data, 16);
    Delay_Ms(5);
    t_elapsed = Timer_GetUs() - t_start;

    USART_Print("Page Write (1x):  ");
    USART_PrintNum((uint16_t)(t_elapsed / 1000));
    USART_Println(" ms");

    // สรุป
    USART_Println("---");
    USART_Println("Page write เร็วกว่า ~16 เท่า");
    USART_Println("เพราะ wait 5ms ครั้งเดียว vs 16 ครั้ง");
}

// AT24C02 page size = 8 bytes
// AT24C08 page size = 16 bytes
// AT24C16 page size = 16 bytes
// AT24C32 page size = 32 bytes
🧠 แนวคิด: AT24Cxx มี write cycle time ~5ms ต่อรอบ — Byte write: 16 ครั้ง × 5ms = 80ms — Page write: 1 ครั้ง × 5ms = 5ms — เร็วกว่า ~16 เท่า — ข้อจำกัด: เขียนข้าม page boundary ไม่ได้ (ต้องเขียนภายใน page เดียว) — page size ขึ้นกับรุ่น: AT24C02=8B, AT24C04/08/16=16B, AT24C32+=32B — ถ้าข้อมูลอยู่ข้าม page ต้องแบ่งเขียนทีละ page
