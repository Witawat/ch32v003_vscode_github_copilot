📝 เฉลย — บทที่ 12: Watchdog

ข้อ 1: ตรวจจับ reset จาก IWDG (IWDG reset detect)

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include "SimpleHAL.h"

volatile uint32_t g_wdg_count = 0;

int main(void) {
    SystemCoreClockUpdate();

    // ——— ตรวจจับ reset จาก IWDG — ใช้ API จริง IWDG_WasResetCause() ———
    if (IWDG_WasResetCause()) {
        // หน่วงเล็กน้อย — ป้องกัน infinite reset loop
        for (volatile uint32_t i = 0; i < 50000; i++);

        Timer_Init();
        USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
        pinMode(PC0, PIN_MODE_OUTPUT);
        g_wdg_count++;

        USART_Print("!!! IWDG WATCHDOG RESET !!!\r\n");
        // LED กระพริบเร็ว 5 ครั้ง — แสดงว่าเกิด reset
        for (int i = 0; i < 5; i++) {
            digitalWrite(PC0, HIGH); Delay_Ms(50);
            digitalWrite(PC0, LOW);  Delay_Ms(50);
        }

        // ⚠ IWDG_ClearResetFlag() เคลียร์ reset flag "ทุกตัวพร้อมกัน"
        //   (POR/PIN/IWDG/WWDG/SW/LP) — ถ้าจะอ่าน flag อื่นต้องทำ "ก่อน" เรียกนี่
        USART_Print("Clearing all reset flags now\r\n");
        IWDG_ClearResetFlag();
    }

    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
    pinMode(PC0, PIN_MODE_OUTPUT);

    // IWDG timeout 1.6 วิ (1600ms) — auto-pick prescaler
    IWDG_SimpleInit(1600);

    USART_Print("IWDG Active — 1.6s timeout\r\n");

    uint32_t counter = 0;
    while (1) {
        // LED ON 500ms
        digitalWrite(PC0, HIGH);
        USART_Print("ON  [#");
        USART_PrintNum((int32_t)++counter);
        USART_Print("]\r\n");
        Delay_Ms(500);

        // LED OFF 500ms
        digitalWrite(PC0, LOW);
        USART_Print("OFF [feed]\r\n");

        // Feed watchdog — no-arg
        IWDG_Feed();
        // รวม 1 cycle = 500ms + feed = ปลอดภัย (< 1.6s)
    }
}
🧠 แนวคิด: IWDG_WasResetCause() คืน 1 ถ้า reset ล่าสุดเกิดจาก IWDG เป็น API จริงของ SimpleHAL — เช็คตั้งแต่ต้น main() ก่อนเริ่ม IWDG ใหม่ — IWDG_SimpleInit(1600) auto-pick prescaler ให้พอดี 1.6s — หลังจากเปิดแล้วปิดไม่ได้ — ทุกรอบ loop ใช้เวลา ~1s → feed ทันเวลาเสมอ — กฎเหล็ก IWDG: feed ใน main loop เท่านั้น ห้าม feed ใน ISR เพราะ ISR อาจทำงานแม้ main loop ค้าง — IWDG timeout ที่ใช้ได้: 0.1ms - 32.7s (แล้วแต่ prescaler) — เลือกให้พอดีกับ worst-case loop time + margin 50% — debugger หยุด IWDG อัตโนมัติ — ต้องรันจริงถึงจะเห็นผล — ⚠ IWDG_ClearResetFlag(): เคลียร์ reset flag ทุกตัวพร้อมกัน (POR/PIN/IWDG/WWDG/SW/LP) ตาม hardware CH32V003 — ถ้าต้องการอ่านสาเหตุ reset อื่นด้วย ให้เรียก RCC_GetFlagStatus() ของ flag อื่นก่อนเรียกฟังก์ชันนี้

ข้อ 2: WWDG window test — เร็วเกิน / ใน window / ช้าเกิน

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include "SimpleHAL.h"

// ——— PCLK1 = 24MHz, prescaler = 8 ———
//   timeout  (counter 0x7F) = WWDG_TIMEOUT_MS(8, 0x7F) ≈ 87.4 ms
//   window opens (counter 0x50) = WWDG_TIMEOUT_MS(8, 0x50) ≈ 23.2 ms
//   → ต้อง WWDG_Refresh(counter) ตอน counter อยู่ระหว่าง 0x50 - 0x40 (≈ 23..87 ms)
uint32_t window_us = WWDG_TIMEOUT_US(8, 0x50);  // ≈ 23211 µs
uint32_t timeout_us = WWDG_TIMEOUT_US(8, 0x7F); // ≈ 87381 µs

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
    pinMode(PC0, PIN_MODE_OUTPUT);
    pinMode(PC1, PIN_MODE_OUTPUT);

    USART_Print("=== WWDG Window Mode Demo ===\r\n");
    USART_Print("timeout ≈ 87 ms, window opens ≈ 23 ms\r\n");
    USART_Print("Valid feed: between 23ms and 87ms\r\n");

    // WWDG_SimpleInit รับ raw register values (counter, window) — ไม่ใช่ (timeout_constant, percentage)
    // ⚠ ถ้าส่ง window > counter (เช่น (0x60, 0x70)) window จะถูก clamp ลงเท่า counter แบบ silent
    WWDG_SimpleInit(0x7F, 0x50);  // counter=127, window=80, prescaler default 8

    uint32_t counter = 0;

    while (1) {
        counter++;
        digitalWrite(PC0, !digitalRead(PC0));

#if defined(TEST_TOO_FAST)
        // เคส 1: feed เร็วเกิน — ตอน counter ยัง > 0x50 → RESET!
        WWDG_Refresh(0x7F);
#elif defined(TEST_IN_WINDOW)
        // เคส 2: ใน window — Delay 30ms (อยู่ระหว่าง 23..87) แล้ว refresh → OK
        Delay_Ms(30);
        digitalWrite(PC1, HIGH);
        WWDG_Refresh(0x7F);   // ต้องส่ง counter เสมอ
        digitalWrite(PC1, LOW);
#elif defined(TEST_TOO_SLOW)
        // เคส 3: ช้าเกิน — Delay 100ms เลย 87ms → RESET!
        Delay_Ms(100);
        WWDG_Refresh(0x7F);   // ตอนนี้ counter ตกไปที่ 0x3F แล้ว — reset ก่อน refresh
#else
        // default: ใน window
        Delay_Ms(30);
        WWDG_Refresh(0x7F);
#endif

        if (counter % 50 == 0) {
            USART_Print("WWDG Feed #");
            USART_PrintNum((int32_t)counter);
            USART_Print(" OK\r\n");
        }
    }
}
🧠 แนวคิด: WWDG window mode ต่างจาก IWDG ตรงที่ต้อง refresh ใน "หน้าต่างเวลา" — เร็วเกินก็ reset, ช้าเกินก็ reset — WWDG_SimpleInit(0x7F, 0x50) ตั้ง raw register values (counter=127, window=80) ไม่ใช่ (timeout constant, percentage) — valid feed คือตอน window < counter < 0x40 ≈ ระหว่าง 23ms ถึง 87ms — WWDG_Refresh(0x7F) ต้องส่ง counter ทุกครั้ง (ไม่ใช่ no-arg) — ทดสอบ 3 เคส: (1) TEST_TOO_FAST → refresh ทันที = counter ยัง > 0x50 → RESET (loop เร็วผิดปกติ), (2) TEST_IN_WINDOW → Delay_Ms(30) แล้ว refresh → OK, (3) TEST_TOO_SLOW → Delay_Ms(100) → counter ไปถึง 0x3F ก่อน → RESET (loop ค้าง) — WWDG ใช้ PCLK1 → แม่นยำกว่า IWDG (LSI) — timeout สั้น (µs-ms) → ตรวจจับ loop anomaly ได้ไวกว่า — ⚠ window-clamp silent: ถ้าตั้ง window > counter (เช่น (0x60, 0x70)) window จะถูกลดลงเท่า counter โดยไม่แจ้งเตือน — ป้องกัน reset loop ไม่รู้จบ — ข้อควรระวัง: Debugger อาจรบกวน timing WWDG — ทดสอบแบบรันจริง — SimpleWWDG ไม่มี timeout constant แบบเสร็จสรรค์ใน header ต้องคำนวณจาก WWDG_TIMEOUT_MS(prescaler, counter) เอง

ข้อ 3: Safe-mode diagnostic — บันทึก reset log + safe mode

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include "SimpleHAL.h"

#define CFG_ADDR   0x08003C00
#define CFG_MAGIC  0x5AFE

typedef struct {
    uint16_t magic;
    uint32_t power_on_count;
    uint32_t wdg_reset_count;
    uint32_t last_reset_ms;
    uint8_t  reset_reason;  // 0=power-on, 1=iwdg, 2=wwdg, 3=soft, 4=pin
    uint16_t crc;
} ResetLog_t;

ResetLog_t g_log;

void log_init(void) {
    Flash_Init();
    if (!Flash_LoadConfig((uint8_t*)&g_log, sizeof(ResetLog_t))) {
        memset(&g_log, 0, sizeof(g_log));
        g_log.magic = CFG_MAGIC;
    }
}

void log_save(void) {
    g_log.crc = Flash_CalculateCRC16((uint8_t*)&g_log,
                     sizeof(ResetLog_t) - sizeof(uint16_t));
    Flash_SaveConfig(CFG_ADDR, (uint8_t*)&g_log, sizeof(ResetLog_t));
}

int main(void) {
    SystemCoreClockUpdate();

    // หน่วงเผื่อ stable power
    for (volatile uint32_t i = 0; i < 50000; i++);

    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
    pinMode(PC0, PIN_MODE_OUTPUT);

    log_init();

    // ——— ตรวจจับ reset source — ใช้ IWDG_WasResetCause() ตาม API จริง ———
    if (IWDG_WasResetCause()) {
        // Reset จาก IWDG — main loop ค้าง!
        g_log.reset_reason = 1;
        g_log.wdg_reset_count++;
        g_log.last_reset_ms = Get_CurrentMs();

        USART_Print("============================\r\n");
        USART_Print("!!! IWDG WATCHDOG RESET !!!\r\n");
        USART_Print("============================\r\n");
        USART_Print("Reset count: ");
        USART_PrintNum((int32_t)g_log.wdg_reset_count);
        USART_Print("\r\n");

        // LED SOS pattern ถ้า reset หลายครั้ง
        if (g_log.wdg_reset_count >= 3) {
            USART_Print("WARNING: Multiple WDG resets!\r\n");
            USART_Print("Safe mode: LED blink SOS\r\n");
            log_save();

            // เคลียร์ flag "หลัง" อ่าน reset cause ครบแล้ว
            // ⚠ IWDG_ClearResetFlag() เคลียร์ flag "ทุกตัว" พร้อมกัน (ไม่แยกเฉพาะ IWDGRST)
            IWDG_ClearResetFlag();

            // SOS: ... --- ...
            while (1) {
                for (int s = 0; s < 3; s++) {
                    digitalWrite(PC0, HIGH); Delay_Ms(150);
                    digitalWrite(PC0, LOW);  Delay_Ms(150);
                }
                Delay_Ms(300);
                for (int l = 0; l < 3; l++) {
                    digitalWrite(PC0, HIGH); Delay_Ms(500);
                    digitalWrite(PC0, LOW);  Delay_Ms(150);
                }
                Delay_Ms(300);
                for (int s = 0; s < 3; s++) {
                    digitalWrite(PC0, HIGH); Delay_Ms(150);
                    digitalWrite(PC0, LOW);  Delay_Ms(150);
                }
                Delay_Ms(1000);
                // (ไม่ได้เปิด IWDG ใน safe mode → ไม่ต้อง feed)
            }
        }
    } else {
        // Power-on reset ธรรมดา
        g_log.reset_reason = 0;
        g_log.power_on_count++;
        g_log.last_reset_ms = 0;

        USART_Print("=== Power-On Reset ===\r\n");
        USART_Print("Boot count: ");
        USART_PrintNum((int32_t)g_log.power_on_count);
        USART_Print("\r\n");
    }

    USART_Print("Last reset reason: ");
    switch (g_log.reset_reason) {
        case 0: USART_Print("POWER-ON\r\n"); break;
        case 1: USART_Print("IWDG\r\n"); break;
        case 2: USART_Print("WWDG\r\n"); break;
        case 3: USART_Print("SOFTWARE\r\n"); break;
        case 4: USART_Print("PIN RESET\r\n"); break;
        default: USART_Print("UNKNOWN\r\n"); break;
    }

    USART_Print("Total IWDG resets: ");
    USART_PrintNum((int32_t)g_log.wdg_reset_count);
    USART_Print("\r\n");

    log_save();

    // เคลียร์ flag หลังอ่าน reset cause ครบแล้ว — เพื่อ boot ครั้งถัดไปอ่านได้สะอาด
    IWDG_ClearResetFlag();

    // เริ่ม IWDG — 4 วิ timeout
    IWDG_SimpleInit(4000);
    USART_Print("IWDG 4s active — running normally\r\n");

    uint32_t tick = 0;
    while (1) {
        digitalWrite(PC0, !digitalRead(PC0));
        USART_Print("Tick ");
        USART_PrintNum((int32_t)++tick);
        USART_Print(" OK\r\n");

        Delay_Ms(1500);
        IWDG_Feed();

        // ถ้า tick == 10 → จงใจค้าง (ทดสอบ)
        if (tick == 10) {
            USART_Print("Intentionally freezing...\r\n");
            Delay_Ms(100);
            while (1);  // ค้าง → 4s → reset → g_log.wdg_reset_count++
        }
    }
}
🧠 แนวคิด: การตรวจจับ reset source เป็นสิ่งจำเป็นสำหรับระบบ production-grade — IWDG_WasResetCause() เป็น API จริงของ SimpleHAL — อ่าน flag ใน RCC → check ตั้งแต่ต้น main() ก่อน init อะไร — ข้อมูล reset log เก็บใน Flash (BKP อาจหายถ้าไฟดับนาน) — บันทึก: จำนวนครั้ง, เวลาล่าสุด, สาเหตุ — วิเคราะห์แนวโน้ม: ถ้า WDG reset ถี่ → safe mode — Safe Mode: ทำงานน้อยที่สุด — กระพริบ LED SOS — ไม่ใช้ peripheral ที่อาจเป็นสาเหตุของ crash — ต้องการ manual reset เพื่อออก — ⚠ IWDG_ClearResetFlag(): ฟังก์ชันนี้เคลียร์ reset flag ทุกตัวพร้อมกัน (POR, PIN, Software, IWDG, WWDG, LowPower) ตาม hardware CH32V003 — ถ้าจะอ่านสาเหตุ reset อื่นด้วย ให้อ่านด้วย RCC_GetFlagStatus() ก่อน เรียกฟังก์ชันนี้ — ข้อควรระวัง: ถ้าไม่ดีเลย์หลังเช็ค IWDG_WasResetCause() → IWDG flag อาจยังไม่ stable → เข้า reset loop ไม่รู้จบ — ดีเลย์ ~50ms ก่อนเริ่ม init ปกติ
