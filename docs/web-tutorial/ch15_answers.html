ข้อ 1: ใช้ millis() ทำ non-blocking LED

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include "SimpleHAL.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();       // ต้องเรียกก่อนใช้ millis()/delay()
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);

    pinMode(PC0, PIN_MODE_OUTPUT);  // LED 1
    pinMode(PC1, PIN_MODE_OUTPUT);  // LED 2
    pinMode(PA1, PIN_MODE_INPUT_PULLUP); // Button

    USART_Print("Non-blocking LEDs with millis()\r\n");

    uint32_t last_blink = 0;
    uint32_t last_fast = 0;
    uint32_t last_print = 0;
    uint8_t  blink_state = 0;
    uint32_t btn_count = 0;

    while (1) {
        uint32_t now = millis();

        // LED1: blink 500ms — ไม่ใช้ delay()!
        if (now - last_blink >= 500) {
            last_blink = now;
            digitalWrite(PC0, !digitalRead(PC0));
        }

        // LED2: fast blink 100ms
        if (now - last_fast >= 100) {
            last_fast = now;
            blink_state = !blink_state;
            digitalWrite(PC1, blink_state ? HIGH : LOW);
        }

        // Print status every 2 seconds
        if (now - last_print >= 2000) {
            last_print = now;
            USART_Print("Uptime: ");
            USART_PrintNum((int32_t)(now / 1000));
            USART_Print("s  Button: ");
            USART_PrintNum((int32_t)btn_count);
            USART_Print("\r\n");
        }

        // Check button — non-blocking
        static uint8_t btn_last = 1;
        uint8_t btn_now = digitalRead(PA1);
        if (btn_last == 1 && btn_now == 0) {  // falling edge
            btn_count++;
            USART_Print("Button pressed! (");
            USART_PrintNum((int32_t)btn_count);
            USART_Print(")\r\n");

            // LED1 ON immediately on press
            digitalWrite(PC0, HIGH);
            last_blink = now;  // reset blink timer
        }
        btn_last = btn_now;

        // CPU 100% free for other tasks!
        // No busy-wait, no delay() blocking
    }
}
🧠 แนวคิด: millis() ใช้ SysTick 1ms — ทำ non-blocking timing: เช็ค if(now - last >= period) แทน delay() — ข้อดี: CPU ว่าง 100% — สามารถทำงานหลายอย่างพร้อมกัน — ตัวอย่างนี้: LED1(500ms) + LED2(100ms) + Print(2s) + Button — ทั้งหมดทำงานพร้อมกันโดยไม่มี delay() — ข้อควรระวัง: millis() overflow ทุก ~49.7 วัน (uint32_t) — ใช้ subtraction แบบ unsigned (now - last) ถึงจะปลอดภัย overflow — ต้องเรียก Timer_Init() ก่อนใช้ millis()

ข้อ 2: random + dtostrf แสดงเลขสุ่ม

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include "SimpleHAL.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    ADC_SimpleInit();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);

    pinMode(PC0, PIN_MODE_OUTPUT);
    pinMode(PC1, PIN_MODE_OUTPUT);

    // Seed random ด้วย ADC noise (unconnected pin)
    randomSeed(analogRead(PD4));

    USART_Print("=== random() + dtostrf Demo ===\r\n\r\n");

    uint32_t counter = 0;

    while (1) {
        counter++;
        digitalWrite(PC0, HIGH);

        // random range: 1-6 (dice), 0-255 (PWM), -1000 to 1000
        int dice = random(1, 7);
        int pwm_val = random(0, 256);
        int temp_sim = random(-200, 801);  // -20.0 to 80.0 C

        analogWrite(PC1, pwm_val);

        // dtostrf: float -> string (ไม่มี sprintf float!)
        float temp_f = temp_sim / 10.0f;
        float vdd = ADC_GetVDD();

        char buf_temp[10];
        char buf_vdd[10];
        char buf_dice[10];

        dtostrf(temp_f, 6, 1, buf_temp);
        dtostrf(vdd, 5, 2, buf_vdd);
        dtostrf((float)dice, 1, 0, buf_dice);

        USART_Print("#");
        USART_PrintNum((int32_t)counter);
        USART_Print(" Dice=");
        USART_Print(buf_dice);
        USART_Print(" Temp=");
        USART_Print(buf_temp);
        USART_Print("C VDD=");
        USART_Print(buf_vdd);
        USART_Print("V PWM=");
        USART_PrintNum(pwm_val);

        // Coin flip simulation
        if (random(0, 2)) {
            USART_Print(" HEAD");
        } else {
            USART_Print(" TAIL");
        }

        USART_Print("\r\n");

        digitalWrite(PC0, LOW);

        // Re-seed occasionally
        if (counter % 100 == 0) {
            randomSeed(analogRead(PD4) ^ millis());
            USART_Print("  -- Re-seeded --\r\n");
        }

        Delay_Ms(500);
    }
}
🧠 แนวคิด: random(min, max) ใช้ LFSR — เร็วกว่า software RNG — seed ด้วย randomSeed(analogRead(unconnected_pin)) — ADC noise จาก pin ที่ไม่ได้ต่อ = entropy จริง — dtostrf(val, width, prec, buf) แปลง float เป็น string — width = จำนวนหลักรวม (รวมจุดทศนิยม), prec = จำนวนทศนิยม — จำเป็นมากเพราะ CH32V003 stdlib แบบ nano ไม่มี sprintf float — buffer ต้องใหญ่พอ: width + 1 — ตัวอย่าง: dtostrf(123.45, 6, 2, buf) → "123.45"

ข้อ 3: yield() ร่วมกับ IWDG

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include "SimpleHAL.h"

// Override yield() — weak function, ถูกเรียกใน delay()
void yield(void) {
    // Feed watchdog — ป้องกัน reset ระหว่าง delay ยาว
    IWDG_Feed();

    // Toggle LED3 แสดงว่า yield ทำงาน (LED กระพริบเบาๆ ระหว่าง delay)
    static uint8_t y_state = 0;
    y_state = !y_state;
    if (y_state) {
        digitalWrite(PC2, HIGH);
    } else {
        digitalWrite(PC2, LOW);
    }
}

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);

    pinMode(PC0, PIN_MODE_OUTPUT);  // LED1: main blink
    pinMode(PC1, PIN_MODE_OUTPUT);  // LED2: event
    pinMode(PC2, PIN_MODE_OUTPUT);  // LED3: yield indicator

    USART_Print("=== yield() + IWDG Demo ===\r\n");

    // เริ่ม IWDG — 2 วิ timeout
    IWDG_SimpleInit(2000);

    USART_Print("IWDG 2s active\r\n");
    USART_Print("yield() feeds watchdog inside delay()\r\n");
    USART_Print("PC2 blinks during delay = yield() is running\r\n\r\n");

    uint32_t counter = 0;

    while (1) {
        counter++;
        digitalWrite(PC0, HIGH);  // LED1 ON

        USART_Print("Loop ");
        USART_PrintNum((int32_t)counter);
        USART_Print(" — entering delay(4000)...\r\n");
        USART_Print("  (yield will feed IWDG every ~1ms)\r\n");

        // delay(4000) > IWDG 2s — ถ้าไม่มี yield() → RESET
        // yield() override ทำให้ IWDG ถูก feed ตลอด → ปลอดภัย
        digitalWrite(PC1, HIGH);
        delay(4000);  // yield() feeds IWDG ~4000 times
        digitalWrite(PC1, LOW);

        digitalWrite(PC0, LOW);  // LED1 OFF
        USART_Print("  delay complete — IWDG still fed!\r\n\r\n");

        // Short tasks
        delay(500);
    }
}
🧠 แนวคิด: yield() เป็น weak function — ถูกเรียกใน delay() ทุก ~1ms — override เพื่อ: (1) feed IWDG → prevent reset during long delay, (2) process background tasks, (3) check flags/buttons — โจทย์นี้: delay(4000) > IWDG timeout(2000) — ถ้าไม่มี yield() → IWDG reset ทุก 4 วิ — yield() feed ทุก ~1ms → ปลอดภัย — ข้อดี: ไม่ต้องเปลี่ยนโค้ด Arduino ที่ใช้ delay() — ข้อเสีย: yield() เรียกบ่อย → overhead — ควรทำงานน้อยที่สุด — alternative: ใช้ millis() non-blocking แทน delay() + yield() เพื่อประสิทธิภาพสูงกว่า
