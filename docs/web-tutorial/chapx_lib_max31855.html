SPI mode 0 — MAX31855 อ่าน 32-bit (14-bit TC + 12-bit CJ + fault), MAX6675 อ่าน 16-bit (12-bit TC)

📚 หลักการทำงาน

Thermocouple K-Type: วัดอุณหภูมิจากแรงดัน mV ที่เกิดขึ้นระหว่างโลหะ 2 ชนิด (Chromel-Alumel) — ช่วงกว้างมาก -270°C ถึง +1372°C

Cold-Junction Compensation (CJC): ฝังในชิป — วัดอุณหภูมิที่ขั้วต่อ PCB แล้วชดเชยอัตโนมัติ

MAX31855: 32-bit frame — thermocouple temp (14-bit) + internal/cold-junction temp (12-bit) + fault bits (OC/SCG/SCV)

MAX6675: 16-bit frame — thermocouple temp (12-bit) — range 0°C ถึง 1024°C, API ง่ายกว่า — MAX6675_ReadTemp(PC4)

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Init SPI + MAX31855

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "MAX31855.h"

MAX31855_Instance therm;

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();

    MAX31855_Init(&therm, PC4);  // CS pin
    USART_Println("MAX31855 Ready!");

🔍 MAX31855_Init(&therm, CS_PIN) — library จัดการ SPI ภายใน — CS เลือกขาไหนก็ได้, ใช้ SPI mode 0

ขั้นที่ 2: อ่าน Thermocouple + Cold-Junction + Fault

    float tc = MAX31855_ReadTemp(&therm);           // thermocouple temp
    float cj = MAX31855_ReadInternalTemp(&therm);    // cold-junction temp
    int fault = MAX31855_GetFault(&therm);           // 0=OK

    USART_Print("TC: "); USART_PrintFloat(tc, 1);
    USART_Print("C  CJ: "); USART_PrintFloat(cj, 1);
    USART_Print("C  Fault: "); USART_PrintNum(fault);
    USART_Println();

    if(fault & 0x01) USART_Println("⚠️ Open Circuit!");
    if(fault & 0x02) USART_Println("⚠️ Short to GND!");
    if(fault & 0x04) USART_Println("⚠️ Short to VCC!");

    // MAX6675 variant (simpler API):
    // #include "MAX6675.h"
    // float t = MAX6675_ReadTemp(PC4);  // 0-1024°C, no fault

🔍 MAX31855_ReadTemp() — thermocouple temp — MAX31855_ReadInternalTemp() — CJ temp — MAX31855_GetFault() — 0=OK, bit0=OC, bit1=SCG, bit2=SCV

ขั้นที่ 3: Loop

    Timer_t tmr;
    Start_Timer(&tmr, 500, 1);

    while(1) {
        if(Is_Timer_Expired(&tmr)) {
            float tc = MAX31855_ReadTemp(&therm);
            float cj = MAX31855_ReadInternalTemp(&therm);
            int fault = MAX31855_GetFault(&therm);

            if(fault == 0) {
                USART_Print("TC: "); USART_PrintFloat(tc, 1);
                USART_Print("°C (CJ:"); USART_PrintFloat(cj, 1);
                USART_Println("°C)");
            } else {
                USART_Print("Fault: "); USART_PrintNum(fault);
                USART_Println(" — Check Thermocouple!");
            }
        }
        IWDG_Feed();
    }
}

🔍 MAX6675 API ง่ายกว่า — MAX6675_ReadTemp(PC4) — ไม่ต้องใช้ instance — แต่ไม่มี fault detection และ range จำกัด 0-1024°C

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
อ่านได้ค่า ~0°C ตลอดต่อ thermocouple ผิดขั้วK-Type: สีเหลือง → T+, สีแดง → T-
MAX31855 Fault bit 0 (OC) ตลอดThermocouple ขาด หรือต่อไม่แน่นเช็คขั้วต่อ — thermocouple ต้องเชื่อมแน่น
อุณหภูมิกระโดด ±10°CNoise coupling บน thermocouple wireใช้ twisted-pair wire, ใส่ ferrite bead, averaging 4-8 ครั้ง
MAX6675 อ่านไม่ออก — คืน 0MAX6675 ใช้ SPI mode 1 (ไม่ใช่ mode 0)MAX6675 ใช้ SPI mode 1 (CPOL=0, CPHA=1)

🛠️ สร้าง Product: เครื่องวัดอุณหภูมิเตาเผา K-Type

MAX31855 + OLED — แสดงอุณหภูมิ real-time + fault warning

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "MAX31855.h"
#include "oled_i2c.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();

    MAX31855_Instance therm;
    MAX31855_Init(&therm, PC4);

    OLED_Handle oled;
    OLED_Init(&oled, OLED_128x64, 0x3C);
    OLED_SetFont(&oled, FONT_8x16);

    Timer_t tmr;
    Start_Timer(&tmr, 500, 1);

    while(1) {
        if(Is_Timer_Expired(&tmr)) {
            float tc = MAX31855_ReadTemp(&therm);
            float cj = MAX31855_ReadInternalTemp(&therm);
            int fault = MAX31855_GetFault(&therm);

            OLED_Clear(&oled);
            if(fault == 0) {
                OLED_SetCursor(&oled, 0, 0);
                OLED_PrintFloat(&oled, "TC:", tc, 1); OLED_Print(&oled, " C");
                OLED_SetCursor(&oled, 0, 2);
                OLED_PrintFloat(&oled, "CJ:", cj, 1); OLED_Print(&oled, " C");
            } else {
                OLED_SetCursor(&oled, 0, 1);
                OLED_Print(&oled, "FAULT! Check");
                OLED_SetCursor(&oled, 0, 2);
                OLED_Print(&oled, "Thermocouple!");
            }
            OLED_Update(&oled);
        }
        IWDG_Feed();
    }
}

✏️ ลองทำดู

  1. วัดอุณหภูมิหัวแร้งบัดกรี — แตะ thermocouple ที่ tip — สังเกต response time
  2. ทดลองสลับขั้ว T+ / T- — ดูว่าอุณหภูมิกลับทิศ
  3. ทำ PID Temperature Controller — MAX31855 + Relay/SSR — ควบคุมอุณหภูมิเตาเผาให้คงที่
  4. เทียบ MAX31855 vs MAX6675 — ต่อบน SPI บัสเดียวกัน (คนละ CS) — สังเกต accuracy และ fault detection
