📝 เฉลย — TM1650 4-Digit 7-Segment

ข้อ 1: Nixie Tube Emulation (ตัวเลขเปลี่ยนแบบมี animation)

#include "TM1650.h"

// Nixie-style digit transition — ไล่เลข 0-9 ขึ้นก่อนหยุด
void nixie_transition(uint8_t pos, uint8_t target) {
    uint8_t current = 0;
    while(current != target) {
        TM1650_DisplayDigit(pos, current);
        Delay_Ms(30);
        current++;
        if(current > 9) current = 0;
    }
    TM1650_DisplayDigit(pos, target);
}

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    TM1650_Init();
    TM1650_SetBrightness(3);

    uint8_t digits[4] = {1, 9, 8, 4};

    while(1) {
        // Nixie boot animation — ใส่ตัวเลขทีละหลัก
        for(uint8_t pos = 0; pos < 4; pos++) {
            nixie_transition(pos, digits[pos]);
        }
        Delay_Ms(2000);

        // เปลี่ยนเลขแบบสุ่ม
        for(uint8_t i = 0; i < 4; i++) {
            digits[i] = (digits[i] + 3) % 10;
        }
    }
}
🧠 แนวคิด: Nixie tube ตอนเปลี่ยนเลขจะมีการไล่เลขก่อนหยุดที่ค่าเป้าหมาย — nixie_transition หมุนเลข 0→1→2...→target ทีละ 30ms — ใส่ animation ทีละหลักจากซ้ายไปขวาเหมือน nixie clock จริง — เปลี่ยนเลขทุก 2 วินาทีด้วยการบวก offset — สำหรับเอฟเฟกต์สมจริงยิ่งขึ้น ให้สุ่มความเร็วไล่เลขแต่ละหลักไม่เท่ากัน และเพิ่ม "overshoot" (เลยเป้าแล้วเด้งกลับ)

ข้อ 2: Voltmeter — อ่าน ADC แสดงแรงดัน

#include "TM1650.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    TM1650_Init();
    TM1650_SetBrightness(3);
    ADC_SimpleInit(ADC_CH2);

    while(1) {
        // อ่าน ADC 10 ครั้งหาค่าเฉลี่ย
        uint32_t sum = 0;
        for(uint8_t i = 0; i < 10; i++) {
            sum += ADC_Read(ADC_CH2);
            Delay_Ms(1);
        }
        uint16_t adc = sum / 10;

        // แปลงเป็นแรงดัน (Vref = 3.3V, 12-bit ADC)
        // V = adc * 3.3 / 4095
        // แสดงในหน่วย mV: mV = adc * 3300 / 4095
        uint16_t mv = (uint32_t)adc * 3300 / 4095;

        // แสดงผล format: X.XXX V
        // หลัก 0 = โวลต์หลักหน่วย, หลัก 1-3 = mV
        TM1650_DisplayDigit(0, mv / 1000);           // V
        TM1650_DisplayDigit(1, (mv / 100) % 10 | 0x80); // . (จุดทศนิยม)
        TM1650_DisplayDigit(2, (mv / 10) % 10);      // 100mV
        TM1650_DisplayDigit(3, mv % 10);              // 10mV

        Delay_Ms(200);
    }
}
🧠 แนวคิด: อ่าน ADC 10 ครั้งหาค่าเฉลี่ยลด noise — แปลง ADC 12-bit (0-4095) เป็นแรงดัน mV ด้วยสูตร adc * 3300 / 4095 — แสดงผล 4 หลักในรูปแบบ V.VVV — ใช้ OR 0x80 เพิ่มจุดทศนิยมที่หลักที่ 2 — การใช้ integer math (mV) แทน float ช่วยประหยัด flash และ RAM บน CH32V003 — ถ้าต้องการวัดแรงดันเกิน 3.3V ต้องใช้ voltage divider

ข้อ 3: 2-Channel Display — สลับแสดง Volt/Amp

#include "TM1650.h"

#define BTN_PIN  PC1

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    TM1650_Init();
    TM1650_SetBrightness(3);
    ADC_SimpleInit(ADC_CH2);   // แรงดัน
    ADC_SimpleInit(ADC_CH3);   // กระแส (ผ่าน shunt)
    pinMode(BTN_PIN, PIN_MODE_INPUT_PULLUP);

    uint8_t mode = 0;  // 0=Volt, 1=Amp
    uint8_t last_btn = 1;

    while(1) {
        // ตรวจจับปุ่มเปลี่ยนโหมด
        uint8_t btn = digitalRead(BTN_PIN);
        if(last_btn && !btn) {
            mode = !mode;
            Delay_Ms(50);  // debounce
        }
        last_btn = btn;

        if(mode == 0) {
            // โหมด Volt
            uint16_t adc = ADC_Read(ADC_CH2);
            uint16_t mv = (uint32_t)adc * 3300 / 4095;
            TM1650_DisplayDigit(0, mv / 1000);
            TM1650_DisplayDigit(1, (mv / 100) % 10 | 0x80);
            TM1650_DisplayDigit(2, (mv / 10) % 10);
            TM1650_DisplayDigit(3, mv % 10);
        } else {
            // โหมด Amp — shunt 0.1Ω, gain 50
            uint16_t adc = ADC_Read(ADC_CH3);
            // I = (adc*3.3/4095) / (0.1*50) = adc * 3300 / (4095*5)
            uint16_t ma = (uint32_t)adc * 3300 / (4095 * 5);
            TM1650_DisplayDigit(0, ma / 1000);
            TM1650_DisplayDigit(1, (ma / 100) % 10 | 0x80);
            TM1650_DisplayDigit(2, (ma / 10) % 10);
            TM1650_DisplayDigit(3, ma % 10);
        }

        Delay_Ms(200);
    }
}
🧠 แนวคิด: TM1650 แสดงผลทีละ 1 channel — ใช้ปุ่มกดสลับระหว่างโหมด Volt และ Amp — แต่ละโหมดอ่าน ADC คนละช่อง — การวัดกระแสใช้ shunt resistor + amplifier (เช่น INA180) — สูตรกระแส: I = V_adc / (R_shunt × gain) — TM1650 ไม่มี colon ทำให้ใช้การสลับโหมดแทนการแสดงพร้อมกัน — สำหรับเวอร์ชัน auto-toggle ให้ใช้ timer แทนการกดปุ่ม
