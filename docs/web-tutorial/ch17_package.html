    ข้อแตกต่างสำคัญระหว่างแพ็กเกจ:
    ├─ SOP8: ถูกสุด (~$0.10), ไม่มี Hardware SPI, ไม่มี USART pins default
    ├─ SOP16: มี SPI, I2C, USART — แต่ไม่มี PD2-PD4
    ├─ TSSOP20: ครบทุก pin — ทุก peripheral ใช้ได้
    └─ QFN20: เท่ากับ TSSOP20 — 3x3mm QFN

สำคัญ: CH32V003_PACKAGE ต้องกำหนดก่อน #include "SimpleHAL.h" — มิฉะนั้นจะ default เป็น TSSOP20

📚 หลักการทำงาน

ระบบแพ็กเกจ — กลไกที่ทำให้โค้ด C เดียวกันคอมไพล์และทำงานได้บนทุกแพ็กเกจของ CH32V003 (SOP8, SOP16, TSSOP20, QFN20) — โดยเปลี่ยนแค่ #define บรรทัดเดียว

CH32V003_PACKAGE

เป็น macro ที่กำหนดว่า MCU ที่ใช้เป็นแพ็กเกจอะไร — SimpleHAL ใช้ค่านี้เพื่อ:

  • Validate pins: เช็คว่า pin ที่ระบุมีอยู่ในแพ็กเกจนั้นหรือไม่ — ถ้าใช้ PC0 บน SOP8 → compile error
  • Validate peripherals: เช็คว่า peripheral บน pin นั้นมีในแพ็กเกจหรือไม่ — เช่น SPI ไม่มีบน SOP8
  • Enable/Disable remap: บางแพ็กเกจต้องใช้ pin remap (เช่น USART บน SOP8 ต้อง remap)
  • Select alternate functions: เลือก AFIO mapping ให้ถูกต้องตามแพ็กเกจ

Macros ที่ใช้ภายใน

Macroคำอธิบาย

IS_ADC_PIN(pin)true ถ้า pin เป็น ADC input
IS_PWM_VALID_PACKAGE(pin, pkg)true ถ้า pin มี PWM output ในแพ็กเกจ

ทำไมถึงสำคัญ?

เขียนโค้ดหนึ่งครั้ง → ขายสินค้าแบบ SOP8 (ถูก) และ TSSOP20 (ฟีเจอร์ครบ) โดยเปลี่ยนแค่ compile flag — ไม่ต้อง maintain โค้ด 2 เวอร์ชัน

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: เลือกแพ็กเกจ + Pin Validation

// เปลี่ยนแค่บรรทัดนี้ — โค้ดที่เหลือเหมือนเดิม
#define CH32V003_PACKAGE PACKAGE_TSSOP20
// #define CH32V003_PACKAGE PACKAGE_SOP8
#include "SimpleHAL.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();

    // IS_VALID_PIN — compile-time check
    #if !IS_VALID_PIN(PC0)
        #error "PC0 not available on this package!"
    #endif

    // ถ้าใช้ SOP8 — บรรทัดนี้จะ compile error
    pinMode(PC0, PIN_MODE_OUTPUT);

🔍 IS_VALID_PIN(pin) — compile-time macro — ใช้กับ #if directive — ถ้า pin ไม่มีในแพ็กเกจ → false → #error แสดงข้อความชัดเจน

ขั้นที่ 2: IS_ADC_PIN — เช็ค Analog Input

    // เช็คว่า pin นี้เป็น ADC ได้หรือไม่
    #if IS_ADC_PIN(PD2)
        ADC_SimpleInit();
        uint16_t val = analogRead(PD2);
        USART_Print("PD2 ADC OK\r\n");
    #else
        USART_Print("PD2 is NOT ADC on this package\r\n");
    #endif

    // PD7 ไม่ใช่ ADC ในทุกแพ็กเกจ
    #if IS_ADC_PIN(PD7)
        analogRead(PD7);  // ไม่มีทางมาถึง
    #endif

🔍 IS_ADC_PIN(pin) — false สำหรับ PD7 และ pin ที่ไม่ใช่ ADC — บน SOP16 PD2 ไม่มี → false

ขั้นที่ 3: IS_PWM_VALID_PACKAGE — PWM Output

    // PWM บน PC4 — มีแค่บน TSSOP20/SOP16/QFN20
    // บน SOP8: PC4 เป็น I2C เท่านั้น

    #if IS_PWM_VALID_PACKAGE(PC4, CH32V003_PACKAGE)
        PWM_SimpleInit(2, PC4);
        PWM_SetDuty(2, 50);
        USART_Print("PWM on PC4 OK\r\n");
    #else
        // SOP8 fallback — ใช้ PA1 หรือ PA2 แทน
        #if IS_VALID_PIN(PA1)
            PWM_SimpleInit(2, PA1);
            PWM_SetDuty(2, 50);
            USART_Print("PWM on PA1 (SOP8 fallback)\r\n");
        #endif
    #endif

🔍 IS_PWM_VALID_PACKAGE(pin, pkg) — ต้องระบุ package — แต่ละแพ็กเกจมี PWM pin ต่างกัน — SOP8: PA1,PA2,PD1,PD4,PD6 — TSSOP20: PC4,PC5,PC6,PA1,PA2,PD3

ขั้นที่ 4: Multi-Package Code — Fallback Chain

    // เลือก pin LED อัตโนมัติตามแพ็กเกจ
    #if IS_VALID_PIN(PC0)
        #define LED_PIN PC0
    #elif IS_VALID_PIN(PC1)
        #define LED_PIN PC1
    #elif IS_VALID_PIN(PA1)
        #define LED_PIN PA1
    #else
        #error "No valid LED pin found"
    #endif

    pinMode(LED_PIN, PIN_MODE_OUTPUT);
    digitalWrite(LED_PIN, HIGH);

🔍 Fallback chain — ลอง pin แรก → ถ้าไม่มี → ลอง pin ถัดไป — โค้ดนี้ทำงานบนทุกแพ็กเกจโดยไม่ต้องแก้ — TSSOP20: PC0, SOP16: PC0, SOP8: PC1

ขั้นที่ 5: ตารางเปรียบเทียบ 4 แพ็กเกจในโค้ด

void print_package_info(void) {
    USART_Print("Package: ");
    #if CH32V003_PACKAGE == PACKAGE_SOP8
        USART_Print("SOP-8\r\n");
        USART_Print("  Pins: PC1,PC2,PC4,PA1,PA2,PD1,PD4,PD6\r\n");
    #elif CH32V003_PACKAGE == PACKAGE_SOP16
        USART_Print("SOP-16\r\n");
        USART_Print("  Pins: PA1,PA2,PC0-PC7,PD1,PD5-PD7\r\n");
    #elif CH32V003_PACKAGE == PACKAGE_TSSOP20
        USART_Print("TSSOP-20\r\n");
        USART_Print("  Pins: all 20 (full)\r\n");
    #elif CH32V003_PACKAGE == PACKAGE_QFN20
        USART_Print("QFN-20\r\n");
        USART_Print("  Pins: all 20 (same as TSSOP20)\r\n");
    #endif

    USART_Print("ADC pins: ");
    USART_PrintNum(IS_ADC_PIN(PD2) ? 8 : 5);
    USART_Print("\r\n");
    USART_Print("Hardware SPI: ");
    USART_Print(IS_VALID_PIN(PC5) ? "YES\r\n" : "NO\r\n");
}

🔍 ใช้ compile-time condition กับ CH32V003_PACKAGE — แสดงข้อมูลแพ็กเกจ — IS_VALID_PIN(PC5) → false บน SOP8 (ไม่มี SCK)

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
"PD2 not available" compile errorใช้ PACKAGE_SOP8 หรือ SOP16 — PD2 ไม่มีใช้ PA1, PA2, PC4, PD1, PD4, PD6 แทน — หรือเปลี่ยนเป็น TSSOP20
"SPI not available" compile errorSOP8 ไม่มี SPI (ไม่มี SCK pin)ใช้ shiftOut()/shiftIn() software SPI — หรือเปลี่ยนเป็น SOP16+
USART ไม่ทำงานบน SOP8Default USART pins (PD5/PD6) ไม่มีบน SOP8ใช้ USART_PINS_SOP8 (PD1=TX, PA2=RX)
PWM เฉพาะบาง pin ไม่ทำงานPWM pin ต่างกันตามแพ็กเกจใช้ IS_PWM_VALID_PACKAGE(pin, CH32V003_PACKAGE) เช็คก่อน
เปลี่ยนแพ็กเกจแต่โค้ดไม่เปลี่ยนลืมแก้ #define CH32V003_PACKAGEอยู่บรรทัดแรกสุด — ก่อน #include "SimpleHAL.h" เสมอ

🛠️ สร้าง Product: โค้ดเดียว รันบน SOP8 SOP16 TSSOP20

ไฟกระพริบ LED + วัดอุณหภูมิ + USART — ทำงานบนทุกแพ็กเกจโดยเปลี่ยนแค่บรรทัดแรก

// ====== เปลี่ยนแค่บรรทัดนี้ ======
#define CH32V003_PACKAGE PACKAGE_TSSOP20
// #define CH32V003_PACKAGE PACKAGE_SOP16
// #define CH32V003_PACKAGE PACKAGE_SOP8
// #define CH32V003_PACKAGE PACKAGE_QFN20
// ================================
#include "SimpleHAL.h"

// เลือก LED pin อัตโนมัติ
#if IS_VALID_PIN(PC0)
    #define LED_PIN PC0
#elif IS_VALID_PIN(PC1)
    #define LED_PIN PC1
#else
    #define LED_PIN PA1
#endif

// เลือก ADC pin อัตโนมัติ
#if IS_ADC_PIN(PD2) && IS_VALID_PIN(PD2)
    #define TEMP_PIN ADC_CH_PD2
#elif IS_ADC_PIN(PD1) && IS_VALID_PIN(PD1)
    #define TEMP_PIN ADC_CH_PD1
#else
    #define TEMP_PIN ADC_CH_PA1
#endif

// เลือก USART pins อัตโนมัติ
#if IS_VALID_PIN(PD5) && IS_VALID_PIN(PD6)
    #define MY_USART_PINS USART_PINS_DEFAULT
#else
    #define MY_USART_PINS USART_PINS_SOP8
#endif

int main(void)
{
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, MY_USART_PINS);
    ADC_SimpleInit();

    pinMode(LED_PIN, PIN_MODE_OUTPUT);

    USART_Print("CH32V003 Multi-Package Demo\r\n");
    USART_Print("Package: ");
    #if CH32V003_PACKAGE == PACKAGE_SOP8
        USART_Print("SOP8\r\n");
    #elif CH32V003_PACKAGE == PACKAGE_SOP16
        USART_Print("SOP16\r\n");
    #elif CH32V003_PACKAGE == PACKAGE_TSSOP20
        USART_Print("TSSOP20\r\n");
    #else
        USART_Print("QFN20\r\n");
    #endif

    while(1)
    {
        digitalWrite(LED_PIN, HIGH);
        Delay_Ms(250);
        digitalWrite(LED_PIN, LOW);
        Delay_Ms(250);

        float temp = ADC_ReadVoltage(TEMP_PIN, 3.3f);
        USART_Print("ADC: ");
        USART_PrintNum((int32_t)(temp * 1000));
        USART_Print("mV  LED: ");
        USART_Print(IS_VALID_PIN(LED_PIN) ? "ON\r\n" : "N/A\r\n");
    }
}

ผลลัพธ์: โค้ดเดียวกัน — คอมไพล์ 4 ครั้งด้วย CH32V003_PACKAGE ต่างกัน — ได้ 4 binary — ทำงานบน 4 แพ็กเกจ — ไม่ต้องแก้โค้ด

✏️ ลองทำดู

  1. คอมไพล์โค้ด "Product" ด้านบนด้วย PACKAGE_SOP8 — ดูว่า LED_PIN เป็น PC1, TEMP_PIN เป็น ADC_CH_PD1 — ทดสอบบนบอร์ด SOP8
  2. คอมไพล์ใหม่ด้วย PACKAGE_TSSOP20 — ดูว่า LED_PIN เป็น PC0, TEMP_PIN เป็น ADC_CH_PD2 — ทดสอบบนบอร์ด TSSOP20
  3. ใช้ #error เพื่อป้องกันไม่ให้ใช้ SPI บน SOP8 — เพิ่มโค้ด: #if CH32V003_PACKAGE == PACKAGE_SOP8 && !IS_VALID_PIN(PC5) #error "SPI requires SOP16 or TSSOP20" #endif
