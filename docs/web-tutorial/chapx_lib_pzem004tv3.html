⚠️ 220VAC! วงจรเหมือน PZEM-004T v1 — แต่ v3 ใช้โปรโตคอลต่างกัน — รองรับ ReadAll ในคำขอเดียว — มี Alarm registers เพิ่ม

📚 หลักการทำงาน

PZEM-004T v3 เป็นรุ่นอัพเกรดจาก v1 — ความแตกต่างหลักคือโปรโตคอล Modbus ที่เปลี่ยนไป — v3 รองรับ ReadAll ในคำขอเดียว (อ่าน 10 registers รวดเดียว) แทนการอ่านทีละ register — ทำให้เร็วขึ้นและลด traffic บน USART

ReadAll Request ส่งคำสั่งอ่าน registers 0x0000 ถึง 0x0009 ในครั้งเดียว — Response มี 25 bytes (1 addr + 1 func + 1 byteCount + 20 data + 2 CRC) — ลด overhead จาก 8 คำขอเหลือ 1 คำขอ

Alarm Registers เพิ่มจาก v1 — มี alarm status (0x0009) และ alarm threshold (0x000A-0x000F) — ตั้งค่า over-voltage, over-current, over-power thresholds ได้

Data Structure ใช้ PZEM_Data_t struct เก็บค่าทั้งหมดไว้ในตัวแปรเดียว — อ่านครั้งเดียว ได้ข้อมูลครบ — ไม่ต้องเรียกทีละฟังก์ชันเหมือน v1

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Init PZEM v3

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "PZEM004Tv3.h"

PZEM004Tv3_t pzem;

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_Init(115200);

    PZEM004Tv3_Init(&pzem, PD6, PD5, 0x01); // 9600 8N1
    USART_Print("PZEM v3 Ready\r\n");

🔍 Init เหมือน v1 — แต่ใช้ lib แยก — 9600 baud 8N1 — address default 0x01

ขั้นที่ 2: ReadAll — อ่านทีเดียวครบ

    PZEM_Data_t data;
    if(PZEM004Tv3_ReadAll(&pzem, &data)) {
        USART_Print("Voltage: "); USART_PrintFloat(data.voltage, 1);
        USART_Print("V\r\n");
        USART_Print("Current: "); USART_PrintFloat(data.current, 2);
        USART_Print("A\r\n");
        USART_Print("Power: "); USART_PrintFloat(data.power, 1);
        USART_Print("W\r\n");
        USART_Print("Energy: "); USART_PrintFloat(data.energy, 1);
        USART_Print("kWh\r\n");
        USART_Print("Freq: "); USART_PrintFloat(data.frequency, 1);
        USART_Print("Hz\r\n");
        USART_Print("PF: "); USART_PrintFloat(data.pf, 2);
        USART_Print("\r\n");
    } else {
        USART_Print("ReadAll failed\r\n");
    }

🔍 ReadAll(dev, &data) คืน bool — true = สำเร็จ — data struct มี voltage, current, power, energy, frequency, pf, alarm — อ่าน 10 registers ใน 1 คำขอ ~200ms

ขั้นที่ 3: Loop + Alarm Threshold

    PZEM004Tv3_SetAlarmThreshold(&pzem, 2500, 5000, 20000);
    // over-V > 250V, over-A > 5A, over-W > 2000W

    Timer_t tmr;
    Start_Timer(&tmr, 2000, 1);

    while(1) {
        if(Is_Timer_Expired(&tmr)) {
            PZEM_Data_t d;
            if(PZEM004Tv3_ReadAll(&pzem, &d)) {
                USART_Print("V:"); USART_PrintFloat(d.voltage, 1);
                USART_Print(" A:"); USART_PrintFloat(d.current, 2);
                USART_Print(" W:"); USART_PrintFloat(d.power, 1);
                USART_Print(" kWh:"); USART_PrintFloat(d.energy, 2);

                if(d.alarm) {
                    USART_Print(" !ALARM!");
                    if(d.alarm & 0x01) USART_Print(" OV");
                    if(d.alarm & 0x02) USART_Print(" OC");
                    if(d.alarm & 0x04) USART_Print(" OP");
                }
                USART_Print("\r\n");
            }
        }
    }
}

🔍 ตั้งค่า alarm threshold ด้วย SetAlarmThreshold() — ค่าที่ส่งเป็น ×10 (2500 = 250.0V) — alarm bit0=over-V, bit1=over-A, bit2=over-W — ใช้ Non-Blocking Timer อ่านทุก 2 วิ

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
ReadAll คืน false ตลอดใช้ PZEM v1 กับ lib v3 หรือกลับกันเช็ค sticker ที่ตัว PZEM — ต้องตรงรุ่น — v3 มีข้อความ "V3.0"
ค่าที่อ่านได้ไม่ตรงใช้ scaling factor ผิดรุ่นv3 ใช้ scaling ต่างจาก v1 — ต้องใช้ lib v3 โดยเฉพาะ
Alarm ไม่ทำงานไม่ได้ตั้ง Alarm Thresholdต้องเรียก SetAlarmThreshold ก่อน — ค่าเป็น ×10
Timeout ในการอ่านBaud rate ไม่ตรง (9600 vs 4800)v3 ใช้ 9600 baud — ถ้าใช้ v1 lib อาจเป็น 4800
Energy reset ไม่ได้Register write-protectedต้องส่งคำสั่ง reset ตามโปรโตคอล v3 — ห้าม power-cycle ระหว่าง reset

🛠️ สร้าง Product: Smart Power Monitor

PZEM v3 + OLED + EEPROM — บันทึกข้อมูลรายชั่วโมง — ตั้ง alarm ตัดไฟอัตโนมัติ

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "PZEM004Tv3.h"
#include "OLED.h"

int main(void)
{
    SystemCoreClockUpdate();
    Timer_Init();
    I2C_Init(PC1, PC2, I2C_SPEED_STANDARD);
    USART_Init(115200);

    PZEM004Tv3_t pzem;
    PZEM004Tv3_Init(&pzem, PD6, PD5, 0x01);
    PZEM004Tv3_SetAlarmThreshold(&pzem, 2500, 10000, 22000);

    OLED_t oled;
    OLED_Init(&oled, 0x3C, OLED_128x64);
    OLED_SetFont(&oled, FONT_8x16);

    Timer_t tmr;
    Start_Timer(&tmr, 2000, 1);
    pinMode(PC0, PIN_MODE_OUTPUT); // Relay control

    while(1)
    {
        if(Is_Timer_Expired(&tmr))
        {
            PZEM_Data_t d;
            if(!PZEM004Tv3_ReadAll(&pzem, &d)) continue;

            OLED_Clear(&oled);
            OLED_SetCursor(&oled, 0, 0);
            OLED_PrintFloat(&oled, "", d.voltage, 1);
            OLED_Print(&oled, "V  ");

            OLED_SetCursor(&oled, 60, 0);
            OLED_PrintFloat(&oled, "", d.current, 2);
            OLED_Print(&oled, "A");

            OLED_SetCursor(&oled, 0, 2);
            OLED_PrintFloat(&oled, "", d.power, 1);
            OLED_Print(&oled, "W  ");

            OLED_SetCursor(&oled, 60, 2);
            OLED_PrintFloat(&oled, "PF:", d.pf, 2);

            OLED_SetCursor(&oled, 0, 4);
            OLED_PrintFloat(&oled, "", d.frequency, 1);
            OLED_Print(&oled, "Hz");

            OLED_SetCursor(&oled, 0, 6);
            OLED_PrintFloat(&oled, "", d.energy, 2);
            OLED_Print(&oled, "kWh");

            if(d.alarm) {
                digitalWrite(PC0, HIGH); // ตัด relay
                OLED_SetCursor(&oled, 90, 0);
                OLED_Print(&oled, "CUT!");
            }

            OLED_Display(&oled);
        }
    }
}

✏️ ลองทำดู

  1. เปลี่ยน Slave Address เป็น 0x02 — ต่อ PZEM 2 ตัวบนบัส USART เดียว (ผ่าน RS485) — อ่านทั้ง 2 ตัว
  2. ส่งข้อมูล JSON ผ่าน USART ไปยัง ESP01 — สร้าง Web Dashboard ด้วย WiFi
  3. บันทึกค่า kWh รายวันลง RTC + EEPROM — สร้างกราฟแท่งแสดงการใช้ไฟ 7 วัน
  4. ตั้ง Alarm แบบ 2 ระดับ — Warning ที่ 80% และ Critical ตัดไฟที่ 100%
