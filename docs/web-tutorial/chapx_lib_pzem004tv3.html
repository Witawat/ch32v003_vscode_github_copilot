PZEM-004T v3 — Modbus-RTU แบบ ReadAll ในครั้งเดียว

ระดับสูง

🧭 แผนภาพการทำงาน

flowchart TD
    A["PZEM004Tv3_Init(&pzem, 1, USART_PINS_DEFAULT)"] --> B["PZEM004Tv3_ReadAll(&pzem)"]
    B --> C["PZEM004Tv3_GetVoltage(&pzem)"]
    C --> D["PZEM004Tv3_GetCurrent(&pzem)"]
    D --> E["แสดงผล V / A / W / kWh / Hz / PF"]

⚠️ 220VAC! — ต่อผ่าน USART (9600 8N1) — ใช้ ReadAll ในคำขอเดียว — มีฟังก์ชัน Get แยกแต่ละค่า

รู้จักขา

  • RX: ขารับคำสั่ง Modbus จาก MCU
  • TX: ขาส่งข้อมูล Modbus ไป MCU
  • VCC: จ่ายไฟ 5V
  • GND: Ground ร่วม
  • CT 100A: Clamp ammeter 100A — หนีบสาย L

วิธีต่อทีละขั้น

  1. ต่อ RX ของ PZEM v3 ไป PD5 (TX) ของ MCU — ส่ง Modbus
  2. ต่อ TX ของ PZEM v3 ไป PD6 (RX) ของ MCU — รับข้อมูล
  3. ต่อ VCC ไป 5V และ GND ไป GND — จ่ายไฟ
  4. หนีบ CT Clamp รอบสาย L โดยให้ ลูกศรชี้ไปทางโหลด

📚 หลักการทำงาน

PZEM-004T v3 เป็นรุ่นอัพเกรดจาก v1 — รองรับ ReadAll ในคำขอเดียว (อ่าน 10 registers รวดเดียว) — ใช้ฟังก์ชัน Get แยกเช่น GetVoltage(), GetCurrent() แทนการเข้าถึง struct โดยตรง

ReadAll Request ส่งคำสั่งอ่าน registers 0x0000 ถึง 0x0009 ในครั้งเดียว — Response มี 25 bytes — ลด overhead จาก 8 คำขอเหลือ 1 คำขอ

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Init PZEM v3

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "PZEM004Tv3.h"

PZEM004Tv3 pzem;

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_Init(115200);

    PZEM004Tv3_Init(&pzem, 1, USART_PINS_DEFAULT);
    // slave address=1, default USART pins
    USART_Println("PZEM v3 Ready!");

🔍 PZEM004Tv3_Init(&pzem, addr, pins) — slave address + USART pins — 9600 baud 8N1 โดย default

ขั้นที่ 2: ReadAll + Get Values

    if(PZEM004Tv3_ReadAll(&pzem)) {
        float v = PZEM004Tv3_GetVoltage(&pzem);
        float a = PZEM004Tv3_GetCurrent(&pzem);
        float w = PZEM004Tv3_GetPower(&pzem);
        float e = PZEM004Tv3_GetEnergy(&pzem);
        float f = PZEM004Tv3_GetFrequency(&pzem);
        float pf = PZEM004Tv3_GetPowerFactor(&pzem);

        USART_Print("V:"); USART_PrintFloat(v, 1);
        USART_Print(" A:"); USART_PrintFloat(a, 2);
        USART_Print(" W:"); USART_PrintFloat(w, 1);
        USART_Print(" kWh:"); USART_PrintFloat(e, 2);
        USART_Print(" Hz:"); USART_PrintFloat(f, 1);
        USART_Print(" PF:"); USART_PrintFloat(pf, 2);
        USART_Println("");
    } else {
        USART_Println("ReadAll failed");
    }

🔍 PZEM004Tv3_ReadAll(&pzem) อ่าน 10 registers ใน 1 คำขอ ~200ms — ใช้ฟังก์ชัน Get แยกค่าทีละตัว — GetVoltage(), GetCurrent(), GetPower(), GetEnergy(), GetFrequency(), GetPowerFactor()

ขั้นที่ 3: Loop + Alarm

    Timer_t tmr;
    Start_Timer(&tmr, 2000, 1);

    while(1) {
        if(Is_Timer_Expired(&tmr)) {
            if(PZEM004Tv3_ReadAll(&pzem)) {
                float v = PZEM004Tv3_GetVoltage(&pzem);
                float a = PZEM004Tv3_GetCurrent(&pzem);
                float w = PZEM004Tv3_GetPower(&pzem);

                USART_Print("V:"); USART_PrintFloat(v, 1);
                USART_Print(" A:"); USART_PrintFloat(a, 2);
                USART_Print(" W:"); USART_PrintFloat(w, 1);

                if(v > 250.0f) USART_Print(" !OV!");
                if(a > 5.0f)   USART_Print(" !OC!");
                USART_Println("");
            }
        }
    }
}

🔍 ใช้ Non-Blocking Timer อ่านทุก 2 วิ — ตรวจสอบ over-voltage/over-current ด้วย threshold ใน code — ฟังก์ชัน Get ให้ค่าที่พร้อมใช้งานทันที

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
ReadAll คืน false ตลอดสาย RX/TX สลับ หรือ baud ไม่ตรงเช็ค PD5-TX→RX, PD6-RX→TX — 9600 8N1
ค่าที่อ่านได้ไม่ตรงใช้ scaling factor ผิดรุ่นเช็ค sticker ที่ตัว PZEM — ต้องตรงรุ่น v3
Timeout ในการอ่านBaud rate ไม่ตรง (9600 vs 4800)v3 ใช้ 9600 baud — ถ้าใช้ v1 lib อาจเป็น 4800

🛠️ สร้าง Product: Smart Power Monitor

PZEM v3 + OLED — แสดงข้อมูลไฟฟ้า real-time

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "PZEM004Tv3.h"
#include "oled_i2c.h"

int main(void)
{
    SystemCoreClockUpdate();
    Timer_Init();
    I2C_Init(PC1, PC2, I2C_SPEED_STANDARD);
    USART_Init(115200);

    PZEM004Tv3 pzem;
    PZEM004Tv3_Init(&pzem, 1, USART_PINS_DEFAULT);

    OLED_Handle oled;
    OLED_Init(&oled, OLED_128x64, 0x3C);
    OLED_SetFont(&oled, FONT_8x16);

    Timer_t tmr;
    Start_Timer(&tmr, 2000, 1);
    pinMode(PC0, PIN_MODE_OUTPUT); // Relay control

    while(1)
    {
        if(Is_Timer_Expired(&tmr))
        {
            if(!PZEM004Tv3_ReadAll(&pzem)) continue;

            float v = PZEM004Tv3_GetVoltage(&pzem);
            float a = PZEM004Tv3_GetCurrent(&pzem);
            float w = PZEM004Tv3_GetPower(&pzem);
            float e = PZEM004Tv3_GetEnergy(&pzem);
            float f = PZEM004Tv3_GetFrequency(&pzem);
            float pf = PZEM004Tv3_GetPowerFactor(&pzem);

            OLED_Clear(&oled);
            char buf[64];
            snprintf(buf, sizeof(buf), "%.1fV  ", v);
            OLED_DrawStringAlign(&oled, 0, 0, buf, ALIGN_LEFT);

            snprintf(buf, sizeof(buf), "%.2fA", a);
            OLED_DrawStringAlign(&oled, 60, 0, buf, ALIGN_LEFT);

            snprintf(buf, sizeof(buf), "%.1fW", w);
            OLED_DrawStringAlign(&oled, 0, 20, buf, ALIGN_LEFT);

            snprintf(buf, sizeof(buf), "%.1fHz  PF: %.2f", f, pf);
            OLED_DrawStringAlign(&oled, 0, 40, buf, ALIGN_LEFT);

            snprintf(buf, sizeof(buf), "%.2fkWh", e);
            OLED_DrawStringAlign(&oled, 0, 60, buf, ALIGN_LEFT);

            if(v > 250.0f || a > 10.0f) {
                digitalWrite(PC0, HIGH); // ตัด relay
                snprintf(buf, sizeof(buf), "CUT!");
                OLED_DrawStringAlign(&oled, 90, 0, buf, ALIGN_LEFT);
            }

            OLED_Update(&oled);
        }
    }
}
