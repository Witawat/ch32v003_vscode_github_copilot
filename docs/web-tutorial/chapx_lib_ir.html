VS1838B มีตัวกรอง 38kHz ในตัว — เอาต์พุตเป็น digital LOW เมื่อได้รับสัญญาณ IR ที่ความถี่ carrier 38kHz เท่านั้น, ใช้ไฟ 3.3V หรือ 5V ก็ได้, ขา OUT ต่อกับ GPIO ธรรมดา

📚 หลักการทำงาน

NEC Protocol เป็นโปรโตคอล IR remote control ที่ใช้กันแพร่หลายที่สุด — ทุกปุ่มของรีโมท NEC จะส่ง frame 32-bit (Address 8-bit + ~Address 8-bit + Command 8-bit + ~Command 8-bit) ผ่านการมอดูเลตด้วย carrier 38kHz

NEC Frame Structure:

  • Lead Code: 9ms LOW + 4.5ms HIGH — แจ้งว่ากำลังจะเริ่ม frame
  • Logic 0: 562.5µs LOW + 562.5µs HIGH
  • Logic 1: 562.5µs LOW + 1687.5µs HIGH (HIGH = 3× LOW)
  • Repeat Code: 9ms LOW + 2.25ms HIGH — ส่งทุก 110ms เมื่อกดปุ่มค้าง

VS1838B demodulate carrier 38kHz ออกจากสัญญาณให้อัตโนมัติ — เราเห็นแต่ LOW/HIGH pulse ไม่ต้องยุ่งกับ carrier เอง

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Init + Setup

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "IR_Remote.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();  // จำเป็นสำหรับวัด pulse width
    IR_Init(PC3);  // VS1838B OUT → PC3
    USART_Init(USART1, BAUD_115200);
    USART_Println("IR Receiver Ready!");
    USART_Println("Press any key on remote...");

🔍 IR_Init(pin) — ตั้ง GPIO เป็น input + interrupt (falling edge), Timer_Init() จำเป็นเพราะใช้ SysTick วัดความกว้าง pulse

ขั้นที่ 2: รับและถอดรหัส

    uint8_t addr, cmd;
    uint32_t last_time = 0;
    while(1) {
        if(IR_Available()) {  // มี NEC frame เข้ามาครบ
            IR_Decode(&addr, &cmd);
            USART_Print("Addr: 0x"); USART_PrintHex(addr);
            USART_Print(" Cmd: 0x"); USART_PrintHex(cmd);
            USART_Println("");
        }
    }
}

🔍 IR_Available() — non-blocking เช็คว่ามี frame ครบหรือยัง, IR_Decode(&addr, &cmd) — ถอดรหัส 32-bit frame ตรวจสอบ checksum (~addr, ~cmd), ใช้ USART_PrintHex() แสดงค่า hex

ขั้นที่ 3: นำไปใช้ — ควบคุม LED ตามปุ่ม

    pinMode(PC0, PIN_MODE_OUTPUT);  // LED1
    pinMode(PC1, PIN_MODE_OUTPUT);  // LED2
    uint8_t addr, cmd;

    while(1) {
        if(IR_Available()) {
            IR_Decode(&addr, &cmd);
            switch(cmd) {
                case 0x45: digitalWrite(PC0, HIGH); break;  // ปุ่ม 1 = LED1 ON
                case 0x46: digitalWrite(PC0, LOW);  break;  // ปุ่ม 2 = LED1 OFF
                case 0x47: digitalToggle(PC1);       break;  // ปุ่ม 3 = Toggle LED2
                case 0x44: digitalWrite(PC0, LOW);   // ปุ่ม 0 = ALL OFF
                           digitalWrite(PC1, LOW); break;
                default: USART_Print("Unknown: 0x");
                         USART_PrintHex(cmd); USART_Println(""); break;
            }
        }
    }
}

🔍 🔴 แต่ละรีโมทมีรหัสปุ่มไม่เหมือนกัน — ต้องอ่านค่าจริงก่อนแล้วจึง map, ใช้ switch(cmd) จับคู่ปุ่ม, Address ใช้กรองกรณีมีรีโมทหลายตัว

ฟังก์ชันคำอธิบายตัวอย่าง

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
ไม่เคยรับ IR ได้เลยVS1838B ขากลับด้าน หรือใช้ GPIO ผิดเช็ค OUT→MCU, VCC→3.3V, GND→GND, VS1838B มองจากด้านโดม: OUT,GND,VCC
รับค่าเดิมซ้ำตลอดลืมเคลียร์ IR buffer หลังอ่านเรียก IR_Decode() ซึ่งจะเคลียร์อัตโนมัติ หรือเรียก IR_Reset()
ค่า cmd ไม่ตรงที่คาดไว้รีโมทแต่ละยี่ห้อใช้รหัสไม่เหมือนกันใช้ code ในขั้นที่ 1 อ่านค่าจริงก่อน แล้วจึง map ใหม่
VS1838B รับสัญญาณรบกวนหลอดไฟ LED/หลอดฟลูออเรสเซนต์ปล่อย IRเพิ่มตัวกรอง (ใส่ท่อหดที่ตัวรับ), หลีกเลี่ยงแสงสว่างจ้าส่องโดยตรง
บางครั้ง decode ผิดNoise หรือ distance ไกลเกินไปลดระยะห่างเหลือ <5m, ใช้ capacitor 100nF ที่ VCC-GND VS1838B

🛠️ สร้าง Product: IR Home Automation Controller

รับคำสั่งจากรีโมท NEC → ควบคุม LED 4 ดวง + บันทึกปุ่มล่าสุด

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "IR_Remote.h"

uint8_t led_pins[4] = {PC0, PC1, PC2, PC3};
uint8_t led_state[4] = {0,0,0,0};

int main(void)
{
    SystemCoreClockUpdate();
    Timer_Init();
    USART_Init(USART1, BAUD_115200);
    IR_Init(PC4);

    for(int i=0; i<4; i++) pinMode(led_pins[i], PIN_MODE_OUTPUT);

    uint8_t addr, cmd;
    USART_Println("IR Controller Ready");

    while(1)
    {
        if(IR_Available()) {
            IR_Decode(&addr, &cmd);
            USART_Print("Cmd: 0x"); USART_PrintHex(cmd);

            if(cmd == 0x45) { led_state[0] = !led_state[0]; // toggle LED1
                digitalWrite(led_pins[0], led_state[0]); }
            else if(cmd == 0x46) { led_state[1] = !led_state[1];
                digitalWrite(led_pins[1], led_state[1]); }
            else if(cmd == 0x47) { led_state[2] = !led_state[2];
                digitalWrite(led_pins[2], led_state[2]); }
            else if(cmd == 0x44) { led_state[3] = !led_state[3];
                digitalWrite(led_pins[3], led_state[3]); }
            else if(cmd == 0x40) { // ปุ่ม "ALL OFF"
                for(int i=0;i<4;i++) { led_state[i]=0; digitalWrite(led_pins[i],LOW); }
            }
            USART_Print(" States:"); for(int i=0;i<4;i++) { USART_PrintInt(led_state[i]); USART_Print(" "); }
            USART_Println("");
        }
    }
}
