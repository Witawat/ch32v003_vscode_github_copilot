IR Remote — NEC Protocol, VS1838B Receiver

ระดับกลาง

🧭 แผนภาพการทำงาน

flowchart TD
    A["IR_ReceiverInit(PC0)"] --> B["รอ interrupt / polling"]
    B --> C["IR_Process() — ตรวจสอบ Lead Code"]
    C --> D["อ่าน 32 bits NEC"]
    D --> E["IR_GetCode()"]
    E --> F["Switch(code) → ทำงานตามปุ่ม"]

VS1838B: OUT=LOW เมื่อเจอ IR, Carrier: 38kHz

รู้จักขา

  • OUT: ขาสัญญาณ ต่อกับ PC3 (อินพุต) — LOW เมื่อเจอ IR 38kHz
  • VCC: ไฟเลี้ยง 3.3V
  • GND: ต่อกราวด์ร่วมกับ MCU

วิธีต่อทีละขั้น

  1. ต่อ PC3 ไป OUT ของ VS1838B — รับสัญญาณ IR
  2. ต่อ 3.3V ไป VCC ของ VS1838B — จ่ายไฟ
  3. ต่อ GND ไป GND ของ VS1838B — กราวด์ร่วม

📚 หลักการทำงาน

NEC Protocol เป็นโปรโตคอล IR remote control — IR_ReceiverInit() ตั้งขารับ, IR_Process() ประมวลผล, IR_GetCode() คืนค่า 32-bit code

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Init + Setup

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "IR.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();  // จำเป็นสำหรับวัด pulse width
    USART_Init(USART1, BAUD_115200);

    IR_ReceiverInit(PC0);  // VS1838B OUT → PC0
    USART_Println("IR Receiver Ready!");

🔍 IR_ReceiverInit(pin) — ตั้ง GPIO เป็น input + interrupt (falling edge) — ไม่มี instance

ขั้นที่ 2: รับและถอดรหัส

    while(1) {
        IR_Process();
        if(IR_Available()) {
            uint32_t code = IR_GetCode();
            USART_Print("Code: 0x");
            USART_PrintHex(code);
            USART_Println("");
        }
    }
}

🔍 IR_Process() — ต้องเรียกใน loop ตลอดเวลา, IR_Available() — เช็คว่ามี frame ครบ, IR_GetCode() — คืนค่า 32-bit NEC code

ขั้นที่ 3: นำไปใช้ — ควบคุม LED ตามปุ่ม

    pinMode(PC0, PIN_MODE_OUTPUT);  // LED1
    pinMode(PC1, PIN_MODE_OUTPUT);  // LED2

    while(1) {
        IR_Process();
        if(IR_Available()) {
            uint32_t code = IR_GetCode();
            uint8_t cmd = code & 0xFF;  // command byte

            switch(cmd) {
                case 0x45: digitalWrite(PC0, HIGH); break;
                case 0x46: digitalWrite(PC0, LOW);  break;
                case 0x47: digitalToggle(PC1);       break;
                case 0x44: digitalWrite(PC0, LOW);
                           digitalWrite(PC1, LOW); break;
                default: USART_Print("Unknown: 0x");
                         USART_PrintHex(cmd); USART_Println(""); break;
            }
        }
    }
}

🛠️ สร้าง Product: IR Home Automation Controller

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "IR.h"

uint8_t led_pins[4] = {PC0, PC1, PC2, PC3};
uint8_t led_state[4] = {0,0,0,0};

int main(void)
{
    SystemCoreClockUpdate();
    Timer_Init();
    USART_Init(USART1, BAUD_115200);

    IR_ReceiverInit(PC4);
    for(int i=0; i<4; i++) pinMode(led_pins[i], PIN_MODE_OUTPUT);

    while(1)
    {
        IR_Process();
        if(IR_Available()) {
            uint32_t code = IR_GetCode();
            uint8_t cmd = code & 0xFF;

            if(cmd == 0x45) { led_state[0] = !led_state[0];
                digitalWrite(led_pins[0], led_state[0]); }
            else if(cmd == 0x46) { led_state[1] = !led_state[1];
                digitalWrite(led_pins[1], led_state[1]); }
            else if(cmd == 0x47) { led_state[2] = !led_state[2];
                digitalWrite(led_pins[2], led_state[2]); }
            else if(cmd == 0x44) { led_state[3] = !led_state[3];
                digitalWrite(led_pins[3], led_state[3]); }

            USART_Print("Cmd: 0x"); USART_PrintHex(cmd); USART_Println("");
        }
    }
}

✏️ ลองทำดู

  1. ทำ IR Repeater — รับสัญญาณจากรีโมท → ส่งซ้ำด้วย IR LED
  2. ทำ IR remote code learner — กดปุ่มบนรีโมท → บันทึก code ลง EEPROM
  3. ทำ IR-controlled servo — ปุ่ม CH+/CH- หมุน servo
