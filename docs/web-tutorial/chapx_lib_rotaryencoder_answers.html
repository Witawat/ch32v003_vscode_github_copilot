📝 เฉลย — Rotary Encoder

ข้อ 1: Volume Control — หมุนปรับ PWM 0–100%, กด mute

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "RotaryEncoder.h"

#define ENC_CLK PC0
#define ENC_DT  PC1
#define ENC_SW  PC2
#define SPK_PIN PC3
#define PWM_CH  1

volatile int16_t volume = 50;      // 0–100%
volatile bool muted = false;
int16_t saved_vol = 50;
RotaryEncoder enc;
int32_t last_pos = 0;

void enc_sw_cb(void) {
    if(muted) { muted = false; volume = saved_vol; }
    else      { muted = true;  saved_vol = volume; volume = 0; }
}

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_Init(115200);

    Rotary_Init(&enc, PC0, PC1, PC2);
    Rotary_SetCallback(&enc, ROTARY_EVENT_SW, enc_sw_cb);

    PWM_SimpleInit(SPK_PIN, PWM_CH, 1000, 0);
    Timer_t update_tmr;
    Start_Timer(&update_tmr, 100, 1);

    while(1) {
        int32_t pos = Rotary_GetPosition(&enc);
        if(pos != last_pos) {
            int8_t diff = (int8_t)(pos - last_pos);
            volume += diff;
            if(volume < 0) volume = 0;
            if(volume > 100) volume = 100;
            last_pos = pos;
        }

        if(Is_Timer_Expired(&update_tmr)) {
            uint16_t duty = (uint16_t)((uint32_t)volume * 1000 / 100);
            PWM_SetDuty(SPK_PIN, PWM_CH, muted ? 0 : duty);
            USART_Printf("Vol:%d%% %s\n", volume, muted ? "[MUTED]" : "");
        }
    }
}
🧠 แนวคิด: volume เก็บเป็น % (0–100) — callback CW/CCW เพิ่มลดค่า clamp 0–100 — enc_sw_cb สลับ mute: เซฟค่าเก่าไว้ saved_vol ก่อนเซ็ต 0, unmute ก็ดึงค่ากลับ — map 0–100 → duty 0–1000 ด้วย (vol*1000/100) — Timer 100ms อัปเดต PWM ป้องกัน jitter — USART_Printf แสดงสถานะ volume พร้อม MUTED flag

ข้อ 2: Menu Scroll — เลือกรายการด้วย rotary

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "RotaryEncoder.h"
#include "oled_i2c.h"

#define ENC_CLK PC0
#define ENC_DT  PC1
#define ENC_SW  PC2

const char *menu[] = {"LED ON", "LED OFF", "BLINK 1Hz", "BLINK 5Hz", "EXIT"};
#define MENU_COUNT 5
volatile int8_t idx = 0;
volatile bool selected = false;
RotaryEncoder enc;
int32_t last_pos2 = 0;

void enc_sw_cb2(void) { selected = true; }

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    OLED_Handle oled;
    OLED_Init(&oled, OLED_128x64, 0x3C);
    Rotary_Init(&enc, PC0, PC1, PC2);
    Rotary_SetCallback(&enc, ROTARY_EVENT_SW, enc_sw_cb2);

    while(1) {
        int32_t pos = Rotary_GetPosition(&enc);
        if(pos != last_pos2) {
            idx = (idx + (int8_t)(pos - last_pos2)) % MENU_COUNT;
            if(idx < 0) idx += MENU_COUNT;
            last_pos2 = pos;
        }

        OLED_Clear(&oled);
        OLED_SetFont(&oled, FONT_8X16);
        char mbuf[64];
        snprintf(mbuf, sizeof(mbuf), "-- Menu --");
        OLED_DrawStringAlign(&oled, 0, 0, mbuf, ALIGN_LEFT);
        for(uint8_t i = 0; i < MENU_COUNT; i++) {
            if(i == idx) { snprintf(mbuf, sizeof(mbuf), ">%s", menu[i]); }
            else          { snprintf(mbuf, sizeof(mbuf), " %s", menu[i]); }
            OLED_DrawStringAlign(&oled, 0, 10 + i * 10, mbuf, ALIGN_LEFT);
        }
        OLED_Update(&oled);

        if(selected) {
            selected = false;
            OLED_Clear(&oled);
            snprintf(mbuf, sizeof(mbuf), "Selected:");
            OLED_DrawStringAlign(&oled, 0, 0, mbuf, ALIGN_LEFT);
            snprintf(mbuf, sizeof(mbuf), "%s", menu[idx]);
            OLED_DrawStringAlign(&oled, 0, 10, mbuf, ALIGN_LEFT);
            OLED_Update(&oled);
            Delay_Ms(1500);
        }

        Delay_Ms(30);
    }
}
🧠 แนวคิด: ใช้ rotary เลือก index ใน array menu[] — CW/CCW เลื่อนขึ้นลงด้วย modulo — แสดงผลด้วย > prefix หน้ารายการที่เลือก — กดปุ่ม → selected flag → แสดงข้อความ selected ค้าง 1.5 วิ — สำหรับการนำไปใช้จริง ควรเปลี่ยนจาก delay เป็น state machine และสั่ง action จริงตาม menu[idx]

ข้อ 3: Acceleration — หมุนเร็วขึ้น step เพิ่มอัตโนมัติ

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "RotaryEncoder.h"

#define ENC_CLK PC0
#define ENC_DT  PC1

volatile int32_t value = 0;
uint32_t last_tick = 0;
int8_t   tick_count = 0;
RotaryEncoder enc;

void enc_tick_cb(int8_t dir) {
    uint32_t now = Timer_GetMs();
    uint32_t dt = now - last_tick;
    last_tick = now;

    tick_count = (dir > 0) ? tick_count + 1 : tick_count - 1;

    int16_t step;
    if(dt < 20)       step = 100;      // เร็วมาก → กระโดด 100
    else if(dt < 50)  step = 10;       // เร็ว → กระโดด 10
    else if(dt < 120) step = 5;        // ปานกลาง → 5
    else              step = 1;        // ช้า → ทีละ 1

    value += dir * step;

    // clamp 0–5000
    if(value < 0)    value = 0;
    if(value > 5000) value = 5000;
}

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_Init(115200);

    Rotary_Init(&enc, PC0, PC1, 0xFF);  // ไม่มีปุ่มกด
    Rotary_SetCallback(&enc, ROTARY_EVENT_TICK, enc_tick_cb);

    Timer_t print_tmr;
    Start_Timer(&print_tmr, 200, 1);

    while(1) {
        if(Is_Timer_Expired(&print_tmr)) {
            USART_Printf("Value: %ld  (speed: %dms)\n", value,
                         tick_count ? (int)(Timer_GetMs() - last_tick) : 0);
        }
    }
}
🧠 แนวคิด: วัด dt (เวลาระหว่าง tick) → ถ้า dt สั้น = หมุนเร็ว → เพิ่ม step ให้ใหญ่ขึ้น — threshold: <20ms→step 100, <50ms→10, <120ms→5, else→1 — ทำให้หมุนช้า ๆ ได้ precision สูง และหมุนเร็ว ๆ เปลี่ยนค่าไว — clamp value 0–5000 ป้องกัน overflow — RotaryEncoder_AttachTick รับ callback พร้อมทิศทาง — ใช้ Timer จับเวลาแทน delay ไม่บล็อก loop — สำหรับ UI จริง เพิ่ม acceleration curve แบบ exponential: step = 1 << (speed_level)
