📝 เฉลย — I2C Scanner

ข้อ 1: Scan I2C Bus + Map อุปกรณ์ที่พบ

#define I2C_MAX_DEV  16

typedef struct {
    uint8_t  addr;
    char     name[20];
    uint8_t  detected;
} i2c_device_t;

i2c_device_t i2c_map[I2C_MAX_DEV];
uint8_t i2c_map_cnt = 0;

void i2c_register_device(uint8_t addr, const char *name) {
    if(i2c_map_cnt >= I2C_MAX_DEV) return;
    i2c_map[i2c_map_cnt].addr = addr;
    strncpy(i2c_map[i2c_map_cnt].name, name, 19);
    i2c_map[i2c_map_cnt].detected = 0;
    i2c_map_cnt++;
}

ข้อ 2: Address Conflict Detector

typedef struct {
    uint8_t addr;
    uint8_t count;
    uint8_t ack_speed;
} i2c_conflict_t;

i2c_conflict_t conflicts[127];

void i2c_conflict_scan(void) {
    I2C_SimpleInit(I2C_100KHZ, I2C_PINS_DEFAULT);
    memset(conflicts, 0, sizeof(conflicts));

    USART_Println("=== Address Conflict Check ===");

    for(uint8_t addr=0x03; addr<0x78; addr++) {
        // เช็ค 3 ครั้งกัน false positive
        uint8_t ack_cnt = 0;
        for(uint8_t attempt=0; attempt<3; attempt++) {
            I2C_Start();
            I2C_SendAddr(addr << 1);
            if(I2C_WaitAck(3)) ack_cnt++;
            I2C_Stop();
            Delay_Ms(1);
        }
        if(ack_cnt >= 2) {
            conflicts[addr].addr = addr;
            conflicts[addr].count = ack_cnt;
        }
    }

    // แสดงอุปกรณ์ที่เจอ
    for(uint8_t a=0x03; a<0x78; a++) {
        if(conflicts[a].count > 0) {
            USART_Print("0x");
            USART_PrintHexByte(a);
            USART_Print(" ack=");
            USART_PrintNum(conflicts[a].count);
            USART_Println("/3");
        }
    }

    // ตรวจ address ที่ซ้ำกัน (พบมากกว่า 1 อุปกรณ์)
    USART_Println("---");
    i2c_check_conflict_pairs();
}

void i2c_check_conflict_pairs(void) {
    // ตรวจ I2C addresses ที่รู้ว่าซ้ำกันบ่อย
    uint8_t conflict_pairs[][2] = {
        {0x50, 0x57},  // EEPROM series
        {0x68, 0x69},  // DS3231 / MPU6050
        {0x76, 0x77},  // BMP280 / BME280
        {0x3C, 0x3D},  // OLED
        {0x27, 0x3F},  // LCD I2C
    };

    for(uint8_t i=0; i<5; i++) {
        uint8_t a1 = conflict_pairs[i][0];
        uint8_t a2 = conflict_pairs[i][1];
        if(conflicts[a1].count > 0 && conflicts[a2].count > 0) {
            USART_Print("WARNING: Both 0x");
            USART_PrintHexByte(a1);
            USART_Print(" and 0x");
            USART_PrintHexByte(a2);
            USART_Println(" found!");
        }
    }
}
🧠 แนวคิด: Scan 3 ครั้งต่อ address กัน false positive — ถ้า ACK >= 2 ครั้งถือว่ามีอุปกรณ์ — เก็บผลลง array conflicts — conflict_pairs เก็บคู่ address ที่รู้ว่ามักชนกัน — ตัวอย่าง: OLED (0x3C) กับ (0x3D), BMP280 (0x76) กับ BME280 (0x77) — ถ้าเจอทั้งคู่ให้ WARNING — บางอุปกรณ์เปลี่ยน address ได้ผ่าน hardware pin (A0, A1) หรือ jumper

ข้อ 3: Auto-Config — ตรวจจับและตั้งค่าอุปกรณ์อัตโนมัติ

typedef enum {
    DEV_NONE = 0,
    DEV_OLED,
    DEV_DS3231,
    DEV_BH1750,
    DEV_BMP280,
    DEV_EEPROM,
} i2c_dev_type_t;

typedef struct {
    i2c_dev_type_t type;
    uint8_t  addr;
    uint8_t  detected;
} dev_config_t;

dev_config_t devs[8];
uint8_t dev_count = 0;

void i2c_auto_discover(void) {
    dev_count = 0;

    // OLED (0x3C, 0x3D)
    if(i2c_probe(0x3C)) add_dev(DEV_OLED, 0x3C);
    else if(i2c_probe(0x3D)) add_dev(DEV_OLED, 0x3D);

    // DS3231 (0x68)
    if(i2c_probe(0x68)) add_dev(DEV_DS3231, 0x68);

    // BH1750 (0x23, 0x5C)
    if(i2c_probe(0x23)) add_dev(DEV_BH1750, 0x23);
    else if(i2c_probe(0x5C)) add_dev(DEV_BH1750, 0x5C);

    // BMP280 (0x76, 0x77)
    if(i2c_probe(0x76)) add_dev(DEV_BMP280, 0x76);
    else if(i2c_probe(0x77)) add_dev(DEV_BMP280, 0x77);

    // EEPROM (0x50-0x57)
    for(uint8_t a=0x50; a<=0x57; a++)
        if(i2c_probe(a)) { add_dev(DEV_EEPROM, a); break; }
}
