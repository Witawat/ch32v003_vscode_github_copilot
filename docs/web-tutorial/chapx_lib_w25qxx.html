📚 หลักการทำงาน

W25Qxx เป็น SPI NOR Flash — ต้อง Erase Sector ก่อนเขียนเสมอ — 1 sector = 4KB, endurance 100K cycles

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Init + Verify JEDEC ID

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "W25Qxx.h"

W25Qxx_Instance flash;

int main(void) {
    SystemCoreClockUpdate();
    SPI_Init(SPI1, SPI_BAUD_4MHZ, SPI_MODE0);
    USART_Init(USART1, BAUD_115200);

    W25Qxx_Init(&flash, PC4);  // CS=PC4

    uint32_t jedec = W25Qxx_ReadJEDEC(&flash);
    USART_Print("JEDEC ID: 0x");
    USART_PrintHex24(jedec);
    USART_Println("");
    // W25Q32: 0xEF4016, W25Q64: 0xEF4017

🔍 W25Qxx_Init(&flash, CS) — instance-based, CS pin — SPI ต้อง init ก่อน

ขั้นที่ 2: Erase Sector → เขียนข้อมูล

    // Erase Sector 0 (4KB)
    W25Qxx_EraseSector(&flash, 0x000000);
    Delay_Ms(50);

    // Write data (max 256 bytes/page)
    uint8_t data[32];
    for(int i=0; i<32; i++) data[i] = i;
    W25Qxx_Write(&flash, 0x000000, data, 32);
    Delay_Ms(5);

🔍 W25Qxx_EraseSector(&flash, addr) — ลบ sector ก่อนเขียน, W25Qxx_Write(&flash, addr, data, len) — เขียนข้อมูล

ขั้นที่ 3: อ่านข้อมูล

    // Read back
    uint8_t rbuf[32];
    W25Qxx_Read(&flash, 0x000000, rbuf, 32);

    for(int i=0; i<32; i++) {
        USART_PrintInt(rbuf[i]); USART_Print(" ");
    }
    USART_Println("");
}

🔍 W25Qxx_Read(&flash, addr, buf, len) — อ่านข้อมูลได้ไม่มี page boundary limit

🛠️ สร้าง Product: Data Logger with SPI Flash

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "W25Qxx.h"

int main(void)
{
    SystemCoreClockUpdate();
    SPI_Init(SPI1, SPI_BAUD_4MHZ, SPI_MODE0);
    USART_Init(USART1, BAUD_115200);

    W25Qxx_Instance flash;
    W25Qxx_Init(&flash, PC4);

    W25Qxx_EraseSector(&flash, 0);
    while(W25Qxx_IsBusy(&flash));

    uint8_t data[256];
    for(int i=0; i<256; i++) data[i] = i;

    W25Qxx_Write(&flash, 0, data, 256);
    while(W25Qxx_IsBusy(&flash));

    uint8_t rbuf[256];
    W25Qxx_Read(&flash, 0, rbuf, 256);

    USART_Println("Verified!");
    while(1);
}
