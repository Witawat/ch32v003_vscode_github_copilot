KY-038 ใช้ Electret Microphone + LM393 Comparator — Analog output เป็นสัญญาณเสียงขยาย — Digital output เป็น comparator เทียบกับ threshold (ปรับด้วย Pot)

📚 หลักการทำงาน

KY-038 Sound Sensor ใช้ Electret Microphone จับเสียง → ขยายผ่าน Op-Amp → แบ่งเป็น 2 outputs — Analog (A0): ค่าแรงดันตาม amplitude ของเสียง — Digital (D0): LM393 comparator — HIGH เมื่อเสียงเกิน threshold

Analog Output วัด amplitude ของเสียง — ใช้ ADC อ่านเพื่อวัดระดับเสียงแบบ continuous

Clap Detection: วัด pattern ของเสียง — ตบมือ 1 ครั้ง = peak สั้น ~100ms — ตบ 2 ครั้ง = peak 2 ครั้งห่าง ~200-500ms — ตรวจจับด้วย state machine

🔍 SoundSensor_Init(&s, ADC_CH_PC4) — ตั้งค่า ADC ที่ขา PC4 — พร้อมใช้งานทันที

ขั้นที่ 2: อ่านระดับเสียง

    float level = SoundSensor_ReadLevel(&s);

    USART_Print("Sound Level: ");
    USART_PrintFloat(level, 1);

    if(level > 0.8f)       USART_Println(" VERY LOUD!");
    else if(level > 0.6f)  USART_Println(" Loud");
    else if(level > 0.4f)  USART_Println(" Normal");
    else                   USART_Println(" Quiet");

🔍 SoundSensor_ReadLevel(&s) — อ่านระดับเสียง normalized (0.0-1.0) — เงียบ ~0.3, ปกติ ~0.5, ดัง ~0.8+

ขั้นที่ 3: Clap Detection State Machine

    pinMode(PC0, PIN_MODE_OUTPUT); // LED ตอบสนอง
    uint8_t clap_count = 0;
    Timer_t tmr_clap;
    uint32_t last_clap = 0;

    while(1) {
        float level = SoundSensor_ReadLevel(&s);

        if(level > 0.7f && (Get_CurrentMs() - last_clap > 200)) {
            last_clap = Get_CurrentMs();
            clap_count++;
            Start_Timer(&tmr_clap, 600, 0);

            if(clap_count == 2) {
                USART_Println("DOUBLE CLAP!");
                digitalToggle(PC0); // toggle LED
                clap_count = 0;
            }
        }

        if(Is_Timer_Expired(&tmr_clap)) {
            if(clap_count == 1) {
                USART_Println("Single clap");
            }
            clap_count = 0;
        }
        Delay_Ms(10);
    }
}

🔍 ตรวจจับการตบมือ — debounce 200ms ระหว่าง clap — timeout 600ms รอ clap ครั้งที่ 2 — double clap = toggle LED

🛠️ สร้าง Product: Clap Switch — ตบมือเปิด/ปิดไฟ

KY-038 + Relay — Double clap = toggle ไฟ — Single clap = ignore

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "SoundSensor.h"

#define RELAY_PIN PC0
#define LED_PIN PC1

int main(void)
{
    SystemCoreClockUpdate();
    Timer_Init();
    USART_Init(115200);

    SoundSensor_Instance s;
    SoundSensor_Init(&s, ADC_CH_PC4);

    pinMode(RELAY_PIN, PIN_MODE_OUTPUT);
    pinMode(LED_PIN, PIN_MODE_OUTPUT);
    digitalWrite(RELAY_PIN, LOW);
    digitalWrite(LED_PIN, LOW);

    uint8_t clap_count = 0;
    uint8_t light_on = 0;
    uint32_t last_clap = 0;
    Timer_t tmr_timeout;

    while(1)
    {
        float level = SoundSensor_ReadLevel(&s);

        if(level > 0.7f && (Get_CurrentMs() - last_clap > 200)) {
            last_clap = Get_CurrentMs();
            clap_count++;
            Start_Timer(&tmr_timeout, 500, 0);

            if(clap_count >= 2) {
                light_on = !light_on;
                digitalWrite(RELAY_PIN, light_on);
                digitalWrite(LED_PIN, light_on);
                USART_Println(light_on ? "LIGHT ON" : "LIGHT OFF");
                clap_count = 0;
            }
        }

        if(clap_count > 0 && Is_Timer_Expired(&tmr_timeout)) {
            if(clap_count == 1) {
                USART_Println("Single clap (ignored)");
            }
            clap_count = 0;
        }
        Delay_Ms(10);
    }
}
