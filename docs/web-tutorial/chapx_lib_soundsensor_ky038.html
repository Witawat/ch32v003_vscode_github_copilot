KY-038 ใช้ Electret Microphone + LM393 Comparator — Analog output เป็นสัญญาณเสียงขยาย — Digital output เป็น comparator เทียบกับ threshold (ปรับด้วย Pot) — HIGH เมื่อเสียงดังเกิน threshold

📚 หลักการทำงาน

KY-038 Sound Sensor ใช้ Electret Microphone จับเสียง → ขยายผ่าน Op-Amp → แบ่งเป็น 2 outputs — Analog (A0): ค่าแรงดันตาม amplitude ของเสียง — Digital (D0): LM393 comparator — HIGH เมื่อเสียงเกิน threshold

Analog Output วัด amplitude ของเสียง — ปกติ ~1.5V (เงียบ) — ดังสูงสุด ~3.3V — ใช้ ADC อ่านเพื่อวัดระดับเสียงแบบ continuous — ใช้ FFT พื้นฐานแยกความถี่ได้

Clap Detection: วัด pattern ของเสียง — ตบมือ 1 ครั้ง = peak สั้น ~100ms — ตบ 2 ครั้ง = peak 2 ครั้งห่าง ~200-500ms — ตรวจจับด้วย state machine

Digital Output ใช้ตรวจจับเสียงดังด้วย GPIO interrupt — ไม่ต้อง polling ต่อเนื่อง — เหมาะกับงานที่ต้องการ low latency (ตบมือปิดไฟ)

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Init ADC + Digital

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "SoundSensor.h"

#define SOUND_ADC PD2
#define SOUND_DIG PC3

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    ADC_Init(SOUND_ADC, ADC_CH0);
    pinMode(SOUND_DIG, PIN_MODE_INPUT);
    USART_Init(115200);

    USART_Print("Sound Sensor Ready\r\n");

🔍 ใช้ทั้ง Analog (PD2) สำหรับวัดระดับเสียง — Digital (PC3) สำหรับตรวจจับเสียงดังแบบ interrupt ได้

ขั้นที่ 2: อ่านระดับเสียง + จับ Digital

    uint16_t adc = ADC_Read(SOUND_ADC);
    uint8_t dig = GPIO_Read(SOUND_DIG);

    if(dig) {
        USART_Print(">> LOUD! << ");
    }

    USART_Print("Sound ADC:"); USART_PrintDec(adc);

    if(adc > 900)  USART_Print(" VERY LOUD\r\n");
    else if(adc > 700) USART_Print(" Loud\r\n");
    else if(adc > 500) USART_Print(" Normal\r\n");
    else USART_Print(" Quiet\r\n");

🔍 อ่านทั้ง Analog แสดงระดับเสียง (Quiet/Normal/Loud/Very Loud) — Digital บอกว่าเกิน threshold หรือไม่ — Analog ใช้บอกความดังต่อเนื่องแม้ไม่ถึง threshold

ขั้นที่ 3: Clap Detection State Machine

    pinMode(PC0, PIN_MODE_OUTPUT); // LED ตอบสนอง
    uint8_t state = 0, clap_count = 0;
    Timer_t tmr_clap;
    Start_Timer(&tmr_clap, 600, 0); // timeout 600ms
    uint32_t last_clap = 0;

    while(1) {
        uint16_t adc = ADC_Read(SOUND_ADC);
        uint8_t dig = GPIO_Read(SOUND_DIG);

        if(dig && (Get_CurrentMs() - last_clap > 200)) {
            last_clap = Get_CurrentMs();
            clap_count++;
            Start_Timer(&tmr_clap, 600, 0);

            if(clap_count == 2) {
                USART_Print("DOUBLE CLAP!\r\n");
                digitalToggle(PC0); // toggle LED
                clap_count = 0;
            }
        }

        if(Is_Timer_Expired(&tmr_clap)) {
            if(clap_count == 1) {
                USART_Print("Single clap\r\n");
            }
            clap_count = 0;
        }
        Delay_Ms(10);
    }
}

🔍 ตรวจจับการตบมือ — debounce 200ms ระหว่าง clap — timeout 600ms รอ clap ครั้งที่ 2 — double clap = toggle LED — single clap = print — ใช้ Timer_t timeout แทน Delay เพื่อ non-blocking

🛠️ สร้าง Product: Clap Switch — ตบมือเปิด/ปิดไฟ

KY-038 + Relay — Double clap = toggle ไฟ — Single clap = ignore — แสดงสถานะด้วย LED

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "SoundSensor.h"

#define SOUND_ADC PD2
#define SOUND_DIG PC3
#define RELAY_PIN PC0
#define LED_PIN PC1

int main(void)
{
    SystemCoreClockUpdate();
    Timer_Init();
    ADC_Init(SOUND_ADC, ADC_CH0);
    pinMode(SOUND_DIG, PIN_MODE_INPUT);
    pinMode(RELAY_PIN, PIN_MODE_OUTPUT);
    pinMode(LED_PIN, PIN_MODE_OUTPUT);
    USART_Init(115200);

    digitalWrite(RELAY_PIN, LOW);
    digitalWrite(LED_PIN, LOW);

    uint8_t clap_count = 0;
    uint8_t light_on = 0;
    uint32_t last_clap = 0;
    Timer_t tmr_timeout;

    while(1)
    {
        uint16_t adc = ADC_Read(SOUND_ADC);
        uint8_t dig = GPIO_Read(SOUND_DIG);

        if(dig && adc > 800 && (Get_CurrentMs() - last_clap > 200)) {
            last_clap = Get_CurrentMs();
            clap_count++;
            Start_Timer(&tmr_timeout, 500, 0);

            if(clap_count >= 2) {
                light_on = !light_on;
                digitalWrite(RELAY_PIN, light_on);
                digitalWrite(LED_PIN, light_on);
                USART_Print(light_on ? "LIGHT ON\r\n" : "LIGHT OFF\r\n");
                clap_count = 0;
            }
        }

        if(clap_count > 0 && Is_Timer_Expired(&tmr_timeout)) {
            if(clap_count == 1) {
                USART_Print("Single clap (ignored)\r\n");
            }
            clap_count = 0;
        }
        Delay_Ms(10);
    }
}
