LCD1602 I2C — จอตัวอักษร 16×2

ระดับเริ่มต้น

🧭 แผนภาพการทำงาน

flowchart TD
    A["LCD_Init(&lcd, 0x27, LCD_16x2)"] --> B["LCD_Clear(&lcd)"]
    B --> C["LCD_SetCursor(&lcd, row, col)"]
    C --> D["LCD_Print(&lcd, str)"]
    D --> E["loop: อ่าน sensor → Print → Delay"]
    E --> C

🔌 การต่อวงจร

          LCD1602 I2C (HD44780 + PCF8574)
    ┌────────────────────────────────┐
    │ MCU                      LCD   │
    │ 5V  ─────────────── VCC        │
    │ GND ─────────────── GND        │
    │ PC1 ─────────────── SDA        │
    │ PC2 ─────────────── SCL        │
    │                 (backpack มี   │
    │                  pull-up ในตัว) │
    └────────────────────────────────┘

LCD1602 ใช้ PCF8574 I2C backpack — address 0x27 หรือ 0x3F ขึ้นอยู่กับรุ่น — ใช้ 5V จ่ายไฟ (backpack มี regulator ในตัว)

📚 หลักการทำงาน

Hitachi HD44780 Controller: เป็นจอตัวอักษร 16 คอลัมน์ × 2 แถว — ภายในใช้ controller HD44780 ที่มี RAM สำหรับตัวอักษร (DDRAM) 80 ตำแหน่ง และ CGRAM สำหรับตัวอักษรที่กำหนดเอง 8 ตัว (5×8 pixel ต่อตัว)

PCF8574 I2C Backpack: แปลงสัญญาณ I2C 2 เส้น → parallel 8-bit สำหรับ HD44780 — ทำให้ใช้สายแค่ 4 เส้น (VCC, GND, SDA, SCL) แทน 8-16 เส้น — address 0x27 (A0-A2 = GND) หรือ 0x3F

LCD1602_Handle: ใช้ struct instance — LCD_Init(&lcd, 0x27, LCD_16x2) — ทุกฟังก์ชันใช้ prefix LCD_ — รับ &lcd เป็นพารามิเตอร์แรก

Backlight Control: LCD_Backlight(&lcd, 1) เปิด, LCD_Backlight(&lcd, 0) ปิด — ควบคุมไฟหลังจอผ่าน PCF8574 ขา P3

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Init + Setup

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "lcd1602_i2c.h"

LCD1602_Handle lcd;

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    I2C_SimpleInit(I2C_SPEED_100K, I2C_PINS_DEFAULT);

    LCD_Init(&lcd, 0x27, LCD_16x2);
    LCD_Clear(&lcd);
    LCD_Backlight(&lcd, 1);

🔍 LCD_Init(&lcd, addr, type) — addr=0x27, type=LCD_16x2 (หรือ LCD_20x4) — LCD_Clear() ล้างหน้าจอ — LCD_Backlight(&lcd, 1) เปิดไฟหลังจอ

ขั้นที่ 2: พิมพ์ข้อความ

    LCD_SetCursor(&lcd, 0, 0);
    LCD_Print(&lcd, "Hello CH32V003!");

    LCD_SetCursor(&lcd, 1, 0);
    LCD_Print(&lcd, "I2C LCD 16x2");

🔍 LCD_SetCursor(&lcd, row, col) — row 0-1, col 0-15 (สำหรับ 16×2) — LCD_Print(&lcd, str) พิมพ์ข้อความ — ทุกฟังก์ชันใช้ prefix LCD_ ไม่ใช่ LCD1602_I2C_

ขั้นที่ 3: Loop แสดงค่า Sensor

    char buf[16];
    uint32_t temp = 0;

    while(1) {
        temp = ADC_Read(ADC_CH0);
        temp = (temp * 3300 / 4096) / 10;  // LM35 → °C

        LCD_SetCursor(&lcd, 1, 6);
        SimpleHAL_Sprintf(buf, "%d C  ", temp);
        LCD_Print(&lcd, buf);

        Delay_Ms(1000);
    }
}

🔍 ไม่ต้อง Clear() ก็ได้ — แค่เขียนทับที่เดิม — ต่อท้ายด้วยช่องว่างเพื่อลบตัวเลขเก่า — ใช้ sprintf สร้างข้อความพร้อมตัวเลข — หน่วง 1 วิ ก่อนอัปเดตใหม่

📋 API Reference

ฟังก์ชันคำอธิบายตัวอย่าง

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
จอดำหมด ไม่มีไฟ backlightJumper backlight หลุด หรือไม่ได้ต่อ 5Vตรวจสอบ jumper บน backpack — ต้องมี 5V จ่ายให้ LED backlight
ตัวหนังสือประหลาด / สี่เหลี่ยมดำContrast ไม่เหมาะสม — V0 potentiometerหมุน potentiometer สีน้ำเงินบน backpack จนเห็นตัวอักษรชัด
I2C ไม่ตอบ — จอดำ+backlight ติดใช้ address ผิด (0x27 vs 0x3F)ใช้ I2C_Scan() หา address — ลองทั้ง 0x27 และ 0x3F
แสดงผลแค่ 8 ตัวอักษรต่อแถวไม่ได้ init เป็น 16×2 modeLCD_Init(&lcd, 0x27, LCD_16x2) ต้องระบุ LCD_16x2
ตัวอักษรกระพริบหรือหายสาย I2C ยาวเกินหรือ pull-up อ่อนเพิ่ม pull-up 4.7kΩ ที่ SDA/SCL หรือลดความเร็ว I2C

🛠️ สร้าง Product: เครื่องวัดอุณหภูมิ LCD

DHT22 อ่านอุณหภูมิ+ความชื้น แสดงผลบน LCD1602 I2C — ใช้ Non-Blocking Timer อัปเดตทุก 2 วินาที

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "lcd1602_i2c.h"
#include "DHT.h"

LCD1602_Handle lcd;
DHT_Instance dht;

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    I2C_SimpleInit(I2C_SPEED_100K, I2C_PINS_DEFAULT);

    LCD_Init(&lcd, 0x27, LCD_16x2);
    LCD_Clear(&lcd);
    LCD_Backlight(&lcd, 1);

    DHT_Init(&dht, PC4, DHT_TYPE_DHT22);
    Delay_Ms(2000);

    Timer_t tmr;
    Start_Timer(&tmr, 2000, 1);
    char buf[16];

    while(1) {
        if(Is_Timer_Expired(&tmr)) {
            if(DHT_Read(&dht) == DHT_OK) {
                float t = DHT_GetTemperature(&dht);
                float h = DHT_GetHumidity(&dht);

                SimpleHAL_Sprintf(buf, "Temp: %.1f C", t);
                LCD_SetCursor(&lcd, 0, 0);
                LCD_Print(&lcd, buf);

                SimpleHAL_Sprintf(buf, "Hum : %.1f %%", h);
                LCD_SetCursor(&lcd, 1, 0);
                LCD_Print(&lcd, buf);
            } else {
                LCD_SetCursor(&lcd, 0, 0);
                LCD_Print(&lcd, "Sensor Error!   ");
            }
        }
    }
}

✏️ ลองทำดู

  1. ทำ Scrolling Text — ข้อความยาวกว่า 16 ตัวอักษรเลื่อนซ้ายอัตโนมัติ — ใช้การเลื่อน cursor ทุก 500ms
  2. สร้าง Custom Characters ตัวเลข 0-9 แบบดิจิทัล — ใช้ CGRAM ถ้ามีใน library
  3. ทำปุ่ม Toggle สลับภาษาไทย/อังกฤษ — ใช้ custom char แสดงภาษาไทย 2-3 ตัว
