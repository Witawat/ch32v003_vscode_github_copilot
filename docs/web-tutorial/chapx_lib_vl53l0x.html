🔌 การต่อวงจร

          VL53L0X — ToF Laser Ranging Sensor I2C
    ┌────────────────────────────────────┐
    │ MCU                       VL53L0X  │
    │ PC1─────────SDA        ┌──────────┤
    │ PC2─────────SCL        │ VCSEL    │
    │ 3.3V────────VCC        │ 940nm    │
    │ GND─────────GND        │ IR Laser │
    │ PC0─────────XSHUT(opt)└──────────┤
    │                                    │
    │ addr=0x29 (default, 1 device)      │
    │ Range: 30mm – 2000mm              │
    └────────────────────────────────────┘
MCU Pin→อุปกรณ์ขาหมายเหตุ
PC1↔SDA-สายข้อมูล I2C — ต้องมี R 4.7kΩ pull-up ไป 3.3V
PC2↔SCL-สายนาฬิกา I2C — ต้องมี R 4.7kΩ pull-up ไป 3.3V
3.3V→VCC-ไฟเลี้ยง 3.3V; I2C addr 0x29 (default)
GND→GND-กราวด์ร่วม
PC0→XSHUT-optional — ใช้เปลี่ยน I2C addr สำหรับ multi-sensor; HIGH=enable

PC1─SDA, PC2─SCL — I2C address 0x29 (default), XSHUT ใช้สลับ address ตอน init สำหรับ multi-sensor

📚 หลักการทำงาน

VL53L0X ใช้หลักการ Time-of-Flight (ToF) — ยิงเลเซอร์ VCSEL 940nm → วัดเวลาเดินทางไป-กลับของโฟตอน → คำนวณระยะทางด้วยความเร็วแสง — แม่นยำกว่า ultrasonic เพราะไม่ขึ้นกับวัสดุเป้าหมาย

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Init + Setup

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "VL53L0X.h"

VL53L0X_Instance tof;

int main(void) {
    SystemCoreClockUpdate();
    USART_Init(USART1, BAUD_115200);
    I2C_Init(I2C1, I2C_SPEED_FAST);

    if(!VL53L0X_Init(&tof, 0x29)) {
        USART_Println("VL53L0X init failed!");
        while(1);
    }
    USART_Println("VL53L0X Ready!");

🔍 VL53L0X_Init(&tof, addr) — คืน FALSE ถ้าไม่พบเซนเซอร์, addr 0x29 คือ default

ขั้นที่ 2: อ่านค่า

    uint16_t mm = VL53L0X_ReadRangeMM(&tof);

    if(mm > 0 && mm < 2000) {
        USART_Print("Range: ");
        USART_PrintInt(mm);
        USART_Println(" mm");
    } else {
        USART_Println("Out of range / error");
    }

🔍 VL53L0X_ReadRangeMM(&tof) — คืนค่าระยะในหน่วย mm (30–2000) — คืน 0 เมื่อวัดไม่ได้

ขั้นที่ 3: Loop + แสดงผล

    OLED_Handle oled;
    OLED_Init(&oled, OLED_128x64, 0x3C);

    char buf[64];
    while(1) {
        uint16_t mm = VL53L0X_ReadRangeMM(&tof);

        OLED_Clear(&oled);

        if(mm > 0) {
            snprintf(buf, sizeof(buf), "Dist: %.1f cm", mm / 10.0f);
            OLED_DrawStringAlign(&oled, 0, 0, buf, ALIGN_LEFT);

            if(mm < 100)       snprintf(buf, sizeof(buf), "Very Close");
            else if(mm < 500)  snprintf(buf, sizeof(buf), "Close");
            else if(mm < 1500) snprintf(buf, sizeof(buf), "Medium");
            else               snprintf(buf, sizeof(buf), "Far");
            OLED_DrawStringAlign(&oled, 0, 20, buf, ALIGN_LEFT);
        } else {
            snprintf(buf, sizeof(buf), "Signal Error");
            OLED_DrawStringAlign(&oled, 0, 0, buf, ALIGN_LEFT);
            snprintf(buf, sizeof(buf), "Out of range / Sun");
            OLED_DrawStringAlign(&oled, 0, 20, buf, ALIGN_LEFT);
        }

        OLED_Update(&oled);
        Delay_Ms(50);
    }
}

🔍 Continuous mode อัพเดท ~33Hz — หน่วง 50ms (~20Hz) — กรองค่า mm > 0 ก่อนใช้

📋 API Reference

ฟังก์ชันคำอธิบายตัวอย่าง
VL53L0X_Init(&tof, addr)เริ่มต้นเซนเซอร์VL53L0X_Init(&tof, 0x29)

VL53L0X_IsReady(&tof)เช็คว่า data พร้อมหรือยังif(VL53L0X_IsReady(&tof))

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
Init ล้มเหลว (ไม่เจอ device)สาย SDA/SCL ผิด หรือ XSHUT ไม่ได้ต่อ HIGHเช็ค I2C scan, ต่อ XSHUT เข้า VCC (3.3V)
อ่านได้ 0 ตลอดวัตถุมืดเกินไป (ดำ) หรืออยู่กลางแดดใช้วัตถุสีอ่อน, หลีกเลี่ยงแสงแดดตรง

🛠️ สร้าง Product: ประตูอัตโนมัติ

VL53L0X วัดระยะ — เมื่อระยะ < 50cm → Servo เปิดประตู

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "VL53L0X.h"

int main(void)
{
    SystemCoreClockUpdate();
    USART_Init(USART1, BAUD_115200);
    I2C_Init(I2C1, I2C_SPEED_FAST);

    VL53L0X_Instance tof;
    VL53L0X_Init(&tof, 0x29);

    Servo_Init(PC0);
    Servo_SetAngle(0);

    Timer_t tmr_close;
    uint8_t doorOpen = 0;

    while(1)
    {
        uint16_t mm = VL53L0X_ReadRangeMM(&tof);

        if(mm > 0 && mm < 500 && !doorOpen) {
            Servo_SetAngle(90);
            doorOpen = 1;
            Start_Timer(&tmr_close, 3000, 0);
        }

        if(doorOpen && Is_Timer_Expired(&tmr_close)) {
            Servo_SetAngle(0);
            doorOpen = 0;
        }

        Delay_Ms(100);
    }
}

✏️ ลองทำดู

  1. ใช้ VL53L0X 2 ตัว — ใช้ XSHUT เปลี่ยน addr เป็น 0x30, 0x31 — วัดระยะ 2 จุดพร้อมกัน
  2. ทำ liquid level sensor — ติด VL53L0X เหนือผิวของเหลว วัดระดับความสูงของเหลว
  3. ทำ gesture sensor — วัดการเปลี่ยนแปลงระยะเร็วๆ ตรวจจับ swipe
