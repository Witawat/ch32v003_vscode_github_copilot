PC1─SDA, PC2─SCL — I2C address 0x29 (default, เปลี่ยนได้เมื่อมีหลายตัวผ่าน XSHUT), XSHUT ใช้สลับ address ตอน init

📚 หลักการทำงาน

VL53L0X ใช้หลักการ Time-of-Flight (ToF) — ยิงเลเซอร์ VCSEL 940nm → วัดเวลาเดินทางไป-กลับของโฟตอน → คำนวณระยะทางด้วยความเร็วแสง — แม่นยำกว่า ultrasonic เพราะไม่ขึ้นกับวัสดุเป้าหมาย

Continuous vs Single Mode:

  • Continuous: วัดต่อเนื่องอัตโนมัติ ~30ms/ครั้ง — เหมาะกับงาน real-time
  • Single: วัดทีละครั้ง → เรียก VL53L0X_StartSingle() แล้วรอ — ประหยัดไฟกว่า

Range Status: ค่า range มาพร้อม status code — 0=OK, 1=early, 2=sigma fail, 4=signal fail, 7=wrap — ใช้กรองค่าที่ผิดพลาดออก

ข้อจำกัดแสงแดด: แสงแดดจ้ามี IR จำนวนมาก → รบกวน VCSEL 940nm → ระยะลดลง หรือวัดไม่ได้เลย — แก้โดยใช้ cover ปิดกันแสง หรือใช้ VL53L1X (IR filter ดีกว่า)

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Init + Setup

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "VL53L0X.h"

int main(void) {
    SystemCoreClockUpdate();
    USART_Init(USART1, BAUD_115200);
    I2C_Init(I2C1, I2C_SPEED_FAST);  // 400kHz recommended

    VL53L0X_Data_t range;
    if(!VL53L0X_Init(0x29)) {  // default address
        USART_Println("VL53L0X init failed!");
        while(1);
    }
    VL53L0X_SetMode(VL53L0X_MODE_CONTINUOUS);

🔍 VL53L0X_Init(addr) — คืน FALSE ถ้าไม่พบเซนเซอร์ (เช็ค I2C), addr 0x29 คือ default — XSHUT ใช้สำหรับ multi-sensor: ปิดทุกตัว → เปิดทีละตัว → เปลี่ยน addr

ขั้นที่ 2: อ่าน/เขียนค่า

    // Continuous mode: อ่านได้เลย
    VL53L0X_ReadRangeMM(&range);

    USART_Print("Range: ");
    USART_PrintInt(range.range_mm);
    USART_Print(" mm | Status: ");
    USART_PrintInt(range.status);
    USART_Println("");

    if(range.status == 0) {
        USART_Println("✓ Valid measurement");
    } else {
        USART_Print("⚠️ Error code: ");
        USART_PrintInt(range.status);
        USART_Println("");
    }

🔍 VL53L0X_ReadRangeMM() — คืน range.range_mm (30–2000) และ range.status — status 0 = OK, 4 = weak signal, 7 = wrap-around — ใช้ status กรองค่าที่ผิดพลาด

ขั้นที่ 3: Loop + แสดงผล

    OLED_Handle oled;
    OLED_Init(&oled, OLED_128x64, 0x3C);

    while(1) {
        VL53L0X_ReadRangeMM(&range);

        OLED_Clear(&oled);
        OLED_SetCursor(&oled, 0, 0);

        if(range.status == 0) {
            OLED_PrintFloat(&oled, "Dist:", (float)range.range_mm / 10.0f, 1);
            OLED_Print(&oled, " cm");

            // ระยะใกล้สุด/ไกลสุดที่วัดได้
            OLED_SetCursor(&oled, 0, 2);
            if(range.range_mm < 100)        OLED_Print(&oled, "Very Close");
            else if(range.range_mm < 500)   OLED_Print(&oled, "Close");
            else if(range.range_mm < 1500)   OLED_Print(&oled, "Medium");
            else                            OLED_Print(&oled, "Far");
        } else {
            OLED_Print(&oled, "Signal Error");
            OLED_SetCursor(&oled, 0, 2);
            OLED_Print(&oled, "Out of range / Sun");
        }

        OLED_Update(&oled);
        Delay_Ms(50);
    }
}

🔍 Continuous mode อัพเดท ~33Hz — หน่วง 50ms (~20Hz) — กรอง status 0 ก่อนใช้ — ถ้าสถานะไม่ใช่ 0 แสดง error — อาจเกิดจากแสงแดด, ระยะเกิน, หรือวัตถุดำเกินไป

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
Init ล้มเหลว (ไม่เจอ device)สาย SDA/SCL ผิด หรือ XSHUT ไม่ได้ต่อ HIGHเช็ค I2C scan, ต่อ XSHUT เข้า VCC (3.3V) ถ้าไม่ใช้ multi-sensor
Status=4 (signal fail) บ่อยวัตถุมืดเกินไป (ดำ) หรืออยู่กลางแดดใช้วัตถุสีอ่อน, หลีกเลี่ยงแสงแดดตรง, เพิ่ม integration time
Status=7 (wrap-around)วัตถุอยู่นอกระยะ 30-2000mmตรวจสอบวัตถุอยู่ในระยะ, ใช้ ultrasonic เสริมระยะไกล
วัดได้ 0mm ทั้งที่อยู่ในระยะCover glass สกปรกหรือมีรอยทำความสะอาด cover glass, ห้ามใช้ cover ที่ไม่ใช่ IR-pass
ค่าแกว่ง ±5mm ตลอดMulti-path reflection (วัตถุมีหลายผิว)เปลี่ยนมุมเซนเซอร์, ใช้ single mode เฉลี่ยหลายค่า

🛠️ สร้าง Product: ประตูอัตโนมัติ — เปิดเมื่อมีคนเข้าใกล้

VL53L0X วัดระยะ — เมื่อระยะ < 50cm → Servo เปิดประตู

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "VL53L0X.h"
#include "Servo.h"

int main(void)
{
    SystemCoreClockUpdate();
    USART_Init(USART1, BAUD_115200);
    I2C_Init(I2C1, I2C_SPEED_FAST);

    VL53L0X_Data_t range;
    VL53L0X_Init(0x29);
    VL53L0X_SetMode(VL53L0X_MODE_CONTINUOUS);

    Servo_Init(PC0);  // Servo for door
    Servo_SetAngle(0);  // door closed

    Timer_t tmr_close;
    uint8_t doorOpen = 0;

    while(1)
    {
        VL53L0X_ReadRangeMM(&range);

        if(range.status == 0 && range.range_mm < 500 && !doorOpen) {
            Servo_SetAngle(90);   // เปิดประตู
            doorOpen = 1;
            Start_Timer(&tmr_close, 3000, 0);  // เปิดค้าง 3 วิ
        }

        if(doorOpen && Is_Timer_Expired(&tmr_close)) {
            Servo_SetAngle(0);    // ปิดประตู
            doorOpen = 0;
        }

        Delay_Ms(100);
    }
}

✏️ ลองทำดู

  1. ใช้ VL53L0X 2 ตัว (multi-sensor) — ใช้ XSHUT เปลี่ยน addr เป็น 0x30, 0x31 — วัดระยะ 2 จุดพร้อมกัน
  2. ทำ liquid level sensor — ติด VL53L0X เหนือผิวของเหลว วัดระดับความสูงของเหลว
  3. ทำ gesture sensor — วัดการเปลี่ยนแปลงระยะเร็วๆ ตรวจจับ swipe left/right/up/down
