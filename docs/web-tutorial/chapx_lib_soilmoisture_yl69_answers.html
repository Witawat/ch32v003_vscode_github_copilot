📝 เฉลย — YL-69 Soil Moisture

ข้อ 1: Smart Plant Watering — วัดความชื้น → เปิดปั๊มอัตโนมัติ

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "SoilMoisture.h"
#include "oled_i2c.h"

SoilMoisture_Instance s;
#define SOIL_VCC PC0
#define SOIL_ADC PD2
#define SOIL_CH  ADC_CH0
#define PUMP_PIN PC3
#define DRY_THRESHOLD 30.0f
#define WATER_DURATION_MS 5000
#define CHECK_INTERVAL_MS 30000

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    I2C_Init(PC1, PC2, I2C_SPEED_STANDARD);
    USART_Init(115200);

    SoilMoisture_Init(&s, SOIL_CH);
    GPIO_SimpleInit(SOIL_VCC, GPIO_OUT_PP);
    GPIO_SimpleInit(PUMP_PIN, GPIO_OUT_PP);
    GPIO_WriteBit(SOIL_VCC, 0);
    GPIO_WriteBit(PUMP_PIN, 0);

    OLED_Handle oled;
    OLED_Init(&oled, OLED_128x64, 0x3C);
    OLED_SetFont(&oled, FONT_8x16);

    uint32_t pump_run_count = 0;
    Timer_t tmr;
    Start_Timer(&tmr, CHECK_INTERVAL_MS, 1);

    USART_Print("Smart Plant Watering Ready\r\n");

    while(1) {
        if(Is_Timer_Expired(&tmr)) {
            GPIO_WriteBit(SOIL_VCC, 1);
            Delay_Ms(80);
            float moisture = SoilMoisture_Read(&s);
            GPIO_WriteBit(SOIL_VCC, 0);

            OLED_Clear(&oled);
            OLED_SetCursor(&oled, 0, 0);
            OLED_Print(&oled, "Smart Watering");

            OLED_SetCursor(&oled, 0, 2);
            OLED_PrintFloat(&oled, "Soil:", moisture, 1);
            OLED_Print(&oled, "%");

            OLED_SetCursor(&oled, 0, 4);
            OLED_Print(&oled, "Pumps:");
            OLED_PrintDec(&oled, pump_run_count);

            if(moisture < DRY_THRESHOLD) {
                OLED_SetCursor(&oled, 0, 6);
                OLED_Print(&oled, "WATERING...");
                OLED_Update(&oled);

                GPIO_WriteBit(PUMP_PIN, 1);
                USART_Print("WATERING! ");
                USART_PrintFloat(moisture, 1);
                USART_Print("%\r\n");
                Delay_Ms(WATER_DURATION_MS);
                GPIO_WriteBit(PUMP_PIN, 0);
                pump_run_count++;
            } else {
                OLED_SetCursor(&oled, 0, 6);
                OLED_Print(&oled, "Soil OK   ");
            }

            OLED_Update(&oled);

            USART_Print("Moisture:"); USART_PrintFloat(moisture, 1);
            USART_Print("%\r\n");
        }
    }
}
🧠 แนวคิด: เปิด VCC → รอ 80ms stabilize → อ่าน ADC → ปิด VCC ทันที — วัดทุก 30 วิ ลดการเปิดปิด VCC ถี่ — ถ้าความชื้น < 30% → เปิดปั๊ม 5 วิ — Map ADC เป็น % ด้วย formula ที่ clamp ค่า min/max — แสดงสถานะ OLED — นับจำนวนครั้งที่รดน้ำ — ป้องกัน probe กัดกร่อนด้วย GPIO VCC control

ข้อ 2: Calibration — บันทึกค่าในน้ำและในอากาศ

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "SoilMoisture.h"
#include "AT24Cxx.h"

SoilMoisture_Instance s;
AT24Cxx_Instance eeprom;
#define SOIL_VCC PC0
#define SOIL_ADC PD2
#define SOIL_CH  ADC_CH0
#define CAL_MAGIC 0xA5A5

typedef struct {
    uint16_t magic;
    uint16_t adc_dry;
    uint16_t adc_wet;
} CalData_t;

CalData_t cal;

void save_calibration(void) {
    AT24Cxx_Write(&eeprom, 0, (uint8_t*)&cal, sizeof(CalData_t));
    USART_Print("Calibration saved to EEPROM\r\n");
}

void load_calibration(void) {
    AT24Cxx_Read(&eeprom, 0, (uint8_t*)&cal, sizeof(CalData_t));
    if(cal.magic != CAL_MAGIC) {
        cal.magic = CAL_MAGIC;
        cal.adc_dry = 200;
        cal.adc_wet = 950;
        USART_Print("No calibration found, using defaults\r\n");
    }
}

uint16_t read_soil_raw(void) {
    GPIO_WriteBit(SOIL_VCC, 1);
    Delay_Ms(50);
    uint16_t adc = ADC_Read(SOIL_ADC);
    GPIO_WriteBit(SOIL_VCC, 0);
    return adc;
}

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    I2C_Init(PC1, PC2, I2C_SPEED_STANDARD);
    ADC_Init(SOIL_ADC, SOIL_CH);
    USART_Init(115200);

    SoilMoisture_Init(&s, SOIL_CH);
    GPIO_SimpleInit(SOIL_VCC, GPIO_OUT_PP);
    GPIO_WriteBit(SOIL_VCC, 0);
    GPIO_SimpleInit(PC3, GPIO_IN_PU);
    GPIO_SimpleInit(PC4, GPIO_IN_PU);

    AT24Cxx_Init(&eeprom, AT24C32, 0x50);
    load_calibration();

    USART_Print("Soil Calibration Menu\r\n");
    USART_Print("Press PC3: Calibrate DRY (in air)\r\n");
    USART_Print("Press PC4: Calibrate WET (in water)\r\n");
    USART_Print("Current: dry="); USART_PrintDec(cal.adc_dry);
    USART_Print(" wet="); USART_PrintDec(cal.adc_wet);
    USART_Print("\r\n");

    while(1) {
        if(GPIO_ReadBit(PC3) == 0) {
            Delay_Ms(100);
            USART_Print("Calibrating DRY...\r\n");
            uint32_t sum = 0;
            for(int i = 0; i < 30; i++) {
                sum += read_soil_raw();
                Delay_Ms(50);
            }
            cal.adc_dry = sum / 30;
            save_calibration();
            USART_Print("Dry ADC: "); USART_PrintDec(cal.adc_dry);
            USART_Print("\r\n");
            while(GPIO_ReadBit(PC3) == 0);
        }

        if(GPIO_ReadBit(PC4) == 0) {
            Delay_Ms(100);
            USART_Print("Calibrating WET...\r\n");
            uint32_t sum = 0;
            for(int i = 0; i < 30; i++) {
                sum += read_soil_raw();
                Delay_Ms(50);
            }
            cal.adc_wet = sum / 30;
            save_calibration();
            USART_Print("Wet ADC: "); USART_PrintDec(cal.adc_wet);
            USART_Print("\r\n");
            while(GPIO_ReadBit(PC4) == 0);
        }

        float moisture = SoilMoisture_Read(&s);

        USART_Print("Moisture:"); USART_PrintFloat(moisture, 1);
        USART_Print("%\r\n");
        Delay_Ms(2000);
    }
}
🧠 แนวคิด: ใช้ EEPROM (AT24Cxx) เก็บค่า calibration — magic number 0xA5A5 ตรวจสอบว่ามีข้อมูล calibration หรือยัง — ปุ่ม PC3 = calibrate แห้ง (จุ่ม probe ในอากาศ), PC4 = calibrate เปียก (จุ่มในน้ำ) — average 30 samples เพื่อความแม่นยำ — save_calibration() บันทึกลง EEPROM — load_calibration() อ่านค่ากลับมาตอนบูท — ค่า default: dry=200, wet=950

ข้อ 3: Transistor Switch — ใช้ 2N2222 ขับ VCC แทน GPIO ตรง

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "SoilMoisture.h"

SoilMoisture_Instance s;
#define SOIL_VCC_EN PC0
#define SOIL_ADC    PD2
#define SOIL_CH     ADC_CH0

/*
  ______ 3.3V
  |
  R(10k)
  |
  +-------- VCC of Sensor
  |
  C (2N2222)
  |
  E --- GND
  |
  B ---[1k]--- PC0 (SOIL_VCC_EN)
*/

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_Init(115200);

    SoilMoisture_Init(&s, SOIL_CH);
    GPIO_SimpleInit(SOIL_VCC_EN, GPIO_OUT_PP);
    GPIO_WriteBit(SOIL_VCC_EN, 0);

    Timer_t tmr;
    Start_Timer(&tmr, 10000, 1);

    USART_Print("Soil Sensor (Transistor VCC)\r\n");
    USART_Print("Transistor ON → 50ms → Read → OFF\r\n");

    while(1) {
        if(Is_Timer_Expired(&tmr)) {
            GPIO_WriteBit(SOIL_VCC_EN, 1);
            Delay_Ms(50);

            float moisture = SoilMoisture_Read(&s);

            GPIO_WriteBit(SOIL_VCC_EN, 0);

            USART_Print(" → "); USART_PrintFloat(moisture, 1);
            USART_Print("%");

            if(moisture > 0) {
                USART_Print(" OK\r\n");
            } else {
                USART_Print(" (check transistor wiring)\r\n");
            }
        }
    }
}
🧠 แนวคิด: GPIO ของ CH32V003 จ่ายกระแสได้จำกัด (~8mA) — ถ้าเซนเซอร์กินกระแสมาก ใช้ NPN Transistor (2N2222) เป็นสวิตช์ — GPIO (PC0) ต่อผ่าน R 1kΩ ไป Base ของ transistor — เมื่อ PC0=HIGH → transistor ON → 3.3V จ่ายเข้าเซนเซอร์ — กระแสไหลผ่าน collector-emitter ไม่ผ่าน GPIO — วงจร: Base มี R 1k จำกัดกระแส, Collector ต่อ VCC ผ่าน R pull-up 10k ไปเซนเซอร์, Emitter ลง GND — ใช้กับเซนเซอร์หลายตัวที่ใช้ VCC ร่วมกันได้
