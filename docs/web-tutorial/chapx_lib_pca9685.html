📚 หลักการทำงาน

PCA9685 เป็น I2C 16-channel PWM controller — แต่ละ channel ความละเอียด 12-bit (4096 steps), ตั้ง servo angle ได้โดยตรงผ่าน PCA9685_SetServoAngle()

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Init + Setup

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "PCA9685.h"

PCA9685_Instance pca;

int main(void) {
    SystemCoreClockUpdate();
    I2C_Init(I2C1, I2C_SPEED_STANDARD);
    PCA9685_Init(&pca, 0x40, 50);  // addr=0x40, 50Hz
    USART_Init(USART1, BAUD_115200);
    USART_Println("PCA9685 Ready @ 50Hz!");

🔍 PCA9685_Init(&pca, addr, freq) — instance-based, 0x40=default, freq=50Hz มาตรฐาน servo RC

ขั้นที่ 2: ควบคุม Servo — SetServoAngle

    PCA9685_SetServoAngle(&pca, 0, 0);    // ch0 = 0°
    Delay_Ms(1000);
    PCA9685_SetServoAngle(&pca, 0, 90);   // ch0 = 90°
    Delay_Ms(1000);
    PCA9685_SetServoAngle(&pca, 0, 180);  // ch0 = 180°
    Delay_Ms(1000);

🔍 PCA9685_SetServoAngle(&pca, ch, angle) — ตั้งมุม servo โดยตรง (0-180) — library คำนวณ pulse width ให้อัตโนมัติ

ขั้นที่ 3: Sweep Servo + Multiple Channels

    // Sweep servo 0 ไปมา 3 รอบ
    for(int round=0; round<3; round++) {
        for(uint8_t a=0; a<=180; a+=5) {
            PCA9685_SetServoAngle(&pca, 0, a);
            Delay_Ms(50);
        }
        for(uint8_t a=180; a>0; a-=5) {
            PCA9685_SetServoAngle(&pca, 0, a);
            Delay_Ms(50);
        }
    }

    // ควบคุมหลาย channel พร้อมกัน
    while(1) {
        PCA9685_SetServoAngle(&pca, 0, 0);
        PCA9685_SetServoAngle(&pca, 1, 90);
        PCA9685_SetServoAngle(&pca, 2, 180);
        Delay_Ms(2000);
        PCA9685_SetServoAngle(&pca, 0, 180);
        PCA9685_SetServoAngle(&pca, 1, 0);
        PCA9685_SetServoAngle(&pca, 2, 90);
        Delay_Ms(2000);
    }
}

🔍 SetServoAngle แปลงมุม 0-180 เป็น pulse width อัตโนมัติ — ไม่ต้องคำนวณ ON/OFF เอง

🛠️ สร้าง Product: 4-Axis Robotic Arm Controller

PCA9685 ควบคุม servo 4 ตัว — sweep อัตโนมัติ

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "PCA9685.h"

int main(void)
{
    SystemCoreClockUpdate();
    I2C_Init(I2C1, I2C_SPEED_STANDARD);

    PCA9685_Instance pca;
    PCA9685_Init(&pca, 0x40, 50);
    USART_Init(USART1, BAUD_115200);

    // Base=ch0, Shoulder=ch1, Elbow=ch2, Gripper=ch3
    while(1)
    {
        for(uint8_t a=0; a<=180; a+=2) {
            PCA9685_SetServoAngle(&pca, 0, a);
            Delay_Ms(20);
        }
        for(uint8_t a=90; a<=150; a+=2) {
            PCA9685_SetServoAngle(&pca, 1, a);
            Delay_Ms(20);
        }
        for(uint8_t a=90; a>=30; a-=2) {
            PCA9685_SetServoAngle(&pca, 3, a);
            Delay_Ms(20);
        }
        Delay_Ms(500);
        // Return to home
        PCA9685_SetServoAngle(&pca, 0, 90);
        PCA9685_SetServoAngle(&pca, 1, 90);
        PCA9685_SetServoAngle(&pca, 2, 90);
        PCA9685_SetServoAngle(&pca, 3, 90);
        Delay_Ms(1000);
    }
}

✏️ ลองทำดู

  1. ทำ servo sequencer — ตั้งลำดับ servo หลายตัวเคลื่อนไหวเป็น sequence
  2. ทำระบบรดน้ำต้นไม้อัตโนมัติ — PCA9685 ควบคุม solenoid valve ผ่าน MOSFET
  3. ควบคุม LED RGB 16 แถบ — แต่ละ channel ต่อ LED strip ผ่าน MOSFET
