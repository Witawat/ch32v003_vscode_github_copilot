PCA9685 มี 2 แหล่งจ่ายไฟแยกอิสระ — VCC (logic) 3.3V จาก CH32V003, V+ (motor/servo power) 5-6V external — ต้องต่อ GND ร่วมกัน, แต่ละ channel รับโหลดได้ 25mA, จ่าย servo 16 ตัวพร้อมกันต้องใช้ external 5V 2A+

📚 หลักการทำงาน

PCA9685 เป็น I2C 16-channel PWM controller — แต่ละ channel ความละเอียด 12-bit (4096 steps), ความถี่ PWM ปรับได้ 24Hz-1526Hz, ใช้ internal oscillator 25MHz + prescaler

การตั้งค่า PWM: ใช้ค่า ON และ OFF — ON = เวลาเริ่ม pulse, OFF = เวลาหยุด pulse (ทั้งคู่ 0-4095) — ความกว้าง pulse = (OFF - ON) / 4096 * period

ตัวอย่าง 50Hz (servo):

  • Period = 20ms (50Hz), Prescaler ~121
  • 0° = pulse 0.5ms → ON=0, OFF=102 (0.5/20*4096)
  • 90° = pulse 1.5ms → ON=0, OFF=307
  • 180° = pulse 2.5ms → ON=0, OFF=512

OE Pin: Output Enable — LOW = เปิด output, HIGH = ปิดทุก channel (high impedance) — ใช้ hardware shutdown

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Init + Setup

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "PCA9685.h"

int main(void) {
    SystemCoreClockUpdate();
    I2C_Init(I2C1, I2C_SPEED_STANDARD);  // 100kHz
    PCA9685_Init(I2C1, 0x40, 50);        // addr=0x40, 50Hz
    USART_Init(USART1, BAUD_115200);
    USART_Println("PCA9685 Ready @ 50Hz!");

🔍 PCA9685_Init(i2c, addr, freq) — 0x40 = default address (A0-A5=GND ทั้งหมด), freq=50Hz มาตรฐานสำหรับ servo RC, ต้องต่อ V+ 5V external สำหรับ servo

ขั้นที่ 2: ควบคุม Servo — แปลงมุมเป็น pulse

    // Servo 3 ตัวบน channel 0, 1, 2
    // SG90: 0.5-2.5ms pulse → 2.5% - 12.5% duty @ 50Hz
    uint16_t servo_min = 102;   // 0.5ms → 0 degree
    uint16_t servo_max = 512;   // 2.5ms → 180 degree

    // ตั้ง servo0 ที่ 0 degrees
    PCA9685_SetPWM(0, 0, servo_min);
    Delay_Ms(1000);

    // ตั้ง servo0 ที่ 90 degrees (center)
    uint16_t center = 307;  // 1.5ms
    PCA9685_SetPWM(0, 0, center);
    Delay_Ms(1000);

    // ตั้ง servo0 ที่ 180 degrees
    PCA9685_SetPWM(0, 0, servo_max);
    Delay_Ms(1000);

🔍 PCA9685_SetPWM(ch, on, off) — ch=0-15, on/off=0-4095, on มักเป็น 0 (เริ่มที่ต้นเฟรม), off กำหนดความกว้าง pulse, 50Hz: off=102=0.5ms, off=307=1.5ms, off=512=2.5ms

ขั้นที่ 3: Sweep Servo + Multiple Channels

    // ฟังก์ชันแปลงมุม → off value (SG90 @ 50Hz)
    uint16_t angle_to_off(uint8_t angle) {
        if(angle > 180) angle = 180;
        return 102 + (uint16_t)((uint32_t)angle * 410 / 180);
    }

    // Sweep servo 0 ไปมา 3 รอบ
    for(int round=0; round<3; round++) {
        for(uint8_t angle=0; angle<=180; angle+=5) {
            PCA9685_SetPWM(0, 0, angle_to_off(angle));
            Delay_Ms(50);  // ให้ servo ทันหมุน
        }
        for(uint8_t angle=180; angle>0; angle-=5) {
            PCA9685_SetPWM(0, 0, angle_to_off(angle));
            Delay_Ms(50);
        }
    }

    // ควบคุมหลาย channel พร้อมกัน
    while(1) {
        PCA9685_SetPWM(0, 0, angle_to_off(0));    // ch0 = 0°
        PCA9685_SetPWM(1, 0, angle_to_off(90));   // ch1 = 90°
        PCA9685_SetPWM(2, 0, angle_to_off(180));  // ch2 = 180°
        Delay_Ms(2000);
        PCA9685_SetPWM(0, 0, angle_to_off(180));  // ch0 = 180°
        PCA9685_SetPWM(1, 0, angle_to_off(0));    // ch1 = 0°
        PCA9685_SetPWM(2, 0, angle_to_off(90));   // ch2 = 90°
        Delay_Ms(2000);
    }
}

🔍 angle_to_off(angle) — แปลงมุม 0-180 เป็น off value (102-512), สูตร: off = min + angle × (max-min) / 180, อย่าลืม delay ให้ servo มีเวลาหมุน (50-100ms)

🛠️ สร้าง Product: 4-Axis Robotic Arm Controller

PCA9685 ควบคุม servo 4 ตัว (base, shoulder, elbow, gripper) — sweep อัตโนมัติ

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "PCA9685.h"

uint16_t angle_to_off(uint8_t a) {
    if(a>180)a=180; return 102+(uint16_t)((uint32_t)a*410/180);
}

int main(void)
{
    SystemCoreClockUpdate();
    I2C_Init(I2C1, I2C_SPEED_STANDARD);
    PCA9685_Init(I2C1, 0x40, 50);
    USART_Init(USART1, BAUD_115200);

    // Base=ch0, Shoulder=ch1, Elbow=ch2, Gripper=ch3
    uint8_t base=90, shoulder=90, elbow=90, gripper=90;

    while(1)
    {
        // Move base: 0→180→0
        for(uint8_t a=0; a<=180; a+=2) {
            PCA9685_SetPWM(0,0,angle_to_off(a));
            Delay_Ms(20);
        }
        // Move shoulder up
        for(uint8_t a=90; a<=150; a+=2) {
            PCA9685_SetPWM(1,0,angle_to_off(a));
            Delay_Ms(20);
        }
        // Close gripper
        for(uint8_t a=90; a>=30; a-=2) {
            PCA9685_SetPWM(3,0,angle_to_off(a));
            Delay_Ms(20);
        }
        Delay_Ms(500);
        // Return to home
        PCA9685_SetPWM(0,0,angle_to_off(90));
        PCA9685_SetPWM(1,0,angle_to_off(90));
        PCA9685_SetPWM(2,0,angle_to_off(90));
        PCA9685_SetPWM(3,0,angle_to_off(90));
        Delay_Ms(1000);
    }
}

✏️ ลองทำดู

  1. ทำ servo sequencer — ตั้งลำดับ servo หลายตัวเคลื่อนไหวเป็น sequence (dance) แบบ time-coded
  2. ทำระบบรดน้ำต้นไม้อัตโนมัติ — PCA9685 ควบคุม solenoid valve ผ่าน MOSFET หลายตัว
  3. ควบคุม LED RGB 16 แถบ — แต่ละ channel ต่อ LED strip ผ่าน MOSFET — สร้าง effect วิ่งแบบต่างๆ
