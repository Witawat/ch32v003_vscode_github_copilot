KY-026 ใช้ IR Photodiode ตรวจจับเปลวไฟในช่วงความยาวคลื่น 760-1100nm — มี LM393 Comparator สำหรับ Digital Output — Pot ปรับ sensitivity — มุมตรวจจับ ~60° ระยะไกลสุด ~1m สำหรับไฟแช็ก

📚 หลักการทำงาน

KY-026 Flame Sensor ใช้ Infrared Photodiode ตรวจจับรังสี IR จากเปลวไฟ — เปลวไฟปล่อย IR ในช่วง 760-1100nm — เซนเซอร์กรองแสง visible ออก — เมื่อมีเปลวไฟ ADC จะสูงขึ้น (แสง IR มากขึ้น)

2 Outputs: Analog (A0) — ใช้ ADC อ่านความเข้ม IR (0-3.3V) — Digital (D0) — LM393 Comparator เทียบกับ threshold ที่ปรับด้วย Pot — D0 = HIGH เมื่อ IR เกิน threshold — ใช้ขัดจังหวะ MCU ได้

Digital vs Analog: Digital ใช้ตรวจจับเร็ว (GPIO interrupt) — Analog ใช้บอกระยะทาง/ความแรงของไฟ — การใช้ทั้งคู่ช่วยลด False Positive (แสงแดด vs เปลวไฟจริง)

ข้อควรระวัง: แสงแดดมี IR สูง อาจ trigger false positive — หลีกเลี่ยงการติดตั้งที่โดนแดดตรง — ใช้ Analog threshold หลายระดับเพื่อยืนยัน

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Init FlameSensor

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "FlameSensor.h"

FlameSensor_Instance f;

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_Init(115200);

    FlameSensor_Init(&f, ADC_CH_PC4, PC1);
    // ADC_CH_PC4 = analog, PC1 = digital
    USART_Println("Flame Sensor Ready!");

🔍 FlameSensor_Init(&f, adc_ch, dig_pin) — ตั้งค่า ADC และ GPIO สำหรับ Digital ในขั้นตอนเดียว — ใช้ ADC channel แทน pin โดยตรง

ขั้นที่ 2: อ่าน Intensity + ตรวจสอบเปลวไฟ

    float intensity = FlameSensor_ReadIntensity(&f);

    USART_Print("Intensity: ");
    USART_PrintFloat(intensity, 1);

    if(FlameSensor_IsFlameDetected(&f)) {
        USART_Println(" >>> FIRE DETECTED!");
    } else if(intensity > 0.5f) {
        USART_Println(" (possible flame)");
    } else {
        USART_Println(" (normal)");
    }

🔍 FlameSensor_ReadIntensity(&f) — อ่านค่าความเข้ม IR แบบ normalized (0.0-1.0) — FlameSensor_IsFlameDetected(&f) ตรวจสอบทั้ง Analog + Digital — คืน true เมื่อตรวจพบเปลวไฟ

ขั้นที่ 3: ตรวจจับไฟ + Debounce + ป้องกัน False Alarm

    pinMode(PC0, PIN_MODE_OUTPUT); // Buzzer
    pinMode(PC3, PIN_MODE_OUTPUT); // Relay ตัดแก๊ส

    uint8_t fire_count = 0;
    Timer_t tmr;
    Start_Timer(&tmr, 200, 1);

    while(1) {
        if(Is_Timer_Expired(&tmr)) {
            float intensity = FlameSensor_ReadIntensity(&f);

            if(FlameSensor_IsFlameDetected(&f)) {
                fire_count++;
                if(fire_count > 5) { // ยืนยัน 5 ครั้งติด (1 วิ)
                    USART_Println("FIRE CONFIRMED!");
                    digitalWrite(PC0, HIGH); // Buzzer ON
                    digitalWrite(PC3, HIGH); // ตัดแก๊ส
                }
            } else {
                if(fire_count > 0) fire_count--;
                if(fire_count == 0) {
                    digitalWrite(PC0, LOW);
                    digitalWrite(PC3, LOW);
                }
            }
            USART_Print("Intensity:"); USART_PrintFloat(intensity, 1);
            USART_Print(" CNT:"); USART_PrintDec(fire_count);
            USART_Println("");
        }
    }
}

🔍 Debounce ตรวจจับไฟ — ต้องเจอต่อเนื่อง 5 ครั้ง (1 วินาที) จึงยืนยัน — ลด false positive จากแสงแดด/ไฟแช็ก — เมื่อยืนยัน → Buzzer + ตัดแก๊ส

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
ตรวจจับตลอดทั้งที่ไม่มีไฟแสงแดด/หลอดไฟมี IR — false triggerหลีกเลี่ยงแดดตรง — ใช้ debounce — เพิ่ม fire_count threshold
Digital ไม่เคย HIGHPot sensitivity ต่ำเกินไปหมุน Pot ทวนเข็มจน LED สีแดงติดตอนมีไฟ — แล้วหมุนกลับเล็กน้อย
ตรวจจับระยะใกล้มากเท่านั้นเซนเซอร์ sensitivity ต่ำ / เปลวไฟเล็กใช้เลนส์รวมแสง — เปลวไฟยิ่งใหญ่ยิ่งตรวจจับไกล — ระยะปกติ 20-100cm
False alarm ตอนกลางคืนหลอดไฟ Incandescent ปล่อย IRใช้ Analog + Digital ร่วม — ตรวจสอบ pattern ของเปลวไฟ (flicker 10-20Hz)

🛠️ สร้าง Product: Fire Alarm System

KY-026 + Buzzer + Relay — ตรวจจับไฟ — แจ้งเตือน 2 ระดับ — ตัดแก๊สอัตโนมัติ

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "FlameSensor.h"

#define BUZZER PC0
#define GAS_RELAY PC3
#define LED_OK PC2

int main(void)
{
    SystemCoreClockUpdate();
    Timer_Init();
    USART_Init(115200);

    FlameSensor_Instance f;
    FlameSensor_Init(&f, ADC_CH_PC4, PC1);

    pinMode(BUZZER, PIN_MODE_OUTPUT);
    pinMode(GAS_RELAY, PIN_MODE_OUTPUT);
    pinMode(LED_OK, PIN_MODE_OUTPUT);
    digitalWrite(LED_OK, HIGH);

    uint8_t fire_streak = 0;
    Timer_t tmr;
    Start_Timer(&tmr, 250, 1);

    while(1)
    {
        if(Is_Timer_Expired(&tmr))
        {
            float intensity = FlameSensor_ReadIntensity(&f);

            if(FlameSensor_IsFlameDetected(&f)) {
                fire_streak++;
                if(fire_streak >= 8) { // 2 seconds confirm
                    digitalWrite(BUZZER, HIGH);
                    digitalWrite(GAS_RELAY, HIGH);
                    digitalWrite(LED_OK, LOW);
                    USART_Println("FIRE! GAS CUT!");
                }
            } else if(intensity > 0.5f) {
                USART_Println("WARNING: Heat detected");
                fire_streak = 0;
            } else {
                if(fire_streak > 0) fire_streak--;
                if(fire_streak == 0) {
                    digitalWrite(BUZZER, LOW);
                    digitalWrite(LED_OK, HIGH);
                }
            }
        }
    }
}

✏️ ลองทำดู

  1. ตรวจจับ flicker ของเปลวไฟ — วัดความถี่ 10-20Hz — แยกเปลวไฟจากแสงคงที่
  2. ใช้ EXTI interrupt จาก Digital pin — ขัดจังหวะทันทีเมื่อมีไฟ — ลด polling overhead
  3. หลายเซนเซอร์ — ใช้ FlameSensor หลาย instance — ตรวจจับทิศทางของไฟ
  4. ส่ง SMS alert ผ่าน SIM800L — เมื่อตรวจจับไฟ → ส่ง SMS + โทรแจ้งเตือน
