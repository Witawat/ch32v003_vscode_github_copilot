KY-026 — Flame Sensor ตรวจจับเปลวไฟ IR + Digital

KY-026 ใช้ IR Photodiode ตรวจจับเปลวไฟในช่วงความยาวคลื่น 760-1100nm — มี LM393 Comparator สำหรับ Digital Output — Pot ปรับ sensitivity — มุมตรวจจับ ~60° ระยะไกลสุด ~1m สำหรับไฟแช็ก

📚 หลักการทำงาน

KY-026 Flame Sensor ใช้ Infrared Photodiode ตรวจจับรังสี IR จากเปลวไฟ — เปลวไฟปล่อย IR ในช่วง 760-1100nm — เซนเซอร์กรองแสง visible ออก — เมื่อมีเปลวไฟ ADC จะสูงขึ้น (แสง IR มากขึ้น)

2 Outputs: Analog (A0) — ใช้ ADC อ่านความเข้ม IR (0-3.3V) — Digital (D0) — LM393 Comparator เทียบกับ threshold ที่ปรับด้วย Pot — D0 = HIGH เมื่อ IR เกิน threshold — ใช้ขัดจังหวะ MCU ได้

Digital vs Analog: Digital ใช้ตรวจจับเร็ว (GPIO interrupt) — Analog ใช้บอกระยะทาง/ความแรงของไฟ — การใช้ทั้งคู่ช่วยลด False Positive (แสงแดด vs เปลวไฟจริง)

ข้อควรระวัง: แสงแดดมี IR สูง อาจ trigger false positive — หลีกเลี่ยงการติดตั้งที่โดนแดดตรง — ใช้ Analog threshold หลายระดับเพื่อยืนยัน

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Init ADC + GPIO สำหรับ Digital

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"

#define FLAME_ADC PD2
#define FLAME_DIG PC3

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    ADC_Init(FLAME_ADC, ADC_CH0);
    pinMode(FLAME_DIG, PIN_MODE_INPUT); // Digital input
    USART_Init(115200);

    USART_Print("Flame Sensor Ready\r\n");

🔍 ใช้ทั้ง Analog (PD2) และ Digital (PC3) — Digital อ่านด้วย GPIO — Analog วัดความเข้ม IR ผ่าน ADC

ขั้นที่ 2: อ่านทั้ง Analog + Digital

    uint16_t adc = ADC_Read(FLAME_ADC);
    uint8_t dig = GPIO_Read(FLAME_DIG);

    USART_Print("ADC:"); USART_PrintDec(adc);
    USART_Print(" Digital:");
    USART_Print(dig ? "FIRE!" : "Normal");
    USART_Print("\r\n");

    // ADC > 800 = มีไฟใกล้
    if(adc > 800) {
        USART_Print(">>> FLAME DETECTED (strong) <<<\r\n");
    } else if(adc > 500) {
        USART_Print(">>> Possible flame (weak) <<<\r\n");
    }

🔍 GPIO_Read(PC3) อ่าน digital status — HIGH = มีไฟ — Analog แบ่งเป็น 3 ระดับ: >800 (ไฟใกล้/แรง), >500 (ไฟไกล/อ่อน), <500 (ปกติ) — ปรับ threshold ตามสภาพแวดล้อม

ขั้นที่ 3: ตรวจจับไฟ + แจ้งเตือน + ป้องกัน False Alarm

    pinMode(PC0, PIN_MODE_OUTPUT); // Buzzer
    pinMode(PC1, PIN_MODE_OUTPUT); // Relay ตัดแก๊ส

    uint8_t fire_count = 0;
    Timer_t tmr;
    Start_Timer(&tmr, 200, 1);

    while(1) {
        if(Is_Timer_Expired(&tmr)) {
            uint16_t adc = ADC_Read(FLAME_ADC);
            uint8_t dig = GPIO_Read(FLAME_DIG);

            if(dig && adc > 700) {
                fire_count++;
                if(fire_count > 5) { // ยืนยัน 5 ครั้งติด
                    USART_Print("FIRE CONFIRMED!\r\n");
                    digitalWrite(PC0, HIGH); // Buzzer ON
                    digitalWrite(PC1, HIGH); // ตัดแก๊ส
                }
            } else {
                if(fire_count > 0) fire_count--;
                if(fire_count == 0) {
                    digitalWrite(PC0, LOW);
                    digitalWrite(PC1, LOW);
                }
            }
            USART_Print("ADC:"); USART_PrintDec(adc);
            USART_Print(" DIG:"); USART_PrintDec(dig);
            USART_Print(" CNT:"); USART_PrintDec(fire_count);
            USART_Print("\r\n");
        }
    }
}

🔍 Debounce ตรวจจับไฟ — ต้องเจอต่อเนื่อง 5 ครั้ง (1 วินาที) จึงยืนยัน — ลด false positive จากแสงแดด/ไฟแช็ก — เมื่อยืนยัน → Buzzer + ตัดแก๊ส

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
ตรวจจับตลอดทั้งที่ไม่มีไฟแสงแดด/หลอดไฟมี IR — false triggerหลีกเลี่ยงแดดตรง — ใช้ Analog threshold สูงขึ้น — เพิ่ม debounce
Digital ไม่เคย HIGHPot sensitivity ต่ำเกินไปหมุน Pot ทวนเข็มจน LED สีแดงติดตอนมีไฟ — แล้วหมุนกลับเล็กน้อย
ตรวจจับระยะใกล้มากเท่านั้นเซนเซอร์ sensitivity ต่ำ / เปลวไฟเล็กใช้เลนส์รวมแสง — เปลวไฟยิ่งใหญ่ยิ่งตรวจจับไกล — ระยะปกติ 20-100cm
ADC ค่านิ่งไม่เปลี่ยนเซนเซอร์เสีย / ต่อสายผิดตรวจสอบ VCC=3.3V — A0 ควรเปลี่ยนเมื่อโบกมือผ่านหรือจุดไฟแช็ก
False alarm ตอนกลางคืนหลอดไฟ Incandescent ปล่อย IRใช้ Analog + Digital ร่วม — ตรวจสอบ pattern ของเปลวไฟ (flicker 10-20Hz)

🛠️ สร้าง Product: Fire Alarm System

KY-026 + Buzzer + Relay — ตรวจจับไฟ 3 จุด — แจ้งเตือน 2 ระดับ (Warning / Fire) — ตัดแก๊สอัตโนมัติ

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"

#define FLAME_A PD2
#define FLAME_D PC3
#define BUZZER PC0
#define GAS_RELAY PC1
#define LED_OK PC2

int main(void)
{
    SystemCoreClockUpdate();
    Timer_Init();
    ADC_Init(FLAME_A, ADC_CH0);
    pinMode(FLAME_D, PIN_MODE_INPUT);
    pinMode(BUZZER, PIN_MODE_OUTPUT);
    pinMode(GAS_RELAY, PIN_MODE_OUTPUT);
    pinMode(LED_OK, PIN_MODE_OUTPUT);
    USART_Init(115200);

    digitalWrite(LED_OK, HIGH);
    uint8_t fire_streak = 0;
    Timer_t tmr;
    Start_Timer(&tmr, 250, 1);

    while(1)
    {
        if(Is_Timer_Expired(&tmr))
        {
            uint16_t adc = ADC_Read(FLAME_A);
            uint8_t dig = GPIO_Read(FLAME_D);

            if(dig && adc > 700) {
                fire_streak++;
                if(fire_streak >= 8) { // 2 seconds confirm
                    digitalWrite(BUZZER, HIGH);
                    digitalWrite(GAS_RELAY, HIGH);
                    digitalWrite(LED_OK, LOW);
                    USART_Print("FIRE! GAS CUT!\r\n");
                }
            } else if(adc > 500) {
                USART_Print("WARNING: Heat detected\r\n");
                fire_streak = 0;
            } else {
                if(fire_streak > 0) fire_streak--;
                if(fire_streak == 0) {
                    digitalWrite(BUZZER, LOW);
                    digitalWrite(LED_OK, HIGH);
                }
            }

            USART_Print("ADC:"); USART_PrintDec(adc);
            USART_Print(" D:"); USART_PrintDec(dig);
            USART_Print(" S:"); USART_PrintDec(fire_streak);
            USART_Print("\r\n");
        }
    }
}

✏️ ลองทำดู

  1. ตรวจจับ flicker ของเปลวไฟ — วัดความถี่ 10-20Hz — แยกเปลวไฟจากแสงคงที่
  2. ใช้ EXTI interrupt จาก Digital pin — ขัดจังหวะทันทีเมื่อมีไฟ — ลด polling overhead
  3. หลายเซนเซอร์ — ใช้ Analog Mux หรือ GPIO หลายตัว — ตรวจจับทิศทางของไฟ
  4. ส่ง SMS alert ผ่าน SIM800L — เมื่อตรวจจับไฟ → ส่ง SMS + โทรแจ้งเตือน
