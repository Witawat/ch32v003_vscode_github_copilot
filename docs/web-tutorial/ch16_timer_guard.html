CH32V003 มีแค่ 2 hardware timers (TIM1, TIM2)! — ต้องใช้ guard system มิฉะนั้น library จะแย่ง timer กัน

📚 หลักการทำงาน

Timer Resource Guard — ระบบจัดการกรรมสิทธิ์ timer แบบ cooperative — ป้องกันไม่ให้หลาย library ใช้ timer ตัวเดียวกันพร้อมกัน

ทำไมต้องมี Resource Guard?

CH32V003 มี hardware timer จำกัด (TIM1, TIM2) — แต่ SimpleHAL มี library หลายตัวที่ต้องการ timer: PWM, Servo, Encoder, Frequency measurement, IR Remote, Input Capture — ถ้าแต่ละ library ตั้งค่า timer โดยไม่เช็คว่าคนอื่นใช้อยู่หรือไม่ → ค่า timer register จะถูกเขียนทับ → ระบบพัง

หลักการ

  1. ทุกครั้งที่ library จะใช้ timer — ต้องเช็ค g_timX_owner
  2. ถ้า owner == TIMER_OWNER_NONE → จอง timer (ตั้ง owner = ตัวเอง)
  3. ถ้า owner != TIMER_OWNER_NONE → คนอื่นใช้อยู่ → คืน error
  4. เมื่อเลิกใช้ → คืน timer (ตั้ง owner = TIMER_OWNER_NONE)

TIMEXT_DEFAULT_TIMER

Global config ที่เลือก timer เริ่มต้นสำหรับฟังก์ชันต่างๆ — ถ้าต้องการให้ฟังก์ชันใช้ TIM2 แทน TIM1 → เปลี่ยน TIMEXT_DEFAULT_TIMER ก่อนเรียก init

ผลลัพธ์

  • ถ้ามี conflict — compile-time หรือ runtime error — ไม่ใช่ silent corruption
  • ผู้ใช้รู้ทันทีว่าต้องใช้ timer คนละตัว — หรือ disable feature ที่ไม่จำเป็น
  • ระบบ fail-safe — ไม่มีทางที่ timer จะถูกแย่งโดยไม่รู้ตัว

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: ดูตัวอย่าง — TIM1 ถูกใช้โดย PWM

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();

    // เริ่ม PWM บน PC4 — ใช้ TIM1
    PWM_SimpleInit(2, PC4);  // success
    // ตอนนี้ g_tim1_owner = TIMER_OWNER_PWM

🔍 เมื่อ PWM_SimpleInit() ถูกเรียก — มันเช็ค g_tim1_owner — ถ้าเป็น NONE → จอง — ตอนนี้ TIM1 เป็นของ PWM

ขั้นที่ 2: Conflict — Encoder พยายามใช้ TIM1

    // พยายามใช้ Encoder บน TIM1
    uint8_t ok = Encoder_Init(TIM1, PA1, PA2);
    if(!ok) {
        // Conflict! TIM1 ถูกใช้โดย PWM แล้ว
        USART_Print("ERROR: TIM1 already used by PWM!\r\n");
        USART_Print("Use TIM2 instead\r\n");

        // ลองใช้ TIM2 แทน
        ok = Encoder_Init(TIM2, PC4, PC5);
        if(ok) {
            USART_Print("Encoder on TIM2 OK\r\n");
        }
    }

🔍 Encoder_Init(TIM1, ...) — จะเช็ค g_tim1_owner — พบว่า != NONE (เป็น PWM) → คืน 0 — ผู้ใช้รู้ทันทีว่ามี conflict — ไม่ใช่เงียบแล้ว PWM เพี้ยน

ขั้นที่ 3: Release Timer — เปลี่ยนเจ้าของ

    // คืน TIM1 (PWM เลิกใช้)
    PWM_DeInit(2);  // หรือ Timer_Release(TIM1)
    // ตอนนี้ g_tim1_owner = TIMER_OWNER_NONE

    // ใช้ Encoder บน TIM1 ได้แล้ว
    Encoder_Init(TIM1, PA1, PA2);  // success
    // g_tim1_owner = TIMER_OWNER_ENCODER

🔍 PWM_DeInit() — คืนกรรมสิทธิ์ TIM1 — หลังจากนี้ library อื่นใช้ TIM1 ได้ — หรือใช้ Timer_Release(TIM1) โดยตรง

ขั้นที่ 4: TIMEXT_DEFAULT_TIMER — เปลี่ยน Timer เริ่มต้น

    // กำหนดให้ library ทั้งหมดใช้ TIM2 เป็น default
    // (ไม่ต้องระบุ timer ทุกครั้ง)
    #define TIMEXT_DEFAULT_TIMER TIM2

    // ก่อน include SimpleHAL.h หรือหลัง include ก็ได้
    // ขึ้นอยู่กับการออกแบบ

    // ต่อจากนี้ — PWM, Servo, Encoder จะพยายามใช้ TIM2
    // โดยอัตโนมัติ — ยกเว้นระบุ timer เอง

    Servo_SimpleInit(1);  // ใช้ TIM2 (default)
    Encoder_Init(TIM1, PA1, PA2);  // ใช้ TIM1 (ระบุเอง)

🔍 TIMEXT_DEFAULT_TIMER — กำหนด timer เริ่มต้น — แต่ละ library เคารพค่านี้ — แต่ถ้าระบุ timer เองในการเรียกฟังก์ชัน → ใช้ที่ระบุ

ขั้นที่ 5: เช็คสถานะ Timer ทั้งระบบ

    // ดูว่าใครใช้ timer อะไรอยู่
    USART_Print("TIM1 owner: ");
    switch(g_tim1_owner) {
        case TIMER_OWNER_NONE:    USART_Print("FREE\r\n"); break;
        case TIMER_OWNER_PWM:     USART_Print("PWM\r\n"); break;
        case TIMER_OWNER_ENCODER: USART_Print("Encoder\r\n"); break;
        case TIMER_OWNER_SERVO:   USART_Print("Servo\r\n"); break;
        case TIMER_OWNER_FREQ:    USART_Print("Frequency\r\n"); break;
        case TIMER_OWNER_CAPTURE: USART_Print("Capture\r\n"); break;
        case TIMER_OWNER_IR:      USART_Print("IR Remote\r\n"); break;
        default:                  USART_Print("Custom\r\n"); break;
    }

    // หรือใช้ helper
    uint8_t t1 = Timer_GetOwner(TIM1);
    uint8_t t2 = Timer_GetOwner(TIM2);
    USART_Print("TIM1=");
    USART_PrintNum(t1);
    USART_Print(" TIM2=");
    USART_PrintNum(t2);
    USART_Print("\r\n");

🔍 Timer_GetOwner(timer) — อ่านค่า owner — ใช้ debug หรือ runtime check — g_timX_owner เป็น global — อ่านได้จากทุกที่

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
Encoder_Init() คืน 0 ตลอดTIM1 ถูกใช้โดย library อื่นเช็ค g_tim1_owner — ใช้ TIM2 แทน หรือ release TIM1 ก่อน
PWM เพี้ยนหลังเรียก Encoderลืมเปิด Timer Guard — library แข่งกันใช้ timerTimer Guard ทำงานอัตโนมัติใน SimpleHAL — ไม่ได้เปิด Guard = ใช้ library นอก SimpleHAL
Servo กระตุก, LED กระพริบไม่ตรงใช้ TIM1 กับหลาย feature พร้อมกันโดยไม่รู้ตัวแยกเป็น TIM1 และ TIM2 — หรือใช้ software PWM แทน hardware PWM
ไม่รู้ว่าใครใช้ TIM1 อยู่ไม่ได้เช็ค g_tim1_ownerเรียก Timer_GetOwner(TIM1) หรืออ่าน g_tim1_owner โดยตรง
Timer ถูกจองแต่ไม่เคยคืนไม่ได้เรียก DeInit เมื่อเลิกใช้เรียก PWM_DeInit() / Encoder_DeInit() / Timer_Release()

🛠️ สร้าง Product: ระบบหลาย Timer ไม่เคยชน

ระบบที่ใช้ PWM (TIM1) + Encoder (TIM2) + Servo (TIM2 แชร์กับ Encoder) — Guard ป้องกัน conflict

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"

int main(void)
{
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);

    // ==== Allocate TIM1: PWM LED ====
    uint8_t pwm_ok = PWM_SimpleInit(2, PC4);
    if(pwm_ok) {
        USART_Print("TIM1: PWM on PC4 OK\r\n");
        PWM_SetDuty(2, 50);  // 50%
    } else {
        USART_Print("TIM1: PWM FAILED (in use)\r\n");
    }

    // ==== Allocate TIM2: Encoder ====
    uint8_t enc_ok = Encoder_Init(TIM2, PC5, PC6);
    if(enc_ok) {
        USART_Print("TIM2: Encoder on PC5/PC6 OK\r\n");
    } else {
        USART_Print("TIM2: Encoder FAILED (in use)\r\n");
    }

    // ==== Try: Servo (default TIM1) — จะชนกับ PWM ====
    uint8_t servo_ok = Servo_SimpleInit(1);
    if(!servo_ok) {
        USART_Print("Servo FAILED: TIM1 owned by PWM\r\n");
        USART_Print("    Fix: Use Servo_SimpleInitEx(TIM2, 1)\r\n");
        // ใช้ TIM2 โดยตรง
        servo_ok = Servo_SimpleInitEx(TIM2, 1);
        if(servo_ok) {
            USART_Print("    Servo on TIM2 OK!\r\n");
        }
    }

    // ==== Report Timer Status ====
    USART_Print("--- Timer Status ---\r\n");
    USART_Print("TIM1 owner: ");
    USART_PrintNum(Timer_GetOwner(TIM1));
    USART_Print(" (PWM=");
    USART_PrintNum(TIMER_OWNER_PWM);
    USART_Print(")\r\n");
    USART_Print("TIM2 owner: ");
    USART_PrintNum(Timer_GetOwner(TIM2));
    USART_Print("\r\n");

    while(1)
    {
        if(enc_ok) {
            int32_t pos = Encoder_GetCount();
            USART_Print("Enc: ");
            USART_PrintNum((int32_t)pos);
            USART_Print("\r\n");
        }
        Delay_Ms(500);
    }
}

✏️ ลองทำดู

  1. เปิด PWM บน TIM1 แล้วเรียก Servo_SimpleInit() — ดูว่า Servo init ล้มเหลว — เช็ค g_tim1_owner ว่าเป็น PWM
  2. เปลี่ยน TIMEXT_DEFAULT_TIMER เป็น TIM2 — สังเกตว่า library ทั้งหมดเปลี่ยนไปใช้ TIM2 โดยอัตโนมัติ
  3. เขียน library ของตัวเองที่ใช้ TIM1 — ตั้ง owner ด้วย Timer_SetOwner(TIM1, TIMER_OWNER_CUSTOM) — เช็คว่า library อื่นใช้ TIM1 ไม่ได้
