🔄 Porting Example — Temp Monitor + LED Alert

Arduino Version (ไฟล์ .ino)

#include <DHT.h>  // หรือ Adafruit DHT library

#define DHTPIN    2
#define LEDPIN    13
#define TEMP_MAX  35.0

DHT dht(DHTPIN, DHT11);

void setup() {
    Serial.begin(9600);
    pinMode(LEDPIN, OUTPUT);
    dht.begin();
    Serial.println("Temp Monitor Ready");
}

void loop() {
    float temp = dht.readTemperature();

    if(!isnan(temp)) {
        Serial.print("Temp: ");
        Serial.print(temp);
        Serial.println(" C");

        if(temp > TEMP_MAX) {
            digitalWrite(LEDPIN, HIGH);  // Alert
        } else {
            digitalWrite(LEDPIN, LOW);
        }
    }

    delay(2000);
}

CH32V003 Version — แปลงแล้ว

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#define ENABLE_USART_PRINTLN   1
#define ENABLE_USART_PRINTFLOAT 1
#include "SimpleHAL.h"
#include "DHT.h"

#define DHTPIN    PC0     // ใช้ชื่อ pin จริง — ไม่ใช่แค่เลข 2
#define LEDPIN    PC1
#define TEMP_MAX  35.0f

int main(void) {                             // ← แทน setup()
    SystemCoreClockUpdate();                  // ← ต้องเพิ่ม
    Timer_Init();                             // ← ต้องเพิ่ม
    USART_SimpleInit(BAUD_9600, USART_PINS_DEFAULT);  // ← Serial.begin()

    pinMode(LEDPIN, PIN_MODE_OUTPUT);
    DHT_Init(DHTPIN, DHT11);                 // ← dht.begin()
    USART_Println("Temp Monitor Ready");

    while(1) {                               // ← แทน loop()
        float temp, hum;
        if(DHT_Read(&temp, &hum) == DHT_OK) {  // ← dht.readTemperature()
            USART_Print("Temp: ");
            USART_PrintFloat(temp, 1);
            USART_Println(" C");

            if(temp > TEMP_MAX) {
                digitalWrite(LEDPIN, HIGH);
            } else {
                digitalWrite(LEDPIN, LOW);
            }
        }

        Delay_Ms(2000);                      // ← แทน delay(2000)
        IWDG_Feed();                         // ← กัน hang (ของแถม CH32V003)
    }
}
✅ สิ่งที่เปลี่ยน: setup()→main(), loop()→while(1), Serial→USART, delay→Delay_Ms, pin number→pin name, เพิ่ม SystemCoreClockUpdate+Timer_Init

⚠️ สิ่งที่ Arduino มีแต่ต้องระวังบน CH32V003

1. String Class — ห้ามใช้

// ❌ Arduino style — ใช้ dynamic memory
String msg = "Hello ";
msg += "World";
Serial.println(msg);

// ✅ CH32V003 style — ใช้ char array
char msg[32];
snprintf(msg, sizeof(msg), "Hello World");
USART_Println(msg);

2. 3.3V vs 5V I/O

    Arduino 5V sensor → CH32V003 (3.3V):
    ┌──────────────┐
    │ Sensor (5V)  │── output 5V ──┬── R 2.2k ──┬── CH32V003 pin
    │              │               └── R 3.3k ── GND
    └──────────────┘
    Voltage divider 2.2k+3.3k → 5V→3.0V (safe)

    หรือใช้ Logic Level Converter module (~10 บาท)

3. EEPROM → Flash

// ❌ Arduino — EEPROM library
#include <EEPROM.h>
EEPROM.write(0, 42);
uint8_t val = EEPROM.read(0);

// ✅ CH32V003 — Flash Storage
#include "SimpleFlash.h"
FlashConfig_t cfg;
cfg.key = 0xABCD;
cfg.data[0] = 42;
Flash_SaveConfig(&cfg);           // write
Flash_LoadConfig(&cfg);           // read
uint8_t val = cfg.data[0];
// ⚠️ ต้อง erase ทั้ง page (1KB) ก่อนเขียน!
// ⚠️ Endurance ~10K writes (น้อยกว่า EEPROM 100K)

✏️ ลองทำดู

  1. เลือก Arduino sketch ที่มีอยู่ — แปลงตามขั้นตอนด้านบน — ทดสอบว่าได้ผลเหมือน Arduino ไหม
  2. เพิ่ม Sleep mode ในโค้ดที่แปลงแล้ว — วัดกระแสเปรียบเทียบก่อน/หลัง
  3. ลองใช้ SimpleScheduler (บท 18) แทน delay() — ทำให้โค้ด Arduino เดิมเป็น non-blocking
  4. แปลง Arduino project ใหญ่ (3+ libraries) → ดูว่าใช้ Flash/RAM เท่าไหร่ → optimize
