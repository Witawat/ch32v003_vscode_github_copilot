ULN2003 เป็น Darlington array ขับกระแสสูง — ต่อ MCU 4 GPIO เข้ากับ IN1–IN4 — ใช้ไฟ 5V เลี้ยงมอเตอร์ (ห้ามใช้ 3.3V) — 28BYJ-48 มี gear ratio 1:64, stride angle 5.625°, 2048 steps/rev (full-step)

📚 หลักการทำงาน

28BYJ-48 เป็น unipolar stepper motor ราคาถูก — ใช้ไฟ 5V, กินกระแส ~240mA ต่อเฟส, gearbox 1:64 — ในโหมด Full-Step (4 steps/cycle) หมุน 1 รอบ = 2048 steps, Half-Step (8 steps/cycle) = 4096 steps/rev — ความละเอียดสูงกว่า 2 เท่าแต่แรงบิดลด

Step Sequence: Full-step ใช้ coil activation pattern 4 state: A→AB→B→BC→C→CD→D→DA — สลับขั้ว coil เพื่อดึง rotor หมุนทีละ step — Half-Step เพิ่ม state กลางอีก 4 state รวมเป็น 8

Speed Control: Delay_Us(speed) ระหว่างแต่ละ step — speed ยิ่งน้อย ยิ่งหมุนเร็ว — ความเร็วสูงสุด ~10 RPM (ประมาณ speed=800μs) ที่ 5V — เกินกว่านี้มอเตอร์จะ missed step

Acceleration: เริ่มจาก speed สูง (หมุนช้า) แล้วค่อยลดลง — ป้องกัน missed step และมอเตอร์กระตุก — ใช้ ramp-up/down profile

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Init Stepper Pins

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"

#define IN1 PC0
#define IN2 PC1
#define IN3 PC2
#define IN4 PC3

// Full-Step Sequence (4 states)
const uint8_t full_step[4] = {
    0b0001,  // IN1
    0b0010,  // IN2
    0b0100,  // IN3
    0b1000   // IN4
};

void stepper_write(uint8_t step) {
    digitalWrite(IN1, step & 0x01);
    digitalWrite(IN2, step & 0x02);
    digitalWrite(IN3, step & 0x04);
    digitalWrite(IN4, step & 0x08);
}

🔍 ใช้ bitmask 4-bit ควบคุม 4 pins — full_step array เก็บ sequence — stepper_write() แยกบิตเขียน GPIO ทีละ pin

ขั้นที่ 2: Step Function + Direction

void stepper_step(uint16_t steps, uint16_t speed_us, int8_t dir) {
    int8_t idx = 0;
    for(uint16_t i = 0; i < steps; i++) {
        stepper_write(full_step[idx]);
        idx += dir;
        if(idx >= 4) idx = 0;
        if(idx < 0) idx = 3;
        Delay_Us(speed_us);  // speed_us = 800-5000μs
    }
}

🔍 stepper_step(n, speed, dir) — dir=+1 หมุนตามเข็ม, dir=-1 ทวนเข็ม — speed_us ควบคุมความเร็ว (ยิ่งน้อยยิ่งเร็ว) — 2048 steps = 1 รอบในโหมด full-step

ขั้นที่ 3: Acceleration + Main Loop

void stepper_accel(uint16_t steps, uint16_t start_speed, uint16_t end_speed) {
    int32_t speed = start_speed;
    int32_t step_inc = ((int32_t)end_speed - start_speed) / steps;
    for(uint16_t i = 0; i < steps; i++) {
        stepper_step(1, (uint16_t)speed, 1);
        speed += step_inc;
        if(speed < end_speed) speed = end_speed;
    }
}

int main(void) {
    SystemCoreClockUpdate();
    pinMode(IN1, PIN_MODE_OUTPUT);
    pinMode(IN2, PIN_MODE_OUTPUT);
    pinMode(IN3, PIN_MODE_OUTPUT);
    pinMode(IN4, PIN_MODE_OUTPUT);

    while(1) {
        // หมุน 1 รอบ (2048 steps) ด้วย acceleration
        stepper_accel(2048, 2000, 900);  // start slow → speed up
        Delay_Ms(1000);
        // หมุนกลับ 1 รอบ
        stepper_step(2048, 1200, -1);
        Delay_Ms(1000);
    }
}

🔍 stepper_accel() เปลี่ยนความเร็วจาก start_speed→end_speed แบบ linear — ป้องกัน motor stall — 1 รอบ = 2048 steps (full-step), ~4096 steps (half-step)

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
มอเตอร์สั่นแต่ไม่หมุนลำดับ step sequence ผิดตรวจสอบลำดับ coil: A→B→C→D หรือ A→AB→B→BC... ห้ามสลับ
มอเตอร์หมุนช้ามากลืม half-step หรือ speed สูงเกิน 5000μsใช้ half-step = 4096 steps/rev — ลด speed ลง 800-1200μs เพื่อ speed up
Missed steps — เสียงดังติ๊กspeed ต่ำเกินไป (เร็วไป) หรือไม่มี accelerationเพิ่ม ramp-up acceleration — เริ่มจาก speed 2000μs ค่อยลดลง
แรงบิดไม่พอ — ไม่สามารถหมุนโหลดใช้ 3.3V เลี้ยงมอเตอร์28BYJ-48 ต้องใช้ 5V — ตรวจสอบ ULN2003 VCC ต่อเข้ากับ 5V
ร้อนมากเมื่อหยุดนิ่งStepper_Hold() จ่ายไฟ coil ตลอดเวลาใช้ Stepper_Stop() เพื่อปล่อย coil — หรือลด duty cycle ใน hold mode

🛠️ สร้าง Product: Programmable Stepper Positioner

หมุน stepper ไปตำแหน่งองศาที่ต้องการด้วย acceleration — ใช้ Potentiometer ตั้งองศา + ปุ่มกด start

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"

#define IN1 PC0
#define IN2 PC1
#define IN3 PC2
#define IN4 PC3

const uint8_t half_step[8] = {
    0b0001,0b0011,0b0010,0b0110,0b0100,0b1100,0b1000,0b1001
};
int32_t current_pos = 0;  // in steps

void stepper_goto(int32_t target, uint16_t speed) {
    int32_t delta = target - current_pos;
    int8_t dir = (delta > 0) ? 1 : -1;
    uint32_t steps = (delta > 0) ? delta : -delta;
    static uint8_t idx = 0;
    for(uint32_t i = 0; i < steps; i++) {
        digitalWrite(IN1, half_step[idx] & 0x01);
        digitalWrite(IN2, half_step[idx] & 0x02);
        digitalWrite(IN3, half_step[idx] & 0x04);
        digitalWrite(IN4, half_step[idx] & 0x08);
        idx += dir; if(idx >= 8) idx = 0; if(idx < 0) idx = 7;
        current_pos += dir;
        Delay_Us(speed);
    }
}

int main(void) {
    SystemCoreClockUpdate();
    ADC_Init_Single(PA1);
    pinMode(PC4, PIN_MODE_INPUT_PULLUP);
    pinMode(IN1, PIN_MODE_OUTPUT);
    pinMode(IN2, PIN_MODE_OUTPUT);
    pinMode(IN3, PIN_MODE_OUTPUT);
    pinMode(IN4, PIN_MODE_OUTPUT);

    while(1) {
        if(!digitalRead(PC4)) {  // กดปุ่ม
            uint16_t adc = ADC_Read(PA1);
            int32_t target_deg = (int32_t)(adc * 360 / 4095);
            int32_t target_steps = (int32_t)(target_deg * 4096 / 360);
            stepper_goto(target_steps, 1000);
            Delay_Ms(200); // debounce
        }
    }
}
