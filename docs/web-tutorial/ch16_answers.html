📝 เฉลย — บทที่ 16: Timer Resource Guard

ข้อ 1: ตรวจจับ timer conflict (PWM + TIM)

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include "SimpleHAL.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);

    USART_Print("=== Timer Conflict Detection ===\r\n\r\n");

    // ==== Step 1: ใช้ TIM1 สำหรับ PWM ====
    uint8_t pwm_ok = PWM_SimpleInit(2, PC4);
    if (pwm_ok) {
        USART_Print("[OK] PWM on TIM1/PC4\r\n");
        PWM_SetDuty(2, 50);
        USART_Print("     g_tim1_owner = ");
        USART_PrintNum(g_tim1_owner);
        USART_Print(" (PWM=");
        USART_PrintNum(TIMER_OWNER_PWM);
        USART_Print(")\r\n");
    } else {
        USART_Print("[FAIL] PWM init failed\r\n");
    }

    // ==== Step 2: พยายามใช้ TIM1 สำหรับ Encoder ====
    USART_Print("\r\n--- Attempting Encoder on TIM1 ---\r\n");
    uint8_t enc_ok = Encoder_Init(TIM1, PA1, PA2);
    if (!enc_ok) {
        USART_Print("[CONFLICT!] Encoder cannot use TIM1\r\n");
        USART_Print("            TIM1 owned by PWM (owner=");
        USART_PrintNum(g_tim1_owner);
        USART_Print(")\r\n");

        USART_Print("            Checking TIM2 instead...\r\n");
        enc_ok = Encoder_Init(TIM2, PC5, PC6);
        if (enc_ok) {
            USART_Print("            [OK] Encoder on TIM2\r\n");
        }
    }

    // ==== Step 3: ตรวจสอบสถานะทั้งระบบ ====
    USART_Print("\r\n--- Timer Status ---\r\n");
    USART_Print("TIM1 owner: ");
    switch (g_tim1_owner) {
        case TIMER_OWNER_NONE:    USART_Print("FREE\r\n"); break;
        case TIMER_OWNER_PWM:     USART_Print("PWM\r\n"); break;
        case TIMER_OWNER_ENCODER: USART_Print("Encoder\r\n"); break;
        default:                  USART_Print("OTHER\r\n"); break;
    }
    USART_Print("TIM2 owner: ");
    switch (g_tim2_owner) {
        case TIMER_OWNER_NONE:    USART_Print("FREE\r\n"); break;
        case TIMER_OWNER_ENCODER: USART_Print("Encoder\r\n"); break;
        default:                  USART_Print("OTHER\r\n"); break;
    }

    // ==== Step 4: Release TIM1 ====
    USART_Print("\r\n--- Releasing TIM1 ---\r\n");
    PWM_DeInit(2);  // or Timer_Release(TIM1)
    USART_Print("TIM1 released — now FREE\r\n");

    // Now TIM1 is free
    USART_Print("g_tim1_owner = ");
    USART_PrintNum(g_tim1_owner);
    USART_Print(" (should be 0 = NONE)\r\n");

    while (1) {
        Delay_Ms(1000);
    }
}
🧠 แนวคิด: ทุกครั้งที่ library ต้องการใช้ timer — เช็ค g_timX_owner — ถ้า != NONE → มีคนใช้อยู่ → คืน error (0) — ผู้ใช้รู้ทันที มี conflict — ไม่ใช่ silent corruption — การตรวจจับ: 1) ดู return value ของ init function 2) อ่าน g_tim1_owner/g_tim2_owner โดยตรง 3) ใช้ Timer_GetOwner() — เมื่อพบ conflict: เลือก timer อื่น, หรือ release timer ที่ใช้อยู่, หรือ disable feature ที่ไม่จำเป็น — CH32V003 มีแค่ TIM1 + TIM2 → ต้องบริหารให้ดี

ข้อ 2: เปลี่ยน TIMEXT_DEFAULT_TIMER เป็น TIM1

// ตั้งค่า default timer ก่อน include SimpleHAL
#define TIMEXT_DEFAULT_TIMER  TIM1

// หรือ: ตั้งในโค้ดหลัง include (ขึ้นอยู่กับ implementation)
// #define TIMEXT_DEFAULT_TIMER TIM1

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include "SimpleHAL.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);

    USART_Print("=== TIMEXT_DEFAULT_TIMER = TIM1 ===\r\n\r\n");

    // Servo จะใช้ TIM1 โดยอัตโนมัติ (ตาม default)
    uint8_t servo_ok = Servo_SimpleInit(1);
    if (servo_ok) {
        USART_Print("[OK] Servo on default timer (TIM1)\r\n");
        USART_Print("     g_tim1_owner = ");
        USART_PrintNum(g_tim1_owner);
        USART_Print("\r\n");
    }

    // แสดงผลของ TIMEXT_DEFAULT_TIMER
    USART_Print("\r\n--- Impact of TIMEXT_DEFAULT_TIMER ---\r\n");
    USART_Print("Default timer: ");
    #if TIMEXT_DEFAULT_TIMER == TIM1
        USART_Print("TIM1\r\n");
    #elif TIMEXT_DEFAULT_TIMER == TIM2
        USART_Print("TIM2\r\n");
    #else
        USART_Print("Unknown\r\n");
    #endif

    USART_Print("\r\nLibraries using default:");
    USART_Print("\r\n  Servo_SimpleInit() -> TIM1");
    USART_Print("\r\n  PWM_SimpleInit()   -> TIM1 (if not specified)");
    USART_Print("\r\n  But Encoder_Init(TIM2,...) -> TIM2 (explicit)\r\n");

    // Explicitly use TIM2 for another feature
    uint8_t pwm_ok = PWM_SimpleInitEx(TIM2, 4, PC5);
    if (pwm_ok) {
        USART_Print("\r\n[OK] PWM on TIM2/PC5 (explicit)");
        USART_Print("\r\n     TIM1: Servo, TIM2: PWM");
        USART_Print("\r\n     No conflict!\r\n");
    }

    while (1) { Delay_Ms(1000); }
}
🧠 แนวคิด: TIMEXT_DEFAULT_TIMER กำหนดว่า library ไหนจะใช้ timer อะไรเป็น default — เปลี่ยนแค่ #define — library ทุกตัวที่รับ default จะเปลี่ยนไปใช้ timer ที่กำหนด — ตัวอย่าง: ถ้า TIMEXT_DEFAULT_TIMER = TIM2 → Servo, PWM จะพยายามใช้ TIM2 → TIM1 ว่างสำหรับ Encoder โดยอัตโนมัติ — Explicit override: บางฟังก์ชันมีเวอร์ชัน Ex (เช่น Servo_SimpleInitEx) ให้ระบุ timer เอง — ข้าม default — กลยุทธ์: ใช้ default สำหรับฟีเจอร์หลัก → ใช้ explicit สำหรับฟีเจอร์รอง → ป้องกัน conflict โดยไม่ต้องตรวจสอบเอง

ข้อ 3: ใช้ SysTick Timer_t แทน TIM2 (หลีกเลี่ยงชน)

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include "SimpleHAL.h"

// ใช้ SysTick แทน hardware timer สำหรับ software tasks
volatile uint32_t g_systick_ms = 0;

void SysTick_Handler(void) {
    g_systick_ms++;
}

// Software timer abstraction using SysTick
typedef struct {
    uint32_t interval_ms;
    uint32_t last_run;
    void (*callback)(void);
    uint8_t  enabled;
    uint8_t  one_shot;
} SoftTimer_t;

SoftTimer_t g_timers[4];

void soft_timer_set(SoftTimer_t *t, uint32_t ms, void (*cb)(void), uint8_t once) {
    t->interval_ms = ms;
    t->callback = cb;
    t->enabled = 1;
    t->one_shot = once;
    t->last_run = g_systick_ms;
}

void soft_timer_poll(void) {
    for (int i = 0; i < 4; i++) {
        if (!g_timers[i].enabled) continue;

        if (g_systick_ms - g_timers[i].last_run >= g_timers[i].interval_ms) {
            g_timers[i].last_run = g_systick_ms;

            if (g_timers[i].callback) {
                g_timers[i].callback();
            }

            if (g_timers[i].one_shot) {
                g_timers[i].enabled = 0;
            }
        }
    }
}

// Callback functions
void cb_led_toggle(void) {
    digitalToggle(PC0);
}

void cb_print_uptime(void) {
    USART_Print("Uptime: ");
    USART_PrintNum((int32_t)(g_systick_ms / 1000));
    USART_Print("s\r\n");
}

void cb_adc_sample(void) {
    uint16_t adc = analogRead(PD2);
    USART_Print("ADC: ");
    USART_PrintNum(adc);
    USART_Print("\r\n");
}

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
    ADC_SimpleInit();

    pinMode(PC0, PIN_MODE_OUTPUT);  // LED toggle

    USART_Print("=== SysTick Software Timers ===\r\n");
    USART_Print("TIM1, TIM2 FREE for hardware features\r\n");
    USART_Print("Software timers use SysTick only\r\n\r\n");

    // Setup software timers — no hardware timer used!
    soft_timer_set(&g_timers[0], 500,  cb_led_toggle,   0);  // blink LED
    soft_timer_set(&g_timers[1], 3000, cb_print_uptime,  0);  // print every 3s
    soft_timer_set(&g_timers[2], 10000,cb_adc_sample,    0);  // ADC every 10s

    // One-shot timer: do something once after 5s
    soft_timer_set(&g_timers[3], 5000, 0,                1);
    g_timers[3].callback = NULL;  // overwrite below
    // (for one-shot, we check manually in loop)

    USART_Print("3 periodic timers + 1 one-shot started\r\n");

    // === Now TIM1 and TIM2 are completely FREE ===
    // Can use PWM on TIM1, Encoder on TIM2, etc.
    if (PWM_SimpleInit(2, PC4)) {
        USART_Print("PWM on TIM1 OK — no conflict!\r\n");
        PWM_SetDuty(2, 50);
    }

    if (Encoder_Init(TIM2, PC5, PC6)) {
        USART_Print("Encoder on TIM2 OK — no conflict!\r\n");
    }

    USART_Print("\r\nAll 3 timers working:\r\n");
    USART_Print("  SysTick: 3 software timers\r\n");
    USART_Print("  TIM1:    PWM\r\n");
    USART_Print("  TIM2:    Encoder\r\n");
    USART_Print("  No resource conflict!\r\n\r\n");

    uint32_t oneshot_start = g_systick_ms;

    while (1) {
        soft_timer_poll();

        // One-shot check (custom)
        if (g_timers[3].enabled &&
            g_systick_ms - oneshot_start >= 5000) {
            USART_Print(">> 5s one-shot fired!\r\n");
            g_timers[3].enabled = 0;  // disable
        }

        // Optional: read encoder
        static uint32_t last_enc = 0;
        if (g_systick_ms - last_enc >= 2000) {
            last_enc = g_systick_ms;
            int32_t pos = Encoder_GetCount();
            USART_Print("Enc: ");
            USART_PrintNum((int32_t)pos);
            USART_Print("\r\n");
        }
    }
}
🧠 แนวคิด: CH32V003 มี hardware timer จำกัด (TIM1, TIM2) — ใช้ SysTick สำหรับ software task — ทำให้ TIM1/TIM2 ว่างสำหรับ PWM, Encoder, Servo — SysTick: interrupt ทุก 1ms — ใช้สร้าง software timer หลายตัวได้ไม่จำกัด (จำกัดด้วย RAM) — ข้อดี: ไม่เปลือง hardware timer, ปรับ interval ได้ละเอียด, one-shot/repeating — ข้อเสีย: interrupt overhead ~1ms, ใช้ CPU นิดหน่อย — ความละเอียด 1ms (จำกัดด้วย SysTick) — เหมาะกับ: LED blink, polling sensor ช้า, periodic print — ไม่เหมาะกับ: PWM (ต้องการ hardware), frequency measurement, encoder — Alternative: ใช้ Timer_Guard จัดการแทน SysTick — ถ้า project ใหญ่ให้ใช้ Guard ล้วน
