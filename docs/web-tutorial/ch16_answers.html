📝 เฉลย — บทที่ 16: Timer Resource Guard

ข้อ 1: ตรวจจับ conflict ระหว่าง PWM และ SimpleTIM

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"

static const char* owner_name(uint8_t o) {
    switch (o) {
        case TIM_OWNER_NONE:    return "NONE";
        case TIM_OWNER_PWM:     return "PWM";
        case TIM_OWNER_TIMER:   return "TIMER";
        case TIM_OWNER_TIMEXT:  return "TIMEXT";
        default:                return "UNKNOWN";
    }
}

void my_tick(void) { }

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);

    USART_Print("=== Timer Conflict Detection ===\r\n\r\n");

    // ==== Step 1: เปิด PWM บน TIM1 (PWM1_CH4 = PC4) ====
    PWM_Init(PWM1_CH4, 1000);
    PWM_SetDutyCycle(PWM1_CH4, 50);
    PWM_Start(PWM1_CH4);

    USART_Print("[PWM] TIM1/PC4  g_tim1_owner = ");
    USART_Print(owner_name(g_tim1_owner));   // PWM (=1)
    USART_Print("\r\n");

    // ==== Step 2: พยายามใช้ SimpleTIM บน TIM1 → ต้องถูก block ====
    USART_Print("\r\n--- Attempting TIM_SimpleInit(TIM_1) ---\r\n");
    TIM_SimpleInit(TIM_1, 1);
    TIM_AttachInterrupt(TIM_1, my_tick);

    USART_Print("After init: g_tim1_owner = ");
    USART_Print(owner_name(g_tim1_owner));   // ยังเป็น PWM อยู่ (SimpleTIM silent)
    USART_Print("\r\n");
    USART_Print("    → SimpleTIM blocked, PWM still safe\r\n");

    // ==== Step 3: เปลี่ยนไปใช้ TIM2 แทน ====
    USART_Print("\r\n--- Try TIM_2 instead ---\r\n");
    if (g_tim2_owner == TIM_OWNER_NONE) {
        TIM_SimpleInit(TIM_2, 2);             // 2 Hz
        TIM_AttachInterrupt(TIM_2, my_tick);
        TIM_Start(TIM_2);
        USART_Print("[SimpleTIM] TIM2 OK   g_tim2_owner = ");
        USART_Print(owner_name(g_tim2_owner)); // TIMER (=2)
        USART_Print("\r\n");
    } else {
        USART_Print("TIM2 ไม่ว่าง ให้ปล่อยก่อน\r\n");
    }

    // ==== Step 4: สรุปสถานะทั้งระบบ ====
    USART_Print("\r\n--- Timer Status ---\r\n");
    USART_Print("TIM1: "); USART_Print(owner_name(g_tim1_owner)); USART_Print("\r\n");
    USART_Print("TIM2: "); USART_Print(owner_name(g_tim2_owner)); USART_Print("\r\n");

    while (1) { Delay_Ms(1000); }
}
🧠 แนวคิด: โมดูลทุกตัวเช็ค g_timX_owner ก่อน init — ถ้าค่า != TIM_OWNER_NONE และไม่ใช่ชื่อตัวเอง → return; เงียบ (SimpleTIM.c:119, SimplePWM.c:218) — วิธีตรวจจับ: อ่าน g_tim1_owner โดยตรงก่อน/หลัง init — ถ้าหลัง init ค่ายังเป็นเจ้าของเดิม แปลว่าโมดูลใหม่ถูก block — วิธีแก้: เลือก timer คนละตัว (TIM1 กับ TIM2) หรือปล่อย timer ที่จองอยู่ก่อน — CH32V003 มีแค่ TIM1 + TIM2 ให้บริหาร ส่วน SysTick เป็นของ SimpleDelay

ข้อ 2: ใช้ SimpleTIM และ Stopwatch พร้อมกันอย่างปลอดภัย

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"

static volatile uint8_t g_led_state = 0;

void blink_tick(void) {
    g_led_state ^= 1;
    digitalToggle(PC0);   // LED blink 1 Hz
}

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
    pinMode(PC0, PIN_MODE_OUTPUT);

    USART_Print("=== SimpleTIM (TIM1) + Stopwatch (TIM2) ===\r\n\r\n");

    // ==== TIM1: SimpleTIM 1 Hz blink LED ====
    TIM_SimpleInit(TIM_1, 1);                  // 1 Hz → callback ทุก 1s
    TIM_AttachInterrupt(TIM_1, blink_tick);
    TIM_Start(TIM_1);
    USART_Print("[SimpleTIM] TIM1  g_tim1_owner = ");
    USART_PrintNum(g_tim1_owner);               // 2 = TIM_OWNER_TIMER
    USART_Print("\r\n");

    // ==== TIM2: Stopwatch จับ uptime (default TIMEXT_DEFAULT_TIMER = TIM_2) ====
    Stopwatch_Init();                          // ภายในเรียก TIM_SimpleInit(TIM_2, 1000)
    Stopwatch_Start();
    USART_Print("[Stopwatch] TIM2  g_tim2_owner = ");
    USART_PrintNum(g_tim2_owner);               // 2 = TIM_OWNER_TIMER (จองผ่าน SimpleTIM)
    USART_Print("\r\n");

    // เช็คว่า Stopwatch จองจริง (ถ้า TIM2 ถูก SimpleTIM/PWM จองก่อน → Stopwatch จะ silent)
    if (g_tim2_owner != TIM_OWNER_TIMER) {
        USART_Print("⚠ Stopwatch จอง TIM2 ไม่สำเร็จ (ถูกใช้อยู่)\r\n");
    }

    char buf[TIME_BUFFER_SIZE_HHMMSS];
    uint32_t last_change = 0;

    while (1) {
        // พิมพ์ uptime ทุก ~2 วินาที (ใช้ Delay_Ms บน SysTick ไม่แย่ง hardware timer)
        Stopwatch_GetTimeString(buf, TIME_FORMAT_HHMMSS, TIME_DISPLAY_NORMALIZED);
        USART_Print("uptime: "); USART_Print(buf);
        USART_Print("  LED=");    USART_PrintNum(g_led_state);
        USART_Print("\r\n");

        Delay_Ms(2000);

        // หลัง ~10 วินาที เปลี่ยนความถี่ SimpleTIM เป็น 2 Hz (toggle เร็วขึ้น)
        // TIM_SetFrequency เช็ค owner ก่อน เนื่องจาก owner เป็น TIMER (ตัวเอง) → ผ่าน
        last_change += 2;
        if (last_change == 10) {
            TIM_SetFrequency(TIM_1, 2);
            USART_Print(">> SimpleTIM TIM1 frequency changed to 2 Hz\r\n");
        }
    }
}
🧠 แนวคิด: กำหนด SimpleTIM ให้ใช้ TIM_1 (LED blink ผ่าน interrupt) และให้ Stopwatch ใช้ TIM_2 (default ของ TIMEXT_DEFAULT_TIMER) — คนละ timer จึงไม่มี conflict — จุดสำคัญ: Stopwatch_Init() ภายในเรียก TIM_SimpleInit(TIM_2, 1000) จึงจอง TIM2 ผ่านกลไกเดียวกับ SimpleTIM → g_tim2_owner กลายเป็น TIM_OWNER_TIMER ไม่ใช่ TIM_OWNER_TIMEXT — ตรวจสอบว่า Stopwatch จองสำเร็จ: ดู if (g_tim2_owner != TIM_OWNER_TIMER) — เปลี่ยนความถี่ SimpleTIM ภายหลัง: TIM_SetFrequency() เช็ค owner ก่อน และเนื่องจาก owner ปัจจุบันเป็น TIM_OWNER_TIMER (ตัวเอง) จึงผ่านได้ — ส่วน Delay ในลูปใช้ Delay_Ms บน SysTick ที่ SimpleDelay ครอบครองอยู่ ไม่แย่ง hardware timer เลย

ข้อ 3: diagnostic ตอน PWM และ SimpleTIM ชนกัน และวิธีปล่อย timer

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"

void tick_a(void) { }
void tick_b(void) { }

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);

    USART_Print("=== Diagnostic: PWM vs SimpleTIM on TIM1 ===\r\n\r\n");

    // ==== A. เปิด PWM บน TIM1 (PWM1_CH4 = PC4) ====
    PWM_Init(PWM1_CH4, 1000);
    PWM_Start(PWM1_CH4);
    USART_Print("A. PWM on TIM1      g_tim1_owner = ");
    USART_PrintNum(g_tim1_owner);               // 1 = TIM_OWNER_PWM
    USART_Print("\r\n");

    // ==== B. พยายาม SimpleTIM บน TIM1 → ต้อง silent ====
    TIM_SimpleInit(TIM_1, 5);
    TIM_AttachInterrupt(TIM_1, tick_a);
    USART_Print("B. after SimpleInit  g_tim1_owner = ");
    USART_PrintNum(g_tim1_owner);               // ยัง 1 (SimpleTIM ถูก block)
    USART_Print("  ← SimpleTIM blocked\r\n");

    // ==== C. ปล่อย TIM1 จาก PWM: Stop + ตั้งค่า global เอง (ไม่มี PWM Release) ====
    PWM_Stop(PWM1_CH4);
    g_tim1_owner = TIM_OWNER_NONE;              // คืนกรรมสิทธิ์ด้วยมือ
    USART_Print("C. manual release   g_tim1_owner = ");
    USART_PrintNum(g_tim1_owner);               // 0 = TIM_OWNER_NONE
    USART_Print("\r\n");

    // ==== D. SimpleTIM บน TIM1 คราวนี้ผ่าน ====
    TIM_SimpleInit(TIM_1, 5);
    TIM_AttachInterrupt(TIM_1, tick_b);
    TIM_Start(TIM_1);
    USART_Print("D. SimpleTIM TIM1   g_tim1_owner = ");
    USART_PrintNum(g_tim1_owner);               // 2 = TIM_OWNER_TIMER
    USART_Print("  ← SimpleTIM OK now\r\n");

    // ==== E. เปรียบเทียบ: TIM_DetachInterrupt คืน owner อัตโนมัติ (สำหรับ SimpleTIM) ====
    TIM_DetachInterrupt(TIM_1);
    USART_Print("E. after detach     g_tim1_owner = ");
    USART_PrintNum(g_tim1_owner);               // 0 = TIM_OWNER_NONE (คืนอัตโนมัติ)
    USART_Print("  ← auto release by TIM_DetachInterrupt\r\n");

    USART_Print("\r\nสรุป:\r\n");
    USART_Print(" - PWM: ต้อง manual release (PWM_SetOwner ไม่มี)\r\n");
    USART_Print(" - SimpleTIM: TIM_DetachInterrupt คืน owner ให้อัตโนมัติ\r\n");

    while (1) { Delay_Ms(1000); }
}
🧠 แนวคิด: การปล่อย timer มี 2 วิธีตามโมดูลที่จอง — กรณี PWM จอง: เพราะ SimplePWM ไม่มี API ปล่อย owner (ไม่มี PWM_Release) จึงต้อง PWM_Stop(channel) แล้วเขียน g_tim1_owner = TIM_OWNER_NONE ด้วยมือ — กรณี SimpleTIM จอง: TIM_DetachInterrupt(timer) จะตั้ง *owner = TIM_OWNER_NONE ให้อัตโนมัติถ้าปัจจุบันเป็น TIM_OWNER_TIMER (SimpleTIM.c:280-282) — การวินิจฉัย conflict: อ่าน g_timX_owner ก่อนและหลัง init — ถ้าหลัง init ค่ายังเป็นเจ้าของเดิม → โมดูลใหม่ถูก block — ข้อควรจำ: ทุกการ block เป็น silent return ไม่มี error code คืน จึงต้องพึ่งการอ่าน global ตรงๆ เสมอ
