📝 เฉลย — PCF8574 IO Expander

ข้อ 1: Keypad 4x4 ผ่าน I2C

PCF8574_Instance pcf;
#define PCF_ADDR  0x20

const char keymap[4][4] = {
    {'1','2','3','A'},
    {'4','5','6','B'},
    {'7','8','9','C'},
    {'*','0','#','D'}
};

char pcf_keypad_read(void) {
    for(uint8_t row=0; row<4; row++) {
        uint8_t row_mask = ~(1 << (row + 4)) & 0xF0;
        PCF8574_WritePort(&pcf, row_mask | 0x0F);
        Delay_Ms(1);

        uint8_t val = PCF8574_ReadPort(&pcf) & 0x0F;
        if(val != 0x0F) {
            for(uint8_t col=0; col<4; col++) {
                if(!(val & (1 << col))) {
                    Delay_Ms(10);
                    uint8_t v2 = PCF8574_ReadPort(&pcf) & 0x0F;
                    if(!(v2 & (1 << col)))
                        return keymap[row][col];
                }
            }
        }
    }
    return 0;
}

void pcf8574_keypad(void) {
    PCF8574_Init(&pcf, PCF_ADDR);
    char last = 0;

    while(1) {
        char k = pcf_keypad_read();
        if(k != 0 && k != last) {
            USART_Print("Key: ");
            USART_Write(k);
            USART_Println("");
            last = k;
        }
        if(k == 0) last = 0;
        Delay_Ms(30);
    }
}
🧠 แนวคิด: PCF8574 เพิ่ม GPIO 8 ขาผ่าน I2C — Keypad 4x4 ใช้ 4 rows (P4-P7 output) + 4 cols (P0-P3 input) — ส่ง row mask ดึง 1 row LOW แล้วอ่าน column — ถ้า column ไม่เป็น HIGH แสดงว่ามีกดที่ row x col — debounce 2 ชั้น (delay 1ms + re-read) — ใช้ I2C แค่ 2 ขาแทน 8 ขา GPIO — PCF8574 มีขา INT แจ้งเมื่อ input เปลี่ยน

ข้อ 3: Output Expansion — ต่อ LED 8 ดวง + Relay ผ่าน I2C

PCF8574_Instance pcf1, pcf2;
#define PCF1_ADDR  0x20
#define PCF2_ADDR  0x21

typedef struct {
    uint8_t leds;
    uint8_t relay;
} pcf_outputs_t;

pcf_outputs_t pcf_out;

void pcf8574_output_write(void) {
    PCF8574_WritePort(&pcf1, ~pcf_out.leds);
    PCF8574_WritePort(&pcf2, ~pcf_out.relay);
}

void pcf8574_relay_control(uint8_t ch, uint8_t on) {
    if(ch >= 8) return;
    if(on) pcf_out.relay |=  (1 << ch);
    else   pcf_out.relay &= ~(1 << ch);
    pcf8574_output_write();
}

void pcf8574_expander_demo(void) {
    PCF8574_Init(&pcf1, PCF1_ADDR);
    PCF8574_Init(&pcf2, PCF2_ADDR);
    pcf_out.leds = 0;
    pcf_out.relay = 0;
    pcf8574_output_write();

    uint8_t pattern = 0x01;

    while(1) {
        // LED chaser 8 ดวง (PCF1)
        pcf_out.leds = pattern;
        pcf8574_output_write();
        pattern = (pattern << 1) | (pattern >> 7);
        Delay_Ms(100);

        // Relay toggle 1-8 (PCF2)
        static uint8_t r = 0;
        pcf8574_relay_control(r, 1);
        Delay_Ms(500);
        pcf8574_relay_control(r, 0);
        r = (r + 1) % 8;
    }
}

// หมายเหตุ: PCF8574 sink current ~25mA ต่อขา
// LED ต้องต่อผ่าน resistor, Relay ต้องใช้ transistor driver
// PCF8574 output = inverted (เขียน 0 = ON)
🧠 แนวคิด: ใช้ PCF8574 2 ตัว (address 0x20, 0x21) ขยาย I/O — ตัวแรก 8 LEDs, ตัวสอง 8 relays — PCF8574 output เป็น open-drain แบบ invert: เขียน 0 = ON, 1 = OFF — struct pcf_out เก็บ state แล้ว write ทั้ง byte — relay_control set/clear เฉพาะบิต — LED chaser 8 ดวงวนขวา — PCF8574 แต่ละขารับกระแส ~25mA, ทั้งชิป ~100mA
