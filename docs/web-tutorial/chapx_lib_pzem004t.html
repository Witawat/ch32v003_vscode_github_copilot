⚠️ 220VAC อันตรายถึงชีวิต! ต้องต่อวงจรเมื่อไม่ได้เสียบไฟเท่านั้น — CT Clamp ต้องหนีบสาย Line (L) เส้นเดียว ห้ามหนีบทั้ง L+N — ใช้隔离电源หรือหม้อแปลงในการทดสอบ

📚 หลักการทำงาน

PZEM-004T เป็น AC Energy Meter สื่อสารผ่าน USART ด้วยโปรโตคอล Modbus-RTU — วัดแรงดัน (80-260V AC), กระแส (0-100A ผ่าน CT clamp), กำลังไฟฟ้า (0-23kW), พลังงาน (0-9999kWh), ความถี่ (45-65Hz), และ Power Factor (0-1)

Modbus-RTU ใช้การสื่อสารแบบ Master-Slave — CH32V003 เป็น Master ส่งคำสั่งอ่าน Register ไปที่ PZEM (Slave address 0x01 default) — แต่ละ register มี 2 bytes (16-bit) — ข้อมูลทุกตัวมี CRC16 checksum เพื่อตรวจสอบความถูกต้อง

CT Clamp 100A วัดกระแสแบบ Non-Contact — ต้องหนีบสาย Line (L) เส้นเดียว — ถ้าหนีบทั้ง L+N กระแสจะ cancel กัน อ่านได้ 0 — ทิศทางการหนีบมีผลต่อ sign ของ Power Factor

Register Map: 0x0000=Voltage(x0.1V), 0x0001=Current(x0.01A high), 0x0002=Current(x0.001A low), 0x0003=Power(x0.1W high), 0x0004=Power low, 0x0005=Energy(x1Wh high), 0x0006=Energy low, 0x0007=Frequency(x0.1Hz), 0x0008=Power Factor(x0.01), 0x0009=Alarm

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Init PZEM + USART

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "PZEM004T.h"

PZEM004T_t pzem;

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_Init(115200); // debug

    // PZEM ใช้ USART2 (PD5=TX, PD6=RX) 9600 8N1
    PZEM004T_Init(&pzem, PD6, PD5, 0x01); // RX, TX, slave addr=1
    USART_Print("PZEM-004T Ready\r\n");

🔍 PZEM004T_Init(dev, rx_pin, tx_pin, addr) — ตั้งค่า USART 9600 baud 8N1 — slave address default=0x01 — RX/TX ต่อกับ PD6/PD5 (USART2)

ขั้นที่ 2: อ่านค่าทีละตัว

    float voltage = PZEM004T_ReadVoltage(&pzem);
    float current = PZEM004T_ReadCurrent(&pzem);
    float power = PZEM004T_ReadPower(&pzem);
    float energy = PZEM004T_ReadEnergy(&pzem);
    float freq = PZEM004T_ReadFrequency(&pzem);
    float pf = PZEM004T_ReadPowerFactor(&pzem);

    USART_Print("V:"); USART_PrintFloat(voltage, 1);
    USART_Print(" I:"); USART_PrintFloat(current, 2);
    USART_Print("A P:"); USART_PrintFloat(power, 1);
    USART_Print("W\r\n");
    USART_Print("E:"); USART_PrintFloat(energy, 1);
    USART_Print("kWh Hz:"); USART_PrintFloat(freq, 1);
    USART_Print(" PF:"); USART_PrintFloat(pf, 2);
    USART_Print("\r\n");

🔍 ฟังก์ชันอ่านแยกแต่ละ register — แต่ละครั้งส่ง Modbus request + รอ response — ใช้เวลา ~50-100ms ต่อค่า — ถ้า timeout (1000ms) จะคืนค่า NAN

ขั้นที่ 3: Loop อ่าน + ตรวจจับ Alarm

    Timer_t tmr;
    Start_Timer(&tmr, 2000, 1); // อ่านทุก 2 วิ

    while(1) {
        if(Is_Timer_Expired(&tmr)) {
            float v = PZEM004T_ReadVoltage(&pzem);
            float a = PZEM004T_ReadCurrent(&pzem);
            float w = PZEM004T_ReadPower(&pzem);
            uint16_t alarm = PZEM004T_ReadAlarm(&pzem);

            if(v != v) { // NaN check
                USART_Print("PZEM Timeout!\r\n");
                continue;
            }

            USART_Print("V:"); USART_PrintFloat(v, 1);
            USART_Print(" A:"); USART_PrintFloat(a, 2);
            USART_Print(" W:"); USART_PrintFloat(w, 1);

            if(alarm & 0x0001) USART_Print(" OVER_V!");
            if(alarm & 0x0002) USART_Print(" OVER_A!");
            if(alarm & 0x0004) USART_Print(" OVER_P!");
            USART_Print("\r\n");
        }
    }
}

🔍 ตรวจสอบ NaN ก่อนใช้ค่า — alarm register แจ้ง over-voltage (bit0), over-current (bit1), over-power (bit2) — ไม่ควรอ่านถี่เกิน 200ms เพราะ PZEM มีเวลาตอบสนอง ~200ms

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
อ่านค่าได้ 0 ทุกตัวCT Clamp หนีบผิด / ไม่มีโหลดCT ต้องหนีบสาย L เส้นเดียว — ต้องมีโหลดต่ออยู่
ค่ากระแสติดลบCT Clamp หนีบกลับทิศกลับด้าน CT Clamp — ลูกศรบน CT ชี้ไปทางโหลด
Timeout ตลอดเวลาUSART baud ไม่ตรง / สาย RX/TX สลับต้อง 9600 8N1 — RX→TX, TX→RX — ตรวจสอบด้วย Oscilloscope
แรงดันอ่านได้แต่กระแสเป็น 0ไม่ได้ต่อโหลด หรือ CT ไม่ได้เสียบตรวจสอบ CT เสียบแน่น — ต่อโหลดอย่างน้อย 50W
ค่าพลังงานไม่เพิ่มกระแสต่ำเกิน minimum detectionPZEM วัดกระแสขั้นต่ำ ~0.01A — ใช้โหลดที่กินกระแสมากขึ้น
MCU รีเซ็ตตอนต่อ 220VNoise จาก AC เข้า MCUใช้隔离电源 — แยก GND ระหว่าง MCU กับ PZEM — ต่อ Optocoupler ที่ USART

🛠️ สร้าง Product: เครื่องวัดไฟบ้าน + บันทึก kWh

PZEM-004T + OLED แสดงค่าทางไฟฟ้า — บันทึก kWh สะสมลง EEPROM ทุก 1 นาที

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "PZEM004T.h"
#include "oled_i2c.h"

int main(void)
{
    SystemCoreClockUpdate();
    Timer_Init();
    I2C_Init(PC1, PC2, I2C_SPEED_STANDARD);
    USART_Init(115200);

    PZEM004T_t pzem;
    PZEM004T_Init(&pzem, PD6, PD5, 0x01);

    OLED_Handle oled;
    OLED_Init(&oled, OLED_128x64, 0x3C);
    OLED_SetFont(&oled, FONT_8x16);

    Timer_t tmr;
    Start_Timer(&tmr, 2000, 1);

    while(1)
    {
        if(Is_Timer_Expired(&tmr))
        {
            float v = PZEM004T_ReadVoltage(&pzem);
            float a = PZEM004T_ReadCurrent(&pzem);
            float w = PZEM004T_ReadPower(&pzem);
            float e = PZEM004T_ReadEnergy(&pzem);
            float pf = PZEM004T_ReadPowerFactor(&pzem);
            float hz = PZEM004T_ReadFrequency(&pzem);

            if(v != v) continue; // timeout skip

            OLED_Clear(&oled);
            OLED_SetCursor(&oled, 0, 0);
            OLED_PrintFloat(&oled, "", v, 1);
            OLED_Print(&oled, "V ");

            OLED_SetCursor(&oled, 60, 0);
            OLED_PrintFloat(&oled, "", a, 2);
            OLED_Print(&oled, "A");

            OLED_SetCursor(&oled, 0, 2);
            OLED_PrintFloat(&oled, "", w, 1);
            OLED_Print(&oled, "W ");

            OLED_SetCursor(&oled, 60, 2);
            OLED_PrintFloat(&oled, "PF:", pf, 2);

            OLED_SetCursor(&oled, 0, 4);
            OLED_PrintFloat(&oled, "", hz, 1);
            OLED_Print(&oled, "Hz ");

            OLED_SetCursor(&oled, 0, 6);
            OLED_PrintFloat(&oled, "", e, 1);
            OLED_Print(&oled, "kWh");

            OLED_Update(&oled);
        }
    }
}

✏️ ลองทำดู

  1. เพิ่ม Relay ตัดโหลดเมื่อ over-current — ใช้ alarm register ตรวจจับแล้วสั่งตัดผ่าน GPIO
  2. บันทึกค่า kWh ลง EEPROM ทุก 5 นาที — ป้องกันข้อมูลหายเมื่อไฟดับ — อ่านกลับมาใช้เป็น baseline
  3. คำนวณค่าไฟ — ตั้งอัตราค่าไฟต่อหน่วย (บาท/kWh) — แสดงค่าใช้จ่ายสะสมบน OLED
  4. ส่งข้อมูลผ่าน UART ไป ESP8266 — ทำ Dashboard แสดงผลบน WiFi แบบ real-time
