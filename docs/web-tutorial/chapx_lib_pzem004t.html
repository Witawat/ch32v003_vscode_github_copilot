PZEM-004T — วัดไฟ AC ผ่าน Modbus-RTU

ระดับสูง

🧭 แผนภาพการทำงาน

flowchart TD
    A["PZEM004T_Init(&pzem, USART_PINS_DEFAULT)"] --> B["ส่ง Modbus Request"]
    B --> C["รอ Response 7-13 bytes"]
    C --> D{"CRC16 OK?"}
    D -->|"Yes"| E["PZEM004T_ReadAll(&pzem)"]
    D -->|"No"| F["Timeout / Retry"]
    E --> G["V, A, W, Wh, Hz, PF"]
    F --> B
    G --> H["แสดงผล / Log"]
    H --> B

🔌 การต่อวงจร

   ⚡ WARNING: 220VAC! อันตรายถึงชีวิต!
    ┌────────────────────────────────────────────┐
    │  MCU              PZEM-004T                │
    │  PD5─TX ────────── RX   VCC──5V            │
    │  PD6─RX ────────── TX   GND──GND           │
    │  CT Clamp ──────── CT ──|Load|── L(220V)   │
    └────────────────────────────────────────────┘
MCU Pin→อุปกรณ์ขาหมายเหตุ
PD5 (TX)→PZEM-004TRXUSART ส่ง Modbus request
PD6 (RX)←PZEM-004TTXUSART รับ Modbus response
5V→PZEM-004TVCCจ่ายไฟ 5V
GND→PZEM-004TGNDGround ร่วม

รู้จักขา

  • RX: ขารับคำสั่ง Modbus จาก MCU (TX)
  • TX: ขาส่งข้อมูล Modbus ไป MCU (RX)
  • VCC: จ่ายไฟ 5V
  • GND: Ground ร่วม
  • CT: Clamp ammeter — หนีบสาย L เส้นเดียว

วิธีต่อทีละขั้น

  1. ต่อ RX ของ PZEM ไป PD5 (TX) ของ MCU — ส่ง Modbus request
  2. ต่อ TX ของ PZEM ไป PD6 (RX) ของ MCU — รับข้อมูล
  3. ต่อ VCC ไป 5V และ GND ไป GND — จ่ายไฟ
  4. หนีบ CT Clamp รอบ สาย L เส้นเดียว — วัดกระแสโหลด

📚 หลักการทำงาน

PZEM-004T เป็น AC Energy Meter สื่อสารผ่าน USART ด้วย Modbus-RTU — PZEM004T_ReadAll() อ่านทุกค่าพร้อมกันในครั้งเดียว

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Init PZEM + USART

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "PZEM004T.h"

PZEM004T pzem;

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_Init(115200); // debug

    PZEM004T_Init(&pzem, USART_PINS_DEFAULT);
    USART_Print("PZEM-004T Ready\r\n");

🔍 PZEM004T_Init(&pzem, pins) — instance-based, ใช้ USART_PINS_DEFAULT — baud 9600 auto

ขั้นที่ 2: อ่านค่าทั้งหมดในครั้งเดียว

    if(PZEM004T_ReadAll(&pzem)) {
        USART_Print("V:"); USART_PrintFloat(pzem.voltage, 1);
        USART_Print(" I:"); USART_PrintFloat(pzem.current, 2);
        USART_Print("A P:"); USART_PrintFloat(pzem.power, 1);
        USART_Print("W\r\n");
        USART_Print("E:"); USART_PrintFloat(pzem.energy, 1);
        USART_Print("kWh Hz:"); USART_PrintFloat(pzem.freq, 1);
        USART_Print(" PF:"); USART_PrintFloat(pzem.pf, 2);
        USART_Print("\r\n");
    }

🔍 PZEM004T_ReadAll(&pzem) — อ่านทุก register ในครั้งเดียว — ข้อมูลเก็บใน fields: voltage, current, power, energy, freq, pf

ขั้นที่ 3: Loop อ่าน

    Timer_t tmr;
    Start_Timer(&tmr, 2000, 1);

    while(1) {
        if(Is_Timer_Expired(&tmr)) {
            if(PZEM004T_ReadAll(&pzem)) {
                USART_Print("V:"); USART_PrintFloat(pzem.voltage, 1);
                USART_Print(" A:"); USART_PrintFloat(pzem.current, 2);
                USART_Print(" W:"); USART_PrintFloat(pzem.power, 1);
                USART_Print("\r\n");
            } else {
                USART_Print("PZEM Timeout!\r\n");
            }
        }
    }
}

🔍 อ่านทุก 2 วิ — ReadAll คืน TRUE ถ้า CRC OK — timeout จะคืน FALSE

📋 API Reference

ฟังก์ชันคำอธิบายตัวอย่าง
PZEM004T_Init(&pzem, pins)เริ่มต้น PZEMPZEM004T_Init(&pzem, USART_PINS_DEFAULT)
PZEM004T_ReadAll(&pzem)อ่านทุกค่าในครั้งเดียวif(PZEM004T_ReadAll(&pzem))
PZEM004T_ReadVoltage(&pzem)อ่านแรงดัน AC (V) แยกfloat v = PZEM004T_ReadVoltage(&pzem)
PZEM004T_ReadCurrent(&pzem)อ่านกระแส (A) แยกfloat a = PZEM004T_ReadCurrent(&pzem)
PZEM004T_ReadPower(&pzem)อ่านกำลังไฟฟ้า (W) แยกfloat w = PZEM004T_ReadPower(&pzem)

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
อ่านค่าได้ 0 ทุกตัวCT Clamp หนีบผิด / ไม่มีโหลดCT ต้องหนีบสาย L เส้นเดียว
Timeout ตลอดเวลาUSART baud ไม่ตรงต้อง 9600 8N1 — RX→TX, TX→RX

🛠️ สร้าง Product: เครื่องวัดไฟบ้าน

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "PZEM004T.h"
#include "oled_i2c.h"

int main(void)
{
    SystemCoreClockUpdate();
    Timer_Init();
    I2C_Init(PC1, PC2, I2C_SPEED_STANDARD);
    USART_Init(115200);

    PZEM004T pzem;
    PZEM004T_Init(&pzem, USART_PINS_DEFAULT);

    OLED_Handle oled;
    OLED_Init(&oled, OLED_128x64, 0x3C);

    Timer_t tmr;
    Start_Timer(&tmr, 2000, 1);

    while(1)
    {
        if(Is_Timer_Expired(&tmr))
        {
            if(PZEM004T_ReadAll(&pzem))
            {
                OLED_Clear(&oled);
                char buf[64];
                snprintf(buf, sizeof(buf), "%.1fV ", pzem.voltage);
                OLED_DrawStringAlign(&oled, 0, 0, buf, ALIGN_LEFT);
                snprintf(buf, sizeof(buf), "%.2fA", pzem.current);
                OLED_DrawStringAlign(&oled, 60, 0, buf, ALIGN_LEFT);
                snprintf(buf, sizeof(buf), "%.1fW ", pzem.power);
                OLED_DrawStringAlign(&oled, 0, 20, buf, ALIGN_LEFT);
                snprintf(buf, sizeof(buf), "PF: %.2f", pzem.pf);
                OLED_DrawStringAlign(&oled, 60, 20, buf, ALIGN_LEFT);
                snprintf(buf, sizeof(buf), "%.1fHz ", pzem.freq);
                OLED_DrawStringAlign(&oled, 0, 40, buf, ALIGN_LEFT);
                snprintf(buf, sizeof(buf), "%.1fkWh", pzem.energy);
                OLED_DrawStringAlign(&oled, 0, 60, buf, ALIGN_LEFT);
                OLED_Update(&oled);
            }
        }
    }
}
