📝 เฉลย — HC-SR04 เซนเซอร์อัลตราโซนิก

ข้อ 1: Parking Sensor — LED เขียว/เหลือง/แดง

#define TRIG_PIN    PC0
#define ECHO_PIN    PC1
#define LED_GREEN   PC4
#define LED_YELLOW  PC5
#define LED_RED     PC3
#define BUZZER      PC6
#define ZONE_SAFE   100.0f  // cm
#define ZONE_WARN   50.0f
#define ZONE_STOP   20.0f

void hcsr04_parking_sensor(void) {
    HCSR04_Instance s;
    HCSR04_Init(&s, TRIG_PIN, ECHO_PIN);
    pinMode(LED_GREEN, PIN_MODE_OUTPUT);
    pinMode(LED_YELLOW, PIN_MODE_OUTPUT);
    pinMode(LED_RED, PIN_MODE_OUTPUT);
    pinMode(BUZZER, PIN_MODE_OUTPUT);

    while(1) {
        float dist = HCSR04_MeasureCm(&s);

        if(dist < 0 || dist > 400) {
            // No valid reading
            digitalWrite(LED_GREEN, LOW);
            digitalWrite(LED_YELLOW, LOW);
            digitalWrite(LED_RED, HIGH);
            Delay_Ms(200);
            digitalWrite(LED_RED, LOW);
            Delay_Ms(200);
            continue;
        }

        digitalWrite(LED_GREEN, dist > ZONE_SAFE);
        digitalWrite(LED_YELLOW, dist <= ZONE_SAFE && dist > ZONE_WARN);
        digitalWrite(LED_RED, dist <= ZONE_WARN);

        if(dist <= ZONE_WARN) {
            uint16_t beep_delay = (uint16_t)(dist * 5);  // ยิ่งใกล้ ยิ่งถี่
            digitalWrite(BUZZER, HIGH); Delay_Ms(50);
            digitalWrite(BUZZER, LOW);  Delay_Ms(beep_delay);
        } else {
            digitalWrite(BUZZER, LOW);
        }
        Delay_Ms(60);
    }
}
🧠 แนวคิด: 3 โซน: SAFE (>100cm เขียว) / WARN (50–100cm เหลือง) / STOP (<50cm แดง) — Buzzer กระพริบถี่ขึ้นเมื่อใกล้ — ใช้ค่า dist map เป็น delay: dist×5 ms — กรอง invalid reading (dist < 0 หรือ > 4m) — HCSR04_MeasureCm คืน cm โดยตรง

ข้อ 2: Average Multiple Readings Filter

#define FILTER_SAMPLES 8

float hcsr04_read_filtered(HCSR04_Instance *s) {
    float readings[FILTER_SAMPLES];
    int count = 0;

    // เก็บตัวอย่าง
    for(int i = 0; i < FILTER_SAMPLES; i++) {
        float d = HCSR04_MeasureCm(s);
        if(d > 0 && d <= 400) {
            readings[count++] = d;
        }
        Delay_Ms(30);
    }

    if(count == 0) return -1;  // ไม่มีค่าที่ใช้ได้เลย

    // Bubble sort
    for(int i = 0; i < count - 1; i++) {
        for(int j = 0; j < count - i - 1; j++) {
            if(readings[j] > readings[j + 1]) {
                float tmp = readings[j];
                readings[j] = readings[j + 1];
                readings[j + 1] = tmp;
            }
        }
    }

    // Median filter — ตัดหัวตัดท้าย แล้วเฉลี่ยกลาง
    int start = count / 4;
    int end   = count - count / 4;
    float sum = 0;
    int valid = 0;
    for(int i = start; i < end; i++) {
        sum += readings[i];
        valid++;
    }

    return sum / valid;
}
🧠 แนวคิด: เก็บ 8 ตัวอย่าง — median filter: sort แล้วตัดหัว-ท้าย 25% (กรอง outlier) — เฉลี่ยค่าที่เหลือ — ดีกว่า simple average เพราะ spike จาก echo ข้างเคียงจะถูกตัดทิ้ง — เพิ่ม 30ms ระหว่างอ่านให้คลื่นจางก่อนยิงใหม่

ข้อ 3: เปรียบเทียบ HC-SR04 vs VL53L0X

void ultrasonic_vs_laser(void) {
    HCSR04_Instance us;
    HCSR04_Init(&us, PC0, PC1);

    VL53L0X_Instance tof;
    VL53L0X_Init(&tof, 0x29);

    OLED_Handle oled;
    OLED_Init(&oled, OLED_128x64, 0x3C);

    char buf[64];
    while(1) {
        float dist_us = HCSR04_MeasureCm(&us);
        uint16_t dist_lx = VL53L0X_ReadRangeMM(&tof);

        snprintf(buf, sizeof(buf), "US: %.1f cm  ", dist_us);
        OLED_DrawStringAlign(&oled, 0, 0, buf, ALIGN_LEFT);

        snprintf(buf, sizeof(buf), "LX: %d mm  ", dist_lx);
        OLED_DrawStringAlign(&oled, 0, 10, buf, ALIGN_LEFT);

        // Comment on accuracy
        if(dist_us < 10 && dist_lx < 100) {
            snprintf(buf, sizeof(buf), "Near \xE2\x80\x94 laser wins ");
        } else if(dist_us > 200) {
            snprintf(buf, sizeof(buf), "Far  \xE2\x80\x94 US wins    ");
        } else {
            snprintf(buf, sizeof(buf), "Mid  \xE2\x80\x94 both OK    ");
        }
        OLED_DrawStringAlign(&oled, 0, 20, buf, ALIGN_LEFT);

        Delay_Ms(500);
    }
}
🧠 แนวคิด: HC-SR04: range 2–400cm, blind zone ~0–2cm, beam กว้าง (~15°) — VL53L0X: range 0–200cm (ขึ้นกับแสง), beam แคบ (~25° FoV) — ระยะใกล้: laser แม่นกว่า เพราะ US มี blind zone — ระยะไกล >2m: US วัดได้ laser อาจไม่ถึง — สรุป: ใช้คู่กันเสริมจุดอ่อน
