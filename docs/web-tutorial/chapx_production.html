📚 2. Read-Out Protection (ROP) — ล็อกเฟิร์มแวร์

ป้องกันไม่ให้ใครอ่านเฟิร์มแวร์จากชิปผ่าน WCH-Link — CH32V003 ROP เป็น binary (ENABLE/DISABLE):

ระดับอ่าน Flashเขียน FlashDebug (GDB)ใช้ตอน
ROP=DISABLE (default)✅ ได้✅ ได้✅ ได้Development
ROP=ENABLE❌ ไม่ได้❌ ไม่ได้❌ ไม่ได้Production
#include "ch32v00x_flash.h"

void enable_readout_protection(void) {
    FLASH_Unlock();
    FLASH_EraseOptionBytes();
    FLASH_ReadOutProtection(ENABLE);   // ล็อกชิป อ่าน Flash ไม่ได้
    FLASH_Lock();
    // ต้อง reset ถึงจะมีผล
    NVIC_SystemReset();
}

void disable_readout_protection(void) {
    FLASH_Unlock();
    FLASH_EraseOptionBytes();
    FLASH_ReadOutProtection(DISABLE);  // ปลดล็อก (ล้าง Flash ทั้งหมด!)
    FLASH_Lock();
    NVIC_SystemReset();
}
⚠️ ข้อควรระวัง: การปลดล็อก ROP จะล้าง Flash ทั้งหมด — ใช้เมื่อต้องการ reprogram เท่านั้น — ฟังก์ชันมีเพียง FLASH_ReadOutProtection(ENABLE/DISABLE) แบบ binary ไม่มี multi-level

for(int i = 0; i < data_len; i++) {
    data[i] ^= xor_key;
}

// Anti-clone check
#define MY_CHIP_IDS  {0x12345678, 0x9ABCDEF0}
const uint32_t allowed[] = MY_CHIP_IDS;
uint8_t authorized = 0;
for(int i = 0; i < sizeof(allowed)/4; i++) {
    if(chip_id == allowed[i]) { authorized = 1; break; }
}
if(!authorized) while(1);  // หยุดถ้าไม่ใช่ชิปที่อนุญาต

📚 4. Brown-out Detection — ป้องกัน Flash เสีย

// PVD = Programmable Voltage Detector
// ตรวจจับไฟตกก่อนที่จะต่ำเกินไป (~2.4V)
PWR_PVDLevelConfig(PWR_PVDLevel_2V4);
PWR_PVDCmd(ENABLE);

// ใน ch32v00x_it.c:
void PVD_IRQHandler(void) {
    if(PWR_GetFlagStatus(PWR_FLAG_PVDO)) {
        // ไฟกำลังตก — save critical data แล้ว stop
        Flash_SaveConfig(&critical_config);
        PWR_ClearFlag(PWR_FLAG_PVDO);

        // รอจนกว่าไฟจะหายหรือกลับมาปกติ
        while(PWR_GetFlagStatus(PWR_FLAG_PVDO)) {
            __NOP();
        }
    }
}

// ตั้งค่าใน main():
NVIC_EnableIRQ(PVD_IRQn);

Test Firmware — ตรวจทุก peripheral ใน 1 โปรแกรม

void run_production_test(void) {
    USART_Println("=== PROD TEST START ===");

    // 1. GPIO
    digitalWrite(PC0, HIGH); Delay_Ms(100); digitalWrite(PC0, LOW);
    USART_Println("[PASS] GPIO");

    // 2. ADC — อ่าน VDD
    uint16_t vdd = ADC_GetVDD();
    if(vdd > 2800 && vdd < 3600) USART_Println("[PASS] ADC");
    else                         USART_Println("[FAIL] ADC");

    // 3. Flash — write + read
    FlashConfig_t test = { .key = 0xBEEF, .data = {42} };
    Flash_SaveConfig(&test);
    FlashConfig_t verify;
    if(Flash_LoadConfig(&verify) && verify.data[0] == 42)
        USART_Println("[PASS] Flash");
    else
        USART_Println("[FAIL] Flash");

    // 4. I2C — scan bus
    uint8_t found = 0;
    for(uint8_t a = 1; a < 127; a++) {
        I2C_Start(); I2C_WriteAddr(a << 1);
        if(I2C_WaitAck()) { found++; }
        I2C_Stop();
    }
    USART_Print("[INFO] I2C devices: ");
    USART_PrintNum(found);
    USART_Println("");

    // 5. PWM
    PWM_SimpleInit(1, PD2, 1000);
    PWM_Start(1);
    PWM_SetDuty(1, 50);
    USART_Println("[PASS] PWM (check with scope)");

    USART_Println("=== PROD TEST COMPLETE ===");
}

📚 6. Bootloader Concept

Bootloader ช่วยให้อัปเดตเฟิร์มแวร์ผ่าน USART/I2C/SPI โดยไม่ต้องใช้ WCH-Link — มีประโยชน์เมื่อสินค้าถึงมือลูกค้าแล้ว

โครงสร้าง Flash กับ Bootloader

    0x00000000 ┌──────────────────┐
               │ Bootloader (2KB) │ ← ไม่เปลี่ยน ป้องกัน brick
    0x00000800 ├──────────────────┤
               │ Application      │ ← เปลี่ยนได้ผ่าน bootloader
    0x00003C00 ├──────────────────┤
               │ Config (1KB)     │ ← เก็บ WiFi password, calib data
    0x00004000 └──────────────────┘

    Boot sequence:
    1. Power on → Bootloader รันก่อน
    2. เช็ค pin (เช่นปุ่มกดค้าง) — ต้องการอัปเดตเฟิร์มแวร์?
    3. ถ้าใช่ → รับเฟิร์มแวร์ผ่าน USART → เขียน Flash → jump
    4. ถ้าไม่ → jump to Application (0x00000800)

🛠️ Firmware Versioning

// ฝัง version ใน firmware — อ่านได้จาก hex หรือ USART
#define FW_VERSION_MAJOR  1
#define FW_VERSION_MINOR  0
#define FW_VERSION_PATCH  0
#define FW_BUILD_DATE     "2026-06-29"

const char __attribute__((section(".rodata"))) firmware_version[] =
    "v" STR(FW_VERSION_MAJOR) "." STR(FW_VERSION_MINOR) "." STR(FW_VERSION_PATCH)
    " (" FW_BUILD_DATE ")";

// STR macro
#define STR_(x) #x
#define STR(x)  STR_(x)

// เช็ค version ใน production test:
USART_Print("Firmware: ");
USART_Println(firmware_version);

✏️ ลองทำดู

  1. สร้าง release build — ปิดทุก print macro → เช็ค Flash/RAM ที่ประหยัดได้
  2. เขียน production test firmware — รันบนบอร์ด 3 บอร์ด → เทียบค่า ADC ว่าต่างกันไม่เกิน ±5%
  3. ตั้ง ROP — ทดสอบว่าปลดล็อกแล้วล้าง Flash แล้วแฟลชใหม่ได้ตามปกติ
  4. ออกแบบ bootloader แบบง่าย — รับเฟิร์มแวร์ผ่าน USART (binary) → เขียนที่ address 0x800 → jump
