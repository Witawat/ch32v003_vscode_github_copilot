NTC10K — วัดอุณหภูมิด้วย Thermistor ADC

ระดับกลาง

🧭 แผนภาพการทำงาน

flowchart TD
    A["NTC_Init(ADC_CH_PC4)"] --> B["NTC_ReadTemperature()"]
    B --> C["Steinhart-Hart / Beta Equation"]
    C --> D["Temperature (°C)"]
    D --> E["Display / Control"]
    E --> B

🔌 การต่อวงจร

          NTC 10kΩ Thermistor — Voltage Divider
    ┌────────────────────────────────────┐
    │             3.3V                   │
    │              │                     │
    │           R_fixed=10kΩ             │
    │              │                     │
    │    PC4───────┤─ NTC (10kΩ @25°C)   │
    │              │                     │
    │              GND                   │
    │                                    │
    │ Vout = 3.3V × NTC/(R_fixed+NTC)    │
    │ R_fixed = 10k (เท่ากับ NTC@25°C)    │
    └────────────────────────────────────┘

PC4─ADC — วงจรแบ่งแรงดัน: R_fixed 10kΩ + NTC 10kΩ — NTC ลดความต้านทานเมื่อร้อนขึ้น → Vout ลดลง

📚 หลักการทำงาน

NTC (Negative Temperature Coefficient) Thermistor — ตัวต้านทานที่ค่าความต้านทานลดลงเมื่ออุณหภูมิสูงขึ้น — ใช้สมการ Beta Equation แปลงค่าความต้านทานเป็นอุณหภูมิ

Beta Equation (แบบง่าย):

1/T = 1/T₀ + (1/B) × ln(R/R₀)

T₀ = 298.15K (25°C), R₀ = 10000Ω, B = 3950 (NTC 10k)

Simple API: NTC10K ใช้ API แบบง่าย — ไม่ต้องสร้าง instance — NTC_Init() เรียกครั้งเดียวแล้วใช้ NTC_ReadTemperature() ได้เลย

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Init + Setup

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "NTC10K.h"

int main(void) {
    SystemCoreClockUpdate();
    USART_Init(115200);

    NTC_Init(ADC_CH_PC4);  // PC4 = ADC input

🔍 NTC_Init(ADC_CH_PC4) — API แบบง่าย ไม่ต้องใช้ instance — ADC channel ที่ต่อ NTC — ใช้ค่า default: R_fixed=10k, R_ref=10k, B=3950

ขั้นที่ 2: อ่านอุณหภูมิ

    float t = NTC_ReadTemperature();

    USART_Print("Temp: ");
    USART_PrintFloat(t, 1);
    USART_Println(" °C");

🔍 NTC_ReadTemperature() — คืนค่าอุณหภูมิ °C — ใช้ Beta equation — ความแม่น ±0.5°C ในช่วง 0–70°C

ขั้นที่ 3: Loop + แสดงผล

    OLED_Handle oled;
    OLED_Init(&oled, OLED_128x64, 0x3C);

    while(1) {
        float temp = NTC_ReadTemperature();

        OLED_Clear(&oled);
        OLED_SetCursor(&oled, 0, 0);
        OLED_PrintFloat(&oled, "Temp:", temp, 1);
        OLED_Print(&oled, " C");

        OLED_SetCursor(&oled, 0, 2);
        if(temp < 0)           OLED_Print(&oled, "COLD!");
        else if(temp < 20)     OLED_Print(&oled, "Cool");
        else if(temp < 30)     OLED_Print(&oled, "Comfortable");
        else if(temp < 40)     OLED_Print(&oled, "Warm");
        else                   OLED_Print(&oled, "HOT!");

        OLED_Update(&oled);
        Delay_Ms(500);
    }
}

🔍 NTC 10k B=3950 — ช่วงวัด -55 ถึง 125°C — ช่วงที่แม่นยำที่สุดคือ 0–70°C — แสดงผลบน OLED พร้อมแถบสถานะ

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
อุณหภูมิสูงกว่าจริง 2-3°CSelf-heating จากกระแสใน dividerใช้ R_fixed ใหญ่ขึ้น (47k-100k), หรือปิด divider หลังอ่าน
อุณหภูมิแกว่งตลอดADC noise — ไม่มี capacitorต่อ 100nF capacitor ขนาน NTC, เก็บค่าเฉลี่ย 8-16 samples
อุณหภูมิผิดมาก (>10°C)R_fixed ไม่ตรงกับที่ configวัด R_fixed ด้วยมัลติมิเตอร์, ใช้ค่าจริงแทนค่าที่ระบุ
Beta ผิดค่า → อุณหภูมิเพี้ยนBeta ไม่ตรง specs ของ NTC ที่ใช้เช็ค datasheet, NTC 10k ทั่วไปใช้ B=3950 หรือ B=3435

🛠️ สร้าง Product: เทอร์โมมิเตอร์ดิจิตอล

NTC 10k + OLED + แจ้งเตือนอุณหภูมิสูง/ต่ำ

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "NTC10K.h"
#include "oled_i2c.h"

int main(void)
{
    SystemCoreClockUpdate();
    USART_Init(115200);
    I2C_Init(PC1, PC2, I2C_SPEED_STANDARD);

    OLED_Handle oled;
    OLED_Init(&oled, OLED_128x64, 0x3C);

    pinMode(PC0, PIN_MODE_OUTPUT);  // Buzzer
    NTC_Init(ADC_CH_PC4);

    while(1)
    {
        float temp = NTC_ReadTemperature();

        OLED_Clear(&oled);
        OLED_SetCursor(&oled, 0, 0);
        OLED_PrintFloat(&oled, "Temperature", temp, 1);
        OLED_Print(&oled, " C");

        if(temp > 50.0f || temp < -10.0f) {
            OLED_SetCursor(&oled, 0, 4);
            OLED_Print(&oled, "ALERT!");
            digitalWrite(PC0, HIGH);  // Buzzer
        } else {
            digitalWrite(PC0, LOW);
        }

        OLED_Update(&oled);
        Delay_Ms(500);
    }
}

✏️ ลองทำดู

  1. เทียบ NTC กับ DS18B20 — วัดอุณหภูมิจุดเดียวกัน บันทึกค่า 2 เซนเซอร์ ดูความต่าง
  2. ทำ PID temperature controller — NTC feedback + MOSFET heater — รักษาอุณหภูมิให้คงที่ ±0.5°C
  3. ทดลอง self-heating — ใช้ R_fixed 1k vs 10k vs 100k วัดอุณหภูมิในห้องเดียวกัน
