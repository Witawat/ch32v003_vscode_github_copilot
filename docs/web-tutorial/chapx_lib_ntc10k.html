NTC10K — วัดอุณหภูมิด้วย Thermistor ADC

ระดับกลาง

🧭 แผนภาพการทำงาน

flowchart TD
    A["ADC_Init() + NTC_Init()"] --> B["NTC_ReadTemp()"]

          NTC 10kΩ Thermistor — Voltage Divider
    ┌────────────────────────────────────┐
    │             3.3V                   │
    │              │                     │
    │           R_fixed=10kΩ             │
    │              │                     │
    │    PA1───────┤─ NTC (10kΩ @25°C)   │

PA1─ADC — วงจรแบ่งแรงดัน: R_fixed 10kΩ + NTC 10kΩ — NTC ลดความต้านทานเมื่อร้อนขึ้น → Vout ลดลง

📚 หลักการทำงาน

NTC (Negative Temperature Coefficient) Thermistor — ตัวต้านทานที่ค่าความต้านทานลดลงเมื่ออุณหภูมิสูงขึ้น — ใช้สมการ Steinhart-Hart แปลงค่าความต้านทานเป็นอุณหภูมิ

สมการ Steinhart-Hart:

1/T = A + B×ln(R) + C×(ln(R))³

T = อุณหภูมิ (Kelvin), R = ความต้านทาน NTC (Ω)
A, B, C = ค่าสัมประสิทธิ์เฉพาะตัว (จาก datasheet)

Beta Equation (แบบง่าย):

1/T = 1/T₀ + (1/B) × ln(R/R₀)

T₀ = 298.15K (25°C), R₀ = 10000Ω, B = 3950 (NTC 10k)

Self-Heating: กระแสไหลผ่าน NTC ทำให้ตัวมันร้อนเอง → อุณหภูมิที่วัดได้สูงกว่าจริง — ใช้ R_fixed ค่าสูงเพื่อลดกระแส — แก้โดยอ่าน ADC แล้วปิดไฟ divider (ใช้ GPIO จ่ายไฟแทน VCC ต่อตรง)

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Init + Setup

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "NTC.h"

// NTC 10kΩ, B=3950, R_fixed=10kΩ, VCC=3.3V, ADC=12-bit
#define NTC_SERIES_R    10000.0f  // R_fixed 10k
#define NTC_REF_R       10000.0f  // NTC @ 25°C
#define NTC_BETA        3950.0f   // B-constant

int main(void) {
    SystemCoreClockUpdate();
    USART_Init(USART1, BAUD_115200);
    ADC_Init(ADC1);

    NTC_Init(PA1, NTC_SERIES_R, NTC_REF_R, NTC_BETA);

🔍 NTC_Init(pin, seriesR, refR, beta) — seriesR ต้องเท่ากับ R_fixed ในวงจร, refR คือค่าที่ 25°C (ปกติ 10k), beta ได้จาก datasheet

ขั้นที่ 2: อ่าน/เขียนค่า

    float tempC = NTC_ReadTemp(NTC_UNIT_CELSIUS);
    float tempF = NTC_ReadTemp(NTC_UNIT_FAHRENHEIT);
    float tempK = NTC_ReadTemp(NTC_UNIT_KELVIN);

    USART_Print("Temp: ");
    USART_PrintFloat(tempC, 1);
    USART_Println(" °C");

    // อ่านค่าดิบ (resistance)
    float resistance = NTC_GetResistance();
    USART_Print("NTC: ");
    USART_PrintFloat(resistance, 1);
    USART_Println(" Ω");

🔍 NTC_ReadTemp() คืนค่าอุณหภูมิในหน่วยที่เลือก — ใช้ Steinhart-Hart หรือ Beta equation ขึ้นอยู่กับการ config — ความแม่น ±0.5°C ในช่วง 0–70°C

ขั้นที่ 3: Loop + แสดงผล

    OLED_t oled;
    OLED_Init(&oled, 0x3C, OLED_128x64);

    while(1) {
        float temp = NTC_ReadTemp(NTC_UNIT_CELSIUS);

        OLED_Clear(&oled);
        OLED_SetCursor(&oled, 0, 0);
        OLED_PrintFloat(&oled, "Temp:", temp, 1);
        OLED_Print(&oled, " C");

        // แสดงสถานะ
        OLED_SetCursor(&oled, 0, 2);
        if(temp < 0)           OLED_Print(&oled, "COLD!");
        else if(temp < 20)     OLED_Print(&oled, "Cool");
        else if(temp < 30)     OLED_Print(&oled, "Comfortable");
        else if(temp < 40)     OLED_Print(&oled, "Warm");
        else                   OLED_Print(&oled, "HOT!");

        OLED_Display(&oled);
        Delay_Ms(500);
    }
}

🔍 NTC 10k B=3950 — ช่วงวัด -55 ถึง 125°C — แต่ช่วงที่แม่นยำที่สุดคือ 0–70°C — แสดงผลบน OLED พร้อมแถบสถานะ

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
อุณหภูมิสูงกว่าจริง 2-3°CSelf-heating จากกระแสใน dividerใช้ R_fixed ใหญ่ขึ้น (47k-100k), หรือปิด divider หลังอ่าน
อุณหภูมิแกว่งตลอดADC noise — ไม่มี capacitorต่อ 100nF capacitor ขนาน NTC, เก็บค่าเฉลี่ย 8-16 samples
อุณหภูมิผิดมาก (>10°C)R_fixed ไม่ตรงกับที่ configวัด R_fixed ด้วยมัลติมิเตอร์, ใช้ค่าจริงแทนค่าที่ระบุ
อ่านค่า resistance ได้ 0 ΩNTC ลัดวงจร หรือสายขาดเช็คการต่อสาย, วัด Vout ที่ divider — ควรอยู่ระหว่าง 0.3V–2.7V
Beta ผิดค่า → อุณหภูมิเพี้ยนBeta ไม่ตรง specs ของ NTC ที่ใช้เช็ค datasheet, NTC 10k ทั่วไปใช้ B=3950 หรือ B=3435

🛠️ สร้าง Product: เทอร์โมมิเตอร์ดิจิตอล

NTC 10k + OLED + แจ้งเตือนอุณหภูมิสูง/ต่ำ

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "NTC.h"
#include "OLED.h"

int main(void)
{
    SystemCoreClockUpdate();
    USART_Init(USART1, BAUD_115200);
    I2C_Init(I2C1, I2C_SPEED_STANDARD);
    ADC_Init(ADC1);

    OLED_t oled;
    OLED_Init(&oled, 0x3C, OLED_128x64);

    pinMode(PC0, PIN_MODE_OUTPUT);  // Buzzer
    NTC_Init(PA1, 10000, 10000, 3950);

    while(1)
    {
        float temp = NTC_ReadTemp(NTC_UNIT_CELSIUS);

        OLED_Clear(&oled);
        OLED_SetCursor(&oled, 0, 0);
        OLED_PrintFloat(&oled, "Temperature", temp, 1);
        OLED_Print(&oled, " C");

        OLED_SetCursor(&oled, 0, 2);
        OLED_PrintFloat(&oled, "NTC:", NTC_GetResistance(), 0);
        OLED_Print(&oled, " ohm");

        // Alert
        if(temp > 50.0f || temp < -10.0f) {
            OLED_SetCursor(&oled, 0, 4);
            OLED_Print(&oled, "ALERT!");
            digitalWrite(PC0, HIGH);  // Buzzer
        } else {
            digitalWrite(PC0, LOW);
        }

        OLED_Display(&oled);
        Delay_Ms(500);
    }
}

✏️ ลองทำดู

  1. เทียบ NTC กับ DS18B20 — วัดอุณหภูมิจุดเดียวกัน บันทึกค่า 2 เซนเซอร์ ดูความต่าง
  2. ทำ PID temperature controller — NTC feedback + MOSFET heater — รักษาอุณหภูมิให้คงที่ ±0.5°C
  3. ทดลอง self-heating — ใช้ R_fixed 1k vs 10k vs 100k วัดอุณหภูมิในห้องเดียวกัน เปรียบเทียบ
