⚠️ ESP8266 กินกระแสสูงสุด ~300mA ตอนส่ง WiFi — ต้องใช้ external 3.3V regulator อย่างน้อย 500mA (AMS1117-3.3 หรือดีกว่า), CH_PD ต้องต่อ HIGH ตลอดเวลา, GPIO0 ต่อ GND = Flash mode (ตอนโปรแกรม ESP), GPIO0 ลอย = Normal mode

📚 หลักการทำงาน

ESP8266 เป็น WiFi SoC ที่สื่อสารผ่าน AT Commands ทาง USART — CH32V003 ส่งคำสั่งแบบ text แล้วรอ response OK หรือ ERROR แต่ละคำสั่งคั่นด้วย \r\n

โหมด WiFi: AT+CWMODE=1 = Station (ต่อ AP), AT+CWMODE=2 = SoftAP (สร้าง hotspot), AT+CWMODE=3 = Station+SoftAP — ส่วนใหญ่ใช้ mode 1

AT Command ทีละขั้น: Test → Set Mode → Connect WiFi → Get IP → Connect TCP → Send/Recv

Baud Rate: Default 115200 bps — ต้อง match กับ CH32V003 USART ถ้าเปลี่ยน baud ใน ESP ต้องเปลี่ยนตาม

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Init + Setup

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "ESP8266.h"

int main(void) {
    SystemCoreClockUpdate();
    USART_Init(USART1, BAUD_115200);      // debug console
    USART_Init(USART2, BAUD_115200);      // ESP communication
    ESP_Init(USART2, BAUD_115200);
    pinMode(PC1, PIN_MODE_OUTPUT);         // CH_PD pin
    digitalWrite(PC1, HIGH);               // enable ESP

🔍 ใช้ USART2 สำหรับ ESP8266, USART1 สำหรับ debug console — CH32V003 มี 2 USART พอดี, CH_PD ต้อง HIGH ก่อน ESP จะเริ่มทำงาน

ขั้นที่ 2: ต่อ WiFi + TCP

    USART_Println("Connecting WiFi...");
    if(!ESP_ConnectWiFi("MyWiFi", "password123")) {
        USART_Println("WiFi Failed!"); while(1);
    }
    USART_Println("WiFi Connected!");
    USART_Print(ESP_GetIP());  // print IP address

    if(!ESP_ConnectTCP("api.thingspeak.com", 80)) {
        USART_Println("TCP Failed!"); while(1);
    }

🔍 ESP_ConnectWiFi(ssid, pwd) — ใช้ AT+CWJAP ภายใน, ต้องรอจนกว่า ESP ตอบ WIFI GOT IP, ESP_ConnectTCP(host, port) — เปิด TCP connection

ขั้นที่ 3: ส่งข้อมูล HTTP + Loop

    while(1) {
        // HTTP GET request to ThingSpeak
        ESP_Send("GET /update?api_key=xxx&field1=25 HTTP/1.1\r\n");
        ESP_Send("Host: api.thingspeak.com\r\n");
        ESP_Send("Connection: close\r\n\r\n");
        Delay_Ms(1000);
        USART_Println(ESP_Read());  // read response
        Delay_Ms(15000);  // ThingSpeak limit: 1 update/15s
    }
}

🔍 ESP_Send(str) — ส่ง raw data ผ่าน TCP (ใช้ AT+CIPSEND), ESP_Read() — อ่านข้อมูลที่ ESP ได้รับและคืนเป็น string, ThingSpeak จำกัด 15 วินาทีต่อ 1 update

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
ESP ไม่ตอบ ATไฟ 3.3V ไม่พอ — ESP ดับขณะ TXใช้ external 3.3V ≥500mA, ต่อ capacitor 100µF ขนาน VCC-GND
บอดไม่ตรงกันESP default baud 115200, แต่ CH32V003 ตั้งผิดเช็ค USART_Init(USART2, BAUD_115200) ให้ตรงกันทั้งสองฝั่ง
WiFi เชื่อมทุกครั้งตกไฟกระเพื่อมตอน TX (current spike)เพิ่ม capacitor 470µF + 0.1µF ที่ขา VCC, ใช้ regulator แยก
TCP ส่งข้อมูลไม่ครบส่งเร็วเกินไป buffer overflowdelay 50-100ms ระหว่าง ESP_Send() แต่ละครั้ง
RX buffer เต็ม ค้างลืมอ่านข้อมูลจาก ESPเรียก ESP_Read() หลังแต่ละการส่งเพื่อ clear buffer

🛠️ สร้าง Product: IoT Temperature Logger

CH32V003 อ่านอุณหภูมิภายใน + ส่งขึ้น ThingSpeak ทุก 20 วินาทีผ่าน ESP8266

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "ESP8266.h"

int main(void)
{
    SystemCoreClockUpdate();
    USART_Init(USART1, BAUD_115200);
    USART_Init(USART2, BAUD_115200);
    ADC_Init(ADC1, ADC_CH_TEMP);
    ESP_Init(USART2, BAUD_115200);
    pinMode(PC1, PIN_MODE_OUTPUT);
    digitalWrite(PC1, HIGH);
    Delay_Ms(3000);

    if(!ESP_ConnectWiFi("MyWiFi", "pass")) {
        USART_Println("WiFi Fail"); while(1);
    }

    while(1)
    {
        uint16_t adc = ADC_Read(ADC1, ADC_CH_TEMP);
        float temp = (float)adc * 3300.0 / 4096.0;  // mV
        temp = (temp - 1430.0) / 4.3 + 25.0;        // °C

        ESP_ConnectTCP("api.thingspeak.com", 80);
        char buf[128];
        sprintf(buf, "GET /update?api_key=XXXXXXXX&field1=%.1f HTTP/1.1\r\nHost: api.thingspeak.com\r\n\r\n", temp);
        ESP_Send(buf);
        Delay_Ms(1000);
        ESP_Read();
        ESP_Disconnect();

        USART_Print("Temp: "); USART_PrintFloat(temp, 1); USART_Println(" C");
        Delay_Ms(20000);
    }
}

✏️ ลองทำดู

  1. ทำ IoT switch — ส่ง AT+CIPSEND พร้อม HTTP GET ไปยัง API server เพื่อเปิด/ปิด LED จาก Web browser
  2. ทำ ESP web server — ใช้ AT+CIPSERVER=1,80 สร้าง HTTP server บน ESP รับคำสั่งจาก browser มา toggle GPIO
  3. ส่งข้อมูลเซนเซอร์ 3 ค่า (temp, humidity, light) ขึ้น ThingSpeak 3 channels — ใช้ sprintf จัดรูปแบบ JSON/POST
