📝 เฉลย — บทที่ 13: OPAMP

ข้อ 1: Voltage Follower อ่านสัญญาณ input

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include "SimpleHAL.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
    ADC_SimpleInit();

    pinMode(PC0, PIN_MODE_OUTPUT);  // LED status

    // Voltage Follower — PA4=Input, PA5=Output
    // Impedance input สูงมาก (~100MΩ) — ไม่ดึงกระแสจากแหล่งสัญญาณ
    OPAMP_SimpleInit(OPAMP_CH1, OPAMP_MODE_FOLLOWER);
    OPAMP_Enable(OPAMP_CH1);

    USART_Print("OPAMP Voltage Follower\r\n");
    USART_Print("PA4(input) → OPAMP Buffer → PA5(output) → ADC\r\n");
    USART_Print("Ready — connect sensor to PA4\r\n\r\n");

    // รอ OPAMP settle
    Delay_Ms(10);

    while (1) {
        digitalWrite(PC0, HIGH);

        // วิธีที่ 1: อ่าน ADC จาก OPAMP output byตรง
        uint16_t adc_raw = ADC_Read(ADC_CH_PA5);
        float v_out = ADC_ToVoltage(adc_raw, 3.3f);

        // วิธีที่ 2: OPAMP_GetOutputVoltage (wrapper)
        float v_opamp = OPAMP_GetOutputVoltage(OPAMP_CH1);

        USART_Print("ADC=");
        USART_PrintNum(adc_raw);
        USART_Print(" V_out=");
        USART_PrintNum((int32_t)(v_out * 1000));
        USART_Print("mV");

        USART_Print("  OPAMP_Get=");
        USART_PrintNum((int32_t)(v_opamp * 1000));
        USART_Print("mV");

        // Follower: Vout ≈ Vin (Gain = 1)
        // ข้อควรรู้: output offset ~5mV typ, output swing: 0.1V ~ VDD-0.1V

        if (v_out > 2.5f) {
            USART_Print(" [HIGH]");
        } else if (v_out < 0.5f) {
            USART_Print(" [LOW]");
        } else {
            USART_Print(" [MID]");
        }

        USART_Print("\r\n");

        digitalWrite(PC0, LOW);
        Delay_Ms(500);
    }
}
🧠 แนวคิด: Voltage Follower (Buffer) มี Gain = 1 — จุดประสงค์หลักคือ Impedance Matching — อินพุต impedance สูง ~100MΩ → ไม่ดึงกระแสจากเซ็นเซอร์ (เช่น pH probe, piezo, high-Z sensor) — เอาต์พุต impedance ต่ำ → ขับ ADC sample/hold ได้ดี — OPAMP_SimpleInit(OPAMP_CH1, OPAMP_MODE_FOLLOWER) ต่อวงจรภายใน — OPAMP_Enable() จ่ายไฟ — ต้องรอ settle ~10µs-10ms — Pin mapping: PA4=OP_IN+, PA5=OP_OUT, PA6=OP_IN- — ADC อ่าน PA5 — offset voltage ~5mV typ → calibration อาจจำเป็น — OPAMP bandwidth ~1MHz — ไม่ใช่ rail-to-rail: output 0.1V ถึง VDD-0.1V

ข้อ 2: Comparator เปรียบเทียบแรงดัน 2 ค่า

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include "SimpleHAL.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
    ADC_SimpleInit();

    pinMode(PC0, PIN_MODE_OUTPUT);  // LED: V_A > V_B → ON
    pinMode(PC1, PIN_MODE_OUTPUT);  // LED: V_A < V_B → ON

    USART_Print("=== OPAMP Comparator ===\r\n");
    USART_Print("PA4 = V_A (non-inverting)\r\n");
    USART_Print("PA6 = V_B (inverting, via R divider)\r\n");
    USART_Print("V_A > V_B → PC0=ON,  V_A < V_B → PC1=ON\r\n");
    USART_Print("V_A = PD2 (potentiometer), V_B = fixed ref\r\n\r\n");

    // OPAMP ในโหมด Open-loop = Comparator
    // ไม่มีโหมด comparator โดยตรงใน SimpleHAL — ใช้ ADC เปรียบเทียบแทน
    // หรือใช้ OPAMP ใน follower + อ่าน ADC แล้วเปรียบเทียบใน software

    // Software Comparator ผ่าน ADC + OPAMP Buffer:
    OPAMP_SimpleInit(OPAMP_CH1, OPAMP_MODE_FOLLOWER);
    OPAMP_Enable(OPAMP_CH1);
    Delay_Ms(10);

    // อ่านแรงดันอ้างอิงจาก PA6 (หรือใช้ ADC_CH_PA4 เทียบ)
    // สมมติ V_B = 1.65V (VDD/2)

    while (1) {
        // อ่าน V_A ผ่าน OPAMP buffer (PA4 → PA5)
        uint16_t raw_a = ADC_Read(ADC_CH_PA5);
        float v_a = ADC_ToVoltage(raw_a, 3.3f);

        // อ่าน V_B โดยตรง (สมมติต่อ voltage divider ที่ PD2)
        uint16_t raw_b = ADC_Read(ADC_CH_PD2);
        float v_b = ADC_ToVoltage(raw_b, 3.3f);

        // Comparator logic
        if (v_a > v_b) {
            digitalWrite(PC0, HIGH);
            digitalWrite(PC1, LOW);
            USART_Print("V_A > V_B  ");
        } else if (v_a < v_b) {
            digitalWrite(PC0, LOW);
            digitalWrite(PC1, HIGH);
            USART_Print("V_A < V_B  ");
        } else {
            digitalWrite(PC0, LOW);
            digitalWrite(PC1, LOW);
            USART_Print("V_A = V_B  ");
        }

        USART_Print("V_A=");
        USART_PrintNum((int32_t)(v_a * 1000));
        USART_Print("mV  V_B=");
        USART_PrintNum((int32_t)(v_b * 1000));
        USART_Print("mV\r\n");

        Delay_Ms(300);
    }
}
🧠 แนวคิด: CH32V003 OPAMP ไม่มีโหมด comparator โดยตรง — ใช้ "Software Comparator" แทน: ADC อ่าน 2 ช่อง → เปรียบเทียบใน software — หรือใช้ hardware: ต่อ OPAMP เป็น open-loop → output จะ swing ไป rail ถ้า input ต่างกัน > offset — ข้อดีของ software comparator: ตั้ง hysteresis ได้, ตั้ง threshold ได้ง่าย, ไม่ต้องต่อวงจรภายนอก — ข้อเสีย: ช้ากว่า hardware comparator (ADC ใช้ ~10µs/sample) — Application: ตรวจจับแรงดันเกิน/ต่ำ, window detector, brown-out warning ก่อน VDD ตกถึง reset level — OPAMP ใช้เป็น buffer ให้ ADC เพื่อลด loading effect เมื่อเทียบแรงดันจาก high-impedance source

ข้อ 3: Non-Inverting Amp gain=2

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include "SimpleHAL.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
    ADC_SimpleInit();

    USART_Print("=== Non-Inverting Amplifier Gain=2 ===\r\n");
    USART_Print("Vout = Vin * (1 + Rf/Rg) = Vin * 2\r\n");
    USART_Print("Requires external Rf, Rg resistors\r\n");
    USART_Print("PA4 = Vin, PA6 = feedback node (Rf/Rg), PA5 = Vout\r\n");
    USART_Print("Rf = 10k, Rg = 10k → Gain = 1 + 10k/10k = 2\r\n\r\n");

    // Non-Inverting Amplifier:
    // ต่อวงจรภายนอก:
    // Vin → PA4 (OP_IN+)
    // PA6 (OP_IN-) ← Rf → PA5 (OP_OUT)
    // PA6 ← Rg → GND
    // Gain = 1 + Rf/Rg

    OPAMP_SimpleInit(OPAMP_CH1, OPAMP_MODE_NON_INVERTING);
    OPAMP_Enable(OPAMP_CH1);
    Delay_Ms(10);

    // สมมติ Vin จาก PD2 (pot divider) — ต่อ PA4 โดยตรง
    // PA5 output ขยาย 2x → อ่านด้วย ADC

    while (1) {
        // อ่าน Vout (PA5 — ขยายแล้ว)
        uint16_t raw_out = ADC_Read(ADC_CH_PA5);
        float v_out = ADC_ToVoltage(raw_out, 3.3f);

        // อ่าน Vin (PD2 — ก่อนขยาย)
        uint16_t raw_in = ADC_Read(ADC_CH_PD2);
        float v_in = ADC_ToVoltage(raw_in, 3.3f);

        // คำนวณ gain จริง
        float gain_measured = (v_in > 0.01f) ? (v_out / v_in) : 0.0f;

        USART_Print("Vin=");
        USART_PrintNum((int32_t)(v_in * 1000));
        USART_Print("mV  Vout=");
        USART_PrintNum((int32_t)(v_out * 1000));
        USART_Print("mV  Gain=");
        USART_PrintNum((int32_t)(gain_measured * 100));
        USART_Print(".");

        int16_t dec = (int16_t)((gain_measured - (int32_t)gain_measured) * 100);
        if (dec < 0) dec = -dec;
        USART_PrintNum(dec);

        // ตรวจสอบ: Gain ควร ≈ 2.0 (±0.1 จาก resistor tolerance)
        if (gain_measured > 1.9f && gain_measured < 2.1f) {
            USART_Print(" [OK]");
        } else if (gain_measured > 0.05f) {
            USART_Print(" [CHECK Rf,Rg!]");
        } else {
            USART_Print(" [Vin too low]");
        }

        USART_Print("\r\n");

        // ตรวจสอบ saturation
        if (v_out > 3.2f) {
            USART_Print("  WARNING: Output saturated (close to VDD)\r\n");
            USART_Print("  Reduce Vin or lower gain\r\n");
        }

        Delay_Ms(500);
    }
}
🧠 แนวคิด: Non-Inverting Amp ใช้ R ภายนอก 2 ตัว: Rf (feedback) และ Rg (to GND) — สูตร: Vout = Vin × (1 + Rf/Rg) — Rf = Rg = 10kΩ → Gain = 2 — ต่อวงจร: PA4 (+), PA6 (−) เชื่อมระหว่าง Rf กับ Rg, PA5 (out) — OPAMP_MODE_NON_INVERTING ตั้งวงจรภายในให้พร้อมรับ R ภายนอก — ข้อควรระวัง: output swing ไม่เกิน VDD-0.1V → ถ้า Vin = 2V, Gain=2 → Vout = 4V → saturated ที่ ~3.2V — ถ้าต้องการขยายมากกว่า VDD → ใช้ PGA แทน (Gain สูงสุด 16x) — ข้อดี: Non-inverting → input impedance สูง, เฟสไม่กลับ — Resistor tolerance: 1% R → Gain error ~2% — Calibrate ใน software ได้
