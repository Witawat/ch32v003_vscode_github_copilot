📝 เฉลย — บทที่ 13: OPAMP

ข้อ 1: Voltage Follower บัฟเฟอร์เซ็นเซอร์ + อ่าน ADC

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include "SimpleHAL.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
    ADC_SimpleInit();  // เปิด ADC (no-arg)

    pinMode(PC0, PIN_MODE_OUTPUT);  // LED status

    // Voltage Follower — buffer สัญญาณเซ็นเซอร์ high-Z
    // SimpleInit รับ mode เดียว (ไม่มี channel enum ปลอม)
    OPAMP_SimpleInit(OPAMP_MODE_VOLTAGE_FOLLOWER);
    OPAMP_Enable();  // no-arg

    USART_Print("OPAMP Voltage Follower Ready\r\n");
    USART_Print("Connect high-Z sensor to CHP0 input\r\n\r\n");

    Delay_Ms(10);  // รอ OPAMP settle

    while (1) {
        digitalWrite(PC0, HIGH);

        // อ่าน OPAMP output ผ่าน ADC ch7 (ต่อภายในอยู่แล้ว)
        // ใช้ ADC_ReadAverage ลด noise จากสัญญาณเซ็นเซอร์
        uint16_t raw = ADC_ReadAverage(ADC_CH_7, 8);
        float v_out = ADC_ToVoltage(raw, 3.3f);

        USART_Print("Vout=");
        USART_PrintNum((int32_t)(v_out * 1000));
        USART_Print(" mV  (raw=");
        USART_PrintNum(raw);
        USART_Print(")");

        // Follower: Vout ≈ Vin (Gain = 1)
        if (v_out > 2.5f) {
            USART_Print(" [HIGH]");
        } else if (v_out < 0.5f) {
            USART_Print(" [LOW]");
        } else {
            USART_Print(" [MID]");
        }
        USART_Print("\r\n");

        digitalWrite(PC0, LOW);
        Delay_Ms(500);
    }
}
🧠 แนวคิด: Voltage Follower (Buffer) มี Gain = 1 — จุดประสงค์หลักคือ Impedance Matching — อินพุต impedance สูงมาก (~100MΩ) → ไม่ดึงกระแสจากเซ็นเซอร์ (เช่น pH probe, piezo, high-Z sensor) — เอาต์พุต impedance ต่ำ → ขับ ADC sample/hold ได้ดีโดยไม่ทำให้แรงดันตก — OPAMP_SimpleInit(OPAMP_MODE_VOLTAGE_FOLLOWER) รับ mode เดียว (ห้ามส่ง channel ปลอมเข้าไป) — OPAMP_Enable() เป็น no-arg ใช้จ่ายไฟ — OPAMP output ต่อภายในกับ ADC ch7 จึงอ่านด้วย ADC_ReadAverage(ADC_CH_7, 8) เพื่อลด noise — ข้อควรระวัง: OPAMP ไม่ใช่ rail-to-rail output swing คือ 0.1V ถึง VDD−0.1V และมี offset ~5mV typ อาจต้อง calibrate ใน software สำหรับการวัดละเอียด

ข้อ 2: Comparator สั่ง LED ที่ PC0

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include "SimpleHAL.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
    ADC_SimpleInit();

    pinMode(PC0, PIN_MODE_OUTPUT);  // LED: V+ > V- → ON

    USART_Print("=== OPAMP Comparator ===\r\n");
    USART_Print("Signal     → CHP0 (+)\r\n");
    USART_Print("Threshold  → CHN0 (-)\r\n");
    USART_Print("V+ > V-  → PC0=ON (LED)\r\n\r\n");

    // Comparator: ทำงานแบบ open-loop → output swing เกือบเต็ม rail
    OPAMP_ConfigComparator(OPAMP_CHP0, OPAMP_CHN0);
    OPAMP_Enable();  // no-arg
    Delay_Ms(10);

    while (1) {
        // Comparator output อ่านผ่าน ADC ch7:
        //   ~0V    (raw ใกล้ 0)   = V+ < V-  → LOW
        //   ~VDD  (raw ใกล้ 1023) = V+ > V-  → HIGH
        uint16_t raw = ADC_Read(ADC_CH_7);
        uint8_t  high = (raw > 512);

        if (high) {
            digitalWrite(PC0, HIGH);
            USART_Print("V+ > V-  (HIGH)  raw=");
        } else {
            digitalWrite(PC0, LOW);
            USART_Print("V+ < V-  (LOW)   raw=");
        }
        USART_PrintNum(raw);
        USART_Print("\r\n");

        Delay_Ms(200);
    }
}
🧠 แนวคิด: โหมด Comparator ทำงานแบบ open-loop (ไม่มี feedback) → ความขยายสูงมาก → output จะ swing ไปแถบ 0V หรือ VDD ทันทีที่ V+ ต่างจาก V− — จึงอ่านเป็น digital ผ่าน ADC ch7 ได้โดยใช้ threshold กึ่งกลาง (raw > 512) — ข้อดีของ Comparator ในชิป: ไม่ต้องต่อ IC comparator ภายนอก, ตอบสนองเร็วกว่า software comparator (ที่ต้อง sample 2 ช่อง) — การประยุกต์: ตรวจจับแรงดันเกิน/ต่ำ (over-voltage / brown-out warning), zero-crossing detection, ประตูทริกเกอร์จาก LDR หรือเซ็นเซอร์ threshold — หมายเหตุ: เนื่องจาก OPAMP ตัวเดียว ถ้าต้องการ hysteresis ให้ต่อ R feedback ขนาดใหญ่จาก output ไปยัง CHP เพื่อสร้าง positive feedback หรือทำ hysteresis ใน software แทน

ข้อ 3: Non-Inverting ขยายสัญญาณเสียง/noise จาก mic

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include "SimpleHAL.h"

// สมมติ mic/noise ให้สัญญาณประมาณ 0–50 mV ต้องขยาย ~50 เท่า
// → ใช้ Non-Inverting: Gain = 1 + R2/R1

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
    ADC_SimpleInit();

    // ค่า R ภายนอกที่เลือก: R1 = 1 kΩ (to GND)
    // ต้องการ gain ~50 → คำนวณ R2 อัตโนมัติ
    uint32_t r1 = 1000;           // 1 kΩ
    float    desired_gain = 50.0f;
    uint32_t r2 = OPAMP_CalculateR2NonInv(r1, desired_gain);  // คืน ~49 kΩ

    // ตรวจสอบ gain จาก R ที่จะต่อจริง
    float gain = OPAMP_CalculateGainNonInv(r1, r2);

    USART_Print("Mic amplifier: gain target = ");
    USART_PrintNum((int32_t)desired_gain);
    USART_Print("  R1=");
    USART_PrintNum(r1);
    USART_Print("  R2(calc)=");
    USART_PrintNum(r2);
    USART_Print("  gain(real)=");
    USART_PrintNum((int32_t)gain);
    USART_Print("\r\n\r\n");

    // วงจร: mic → CHP0(+),  CHN0(−) ← R2 ← Output,  CHN0 ← R1 ← GND
    OPAMP_ConfigNonInverting(OPAMP_CHP0, OPAMP_CHN0);
    OPAMP_Enable();  // no-arg
    Delay_Ms(10);

    uint16_t baseline = 0;
    uint8_t  cal = 0;

    while (1) {
        // อ่าน output ที่ขยายแล้ว
        float v_out = ADC_ReadVoltage(ADC_CH_7, 3.3f);

        // First run: เก็บ baseline (offset จาก opamp + bias)
        if (!cal) {
            baseline = (uint16_t)(v_out * 1000);  // mV
            cal = 1;
        }

        int32_t delta_mv = (int32_t)(v_out * 1000) - baseline;

        // Clipping check: ถ้า output ชิดเพดาน → gain สูงเกิน
        if (v_out > 3.2f) {
            USART_Print("CLIP-high ");
        } else if (v_out < 0.1f) {
            USART_Print("CLIP-low ");
        }

        USART_Print("Vout=");
        USART_PrintNum((int32_t)(v_out * 1000));
        USART_Print(" mV  d=");
        USART_PrintNum(delta_mv);
        USART_Print(" mV\r\n");

        Delay_Ms(50);  // sample 20 Hz
    }
}
🧠 แนวคิด: Non-Inverting Amp ใช้ R ภายนอก 2 ตัว R1 (to GND) และ R2 (feedback) — สูตร Gain = 1 + (R2/R1) — เฟสเอาต์พุตตรงกับอินพุต (ไม่กลับ) — input impedance สูงเหมาะกับ mic ที่สัญญาณเล็ก — OPAMP_CalculateR2NonInv(r1, desired_gain) คำนวณค่า R2 ที่ต้องการจาก gain เป้าหมาย (สูตร R2 = R1 × (gain − 1)) — หลังต่อ R จริงแล้วใช้ OPAMP_CalculateGainNonInv(r1, r2) เพื่อยืนยัน gain จริงอีกครั้ง — ข้อควรระวัง: สัญญาณ mic มักต้อง bias ที่ VDD/2 ก่อนเข้า OPAMP เพื่อให้อยู่กลาง range ไม่งั้น output ติด rail — ถ้า output ชิดเพดาน (clipping) ให้ลด gain (เช่นเปลี่ยนเป็น gain ~20) โดยลด desired_gain หรือเพิ่ม R1 — ขั้นตอนปรับปรุง: ใช้ running average ลด noise และใช้ HPF ใน software ตัดส่วน DC ออกก่อนวัด amplitude จริง — ไม่ใช้ ชื่อโหมด/ฟังก์ชันปลอมแบบ MODE_PGA หรือ ConfigPGA (เมื่อเติม prefix OPAMP_ ข้างหน้าจะเป็นชื่อปลอมที่ไม่มีใน header จริง) เพราะไม่มีใน API จริง gain ตั้งด้วย R ภายนอกเท่านั้น
