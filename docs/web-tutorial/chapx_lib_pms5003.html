PMS5003 — เซนเซอร์วัดฝุ่น PM2.5 USART

ระดับสูง

🧭 แผนภาพการทำงาน

flowchart TD
    A["PMS5003_Init(&sensor, USART_PINS_DEFAULT)"] --> B["PMS5003_GetData(&sensor, &data)"]
    B --> C["data.pm1_0_std, pm2_5_std, pm10_std"]
    C --> D["Display / Log"]
    D --> B

🔌 การต่อวงจร

           PMS5003 — PM2.5 Laser Particle Sensor
    ┌────────────────────────────────────┐
    │ MCU                         PMS5003│
    │ PD6(RX)─────TX(pin4)    ┌─────────┤
    │ PD5(TX)─────RX(pin5)    │ Laser   │
    │ GND─────────GND(pin2)   │ counter │
    │ 5V──────────VCC(pin1)   └─────────┤
    │                                    │
    │ Frame: 32 bytes, baud=9600         │
    └────────────────────────────────────┘
MCU Pin→อุปกรณ์ขาหมายเหตุ
PD5 (TX)→PMS5003RX (pin5)MCU→เซนเซอร์
PD6 (RX)←PMS5003TX (pin4)เซนเซอร์→MCU
5V↔PMS5003VCC (pin1)จ่ายไฟ 5V
GND↔PMS5003GND (pin2)กราวด์ร่วม

รู้จักขา

  • VCC (pin1): ไฟเลี้ยง 5V
  • GND (pin2): กราวด์
  • RX (pin5): รับข้อมูลจาก MCU — ต่อขา PD5 (TX)
  • TX (pin4): ส่งข้อมูลไป MCU — ต่อขา PD6 (RX)

วิธีต่อทีละขั้น

  1. ต่อ VCC (pin1) PMS5003 ไป 5V ของ MCU
  2. ต่อ GND (pin2) PMS5003 ไป GND ของ MCU
  3. ต่อ RX (pin5) PMS5003 ไปขา PD5 (TX) ของ MCU
  4. ต่อ TX (pin4) PMS5003 ไปขา PD6 (RX) ของ MCU

📚 หลักการทำงาน

PMS5003 เป็นเซนเซอร์วัดฝุ่นละออง PM1.0/PM2.5/PM10 แบบ laser scattering — สื่อสารผ่าน USART 9600 baud — 32-byte binary frame

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Init + Setup

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "PMS5003.h"

PMS5003_Instance sensor;
PMS5003_Data_t data;

int main(void) {
    SystemCoreClockUpdate();
    USART_Init(USART1, BAUD_9600);  // PMS5003 = 9600 fixed

    PMS5003_Init(&sensor, USART_PINS_DEFAULT);
    USART_Println("PMS5003 Ready!");

🔍 PMS5003_Init(&sensor, USART_PINS_DEFAULT) — instance-based, ใช้ USART pins default — ตั้งค่า baud 9600 อัตโนมัติ

ขั้นที่ 2: อ่านค่า

    if(PMS5003_GetData(&sensor, &data)) {
        USART_Print("PM1.0: "); USART_PrintInt(data.pm1_0_std);
        USART_Print(" PM2.5: "); USART_PrintInt(data.pm2_5_std);
        USART_Print(" PM10: "); USART_PrintInt(data.pm10_std);
        USART_Println(" ug/m3");

        if(data.pm2_5_std <= 25)      USART_Println("Air: GOOD");
        else if(data.pm2_5_std <= 50) USART_Println("Air: MODERATE");
        else                          USART_Println("Air: UNHEALTHY!");
    }

🔍 PMS5003_GetData(&sensor, &data) — อ่าน frame 32-byte + verify checksum — คืน TRUE ถ้า valid

ขั้นที่ 3: Loop + แสดงผล

    OLED_Handle oled;
    OLED_Init(&oled, OLED_128x64, 0x3C);

    char buf[64];
    while(1) {
        if(PMS5003_GetData(&sensor, &data)) {
            OLED_Clear(&oled);
            snprintf(buf, sizeof(buf), "PM2.5: %d ug/m3", data.pm2_5_std);
            OLED_DrawStringAlign(&oled, 0, 0, buf, ALIGN_LEFT);

            if(data.pm2_5_std <= 25)      snprintf(buf, sizeof(buf), "AIR: GOOD");
            else if(data.pm2_5_std <= 50) snprintf(buf, sizeof(buf), "AIR: MODERATE");
            else                          snprintf(buf, sizeof(buf), "AIR: UNHEALTHY!");
            OLED_DrawStringAlign(&oled, 0, 20, buf, ALIGN_LEFT);

            OLED_Update(&oled);
        }
        Delay_Ms(1000);
    }
}

🔍 PMS5003 อัพเดททุก ~800ms — หน่วง 1 วิเพื่อ sync

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
GetData คืน FALSE ตลอดBaud ผิด (ต้อง 9600) หรือ RX/TX สลับเช็ค PD6→TX(sensor), PD5→RX(sensor), baud=9600
PM2.5 = 0 ตลอดเวลาBuffer ยังไม่เต็มรออย่างน้อย 1 วิหลัง init

🛠️ สร้าง Product: เครื่องวัดฝุ่น PM2.5 พกพา

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "PMS5003.h"
#include "oled_i2c.h"

int main(void)
{
    SystemCoreClockUpdate();
    USART_Init(USART1, BAUD_9600);
    I2C_Init(I2C1, I2C_SPEED_STANDARD);

    OLED_Handle oled;
    OLED_Init(&oled, OLED_128x64, 0x3C);

    PMS5003_Instance sensor;
    PMS5003_Data_t data;
    PMS5003_Init(&sensor, USART_PINS_DEFAULT);

    char buf[64];
    OLED_Clear(&oled);
    snprintf(buf, sizeof(buf), "PM2.5 Monitor");
    OLED_DrawStringAlign(&oled, 0, 0, buf, ALIGN_LEFT);
    snprintf(buf, sizeof(buf), "Warming up...");
    OLED_DrawStringAlign(&oled, 0, 20, buf, ALIGN_LEFT);
    OLED_Update(&oled);
    Delay_Ms(5000);

    while(1)
    {
        if(PMS5003_GetData(&sensor, &data)) {
            OLED_Clear(&oled);
            snprintf(buf, sizeof(buf), "PM1.0: %d", data.pm1_0_std);
            OLED_DrawStringAlign(&oled, 0, 0, buf, ALIGN_LEFT);
            snprintf(buf, sizeof(buf), "PM2.5: %d", data.pm2_5_std);
            OLED_DrawStringAlign(&oled, 0, 10, buf, ALIGN_LEFT);
            snprintf(buf, sizeof(buf), "PM10: %d", data.pm10_std);
            OLED_DrawStringAlign(&oled, 0, 20, buf, ALIGN_LEFT);
            OLED_Update(&oled);
        }
        Delay_Ms(1000);
    }
}
