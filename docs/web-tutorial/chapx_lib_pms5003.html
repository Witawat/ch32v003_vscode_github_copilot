PD6(RX)─TX(sensor), PD5(TX)─RX(sensor) — USART 9600 baud, ไฟเลี้ยง 5V, Frame 32-byte แบบ binary

📚 หลักการทำงาน

PMS5003 เป็นเซนเซอร์วัดฝุ่นละออง PM1.0/PM2.5/PM10 แบบ laser scattering — พัดลมดูดอากาศผ่านลำแสงเลเซอร์ → วัดอนุภาคที่กระเจิงแสง — แม่นยำกว่าเซนเซอร์ IR ราคาถูก

32-byte Frame Protocol:

  • Start: 0x42 0x4D (2 bytes)
  • Frame Length: 2 bytes (always 0x00 0x1C = 28)
  • Data: 26 bytes — PM1.0, PM2.5, PM10 (หน่วย μg/m³), particle counts per 0.1L
  • Checksum: 2 bytes — sum of all previous bytes

Active vs Passive Mode:

  • Active: เซนเซอร์ส่ง data ทุก ~800ms อัตโนมัติ — ง่ายสุด ไม่ต้องควบคุม
  • Passive: ต้องส่ง command ขอ data ทุกครั้ง — ประหยัดไฟ, ควบคุมจังหวะได้

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Init + Setup

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "PMS5003.h"

int main(void) {
    SystemCoreClockUpdate();
    USART_Init(USART1, BAUD_9600);  // PMS5003 = 9600 fixed

    PMS5003_Data_t data;
    PMS5003_Init(PMS5003_MODE_ACTIVE);  // auto send ทุก ~800ms

🔍 PMS5003_Init(mode) — ACTIVE: เซนเซอร์ส่ง data อัตโนมัติ, PASSIVE: ต้องสั่ง PMS5003_RequestRead() ทุกครั้ง — ACTIVE ง่ายกว่าและแนะนำสำหรับงานทั่วไป

ขั้นที่ 2: อ่าน/เขียนค่า

    if(PMS5003_ReadFrame(&data)) {
        USART_Print("PM1.0: "); USART_PrintInt(data.pm1_0_std);
        USART_Print(" PM2.5: "); USART_PrintInt(data.pm2_5_std);
        USART_Print(" PM10: "); USART_PrintInt(data.pm10_std);
        USART_Println(" ug/m3");

        // เช็คคุณภาพอากาศ
        if(data.pm2_5_std <= 25)      USART_Println("✓ Air Quality: GOOD");
        else if(data.pm2_5_std <= 50) USART_Println("⚡ Air Quality: MODERATE");
        else                          USART_Println("⚠️ Air Quality: UNHEALTHY!");
    } else {
        USART_Println("Frame error / timeout");
    }

🔍 PMS5003_ReadFrame() รอ start byte 0x42 0x4D → อ่าน 32 bytes → verify checksum — คืน TRUE ถ้า valid — timeout ~2 วิ — frame error เกิดจาก baud ผิดหรือสายหลวม

ขั้นที่ 3: Loop + แสดงผล

    OLED_t oled;
    OLED_Init(&oled, 0x3C, OLED_128x64);

    while(1) {
        if(PMS5003_ReadFrame(&data)) {
            OLED_Clear(&oled);
            OLED_SetCursor(&oled, 0, 0);
            OLED_PrintFloat(&oled, "PM2.5:", (float)data.pm2_5_std, 0);
            OLED_Print(&oled, " ug/m3");

            OLED_SetCursor(&oled, 0, 2);
            if(data.pm2_5_std <= 25)      OLED_Print(&oled, "AIR: GOOD      ");
            else if(data.pm2_5_std <= 50) OLED_Print(&oled, "AIR: MODERATE  ");
            else                          OLED_Print(&oled, "AIR: UNHEALTHY!");

            OLED_Display(&oled);
        }
        Delay_Ms(1000);  // PMS5003 update ~800ms
    }
}

🔍 PMS5003 อัพเดททุก ~800ms ใน active mode — หน่วง 1 วิเพื่อ sync — แสดงผลบน OLED ทำให้พกพาได้ เป็นเครื่องวัดฝุ่นขนาดเล็ก

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
ReadFrame คืน FALSE ตลอดBaud ผิด (ต้อง 9600) หรือ RX/TX สลับเช็ค PD6→TX(sensor), PD5→RX(sensor), baud=9600
ค่าที่อ่านได้เกินจริงมากรบกวนจากพัดลมเซนเซอร์ (เสียง/แรงสั่น)แยกเซนเซอร์จากแหล่งสั่น, ใช้ housing ป้องกันลม
PM2.5 = 0 ตลอดเวลาActive mode แต่ USART buffer ยังไม่เต็มรออย่างน้อย 1 วิหลัง init, frame มาทุก ~800ms
Checksum error บ่อยUSART noise หรือสายยาวเกินใช้สายสั้น (<30cm), ลด baud, ต่อ GND ให้แน่น
เซนเซอร์ร้อน/พัดลมเสียงดังผิดปกติพัดลมสกปรกหรือลูกปืนเสื่อมทำความสะอาดพัดลม, เปลี่ยนเซนเซอร์ใหม่ (อายุ ~8000 ชม.)

🛠️ สร้าง Product: เครื่องวัดฝุ่น PM2.5 พกพา

PMS5003 + OLED — แสดง PM2.5 แบบ real-time พร้อม AQI ระดับสี

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "PMS5003.h"
#include "OLED.h"

int main(void)
{
    SystemCoreClockUpdate();
    USART_Init(USART1, BAUD_9600);
    I2C_Init(I2C1, I2C_SPEED_STANDARD);

    OLED_t oled;
    OLED_Init(&oled, 0x3C, OLED_128x64);

    PMS5003_Data_t data;
    PMS5003_Init(PMS5003_MODE_ACTIVE);

    OLED_Clear(&oled);
    OLED_SetCursor(&oled, 0, 0);
    OLED_Print(&oled, "PM2.5 Monitor");
    OLED_SetCursor(&oled, 0, 2);
    OLED_Print(&oled, "Warming up...");
    OLED_Display(&oled);
    Delay_Ms(5000);  // รอเซนเซอร์ warm-up

    while(1)
    {
        if(PMS5003_ReadFrame(&data)) {
            OLED_Clear(&oled);

            OLED_SetCursor(&oled, 0, 0);
            OLED_PrintFloat(&oled, "PM1.0:", (float)data.pm1_0_std, 0);
            OLED_Print(&oled, " ug/m3");

            OLED_SetCursor(&oled, 0, 1);
            OLED_PrintFloat(&oled, "PM2.5:", (float)data.pm2_5_std, 0);
            OLED_Print(&oled, " ug/m3");

            OLED_SetCursor(&oled, 0, 2);
            OLED_PrintFloat(&oled, "PM10:", (float)data.pm10_std, 0);
            OLED_Print(&oled, " ug/m3");

            OLED_SetCursor(&oled, 0, 4);
            if(data.pm2_5_std <= 12)       OLED_Print(&oled, "");
            else if(data.pm2_5_std <= 35)   OLED_Print(&oled, "");
            else if(data.pm2_5_std <= 55)   OLED_Print(&oled, "");
            else                            OLED_Print(&oled, "");

            OLED_Display(&oled);
        }
        Delay_Ms(1000);
    }
}
