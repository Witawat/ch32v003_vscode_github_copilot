74HC595 ใช้ 3 ขา (DATA, CLK, LATCH) — 8 outputs ต่อ chip — ต่อพ่วงหลายตัวได้ — output กระแส ~6mA ต่อขา — ใช้ external 5V ได้ (CH32V003 3.3V logic ใช้ได้กับ 74HC595 5V)

📚 หลักการทำงาน

Serial-In Parallel-Out (SIPO): 74HC595 รับข้อมูล serial 1 bit ต่อครั้งผ่านขา DATA — CLK ใช้ shift ข้อมูลเข้า register — LATCH pulse คัดลอกข้อมูลจาก shift register ไป output register — output เปลี่ยนพร้อมกันทุกขา

8 Outputs per Chip: แต่ละ 74HC595 มี 8 outputs (Q0-Q7) — เขียนค่า 8-bit (0-255) เข้า shift register — Latch() ส่ง LATCH pulse — LEDs เปลี่ยนพร้อมกัน (no flicker)

Chain Multiple Chips: ต่อ Q7' (pin 9) ของตัวแรกเข้า DATA ของตัวที่สอง — ส่งข้อมูล 16-bit ต่อเนื่อง — ใช้ count parameter ในการ Init — ShiftReg595_WriteByte หลายครั้งแล้ว Latch ทีเดียว

No PWM: 74HC595 เป็น digital output เท่านั้น — ไม่มี PWM ในตัว — ถ้าต้องการ PWM ต้องใช้ software PWM หรือใช้ GPIO ของ MCU โดยตรง — แต่ต่อ LEDs มากกว่า 8 ดวงด้วย GPIO 3 ขาเท่านั้น

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: สร้าง Instance + Init

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "ShiftReg595.h"

ShiftReg595_Instance sr;

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();

    ShiftReg595_Init(&sr, PC0, PC1, PC2, 1);  // DATA, CLK, LATCH, 1 chip

🔍 ShiftReg595_Init(&sr, data, clk, latch, count) — ใช้ instance-based — DATA=PC0, CLK=PC1, LATCH=PC2 — count=1 สำหรับ 1 ชิป — 3.3V logic ใช้กับ 74HC595 ได้

ขั้นที่ 2: เขียนค่า + Latch

    // LED สลับฟันปลา
    ShiftReg595_WriteByte(&sr, 0x55);  // 0b01010101
    ShiftReg595_Latch(&sr);
    Delay_Ms(500);

    // LED ตรงข้าม
    ShiftReg595_WriteByte(&sr, 0xAA);  // 0b10101010
    ShiftReg595_Latch(&sr);
    Delay_Ms(500);

    // ควบคุมทีละ bit
    ShiftReg595_WriteByte(&sr, 0x00);
    ShiftReg595_SetBit(&sr, 0, 1);  // Q0 = HIGH
    ShiftReg595_SetBit(&sr, 7, 1);  // Q7 = HIGH
    ShiftReg595_Latch(&sr);

    // 2 chips: WriteByte 2 ครั้ง
    ShiftReg595_WriteByte(&sr, 0x0F);  // chip 2 (ตัวหลัง)
    ShiftReg595_WriteByte(&sr, 0xF0);  // chip 1 (ตัวหน้า)
    ShiftReg595_Latch(&sr);

🔍 ShiftReg595_WriteByte(&sr, value) เขียน 8-bit — ShiftReg595_Latch(&sr) เพื่อ latch output — ShiftReg595_SetBit(&sr, bit, val) เปลี่ยนทีละบิต — ต่อพ่วง 2 ชิป: เขียน 2 ครั้งแล้ว Latch ทีเดียว

ขั้นที่ 3: LED Chaser Binary Counter

    uint8_t counter = 0;

    while(1) {
        ShiftReg595_WriteByte(&sr, counter);
        ShiftReg595_Latch(&sr);
        Delay_Ms(200);
        counter++;
    }
}

🔍 Binary counter 0-255 — LEDs แสดงผลเป็น binary — 8-bit counter วิ่งวน — Delay_Ms 200ms ต่อขั้น — ดูจังหวะ shift อย่างชัดเจน

🛠️ สร้าง Product: LED Chaser 8 ดวง

LED 8 ดวงวิ่งไปมา (Knight Rider / Larson Scanner) — ใช้ Non-Blocking Timer

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "ShiftReg595.h"

ShiftReg595_Instance sr;

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    ShiftReg595_Init(&sr, PC0, PC1, PC2, 1);

    Timer_t tmr;
    Start_Timer(&tmr, 100, 1);
    int8_t pos = 0, dir = 1;

    while(1) {
        if(Is_Timer_Expired(&tmr)) {
            uint8_t pattern = 1 << pos;
            ShiftReg595_WriteByte(&sr, pattern);
            ShiftReg595_Latch(&sr);

            pos += dir;
            if(pos >= 7 || pos <= 0) dir = -dir;
        }
    }
}

✏️ ลองทำดู

  1. Binary Counter — นับ 0-255 แสดงผลบน LED 8 ดวง — ใช้ Non-Blocking Timer 100ms
  2. Knight Rider — LED วิ่งไป-กลับ 8 ดวง — smooth acceleration ด้วย delay แปรผัน
  3. Chain 2 Chips — ต่อ 16 LEDs — ใช้ count=2 — ทำ pattern 16-bit (เช่น ไฟวิ่ง 2 ดวงสวนกัน)
