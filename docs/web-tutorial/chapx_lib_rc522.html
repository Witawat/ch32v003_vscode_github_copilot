รู้จักขา

  • VCC: ไฟเลี้ยง 3.3V เท่านั้น — ห้าม 5V!
  • GND: กราวด์
  • SCK: สัญญาณนาฬิกา SPI — ต่อขา PC5
  • MISO: ข้อมูลจาก RC522 ส่งเข้า MCU — ต่อขา PC6
  • MOSI: ข้อมูลจาก MCU เข้า RC522 — ต่อขา PC7
  • SDA (CS): เลือกชิป (Chip Select) — ต่อขา PC3
  • RST: รีเซ็ต — ต่อขา PC4
  • IRQ: สัญญาณขัดจังหวะ (ไม่ต้องต่อก็ได้) — ต่อขา PC0

วิธีต่อทีละขั้น

  1. ต่อ VCC RC522 ไป 3.3V ของ MCU
  2. ต่อ GND RC522 ไป GND ของ MCU
  3. ต่อ SCK RC522 ไปขา PC5 ของ MCU
  4. ต่อ MISO RC522 ไปขา PC6 ของ MCU
  5. ต่อ MOSI RC522 ไปขา PC7 ของ MCU
  6. ต่อ SDA (CS) RC522 ไปขา PC3 ของ MCU
  7. ต่อ RST RC522 ไปขา PC4 ของ MCU
  8. (ไม่บังคับ) ต่อ IRQ RC522 ไปขา PC0 ของ MCU

RC522 เป็น RFID reader/writer 13.56MHz — อ่านบัตร MIFARE Classic 1K/4K, ISO14443A, รองรับการเข้ารหัส (authentication) ด้วย key A/B, ต้องทำ self-test เพื่อตรวจสอบการเชื่อมต่อ SPI

📚 หลักการทำงาน

MFRC522 เป็น RFID IC ที่สื่อสารผ่าน SPI — รองรับ ISO/IEC 14443A/MIFARE, ระยะอ่าน ~5-8cm กับบัตรมาตรฐาน, ใช้ไฟ 3.3V

ขั้นตอนอ่านบัตร MIFARE Classic:

  • Request (REQA): ส่งคำสั่งถามว่ามีบัตรอยู่ในสนามหรือไม่
  • Anticollision (AntiColl): อ่าน UID (Unique ID) 4 หรือ 7 bytes
  • Select Tag: เลือกบัตรที่จะสื่อสาร (กรณีหลายใบทับกัน)
  • Authentication: พิสูจน์ตัวตนด้วย Key A หรือ Key B (default: 0xFFFFFFFFFFFF)
  • Read/Write Block: อ่านหรือเขียนข้อมูล block ละ 16 bytes

MIFARE Classic 1K: 1024 bytes แบ่งเป็น 16 sectors, sector ละ 4 blocks, block สุดท้ายของแต่ละ sector = Sector Trailer (เก็บ key + access bits)

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Init + SelfTest

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "RC522.h"

RC522_Instance rfid;

int main(void) {
    SystemCoreClockUpdate();
    SPI_Init(SPI1, SPI_BAUD_4MHZ, SPI_MODE0);
    RC522_Init(&rfid, PC4, PC3);  // CS=PC4, RST=PC3
    USART_Init(USART1, BAUD_115200);

    if(RC522_SelfTest()) {
        USART_Println("RC522 Ready!");
    } else {
        USART_Println("RC522 SelfTest FAILED!");
        while(1);
    }

🔍 RC522_Init(&rfid, CS, RST) — CS=PC4, RST=PC3, RC522_SelfTest() — ทำ self-test ภายใน RC522 ตรวจสอบ SPI communication และ register response

ขั้นที่ 2: ตรวจจับบัตร + อ่าน UID

    uint8_t uid[10];
    while(1) {
        if(RC522_IsCardPresent(&rfid)) {
            USART_Println("Card detected!");
            RC522_ReadUID(&rfid, uid);

            USART_Print("UID: ");
            for(uint8_t i=0; i<4; i++) {
                USART_PrintHex(uid[i]); USART_Print(" ");
            }
            USART_Println("");

            RC522_SelectTag(uid, 4);
        }
        Delay_Ms(200);
    }
}

🔍 RC522_IsCardPresent(&rfid) — ตรวจจับบัตร, RC522_ReadUID(&rfid, uid) — อ่าน UID 4-7 byte

ขั้นที่ 3: Authenticate + Read/Write Block

    // Default Key A for MIFARE Classic (new cards)
    uint8_t default_key[6] = {0xFF,0xFF,0xFF,0xFF,0xFF,0xFF};
    uint8_t block = 4;  // sector 1, block 0 (first user block)
    uint8_t buf[16];

    // Auth with Key A
    if(RC522_Auth(PICC_AUTHENT1A, block, default_key, uid) == MI_OK) {
        USART_Println("Auth OK!");

        // Read block 4 (16 bytes)
        uint8_t size = 16;
        if(RC522_Read(block, buf, &size) == MI_OK) {
            USART_Print("Block 4: ");
            for(int i=0; i<16; i++) {
                USART_PrintHex(buf[i]); USART_Print(" ");
            }
        }

        // Write to block 4
        uint8_t data[16] = "HELLO CH32V003!";
        if(RC522_Write(block, data) == MI_OK) {
            USART_Println("Write OK!");
        }

        RC522_Halt(&rfid);  // put card to sleep
    }

🔍 RC522_Auth(mode, block, key, uid) — authenticate ด้วย Key A หรือ B ก่อนเขียน/อ่าน, RC522_Read(block, buf, &len) — อ่าน 16 bytes, RC522_Write(block, data) — เขียน 16 bytes, RC522_Halt(&rfid) — สั่งการ์ด sleep

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
SelfTest ล้มเหลวSPI mode ผิด หรือ CS/RST สลับต้องใช้ SPI_MODE0 (CPOL=0,CPHA=0), เช็ค SPI clock <10MHz
ไม่เจอการ์ดเลยไฟ 3.3V ไม่พอ — RC522 กิน ~100mAใช้ external 3.3V regulator, อย่าจ่ายจาก WCH-Link
Auth ล้มเหลว (MI_ERR)Key ไม่ถูกต้องการ์ดใหม่ใช้ default key 0xFFFFFFFFFFFF — ถ้าเปลี่ยนแล้วต้องรู้ key
เขียนแล้ว block อ่านไม่ได้อีกเขียนทับ Sector Trailer โดยไม่ตั้งใจBlock 3,7,11,15... = Sector Trailer — ห้ามเขียนทับโดยไม่ตั้งใจ
อ่าน UID ได้แต่ Auth ไม่ผ่านลืม SelectTag ก่อน Authต้องเรียก RC522_SelectTag() หลังจาก Anticoll และก่อน Auth

🛠️ สร้าง Product: RFID Access Control

อ่าน UID บัตร → เทียบกับ whitelist → เปิด servo (ปลดล็อค) ถ้าตรง

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "RC522.h"
#include 

uint8_t whitelist[2][4] = {
    {0xA1,0xB2,0xC3,0xD4},  // card 1
    {0xE5,0xF6,0xA7,0xB8}   // card 2
};

int main(void)
{
    SystemCoreClockUpdate();
    SPI_Init(SPI1, SPI_BAUD_4MHZ, SPI_MODE0);
    RC522_Instance rfid;
    RC522_Init(&rfid, PC4, PC3);
    USART_Init(USART1, BAUD_115200);
    pinMode(PC0, PIN_MODE_OUTPUT);  // lock solenoid

    if(!RC522_SelfTest()) { USART_Println("RC522 FAIL"); while(1); }

    uint8_t uid[10];
    while(1)
    {
        if(!RC522_IsCardPresent(&rfid)) {
            Delay_Ms(100); continue;
        }
        RC522_ReadUID(&rfid, uid);
        RC522_SelectTag(uid, 4);

        USART_Print("UID: ");
        for(int i=0;i<4;i++){ USART_PrintHex(uid[i]); }

        uint8_t access = 0;
        for(int i=0;i<2;i++) {
            if(memcmp(uid, whitelist[i], 4) == 0) {
                access = 1; break;
            }
        }
        if(access) {
            USART_Println(" => ACCESS GRANTED");
            digitalWrite(PC0, HIGH); Delay_Ms(3000);
            digitalWrite(PC0, LOW);
        } else {
            USART_Println(" => ACCESS DENIED");
        }
        RC522_Halt(&rfid);
        Delay_Ms(500);
    }
}

✏️ ลองทำดู

  1. ทำ RFID wallet — อ่าน balance จาก block 4 (เก็บเป็น uint32) → ลดค่าเมื่อแตะ → เขียนกลับ
  2. ทำ multi-card reader — วนอ่าน UID ทุก 200ms → แสดง UID การ์ดล่าสุดบน OLED
  3. ทำ key changer — เปลี่ยน default key ของ sector 1 เป็น custom key → ใช้ new key auth ครั้งต่อไป
