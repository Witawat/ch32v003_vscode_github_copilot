📚 หลักการทำงาน

RCWL-0516 ใช้หลักการ Doppler Radar — ส่งคลื่นไมโครเวฟ 3.18 GHz (10 cm wavelength) → กระทบวัตถุเคลื่อนที่ → ความถี่สะท้อนเปลี่ยนตามความเร็ววัตถุ → วงจรตรวจจับ frequency shift → output HIGH

ข้อดีเหนือ PIR:

  • ทะลุผ่านวัสดุ: ตรวจจับผ่านกำแพงไม้, แผ่นยิปซัม, กระจก (แต่ไม่ทะลุโลหะ/คอนกรีตหนา)
  • ไม่ต้องใช้ lens: ไม่มี Fresnel lens → ไม่ต้องหาทิศทาง
  • ตรวจจับได้แม้วัตถุไม่เคลื่อนที่ผ่านโซน: ตรวจจับการเคลื่อนที่เข้าหา/ออกห่าง (Doppler effect)
  • ไม่ไวต่อความร้อน: ไม่มี false trigger จากแหล่งความร้อน (ต่างจาก PIR)

ข้อเสีย: ตรวจจับได้ไกลกว่า → false trigger มากขึ้นจากสิ่งแวดล้อม — กินไฟมากกว่า PIR (~30mA vs 0.05mA)

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Init + Set Callbacks

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "RCWL0516.h"

RCWL0516_Instance r;
volatile uint8_t radarFlag = 0;

void on_motion(void) {
    radarFlag = 1;
}

void on_no_motion(void) {
    radarFlag = 0;
}

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_Init(USART1, BAUD_115200);

    pinMode(PC1, PIN_MODE_OUTPUT);  // Buzzer/LED

    // RCWL-0516: instance-based init
    RCWL0516_Init(&r, PC0);
    RCWL0516_SetCallbacks(&r, on_motion, on_no_motion);
    USART_Println("RCWL-0516 Ready");

🔍 RCWL0516_Init(&r, pin) เริ่มต้นแบบ instance — RCWL0516_SetCallbacks(&r, on_motion, on_no_motion) ตั้ง 2 callback: เมื่อตรวจจับ และเมื่อไม่ตรวจจับ — OUT pin อาจ pulse สั้น (~2-3ms)

ขั้นที่ 2: Update + ตรวจจับ

    while(1) {
        RCWL0516_Update(&r);

        if(radarFlag) {
            USART_Print("Motion @ ");
            USART_PrintInt(Get_CurrentMs());
            USART_Println(" ms");
            digitalWrite(PC1, HIGH);  // Alert
        }

        // เช็ค motion โดยตรง
        if(RCWL0516_IsMotionDetected(&r)) {
            // มีวัตถุเคลื่อนที่
        }

        Delay_Ms(50);
    }
}

🔍 RCWL0516_Update(&r) ต้องเรียกใน loop — RCWL0516_IsMotionDetected(&r) เช็คสถานะโดยตรง — callbacks on_motion/on_no_motion ถูกเรียกตามสถานะ

ขั้นที่ 3: Loop + แสดงผล

    Timer_t tmr_debounce;
    uint32_t lastTrigger = 0;
    uint16_t totalTriggers = 0;

    while(1) {
        RCWL0516_Update(&r);

        if(radarFlag) {
            radarFlag = 0;

            // Debounce: ignore triggers within 500ms
            if(Get_CurrentMs() - lastTrigger > 500) {
                lastTrigger = Get_CurrentMs();
                totalTriggers++;

                USART_Print("Trigger #");
                USART_PrintInt(totalTriggers);
                USART_Println("");

                digitalWrite(PC1, HIGH);  // LED ON
                Start_Timer(&tmr_debounce, 200, 0);
            }
        }

        if(Is_Timer_Expired(&tmr_debounce)) {
            digitalWrite(PC1, LOW);  // LED auto off
        }

        Delay_Ms(50);
    }
}

🔍 Debounce สำคัญ — RCWL-0516 อาจ trigger หลายครั้งต่อ 1 motion (multi-path reflection) — ใช้ timeout 500ms กรอง trigger ซ้ำ — LED ติด 200ms ทุกครั้งที่ตรวจจับได้

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
Trigger ตลอดเวลาไม่หยุดRange/sensitivity สูงเกิน, มีแหล่ง RF ใกล้ลด range ด้วย jumper, ย้าย WiFi router ห่างออกไป
Trigger หลายครั้งต่อ 1 การเคลื่อนที่Multi-path reflection (คลื่นสะท้อนหลายทาง)ใช้ debounce 500ms-1s, ลด sensitivity
ไม่ trigger แม้คนเดินอยู่CDS (light sensor) ปิดกั้นการตรวจจับตอนสว่างถอด jumper CDS (disable light sensor feature)
Trigger แม้ไม่มีใครอยู่ในห้องทะลุกำแพงตรวจจับคนข้างนอกลด range, หันเซนเซอร์ออกจากผนัง, ใช้ PIR แทน
รบกวน WiFi 2.4GHzRCWL-0516 ใช้ 3.18GHz → harmonics ใกล้ 2.4GHzวางเซนเซอร์ห่างจาก WiFi antenna อย่างน้อย 1 เมตร

🛠️ สร้าง Product: Radar Security System

RCWL-0516 ตรวจจับผ่านกำแพง + Buzzer + LED + Count log

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "RCWL0516.h"
#include "oled_i2c.h"

RCWL0516_Instance r;
volatile uint8_t radarFlag = 0;
void on_motion(void) { radarFlag = 1; }
void on_no_motion(void) { radarFlag = 0; }

int main(void)
{
    SystemCoreClockUpdate();
    Timer_Init();
    I2C_Init(I2C1, I2C_SPEED_STANDARD);

    OLED_Handle oled;
    OLED_Init(&oled, OLED_128x64, 0x3C);

    pinMode(PC1, PIN_MODE_OUTPUT);  // Buzzer
    RCWL0516_Init(&r, PC0);
    RCWL0516_SetCallbacks(&r, on_motion, on_no_motion);

    uint16_t count = 0;
    uint32_t lastTrigger = 0;

    char buf[64];
    OLED_Clear(&oled);
    snprintf(buf, sizeof(buf), "Radar Security");
    OLED_DrawStringAlign(&oled, 0, 0, buf, ALIGN_LEFT);
    snprintf(buf, sizeof(buf), "Active...");
    OLED_DrawStringAlign(&oled, 0, 20, buf, ALIGN_LEFT);
    OLED_Update(&oled);
    Delay_Ms(2000);

    while(1)
    {
        RCWL0516_Update(&r);

        if(radarFlag) {
            radarFlag = 0;

            if(Get_CurrentMs() - lastTrigger > 1000) {
                lastTrigger = Get_CurrentMs();
                count++;

                digitalWrite(PC1, HIGH);  // Buzzer ON
                Delay_Ms(300);
                digitalWrite(PC1, LOW);

                OLED_Clear(&oled);
                snprintf(buf, sizeof(buf), "ALERT!");
                OLED_DrawStringAlign(&oled, 0, 0, buf, ALIGN_LEFT);
                snprintf(buf, sizeof(buf), "Motion #%d", count);
                OLED_DrawStringAlign(&oled, 0, 20, buf, ALIGN_LEFT);
                OLED_Update(&oled);

                USART_Print("Alert #");
                USART_PrintInt(count);
                USART_Println("");
            }
        }

        Delay_Ms(100);
    }
}
