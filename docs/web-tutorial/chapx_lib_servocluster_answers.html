📝 เฉลย — Servo Cluster

ข้อ 1: Robot Arm 3-DOF Pick & Place

#include "ServoCluster.h"

ServoCluster_Instance arm;

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    ServoCluster_Init(&arm, SERVOCLUSTER_PCA9685, 3);
    ServoCluster_AddServo(&arm, 0);  // base
    ServoCluster_AddServo(&arm, 1);  // shoulder
    ServoCluster_AddServo(&arm, 2);  // gripper

    while (1) {
        // Home position
        ServoCluster_MoveTo(&arm, 0, 90,  EASE_LINEAR);
        ServoCluster_MoveTo(&arm, 1, 45,  EASE_LINEAR);
        ServoCluster_MoveTo(&arm, 2, 160, EASE_LINEAR);
        while (!ServoCluster_IsAllDone(&arm)) { Delay_Ms(5); }
        Delay_Ms(300);

        // Reach for object
        ServoCluster_MoveTo(&arm, 0, 135, EASE_IN_OUT_CUBIC);
        ServoCluster_MoveTo(&arm, 1, 30,  EASE_IN_OUT_CUBIC);
        ServoCluster_MoveTo(&arm, 2, 160, EASE_LINEAR);
        while (!ServoCluster_IsAllDone(&arm)) { Delay_Ms(5); }
        Delay_Ms(200);

        // Gripper close
        ServoCluster_MoveTo(&arm, 2, 90, EASE_LINEAR);
        while (!ServoCluster_IsAllDone(&arm)) { Delay_Ms(5); }

        // Lift
        ServoCluster_MoveTo(&arm, 0, 90,  EASE_IN_OUT_CUBIC);
        ServoCluster_MoveTo(&arm, 1, 135, EASE_IN_OUT_CUBIC);
        while (!ServoCluster_IsAllDone(&arm)) { Delay_Ms(5); }
        Delay_Ms(200);

        // Place
        ServoCluster_MoveTo(&arm, 0, 45,  EASE_IN_OUT_CUBIC);
        ServoCluster_MoveTo(&arm, 1, 30,  EASE_IN_OUT_CUBIC);
        while (!ServoCluster_IsAllDone(&arm)) { Delay_Ms(5); }

        // Gripper open
        ServoCluster_MoveTo(&arm, 2, 160, EASE_LINEAR);
        while (!ServoCluster_IsAllDone(&arm)) { Delay_Ms(5); }

        Delay_Ms(1000);
    }
}
🧠 แนวคิด: ใช้ ServoCluster จัดการ 3 servo พร้อมกัน — base หมุนซ้าย-ขวา, shoulder ก้ม-เงย, gripper จับ-ปล่อย — MoveTo แบบ non-blocking เรียก 3 ตัวพร้อมกันได้ — ใช้ IsDone() เช็คว่าทุก servo เคลื่อนที่เสร็จก่อนไปขั้นตอนถัดไป — เลือก easing ต่างกัน: LINEAR สำหรับ gripper (แรงบิดตรง ๆ), CUBIC สำหรับ arm (ลดแรงกระชาก) — ต้องรอ delay ระหว่างท่าเพื่อให้ servo มีเวลาจับวัตถุจริง

ข้อ 2: Synchronized Movement — หลาย Servo ขยับพร้อมกัน

#include "ServoCluster.h"

ServoCluster_Instance cluster;

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    ServoCluster_Init(&cluster, SERVOCLUSTER_PCA9685, 4);
    for (uint8_t i = 0; i < 4; i++)
        ServoCluster_AddServo(&cluster, i);

    uint8_t wave[4] = {0, 45, 90, 135};
    int8_t  dir[4]  = {1, 1, 1, 1};

    while (1) {
        for (uint8_t i = 0; i < 4; i++) {
            ServoCluster_MoveTo(&cluster, i, wave[i], EASE_LINEAR);

            wave[i] += dir[i];
            if (wave[i] >= 180) { wave[i] = 180; dir[i] = -1; }
            if (wave[i] == 0)   { dir[i] = 1; }
        }
        Delay_Ms(30);
    }
}
🧠 แนวคิด: เรียก MoveTo ให้ทุก servo ทุกรอบ loop — ใช้ array เก็บค่ามุมและทิศทางของแต่ละตัว — แต่ละ servo เริ่มที่มุมต่างกันเล็กน้อย ทำให้เกิดเอฟเฟกต์คลื่น (wave) — MoveTo แบบ non-blocking ทำให้ทุกตัวขยับพร้อมกันโดยไม่ต้องรอตัวใดตัวหนึ่ง — ถ้าต้องการ synchronized ที่แม่นยำระดับเฟรม ให้ใช้ ServoCluster_SyncUpdate หรือคำนวณ trajectory ล่วงหน้าใน buffer — เพิ่ม condition เช็ค IsDone เพื่อเปลี่ยนท่าถัดไปเมื่อทุกตัวถึงเป้าหมาย

ข้อ 3: Easing Comparison — LINEAR vs EASE_IN_OUT_CUBIC

#include "ServoCluster.h"

ServoCluster_Instance cluster;

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    ServoCluster_Init(&cluster, SERVOCLUSTER_PCA9685, 2);
    ServoCluster_AddServo(&cluster, 0);
    ServoCluster_AddServo(&cluster, 1);

    const uint8_t targets[] = {0, 180};
    uint8_t idx = 0;

    while (1) {
        // Servo 0: LINEAR (ความเร็วคงที่)
        ServoCluster_MoveTo(&cluster, 0, targets[idx], EASE_LINEAR);
        // Servo 1: EASE_IN_OUT_CUBIC (เร่ง-ชะลอ นุ่มนวล)
        ServoCluster_MoveTo(&cluster, 1, targets[idx], EASE_IN_OUT_CUBIC);

        while (!ServoCluster_IsAllDone(&cluster))
            Delay_Ms(5);

        Delay_Ms(500);
        idx ^= 1;  // toggle 0 ↔ 1
    }
}
🧠 แนวคิด: ใช้ servo 2 ตัวเดินทางจาก 0↔180 เหมือนกัน แต่คนละ easing — LINEAR มีความเร็วคงที่ตลอดการเดินทาง กระชากตอนเริ่มและหยุด — EASE_IN_OUT_CUBIC เร่งช้า → เร็วกลางทาง → ชะลอก่อนถึง รู้สึกนุ่มนวลกว่า — สังเกตด้วยตาเปล่า: LINEAR จะกระตุกตอนออกตัว EASE จะลื่นไหล — ในการใช้งานจริง: LINEAR เหมาะกับ gripper/linear slide, EASE เหมาะกับ camera pan-tilt/robot arm — ความเร็วรวม (ระยะเวลา) ควรเท่ากันทั้ง 2 easing เพื่อเปรียบเทียบได้ยุติธรรม
