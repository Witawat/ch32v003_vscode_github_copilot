รู้จักขา

  • PC4 (NSS): ขาเลือกอุปกรณ์ (Chip Select) — LOW = เลือก
  • PC5 (SCK): ขานาฬิกา SPI
  • PC6 (MOSI): ขาส่งข้อมูล — Master Out Slave In
  • PC7 (MISO): ขารับข้อมูล — Master In Slave Out

วิธีต่อทีละขั้น

  1. ต่อ PC4 (CS) ไป CS ของ MAX7219 และ /CS ของ W25Qxx — ใช้คนละ pin ถ้ามีหลายอุปกรณ์
  2. ต่อ PC5 (SCK) ไป CLK ของ MAX7219 และ CLK ของ W25Qxx
  3. ต่อ PC6 (MOSI) ไป DIN ของ MAX7219 และ DI ของ W25Qxx
  4. ต่อ PC7 (MISO) ไป DO ของ W25Qxx — MAX7219 ไม่ใช้ MISO
  5. ต่อ 3.3V ไป /WP, /HOLD, VCC ของ W25Qxx
  6. ต่อ GND ร่วมกันทุกตัว

SOP-8/SOP-16 ไม่มี Hardware SPI! — ต้องใช้ shiftOut() / shiftIn() แบบ software แทน

📚 หลักการทำงาน

SPI (Serial Peripheral Interface) — การสื่อสาร synchronous แบบ full-duplex ผ่าน 4 สาย (SCK, MOSI, MISO, CS)

SPI Mode (CPOL/CPHA):

  • Mode 0: CPOL=0, CPHA=0 — Clock idle LOW, sample rising edge (ใช้บ่อยสุด)
  • Mode 3: CPOL=1, CPHA=1 — Clock idle HIGH, sample rising edge

CS (Chip Select) — แต่ละอุปกรณ์ต้องมี CS แยก — HIGH = ไม่เลือก, LOW = เลือก — ต้องควบคุมเอง

SPI vs I2C: SPI เร็วกว่า (สูงสุด 24MHz vs 400kHz) ใช้ 4 สาย, full-duplex, ไม่มี addressing ในโปรโตคอล — หมายเหตุ: enum SPI_12MHZ ใน SimpleSPI.h อิงจากสมมติฐาน PCLK2=24MHz แต่จริง PCLK2=48MHz (HSI+PLL) ทำให้ความเร็วจริงเป็น 2 เท่าของชื่อ enum (ดู SimpleSPI.h:64-79)

Full-Duplex: ทุกครั้งที่ส่ง 1 byte จะได้รับกลับมา 1 byte พร้อมกัน — SPI_Transfer() ทำทั้งส่งและรับ

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Init + Setup CS

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
    SPI_SimpleInit(SPI_MODE0, SPI_1MHZ, SPI_PINS_DEFAULT);

    pinMode(PC4, PIN_MODE_OUTPUT);       // CS pin
    SPI_SetCSPin(GPIOC, GPIO_Pin_4);     // บอก SPI ว่าใช้ PC4 เป็น CS

🔍 SPI_SimpleInit(mode, speed, pins) — เริ่มต้น SPI hardware — CS ต้องตั้งเอง! — pinMode() + SPI_SetCSPin()

ขั้นที่ 2: Loopback Test

    while(1) {
        SPI_SetCS(0);                    // LOW = select
        uint8_t rx = SPI_Transfer(0xAA); // ส่ง 0xAA รับกลับ
        SPI_SetCS(1);                    // HIGH = deselect

        USART_Print("Sent 0xAA, got 0x");
        USART_PrintHex(rx, 1);
        USART_Print("\r\n");
        Delay_Ms(1000);
    }
}

🔍 SPI_SetCS(0) — เลือกอุปกรณ์ (active LOW) — SPI_Transfer(tx) — ส่ง 1 byte พร้อมรับ 1 byte — SPI_SetCS(1) — ปล่อย CS (สำคัญ!)

ขั้นที่ 3: อ่าน JEDEC ID จาก W25Qxx Flash

    SPI_SetCS(0);
    SPI_Transfer(0x9F);  // Command: Read JEDEC ID
    uint8_t mfg = SPI_Transfer(0x00);
    uint8_t mem = SPI_Transfer(0x00);
    uint8_t cap = SPI_Transfer(0x00);
    SPI_SetCS(1);

    USART_Print("JEDEC ID: ");
    USART_PrintHex(mfg, 1); USART_Print(" ");
    USART_PrintHex(mem, 1); USART_Print(" ");
    USART_PrintHex(cap, 1); USART_Print("\r\n");

🔍 ส่ง command 0x9F แล้วอ่าน 3 bytes ตอบกลับ — SPI ให้ส่ง dummy byte (0x00) เพื่อสร้าง clock สำหรับรับข้อมูล

ขั้นที่ 4: MAX7219 (LED Driver)

    // MAX7219: ส่ง address + data (16-bit)
    SPI_SetCS(0);
    SPI_Transfer(0x0C);  // Shutdown register
    SPI_Transfer(0x01);  // Normal operation
    SPI_SetCS(1);

    SPI_SetCS(0);
    SPI_Transfer(0x01);  // Digit 0
    SPI_Transfer(0x7F);  // Show "8" (7-segment)

🔍 MAX7219 รับ 16-bit (address + data) — ส่ง 2 SPI_Transfer() ต่อเนื่อง โดย CS ค้าง LOW ตลอด

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
SPI ไม่ทำงานเลยลืมตั้งค่า CS pinต้อง pinMode(CS, OUTPUT) + SPI_SetCSPin()
รับข้อมูลผิดเพี้ยนSPI Mode ไม่ตรงกับอุปกรณ์เช็ค datasheet ว่าเป็น Mode 0,1,2,3
ข้อมูลส่งไม่ออกลืม SPI_SetCS(0) ก่อน transferCS ต้องเป็น LOW ก่อนเริ่ม transfer
SOP-8/SOP-16: SPI compile errorไม่มี Hardware SPISOP-8/16 ไม่มี SCK pin — ใช้ shiftOut()/shiftIn()
MAX7219 แสดงผลผิดลืม init registers (Shutdown, Intensity, Scan Limit)ต้อง config MAX7219 registers ก่อนส่งข้อมูล display
W25Qxx อ่าน ID ได้ 0xFFCS ไม่ถูกปล่อยก่อนส่ง commandCS HIGH → LOW → command → HIGH (ต้อง reset state machine)

🛠️ สร้าง Product: จอ LED 8 หลัก + อ่าน Flash ID

แสดง "12345678" บน MAX7219 8-digit display และแสดง JEDEC ID ของ W25Qxx Flash

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"

#define CS_FLASH  PC3
#define CS_MAX    PC4

void max7219_write(uint8_t addr, uint8_t data) {
    SPI_SetCSPin(GPIOC, GPIO_Pin_4);  // MAX7219 CS = PC4
    SPI_SetCS(0);
    SPI_Transfer(addr);
    SPI_Transfer(data);
    SPI_SetCS(1);
}

int main(void)
{
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
    SPI_SimpleInit(SPI_MODE0, SPI_1MHZ, SPI_PINS_DEFAULT);

    pinMode(CS_FLASH, PIN_MODE_OUTPUT);
    pinMode(CS_MAX, PIN_MODE_OUTPUT);

    // MAX7219 Init
    max7219_write(0x0C, 0x01);  // Shutdown: Normal
    max7219_write(0x09, 0x00);  // Decode: no decode
    max7219_write(0x0B, 0x07);  // Scan Limit: 8 digits
    max7219_write(0x0A, 0x04);  // Intensity: 9/32

    // Show digits 1-8
    const uint8_t seg[] = {0x06,0x5B,0x4F,0x66,0x6D,0x7D,0x07,0x7F};
    for(int i=0; i<8; i++) max7219_write(i+1, seg[i]);

    // Read W25Qxx JEDEC ID
    SPI_SetCSPin(GPIOC, GPIO_Pin_3);  // Flash CS = PC3
    SPI_SetCS(0);
    SPI_Transfer(0x9F);
    uint8_t id[3] = {SPI_Transfer(0), SPI_Transfer(0), SPI_Transfer(0)};
    SPI_SetCS(1);

    USART_Print("Flash JEDEC: ");
    USART_PrintHex(id[0],1); USART_Print(" ");
    USART_PrintHex(id[1],1); USART_Print(" ");
    USART_PrintHex(id[2],1); USART_Print("\r\n");

    while(1);
}

SOP-8 Alternative (Software SPI):

// SOP-8 ไม่มี Hardware SPI — ใช้ shiftOut แทน
#define DATA_PIN PC1
#define CLOCK_PIN PC2
#define LATCH_PIN PC4

digitalWrite(LATCH_PIN, LOW);
shiftOut(DATA_PIN, CLOCK_PIN, MSBFIRST, 0xAA);  // 74HC595
digitalWrite(LATCH_PIN, HIGH);

✏️ ลองทำดู

  1. เปลี่ยน SPI Mode เป็น Mode 3 (SPI_MODE3) — ทดสอบ loopback ยังได้ผลเหมือนกันหรือไม่
  2. ใช้ SPI_TransferBuffer() แทนการเรียก SPI_Transfer() ทีละ byte
  3. ถ้าใช้ SOP-8 — ทำวงจร 74HC595 + shiftOut() ควบคุม LED 8 ดวง
