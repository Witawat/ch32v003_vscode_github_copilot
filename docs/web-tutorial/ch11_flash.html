บทที่ 11: Flash Storage — เก็บข้อมูลไม่หายหลังปิดเครื่อง

ระดับสูง

🔌 การต่อวงจร

    Flash เป็นหน่วยความจำภายใน — ไม่ต้องต่อวงจรภายนอก

    CH32V003 Flash Layout (16KB Total)
    ┌────────────────────────────────────────────┐
    │ 0x08000000 ┌──────────────────────────────┐│
    │            │  Application Code (~12KB)    ││
    │            │                              ││
    │ 0x08003000 ├──────────────────────────────┤│
    │            │  User Config Storage (2KB)   ││
    │            │  Page 60: Settings           ││ 1 page = 1KB
    │            │  Page 61: Data Log           ││ (CH32V003 fast
    │ 0x08003FFF └──────────────────────────────┘│ erase mode)
    └────────────────────────────────────────────┘

    ข้อจำกัดของ Flash:
    - ต้อง Erase ก่อน Write (Erase ทีละ 1KB page)
    - Erase ใช้เวลา ~1ms ต่อ page
    - เขียนได้ทีละ 16-bit (half-word)
    - เขียนข้าม page boundary ไม่ได้ในครั้งเดียว
    - Endurance: 10,000+ cycles ต่อ page (WCH spec)
    - อ่านได้เร็วเท่า RAM (ไม่ต้อง unlock)

สำคัญ: ห้ามเขียนทับพื้นที่โค้ด (0x08000000-0x08003FFF โดยประมาณ) — ใช้เฉพาะพื้นที่ว่างที่กำหนดไว้

📚 หลักการทำงาน

Flash Memory — หน่วยความจำแบบ non-volatile — ข้อมูลไม่หายเมื่อไฟดับ — ใช้เก็บค่าคอนฟิก, calibration data, serial number, log ข้อมูล

Erase-Before-Write: หลักการสำคัญของ Flash — บิตใน Flash เปลี่ยนจาก 1 → 0 ได้โดยตรง แต่เปลี่ยน 0 → 1 ต้อง Erase ทั้ง page — Erase คือการคืนค่าทั้งหน้าเป็น 0xFF

SimpleHAL API แนว Struct: แทนที่จะเขียนทีละ byte — ใช้ Flash_WriteStruct() เขียน struct ทั้งก้อน — Flash_ReadStruct() อ่าน struct ทั้งก้อน — ปลอดภัยกว่าเพราะมีการตรวจสอบ size

CRC16: Flash_CalculateCRC16() — คำนวณ checksum 16-bit — ใช้ตรวจสอบความถูกต้องของข้อมูล — ถ้า CRC ไม่ตรง = ข้อมูลเสียหาย (อาจจากไฟดับระหว่างเขียน)

CH32V003 specific: Flash page size = 1KB (fast erase mode) หรือ 128 bytes (standard mode) — SimpleHAL ใช้ 1KB page

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: นิยาม Config Struct

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"

#define FLASH_CONFIG_ADDR  0x08003C00   // Page 60
#define FLASH_MAGIC        0x55AA

typedef struct {
    uint16_t magic;        // Magic number (ตรวจสอบว่ามีข้อมูล)
    uint8_t  brightness;   // 0-100
    uint8_t  volume;       // 0-100
    uint16_t crc;          // CRC16 checksum
} Config_t;

Config_t g_config;

🔍 ใช้ struct รวมค่าคอนฟิกทั้งหมด — magic ตรวจสอบว่า Flash มีข้อมูลหรือยัง (0xFFFF = empty) — crc ตรวจสอบความถูกต้องของข้อมูล

ขั้นที่ 2: Load Config เมื่อเริ่มต้น

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
    Flash_Init();

    if(Flash_LoadConfig((uint8_t*)&g_config, sizeof(g_config)))
    {
        USART_Print("Config loaded from Flash!\r\n");
    }
    else
    {
        // ไม่มี config — ใช้ default
        USART_Print("No saved config — using defaults\r\n");
        g_config.magic = FLASH_MAGIC;
        g_config.brightness = 50;
        g_config.volume = 80;
    }

🔍 Flash_LoadConfig(data, size) — อ่าน config จาก Flash — คืนค่า 1 ถ้าสำเร็จ (magic ตรง, CRC ผ่าน) — คืนค่า 0 ถ้าไม่มีข้อมูลหรือเสียหาย

ขั้นที่ 3: Save Config เมื่อแก้ไข

    // ผู้ใช้เปลี่ยน brightness
    g_config.brightness = 75;

    // คำนวณ CRC16
    g_config.crc = Flash_CalculateCRC16((uint8_t*)&g_config,
                        sizeof(g_config) - sizeof(uint16_t));

    // บันทึกลง Flash
    if(Flash_SaveConfig(FLASH_CONFIG_ADDR,
                        (uint8_t*)&g_config,
                        sizeof(g_config)))
    {
        USART_Print("Config saved!\r\n");
    }
    else
    {
        USART_Print("Save failed!\r\n");
    }

🔍 Flash_SaveConfig(addr, data, size) — Erase page อัตโนมัติ + เขียนข้อมูล + verify — คืนค่า 1 เมื่อสำเร็จ — สำคัญ: Flash_CalculateCRC16() ต้องไม่รวมฟิลด์ crc เอง

ขั้นที่ 4: อ่าน/เขียน Raw Flash

    // อ่าน raw Flash (เหมือนอ่าน RAM — ไม่ต้อง unlock)
    uint16_t val = *(uint16_t*)FLASH_CONFIG_ADDR;
    USART_Print("Raw value at 0x");
    USART_PrintHex(val >> 8, 1);
    USART_PrintHex(val & 0xFF, 1);
    USART_Print("\r\n");

    // เขียน raw (ต้อง erase ก่อน)
    Flash_ErasePage(FLASH_CONFIG_ADDR);
    uint16_t data = 0xABCD;
    Flash_WriteHalfWord(FLASH_CONFIG_ADDR, data);

🔍 Flash_ErasePage(addr) — ลบ 1KB page — Flash_WriteHalfWord(addr, data) — เขียน 16-bit — ใช้เมื่อต้องการควบคุมเอง

ขั้นที่ 5: Flash Read/Write Struct โดยตรง

    // เก็บ log ง่ายๆ ด้วย struct
    typedef struct {
        uint32_t timestamp;
        uint16_t temperature;
        uint16_t humidity;
    } LogEntry_t;

    LogEntry_t log = {.timestamp=Get_CurrentMs(), .temperature=250, .humidity=60};

    Flash_ErasePage(FLASH_LOG_ADDR);
    Flash_WriteStruct(FLASH_LOG_ADDR, (uint8_t*)&log, sizeof(log));

    // อ่านกลับ
    LogEntry_t readback;
    Flash_ReadStruct(FLASH_LOG_ADDR, (uint8_t*)&readback, sizeof(readback));

🔍 Flash_WriteStruct(addr, data, size) — เขียน struct ทั้งก้อน — Flash_ReadStruct(addr, data, size) — อ่าน struct ทั้งก้อน — สะดวกและปลอดภัยกว่าเขียนทีละ byte

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
Flash เขียนไม่ติดลืมเรียก Flash_Init()Flash ต้อง unlock ก่อนเขียน — เรียก Flash_Init() ก่อน
ข้อมูลเก่าไม่หายลืม Erase ก่อน WriteFlash ต้อง erase ก่อนเขียน — ใช้ Flash_SaveConfig() ที่จัดการให้อัตโนมัติ
Config โหลดไม่สำเร็จFlash ว่างเปล่า (0xFF) — magic ไม่ตรงเช็ค magic number — ถ้าไม่ตรงให้ใช้ค่า default
HardFault ตอนอ่าน Flashอ่าน address ที่ไม่มี Flash (นอก 0x08000000-0x08003FFF)เช็ค address — CH32V003 มี Flash 16KB (0x08000000-0x08003FFF)
เขียนแล้ว CRC ไม่ตรงไฟดับระหว่างเขียน Flashใช้ dual-page storage — เขียน 2 หน้า ตรวจสอบทั้งคู่
Flash เสื่อมเร็วErase/Write บ่อยเกินไป (ทุก loop)เขียนเฉพาะเมื่อค่าปลี่ยน — ใช้ RAM cache ระหว่างรัน

🛠️ สร้าง Product: ตั้งค่าผู้ใช้ที่จำได้ตลอด

อุปกรณ์ที่มี brightness/volume config — จำค่าได้แม้ปิดไฟ — มี factory reset

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"

#define CFG_ADDR   0x08003C00
#define CFG_MAGIC  0xA5A5

typedef struct {
    uint16_t magic;
    uint8_t  brightness;   // 0-100
    uint8_t  volume;       // 0-100
    uint8_t  channel;      // 1-10
    uint8_t  reserved;
    uint16_t crc;
} AppConfig_t;

AppConfig_t g_cfg;

void cfg_defaults(void) {
    g_cfg.magic = CFG_MAGIC;
    g_cfg.brightness = 50;
    g_cfg.volume = 80;
    g_cfg.channel = 1;
}

void cfg_save(void) {
    g_cfg.crc = Flash_CalculateCRC16(
        (uint8_t*)&g_cfg,
        sizeof(AppConfig_t) - sizeof(uint16_t)
    );
    if(Flash_SaveConfig(CFG_ADDR, (uint8_t*)&g_cfg, sizeof(g_cfg))) {
        USART_Print("Saved OK\r\n");
    }
}

void cfg_load(void) {
    if(!Flash_LoadConfig((uint8_t*)&g_cfg, sizeof(g_cfg))) {
        USART_Print("No saved config — factory defaults\r\n");
        cfg_defaults();
    } else {
        USART_Print("Config loaded\r\n");
    }
}

void cfg_factory_reset(void) {
    Flash_ErasePage(CFG_ADDR);
    cfg_defaults();
    USART_Print("Factory reset!\r\n");
}

int main(void)
{
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
    Flash_Init();

    cfg_load();

    USART_Print("Brightness: ");
    USART_PrintNum(g_cfg.brightness);
    USART_Print("\r\nVolume: ");
    USART_PrintNum(g_cfg.volume);
    USART_Print("\r\nChannel: ");
    USART_PrintNum(g_cfg.channel);
    USART_Print("\r\n");

    // จำลองการเปลี่ยนค่าแล้ว save
    g_cfg.brightness = 75;
    g_cfg.channel = 5;
    cfg_save();

    // Factory reset (ถ้าต้องการ)
    // cfg_factory_reset();

    while(1) { Delay_Ms(1000); }
}

✏️ ลองทำดู

  1. เพิ่มฟิลด์ device_name[16] ใน struct — เก็บชื่ออุปกรณ์ — ทดสอบ save/load
  2. ทำ dual-page storage — เขียน config 2 หน้า (page 60 และ 61) — โหลดหน้าที่ CRC ผ่าน — ป้องกันไฟดับระหว่างเขียน
  3. เขียนฟังก์ชัน flash_dump() — แสดงข้อมูล raw ใน Flash ทีละ 16 bytes ผ่าน USART — ใช้ debug ว่า Flash เก็บอะไรอยู่
