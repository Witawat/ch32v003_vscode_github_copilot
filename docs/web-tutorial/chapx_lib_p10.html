รู้จักขา

  • A (PC0): เลือกแถวบน P10 บิต 0
  • B (PC1): เลือกแถวบน P10 บิต 1
  • CLK (PC2): สัญญาณนาฬิกาเลื่อนข้อมูลเข้าชิฟต์รีจิสเตอร์
  • SCLK (PC3): สัญญาณล็อคข้อมูลออกสู่ขาแสดงผล
  • R1 (PC4): ข้อมูลสีแดงของ LED แต่ละดวง
  • OE (PC5): สั่งให้ LED สว่าง (Active LOW = สว่าง)
  • VCC (5V): จ่ายไฟ 5V กระแส 2A+ แยกจาก MCU
  • GND: กราวด์ร่วม P10 กับ MCU

วิธีต่อทีละขั้น

  1. ต่อ PC0 ไป A ของ P10 — เลือกแถวที่ 1
  2. ต่อ PC1 ไป B ของ P10 — เลือกแถวที่ 2
  3. ต่อ PC2 ไป CLK ของ P10 — ป้อนนาฬิกาชิฟต์รีจิสเตอร์
  4. ต่อ PC3 ไป SCLK ของ P10 — ล็อคข้อมูลออกจอ
  5. ต่อ PC4 ไป R1 ของ P10 — ส่งข้อมูลสีแดง
  6. ต่อ PC5 ไป OE ของ P10 — เปิดให้ LED ติด
  7. ต่อ แหล่งจ่าย 5V 2A+ ไป VCC ของ P10 — จ่ายไฟแยกจาก MCU
  8. ต่อ GND จากแหล่งจ่ายและ MCU ไป GND ของ P10 — กราวด์ร่วม

P10 ต้องใช้ external 5V 2A+ จ่ายไฟ — CH32V003 ใช้ 3.3V logic — ใช้ config struct กำหนดขา

📚 หลักการทำงาน

Multiplexed LED Panel: P10 32×16 ใช้การสแกน 1/4 scan — แบ่งจอเป็น 4 แถว (A,B เลือก) — P10_Init() ตั้ง TIM2 interrupt ให้ทำงานอัตโนมัติ — ISR จะวนสแกน 4 แถว

64-Byte Buffer: buffer ขนาด 32×16 = 64 ไบต์ — การเปลี่ยนแปลงข้อมูลใน main loop จะถูกแสดงผลโดยอัตโนมัติผ่าน ISR

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Init + Setup

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "P10.h"

P10_Instance inst;

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();

    P10_Config cfg = {
        .pin_a   = PC0,
        .pin_b   = PC1,
        .pin_clk  = PC2,
        .pin_sclk = PC3,
        .pin_r1   = PC4,
        .pin_oe   = PC5
    };
    P10_Init(&inst, &cfg);
    P10_Clear(&inst);

🔍 ใช้ P10_Config struct กำหนดขา — P10_Init(&inst, &cfg) — ฟังก์ชันตั้งค่า GPIO และ TIM2 interrupt โดยอัตโนมัติ

ขั้นที่ 2: แสดงข้อความ + กราฟิก

    P10_SetPixel(&inst, 0, 0, 1);   // เปิด pixel ที่ (0,0)
    P10_SetPixel(&inst, 16, 8, 1);  // เปิด pixel ตรงกลาง

    USART_Println("P10 Ready!");

🔍 P10_SetPixel(&inst, x, y, 1) — เปิด/ปิด pixel เดี่ยว — P10_Clear(&inst) ล้างจอทั้งหมด

ขั้นที่ 3: Loop อัปเดต

    Timer_t tmr;
    Start_Timer(&tmr, 1000, 1);
    uint8_t sec = 0;

    while(1) {
        if(Is_Timer_Expired(&tmr)) {
            P10_Clear(&inst);
            sec++;
        }
        // TIM2 ISR จัดการสแกนจออัตโนมัติ
    }
}

🔍 P10 auto-updates ผ่าน TIM2 ISR — main loop แค่เปลี่ยน buffer ก็พอ — ไม่ต้องเรียก Display()

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
จอไม่มีไฟเลยP10 ต้องใช้ external 5V 2A+ใช้ adapter 5V 2A ขึ้นไป — อย่าพึ่งไฟจาก MCU
ครึ่งจอดับ / บางแถวไม่ติดA/B pin ผิดตรวจสอบ config การต่อ A (PC0) และ B (PC1)
จอ flicker / กระพริบScan rate ต่ำตรวจสอบว่า SystemCoreClockUpdate() ถูกเรียกก่อน P10_Init()
3.3V logic กับ P10 5V3.3V อาจไม่พอให้ P10 ตีความเป็น HIGHMarginal แต่ใช้ได้กับ P10 ส่วนใหญ่ — ถ้าไม่เวิร์กใช้ 74HCT245 level shifter

🛠️ สร้าง Product: ป้ายแสดงเวลา+วันที่

P10 + DS3231 RTC — แสดงเวลาสลับกับวันที่ทุก 5 วินาที

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "P10.h"

RTC_DS3231_t rtc;

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    I2C_SimpleInit(I2C_100KHZ, I2C_PINS_DEFAULT);
    DS3231_Init(&rtc);

    P10_Instance inst;
    P10_Config cfg = {
        .pin_a=PC0, .pin_b=PC1, .pin_clk=PC2,
        .pin_sclk=PC3, .pin_r1=PC4, .pin_oe=PC5
    };
    P10_Init(&inst, &cfg);
    P10_Clear(&inst);

    RTC_DateTime_t dt;
    Timer_t tmr;
    Start_Timer(&tmr, 1000, 1);

    while(1) {
        if(Is_Timer_Expired(&tmr)) {
            DS3231_GetDateTime(&rtc, &dt);
        }
    }
}
