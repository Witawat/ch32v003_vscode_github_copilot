TM1637 ใช้สายแค่ 2 เส้น (CLK + DIO) — ใช้ 3.3V หรือ 5V ก็ได้ — มี pull-up ภายใน chip — 4 หลัก 7-segment + colon (:) ตรงกลาง

📚 หลักการทำงาน

Bit-Bang 2-Wire Protocol: TM1637 ใช้โปรโตคอล 2 เส้นแบบ proprietary — ไม่ใช่ I2C มาตรฐาน — CLK ควบคุมจังหวะ DIO ส่งข้อมูล — ใช้ GPIO bit-bang (Software I2C-like) — เริ่มด้วย start condition → ส่ง command → ส่ง 4 ไบต์ data → stop condition

4-Digit 7-Segment: แต่ละ digit แสดงตัวเลข 0-9 และอักษรบางตัว (A-F, H, P, U) — segment a-g + dp — ส่งข้อมูล digit ละ 1 ไบต์ (8-bit segment map) — หน่วงเวลาเล็กน้อยระหว่าง byte

Brightness Control: PWM 8 ระดับ (0=mืด, 7=สว่างสุด) — TM1637_SetBrightness(0-7) — ระดับ 4-5 เหมาะสำหรับใช้งานปกติ

Colon: TM1637 มี colon ระหว่างหลัก 2 และ 3 — ติดตลอดเวลาเมื่อเปิด display — ไม่สามารถควบคุมแยกได้ (แต่บาง library มี SetColon ถ้าใช้ pin แยก)

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Init + Setup

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();

    TM1637_Init(PC0, PC1);      // CLK=PC0, DIO=PC1
    TM1637_SetBrightness(5);    // 5/7 brightness

🔍 TM1637_Init(clk_pin, dio_pin) ตั้งขา 2 เส้น — ใช้ GPIO bit-bang — SetBrightness 0-7 — ไม่ต้องใช้ I2C หรือ SPI

ขั้นที่ 2: แสดงตัวเลขรูปแบบต่างๆ

    TM1637_DisplayNumber(1234);     // แสดง 1234
    Delay_Ms(2000);

    TM1637_DisplayTime(12, 30);     // แสดง 12:30
    Delay_Ms(2000);

    TM1637_DisplayFloat(25.5f, 1);  // แสดง 25.5
    Delay_Ms(2000);

    TM1637_DisplayHex(0xAB);        // แสดง "AB" hex

🔍 DisplayNumber(n) แสดงเลข 0-9999 — DisplayTime(h, m) แสดง hh:mm (12:30) — DisplayFloat(f, dec) แสดงทศนิยม 1 ตำแหน่ง — DisplayHex แสดงค่า hex 2 หลัก

ขั้นที่ 3: Countdown Timer

    uint16_t count = 9999;

    while(1) {
        TM1637_DisplayNumber(count);
        Delay_Ms(1000);

        if(count > 0) {
            count--;
        } else {
            // Blink เมื่อถึง 0
            for(uint8_t i = 0; i < 5; i++) {
                TM1637_Clear();
                Delay_Ms(300);
                TM1637_DisplayNumber(0);
                Delay_Ms(300);
            }
            count = 9999;
        }
    }
}

🔍 countdown จาก 9999 → 0 — เมื่อถึง 0 กระพริบ 5 ครั้งแล้ว reset — Clear() ล้างหน้าจอ — ใช้ Delay_Ms แบบ blocking เพื่อความเรียบง่าย

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
ตัวเลขกลับหัว / 7-segment เพี้ยนCLK/DIO กลับข้างกันสลับขา CLK กับ DIO — ตรวจสอบ pinout ของโมดูล
ความสว่างไม่พอBrightness < 3SetBrightness(5) หรือ 6 — ระดับ 7 สว่างสุดแต่กินไฟเพิ่ม
แสดงค่าเกิน 9999 เพี้ยนTM1637 รองรับ 4 หลักเท่านั้นเช็คค่า input ไม่ให้เกิน 9999 — หรือแบ่งแสดงแบบสลับ
Colon ควบคุมไม่ได้ (ติดตลอด)TM1637 ไม่มี pin แยก colonเป็นข้อจำกัดของฮาร์ดแวร์ — ใช้โมดูลที่มี pin colon แยก หรือใช้ TM1650
Display กระตุกตอนอัปเดตProtocol bit-bang ช้าเพราะ interrupt ขัดจังหวะปิด interrupt ก่อนเรียก DisplayNumber() — หรือใช้ TM1650 I2C แทน

🛠️ สร้าง Product: นาฬิกาจับเวลา Rotary Encoder ตั้งเวลา

TM1637 + Rotary Encoder — หมุนตั้งนาที, กดเริ่มนับถอยหลัง — ใช้ Non-Blocking Timer สำหรับ countdown

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    TM1637_Init(PC0, PC1);
    TM1637_SetBrightness(5);

    uint16_t target_sec = 60;
    uint16_t remaining = 0;
    uint8_t running = 0;

    // Setup Rotary Encoder on PC2(CLK), PC3(DT), PC4(SW)
    Rotary_Init(PC2, PC3, PC4);

    Timer_t tmr_countdown;
    TM1637_DisplayTime(target_sec / 60, target_sec % 60);

    while(1) {
        int8_t rot = Rotary_Read();
        if(rot != 0 && !running) {
            if(rot > 0 && target_sec < 5999) target_sec += 30;
            if(rot < 0 && target_sec > 0) target_sec -= 30;
            TM1637_DisplayTime(target_sec / 60, target_sec % 60);
        }

        if(Rotary_ButtonPressed() && !running) {
            remaining = target_sec;
            Start_Timer(&tmr_countdown, 1000, 1);
            running = 1;
        }

        if(running && Is_Timer_Expired(&tmr_countdown)) {
            if(remaining > 0) {
                remaining--;
                TM1637_DisplayTime(remaining / 60, remaining % 60);
            } else {
                running = 0;
                // Blink 3 times
                for(int i = 0; i < 3; i++) {
                    TM1637_Clear(); Delay_Ms(200);
                    TM1637_DisplayTime(0, 0); Delay_Ms(200);
                }
                TM1637_DisplayTime(target_sec / 60, target_sec % 60);
            }
        }
    }
}

✏️ ลองทำดู

  1. ทำ Countdown 99:59 — รองรับการตั้งเวลามากกว่า 59:59 โดยใช้ DisplayNumber แทน DisplayTime
  2. แสดงอุณหภูมิ — อ่าน DHT11 แล้ว DisplayFloat(temp, 1) — อัปเดตทุก 2 วิ
  3. RPM Counter — ใช้ External Interrupt นับรอบ — คำนวณ RPM แล้ว DisplayNumber
