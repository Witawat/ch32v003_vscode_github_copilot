📚 หลักการทำงาน

TM1637 ใช้โปรโตคอล 2 เส้นแบบ proprietary — ไม่ใช่ I2C — TM1637_Init() รับ CLK, DIO, และจำนวนหลัก — ไม่มี instance (global functions)

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Init + Setup

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "TM1637.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();

    TM1637_Init(PC0, PC1, 4);   // CLK=PC0, DIO=PC1, 4 digits
    TM1637_SetBrightness(5);

🔍 TM1637_Init(clk, dio, digits) — ไม่มี instance, ระบุจำนวนหลัก — SetBrightness 0-7

ขั้นที่ 2: แสดงตัวเลขรูปแบบต่างๆ

    TM1637_DisplayNumber(1234);
    Delay_Ms(2000);

    TM1637_DisplayTime(12, 30);     // แสดง 12:30
    Delay_Ms(2000);

    TM1637_DisplayFloat(25.5f, 1);  // แสดง 25.5
    Delay_Ms(2000);

    TM1637_DisplayHex(0xAB);
    Delay_Ms(2000);

🔍 DisplayNumber, DisplayTime, DisplayFloat, DisplayHex — API เดิม ไม่ต้องเปลี่ยน

ขั้นที่ 3: Countdown Timer

    uint16_t count = 9999;

    while(1) {
        TM1637_DisplayNumber(count);
        Delay_Ms(1000);

        if(count > 0) {
            count--;
        } else {
            for(uint8_t i = 0; i < 5; i++) {
                TM1637_Clear();
                Delay_Ms(300);
                TM1637_DisplayNumber(0);
                Delay_Ms(300);
            }
            count = 9999;
        }
    }
}

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้

🛠️ สร้าง Product: นาฬิกาจับเวลา

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "TM1637.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    TM1637_Init(PC0, PC1, 4);
    TM1637_SetBrightness(5);

    uint16_t remaining = 120;
    Timer_t tmr;
    Start_Timer(&tmr, 1000, 1);

    TM1637_DisplayTime(remaining / 60, remaining % 60);

    while(1) {
        if(Is_Timer_Expired(&tmr)) {
            if(remaining > 0) {
                remaining--;
                TM1637_DisplayTime(remaining / 60, remaining % 60);
            } else {
                for(int i = 0; i < 3; i++) {
                    TM1637_Clear(); Delay_Ms(200);
                    TM1637_DisplayTime(0, 0); Delay_Ms(200);
                }
                remaining = 120;
            }
        }
    }
}

✏️ ลองทำดู

  1. ทำ Countdown 99:59 — รองรับการตั้งเวลามากกว่า 59:59
  2. แสดงอุณหภูมิ — อ่าน DHT11 แล้ว DisplayFloat(temp, 1)
  3. RPM Counter — ใช้ External Interrupt นับรอบ
