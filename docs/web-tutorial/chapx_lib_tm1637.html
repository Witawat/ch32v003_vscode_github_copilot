⚠️ TM1637 ใช้โปรโตคอล 2 เส้นแบบ proprietary — ไม่ใช่ I2C ไม่มี address ไม่ต้อง pull-up แบบ I2C

รู้จักขา

  • VCC: ไฟเลี้ยง 3.3V–5V
  • GND: ต่อกราวด์ร่วมกับ MCU
  • CLK: ขาสัญญาณนาฬิกา ต่อกับ PC0
  • DIO: ขาข้อมูล 2 ทิศทาง ต่อกับ PC1

วิธีต่อทีละขั้น

  1. ต่อ PC0 ไป CLK — สัญญาณนาฬิกา
  2. ต่อ PC1 ไป DIO — ส่ง/รับข้อมูล
  3. ต่อ 3.3V ไป VCC — จ่ายไฟ
  4. ต่อ GND ไป GND — กราวด์ร่วม

📚 หลักการทำงาน

TM1637 เป็น IC driver สำหรับ 7-segment display 4 หรือ 6 หลัก ใช้โปรโตคอล 2 เส้นแบบ proprietary คุยกับ MCU ด้วย CLK + DIO — API เป็น handle-based เหมือน I2C display ทั่วไป: TM1637_Init คืน TM1637_Handle* (pointer) และทุกฟังก์ชัน display ต้องส่ง handle กลับไปเป็น argument แรกเสมอ — ทำให้ต่อหลายจอในเวลาเดียวกันได้ (แต่ละจอมี handle ของตัวเอง)

⚠️ ข้อจำกัดจาก LIMITATIONS.md: TM1637_DisplayNumber รองรับ INT16_MIN — ระวัง overflow ถ้าเลขติดลบเกินช่วง int16_t

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Init + ตั้งค่า handle

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "TM1637.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();                                  // จำเป็น — TM1637 ใช้ Delay

    // TM1637_Init คืน POINTER — เก็บไว้ในตัวแปร pointer
    TM1637_Handle* disp = TM1637_Init(PC0, PC1, 4);
    // CLK=PC0, DIO=PC1, 4 digits
    if (disp == NULL) { while (1); }               // init fail guard

    TM1637_SetBrightness(disp, 7);   // สว่างสุด (0–7)
    TM1637_Clear(disp);

🔍 TM1637_Init(clk, dio, digits) คืน TM1637_Handle* — เป็น handle-based ไม่ใช่ global — ทุกฟังก์ชันต่อไปรับ disp เป็น arg แรก — ต้องเรียก Timer_Init() ก่อน

ขั้นที่ 2: Counter 0→9999 + leading_zero

    for (int16_t n = 0; n <= 9999; n++) {
        // leading_zero=true → "0012", false → "  12"
        TM1637_DisplayNumber(disp, n, true);
        Delay_Ms(100);
    }
    // แสดงด้วย leading_zero=false เปรียบเทียบ
    TM1637_DisplayNumber(disp, 42, false);   // → "  42"
    Delay_Ms(2000);
    TM1637_DisplayNumber(disp, 42, true);    // → "0042"
    Delay_Ms(2000);

    // ทดสอบเลขติดลบ (รองรับถึง INT16_MIN)
    TM1637_DisplayNumber(disp, -123, false); // → "-123"
    Delay_Ms(2000);

🔍 TM1637_DisplayNumber(disp, number, leading_zero) — มี 3 args เสมอ — leading_zero=true → แสดง 0 นำหน้า ("0042"), false → ช่องว่าง (" 42") — รองรับเลขติดลบ (int16_t)

ขั้นที่ 3: Time HH:MM + dim ตอนกลางคืน

    // แสดงเวลา 12:34 (DisplayTime มี 3 args รวม show_colon)
    TM1637_DisplayTime(disp, 12, 34, true);   // → "12:34"
    Delay_Ms(2000);
    TM1637_DisplayTime(disp, 8, 5, false);    // → "08 5" (ไม่มี colon)
    Delay_Ms(2000);

    // ปรับความสว่างตามช่วงเวลา (dim ตอนกลางคืน)
    TM1637_SetBrightness(disp, 7);    // กลางวัน — สว่างสุด
    Delay_Ms(1000);
    TM1637_SetBrightness(disp, 1);    // กลางคืน — มืดสุด
    Delay_Ms(1000);
    TM1637_DisplayControl(disp, false);  // ปิดจอเลย
    Delay_Ms(1000);
    TM1637_DisplayControl(disp, true);   // เปิดใหม่
}

🔍 TM1637_DisplayTime(disp, hours, minutes, show_colon) — มี 3 args (ครบทั้ง hours, minutes, show_colon) — SetBrightness รับ 0–7 — DisplayControl(disp, on) เปิด/ปิดจอโดยไม่ล้าง buffer — ทุกฟังก์ชันต้องส่ง disp

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
Display ปิด / มืด / ไม่ติดเลยลืมเรียก TM1637_SetBrightness(disp, n) หรือเรียก DisplayControl(disp, false) ค้างไว้เรียก SetBrightness(disp, 7) หลัง Init เสมอ — เช็คว่าไม่ได้ DisplayControl(false) โดยไม่เปิดกลับ
ตัวเลขแสดงผิด / หายหลักnum_digits ผิดขนาด module (ตั้ง 4 แต่จอเป็น 6 หรือกลับกัน)ตั้ง num_digits ให้ตรงกับ module จริง (4 หรือ 6)
เลขติดลบแสดงผิดเกินช่วง int16_t (-32768–32767)เช็คค่าก่อนส่ง — LIMITATIONS.md บอกรองรับถึง INT16_MIN
ลืมส่ง handle arg → build failเขียน TM1637_SetBrightness(5) แบบเดิม (global-style)ทุกฟังก์ชันต้องส่ง disp เป็น arg แรก — เช่น TM1637_SetBrightness(disp, 5)
DisplayTime ไม่มี colonเขียน 2 args (เดิม) หรือลืม show_colonใช้ 3 args: DisplayTime(disp, h, m, true) — ใช้ SetColon(disp, true) แยกถ้าจอมี colon
จอพริก / กระพริบเองเรียก SetBlink(disp, true, n) โดยไม่เรียก UpdateBlink(disp) ใน loopUpdateBlink(disp) ต้องอยู่ใน main loop — มิฉะนั้น state ไม่ sync
CLK/DIO กลับด้านสลับขาสลับ CLK กับ DIO — ไม่ใช่ I2C ไม่มี address ให้สลับเลย

🛠️ สร้าง Product: นาฬิกาจับเวลาพร้อม dim ตอนกลางคืน

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "TM1637.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();

    TM1637_Handle* disp = TM1637_Init(PC0, PC1, 4);
    TM1637_SetBrightness(disp, 7);
    TM1637_Clear(disp);

    uint16_t remaining = 120;       // นาที:วินาที 02:00 → 120 วินาที
    Timer_t tmr;
    Start_Timer(&tmr, 1000, 1);     // 1s non-blocking timer

    TM1637_DisplayTime(disp, remaining / 60, remaining % 60, true);

    while (1) {
        if (Is_Timer_Expired(&tmr)) {
            if (remaining > 0) {
                remaining--;
                // dim ถ้าเหลือ < 30 วินาที (เตือนว่าใกล้หมด)
                if (remaining < 30) TM1637_SetBrightness(disp, 2);
                TM1637_DisplayTime(disp, remaining / 60, remaining % 60, true);
            } else {
                for (int i = 0; i < 3; i++) {
                    TM1637_Clear(disp);                    Delay_Ms(200);
                    TM1637_DisplayTime(disp, 0, 0, true);   Delay_Ms(200);
                }
                remaining = 120;
                TM1637_SetBrightness(disp, 7);   // คืนค่าสว่างเดิม
            }
        }
    }
}

🔍 ใช้ TM1637_Handle* disp ครบทุก call — DisplayTime(disp, min, sec, true) สาม args — SetBrightness(disp, n) ลดแสงตอนใกล้หมดเวลา — เปรียบเทียบกับบทเดิมที่เขียน TM1637_SetBrightness(5) ผิด (global-style)

✏️ ลองทำดู

  1. ทำ Countdown 99:59 — ใช้ DisplayTime(disp, 99, 59, true) และเช็ค colon ที่จอของคุณ
  2. แสดงอุณหภูมิจาก DHT11 — อ่านแล้ว TM1637_DisplayFloat(disp, temp, 1)
  3. RPM Counter — ใช้ external interrupt นับ pulse รอบ แล้ว TM1637_DisplayNumber(disp, rpm, true) หรือ DisplayHex(disp, rpm, true)
