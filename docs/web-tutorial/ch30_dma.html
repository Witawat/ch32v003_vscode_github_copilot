ex01_MemCopy_Sync — คัดลอกหน่วยความจำแบบ DMA แบบรอเสร็จ

📖 แนวคิด

ใช้ DMA_MemCopy คัดลอกข้อมูลจาก buffer หนึ่งไปอีก buffer หนึ่งด้วย DMA — รอจนกว่าจะเสร็จ (blocking) — ใช้ MemSet เติมค่าให้ buffer ด้วย DMA เช่นกัน

🧭 ผังการทำงาน

flowchart TD
    A["DMA_MemSet(src, 0xFF, 256)"] --> B["DMA_MemCopy(src, dst, 256)"]
    B --> C["Verify memcmp(src, dst)"]
    C --> D{"Match?"}
    D -->|"Yes"| E["Print PASS"]
    D -->|"No"| F["Print FAIL"]
    E --> G["while(1)"]
    F --> G

💻 โค้ด + อธิบายทีละบรรทัด

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include <SimpleHAL.h>

#define BUF_SIZE  64

int main(void)
{
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);

    uint8_t src[BUF_SIZE];
    uint8_t dst[BUF_SIZE];

    // เติม src ด้วยลาย test
    for(uint8_t i = 0; i < BUF_SIZE; i++) src[i] = i;

    // DMA MemCopy — blocking รอจนเสร็จ
    DMA_MemCopy(dst, src, BUF_SIZE);                 // แบบรอ
    USART_Print("MemCopy done\r\n");

    // DMA MemSet — เซตค่าทั้ง buffer
    DMA_MemSet(dst, 0xFF, BUF_SIZE);                 // เติม 0xFF
    USART_Print("MemSet done\r\n");

    while(1) { IWDG_Feed(); }
}
  • DMA_MemCopy(dst, src, len): ใช้ DMA channel 1 — ตั้งค่า src/dst address, length = len — polling จน TC flag set
  • DMA_MemSet(buf, val, len): ใช้ DMA+constant source — เขียนค่า val ทุก byte โดยไม่ต้อง loop CPU
  • Blocking (sync): CPU หยุดรอ DMA เสร็จ — เหมาะกับช่วง initialization

⚠️ จุดควรระวัง

  • แหล่ง DMA channel ไม่เหมือนกันทุกชิป — เช็คคู่มือว่า channel ไหนใช้กับ peripheral ใด
  • DMA address ต้องเป็น RAM (SRAM) — ไม่ใช่ Flash หรือ Peripheral register
  • DMA_MemSet มีเฉพาะบาง implementation — ถ้าไม่มีให้ใช้ loop แทน

✅ ตรวจสอบก่อนรัน

  • dst buffer ต้องมีขนาดเท่ากับ src — DMA เขียนเกิน = ทับ memory ข้างเคียง
  • เปรียบเทียบ dst กับ src หลัง MemCopy ว่าตรงกัน

ex02_MemCopy_Async — คัดลอกแบบไม่รอ + Callback

📖 แนวคิด

ใช้ MemCopyAsync เริ่ม DMA แล้วคืนทันที — CPU ทำงานต่อ — เมื่อ DMA เสร็จจะเรียก callback — เหมาะกับงานที่ต้องการทำต่อเนื่องโดยไม่รอ

🧭 ผังการทำงาน

flowchart TD
    A["DMA_SetCallback(Copy_Callback)"] --> B["MemCopyAsync(dst, src, 128)"]
    B --> C["CPU continues working"]
    C --> D{"DMA complete?"}
    D -->|"No"| C
    D -->|"Yes"| E["Callback → dma_done = 1"]
    E --> F["Print DMA complete!"]
    F --> C

💻 โค้ด

volatile uint8_t dma_done = 0;

void DMA_Copy_Callback(void) {                       // เรียกเมื่อ DMA TC interrupt
    dma_done = 1;
    digitalToggle(PC0);
}

int main(void)
{
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
    pinMode(PC0, PIN_MODE_OUTPUT);

    uint8_t src[128];
    uint8_t dst[128];
    for(uint8_t i = 0; i < 128; i++) src[i] = i;

    DMA_SetCallback(DMA_Copy_Callback);              // ลงทะเบียน callback
    MemCopyAsync(dst, src, 128);                     // เริ่ม DMA + คืนทันที
    USART_Print("DMA started — CPU is free!\r\n");

    while(1)
    {
        if(dma_done) {
            USART_Print("DMA complete!\r\n");
            dma_done = 0;
        }
        IWDG_Feed();
    }
}
  • MemCopyAsync(dst, src, len): เริ่ม DMA transfer — ไม่รอ — คืนทันที
  • DMA_SetCallback(cb): ฟังก์ชันนี้จะถูกเรียกใน DMA interrupt เมื่อ transfer complete
  • dma_done flag: ตั้งค่าใน ISR — main loop ตรวจสอบ — ใช้ volatile

⚠️ จุดควรระวัง

  • ห้ามใช้ src/dst buffer จนกว่า DMA จะเสร็จ — ถ้า CPU เขียนทับ = data corruption
  • callback ทำงานใน ISR — ต้องสั้น ใช้แค่ตั้ง flag
  • MemCopyAsync ใช้ DMA interrupt — ต้องเปิด NVIC ให้ DMA channel

ex03_DMA_USART_Transmit — ส่ง USART ด้วย DMA

📖 แนวคิด

ใช้ DMA_USART_InitTx ตั้งค่า USART TX DMA — ส่งข้อมูลยาว ๆ โดย CPU ไม่ต้องรอทีละ byte — เหมาะกับ logging, file transfer

🧭 ผังการทำงาน

flowchart TD
    A["DMA_USART_InitTx()"] --> B["while(1)"]
    B --> C["DMA_USART_Transmit(msg, len)"]
    C --> D["DMA sends via USART\n(CPU free)"]
    D --> E["Delay_Ms(1000)"]
    E --> B

💻 โค้ด

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include <SimpleHAL.h>

int main(void)
{
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);

    DMA_USART_InitTx();                              // จับคู่ DMA channel กับ USART TX

    uint8_t msg[] = "Hello from DMA! This is a long string.\r\n";
    while(1)
    {
        DMA_USART_Transmit(msg, sizeof(msg) - 1);    // ส่งแบบ DMA — ไม่ block
        Delay_Ms(1000);
    }
}
  • DMA_USART_InitTx(): ตั้งค่า DMA channel สำหรับ USART TX — กำหนด source = buffer, dest = USART DR register
  • DMA_USART_Transmit(buf, len): เริ่ม DMA transfer — ข้อมูลถูกส่งอัตโนมัติโดยไม่ต้อง CPU รอ TXE

⚠️ จุดควรระวัง

  • DMA_USART_Transmit ไม่รอจนส่งเสร็จ — ถ้าต้องการรอ ใช้ while(!DMA_GetFlagStatus(...))
  • buffer ต้องคงอยู่จนกว่า DMA ส่งเสร็จ — อย่าใช้ local variable ถ้าไม่รอ
  • USART baud rate ต้องตรง — DMA ส่งเร็วแต่ baud จำกัดอัตรา

ex04_DMA_SPI_Transfer — โอนข้อมูล SPI ผ่าน DMA

📖 แนวคิด

ใช้ DMA_SPI_Init + DMA_SPI_TransferBuffer ส่ง/รับข้อมูล SPI แบบไม่ต้อง CPU รอทีละ byte — เหมาะกับ display, SD card, sensor

🧭 ผังการทำงาน

flowchart TD
    A["SPI_SimpleInit()"] --> B["DMA_SPI_Init()"]
    B --> C["CS = LOW"]
    C --> D["DMA_SPI_TransferBuffer(tx, rx, 256)"]
    D --> E["CS = HIGH"]
    E --> F["Print SPI DMA done"]
    F --> G["while(1)"]

💻 โค้ด

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include <SimpleHAL.h>

#define SPI_CS_PIN  PC1

int main(void)
{
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
    pinMode(SPI_CS_PIN, PIN_MODE_OUTPUT);
    digitalWrite(SPI_CS_PIN, HIGH);

    SPI_SimpleInit(SPI_MODE_MASTER, SPI_SPEED_4MHZ);

    DMA_SPI_Init();                                  // ตั้งค่า DMA สำหรับ SPI

    uint8_t tx_buf[256];
    uint8_t rx_buf[256];
    for(uint16_t i = 0; i < 256; i++) tx_buf[i] = i;

    digitalWrite(SPI_CS_PIN, LOW);
    DMA_SPI_TransferBuffer(tx_buf, rx_buf, 256);     // ส่ง+รับพร้อมกัน
    digitalWrite(SPI_CS_PIN, HIGH);

    USART_Print("SPI DMA transfer done\r\n");

    while(1) { IWDG_Feed(); }
}
  • DMA_SPI_Init(): จับคู่ DMA channel 1 (TX) และ channel 2 (RX) กับ SPI
  • DMA_SPI_TransferBuffer(tx, rx, len): ส่ง tx_buf และรับ rx_buf พร้อมกัน — ใช้ DMA สอง channel
  • SPI กับ DMA = throughput สูงสุด: SPI clock เร็วสุด 8MHz (หรือมากกว่า) — DMA ช่วยไม่ให้ CPU รอ

ex05_DMA_PWM_Waveform — สร้างรูปคลื่น PWM ด้วย DMA + Sine Table

📖 แนวคิด

ใช้ TIM update event trigger DMA — DMA อ่านค่า sine table → เขียน CCR — สร้างสัญญาณ sine wave บน PWM output โดย CPU ไม่ต้องจัดการ

🧭 ผังการทำงาน

flowchart TD
    A["build_sine_table()"] --> B["TIM1_SimpleInit(PWM)"]
    B --> C["DMA_PWM_Waveform(TIM1, table)"]
    C --> D["TIM update →\nDMA trigger"]
    D --> E["DMA reads sine_table\n→ writes CCR1"]
    E --> F["PWM output = sine wave"]
    F --> D

💻 โค้ด

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include <SimpleHAL.h>

#define SINE_TABLE_SIZE  128
uint16_t sine_table[SINE_TABLE_SIZE];                // ค่า sine duty cycle

static void build_sine_table(void)
{
    for(uint16_t i = 0; i < SINE_TABLE_SIZE; i++) {
        float angle = 2.0f * 3.14159f * i / SINE_TABLE_SIZE;
        // duty cycle 10-90% (ประมาณ 100-900 สำหรับ TIM ARR=1000)
        sine_table[i] = (uint16_t)(500 + 400 * sinf(angle));
    }
}

int main(void)
{
    SystemCoreClockUpdate();
    Timer_Init();
    build_sine_table();

    TIM1_SimpleInit(PWM_MODE, 1000, 0);              // ARR=1000, prescaler=0
    TIM1_PWM_InitPin(PA1);                           // PA1 = CH1 output

    DMA_PWM_Waveform(TIM1, sine_table, SINE_TABLE_SIZE);
    // เมื่อ TIM1 update → DMA อ่านค่า sine_table → เขียน CCR1 อัตโนมัติ

    while(1) { IWDG_Feed(); }
}
  • DMA_PWM_Waveform(tim, table, size): ตั้งค่า DMA ให้ TIM update event trigger — โอนค่า table → TIM_CCR1
  • Circular mode: DMA วนซ้ำ table เมื่อถึงท้าย — สัญญาณต่อเนื่อง ไม่ต้อง CPU แทรก
  • Sine table: คำนวณ duty cycle ล่วงหน้า เก็บใน RAM — ปรับความถี่ด้วย TIM prescaler

⚠️ จุดควรระวัง

  • sine table ต้องอยู่ใน RAM (ไม่ใช่ Flash) — DMA อ่านจาก RAM เท่านั้น
  • DMA_PWM_Waveform ใช้ DMA channel ที่ TIM match — เช็คคู่มือ DMA mapping
  • TIM update frequency = TIM_clk / (ARR+1) / (PSC+1) — ปรับให้ตรงกับขนาด table

ex06_DMA_I2C_Transfer — โอนข้อมูล I2C ผ่าน DMA

📖 แนวคิด

ใช้ DMA_I2C_InitTx/Rx ส่ง/รับข้อมูล I2C ด้วย DMA — เหมาะกับ sensor ที่อ่านค่าต่อเนื่องหลาย byte

🧭 ผังการทำงาน

flowchart TD
    A["I2C_SimpleInit()"] --> B["DMA_I2C_InitTx()"]
    B --> C["DMA_I2C_InitRx()"]
    C --> D["I2C write reg address\n(DMA TX)"]
    D --> E["I2C read data\n(DMA RX)"]
    E --> F["Print I2C DMA done"]
    F --> G["while(1)"]

💻 โค้ด

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include <SimpleHAL.h>

#define I2C_ADDR  0x68  // ตัวอย่าง MPU6050

int main(void)
{
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);

    I2C_SimpleInit(I2C_SPEED_400KHZ);

    DMA_I2C_InitTx();                                // ตั้งค่า DMA TX สำหรับ I2C
    DMA_I2C_InitRx();                                // ตั้งค่า DMA RX สำหรับ I2C

    uint8_t cmd = 0x3B;                              // register address
    uint8_t rx_buf[14];                              // รับ 14 byte

    I2C_Start();
    I2C_Write(I2C_ADDR << 1);                        // address + write
    DMA_I2C_Transmit(&cmd, 1);                       // ส่ง register address
    I2C_Stop();

    I2C_Start();
    I2C_Write((I2C_ADDR << 1) | 1);                  // address + read
    DMA_I2C_Receive(rx_buf, 14);                     // รับ data
    I2C_Stop();

    USART_Print("I2C DMA read done\r\n");

    while(1) { IWDG_Feed(); }
}
  • DMA_I2C_InitTx() / InitRx(): จับคู่ DMA channel กับ I2C TX/RX
  • DMA_I2C_Transmit(buf, len): ส่งข้อมูล I2C ด้วย DMA — ไม่ต้องรอ TXE
  • DMA_I2C_Receive(buf, len): รับข้อมูล I2C ด้วย DMA — ไม่ต้องรอ RXNE

⚠️ จุดควรระวัง

  • I2C + DMA ต้องจัดการ NACK และ STOP ด้วยตัวเอง — DMA ไม่ได้จัดการ I2C protocol
  • I2C speed 400kHz = I2C_SPEED_400KHZ — ต้องตรงกับอุปกรณ์ slave
  • DMA_I2C_InitTx/Rx ใช้ DMA channel ต่างกัน — อย่าสลับ

ex07_DMA_Advanced — DMA_SimpleInit + Half/Complete Callbacks

📖 แนวคิด

ใช้ DMA_SimpleInit ตั้งค่า DMA ด้วยตัวเอง — กำหนด peripheral address, memory address, direction, data size — ใช้ half-transfer และ complete-transfer callbacks

🧭 ผังการทำงาน

flowchart TD
    A["SetCallbackHalf(HalfCplt)"] --> B["SetCallback(Cplt)"]
    B --> C["DMA_SimpleInit(MEM_TO_MEM, ...)"]
    C --> D["DMA starts"]
    D --> E{"Half transfer?"}
    E -->|"Yes"| F["Print Half done"]
    F --> G{"Full transfer?"}
    E -->|"No"| G
    G -->|"Yes"| H["Print Full done"]
    G -->|"No"| D

💻 โค้ด

volatile uint8_t half_done = 0;
volatile uint8_t full_done = 0;

void DMA_HalfCplt_Callback(void) {
    half_done = 1;                                   // transfer ไปครึ่งทาง
}

void DMA_Cplt_Callback(void) {
    full_done = 1;                                   // transfer เสร็จสมบูรณ์
}

int main(void)
{
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);

    uint8_t src[256];
    uint8_t dst[256];
    for(uint16_t i = 0; i < 256; i++) src[i] = i;

    DMA_SetCallbackHalf(DMA_HalfCplt_Callback);
    DMA_SetCallback(DMA_Cplt_Callback);

    // ตั้งค่า DMA ด้วยตนเอง — กำหนดทุกพารามิเตอร์
    DMA_SimpleInit(
        DMA_MEM_TO_MEM,                              // ทิศทาง
        DMA_DATA_SIZE_8BIT,                          // ขนาดข้อมูล
        DMA_PRIORITY_HIGH,                           // priority
        DMA_MODE_NORMAL,                             // ไม่วน loop
        (uint32_t)dst,                               // destination (peripheral)
        (uint32_t)src,                               // source (memory)
        256                                          // length
    );

    USART_Print("DMA Advanced started\r\n");

    while(1)
    {
        if(half_done) { USART_Print("Half done\r\n"); half_done = 0; }
        if(full_done) { USART_Print("Full done\r\n"); full_done = 0; }
        IWDG_Feed();
    }
}
  • DMA_SimpleInit(dir, size, prio, mode, periph, mem, len): กำหนดค่า DMA register โดยตรง — ไม่ใช้ wrapper
  • Half callback: ทำงานเมื่อ DMA transfer ไปครึ่งทาง (TCIF ครึ่ง buffer) — เหมาะกับ double-buffer
  • Complete callback: ทำงานเมื่อ transfer เสร็จทั้งหมด

⚠️ จุดควรระวัง

  • DMA_SimpleInit เขียนทับค่าที่ตั้งไว้ก่อนหน้า — เรียกครั้งเดียวต่อ DMA channel
  • ครึ่งทาง callback ใช้ได้กับ DMA_MODE_NORMAL เท่านั้น (ไม่ใช่ circular)
  • MEM_TO_MEM ต้องใช้ DMA channel ที่ support — ไม่ทุก channel ทำได้

ex08_DMA_ADC_Init — อ่าน ADC แบบต่อเนื่องด้วย DMA + Circular Buffer

📖 แนวคิด

ใช้ DMA_ADC_Init ตั้งค่า DMA ให้รับข้อมูล ADC อัตโนมัติเมื่อ ADC conversion เสร็จ — ใช้ circular buffer เพื่อเก็บค่าล่าสุดตลอดเวลา

🧭 ผังการทำงาน

flowchart TD
    A["ADC_SimpleInit()"] --> B["DMA_ADC_Init(buf, size)"]
    B --> C["ADC_Start()"]
    C --> D["DMA_ADC_Enable()"]
    D --> E["ADC ready → DMA\nwrites to buffer"]
    E --> F["while(1): read latest"]
    F --> G["DMA_ADC_GetLatest()"]
    G --> E

💻 โค้ด

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include <SimpleHAL.h>

#define ADC_BUF_SIZE  16
uint16_t adc_buffer[ADC_BUF_SIZE];                   // circular buffer

int main(void)
{
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);

    ADC_SimpleInit(ADC_PIN_PC4);                     // ตั้งค่า ADC pin
    ADC_SetSampleTime(ADC_SAMPLETIME_239CYCLES);     // sample time สูง

    DMA_ADC_Init(adc_buffer, ADC_BUF_SIZE);          // DMA รับค่า ADC → buffer

    ADC_Start();                                      // เริ่ม conversion ต่อเนื่อง
    DMA_ADC_Enable();                                 // เริ่ม DMA (circular)

    USART_Print("ADC DMA circular started\r\n");

    while(1)
    {
        // อ่านค่าล่าสุดจาก circular buffer
        static uint16_t last_val = 0;
        uint16_t val = DMA_ADC_GetLatest();
        if(val != last_val) {
            USART_Print("ADC: ");
            USART_PrintNum(val);
            USART_Print("\r\n");
            last_val = val;
        }
        Delay_Ms(100);
    }
}
  • DMA_ADC_Init(buf, size): ตั้งค่า DMA channel สำหรับ ADC — mode = CIRCULAR — ทุกครั้งที่ ADC ready → DMA อ่าน DR → เก็บใน buf
  • DMA_ADC_Enable(): เปิด DMA — ADC จะ push ค่าไปเรื่อย ๆ ไม่ต้อง CPU สั่ง
  • DMA_ADC_GetLatest(): อ่านค่า ADC ล่าสุดจาก circular buffer โดยใช้ index ที่ DMA update

⚠️ จุดควรระวัง

  • Circular buffer อาจเขียนทับข้อมูลเก่า — อ่านให้ทันหรือใช้ callback เมื่อ buffer เต็มครึ่ง
  • ADC sample time ยิ่งนาน ค่ายิ่งแม่นยำแต่ช้าลง — ปรับตามความต้องการ
  • DMA_ADC_Init จับคู่ ADC กับ DMA channel ที่ถูกต้องเท่านั้น
