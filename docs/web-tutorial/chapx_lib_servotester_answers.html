📝 เฉลย — Servo Tester

ข้อ 1: Calibrate เซอร์โวที่ไม่รู้จัก — หา min/max pulse

#include "simple_servotester.h"

ServoTester_t tester;

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    ServoTester_Init(&tester, GPIOA, GPIO_Pin_0);

    // กวาด pulse ตั้งแต่ 500us ถึง 2500us ทีละ 10us
    for (uint16_t pulse = 500; pulse <= 2500; pulse += 10) {
        ServoTester_SetPulse(&tester, pulse);
        Delay_Ms(30);  // รอให้เซอร์โวตอบสนอง
    }

    while (1) {
        // วนกลับไปกลับมา ระหว่าง min กับ max ที่หาได้
        for (uint16_t p = 500; p <= 2500; p += 50)
            { ServoTester_SetPulse(&tester, p); Delay_Ms(40); }
        for (uint16_t p = 2500; p >= 500; p -= 50)
            { ServoTester_SetPulse(&tester, p); Delay_Ms(40); }
    }
}
🧠 แนวคิด: เริ่มด้วยการสแกน pulse แบบเต็มช่วง 500–2500µs — สังเกตตำแหน่งที่ servo เริ่มหมุนและหยุดหมุน — ค่า min คือ pulse ที่ servo เพิ่งเริ่มขยับจากปลายหนึ่ง ค่า max คือ pulse ที่ servo หยุดหมุนอีกฝั่ง — ขั้นตอน calibrate ทำครั้งเดียวตอนเริ่ม จากนั้นใช้ค่า min/max ในการคำนวณมุม — หลัง calibrate ให้ใช้ for loop สลับทิศทางเพื่อทดสอบการตอบสนอง

ข้อ 2: ทดสอบ Center Accuracy — วนกลับจุดกึ่งกลางซ้ำ ๆ

#include "simple_servotester.h"

ServoTester_t tester;

#define CENTER_PULSE  1500
#define LEFT_PULSE    1000
#define RIGHT_PULSE   2000

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    ServoTester_Init(&tester, GPIOA, GPIO_Pin_0);

    uint16_t positions[] = {LEFT_PULSE, CENTER_PULSE, RIGHT_PULSE, CENTER_PULSE};
    uint8_t  idx = 0;

    while (1) {
        ServoTester_SetPulse(&tester, positions[idx]);
        Delay_Ms(800);

        if (positions[idx] == CENTER_PULSE) {
            // หน่วงนานขึ้นที่ตำแหน่งกลางเพื่อวัดความแม่นยำ
            Delay_Ms(500);
        }

        idx = (idx + 1) % 4;
    }
}
🧠 แนวคิด: ทดสอบความแม่นยำในการกลับจุดกึ่งกลาง — ใช้ array เก็บลำดับตำแหน่ง Left → Center → Right → Center — หน่วงนานที่ตำแหน่งกลางเป็นพิเศษเพื่อให้ผู้ใช้สังเกตด้วยตาเปล่าว่าเซอร์โวกลับมาตรงเดิมหรือไม่ — ถ้าเซอร์โวมี backlash (ฟันเฟืองหลวม) จะเห็นตำแหน่งกลางจากทางซ้ายไม่เท่ากับจากทางขวา — ในทางปฏิบัติให้ใช้เครื่องหมายเทียบกับไม้บรรทัดหรือ protractor — ค่า pulse ที่ใช้ควรมาจากการ calibrate จริง ไม่ใช่ค่ามาตรฐาน

ข้อ 3: วัด Deadband ของเซอร์โว — หา pulse ขั้นต่ำที่ทำให้ขยับ

#include "simple_servotester.h"

ServoTester_t tester;

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    ServoTester_Init(&tester, GPIOA, GPIO_Pin_0);

    uint16_t base_pulse = 1500;

    ServoTester_SetPulse(&tester, base_pulse);
    Delay_Ms(1000);

    while (1) {
        uint16_t delta = 1;
        uint16_t test_pulse;

        // เพิ่ม pulse ทีละ 1 จนกว่าเซอร์โวจะขยับ
        for (delta = 1; delta <= 30; delta++) {
            test_pulse = base_pulse + delta;
            ServoTester_SetPulse(&tester, test_pulse);
            Delay_Ms(200);

            test_pulse = base_pulse - delta;
            ServoTester_SetPulse(&tester, test_pulse);
            Delay_Ms(200);
        }

        Delay_Ms(2000);  // พักระหว่างรอบ
    }
}
🧠 แนวคิด: Deadband คือช่วง pulse ที่ servo ยังไม่ตอบสนอง — เริ่มจากตำแหน่งกลาง 1500µs — เพิ่ม delta ทีละ 1µs ทั้งสองทิศทาง — สังเกตด้วยตาว่า delta เท่าไหร่ที่ servo เริ่มกระตุก — เซอร์โวคุณภาพดี deadband ≈ 1-2µs, คุณภาพต่ำ ≈ 4-8µs — ข้อมูล deadband มีประโยชน์ในการตั้งค่า PID และการปรับ smoothing — อย่าลืมรอ delay เพียงพอ (200ms) ให้เซอร์โวมีเวลาตอบสนอง
