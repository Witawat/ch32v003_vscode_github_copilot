📝 เฉลย — Circular Buffer

ข้อ 1: USART RX Buffer — ISR เขียน, main loop อ่าน

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "CircularBuffer.h"

#define BUF_SIZE 128
CircularBuffer_t rx_buf;
uint8_t buf_data[BUF_SIZE];

void USART1_IRQHandler(void) __attribute__((interrupt("WCH-Interrupt-fast")));
void USART1_IRQHandler(void) {
    if(USART_GetITStatus(USART1, USART_IT_RXNE)) {
        uint8_t byte = USART_ReceiveData(USART1);
        CircularBuffer_Push(&rx_buf, byte);   // ISR-safe write
    }
}

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();

    CircularBuffer_Init(&rx_buf, buf_data, BUF_SIZE);

    NVIC_EnableIRQ(USART1_IRQn);
    USART_Init(115200);
    USART_ITConfig(USART1, USART_IT_RXNE, ENABLE);

    USART_Println("Rx buffer ready!");

    while(1) {
        // อ่านทั้งก้อนเมื่อมีข้อมูล ≥ 1 byte
        while(CircularBuffer_Available(&rx_buf) > 0) {
            uint8_t b;
            CircularBuffer_Pop(&rx_buf, &b);
            USART_SendData(USART1, b);         // echo back
            while(!USART_GetFlagStatus(USART1, USART_FLAG_TC));
        }

        // ทำงานอื่นได้ในระหว่างนี้
        IWDG_Feed();
    }
}
🧠 แนวคิด: ISR (USART1_IRQHandler) รับ byte จาก USART แล้วเขียนลง circular buffer ทันที — เร็ว ไม่บล็อก — main loop ใช้ while(Available > 0) อ่านทีละ byte ประมวลผล — โค้ดนี้ echo กลับแบบ non-blocking — CircularBuffer_Write thread-safe เมื่อใช้จาก ISR เพียงที่เดียว — buffer 128 bytes เพียงพอสำหรับ burst ข้อมูล — IWDG_Feed() ใน main loop กัน watchdog reset — สำหรับ high-speed UART (≥115200) เพิ่ม buffer เป็น 256 หรือใช้ DMA

ข้อ 2: Command Parser — แยกคำสั่งจาก buffer แบบ line-delimited

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "CircularBuffer.h"

#define CMD_BUF_SIZE 256
CircularBuffer_t cmd_buf;
uint8_t cmd_data[CMD_BUF_SIZE];

char line[64];
uint8_t li = 0;

void process_line(const char *line) {
    if(strncmp(line, "LED ON", 6) == 0) {
        digitalWrite(PC0, HIGH);
        USART_Println("OK: LED ON");
    }
    else if(strncmp(line, "LED OFF", 7) == 0) {
        digitalWrite(PC0, LOW);
        USART_Println("OK: LED OFF");
    }
    else if(strncmp(line, "TEMP?", 5) == 0) {
        float t = NTC_ReadTemp_C(PA0);
        USART_Printf("TEMP: %.1f C\n", t);
    }
    else USART_Println("ERR: Unknown command");
}

void USART1_IRQHandler(void) __attribute__((interrupt("WCH-Interrupt-fast")));
void USART1_IRQHandler(void) {
    if(USART_GetITStatus(USART1, USART_IT_RXNE)) {
        CircularBuffer_Push(&cmd_buf, USART_ReceiveData(USART1));
    }
}

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    CircularBuffer_Init(&cmd_buf, cmd_data, CMD_BUF_SIZE);

    NVIC_EnableIRQ(USART1_IRQn);
    USART_Init(115200);
    USART_ITConfig(USART1, USART_IT_RXNE, ENABLE);
    pinMode(PC0, PIN_MODE_OUTPUT);
    ADC_SimpleInit(PA0);

    USART_Println("CMD> Ready. (LED ON/OFF, TEMP?)");

    while(1) {
        while(CircularBuffer_Available(&cmd_buf)) {
            uint8_t b;
            CircularBuffer_Pop(&cmd_buf, &b);

            if(b == '\n' || b == '\r') {
                if(li > 0) { line[li] = '\0'; process_line(line); li = 0; }
            } else if(li < 63) {
                line[li++] = b;
            }
        }

        IWDG_Feed();
    }
}
🧠 แนวคิด: ISR เก็บ byte ลง buffer → main loop อ่านทีละ byte สะสมใน line[] — เมื่อเจอ \n หรือ \r → terminate string → process_line() — ใช้ strncmp เทียบคำสั่ง (case-sensitive) — รองรับ LED ON, LED OFF, TEMP? — ถ้าคำสั่งไม่รู้จัก → ERR — line buffer 64 bytes ป้องกัน buffer overflow — เทคนิคนี้ใช้ได้กับ GPS NMEA, AT Commands, หรือ protocol ใดๆ ที่เป็น line-based — สามารถ extend เพิ่ม command โดยเพิ่ม else if ใน process_line

ข้อ 3: Buffer Overflow Detection — ตรวจจับและจัดการข้อมูลล้น

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "CircularBuffer.h"

#define BUF_SIZE  64
CircularBuffer_t rx;
uint8_t rx_data[BUF_SIZE];

volatile uint16_t overflow_cnt = 0;
volatile uint16_t dropped_bytes = 0;

void USART1_IRQHandler(void) __attribute__((interrupt("WCH-Interrupt-fast")));
void USART1_IRQHandler(void) {
    if(USART_GetITStatus(USART1, USART_IT_RXNE)) {
        uint8_t b = USART_ReceiveData(USART1);
        if(!CircularBuffer_Push(&rx, b)) {
            overflow_cnt++;              // นับครั้งที่ buffer เต็ม
            dropped_bytes++;             // นับ byte ที่หายไป
        }
    }
}

void flush_buffer(void) {
    uint8_t dummy;
    while(CircularBuffer_Available(&rx)) CircularBuffer_Pop(&rx, &dummy);
}

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    CircularBuffer_Init(&rx, rx_data, BUF_SIZE);

    NVIC_EnableIRQ(USART1_IRQn);
    USART_Init(115200);
    USART_ITConfig(USART1, USART_IT_RXNE, ENABLE);

    Timer_t report_tmr;
    Start_Timer(&report_tmr, 2000, 1);

    while(1) {
        uint16_t avail = CircularBuffer_Available(&rx);

        // ถ้า buffer ใกล้เต็ม (>80%) → flush ทิ้ง
        if(avail > BUF_SIZE * 80 / 100) {
            USART_Println("⚠ BUFFER OVERLOAD — flushing!");
            flush_buffer();
            overflow_cnt++;
        }

        if(Is_Timer_Expired(&report_tmr)) {
            USART_Printf("Buf: %d/%d  Overflow: %d  Dropped: %d\n",
                         avail, BUF_SIZE, overflow_cnt, dropped_bytes);
        }

        // ประมวลผลเฉพาะเมื่อ buffer ไม่เกิน 50%
        if(avail <= BUF_SIZE / 2) {
            uint8_t b;
            while(CircularBuffer_Pop(&rx, &b)) {
                USART_SendData(USART1, b);
                while(!USART_GetFlagStatus(USART1, USART_FLAG_TC));
            }
        }

        IWDG_Feed();
    }
}
🧠 แนวคิด: CircularBuffer_Write return false เมื่อ buffer เต็ม → นับ overflow_cnt และ dropped_bytes — ตรวจจับ buffer >80% → flush ทิ้งทั้งก้อน ป้องกันข้อมูล stale — แสดงสถิติทุก 2 วิ: used/total, overflow count, dropped — ประมวลผลเฉพาะเมื่อ buffer ≤50% (flow control) — ดีกว่าใช้ delay เพราะยังทำงานอื่นได้ — วิธีแก้ overflow จริง: เพิ่ม hardware flow control (RTS/CTS) หรือลด baud rate — สำหรับข้อมูลสำคัญ ให้ใช้ protocol checksum แล้ว request retransmit แทนการ flush
