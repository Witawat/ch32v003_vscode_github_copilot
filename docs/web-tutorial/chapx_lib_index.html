🧭 Universal Pattern — ใช้ได้กับทุก Lib

flowchart TD
    A["SystemCoreClockUpdate()+Timer_Init()"] --> B["Init Interface: I2C/SPI/USART"]
    B --> C["Lib_Init(&inst, params)"]
    C --> D{"Init สำเร็จ?"}
    D -->|"✅ OK"| E["while(1)"]
    D -->|"❌ Fail"| F["จัดการ Error"]
    E --> G["Lib_Read/Update(&inst)"]
    G --> H["ประมวลผล / แสดงผล"]
    H --> I["Delay / Yield"]
    I --> E

📚 3 ขั้นตอนใช้งานทุก Lib

// ขั้น 1: System + Communication Init
SystemCoreClockUpdate(); Timer_Init();
I2C_SimpleInit(I2C_SPEED_100K, I2C_PINS_DEFAULT); // หรือ SPI/USART

// ขั้น 2: Lib Init + Instance
Lib_Instance_t dev;
if (Lib_Init(&dev, params...) != LIB_OK) { /* error */ }

// ขั้น 3: Main Loop
while(1) { Lib_Read(&dev, &data); Delay_Ms(interval); }
