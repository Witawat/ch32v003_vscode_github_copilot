หมายเหตุ: SimpleHAL รองรับ Arduino API โดยตรง — ไม่ต้องใช้ Arduino core — คอมไพล์ด้วย GCC RISC-V — โค้ดเล็กกว่า Arduino หลายเท่า

📚 หลักการทำงาน

Arduino API — ชุดฟังก์ชันมาตรฐานที่คุ้นเคย — SimpleHAL รองรับฟังก์ชันหลักทั้งหมด — ทำให้พอร์ตโค้ดจาก Arduino มา CH32V003 ได้ง่าย

ฟังก์ชันที่รองรับ

  • millis() / micros(): คืนค่าเวลาตั้งแต่เริ่มต้น — ใช้ SysTick 1ms — micros() ใช้ DWT cycle counter — ความละเอียด ~21ns ที่ 48MHz
  • delay() / delayMicroseconds(): หน่วงเวลา — เหมือน Arduino ทุกประการ
  • pinMode() / digitalWrite() / digitalRead(): ควบคุม GPIO — รองรับ INPUT_PULLUP, INPUT_PULLDOWN
  • analogRead(): อ่าน ADC — 10-bit (0-1023) — รองรับเฉพาะ pin ที่เป็น ADC
  • analogWrite(): PWM output — 8-bit (0-255) — ใช้ TIM1/TIM2
  • attachInterrupt(): ผูก interrupt กับ pin — รองรับ RISING, FALLING, CHANGE
  • Serial: เริ่มด้วย USART_SimpleInit() — ใช้ Serial.print() ได้
  • dtostrf(): แปลง float เป็น string — มีประโยชน์มากบน CH32V003 ที่ไม่มี sprintf float ใน stdlib จิ๋ว
  • yield(): weak function — ถูกเรียกใน delay() — override ได้สำหรับ cooperative multitasking
  • random() / randomSeed(): สุ่มเลข — ใช้ LFSR บน CH32V003
  • map() / constrain(): แมปช่วงค่า — ใช้ int arithmetic (ไม่มี float overhead)

ข้อแตกต่างสำคัญ: Arduino Uno = ATmega328P (8-bit AVR 16MHz), CH32V003 = RISC-V 32-bit 48MHz — เร็วกว่าแต่ไม่มี FPU — float ทำงานช้ากว่า int มาก

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Blink — Arduino Style

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"

// นี่คือโค้ด Arduino มาตรฐาน!
void setup() {
    pinMode(PC0, OUTPUT);
}

void loop() {
    digitalWrite(PC0, HIGH);
    delay(500);
    digitalWrite(PC0, LOW);
    delay(500);
}

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();  // จำเป็นสำหรับ delay()/millis()
    setup();
    while(1) loop();
}

🔍 setup() + loop() — แพทเทิร์น Arduino — SimpleHAL ไม่ได้บังคับให้มี setup()/loop() โดยอัตโนมัติ — แต่คุณเขียนแบบนี้ได้ — โค้ดนี้พอร์ตจาก Arduino มาโดยแทบไม่ต้องเปลี่ยนอะไร

ขั้นที่ 2: Non-blocking Blink (millis)

uint32_t last_toggle = 0;
bool led_state = false;

void setup() {
    pinMode(PC0, OUTPUT);
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
    Serial.print("Non-blocking blink with millis()\r\n");
}

void loop() {
    uint32_t now = millis();

    if(now - last_toggle >= 500) {
        last_toggle = now;
        led_state = !led_state;
        digitalWrite(PC0, led_state ? HIGH : LOW);

        Serial.print("Uptime: ");
        Serial.print(now);
        Serial.print(" ms\r\n");
    }

    // CPU ว่างทำงานอื่นได้ — ไม่ต้อง busy-wait!
    // เช่น: เช็ค button, อ่าน sensor, สื่อสาร
}

🔍 millis() — คืนค่า ms ตั้งแต่เริ่ม — ใช้ทำ non-blocking timing — ต่างจาก delay() ที่ block — Serial.print() — พิมพ์ผ่าน USART — ต้องเรียก USART_SimpleInit() ก่อน

ขั้นที่ 3: analogRead + analogWrite

void setup() {
    pinMode(PC4, OUTPUT);  // PWM output
    ADC_SimpleInit();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
}

void loop() {
    // อ่าน Potentiometer — Arduino style
    int sensor = analogRead(PD2);  // 0-1023

    // Map ไปเป็นความสว่าง LED 0-255
    int brightness = map(sensor, 0, 1023, 0, 255);

    // PWM output (Arduino style)
    analogWrite(PC4, brightness);

    Serial.print("Sensor: ");
    Serial.print(sensor);
    Serial.print(" -> PWM: ");
    Serial.print(brightness);
    Serial.print("\r\n");

    delay(50);
}

🔍 analogRead(pin) — Arduino-style ADC — ต้องเรียก ADC_SimpleInit() ก่อน — analogWrite(pin, val) — PWM output 0-255 — map(val, inMin, inMax, outMin, outMax) — แมปช่วงค่า

ขั้นที่ 4: Interrupt + Button

volatile int button_pressed = 0;

void button_isr(void) {
    button_pressed = 1;  // ISR: แค่ตั้ง flag
}

void setup() {
    pinMode(PC0, OUTPUT);
    pinMode(PA1, INPUT_PULLUP);

    // Arduino-style interrupt
    attachInterrupt(digitalPinToInterrupt(PA1),
                    button_isr, FALLING);

    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
}

void loop() {
    if(button_pressed) {
        button_pressed = 0;
        digitalWrite(PC0, !digitalRead(PC0));
        Serial.print("Button pressed!\r\n");
    }

    // CPU อิสระ — ไม่ต้อง poll button
    delay(10);
}

🔍 attachInterrupt(pin, func, mode) — Arduino-style interrupt — digitalPinToInterrupt(pin) — แปลง pin เป็น interrupt number — รองรับ RISING, FALLING, CHANGE, LOW

ขั้นที่ 5: dtostrf — Float to String

void loop() {
    float vdd = ADC_GetVDD();
    float temp = 25.6f;

    char buf[16];

    // แปลง float -> string (ไม่มี sprintf float ใน stdlib จิ๋ว)
    dtostrf(vdd, 5, 2, buf);  // " 3.28"
    Serial.print("VDD: "); Serial.print(buf);

    dtostrf(temp, 5, 1, buf);  // " 25.6"
    Serial.print(" Temp: "); Serial.print(buf);
    Serial.print("\r\n");

    delay(1000);
}

🔍 dtostrf(val, width, prec, buf) — แปลง float เป็น string — สำคัญมากเพราะ CH32V003 stdlib แบบ nano ไม่รองรับ sprintf float — width=ความกว้างรวม, prec=จำนวนทศนิยม

ขั้นที่ 6: yield() — Cooperative Multitasking

// Override yield() — ถูกเรียกใน delay()
void yield(void) {
    // ทำงานอื่นขณะ delay
    // เช่น: เช็ค button, อ่าน sensor, feed watchdog
    IWDG_Feed();
}

void loop() {
    // delay() จะเรียก yield() ระหว่างรอ
    // ทำให้ CPU ไม่ได้ idle เปล่าๆ
    digitalWrite(PC0, HIGH);
    delay(1000);  // yield() ถูกเรียกหลายครั้งในนี้
    digitalWrite(PC0, LOW);
    delay(1000);
}

🔍 yield() — weak function — ถูกเรียกใน delay() — override เพื่อทำงานอื่นระหว่าง delay — ใช้ feed watchdog, process data, check flags ได้

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
delay() ไม่ทำงานลืมเรียก Timer_Init()Timer_Init() ต้องอยู่ก่อน delay() หรือ millis()
analogRead() ได้ค่า 0 ตลอดลืมเรียก ADC_SimpleInit()ต้อง init ADC ก่อนใช้ analogRead()
analogWrite() ไม่ทำงานใช้ pin ที่ไม่ใช่ PWMPWM pins: PC4, PC5, PC6, PA1, PA2, PD3 — เช็คด้วย IS_PWM_VALID_PACKAGE
delay(5000) แล้ว IWDG resetdelay() ไม่ feed watchdogOverride yield() เพื่อ feed watchdog — หรือใช้ millis() แบบ non-blocking
Serial.print() ไม่แสดงลืมเรียก USART_SimpleInit() ก่อนUSART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT)
dtostrf() compile errorลืม include SimpleHAL.hdtostrf() define ใน SimpleHAL — ไม่ใช่ stdlib
attachInterrupt() HardFaultใช้ pin ที่ไม่มี interruptทุก pin ใน CH32V003 มี interrupt — เช็คแค่ว่า pin valid

🛠️ สร้าง Product: พอร์ตโค้ด Arduino มาลง CH32V003

ตัวอย่างพอร์ต Arduino sketch — อ่านเซ็นเซอร์ + PWM ควบคุม LED + Serial Monitor

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"

void setup()
{
    SystemCoreClockUpdate();
    Timer_Init();
    ADC_SimpleInit();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);

    pinMode(PC0, OUTPUT);      // LED
    pinMode(PC4, OUTPUT);       // PWM LED
    pinMode(PA1, INPUT_PULLUP); // Button

    Serial.print("CH32V003 Arduino Compatible\r\n");

    // ใช้ random seed จาก ADC noise
    randomSeed(analogRead(PD4));
}

void loop()
{
    // อ่าน potentiometer
    int pot = analogRead(PD2);

    // แมปเป็นความสว่าง LED
    int brightness = map(pot, 0, 1023, 0, 255);
    brightness = constrain(brightness, 0, 255);

    analogWrite(PC4, brightness);

    // เช็ค button
    if(digitalRead(PA1) == LOW) {
        digitalWrite(PC0, HIGH);
        Serial.print("Button! Pot=");
        Serial.print(pot);
        Serial.print("\r\n");
    } else {
        digitalWrite(PC0, LOW);
    }

    // Non-blocking print ทุก 5 วินาที
    static uint32_t last = 0;
    if(millis() - last > 5000) {
        last = millis();
        float vdd = ADC_GetVDD();
        char buf[10];
        dtostrf(vdd, 4, 2, buf);
        Serial.print("VDD: ");
        Serial.print(buf);
        Serial.print("V Uptime: ");
        Serial.print(millis() / 1000);
        Serial.print("s\r\n");
    }

    delay(50);
}

int main(void) {
    setup();
    while(1) loop();
}

เปรียบเทียบขนาดโค้ด: Arduino Uno Blink ~924 bytes — SimpleHAL Blink ~400 bytes — เล็กกว่า 57% — เหลือพื้นที่ Flash ให้โปรเจกต์ใหญ่ขึ้น
