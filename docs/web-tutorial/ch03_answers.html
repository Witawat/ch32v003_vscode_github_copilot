📝 เฉลย — บทที่ 3: Delay & Timer

ข้อ 1: นาฬิกาจับเวลาแบบ non-blocking แสดงผลผ่าน USART ทุก 1 วิ

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include "SimpleHAL.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
    pinMode(PC0, PIN_MODE_OUTPUT);

    Timer_t tmr_sec;
    Start_Timer(&tmr_sec, 1000, 1);  // repeat ทุก 1 วิ
    uint32_t seconds = 0;

    while (1) {
        if (Is_Timer_Expired(&tmr_sec)) {
            seconds++;
            USART_Print("[");
            USART_PrintNum(seconds);
            USART_Print("s]\r\n");
        }

        // LED กระพริบ 200ms — พิสูจน์ว่า main loop ไม่เคยหยุด
        static uint32_t last_toggle = 0;
        uint32_t now = Get_CurrentMs();
        if (ELAPSED_TIME(last_toggle, now) >= 200) {
            last_toggle = now;
            digitalToggle(PC0);
        }
    }
}
🧠 แนวคิด: ใช้ Timer_t แบบ repeat=1 หมดอัตโนมัติทุก 1 วิ — Is_Timer_Expired() ตรวจสอบโดยไม่หยุดรอ — LED กระพริบ 200ms ใช้ ELAPSED_TIME() macro (overflow-safe) พิสูจน์ว่า non-blocking ทำหลายงานพร้อมกันได้ — ต่างจาก Delay_Ms() ที่หยุดทุกอย่าง

ข้อ 2: Countdown 30 วิ พร้อม Buzzer เมื่อหมดเวลา

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include "SimpleHAL.h"

#define BUZZER  PC0
#define LED     PC1

void buzzer_beep(uint16_t ms) {
    digitalWrite(BUZZER, HIGH);
    Delay_Ms(ms);
    digitalWrite(BUZZER, LOW);
}

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);

    pinMode(BUZZER, PIN_MODE_OUTPUT);
    pinMode(LED, PIN_MODE_OUTPUT);

    Timer_t tmr_countdown, tmr_led;
    Start_Timer(&tmr_countdown, 1000, 1);
    Start_Timer(&tmr_led, 250, 1);

    int32_t count = 30;

    USART_Print("Countdown Start!\r\n");

    while (1) {
        if (count > 0 && Is_Timer_Expired(&tmr_countdown)) {
            count--;
            USART_Print("Remaining: ");
            USART_PrintNum(count);
            USART_Print("s\r\n");
        }

        if (count == 0) {
            Stop_Timer(&tmr_countdown);
            Stop_Timer(&tmr_led);

            // Buzzer ดัง 3 ครั้ง + LED ติดค้าง
            for (int i = 0; i < 3; i++) {
                buzzer_beep(200);
                Delay_Ms(100);
            }
            digitalWrite(LED, HIGH);
            USART_Print("TIME'S UP!\r\n");

            while (1);  // หยุด loop
        }

        // LED กระพริบระหว่างนับถอยหลัง
        if (Is_Timer_Expired(&tmr_led)) {
            digitalToggle(LED);
        }
    }
}
🧠 แนวคิด: ใช้ Timer_t นับถอยหลังทีละวิ — Stop_Timer() หยุดทั้ง countdown และ LED เมื่อหมดเวลา — Buzzer เป็น active buzzer (มี oscillator ในตัว) แค่จ่าย HIGH ก็ดัง — ใช้ for loop สร้างเสียง 3 ครั้ง — หลังจากนั้น while(1); หยุดโปรแกรม

ข้อ 3: ใช้ Timer_t สร้าง LED 3 ดวงกระพริบที่ความถี่ต่างกัน

#define CH32V003_PACKAGE  PACKAGE_TSSOP20
#include "SimpleHAL.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();

    uint8_t leds[] = {PC0, PC1, PC2};
    pinModeMultiple(leds, PIN_MODE_OUTPUT);

    Timer_t tmr[3];
    Start_Timer(&tmr[0], 100, 1);   // 5Hz
    Start_Timer(&tmr[1], 300, 1);   // ~1.67Hz
    Start_Timer(&tmr[2], 700, 1);   // ~0.7Hz

    while (1) {
        for (int i = 0; i < 3; i++) {
            if (Is_Timer_Expired(&tmr[i])) {
                digitalToggle(leds[i]);
            }
        }
        // Main loop วนอิสระ — ไม่มี Delay_Ms เลย!
    }
}
🧠 แนวคิด: Timer_t แต่ละตัวทำงานอิสระ — Start_Timer() ตั้งเวลา + repeat=1 ให้วนซ้ำ — array tmr[3] จับคู่กับ leds[3] — main loop วนตรวจสอบทั้ง 3 timer โดยไม่มี Delay_Ms() เลย CPU ไม่เคยหยุดรอ — ถ้าใช้ Delay_Ms() แต่ละดวงจะแย่งเวลากัน
