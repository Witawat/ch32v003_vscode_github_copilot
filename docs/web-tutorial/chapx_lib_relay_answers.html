📝 เฉลย — Relay Module

ข้อ 1: Timer Switch — เปิด 5 นาที / ปิด 5 นาที

#include "simple_relay.h"

Relay_Instance relay;

#define ON_TIME_MS   300000   // 5 นาที
#define OFF_TIME_MS  300000

typedef enum { STATE_ON, STATE_OFF } TimerState_t;

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    Relay_Init(&relay, PC0, RELAY_ACTIVE_HIGH);

    TimerState_t state = STATE_ON;
    uint32_t     last_toggle = 0;

    while (1) {
        uint32_t now = Timer_GetMs();

        if (state == STATE_ON) {
            Relay_On(&relay);
            if (now - last_toggle >= ON_TIME_MS) {
                state = STATE_OFF;
                last_toggle = now;
            }
        } else {
            Relay_Off(&relay);
            if (now - last_toggle >= OFF_TIME_MS) {
                state = STATE_ON;
                last_toggle = now;
            }
        }

        Delay_Ms(100);  // ไม่ต้อง poll ถี่มาก
    }
}
🧠 แนวคิด: ใช้สถานะ 2 state สลับไปมา — Timer_GetMs เป็น millisecond counter ใช้จับเวลาแบบ non-blocking — เมื่อครบเวลา 5 นาที state เปลี่ยนและ last_toggle ถูกรีเซ็ต — วิธี non-blocking ดีกว่า Delay_Ms เพราะระบบยังทำงานอื่นได้ — ถ้าต้องการปรับเวลาแบบ runtime: อ่านค่าจาก potentiometer หรือปุ่มกด — ระวัง millis overflow ทุก ~49 วัน — ใช้ unsigned subtraction now - last_toggle ทำให้ overflow-safe — สำหรับงานจริง: เพิ่ม RTC (DS3231) เพื่อความแม่นยำสูงข้ามการรีเซ็ต — ต่อ snubber diode ขนาน coil relay เพื่อป้องกันไฟ反向

ข้อ 2: Button Toggle — กดเปิด กดซ้ำปิด

#include "simple_relay.h"
#include "simple_button.h"

Relay_Instance relay;
Button_Instance btn;
volatile bool btn_pressed = false;

void on_btn_press(void) { btn_pressed = true; }

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    Relay_Init(&relay, PC0, RELAY_ACTIVE_HIGH);
    Button_Init(&btn, PC1, BUTTON_ACTIVE_LOW);
    Button_SetCallback(&btn, BUTTON_EVENT_PRESS, on_btn_press);

    uint8_t relay_state = 0;  // 0=OFF, 1=ON

    while (1) {
        if (btn_pressed) {
            btn_pressed = false;
            relay_state ^= 1;   // toggle
            if (relay_state) Relay_On(&relay);
            else             Relay_Off(&relay);
        }

        Delay_Ms(10);
    }
}
🧠 แนวคิด: ใช้ XOR toggle (^= 1) สลับสถานะทุกครั้งที่กดปุ่ม — Button_WasPressed ตรวจจับขอบขาขึ้นเพียงครั้งเดียว (ไม่ใช่การกดค้าง) — Button_Update ต้องเรียกใน loop หลักเพื่อทำ debounce — ถ้าใช้ปุ่มแบบ toggle switch (กดติด-ปล่อยดับ) ให้ใช้ Button_IsDown แทน — ควรเพิ่ม capacitor 0.1µF ที่ขา button เพื่อ hardware debounce ลด noise — ในตู้คอนโทรล: ต่อ LED แสดงสถานะ relay เพื่อให้ผู้ใช้เห็นว่าเปิดหรือปิด — ถ้า relay หลายตัวใช้หลายปุ่ม: เพิ่ม array of Button_t และ loop ตรวจสอบ

ข้อ 3: Countdown Timer พร้อม Relay — นับถอยหลังแล้วตัด

#include "simple_relay.h"
#include "simple_oled.h"

Relay_Instance relay;

#define COUNTDOWN_SEC  30

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    Relay_Init(&relay, PC0, RELAY_ACTIVE_HIGH);
    OLED_Handle oled;
    OLED_Init(&oled, OLED_128x64, 0x3C);

    uint16_t remaining = COUNTDOWN_SEC;
    uint32_t last_tick = 0;
    uint8_t  running   = 0;

    OLED_SetFont(FONT_8X16);
    OLED_Print(0, 0, "Countdown:");
    OLED_Update(&oled);

    while (1) {
        uint32_t now = Timer_GetMs();

        // เริ่มนับถอยหลังด้วยปุ่ม (simulate)
        if (!running) {
            running  = 1;
            remaining = COUNTDOWN_SEC;
            Relay_On(&relay);
            last_tick = now;
        }

        if (running && now - last_tick >= 1000) {
            last_tick = now;
            if (remaining > 0) {
                remaining--;
            }
            if (remaining == 0) {
                Relay_Off(&relay);  // หมดเวลา → ตัด relay
                running = 0;
            }
        }

        // แสดงผลบน OLED
        OLED_Clear();
        OLED_Print(0, 0, "Countdown:");
        OLED_SetFont(FONT_16X16_THAI);
        OLED_PrintNum(20, 30, remaining);
        OLED_Print(60, 30, "s");
        OLED_SetFont(FONT_8X16);
        if (!running && remaining == 0)
            OLED_Print(10, 50, "DONE!");
        OLED_Update(&oled);

        Delay_Ms(50);
    }
}
🧠 แนวคิด: ใช้ millisecond timer จับเวลา 1 วินาทีต่อครั้ง — ลด remaining ทุกวิ — เมื่อถึง 0: ปิดรีเลย์ + หยุดนับ — แสดงผลถอยหลังบน OLED — relay ทำหน้าที่เป็น switch ตัดกระแสอุปกรณ์เมื่อหมดเวลา — ถ้าต้องการเริ่มนับจากปุ่มกด: ใช้ Button_WasPressed ใน loop ตรวจสอบ — ถ้าต้องการ pause/resume: เพิ่ม state PAUSE แยกต่างหาก — ระวัง: Relay_Off ใช้กับ relay แบบ active-low (LOW=ON) ต้องเช็คชนิดของ relay module (HIGH trigger vs LOW trigger) — สำหรับ application จริง: ใช้กับเตารีด เครื่องชงกาแฟ หรือไฟส่องสว่างที่ต้องการปิดอัตโนมัติ — เพิ่มปุ่มยกเลิกกลางทางเพื่อปิดก่อนหมดเวลา
