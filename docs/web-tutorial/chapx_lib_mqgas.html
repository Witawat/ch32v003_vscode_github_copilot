            MQ Gas Sensor — เชื่อมต่อ ADC
    ┌────────────────────────────────────┐
    │ MCU                        MQ-xxx  │
    │ PA1────────AOUT     ┌─────────────┤
    │ 5V─────────VCC     │    Heater    │
    │ GND────────GND     └──────────────┤

MCU Pin→อุปกรณ์ขาหมายเหตุ
PA1 (ADC)←MQ-xxxAOUTAnalog Output ความเข้มข้นแก๊ส
5V→MQ-xxxVCCHeater 5V ~150mA
GND→MQ-xxxGNDGround ร่วม

รู้จักขา

  • AOUT: แรงดัน analog แปรผันตามความเข้มข้นแก๊ส
  • VCC: จ่ายไฟ 5V สำหรับ heater (~150mA)
  • GND: Ground ร่วม

วิธีต่อทีละขั้น

  1. ต่อ AOUT ไป PA1 (ADC) — อ่านค่าแก๊ส
  2. ต่อ VCC ไป 5V และ GND ไป GND — จ่ายไฟ heater

📚 หลักการทำงาน

MQ Gas Sensors ใช้หลักการเปลี่ยนความต้านทานของวัสดุ SnO₂ เมื่อสัมผัสแก๊ส — ต้อง preheat 24–48 ชม. ก่อน Calibrate — MQGas_Init รับ channel, type, VCC, VREF

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Init + Setup

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "MQGas.h"

MQGas_Instance mq;

int main(void) {
    SystemCoreClockUpdate();
    USART_Init(USART1, BAUD_115200);
    ADC_Init(ADC1);

    MQGas_Init(&mq, ADC_CH_PC4, MQ2, 5.0f, 3.3f);
    //         inst  adc_ch    type VCC  VREF
    USART_Println("MQGas Ready!");

🔍 MQGas_Init(&mq, adc_ch, type, VCC, VREF) — instance-based, adc_ch คือช่อง ADC, type=MQ2/MQ3/MQ4/MQ7/MQ135, VCC=แรงดัน heater (5V), VREF=แรงดันอ้างอิง ADC (3.3V)

ขั้นที่ 2: Calibrate + อ่านค่า

    // Calibrate ในอากาศสะอาด (ต้อง preheat ≥24 ชม.)
    MQGas_Calibrate(&mq);

    float ppm = MQGas_GetPPM(&mq);
    USART_Print("Gas PPM: ");
    USART_PrintFloat(ppm, 2);
    USART_Println("");

🔍 MQGas_Calibrate(&mq) — calibrate Ro ในอากาศสะอาด, MQGas_GetPPM(&mq) — คืนค่า PPM

ขั้นที่ 3: Loop + Alarm

    while(1) {
        float ppm = MQGas_GetPPM(&mq);

        USART_Print("PPM: "); USART_PrintFloat(ppm, 2);
        if(ppm > 1000) {
            USART_Println(" DANGER! Gas leak!");
        } else if(ppm > 200) {
            USART_Println(" WARNING!");
        } else {
            USART_Println(" Normal");
        }

        Delay_Ms(1000);
    }
}

🔍 CalibrateRo ต้องทำในอากาศสะอาด — เซนเซอร์ใช้เวลา warm-up อย่างน้อย 30 นาทีก่อนวัด

🛠️ สร้าง Product: เครื่องตรวจจับแก๊สรั่ว

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "MQGas.h"

int main(void)
{
    SystemCoreClockUpdate();
    USART_Init(USART1, BAUD_115200);
    ADC_Init(ADC1);

    pinMode(PC0, PIN_MODE_OUTPUT);  // Buzzer
    pinMode(PC1, PIN_MODE_OUTPUT);  // LED Alarm

    MQGas_Instance mq;
    MQGas_Init(&mq, ADC_CH_PC4, MQ2, 5.0f, 3.3f);
    MQGas_Calibrate(&mq);

    while(1)
    {
        float ppm = MQGas_GetPPM(&mq);
        USART_Print("PPM: "); USART_PrintFloat(ppm, 2); USART_Println("");

        if(ppm > 500) {
            digitalWrite(PC1, HIGH);
            digitalToggle(PC0);
        } else {
            digitalWrite(PC1, LOW);
            digitalWrite(PC0, LOW);
        }

        Delay_Ms(1000);
    }
}

✏️ ลองทำดู

  1. ทดลอง calibrate MQ-2 ในอากาศสะอาด — บันทึกค่า Ro แล้วทดสอบกับควันธูป
  2. ทำ gas logger — บันทึก PPM ทุกวินาทีผ่าน USART ลง PC
  3. ใช้ MQ-135 วัด Air Quality
