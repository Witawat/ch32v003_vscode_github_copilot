Button — Push Button with Debounce

PC1─Button─GND (active LOW, ใช้ internal pull-up)

#include "Button.h"
Button_Handle btn;
Button_Init(&btn, PC1, BUTTON_ACTIVE_LOW, 50); // debounce 50ms

void on_press(void)    { USART_Println("Pressed"); }
void on_long(void)     { USART_Println("Long press 1s"); }
void on_double(void)   { USART_Println("Double click"); }

Button_SetCallback(&btn, on_press);
Button_SetAdvancedCallback(&btn, on_long, 1000);
Button_OnDoubleClick(&btn, on_double, 400);

while(1) { Button_Update(&btn); IWDG_Feed(); }

⚠️ debounce 50ms=default, ACTIVE_LOW=button→GND (use pull-up), ACTIVE_HIGH=button→VCC (pull-down), long press threshold ms, double click window ms, callback-based (ไม่ต้อง poll flag), ต้อง Update() ใน loop

KeyMatrix — 4x4 Keypad

PC0-PC3─ROW1-4, PC4-PC7─COL1-4 (pull-up on columns)

#include "KeyMatrix.h"
KeyMatrix_Handle keypad;
uint8_t rows[] = {PC0,PC1,PC2,PC3}, cols[] = {PC4,PC5,PC6,PC7};
char keys[4][4] = {{'1','2','3','A'},{'4','5','6','B'},{'7','8','9','C'},{'*','0','#','D'}};
KeyMatrix_Init(&keypad, rows, cols, 4, 4, keys);

while(1) {
    KeyMatrix_Update(&keypad);
    char key = KeyMatrix_GetKey(&keypad);
    if(key) { USART_WriteByte(key); USART_Println(""); }
    Delay_Ms(20);
}

⚠️ row scan: activate one row LOW, read columns, ghost key prevention built-in, debounce ~20ms, max 8x8, ใช้ internal pull-up

RotaryEncoder — Knob + Button

PC0─CLK, PC1─DT, PC2─SW (pull-up on all)

#include "RotaryEncoder.h"
int32_t pos = 0;
void on_rotate(int8_t dir) { pos += dir; USART_PrintNum(pos); USART_Println(""); }
void on_button(void)       { pos = 0; USART_Println("Zero"); }

Rotary_Init(PC0, PC1, PC2);
Rotary_SetCallback(on_rotate);
Rotary_SetCallback(on_button);
// OR interrupt: RotaryEncoder_InitInt(PC0, PC1, PC2); // no polling

while(1) { Rotary_Update(); IWDG_Feed(); }

⚠️ quadrature encoding, CLK lead=±, 20 steps/rev common, interrupt mode saves CPU, acceleration: if speed > threshold → multiply step

PID — Control Algorithm

#include "PID.h"
PID_Handle pid;
PID_Init(&pid, PID_DIRECT); // heater
PID_SetTunings(&pid, 2.0f, 3.0f, 0.5f); // Kp, Ki, Kd
PID_SetLimits(&pid, 0, 100);
PID_SetSetpoint(&pid, 25.0f);

while(1) {
    float input = read_temperature();
    float output = PID_Compute(&pid, input);
    set_heater_pwm(output); // 0-100%
    Delay_Ms(1000); // sample time
}

⚠️ DIRECT=heater, REVERSE=cooler, Ki in minutes (integral time), anti-windup via output limits, tuning: manual or Ziegler-Nichols

LCDMenu — Interactive Menu

LCD1602 I2C: PC1─SDA, PC2─SCL
Buttons: PC3─UP, PC4─DOWN, PC5─ENTER, PC6─BACK (pull-up)

#include "LCDMenu.h"
#include "lcd1602_i2c.h"
LCDMenu_Handle menu;

void on_temp(void) { USART_Println("Temp menu"); }
void on_fan_toggle(LCDMenu_Handle *m, uint8_t val) {
    USART_Print(val ? "Fan ON\r\n" : "Fan OFF\r\n");
}

LCD_Init(0x27);
LCDMenu_Init(&menu, 0x27, 16, 2);
LCDMenu_AddItem(&menu, "Temperature",   LCDMENU_SUBMENU, NULL);
LCDMenu_AddItem(&menu, "  Set Max",     LCDMENU_EDIT,    on_temp);
LCDMenu_AddItem(&menu, "Fan Control",   LCDMENU_TOGGLE,  on_fan_toggle);
LCDMenu_AddItem(&menu, "Back",          LCDMENU_BACK,    NULL);

while(1) {
    LCDMenu_Update(&menu, !digitalRead(PC3), !digitalRead(PC4),
                   !digitalRead(PC5), !digitalRead(PC6));
    Delay_Ms(50);
}

⚠️ 4 callback types: SUBMENU/EDIT/TOGGLE/ACTION/BACK, static memory pool, LCD1602 16x2, edit mode: UP/DOWN change value, refresh 50ms

CircularBuffer — FIFO Queue

#include "CircularBuffer.h"
#define BUF_SIZE 64
CircularBuffer_Handle cb;
uint8_t buf[BUF_SIZE];
CircularBuffer_Init(&cb, buf, BUF_SIZE);

// ISR: write
void USART1_IRQHandler(void) {
    if(USART_GetFlagStatus(USART1, USART_FLAG_RXNE)) {
        char c = USART_ReceiveData(USART1);
        CircularBuffer_Push(&cb, c); // safe (no ISR conflicts)
    }
}
// Main: read
while(1) {
    while(CircularBuffer_Available(&cb)) {
        char c = CircularBuffer_Pop(&cb);
        process_char(c);
    }
    IWDG_Feed();
}

⚠️ FIFO, head/tail wrap, write fails when full (return 0), ISR-write + main-read = thread-safe, 0 RAM overhead beyond buffer, ใช้ bridge between ISR↔main loop
