nRF24L01+ ใช้ SPI + 2 control pins (CE, CSN) + optional IRQ, ห้ามต่อ 5V เด็ดขาด — module พังทันที, capacitor ที่ VCC สำคัญมากเพราะ module กินกระแสกระชากตอน TX สูงถึง ~13mA

📚 หลักการทำงาน

nRF24L01+ เป็น 2.4GHz RF transceiver สื่อสารผ่าน SPI — รองรับ 125 channels, data rate 250kbps/1Mbps/2Mbps, ระยะ ~100m line-of-sight (พร้อม PCB antenna), ใช้ Enhanced ShockBurst protocol: auto-ACK, auto-retransmit

6 Data Pipes: RX side รับข้อมูลจาก 6 address พร้อมกันได้ (Pipe 0-5) — Pipe 0-1 มี 5-byte address, Pipe 2-5 มี 1-byte unique + share 4-byte base address

Auto-ACK: ทุก packet ที่ส่งไป TX จะรอ ACK จาก RX — ถ้าไม่ได้รับภายใน timeout (auto-retransmit delay + max retries) จะ set MAX_RT flag — ทำให้ส่งข้อมูลเชื่อถือได้สูง

Payload: 1-32 bytes per packet — ถ้าต้องส่งข้อมูลมากต้องแบ่งเป็นหลาย packets เอง

💻 เขียน Code ทีละขั้น

ขั้นที่ 1: Init + Setup (Transmitter)

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "nRF24L01.h"

nRF24_Instance radio;
uint8_t tx_addr[5] = {0xE7,0xE7,0xE7,0xE7,0xE7}; // 5-byte address

int main(void) {
    SystemCoreClockUpdate();
    SPI_Init(SPI1, SPI_BAUD_4MHZ, SPI_MODE0);
    nRF24_Init(&radio, PC4, PC3);  // CSN=PC4, CE=PC3
    USART_Init(USART1, BAUD_115200);

🔍 nRF24_Init(&radio, CSN, CE) — CSN=PC4, CE=PC3, address 5-byte ต้องตรงกันทั้ง TX และ RX, SPI Mode 0 (CPOL=0, CPHA=0)

ขั้นที่ 2: TX — ส่งข้อมูล

    nRF24_SetTxAddr(&radio, tx_addr);        // set TX addr
    nRF24_SetChannel(0x60);                   // channel 0x60
    NRF24_SetDataRate(NRF24_1MBPS);

    uint8_t data[32];
    sprintf((char*)data, "Hello from CH32V003!");
    uint8_t len = strlen((char*)data);

    if(nRF24_Transmit(&radio, data, len)) {
        USART_Println("Sent OK!");
    } else {
        USART_Println("Send failed (MAX_RT)"); // max retransmit reached
    }

🔍 nRF24_Transmit(&radio, data, len) — เขียน payload ถึง 32 bytes, คืน 0 ถ้า MAX_RT (ส่งซ้ำจนเกิน limit), คืน 1 ถ้า TX_DS (ได้รับ ACK), ต้องตั้ง channel เดียวกันทั้ง TX และ RX

ขั้นที่ 3: RX — รับข้อมูลหลาย Pipe

    nRF24_SetRxAddr(&radio, tx_addr);      // same addr for pipe0
    uint8_t rx_addr2[5] = {0xC2,0xC2,0xC2,0xC2,0xC2};
    nRF24_SetRxAddr(&radio, rx_addr2);      // different addr
    nRF24_StartListening(&radio);

    while(1) {
        if(nRF24_DataAvailable(&radio)) {        // data ready?
            uint8_t rx_buf[32];
            nRF24_Receive(&radio, rx_buf, 32);   // read payload
            USART_Print("RX: ");
            USART_Println((char*)rx_buf);
        }
    }
}

🔍 nRF24_DataAvailable(&radio) — non-blocking เช็ค RX FIFO, nRF24_Receive(&radio, buf, len) — อ่าน payload + clear RX_DR flag, RX รองรับ 6 pipes — ใช้ pipe ต่างกันรับข้อมูลจาก 6 nodes พร้อมกัน

⚠️ ข้อผิดพลาดที่พบบ่อย

ปัญหาสาเหตุวิธีแก้
สื่อสารกันไม่ได้เลยAddress ไม่ตรงกัน TX/RXเช็ค 5-byte address ให้ตรงกันเป๊ะทั้งสองฝั่ง
MAX_RT ทุกครั้งRX ไม่อยู่ใน listening modeเรียก nRF24_StartListening(&radio) ที่ฝั่ง RX
ส่งได้บางครั้ง หลุดบ่อยไฟ 3.3V ไม่สะอาด — noiseต่อ capacitor 10µF + 0.1µF ขนาน VCC-GND ใกล้ module
อ่าน/เขียน register ไม่ได้SPI mode ไม่ตรง (ต้อง Mode 0)ใช้ SPI_MODE0 (CPOL=0, CPHA=0) เท่านั้น
ระยะสั้นมาก <5mPA+LNA module หรือ antenna พังลด data rate เหลือ 250kbps เพิ่มระยะ, เช็ค antenna

🛠️ สร้าง Product: Multi-Sensor Wireless Star Network

1 Receiver + 3 Transmitter nodes — TX แต่ละตัวส่งค่าอุณหภูมิไปยัง RX ผ่าน pipe แยกกัน

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#include "SimpleHAL.h"
#include "nRF24L01.h"

// === RX Node (Collector) ===
int main(void)
{
    SystemCoreClockUpdate();
    SPI_Init(SPI1, SPI_BAUD_4MHZ, SPI_MODE0);
    nRF24_Instance radio;
    nRF24_Init(&radio, PC4, PC3);
    USART_Init(USART1, BAUD_115200);

    uint8_t addr0[5] = {0xAA,0xAA,0xAA,0xAA,0xAA};
    uint8_t addr1[5] = {0xBB,0xBB,0xBB,0xBB,0xBB};
    uint8_t addr2[5] = {0xCC,0xCC,0xCC,0xCC,0xCC};

    nRF24_SetRxAddr(&radio, addr0);  // Node 0
    nRF24_SetRxAddr(&radio, addr1);  // Node 1  (multi-pipe)
    nRF24_SetRxAddr(&radio, addr2);  // Node 2
    nRF24_StartListening(&radio);

    uint8_t buf[32];
    while(1)
    {
        if(nRF24_DataAvailable(&radio)) {
            nRF24_Receive(&radio, buf, 32);
            USART_Print("RX: "); USART_Println((char*)buf);
        }
    }
}

✏️ ลองทำดู

  1. ทำ wireless remote control — TX ส่ง joystick ADC (x,y) 2 bytes → RX รับ → ควบคุม servo 2 ตัว
  2. ทำ mesh network อย่างง่าย — node A ส่งถึง node B, node B forward ไป node C (repeater)
  3. ทำ ACK payload — RX แนบ response ไปกับ ACK packet (ใช้ NRF24_WriteAckPayload())
