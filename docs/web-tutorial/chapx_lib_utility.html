🛠️ Full Product — Temperature Controller with OLED UI

CH32V003 TSSOP-20
├── I2C Bus (PC1=SDA, PC2=SCL)
│   ├── BMP280 (addr 0x76) — measure room temp
│   └── OLED SSD1306 (addr 0x3C) — display
├── 1-Wire (PC0) ── DS18B20 — measure target zone temp
├── Relay (PC5) ── Heater — activate when cold
├── RotaryEncoder (PC3=CLK, PC4=DT, PC6=SW) — adjust setpoint
├── Buzzer (PA1) — alarm when over-temp
├── USART (PD5=TX, PD6=RX) — serial monitor / IoT extension
└── WCH-Link (PD1=SWIO) — flash + debug

โค้ดสมบูรณ์ (~140 บรรทัด)

#define CH32V003_PACKAGE PACKAGE_TSSOP20
#define ENABLE_USART_PRINTLN  1
#define ENABLE_USART_PRINTFLOAT 1
#include "SimpleHAL.h"
#include "BMP280.h"
#include "DS18B20.h"
#include "oled_i2c.h"
#include "Relay.h"
#include "RotaryEncoder.h"
#include "PID.h"
#include "SimpleScheduler.h"
#include "Buzzer.h"

OLED_Handle oled;
PID_Handle pid;
DS18B20_Device* ds;
RotaryEncoder enc;
Relay_Instance relay;
float setpoint = 25.0f, current_temp = 0, pid_output = 0;
uint8_t heater_on = 0, alarm_active = 0;
uint32_t alarm_start = 0;
char buf[32];

// ---- Tasks (called by Scheduler) ----
void task_read_sensors(void) {
    float t1 = BMP280_ReadTemperature();           // room
    float t2 = DS18B20_ReadTemperatureBlocking(ds); // target zone
    current_temp = (t1 + t2) / 2.0f;
}

void task_pid_control(void) {
    pid_output = PID_Compute(&pid, current_temp);
    if(pid_output > 5) { Relay_On(&relay); heater_on = 1; }
    else { Relay_Off(&relay); heater_on = 0; }
    // Alarm if 2°C above setpoint for 10 seconds
    if(current_temp > setpoint + 2.0f) {
        if(!alarm_active) { alarm_active = 1; alarm_start = Get_CurrentMs(); }
        else if(Get_CurrentMs() - alarm_start > 10000) {
            Buzzer_Beep(PA1, 200); // beep every 2s
            USART_Println("WARNING: Over temperature!");
            alarm_start = Get_CurrentMs();
        }
    } else { alarm_active = 0; }
}

void task_update_display(void) {
    OLED_Clear(&oled);
    snprintf(buf, sizeof(buf), "T:%.1f C  %s", current_temp, heater_on ? "[ON]" : "[  ]");
    OLED_DrawStringAlign(&oled, 0, 0, buf, ALIGN_LEFT);
    snprintf(buf, sizeof(buf), "Set:%.1f C", setpoint);
    OLED_DrawStringAlign(&oled, 0, 16, buf, ALIGN_LEFT);
    snprintf(buf, sizeof(buf), "PWR:%.1f%%  %lus", pid_output, (uint32_t)(Get_CurrentMs()/1000));
    OLED_DrawStringAlign(&oled, 0, 32, buf, ALIGN_LEFT);
    OLED_Update(&oled);
}

// ---- Rotary Encoder Callbacks ----
void on_rotate(RotaryEncoder* e) {
    int32_t dir = Rotary_GetPosition(e);
    setpoint += dir * 0.5f;
    if(setpoint < 0)  setpoint = 0;
    if(setpoint > 50) setpoint = 50;
    PID_SetSetpoint(&pid, setpoint);
    USART_Print("Setpoint: "); USART_PrintFloat(setpoint, 1); USART_Println("C");
    Buzzer_Tone(PA1, 1000, 20); // short click feedback
}

void on_button(RotaryEncoder* e) {
    heater_on = !heater_on;
    USART_Println(heater_on ? "Controller ON" : "Controller OFF");
}

// ---- Main ----
int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();

    // Init communication
    I2C_SimpleInit(I2C_SPEED_400K, I2C_PINS_DEFAULT);
    ds = DS18B20_Init(PC0);
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);

    // Init sensors
    BMP280_Init(0x76);
    BMP280_SetMode(BMP280_MODE_NORMAL);

    // Init display
    OLED_Init(&oled, OLED_128x64, 0x3C);
    OLED_Clear(&oled);
    snprintf(buf, sizeof(buf), "Temp Controller");
    OLED_DrawStringAlign(&oled, 0, 16, buf, ALIGN_LEFT);
    snprintf(buf, sizeof(buf), "Booting...");
    OLED_DrawStringAlign(&oled, 0, 32, buf, ALIGN_LEFT);
    OLED_Update(&oled);
    Delay_Ms(2000);

    // Init actuators
    Relay_Init(&relay, PC5, RELAY_ACTIVE_HIGH);
    Buzzer_Init(PA1, BUZZER_ACTIVE);

    // Init PID
    PID_Init(&pid, PID_DIRECT);
    PID_SetTunings(&pid, 2.0f, 3.0f, 0.5f);
    PID_SetLimits(&pid, 0, 100);
    PID_SetSetpoint(&pid, setpoint);

    // Init input
    Rotary_Init(&enc, PC3, PC4, PC6);
    Rotary_SetCallback(&enc, ROTARY_EVENT_ROTATE, on_rotate);
    Rotary_SetCallback(&enc, ROTARY_EVENT_PRESS, on_button);

    // Init scheduler
    Scheduler_Init();
    Scheduler_AddTask(task_read_sensors,  2000); // 2s
    Scheduler_AddTask(task_pid_control,   1000); // 1s
    Scheduler_AddTask(task_update_display, 500); // 0.5s

    Buzzer_Beep(PA1, 100); // startup beep
    USART_Println("=== Temp Controller Ready ===");

    while(1) {
        Rotary_Update(&enc);  // fast poll for responsiveness
        Scheduler_Run();      // run timed tasks
        yield();              // feed IWDG

        // Optional: sleep when idle
        // PWR_Sleep();
    }
}

⚠️ ให้เหลือ Flash > 15%, RAM > 20% สำหรับ stack และการเพิ่มฟีเจอร์ทีหลัง — เปิด ENABLE_PRINTF=0 ใน release จะประหยัด ~2KB Flash

🧩 Design Patterns สรุป

PatternInitLoopเหมาะกับ
SensorLib_Init → Lib_SetModeLib_Read → ProcessBMP280, DHT, MQGas
ActuatorLib_Init → Lib_StartLib_Set/MoveTo → Update → CheckDoneServo, Stepper, ESC
DisplayLib_Init → Lib_ClearClear → Draw → DisplayOLED, MAX7219, TJC
ControlPID_Init → PID_SetTuningsRead → Compute → OutputPID, SimpleScheduler
InputLib_Init → Lib_OnCallbackLib_Update → event fires callbackButton, RotaryEncoder, KeyMatrix

✏️ ลองทำดู

  1. เพิ่ม WiFi logging — ESP01 send data to ThingSpeak every 30s
  2. บันทึก setpoint ลง Flash — โหลดกลับตอน boot
  3. เพิ่ม alarm escalation — เกิน setpoint+2=beep, +5=relay off (emergency stop)
  4. เปลี่ยน RotaryEncoder เป็น KeyMatrix — input setpoint เป็นเลข 4 หลัก
  5. ทำ battery-powered version — Standby+AWU 10s, wake→measure→control→sleep
