📝 เฉลย — TJC HMI Display

ข้อ 1: 2-Page UI — สลับหน้าด้วยปุ่ม

#include "TJC.h"

#define BTN_PAGE  PC1

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
    TJC_Init(BAUD_115200, USART_PINS_DEFAULT);
    pinMode(BTN_PAGE, PIN_MODE_INPUT_PULLUP);

    uint8_t page = 0;
    uint8_t last_btn = 1;

    // หน้าแรก — Main
    TJC_SendCommand("page 0");

    while(1) {
        uint8_t btn = digitalRead(BTN_PAGE);
        if(last_btn && !btn) {
            page = !page;
            if(page == 0) {
                TJC_SendCommand("page 0");  // ไปหน้า Main
            } else {
                TJC_SendCommand("page 1");  // ไปหน้า Settings
                // อัปเดตค่าในหน้า settings
                TJC_SendCommand("t0.txt=\"Settings\"");
                TJC_SendCommand("n0.val=25");
                TJC_SendCommand("n1.val=60");
            }
            Delay_Ms(50);
        }
        last_btn = btn;

        // หน้า Main — อัปเดตค่าจากเซ็นเซอร์
        if(page == 0) {
            uint16_t adc = ADC_Read(ADC_CH0);
            uint16_t temp = (uint32_t)adc * 330 / 4095;
            char cmd[32];
            sprintf(cmd, "t1.txt=\"Temp: %d.%d C\"", temp/10, temp%10);
            TJC_SendCommand(cmd);
        }

        Delay_Ms(200);
    }
}
🧠 แนวคิด: TJC HMI แต่ละหน้าคือ "page" ที่ออกแบบไว้ใน Nextion Editor — TJC_SendCommand("page N") สั่งเปลี่ยนหน้า — ตอนเปลี่ยนไปหน้า settings ให้ส่งค่าปัจจุบันไปอัปเดต components (t0=text, n0=number) — หน้า Main อัปเดต sensor value ทุก 200ms — ใช้ sprintf สร้าง command string — TJC สื่อสารผ่าน UART (115200 baud) — ตั้งชื่อ component ให้สื่อความหมาย (t=text, n=number, b=button)

ข้อ 2: Graph แบบ Real-time จากเซ็นเซอร์

#include "TJC.h"

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
    TJC_Init(BAUD_115200, USART_PINS_DEFAULT);
    ADC_SimpleInit(ADC_CH0);

    TJC_SendCommand("page 2");  // หน้า graph

    uint8_t x = 0;

    while(1) {
        uint16_t adc = ADC_Read(ADC_CH0);
        // Map ADC 0-4095 → graph Y 0-200
        uint16_t y = (uint32_t)adc * 200 / 4095;

        // เพิ่มจุดข้อมูลลง waveform component (wv0)
        char cmd[64];
        sprintf(cmd, "add 1,%d,%d", x, y);
        TJC_SendCommand(cmd);

        x++;
        if(x >= 300) {  // waveform เต็ม
            TJC_SendCommand("cle 1,0");  // ล้าง channel 1
            x = 0;
        }

        // อัปเดตตัวเลขแสดงค่าปัจจุบัน
        sprintf(cmd, "n0.val=%d", adc);
        TJC_SendCommand(cmd);

        // แสดงค่าแรงดัน
        uint16_t mv = (uint32_t)adc * 3300 / 4095;
        sprintf(cmd, "t0.txt=\"%d.%03d V\"", mv/1000, mv%1000);
        TJC_SendCommand(cmd);

        Delay_Ms(100);
    }
}
🧠 แนวคิด: TJC มี waveform component ในตัว — add ch,x,y เพิ่มจุดข้อมูลลง channel — map ADC 0-4095 เป็นค่า y 0-200 — เลื่อน x ทีละ 1 จนเต็ม 300 จุดแล้ว clear — อัปเดตตัวเลข (n0) และข้อความ (t0) ประกอบกราฟ — Refresh rate 100ms = 10Hz — waveform component รองรับหลาย channel (สูงสุด 4) — สำหรับข้อมูลเร็ว (kHz) ควรส่งแบบ batch ด้วยคำสั่ง addt แทน

ข้อ 3: รับ Touch Event จาก TJC

#include "TJC.h"

#define LED_PIN  PC0

volatile uint8_t led_state = 0;
volatile uint8_t slider_value = 0;

// Parse touch event จาก TJC
// รูปแบบ: 0x65 btn_id event_type\xFF\xFF\xFF
void TJC_ParseEvent(uint8_t *data, uint8_t len) {
    if(len >= 4 && data[0] == 0x65) {  // touch event
        uint8_t component = data[1];
        uint8_t event = data[2];  // 1=press, 0=release

        if(component == 0 && event == 1) {  // btn0 press
            led_state = !led_state;
            digitalWrite(LED_PIN, led_state);
        }
        if(component == 1 && event == 1) {  // btn1 press
            TJC_SendCommand("page 0");  // back to main
        }
    }
    if(len >= 4 && data[0] == 0x71) {  // slider/value event
        slider_value = data[1];
        // ใช้ slider_value ควบคุม PWM
        PWM_SetDuty(TIM2_CH1, slider_value);
    }
}

int main(void) {
    SystemCoreClockUpdate();
    Timer_Init();
    USART_SimpleInit(BAUD_115200, USART_PINS_DEFAULT);
    TJC_Init(BAUD_115200, USART_PINS_DEFAULT);
    pinMode(LED_PIN, PIN_MODE_OUTPUT);

    TJC_SendCommand("page 0");

    uint8_t rx_buf[32];

    while(1) {
        // อ่านข้อมูลจาก TJC
        uint8_t len = 0;
        while(USART_Available(&USART2)) {
            if(len < sizeof(rx_buf)) {
                rx_buf[len++] = USART_Read(&USART2);
            }
        }

        if(len > 0) {
            TJC_ParseEvent(rx_buf, len);
        }

        // อัปเดต slider value บนจอ
        char cmd[16];
        sprintf(cmd, "n0.val=%d", slider_value);
        TJC_SendCommand(cmd);

        Delay_Ms(50);
    }
}
🧠 แนวคิด: TJC ส่ง touch event กลับมาผ่าน UART — ข้อมูลขึ้นต้นด้วย 0x65 (touch) หรือ 0x71 (slider/value) — ไบต์ที่ 2 คือ ID ของ component, ไบต์ที่ 3 คือ event type (1=press, 0=release) — อ่านข้อมูลจาก UART buffer แล้ว parse ทีละ event — ใช้ touch event ควบคุม LED, เปลี่ยนหน้า, ปรับ PWM — การ poll UART แบบนี้เหมาะกับ MCU ที่มี RAM น้อย — TJC ยังรองรับ pass-through mode เพื่อ debug คำสั่งผ่าน USB
